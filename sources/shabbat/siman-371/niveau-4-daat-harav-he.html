דעת הרב

דעת הרב — שולחן ערוך אדמו״ר הזקן על סימן שע״א

אחד מבני החצר שהלך לשבות במקום אחר, או שמת: האם ביתו עדיין אוסר על בני החצר? בית שירשו מבעוד יום או משחשכה, גר שמת, גוסס, קטן ואורח. ששת הסעיפים של שולחן ערוך אדמו״ר הזקן, מושווים לששת הסעיפים של המחבר.

מדוע רמה 4 מוקדשת לאדמו״ר הזקן? שולחן ערוך אדמו״ר הזקן אינו פירוש על המחבר: זהו שולחן ערוך בפני עצמו, שנכתב על ידי אדמו״ר הזקן, בחלוקה ובמספור סעיפים משלו. כאן יש בשולחן ערוך הרב שישה סעיפים, וגם בשולחן ערוך של המחבר שישה, אבל החלוקה שונה: סעיף א של הרב כולל את מה שאומר המחבר בסעיפים א ו־ב (ודין הנכרי, שהמחבר דן בו בסעיף א, נדחה שם לסימן שפ״ג); סעיפים ב, ג ו־ד של הרב מקבילים לסעיפים ג, ד ו־ה של המחבר; וסעיף ו של המחבר מתחלק בין סעיפים ה ו־ו של הרב. ההפניות בעמוד זה הולכות לפי מספור הרב, אלא כשהשולחן ערוך של המחבר נזכר בשמו. בעמוד זה שני המספורים כתובים באותיות, ולכן כל הפניה לסעיף נזכרת עם שם החיבור: של המחבר (שו״ע או״ח) או של הרב (שו״ע הרב).

עמוד זה מקבץ, לסימן שע״א, את הטקסט המלא של הרב, את מה ששונה בלשונו מדברי המחבר, ואת ההנהגה המעשית העולה מדבריו.

שלחן ערוך הרב — סימן שע״א

הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן שע״א — כְּשֶׁאֶחָד מִבְּנֵי חָצֵר נִפְרָד מִשָּׁם אוֹ מֵת — וּבוֹ ו׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. שישה סעיפים (גם בשולחן ערוך של המחבר שישה, במספור ובחלוקה משלו).

סעיף א כְּשֶׁאֶחָד מִבְּנֵי חָצֵר נִפְרָד מִשָּׁם אוֹ מֵת וּבוֹ ו' סְעִיפִים: דִּירָה בְּלֹא בְעָלִים אֵינָהּ דִּירָה…

כְּשֶׁאֶחָד מִבְּנֵי חָצֵר נִפְרָד מִשָּׁם אוֹ מֵת וּבוֹ ו' סְעִיפִים:

דִּירָה בְּלֹא בְעָלִים אֵינָהּ דִּירָה וְאֵינָהּ אוֹסֶרֶת בְּחָצֵר, וְגַם מֻתָּר לְטַלְטֵל מִמֶּנָּה לְחָצֵר שֶׁיֵּשׁ בָּהּ דִּיּוּרִים אֲחֵרִים שֶׁעֵרְבוּ בֵּינֵיהֶם, וְאֵינוֹ כְּמוֹצִיא מֵרְשׁוּת לִרְשׁוּת. לְפִיכָךְ אֶחָד מִבְּנֵי חָצֵר שֶׁהִנִּיחַ בֵּיתוֹ וְהָלַךְ עִם כָּל בְּנֵי בֵיתוֹ וְשָׁבַת בְּחָצֵר אַחֶרֶת, אֲפִלּוּ הָיְתָה סְמוּכָה לַחֲצֵרוֹ — אִם הִסִּיחַ לִבּוֹ וְאֵין דַּעְתּוֹ לַחֲזוֹר לְבֵיתוֹ בְּשַׁבָּת, הֲרֵי זֶה אֵינוֹ אוֹסֵר עַל בְּנֵי הֶחָצֵר שֶׁבֵּיתוֹ שָׁמָּה אִם עֵרְבוּ בֵּינֵיהֶם, אַף עַל פִּי שֶׁהוּא לֹא עֵרֵב עִמָּהֶם. וּמֻתָּרִים לְטַלְטֵל גַּם מִבֵּיתוֹ לְחָצֵר.

וַאֲפִלּוּ אִם אַחַר כָּךְ נִמְלַךְ וְחָזַר לְבֵיתוֹ בְּשַׁבָּת, אֵינוֹ אוֹסֵר, שֶׁכֵּיוָן שֶׁכְּבָר הִסִּיחַ מִלִּבּוֹ בִּכְנִיסַת הַשַּׁבָּת, הֲרֵי שַׁבָּת זוֹ — כֵּיוָן שֶׁהֻתְּרָה הֻתְּרָה, וְשׁוּב לֹא תֵאָסֵר, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שס״ב.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁהָלַךְ מִבֵּיתוֹ מִבְּעוֹד יוֹם. אֲבָל אִם הָלַךְ מִשֶּׁחָשֵׁכָה — אוֹסֵר כְּשֶׁחוֹזֵר.

וְנָכְרִי שֶׁהָלַךְ מִבֵּיתוֹ יִתְבָּאֵר בְּסִמָּן שפ״ג:

סעיף ב אֶחָד מִן הַשּׁוּק שֶׁהָיָה לוֹ בַּיִת בְּחָצֵר שֶׁלֹּא הָיָה…

אֶחָד מִן הַשּׁוּק שֶׁהָיָה לוֹ בַּיִת בְּחָצֵר שֶׁלֹּא הָיָה דָר בָּהּ וְלָכֵן לֹא עֵרֵב בָּהּ, וּמֵת וְהִנִּיחַ רְשׁוּתוֹ לְאֶחָד מִבְּנֵי אוֹתָהּ חָצֵר — אִם מֵת מִבְּעוֹד יוֹם, אֵין הַזּוֹכֶה אוֹסֵר, לְפִי שֶׁהָעֵרוּב שֶׁעֵרֵב עַל בַּיִת שֶׁלּוֹ מַתִּיר גַּם בַּיִת שֶׁיָּרַשׁ מִבְּעוֹד יוֹם, שֶׁאַף שֶׁהָעֵרוּב נַעֲשָׂה קֹדֶם הַיְרֻשָּׁה, מִכָּל מָקוֹם הֲרֵי קְנִיַּת הָעֵרוּב הוּא בִּכְנִיסַת הַשַּׁבָּת, וְאָז יְרָשׁוֹ כְּבָר. אֲבָל אִם מֵת מִשֶּׁחָשֵׁכָה — אוֹסֵר הַזּוֹכֶה בּוֹ עַל בְּנֵי הֶחָצֵר, וְגַם הוּא אָסוּר לְטַלְטֵל בֶּחָצֵר, בֵּין מִבַּיִת שֶׁלּוֹ בֵּין מִבַּיִת זֶה, וְאַף עַל פִּי שֶׁעֵרֵב עִמָּהֶם, אֵינוֹ מוֹעִיל לְזֶה שֶׁיָּרַשׁ בְּשַׁבָּת אַחַר קְנִיַּת הָעֵרוּב. וַאֲפִלּוּ אִם זֶה הַזּוֹכֶה לֹא נִכְנַס עֲדַיִן בְּבַיִת זֶה לָדוּר בּוֹ, אֵין זוֹ דִירָה בְּלֹא בְעָלִים, שֶׁכֵּיוָן שֶׁהוּא דָר בְּחָצֵר זוֹ וְרָאוּי לְהִשְׁתַּמֵּשׁ בְּבַיִת זֶה שֶׁיְּרָשׁוֹ — אָנוּ חוֹשְׁבִין אוֹתוֹ כְּאִלּוּ נִכְנַס לָדוּר בּוֹ.

וְאַף עַל פִּי שֶׁבִּכְנִיסַת הַשַּׁבָּת כְּשֶׁהָיָה מוֹרִישׁוֹ חַי וְלֹא הָיָה דָר בּוֹ הָיְתָה דִירָה בְּלֹא בְעָלִים, וְלֹא הָיְתָה אוֹסֶרֶת עַל בְּנֵי הֶחָצֵר — אֵין אוֹמְרִים בְּזֶה "שַׁבָּת כֵּיוָן שֶׁהֻתְּרָה הֻתְּרָה", לְפִי שֶׁאֵין אוֹמְרִים כֵּן אֶלָּא כְּשֶׁבְּשָׁעָה שֶׁהֻתְּרָה הָיָה הַדָּבָר בָּרוּר שֶׁלֹּא תַחֲזוֹר וְתֵאָסֵר, כְּגוֹן מִי שֶׁהָלַךְ מִבֵּיתוֹ מִבְּעוֹד יוֹם וְהִסִּיחַ מִלִּבּוֹ שֶׁלֹּא לַחֲזוֹר עַד מוֹצָאֵי שַׁבָּת, שֶׁבְּשָׁעָה שֶׁהִסִּיחַ מִלִּבּוֹ הַדָּבָר יָדוּעַ בְּוַדַּאי שֶׁלֹּא יַחֲזוֹר וְהֻתְּרָה אָז לְכָל הַשַּׁבָּת, וְלָכֵן אֵינָהּ חוֹזֶרֶת וְנֶאֱסֶרֶת כְּשֶׁנִּמְלַךְ אַחַר כָּךְ וְחָזַר. אֲבָל כַּאן אֵין הַדָּבָר יָדוּעַ בְּוַדַּאי בִּכְנִיסַת הַשַּׁבָּת שֶׁלֹּא יָבֹא בַּעַל הַבַּיִת זֶה לְבֵיתוֹ בְּשַׁבָּת, וְאֵין אוֹמְרִים שֶׁכֵּיוָן שֶׁלֹּא בָא בִּכְנִיסַת שַׁבָּת, הִסִּיחַ מִלִּבּוֹ וּבְוַדַּאי לֹא יָבֹא כָּל הַשַּׁבָּת:

סעיף ג אֶחָד מִבְּנֵי הֶחָצֵר שֶׁמֵּת בְּשַׁבָּת וְלֹא הִנִּיחַ בְּנֵי בַיִת…

אֶחָד מִבְּנֵי הֶחָצֵר שֶׁמֵּת בְּשַׁבָּת וְלֹא הִנִּיחַ בְּנֵי בַיִת בְּבֵיתוֹ, אִם עֵרֵב, אֲפִלּוּ יְרָשׁוֹ אֶחָד מִבְּנֵי הַשּׁוּק וְנִכְנַס לָדוּר בּוֹ בְּשַׁבָּת — אֵינוֹ אוֹסֵר, אַף עַל פִּי שֶׁהוּא לֹא עֵרֵב בְּחָצֵר זוֹ, לְפִי שֶׁשַּׁבָּת כֵּיוָן שֶׁהֻתְּרָה בְּעֵרוּב הַמּוֹרִישׁ — הֻתְּרָה, וּבִלְבָד שֶׁלֹּא הָיָה הַמּוֹרִישׁ גּוֹסֵס מִבְּעוֹד יוֹם, כְּמוֹ שֶׁיִּתְבָּאֵר.

וְאִם הַמּוֹרִישׁ לֹא עֵרֵב, וִירָשׁוֹ אֶחָד מִן הַשּׁוּק, כָּל זְמַן שֶׁאֵינוֹ בָא לָדוּר בּוֹ — אֵינוֹ אוֹסֵר. בָּא לָדוּר בּוֹ בְּשַׁבָּת — אוֹסֵר. שֶׁלֹּא שַׁיָּךְ כַּאן "שַׁבָּת כֵּיוָן שֶׁהֻתְּרָה הֻתְּרָה" (כֵּיוָן שֶׁלֹּא הֻתְּרָה בִּתְחִלָּתָהּ, וְעוֹד) מִטַּעַם שֶׁנִּתְבָּאֵר לְמַעְלָה.

וְאִם יְרָשׁוֹ אֶחָד מִבְּנֵי הֶחָצֵר — אוֹסֵר אַף עַל פִּי שֶׁלֹּא נִכְנַס עֲדַיִן לָדוּר בּוֹ, מִטַּעַם שֶׁנִּתְבָּאֵר לְמַעְלָה. וַאֲפִלּוּ אִם עֵרֵב עִם בְּנֵי הֶחָצֵר בִּשְׁבִיל בַּיִת שֶׁלּוֹ, אֵין עֵרוּב זֶה מוֹעִיל לְבַיִת זֶה שֶׁיְּרָשׁוֹ בְּשַׁבָּת אַחַר קְנִיַּת הָעֵרוּב.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁהַיּוֹרֵשׁ דָּר בְּבַיִת אַחֵר בְּחָצֵר זוֹ. אֲבָל אִם הָיָה דָר בְּבַיִת זֶה עִם הַמּוֹרִישׁ, וְנָתַן לָעֵרוּב מִשֶּׁלּוֹ בִּשְׁבִיל בַּיִת זֶה שֶׁל הַמּוֹרִישׁ — הֲרֵי זֶה מוֹעִיל, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן ש״ע, שֶׁבְּנֵי בֵיתוֹ שֶׁל אָדָם יְכוֹלִים לְעָרֵב מִשֶּׁלָּהֶן בְּעַד בֵּיתוֹ.

וְאִם מִבְּעוֹד יוֹם מֵת הַמּוֹרִישׁ, וִירָשׁוֹ אֶחָד מִן הַשּׁוּק שֶׁאֵינוֹ מְעָרֵב עִם בְּנֵי הֶחָצֵר — הֲרֵי זֶה אוֹסֵר אִם נִכְנַס לָדוּר בּוֹ בְּשַׁבָּת, אֲפִלּוּ אִם הַמּוֹרִישׁ עֵרֵב כְּבָר בְּחַיָּיו, לְפִי שֶׁכֵּיוָן שֶׁמֵּת — בָּטֵל עֵרוּבוֹ:

סעיף ד יִשְׂרָאֵל וְגֵר שֶׁהָיוּ דָרִים בֶּחָצֵר וְעֵרְבוּ יַחַד, וּמֵת הַגֵּר…

יִשְׂרָאֵל וְגֵר שֶׁהָיוּ דָרִים בֶּחָצֵר וְעֵרְבוּ יַחַד, וּמֵת הַגֵּר מִבְּעוֹד יוֹם, וְהֶחֱזִיק יִשְׂרָאֵל אַחֵר בִּנְכָסָיו, אֲפִלּוּ לֹא הֶחֱזִיק עַד שֶׁחָשֵׁכָה — אוֹסֵר עַד שֶׁיְּבַטֵּל רְשׁוּתוֹ כִּשְׁאָר יוֹרֵשׁ שֶׁיּוּכַל לְבַטֵּל, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שפ״א. וְאַף עַל פִּי שֶׁבִּכְנִיסַת הַשַּׁבָּת, כָּל זְמַן שֶׁלֹּא נִכְנַס יִשְׂרָאֵל זֶה בְּבֵיתוֹ שֶׁל גֵּר, הָיְתָה דִירָה בְּלֹא בְעָלִים וְלֹא הָיְתָה אוֹסֶרֶת — אֵין אוֹמְרִים בָּהּ "שַׁבָּת כֵּיוָן שֶׁהֻתְּרָה הֻתְּרָה", הוֹאִיל וְהָיְתָה רְאוּיָה לְהַחֲזִיק בָּהּ מִבְּעוֹד יוֹם, וְהָיְתָה תְּלוּיָה וְעוֹמֶדֶת, וְלֹא הָיָה אָז הֶתֵּרָהּ בָּרוּר שֶׁתְּהֵא מֻתֶּרֶת כָּל הַשַּׁבָּת.

אֲבָל אִם מִשֶּׁחָשֵׁכָה מֵת הַגֵּר — אֵין הַמַּחֲזִיק אוֹסֵר, שֶׁכֵּיוָן שֶׁבִּכְנִיסַת הַשַּׁבָּת הָיָה מֻתָּר מֵחֲמַת עֵרוּבוֹ שֶׁל גֵּר, אָנוּ אוֹמְרִים "שַׁבָּת כֵּיוָן שֶׁהֻתְּרָה הֻתְּרָה". וּבִלְבָד שֶׁלֹּא הָיָה גוֹסֵס מִבְּעוֹד יוֹם, כְּמוֹ שֶׁיִּתְבָּאֵר:

סעיף ה אֶחָד מִבְּנֵי הֶחָצֵר שֶׁהוּא גוֹסֵס, אַף עַל פִּי שֶׁאֵינוֹ…

אֶחָד מִבְּנֵי הֶחָצֵר שֶׁהוּא גוֹסֵס, אַף עַל פִּי שֶׁאֵינוֹ יָכוֹל לִחְיוֹת בּוֹ בַיּוֹם — אוֹסֵר עַד שֶׁיְּזַכּוּ לוֹ בְּפַת וִיעָרְבוּ עָלָיו. וְאִם מֵת בְּשַׁבָּת — אֵין אוֹמְרִים בּוֹ "שַׁבָּת כֵּיוָן שֶׁהֻתְּרָה הֻתְּרָה", כֵּיוָן שֶׁאֵין הָעֵרוּב רָאוּי לְהִתְקַיֵּם כָּל הַשַּׁבָּת, שֶׁרֹב גּוֹסְסִין לְמִיתָה. וּלְפִיכָךְ אִם יְרָשׁוֹ אֶחָד מִבְּנֵי הֶחָצֵר שֶׁלֹּא עֵרֵב עִמָּהֶם, אוֹסֵר אֲפִלּוּ אִם לֹא נִכְנַס עֲדַיִן לָדוּר בּוֹ, אוֹ אִם יְרָשׁוֹ אֶחָד מִן הַשּׁוּק וְנִכְנַס לָדוּר בּוֹ:

סעיף ו וְכֵן קָטָן אוֹסֵר, אַף עַל פִּי שֶׁאֵינוֹ יָכוֹל לֶאֱכוֹל…

וְכֵן קָטָן אוֹסֵר, אַף עַל פִּי שֶׁאֵינוֹ יָכוֹל לֶאֱכוֹל כְּזַיִת.

אֲבָל אוֹרֵחַ אֵינוֹ אוֹסֵר, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן ש״ע:

מקור: ספריא — שולחן ערוך הרב, אורח חיים שע״א. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה.

2 · לשון הרב ולשון המחבר

לשון הרב מול סעיף המחבר

במה שונה לשון הרב מסעיף המחבר

כל נקודה מעמידה זה לצד זה סעיף של שולחן ערוך המחבר (במספור שלו, שישה סעיפים, המסומנים באותיותיהם) ואת הסעיף המקביל בשולחן ערוך אדמו״ר הזקן (במספור שלו, גם הוא שישה סעיפים, בחלוקה אחרת), לפי סדר סעיפי הרב, ואינה מציינת אלא מה שעולה מהשוואת שני הטקסטים. אין בכך שנקודה מסוימת מיוחדת לרב: בכל נקודה הדף מזכיר את מפרשי המחבר שכבר כותבים אותה — המגן אברהם והט״ז, שקדמו לו; המשנה ברורה והביאור הלכה, המאוחרים לרב — וכשהם קוראים את הנקודה בדברי המחבר עצמם, הדף אומר זאת. הטור והבית יוסף נזכרים במקום שהם כבר כותבים את מה שכותב הרב (בין השאר בנקודות א, ב ו־ד); את הראשונים שהם מביאים לא השווינו אחד אחד.

אין נמנים כאן כנקודות בפני עצמן פרטים בדברי הרב המפרשים את הסעיף המקביל של המחבר. בסעיף ג כותב הרב ביורש שהיה דר עם המוריש « וְנָתַן לָעֵרוּב מִשֶּׁלּוֹ בִּשְׁבִיל בַּיִת זֶה שֶׁל הַמּוֹרִישׁ » (שו״ע הרב שע״א:ג), ונותן טעם « שֶׁבְּנֵי בֵיתוֹ שֶׁל אָדָם יְכוֹלִים לְעָרֵב מִשֶּׁלָּהֶן בְּעַד בֵּיתוֹ » (שו״ע הרב שע״א:ג), ואילו המחבר כותב « אם עירב היורש והיה דר עם המוריש בבי׳ אינו אוסר אע״פ שלא עירב המוריש » (שו״ע או״ח שע״א:ד); המשנה ברורה, המאוחר לרב, נותן אותו טעם: « שבני ביתו של אדם מערבין לו שלא מדעתו » (משנה ברורה שע״א ס״ק כ׳). בסעיף ד כותב הרב שהישראל והגר עירבו יחד — כך מפרש הט״ז את סעיף ה של המחבר: « פי׳ ועירבו ביחד » (ט״ז או״ח שע״א ס״ק ה) — ונותן את הטעם שאין אומרים, כשמת הגר מבעוד יום, שבת כיון שהותרה הותרה, « הוֹאִיל וְהָיְתָה רְאוּיָה לְהַחֲזִיק בָּהּ מִבְּעוֹד יוֹם, וְהָיְתָה תְּלוּיָה וְעוֹמֶדֶת » (שו״ע הרב שע״א:ד), וכבר כתב המגן אברהם: « כיון דראוי להחזיק מבע״י ה״ל רשות זו תלוי׳ ועומדת ולא הותר למקצת שבת » (מגן אברהם שע״א ס״ק ז). וכשמת הגר משחשכה, המחבר כבר נותן את הדין ואת טעמו: « מת משחשיכה אין המחזיק אוסר שבהיתר הראשון עומד » (שו״ע או״ח שע״א:ה).

נקודה א — דירה בלא בעלים, ותנאי הדין (שו״ע הרב שע״א:א)

דירה בלא בעלים

דברי המחבר: « א׳ מבני חצר שהניח ביתו והלך ושבת בחצר אחר׳ אפי׳ היתה סמוכה לחצרו אם הסיח מלבו ואין דעתו לחזור לביתו בשב׳ הרי זה אינו אוסר עליהם » (שו״ע או״ח שע״א:א). סעיף א שלו נותן את הדין בלי לכתוב את טעמו.

דברי הרב: הוא פותח את הסימן בעיקרון, « דִּירָה בְּלֹא בְעָלִים אֵינָהּ דִּירָה וְאֵינָהּ אוֹסֶרֶת בְּחָצֵר » (שו״ע הרב שע״א:א), ומוסיף עליו דין שני, « וְגַם מֻתָּר לְטַלְטֵל מִמֶּנָּה לְחָצֵר שֶׁיֵּשׁ בָּהּ דִּיּוּרִים אֲחֵרִים שֶׁעֵרְבוּ בֵּינֵיהֶם » (שו״ע הרב שע״א:א). ואחר כך, במי שהלך לשבות במקום אחר, הוא כותב שהלך « וְהָלַךְ עִם כָּל בְּנֵי בֵיתוֹ » (שו״ע הרב שע״א:א), שאינו אוסר « אִם עֵרְבוּ בֵּינֵיהֶם, אַף עַל פִּי שֶׁהוּא לֹא עֵרֵב עִמָּהֶם » (שו״ע הרב שע״א:א), וכן « וּמֻתָּרִים לְטַלְטֵל גַּם מִבֵּיתוֹ לְחָצֵר » (שו״ע הרב שע״א:א).

ההבדל: סעיף א של המחבר אינו קובע את עיקרון הדירה בלא בעלים; ואין עיקרון זה מיוחד לרב: הטור כבר פותח בו את הסימן ומוציא ממנו את הדין, « דירה בלא בעלים אינה אוסרת הלכך » (טור או״ח שע״א), והבית יוסף פותח במילים אלה את פירושו לסימן, « דירה בלא בעלים אינה אוסרת וכו׳ » (בית יוסף או״ח שע״א). לא סעיף א של המחבר ולא הטור אומרים שהלך עם כל בני ביתו, שבני החצר האחרים עירבו ביניהם, ולא שמותר לטלטל מן הדירה בלא בעלים, ומביתו, לחצר; לשון הרב כותבת זאת.

אצל מפרשי המחבר: המשנה ברורה, המאוחר לרב, נותן אותו טעם, « והטעם דדירה בלא בעלים לא שמה דירה ואינה אוסרת בחצר אף שלא נתנה חלק בעירוב » (משנה ברורה שע״א ס״ק א), וקורא את התנאי בסעיף המחבר: « מיירי שהלך עם כל ב״ב משם ומבעוד יום » (משנה ברורה שע״א ס״ק ב). כך הוא מפרש אפוא את סעיף המחבר.

השלכה מעשית: לפי לשון הרב, ביתו של מי שהלך עם כל בני ביתו, ואין דעתו לחזור בשבת, אינו אוסר על בני החצר שעירבו ביניהם, ומותרים הם לטלטל מביתו לחצר (שו״ע הרב שע״א:א); ראה פרק 4, נקודה ①.

נקודה ב — מי שהלך משחשכה (שו״ע הרב שע״א:א)

מי שהלך משחשכה

דברי המחבר: « ישראל בן חצר זו שהלך לשבו׳ בחצר אחר׳ ואין דעתו לחזור בשב׳ ואח״כ נמלך בשב׳ וחזר אינו אוסר » (שו״ע או״ח שע״א:ב). אינו אומר באיזו שעה הלך.

דברי הרב: « וַאֲפִלּוּ אִם אַחַר כָּךְ נִמְלַךְ וְחָזַר לְבֵיתוֹ בְּשַׁבָּת, אֵינוֹ אוֹסֵר » (שו״ע הרב שע״א:א), משום « שֶׁכֵּיוָן שֶׁכְּבָר הִסִּיחַ מִלִּבּוֹ בִּכְנִיסַת הַשַּׁבָּת » (שו״ע הרב שע״א:א); ואחר כך הוא מגביל את הדין: « בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁהָלַךְ מִבֵּיתוֹ מִבְּעוֹד יוֹם. אֲבָל אִם הָלַךְ מִשֶּׁחָשֵׁכָה — אוֹסֵר כְּשֶׁחוֹזֵר » (שו״ע הרב שע״א:א).

ההבדל: סעיף ב של המחבר אינו מחלק לפי שעת ההליכה; לשון הרב מגבילה את הדין למי שהלך מבעוד יום, וכותבת שמי שהלך משחשכה אוסר כשחוזר. הגבלה זו למי שהלך מבעוד יום אינה מיוחדת לרב: הטור כבר כותב אותה, עם דין מי שחוזר, « והולך מבע״י לשבות אצל בתו אפילו באות׳ העיר אינו אוסר אפילו חזר בשבת » (טור או״ח שע״א) — והוא מדבר במי שהולך לשבות אצל בתו, ומיד מגביל את הדין למקרה זה; שמי שהלך משחשכה אוסר כשחוזר — הטור אינו כותב. מה דעת הרב בשעה שהוא עדיין אינו שם — אין סעיף א שלו אומר זאת, והדף אינו מסיק זאת.

אצל מפרשי המחבר: המגן אברהם מפרש את לשון הטור, שהניח ביתו מבע״י (וכך מציין גם הביאור הלכה: כן כתב הטור), וכותב: « נ״ל דנקט כן לאשמועי׳ דאפי׳ חזר אין אוסר » (מגן אברהם שע״א ס״ק א) — הראשון משני פירושיו מתאים למה שכותבת לשון הרב; ומיד הוא מציע פירוש אחר, הנוגע לעכו״ם: « א״נ נקט מבע״י משום עכו״ם עססי׳ שפ״ג » (מגן אברהם שע״א ס״ק א); ואחר כך מוסיף: « אבל באמת אפי׳ הי׳ כאן ולא עירב והלך בשבת אינו אוסר » (מגן אברהם שע״א ס״ק א). הביאור הלכה מבין כך את דברי המגן אברהם, « ואף דכתב המגן אברהם דהטור לאו דוקא קאמר שה״ה אם הלך בשבת כשהסיח דעתו שלא לחזור היום » (ביאור הלכה שע״א), וחולק עליו, בין השאר מדברי רש״י: « וכ״כ רש״י בדף מ״ז בהדיא דמיירי מבע״י » (ביאור הלכה שע״א), ומסיק: « וע״כ נראה פשוט דאין להקל כשהלך בשבת אם לא בשעת הדחק אפשר דיש לסמוך על דעת המגן אברהם כשהסיח דעתו בהדיא » (ביאור הלכה שע״א). המשנה ברורה, המאוחר לרב, כותב « מיירי שהלך עם כל ב״ב משם ומבעוד יום » (משנה ברורה שע״א ס״ק ב), ומביא « ויש מקילין אפילו כשהלך בשבת אם לא חזר בו ביום » (משנה ברורה שע״א ס״ק ב). ובדין סעיף ב של המחבר עצמו — מי שנמלך וחזר — כותב הביאור הלכה « עיין בב״י שהביא דעות המחמירין בזה אך הוא נטה להקל » (ביאור הלכה שע״א), ואחר כך « והגר״א בביאורו משמע דדעתו לפסוק כדעת המחמירים » (ביאור הלכה שע״א), ומסיק: « וע״כ יש להחמיר אם לא בשעת הדחק » (ביאור הלכה שע״א); לשון הרב, כמו המחבר, כותבת שאינו אוסר.

השלכה מעשית: לפי לשון הרב, מי שהלך מבעוד יום ואין דעתו לחזור בשבת אינו אוסר, אפילו נמלך וחזר; ומי שהלך משחשכה אוסר כשחוזר (שו״ע הרב שע״א:א). במי שחוזר, הביאור הלכה מסיק שיש להחמיר, אם לא בשעת הדחק (לעיל): היעזר ברב שלך. ראה פרק 4, נקודה ②.

נקודה ג — הנכרי שהלך מביתו (שו״ע הרב שע״א:א)

הנכרי שהלך מביתו

דברי המחבר: באותו סעיף א כותב המחבר « אבל בעכו״ם אפי׳ הלך לשבו׳ בעיר אחר׳ אוסר עליהם עד שישכרו ממנו מקומו » (שו״ע או״ח שע״א:א), ומביא את הדעה « וי״א שגם עכו״ם אם הלך לשבו׳ בחצר אחר׳ אינו אוסר » (שו״ע או״ח שע״א:א); ואחר כך, באותיות קטנות, בסוגריים מרובעים ובלי המילה הגה, כתוב « וכן נראה עיקר » (שו״ע או״ח שע״א:א), וכן « ואם בא העכו״ם בשבת ע״ל סי׳ שפ״ג » (שו״ע או״ח שע״א:א). הטקסט אינו מזכיר את בעל הדברים; הט״ז מייחס את ההכרעה לרמ״א: « והכריע רמ״א להקל בעכום בספק » (ט״ז או״ח שע״א ס״ק א).

דברי הרב: « וְנָכְרִי שֶׁהָלַךְ מִבֵּיתוֹ יִתְבָּאֵר בְּסִמָּן שפ״ג » (שו״ע הרב שע״א:א).

ההבדל: סעיף א של המחבר דן כאן בנכרי שהלך, והמובא הראשון בסוגריים, שהט״ז מייחס לרמ״א, מכריע; והשני אינו מפנה לסימן שפ״ג אלא בנכרי שבא בשבת; לשון הרב אינה דנה בדין זה בסימן זה, ומפנה את כולו לסימן שפ״ג. הדף אינו מביא כאן מה שכותב הרב שם.

השלכה מעשית: בנכרי שהלך מביתו, לשון הרב מפנה לסימן שפ״ג (שו״ע הרב שע״א:א); הדף אינו מסיק כאן הנהגה.

נקודה ד — בית שירשו משחשכה: למה אין אומרים שבת כיון שהותרה הותרה (שו״ע הרב שע״א:ב)

בית שירשו משחשכה

דברי המחבר: « ואם מת משחשיכה אוסר אע״פ שעירב הזוכה עמהם אינו מועיל למה שירש אח״כ בשב׳ » (שו״ע או״ח שע״א:ג).

דברי הרב: הוא מגדיר את המקרה, אחד מן השוק « שֶׁהָיָה לוֹ בַּיִת בְּחָצֵר שֶׁלֹּא הָיָה דָר בָּהּ וְלָכֵן לֹא עֵרֵב בָּהּ » (שו״ע הרב שע״א:ב); ומוסיף ביורש « וְגַם הוּא אָסוּר לְטַלְטֵל בֶּחָצֵר, בֵּין מִבַּיִת שֶׁלּוֹ בֵּין מִבַּיִת זֶה » (שו״ע הרב שע״א:ב), וכן « וַאֲפִלּוּ אִם זֶה הַזּוֹכֶה לֹא נִכְנַס עֲדַיִן בְּבַיִת זֶה לָדוּר בּוֹ, אֵין זוֹ דִירָה בְּלֹא בְעָלִים » (שו״ע הרב שע״א:ב). ואחר כך הוא כותב למה אין אומרים כאן שבת כיון שהותרה הותרה: « לְפִי שֶׁאֵין אוֹמְרִים כֵּן אֶלָּא כְּשֶׁבְּשָׁעָה שֶׁהֻתְּרָה הָיָה הַדָּבָר בָּרוּר שֶׁלֹּא תַחֲזוֹר וְתֵאָסֵר » (שו״ע הרב שע״א:ב), וכן « וְאֵין אוֹמְרִים שֶׁכֵּיוָן שֶׁלֹּא בָא בִּכְנִיסַת שַׁבָּת, הִסִּיחַ מִלִּבּוֹ וּבְוַדַּאי לֹא יָבֹא כָּל הַשַּׁבָּת » (שו״ע הרב שע״א:ב). ובסעיף ג, ביורש מן השוק כשהמוריש לא עירב, הוא נותן שני טעמים: « (כֵּיוָן שֶׁלֹּא הֻתְּרָה בִּתְחִלָּתָהּ, וְעוֹד) מִטַּעַם שֶׁנִּתְבָּאֵר לְמַעְלָה » (שו״ע הרב שע״א:ג).

ההבדל: סעיף ג של המחבר כותב שהיורש אוסר, ואינו אומר שגם הוא עצמו אסור לטלטל, ולא למה אין אומרים כאן שבת כיון שהותרה הותרה; לשון הרב כותבת את שניהם. שהוא אוסר אף קודם שנכנס לבית — לשון הרב כותבת זאת גם במקרה זה, והמשנה ברורה, המאוחר לרב, כותב זאת בסעיף זה (ס״ק י״ב, להלן); והטעם אינו מיוחד לרב: הרא״ש, שמביא הבית יוסף, כותב אותו ביורש הדר בבית אחר שבחצר — המקרה של סעיף ד של המחבר —, « ומיהו אי הוה דייר היורש בחצר בבית אחר אוסר אע״פ שלא נכנס לדור בבית כיון שהוא בחצר וראוי להשתמש בו חשבינן כאילו נכנס לדור בבית ואוסר עליהם כיון שלא עירב » (בית יוסף או״ח שע״א), ומוסיף שעירובו של היורש אינו מועיל לבית שירש בשבת, ועל דין זה הוא מביא ראיה מן הברייתא של סעיף ג של המחבר: « ואע״פ שהיורש עירב עמהם אין אותו עירוב מועיל לבית שירש בשבת כדתניא » (בית יוסף או״ח שע״א); ובסעיף ד כותב המחבר ביורש מבני החצר שהיה דר בבית אחר שהוא אוסר, בלי התנאי שהוא מציב ביורש מן השוק, שיבוא לדור בו, « ואם לא היה דר עם המוריש אלא בבי׳ אחר שבחצר אוסר אע״פ שעירב היורש » (שו״ע או״ח שע״א:ד), ושם קורא המשנה ברורה, המאוחר לרב, אותו טעם: « כיון דיש לו בית באותו חצר שדר בו וראוי לו להשתמש גם באותו בית שירש חשבינן כאלו נכנס לדור בו » (משנה ברורה שע״א ס״ק כ״א).

אצל מפרשי המחבר: הקושיה היא של הב״ח, שמביאים אותה המגן אברהם, « וא״ת נימא שבת הואיל והותרה הותרה (ב״ח) » (מגן אברהם שע״א ס״ק ד), והט״ז, הקורא לו חמיו: « ומו״ח ז״ל הקשה כאן נימא כיון דהותר׳ שבת הותרה » (ט״ז או״ח שע״א ס״ק ג). הט״ז מתרץ: « ולא דק דכאן לא הותר בכניסת שבת שהרי לא היה שם עירוב » (ט״ז או״ח שע״א ס״ק ג). המגן אברהם מתרץ אחרת, על פי רש״י ותוספות, « ולא ראה דברי רש״י ותוס׳ שכתבו דמיירי שהי׳ לו דיר׳ בחצר זו וצריך לערב » (מגן אברהם שע״א ס״ק ד), וכן « ובא לאשמועי׳ דלא נימא כיון דלא בא בכניסת השב׳ הסיח מלבו » (מגן אברהם שע״א ס״ק ד); ובסעיף הקטן שאחריו הוא כותב ביורש מן השוק: « דבדידיה לא אמרי׳ שהסיח מלבו דדוקא כשהי׳ כאן והלך לו אמרי׳ כן » (מגן אברהם שע״א ס״ק ה). לפי לשון הרב אין אומרים שבת כיון שהותרה הותרה אלא כשהיה הדבר ברור בשעה שהותרה שלא תחזור ותיאסר, והוא נותן לדוגמה את מי שהלך מביתו מבעוד יום; לשונו « וְאֵין אוֹמְרִים שֶׁכֵּיוָן שֶׁלֹּא בָא בִּכְנִיסַת שַׁבָּת, הִסִּיחַ מִלִּבּוֹ » (שו״ע הרב שע״א:ב) קרובה ללשון המגן אברהם בסעיף קטן ד שהובאה לעיל, והדוגמה שלו מתאימה למה שכתב בסעיף קטן ה. אין הדף אומר בכך שהרב הולך בכל דבר אחר פירוש המגן אברהם לסוגיה. הביאור הלכה מביא שתירוץ הט״ז נדחה, « וכבר השיג הגר״א וכן שארי אחרונים על תירוץ הט״ז » (ביאור הלכה שע״א), וכן תירוץ המגן אברהם בסעיף קטן ד, « ועיין מה שתירץ המגן אברהם בסק״ד וגם ע״ז השיג הגר״א » (ביאור הלכה שע״א), וכותב: « אכן לפי מה שכתב המגן אברהם בסק״ה מתורץ קושית הב״ח ואין אנו צריכין כלל לתירוצו שתירץ בסק״ד » (ביאור הלכה שע״א). ובסעיף ג של שו״ע הרב, הטעם הראשון מבין שני טעמי הרב, שבסוגריים בלשונו, מתאים לטעם הט״ז. המשנה ברורה, המאוחר לרב, כותב כמוהו שהיורש אוסר אף קודם שנכנס: « אפילו אם לא בא היורש לדור באותו בית מ״מ מקרי דירה בבעלים כיון שהוא דר באותה חצר » (משנה ברורה שע״א ס״ק י״ב).

השלכה מעשית: לפי לשון הרב, אם מת האחד מן השוק משחשכה, בן החצר שיורש את ביתו אוסר על בני החצר, וגם הוא עצמו אסור לטלטל בחצר, בין מביתו בין מבית זה, אף אם לא נכנס עדיין לדור בו ואף אם עירב עמהם (שו״ע הרב שע״א:ב); ראה פרק 4, נקודה ③.

נקודה ה — גוסס שמת בשבת (שו״ע הרב שע״א:ה)

גוסס שמת בשבת

דברי המחבר: בסעיף ד: « א׳ מן החצר שמת בשב׳ אם עירב אפי׳ ירשו א׳ מבני השוק אינו אוסר » (שו״ע או״ח שע״א:ד); ובסעיף ו: « א׳ מבני חצר שהיה גוסס אף על פי שאינו יכול לחיות בו ביום אוסר עד שיזכו לו בפת ויערבו עליו » (שו״ע או״ח שע״א:ו).

דברי הרב: נקודה זו מקרבת לסעיף ה של הרב את התנאים שהרב מתנה, באותו ענין, בסעיפים ג ו־ד שלו; המובאות כאן לפי סדר הסעיפים. בסעיפים ג ו־ד, כשהוא כותב שהשבת הותרה בעירוב המוריש, הוא מוסיף « וּבִלְבָד שֶׁלֹּא הָיָה הַמּוֹרִישׁ גּוֹסֵס מִבְּעוֹד יוֹם » (שו״ע הרב שע״א:ג), וכן « וּבִלְבָד שֶׁלֹּא הָיָה גוֹסֵס מִבְּעוֹד יוֹם » (שו״ע הרב שע״א:ד). ובסעיף ה של הרב: אם מת בשבת אין אומרים בו שבת כיון שהותרה הותרה, « כֵּיוָן שֶׁאֵין הָעֵרוּב רָאוּי לְהִתְקַיֵּם כָּל הַשַּׁבָּת, שֶׁרֹב גּוֹסְסִין לְמִיתָה » (שו״ע הרב שע״א:ה), ומכאן הוא כותב: « וּלְפִיכָךְ אִם יְרָשׁוֹ אֶחָד מִבְּנֵי הֶחָצֵר שֶׁלֹּא עֵרֵב עִמָּהֶם, אוֹסֵר אֲפִלּוּ אִם לֹא נִכְנַס עֲדַיִן לָדוּר בּוֹ, אוֹ אִם יְרָשׁוֹ אֶחָד מִן הַשּׁוּק וְנִכְנַס לָדוּר בּוֹ » (שו״ע הרב שע״א:ה).

ההבדל: סעיף ד של המחבר כותב שהיורש אינו אוסר, בלי להוציא את המקרה שהמוריש היה גוסס מבעוד יום; וסעיף ו שלו כותב שהגוסס אוסר עד שיזכו לו בפת ויערבו עליו, ואינו אומר מה הדין אם מת בשבת. לשון הרב כותבת את שניהם.

אצל מפרשי המחבר: המגן אברהם כותב זאת לפניו, על סעיף ו של המחבר: « ואם מת בשבת אוסר כיון שאין העירוב ראוי להתקיים כל השבת דרוב גוססין למיתה » (מגן אברהם שע״א ס״ק ח), ומציין « הגהות תוספות דף ס״ה » (מגן אברהם שע״א ס״ק ח). המשנה ברורה, המאוחר לרב, מביא את דבריו ומפרט את המקרים: « אך המ״א מיירי שירשו אחד מן השוק ובא לדור שם בשבת או שירשו אחד מבני החצר שדר בבית אחר » (משנה ברורה שע״א ס״ק ל״ג). גם לשון הרב מזכירה את האחד מן השוק שבא לדור בו; אבל ביורש מבני החצר היא כותבת « שֶׁלֹּא עֵרֵב עִמָּהֶם » (שו״ע הרב שע״א:ה), ואילו המשנה ברורה כותב « שדר בבית אחר » (משנה ברורה שע״א ס״ק ל״ג). הביאור הלכה מסיק מדברי המגן אברהם לסעיף ד של המחבר: « ע״כ מיירי הכא שבעת כניסת השבת לא היה עדיין גוסס » (ביאור הלכה שע״א); אבל הוא מביא שבעל עצי אלמוגים חולק על הדין, « ועיין בספר עצי אלמוגים שמפקפק מאד על עיקר דין זה » (ביאור הלכה שע״א), ובין השאר משום שבסעיף ד הטור והשולחן ערוך לא אמרו בו דבר, « ולא התנו דדוקא כשלא היה גוסס » (ביאור הלכה שע״א), ומסיק: « ובשעת הדחק נראה דיש לסמוך ע״ז » (ביאור הלכה שע״א). אם שתיקת המחבר בסעיף ד היא מחלוקת או מקרה שלא כתבו — הדף אינו מכריע.

השלכה מעשית: לפי לשון הרב, הגוסס אוסר עד שיזכו לו בפת ויערבו עליו; אם מת בשבת, בן החצר שלא עירב עמהם ויורש אותו אוסר אף קודם שנכנס לדור בו, ואחד מן השוק שיורש אותו אוסר אם נכנס לדור בו (שו״ע הרב שע״א:ה); ומוריש שהיה גוסס מבעוד יום — אין אומרים בו שבת כיון שהותרה הותרה (שו״ע הרב שע״א:ג, שו״ע הרב שע״א:ד). הביאור הלכה כותב שבשעת הדחק נראה שיש לסמוך על הדעה החולקת (לעיל): היעזר ברב שלך. ראה פרק 4, נקודה ⑥.

4 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן שע״א, ברגישות החב״דית. כל נקודה מפנה לסעיף שלה בשולחן ערוך הרב, לפי סדר הלשון.

לחסיד חב״ד — סימן שע״א

  • ① דירה בלא בעלים: אינה אוסרת בחצר, ומותר לטלטל ממנה לחצר שבניה עירבו ביניהם. מי שהניח ביתו והלך עם כל בני ביתו לשבות בחצר אחרת, אפילו סמוכה, ואין דעתו לחזור בשבת — אינו אוסר על בני חצרו אם עירבו ביניהם, אף על פי שהוא לא עירב עמהם; ומותרים הם לטלטל מביתו לחצר (שו״ע הרב שע״א:א). הפניה: שולחן ערוך אדמו״ר הזקן שע״א:א.
  • ② אם חזר בשבת: אם הלך מבעוד יום, והסיח מלבו ואין דעתו לחזור בשבת, ונמלך וחזר בשבת — אינו אוסר; אם הלך משחשכה — אוסר כשחוזר (שו״ע הרב שע״א:א). במי שחוזר — היעזר ברב שלך (מה שכותבים בזה מפרשי המחבר — ראה פרק 2, נקודה ב). דין הנכרי שהלך מביתו נדחה לסימן שפ״ג. ראה פרק 2, נקודות ב ו־ג. הפניה: שולחן ערוך אדמו״ר הזקן שע״א:א.
  • ③ ביתו של אחד מן השוק: מי שהיה לו בית בחצר שלא היה דר בה ולא עירב בה, ומת והניח רשותו לאחד מבני החצר: אם מת מבעוד יום — היורש אינו אוסר, שעירובו מתיר גם את הבית שירש; אם מת משחשכה — היורש אוסר על בני החצר, וגם הוא עצמו אסור לטלטל בחצר, בין מביתו בין מבית זה, אף אם עירב עמהם ואף אם לא נכנס עדיין לדור בבית זה (שו״ע הרב שע״א:ב). הפניה: שולחן ערוך אדמו״ר הזקן שע״א:ב.
  • ④ בן החצר שמת בשבת: ולא הניח בני בית בביתו. אם עירב — היורש מן השוק אינו אוסר, אפילו נכנס לדור בו בשבת, ובלבד שלא היה המוריש גוסס מבעוד יום. אם לא עירב — היורש מן השוק אינו אוסר כל זמן שאינו בא לדור בו, ואוסר אם בא לדור בו בשבת; היורש הדר בבית אחר בחצר אוסר אף קודם שנכנס לדור בו, ואף אם עירב בשביל ביתו; אבל אם היה דר בבית זה עם המוריש ונתן לעירוב משלו בשביל בית זה — הרי זה מועיל. ואם מת המוריש מבעוד יום, היורש מן השוק שאינו מערב עם בני החצר אוסר אם נכנס לדור בו בשבת, אפילו עירב המוריש בחייו (שו״ע הרב שע״א:ג). הפניה: שולחן ערוך אדמו״ר הזקן שע״א:ג.
  • ⑤ ישראל וגר: שעירבו יחד. אם מת הגר מבעוד יום — ישראל אחר שהחזיק בנכסיו אוסר, אפילו לא החזיק עד שחשכה, עד שיבטל רשותו כשאר יורש (הרב מפנה לסימן שפ״א); אם מת הגר משחשכה — המחזיק אינו אוסר, ובלבד שלא היה הגר גוסס מבעוד יום (שו״ע הרב שע״א:ד). הפניה: שולחן ערוך אדמו״ר הזקן שע״א:ד.
  • ⑥ גוסס: בן החצר שהוא גוסס, אף על פי שאינו יכול לחיות בו ביום, אוסר עד שיזכו לו בפת ויערבו עליו; אם מת בשבת אין אומרים בו שבת כיון שהותרה הותרה: בן החצר שלא עירב עמהם ויורש אותו אוסר אף קודם שנכנס לדור בו, ואחד מן השוק שיורש אותו אוסר אם נכנס לדור בו (שו״ע הרב שע״א:ה). אם מת בשבת — היעזר ברב שלך (מה שכותבים בזה מפרשי המחבר — ראה פרק 2, נקודה ה). הפניה: שולחן ערוך אדמו״ר הזקן שע״א:ה.
  • ⑦ קטן ואורח: הקטן אוסר, אף על פי שאינו יכול לאכול כזית; האורח אינו אוסר — הרב מפנה לסימן ש״ע · 370 (שו״ע הרב שע״א:ו). הפניה: שולחן ערוך אדמו״ר הזקן שע״א:ו.
