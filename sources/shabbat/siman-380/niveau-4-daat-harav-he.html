הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן ש״פ · 380 — דִּינֵי בִּטּוּל רְשׁוּת — וּבוֹ ז׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. ז׳ סעיפים (בשולחן ערוך של המחבר ד׳, במספור שלו).

סעיף א דִּינֵי בִּטּוּל רְשׁוּת וּבוֹ ז' סְעִיפִים: אֶחָד מִבְּנֵי הֶחָצֵר שֶׁשָּׁכַח וְלֹא עֵרֵב עִמָּהֶם…

דִּינֵי בִּטּוּל רְשׁוּת וּבוֹ ז' סְעִיפִים:

אֶחָד מִבְּנֵי הֶחָצֵר שֶׁשָּׁכַח וְלֹא עֵרֵב עִמָּהֶם — אוֹסֵר עֲלֵיהֶם. מַה תַּקָּנָתָם? יְבַטֵּל לָהֶם רְשׁוּת שֶׁיֵּשׁ לוֹ בֶּחָצֵר, שֶׁיֹּאמַר "רְשׁוּתִי מְבֻטֶּלֶת לָכֶם" אוֹ "רְשׁוּתִי קְנוּיָה לָכֶם". וְאֵין צָרִיךְ לִקְנוֹת בְּקִנְיַן סוּדָר. וְיָכוֹל לְבַטֵּל מִשֶּׁתֶּחְשַׁךְ. וְאַף עַל פִּי שֶׁאֵין קוֹנִין קִנְיָן בְּשַׁבָּת, מִכָּל מָקוֹם כַּאן אֵין צָרִיךְ שֶׁיִּקְנוּ הֵם רְשׁוּתוֹ, אֶלָּא שֶׁהוּא יְסַלֵּק עַצְמוֹ מֵרְשׁוּתוֹ, וּמִמֵּילָא אֵין מִי שֶׁיֶּאֱסֹר עֲלֵיהֶם; וּלְסַלֵּק עַצְמוֹ מֵרְשׁוּתוֹ מֻתָּר אַף בְּשַׁבָּת.
וְאִם דָּר עִם ד' אוֹ ה' — צָרִיךְ לְבַטֵּל לְכָל אֶחָד וְאֶחָד, שֶׁיֹּאמַר "רְשׁוּתִי מְבֻטֶּלֶת לְכֻלְּכֶם", וְלֹא דַי שֶׁיְּבַטֵּל לְאֶחָד מֵהֶם, אַף עַל פִּי שֶׁזֶּה שֶׁמְּבַטֵּל לוֹ כְּבָר עֵרֵב עִם הַשְּׁאָר. וְאִם לֹא עָשָׂה כֵּן, אַף לְמִי שֶׁבִּטֵּל לוֹ אָסוּר. וְיֵשׁ אוֹמְרִים שֶׁלֹּא דַי שֶׁיֹּאמַר "רְשׁוּתִי מְבֻטֶּלֶת לְכֻלְּכֶם", אֶלָּא צָרִיךְ שֶׁיְּפָרֵשׁ וְיֹאמַר "מְבֻטֶּלֶת לְךָ וּלְךָ". וּלְעִנְיַן הֲלָכָה, בְּדִבְרֵי סוֹפְרִים הַלֵךְ אַחַר הַמֵּקֵל:

סעיף ב אִם בִּטֵּל לָהֶם רְשׁוּתוֹ סְתָם — לֹא בִטֵּל אֶלָּא רְשׁוּת…

אִם בִּטֵּל לָהֶם רְשׁוּתוֹ סְתָם — לֹא בִטֵּל אֶלָּא רְשׁוּת שֶׁיֵּשׁ לוֹ בֶּחָצֵר. לְפִיכָךְ הֵם מֻתָּרִים לְהוֹצִיא מִבָּתֵּיהֶם לֶחָצֵר וְגַם הוּא, לְפִי שֶׁהוּא נַעֲשֶׂה אֶצְלָם כְּאוֹרֵחַ בְּעָלְמָא. אֲבָל אֲסוּרִים לְהוֹצִיא מִבֵּיתוֹ לֶחָצֵר וְכֵן לְהַכְנִיס, שֶׁהֲרֵי בֵּיתוֹ הוּא מְיֻחָד לוֹ וְהֶחָצֵר לָהֶם, וַהֲרֵי זֶה מוֹצִיא מֵרְשׁוּת לִרְשׁוּת. וַאֲפִלּוּ הוּא, שֶׁבֵּיתוֹ שֶׁלּוֹ וְגַם בֶּחָצֵר הוּא כְּאוֹרֵחַ אֶצְלָם, אַף עַל פִּי כֵן אָסוּר לוֹ לְהוֹצִיא מִבֵּיתוֹ לֶחָצֵר (וְכֵן לְהַכְנִיס), לְפִי שֶׁאֵין אוֹמְרִים שֶׁהוּא כְּאוֹרֵחַ אֶצְלָם אֶלָּא לְעִנְיַן תַּשְׁמִישׁ מִבָּתֵּיהֶם לֶחָצֵר, אֲבָל תַּשְׁמִישׁ שֶׁמִּבֵּיתוֹ אוֹ לְבֵיתוֹ — אֵין לוֹמַר שֶׁהוּא אוֹרֵחַ. וְיֵשׁ אוֹמְרִים שֶׁצָּרִיךְ לִנְעוֹל בֵּיתוֹ, כְּדֵי שֶׁלֹּא יָבֹא לְהוֹצִיא בְּאִסּוּר, וְלֹא יִפְתָּחֶנּוּ אֶלָּא כְּשֶׁרוֹצֶה לָצֵאת וְלָבֹא, וְיִנְעָלֶנּוּ מִיָּד אַחַר צֵאתוֹ וּבוֹאוֹ.ֹ וְהַמֵּקֵל לֹא הִפְסִיד, כִּי יֵשׁ לוֹ עַל מִי לִסְמוֹךְ:

סעיף ג אִם בִּטֵּל לָהֶם גַּם רְשׁוּת בֵּיתוֹ — מֻתָּרִים גַּם בְּבֵיתוֹ…

אִם בִּטֵּל לָהֶם גַּם רְשׁוּת בֵּיתוֹ — מֻתָּרִים גַּם בְּבֵיתוֹ לְהוֹצִיא מִמֶּנּוּ לֶחָצֵר וְכֵן לְהַכְנִיס כְּמוֹ בְּבָתֵּיהֶם, וְגַם הוּא מֻתָּר בְּבֵיתוֹ כְּמוֹ בְּבָתֵּיהֶם, לְפִי שֶׁנַּעֲשֶׂה אוֹרֵחַ גָּמוּר אֶצְלָם, כֵּיוָן שֶׁלֹּא נִשְׁאַר לוֹ שׁוּם רְשׁוּת שֶׁלֹּא בִטְּלָה לָהֶם. רַק שֶׁיִּזָּהֵר שֶׁלֹּא יוֹצִיא מִבֵּיתוֹ לֶחָצֵר קֹדֶם שֶׁיּוֹצִיאוּ [הֵ]ם מִבָּתֵּיהֶם לֶחָצֵר, כְּדֵי שֶׁיַּחֲזִיקוּ הֵם בֶּחָצֵר תְּחִלָּה, וְלֹא יַכְנִיס מֵחָצֵר לְבֵיתוֹ קֹדֶם שֶׁיַּכְנִיסוּ הֵם מֵחָצֵר לְבֵיתוֹ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שפ״א:

סעיף ד אִם אֵינוֹ רוֹצֶה לְבַטֵּל לָהֶם רְשׁוּתוֹ, אֶלָּא לְהַשְׂכִּירוֹ…

אִם אֵינוֹ רוֹצֶה לְבַטֵּל לָהֶם רְשׁוּתוֹ, אֶלָּא לְהַשְׂכִּירוֹ — יֵשׁ אוֹמְרִים שֶׁשְּׂכִירוּת מוֹעֶלֶת בְּיִשְׂרָאֵל כְּמוֹ שֶׁמּוֹעֶלֶת בְּנָכְרִי, לְפִי שֶׁהַשְּׂכִירוּת אֵינָהּ גָּרוּעַ מֵהַבִּטּוּל שֶׁהִיא מַתָּנָה בְּחִנָּם. וְיֵשׁ אוֹמְרִים שֶׁלֹּא תִקְּנוּ שְׂכִירוּת אֶלָּא בְּנָכְרִי שֶׁאֵין בִּטּוּלוֹ מוֹעִיל, אֲבָל לֹא בְּיִשְׂרָאֵל. וּלְעִנְיַן הֲלָכָה, בְּדִבְרֵי סוֹפְרִים הַלֵּךְ אַחַר הַמֵּקֵל:

סעיף ה אִם בְּנֵי הֶחָצֵר שֶׁעֵרְבוּ מְבַטְּלִין רְשׁוּתָם לְאֶחָד שֶׁלֹּא עֵרֵב עִמָּהֶם…

אִם בְּנֵי הֶחָצֵר שֶׁעֵרְבוּ מְבַטְּלִין רְשׁוּתָם לְאֶחָד שֶׁלֹּא עֵרֵב עִמָּהֶם — הוּא מֻתָּר לְהוֹצִיא מִבֵּיתוֹ לֶחָצֵר, וְלֹא מִבָּתֵּיהֶם, אֶלָּא אִם כֵּן בִּטְּלוּ לוֹ בְּפֵרוּשׁ גַּם רְשׁוּת בָּתֵּיהֶם.
וְהֵם אֲסוּרִים אֲפִלּוּ מִבֵּיתוֹ לֶחָצֵר. וְאַף עַל פִּי שֶׁרְשׁוּת אֶחָד הֵן לְאַחַר שֶׁבִּטְּלוּהוּ לוֹ, מִכָּל מָקוֹם כֵּיוָן שֶׁחָצֵר זוֹ הָיְתָה שֶׁלָּהֶם, אִם כֵּן כְּשֶׁחוֹזְרִין וּמוֹצִיאִים לְתוֹכָהּ הֲרֵי זֶה כְּאִלּוּ חוֹזְרִים בָּהֶם וְרוֹצִים לִזְכּוֹת בָּהּ, שֶׁאֵין לוֹמַר שֶׁהֵם מִשְׁתַּמְּשִׁים בָּהּ כְּאוֹרְחִים בְּעָלְמָא, לְפִי שֶׁהָרַבִּים לְגַבֵּי יָחִיד אֵינָן דּוֹמִין כְּאוֹרְחִים אֶצְלוֹ. וַאֲפִלּוּ אִם כְּבָר הֶחֱזִיק הוּא בֶּחָצֵר — אֵין זֶה מוֹעִיל שֶׁלֹּא יוּכְלוּ הָרַבִּים לַחֲזוֹר בָּהֶם.
אֲבָל מֻתָּרִים לְהַכְנִיס מֵחָצֵר לְבֵיתוֹ, לְפִי שֶׁאֵין נִרְאֶה חֲזָרָה אֶלָּא כְּשֶׁמַּכְנִיסִים לִרְשׁוּת שֶׁלָּהֶם שֶׁבִּטְּלוּהָ, דְּהַיְנוּ לֶחָצֵר, אֲבָל כְּשֶׁמּוֹצִיאִים מֵרְשׁוּת זוֹ וּמַכְנִיסִים לִרְשׁוּת אַחֶרֶת — אֵין זוֹ נִרְאֵית חֲזָרָה.
אֲבָל אֲסוּרִים לְהַכְנִיס מֵחָצֵר לְבָתֵּיהֶם אֲפִלּוּ אִם בִּטְּלוּ לוֹ גַּם רְשׁוּת בָּתֵּיהֶם, שֶׁנַּעֲשָׂה הַכֹּל רְשׁוּת אֶחָד שֶׁלּוֹ — מִכָּל מָקוֹם כֵּיוָן שֶׁהֵם אֵינָן כְּאוֹרְחִים לְגַבֵּי הַיָּחִיד, אִם כֵּן כְּשֶׁמַּכְנִיסִים לְבָתֵּיהֶם הֲרֵי זֶה כְּאִלּוּ חוֹזְרִים וְזוֹכִים בְּבָתֵּיהֶם. וְאֵין צָרִיךְ לוֹמַר שֶׁאֲסוּרִים לְהוֹצִיא מִבָּתֵּיהֶם לֶחָצֵר אֲפִלּוּ בִּטְּלוּ לוֹ גַּם רְשׁוּת בָּתֵּיהֶם, כְּמוֹ שֶׁאֲסוּרִים לְהוֹצִיא אֲפִלּוּ מִבֵּיתוֹ מַמָּשׁ לֶחָצֵר:

סעיף ו וְכֵן אִם הָיוּ שְׁנַיִם לְבָד דָּרִים בֶּחָצֵר וְלֹא עֵרְבוּ…

וְכֵן אִם הָיוּ שְׁנַיִם לְבָד דָּרִים בֶּחָצֵר וְלֹא עֵרְבוּ, וּבִטֵּל אֶחָד מֵהֶם לַחֲבֵרוֹ — דִּינוֹ כְּרַבִּים לְגַבֵּי יָחִיד, שֶׁאַף הַיָּחִיד אֵינוֹ נַעֲשֶׂה אוֹרֵחַ לְגַבֵּי יָחִיד אֶלָּא לְגַבֵּי רַבִּים, וְאֵין רַבִּים פְּחוּתִים מִשְּׁנַיִם. וְכֵן רַבִּים שֶׁלֹּא עֵרְבוּ שֶׁבִּטְּלוּ רְשׁוּתָם לְרַבִּים שֶׁעֵרְבוּ, אֵינָם נַעֲשִׂים אוֹרְחִים אֶצְלָם, וְדִינָם כְּרַבִּים לְגַבֵּי יָחִיד:

סעיף ז רַבִּים שֶׁלֹּא עֵרְבוּ יְכוֹלִים לְבַטֵּל רְשׁוּתָם בֵּין…

רַבִּים שֶׁלֹּא עֵרְבוּ יְכוֹלִים לְבַטֵּל רְשׁוּתָם בֵּין לְיָחִיד שֶׁלֹּא עֵרֵב בֵּין לְרַבִּים שֶׁעֵרְבוּ בֵּינֵיהֶם, וּבִלְבָד שֶׁכָּל אֶחָד מֵהַמְבַטְּלִים יְבַטֵּל לְכָל אֶחָד מֵהֶם, כְּמוֹ שֶׁנִּתְבָּאֵר לְמַעְלָה בְּיָחִיד הַמְבַטֵּל. אֲבָל לְרַבִּים שֶׁלֹּא עֵרְבוּ, אֲפִלּוּ לִשְׁנַיִם שֶׁלֹּא עֵרְבוּ — אֵין מְבַטְּלִין לָהֶם, בֵּין שֶׁהַמְבַטְּלִים הֵם רַבִּים שֶׁעֵרְבוּ בֵּין שֶׁהוּא יָחִיד שֶׁלֹּא עֵרֵב, שֶׁהֲרֵי אַף לְאַחַר שֶׁיְּבַטְּלוּ לָהֶם הֵם שְׁנַיִם, וְאֶחָד אוֹסֵר עַל חֲבֵרוֹ.
וַאֲפִלּוּ אִם חָזַר הָאֶחָד וּבִטֵּל לַחֲבֵרוֹ — אֵינוֹ מוֹעִיל כְּלוּם, שֶׁכֵּיוָן שֶׁבְּשָׁעָה שֶׁבִּטְּלוּ הָרִאשׁוֹנִים הֶחָצֵר לְזֶה הַחוֹזֵר וּמְבַטֵּל לֹא הָיָה לוֹ אָז בָּהּ זְכוּת שֶׁל הֶתֵּר טִלְטוּל, מִפְּנֵי שֶׁאוֹסְרִים זֶה עַל זֶה, אִם כֵּן בִּטּוּל זֶה לֹא הוֹעִיל כְּלוּם, וּכְשֶׁחוֹזֵר וּמְבַטֵּל לְזֶה אֵינוֹ יָכוֹל לְבַטֵּל רְשׁוּת שֶׁלָּהֶם, שֶׁהֲרֵי לֹא זָכוּ בָּהּ. וַאֲפִלּוּ כְּשֶׁבִּטְּלוֹ לוֹ הָרִאשׁוֹנִים אָמְרוּ לוֹ "אָנוּ מְבַטְּלִים לְךָ עַל מְנָת שֶׁתַּחֲזוֹר וּתְבַטֵּל לַחֲבֵרְךָ", שֶׁנָּתְנוּ לָהֶם רְשׁוּת לְבַטֵּל לוֹ — אֵינוֹ מוֹעִיל כְּלוּם, שֶׁכֵּיוָן שֶׁלֹּא זָכָה בְּחֵלֶק שֶׁבִּטְּלוּ לוֹ בְּעִנְיָן שֶׁיּוּכַל לְטַלְטֵל בֶּחָצֵר, אֵינוֹ יָכוֹל לְהַקְנוֹת לַחֲבֵרוֹ:

מקור: ספריא — שולחן ערוך הרב, אורח חיים סימן ש״פ · 380. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה. הטקסט מועתק כפי שספריא מגישה אותו: בסעיף ב, אחרי הנקודה שבסוף המשפט על נעילת הבית, בא סימן ניקוד (חולם); בסעיף ג, אות אחת בתוך סוגריים מרובעים; שניהם הושארו כמות שהם.

2 · לשון הרב ולשון המחבר

לשון הרב מול סעיף המחבר

במה שונה לשון הרב מסעיף המחבר

כל נקודה מעמידה זה לצד זה סעיף מן השולחן ערוך של המחבר (במספור שלו, ד׳ סעיפים) ואת הסעיף המקביל בשולחן ערוך אדמו״ר הזקן (במספור שלו, ז׳ סעיפים), לפי סדר סעיפי הרב, ומציינת רק מה שההשוואה בין הלשונות מראה. בעיקרו של דבר החלוקות מקבילות כך (סעיף הרב ← סעיף המחבר): א ו־ב ← א; ג ← ב; ד ← ג; ה, ו ו־ז ← ד. מה שחסר בלשון המחבר אינו חידוש בשל כך: בסימן זה כמה מן הפירושים שבלשון הרב כתובים כבר במשנה ובגמרא, ברש״י וברמב״ם, בטור, בבית יוסף, במגן אברהם או בט״ז, והמשנה ברורה, שכתב אחריו, מביא כמה מהם; ואחד מהם הוא אף לומד מלשון המחבר עצמו (נקודה ו). בכל נקודה מצוין היכן הם כתובים.

נקודה א — נוסח הביטול לדר עם ארבעה או חמשה (שו״ע הרב ש״פ:א)

נוסח הביטול

דברי המחבר: בסעיף א הוא כותב « ואם דר עם ארבעה או חמשה צריך לבטל לכל א' וא' שיאמר רשותי מבוטלת לך ולך וי״א שדי שיאמר רשותי מבוטל' לכולכם » (שו״ע או״ח ש״פ:א): הנוסח „לך ולך” נאמר בסתם, והנוסח „לכולכם” מובא כדעת יש אומרים.

דברי הרב: לביטול משתחשך הוא נותן טעם: « וְאַף עַל פִּי שֶׁאֵין קוֹנִין קִנְיָן בְּשַׁבָּת, מִכָּל מָקוֹם כַּאן אֵין צָרִיךְ שֶׁיִּקְנוּ הֵם רְשׁוּתוֹ, אֶלָּא שֶׁהוּא יְסַלֵּק עַצְמוֹ מֵרְשׁוּתוֹ, וּמִמֵּילָא אֵין מִי שֶׁיֶּאֱסֹר עֲלֵיהֶם; וּלְסַלֵּק עַצְמוֹ מֵרְשׁוּתוֹ מֻתָּר אַף בְּשַׁבָּת » (שו״ע הרב ש״פ:א); ואחר כך, בדר עם ארבעה או חמשה, הוא קובע כעיקר את הנוסח הפונה אל כולם, ומפרש שאין די בביטול לאחד: « וְאִם דָּר עִם ד' אוֹ ה' — צָרִיךְ לְבַטֵּל לְכָל אֶחָד וְאֶחָד, שֶׁיֹּאמַר "רְשׁוּתִי מְבֻטֶּלֶת לְכֻלְּכֶם", וְלֹא דַי שֶׁיְּבַטֵּל לְאֶחָד מֵהֶם, אַף עַל פִּי שֶׁזֶּה שֶׁמְּבַטֵּל לוֹ כְּבָר עֵרֵב עִם הַשְּׁאָר » (שו״ע הרב ש״פ:א); ומוסיף « וְאִם לֹא עָשָׂה כֵּן, אַף לְמִי שֶׁבִּטֵּל לוֹ אָסוּר » (שו״ע הרב ש״פ:א); ואחר כך מביא את הנוסח האחר כדעת יש אומרים, ופוסק: « וְיֵשׁ אוֹמְרִים שֶׁלֹּא דַי שֶׁיֹּאמַר "רְשׁוּתִי מְבֻטֶּלֶת לְכֻלְּכֶם", אֶלָּא צָרִיךְ שֶׁיְּפָרֵשׁ וְיֹאמַר "מְבֻטֶּלֶת לְךָ וּלְךָ". וּלְעִנְיַן הֲלָכָה, בְּדִבְרֵי סוֹפְרִים הַלֵךְ אַחַר הַמֵּקֵל » (שו״ע הרב ש״פ:א).

מה שכתוב כבר לפניו: הטור כותב כבר את הנוסח „לכולכם” לצד „לך ולך”, בלי להביא חולק: « צריך לבטל לכל א' ואחד שיאמר רשותי מבוטלת לך ולך או יאמר לכולכם בכלל » (טור או״ח ש״פ); הבית יוסף מביא שהמגיד משנה לומד אותו מדברי רש״י: « ומ״ש או יאמר לכולכם בכלל בפ״ב מה״ע כתב ה״ה דהכי משמע מדברי רש״י בספ״ב דעירובין » (בית יוסף או״ח ש״פ), ושלדבריו לשון הרמב״ם בזה צריך תלמוד. לביטול משתחשך כותב המגן אברהם « דאסתלוקי רשותא בשבתא שרי » (מגן אברהם ש״פ ס״ק א); לביטול לכל אחד הוא כותב, בשם המגיד משנה, « ואם לא עשה כן אף למי שביטל אסור » (מגן אברהם ש״פ ס״ק ב). הט״ז לומד מסעיף המחבר שלדעה הראשונה אין „לכולכם” מועיל, « משמע דלדיע' זו לא מהני לכולכם » (ט״ז או״ח ש״פ ס״ק א), ומציע לכך טעם: « ונרא' הטעם דמצינו רובו ככולו וע״כ אי אמר לכולכם אפשר דלא נתכוין רק על הרוב » (ט״ז או״ח ש״פ ס״ק א). שאין די בביטול לאחד אף כשזה כבר עירב עם השאר, נותן רש״י טעם: « דאע״ג דהאי דבטיל ליה עירב עם חבירו כי עירב מאתמול אדעתא דההיא חולקא לא עירב דההיא שעתא לאו דידיה הוא » (רש״י עירובין כ״ו ע״ב); והביאור הלכה, שכתב אחרי הרב, מביאו על סעיף זה.

ההבדל: המחבר כותב בסתם את הנוסח „לך ולך”, ומביא כדעת יש אומרים ש„לכולכם” מועיל; הרב כותב כעיקר את הנוסח „לכולכם” — שהטור כותב כבר לצד „לך ולך” —, מביא כדעת יש אומרים שצריך לפרש „לך ולך”, וכותב שלענין הלכה בדברי סופרים הולכים אחר המקל. המשנה ברורה, שכתב אחריו, מביא שגם האליה רבה מסיק ש„לכולכם” מועיל, « ולדינא מסיק הא״ר דהעיקר כהי״א דמהני גם כשאמר לכולכם » (משנה ברורה ש״פ ס״ק ה), ובהמשך דבריו « וכ״ז כשעומד נגד כולם ואמר להם לשון זה » (משנה ברורה ש״פ ס״ק ה) — אין בלשונו הכרע אם גם מלים אלה הן מדברי האליה רבה; פירוש זה אינו כתוב בסעיף א של הרב. טעם הביטול משתחשך, והאיסור הנשאר אף למי שביטל לו כשלא ביטל לכל אחד, כתובים כבר במגן אברהם, והשני בשם המגיד משנה; ושאין די בביטול לאחד אף כשזה כבר עירב עם השאר — כבר מבאר רש״י (לעיל).

השלכה מעשית: הדר עם ארבעה או חמשה ולא עירב צריך לבטל רשותו לכל אחד מהם; לדעת הרב די בנוסח „רשותי מבוטלת לכולכם”, שהרי לענין הלכה בדברי סופרים הולכים אחר המקל; ביטול לאחד מהם אינו מועיל, אף אם זה כבר עירב עם השאר, ואז אף למי שביטל לו אסור (380:1). לפירוש שכותב המשנה ברורה (שיעמוד נגד כולם ויאמר להם לשון זה) — היעזר ברב שלך.

נקודה ב — נעילת ביתו, כשביטל רשותו סתם (שו״ע הרב ש״פ:ב)

נעילת הבית

דברי המחבר: בסעיף א, במי שביטל רשותו סתם, הוא כותב « אבל אסורין להוציא מביתו לחצר וגם הוא וי״א שצריך לנעול ביתו כדי שלא יבא להוציא באיסור ולא יפתחנו אלא כשרוצה לצאת ולבא וינעלנו מיד אחר צאתו ובואו » (שו״ע או״ח ש״פ:א): דעת המצריכים נעילת בית מובאת בלי הכרעה.

דברי הרב: הוא כותב את האיסור להוציא מביתו לחצר, להם ולו, ומוסיף בכל פעם שכן אסור להכניס [מן החצר] — לו, בתוך סוגריים; ונותן לכך טעם, « לְפִי שֶׁאֵין אוֹמְרִים שֶׁהוּא כְּאוֹרֵחַ אֶצְלָם אֶלָּא לְעִנְיַן תַּשְׁמִישׁ מִבָּתֵּיהֶם לֶחָצֵר, אֲבָל תַּשְׁמִישׁ שֶׁמִּבֵּיתוֹ אוֹ לְבֵיתוֹ — אֵין לוֹמַר שֶׁהוּא אוֹרֵחַ » (שו״ע הרב ש״פ:ב); מביא את דעת הנעילה, « וְיֵשׁ אוֹמְרִים שֶׁצָּרִיךְ לִנְעוֹל בֵּיתוֹ, כְּדֵי שֶׁלֹּא יָבֹא לְהוֹצִיא בְּאִסּוּר » (שו״ע הרב ש״פ:ב), ומסיים « וְהַמֵּקֵל לֹא הִפְסִיד, כִּי יֵשׁ לוֹ עַל מִי לִסְמוֹךְ » (שו״ע הרב ש״פ:ב).

מה שכתוב כבר לפניו: על דברי המחבר „מביתו לחצר” כותב המגן אברהם « וגם הוא אסור שהרי בביתו אין לומר שהוא אורח » (מגן אברהם ש״פ ס״ק ג). שהאיסור הוא גם להכניס לביתו, כבר כתוב במשנה על ביתו: « בֵּיתוֹ אָסוּר מִלְּהַכְנִיס וּמִלְּהוֹצִיא לוֹ וְלָהֶם » (עירובין ס״ט ע״ב); וברמב״ם: « בִּטֵּל לָהֶן זֶה שֶׁלֹּא עֵרֵב רְשׁוּת חֲצֵרוֹ בִּלְבַד הֲרֵי אֵלּוּ מֻתָּרִין לְהוֹצִיא וּלְהַכְנִיס מִבָּתֵּיהֶן לֶחָצֵר וּמֵחָצֵר לְבָתֵּיהֶן אֲבָל לְבֵיתוֹ אָסוּר » (הלכות עירובין פ״ב ה״א). ולענין הנעילה מביא הבית יוסף פוסקים המחלקים בין מקרה זה לברייתא של נעילת הבית, ואת הרמב״ם שלא הזכיר נעילה: « והתוספות והרא״ש בדף קכ״ה והמרדכי כתבו לחלק בין ההיא לאנשי חצר ששכח אחד מהם ולא עירב וגם הרמב״ם בפ״ב לא הזכיר הא דנועל ביתו » (בית יוסף או״ח ש״פ); ומביא גם שהרא״ש כתב אחר כך שיש להחמיר, « מיהו אח״כ כתב הרא״ש ואפשר דבעי נעילת ביתו כל היום כשא״צ ליכנס ולצאת אבל בשעת כניסה ויציאה פותח ואח״כ נועל ולעיל מצינו דחיישינן דילמא אתא לאפוקי לאחר שביטל הילכך יש להחמיר כדפירש״י » (בית יוסף או״ח ש״פ), ומעמיד זאת במי שלא ביטל אלא רשות חצרו: « מיהו נראה דהיינו דוקא במבטל רשות חצירו ולא רשות ביתו » (בית יוסף או״ח ש״פ).

ההבדל: המחבר משאיר את דעת הנעילה בלי הכרעה; הרב כותב שהמקל לא הפסיד, כי יש לו על מי לסמוך. המשנה ברורה, שכתב אחריו, כותב « ובמקום הדחק יש לסמוך להקל שאין צריך נעילת בית » (משנה ברורה ש״פ ס״ק י״א): הוא כותב קולא זו במקום הדחק; בסעיף ב של הרב היא כתובה בלי הזכרה זו.

השלכה מעשית: אם ביטל רשותו סתם, לא ביטל אלא את החצר: הם והוא מותרים להוציא מבתיהם לחצר; להוציא מביתו לחצר, או להכניס מן החצר לביתו, אסור להם וגם לו; יש מצריכים שינעל ביתו ולא יפתחנו אלא כשרוצה לצאת ולבוא; הרב כותב שהמקל לא הפסיד (380:2). המשנה ברורה כותב קולא זו רק במקום הדחק (ס״ק י״א) — היעזר ברב שלך.

נקודה ג — ביטל גם רשות ביתו: שיחזיקו הם תחלה (שו״ע הרב ש״פ:ג)

מי מוציא תחלה

דברי המחבר: בסעיף ב הוא כותב « אם ביטל להם גם רשות ביתו מותרין בין הוא בין הם להוציא לחצר בין מביתו בין מבתיהם » (שו״ע או״ח ש״פ:ב), בלי תנאי של סדר בגוף הסעיף.

דברי הרב: הוא נותן טעם להיתר, « לְפִי שֶׁנַּעֲשֶׂה אוֹרֵחַ גָּמוּר אֶצְלָם, כֵּיוָן שֶׁלֹּא נִשְׁאַר לוֹ שׁוּם רְשׁוּת שֶׁלֹּא בִטְּלָה לָהֶם » (שו״ע הרב ש״פ:ג), וקובע זהירות: שלא יוציא מביתו לחצר קודם שיוציאו הם מבתיהם לחצר, « כְּדֵי שֶׁיַּחֲזִיקוּ הֵם בֶּחָצֵר תְּחִלָּה, וְלֹא יַכְנִיס מֵחָצֵר לְבֵיתוֹ קֹדֶם שֶׁיַּכְנִיסוּ הֵם מֵחָצֵר לְבֵיתוֹ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שפ״א » (שו״ע הרב ש״פ:ג).

מה שכתוב כבר לפניו: בבית יוסף המחבר עצמו נותן טעם להיתר ומתנה בו תנאי זה: « אבל במבטל רשות חצירו ורשות ביתו הוי אורח גמור לגבייהו ומותר להוציא אף מביתו כיון דלא נשאר לו שום רשות שלא ביטלו והוא שלא יוציא עד שיחזיקו הם וכמ״ש בסי' שאחר זה » (בית יוסף או״ח ש״פ). על דברי הסעיף „בין מביתו” כותב המגן אברהם « היינו לאחר שהחזיקו בחצר » (מגן אברהם ש״פ ס״ק ה), ומפנה לסימן שפ״א · 381.

ההבדל: סעיף המחבר מתיר בלי תנאי; שההיתר הוא לאחר שהחזיקו בחצר, כותב המחבר עצמו בבית יוסף, עם טעם האורח הגמור, וכותב המגן אברהם על הסעיף; הרב כותב זאת בגוף הסעיף, להוצאה לחצר ולהכנסה לביתו, בהפניה לסימן שפ״א · 381. המשנה ברורה, שכתב אחריו, כותב בס״ק י״ב את אותה זהירות, « וכן יזהר שלא יכניס מחצר לביתו קודם שיכניסו הם מהחצר לביתו » (משנה ברורה ש״פ ס״ק י״ב), ואת אותו טעם, « וטעם ההיתר דכיון שביטל גם רשות ביתו נעשה אורח גמור אצלם כיון שלא נשאר לו רשות שלא ביטל » (משנה ברורה ש״פ ס״ק י״ב).

השלכה מעשית: אם ביטל להם גם רשות ביתו, הם והוא מותרים להוציא מביתו לחצר ולהכניס, כמו בבתיהם; אך לא יוציא מביתו לחצר קודם שיוציאו הם מבתיהם לחצר, ולא יכניס מן החצר לביתו קודם שיכניסו הם מן החצר לביתו שלו (380:3; הרב מפנה לסימן שפ״א · 381, ושם הוא מונה הכנסה זו בין ההחזקות, 381:1). וזה כשהוא מבטל לרבים: בסימן שפ״א · 381 כותב הרב על החזקה זו « וַאֲפִלּוּ אִם הֶחֱזִיקוּ אֵינוֹ מוֹעִיל אֶלָּא בְּיָחִיד שֶׁבִּטֵּל לְרַבִּים, אֲבָל לֹא בְּיָחִיד לְיָחִיד, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן ש״פ » (שו״ע הרב שפ״א:ד); לשנים בלבד, ראה סעיף ו (פרק 4, ⑥).

נקודה ד — שכירות במקום ביטול (שו״ע הרב ש״פ:ד)

השכירות

דברי המחבר: בסעיף ג הוא כותב « אם אינו רוצה לבטל להם רשותו אלא להשכירו יש אומרים שמועיל כמו ביטול וי״א שאינו מועיל » (שו״ע או״ח ש״פ:ג): שתי דעות, בלי הכרעה.

דברי הרב: הוא מביא את שתי הדעות, כל אחת בטעמה, « יֵשׁ אוֹמְרִים שֶׁשְּׂכִירוּת מוֹעֶלֶת בְּיִשְׂרָאֵל כְּמוֹ שֶׁמּוֹעֶלֶת בְּנָכְרִי, לְפִי שֶׁהַשְּׂכִירוּת אֵינָהּ גָּרוּעַ מֵהַבִּטּוּל שֶׁהִיא מַתָּנָה בְּחִנָּם » (שו״ע הרב ש״פ:ד), ואחר כך « וְיֵשׁ אוֹמְרִים שֶׁלֹּא תִקְּנוּ שְׂכִירוּת אֶלָּא בְּנָכְרִי שֶׁאֵין בִּטּוּלוֹ מוֹעִיל, אֲבָל לֹא בְּיִשְׂרָאֵל » (שו״ע הרב ש״פ:ד); ופוסק: « וּלְעִנְיַן הֲלָכָה, בְּדִבְרֵי סוֹפְרִים הַלֵּךְ אַחַר הַמֵּקֵל » (שו״ע הרב ש״פ:ד).

מה שכתוב כבר לפניו: הטור כותב כבר, בלי להביא חולק: « ואם אינו רוצה לבטל להם רשותו אלא להשכירו מועיל כמו ביטול » (טור או״ח ש״פ); הבית יוסף מביא שזו דעת הרא״ש, שלא כדעת ר״מ: « ומ״ש ואם אינו רוצה לבטל להם רשותו אלא להשכירו מועיל כ״כ הרא״ש בר״פ הדר (דף קנג) גבי צדוקי ושלא כדברי ר״מ שסובר דלא מהני שכירות בישראל » (בית יוסף או״ח ש״פ), ושהמרדכי כותב כדברי ר״מ, ודברי הרמב״ם נוטים לדעתו. הט״ז כותב על הדעה השנייה « זהו דעת הר״מ באשר״י ולא זכר טעם ע״ז » (ט״ז או״ח ש״פ ס״ק ב), ומציע לה טעם, מן השכירות בפחות משוה פרוטה (ומפנה לסימן שפ״ב · 382). המגן אברהם מציע, בלשון של יראה, שאף הדעה השנייה מתירה שכירות בחול, ואוסרת רק בשבת: « לולי דמסתפינא הייתי אומר דהר״מ מודה דבחול שרי דלמה יגרע מביטול אלא דשבת אסו' לשכור דהוי כמקח וממכר » (מגן אברהם ש״פ ס״ק ו). הרב נותן לדעה הראשונה אותה השוואה לביטול.

ההבדל: המחבר מביא את שתי הדעות בלי הכרעה; הרב פוסק כדעת המקל: השכירות מועילה בישראל — כמו שהטור כותב בלי להביא חולק, והבית יוסף מביא בשם הרא״ש. המשנה ברורה, שכתב אחריו, מביא « ולענין הלכה הא״ר פוסק דאין שכירות מועלת בישראל כלל אבל שארי אחרונים מקילין בזה לענין ימי החול כהמ״א » (משנה ברורה ש״פ ס״ק י״ג). בסעיף ד הרב אינו מחלק בין שבת לחול; ובסימן שפ״ב · 382 הוא כותב על השכירות מן הנכרי: « מֻתָּר לִשְׂכּוֹר מִן הַנָּכְרִי אֲפִלּוּ בְּשַׁבָּת, וְאֵין בְּזֶה מִשּׁוּם מִקָּח וּמִמְכָּר » (שו״ע הרב שפ״ב:ח); ואף שהשכירות גם היא אסורה בשבת, מפני שהיא דומה למקח וממכר: « מִכָּל מָקוֹם כַּאן, כֵּיוָן שֶׁאֵינוֹ שׂוֹכֵר רְשׁוּתוֹ כְּדֵי לְהִשְׁתַּמֵּשׁ בָּהּ אֶלָּא לְהַתִּיר הַטִּלְטוּל בִּלְבָד, אִם כֵּן הַדָּמִים שֶׁנּוֹתֵן לוֹ אֵינוֹ אֶלָּא מַתָּנָה בְּעָלְמָא » (שו״ע הרב שפ״ב:ח); ואחר כך: « אֲבָל לִקְנוֹת רְשׁוּת בְּשַׁבָּת — אָסוּר, אַף שֶׁאֵינוֹ קוֹנֶה אֶלָּא לְהַתִּיר טִלְטוּל, וַאֲפִלּוּ בְּמַתְּנַת חִנָּם, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן ש״פ וּכְמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן תקכ״ח » (שו״ע הרב שפ״ב:ח). אין הוא כותב שם בפירוש שכירות מישראל בשבת. ובאותו סימן הוא כותב על ישראל הרוצה להשכיר ולא לבטל: « אֲבָל יִשְׂרָאֵל שֶׁאֵינוֹ רוֹצֶה לְבַטֵּל רְשׁוּתוֹ אֶלָּא לְהַשְׂכִּיר — אֵין פָּחוֹת מִשָּׁוֶה פְּרוּטָה נֶחְשָׁב לוֹ מָמוֹן כְּלָל, אַף עַל פִּי שֶׁהוּא מִתְרַצֶּה בּוֹ. וְאַף עַל פִּי שֶׁאִם הָיָה מְבַטֵּל בְּחִנָּם הָיָה מוֹעִיל, מִכָּל מָקוֹם כֵּיוָן שֶׁאֵינוֹ רוֹצֶה אֶלָּא לְהַשְׂכִּיר, וּשְׂכִירוּת אֵין כַּאן וּבִטּוּל אֵין כַּאן » (שו״ע הרב שפ״ב:ז) — ומאותה הבחנה יוצא הט״ז לתת טעם לדעה השנייה (ס״ק ב).

השלכה מעשית: לדעת הרב, מי שרוצה להשכיר רשותו במקום לבטלה רשאי, והשכירות מועילה, שהרי לענין הלכה בדברי סופרים הולכים אחר המקל (380:4); אבל בישראל, שכירות בפחות משוה פרוטה אינה לא שכירות ולא ביטול (382:7). האם מותר לשכור בשבת עצמה? המגן אברהם מציע, בלשון של יראה, שלא, משום שהשכירות היא כמקח וממכר (ס״ק ו); הרב מתיר לשכור מן הנכרי אף בשבת, שהדמים אינם אלא מתנה, ואינו כותב בפירוש את דין ישראל (382:8) — היעזר ברב שלך.

נקודה ה — רבים שביטלו ליחיד: להכניס מחצר לביתו (שו״ע הרב ש״פ:ה)

מה נשאר מותר לרבים

דברי המחבר: בסעיף ד הוא כותב « אם בני החצר שעירבו מבטלים רשותם לאחר שלא עירב הוא מותר להוציא מביתו לחצר ולא מבתיהם » (שו״ע או״ח ש״פ:ד); סוגריים באותיות קטנות, המפנים לבית יוסף, מוסיפים « אם לא שבטלו בפירוש גם רשות ביתם » (שו״ע או״ח ש״פ:ד); ואחר כך « והם אסורים אף מביתו לחצר ולא אמרינן שיהיו כאורחים שאין רבים נעשו אורחים אצל יחיד » (שו״ע או״ח ש״פ:ד). אין הוא מדבר לא במי שכבר החזיק בחצר ולא בהכנסה מן החצר לבית.

דברי הרב: הוא כותב בגוף הסעיף את מה שבסוגריים, « אֶלָּא אִם כֵּן בִּטְּלוּ לוֹ בְּפֵרוּשׁ גַּם רְשׁוּת בָּתֵּיהֶם » (שו״ע הרב ש״פ:ה); ומוסיף « וַאֲפִלּוּ אִם כְּבָר הֶחֱזִיק הוּא בֶּחָצֵר — אֵין זֶה מוֹעִיל שֶׁלֹּא יוּכְלוּ הָרַבִּים לַחֲזוֹר בָּהֶם » (שו״ע הרב ש״פ:ה); ואחר כך « אֲבָל מֻתָּרִים לְהַכְנִיס מֵחָצֵר לְבֵיתוֹ, לְפִי שֶׁאֵין נִרְאֶה חֲזָרָה אֶלָּא כְּשֶׁמַּכְנִיסִים לִרְשׁוּת שֶׁלָּהֶם שֶׁבִּטְּלוּהָ, דְּהַיְנוּ לֶחָצֵר » (שו״ע הרב ש״פ:ה); וגם « אֲבָל אֲסוּרִים לְהַכְנִיס מֵחָצֵר לְבָתֵּיהֶם אֲפִלּוּ אִם בִּטְּלוּ לוֹ גַּם רְשׁוּת בָּתֵּיהֶם » (שו״ע הרב ש״פ:ה).

מה שכתוב כבר לפניו: המגן אברהם כותב « ואפי' חזקה לא מהני » (מגן אברהם ש״פ ס״ק ז), ואומר שכן משמע בה״ר יהונתן, ומסיים, בסוף אותו ס״ק, « וא״כ משמע דמותרים להכניס מחצר לביתו » (מגן אברהם ש״פ ס״ק ז).

ההבדל: בסעיף ד המחבר אינו מדבר לא בהחזקה ולא בהכנסה לבתים; הרב כותב את שתיהן, כמו שכתב המגן אברהם — את ההחזקה, באמרו שכן משמע בה״ר יהונתן; את ההכנסה לביתו, מדיוק (וא״כ משמע) —, ומוסיף את האיסור להכניס מן החצר לבתיהם, אף אם ביטלו לו גם את בתיהם. המשנה ברורה, שכתב אחריו, מביא את המגן אברהם, « וכתב המ״א דלא אמרינן זה אלא כשהם מוציאים מבית לחצר אבל כשהם מכניסים מחצר לבית אין זה נקרא השתמשות בחצר ולא הוי כרוצים לזכות בחצר » (משנה ברורה ש״פ ס״ק ט״ז), ואחר כך « אבל בא״ר כתב בשם הרבה פוסקים שגם להכניס מחצר לבית אסור וכ״כ באבן עוזר » (משנה ברורה ש״פ ס״ק ט״ז).

השלכה מעשית: אם בני החצר שעירבו מבטלים רשותם ליחיד שלא עירב, הוא מוציא מביתו לחצר, ולא מבתיהם, אלא אם כן ביטלו לו בפירוש גם את בתיהם; הם אינם מוציאים אף מביתו לחצר, ואף אם כבר החזיק הוא בחצר; מותרים הם להכניס מן החצר לביתו, אך לא לבתיהם, אף אם ביטלו לו גם אותם; ואינם מוציאים מבתיהם לחצר (380:5). להכנסה מן החצר לביתו מביא המשנה ברורה שהאליה רבה אוסר בשם הרבה פוסקים (ס״ק ט״ז) — היעזר ברב שלך.

נקודה ו — המקבל ביטול שחוזר ומבטל לחבירו (שו״ע הרב ש״פ:ז)

ביטול שחוזר ומבטל

דברי המחבר: בסוף סעיף ד הוא כותב « ואפילו אמר לא' אני מבטל לך על מנת שתחזור ותבטל לחבירך אינו מועיל » (שו״ע או״ח ש״פ:ד): הסעיף כותב בפירוש רק את המקרה של התנאי, ופותח בו ב„ואפילו”.

דברי הרב: הוא כותב תחלה את המקרה בלי תנאי, « וַאֲפִלּוּ אִם חָזַר הָאֶחָד וּבִטֵּל לַחֲבֵרוֹ — אֵינוֹ מוֹעִיל כְּלוּם » (שו״ע הרב ש״פ:ז), עם טעמו; ואחר כך את המקרה של התנאי, שגם הוא אינו מועיל, ונותן לכך טעם: « שֶׁכֵּיוָן שֶׁלֹּא זָכָה בְּחֵלֶק שֶׁבִּטְּלוּ לוֹ בְּעִנְיָן שֶׁיּוּכַל לְטַלְטֵל בֶּחָצֵר, אֵינוֹ יָכוֹל לְהַקְנוֹת לַחֲבֵרוֹ » (שו״ע הרב ש״פ:ז).

מה שכתוב כבר לפניו: המקרה בלי תנאי, וטעמו, כתובים כבר בגמרא: « לָא צְרִיכָא, דַּהֲדַר חַד מִינַּיְיהוּ וּבַטֵּיל לֵיהּ לְחַבְרֵיהּ. מַהוּ דְּתֵימָא לִישְׁתְּרֵי, קָא מַשְׁמַע לַן דְּכֵיוָן דִּבְעִידָּנָא דְּבַטֵּיל לָא הֲוָה לֵיהּ שַׁרְיוּתָא בְּהַאי חָצֵר » (עירובין ס״ט ע״ב); וברמב״ם: « וַאֲפִלּוּ חָזַר הָאֶחָד שֶׁלֹּא עֵרֵב וּבִטֵּל רְשׁוּתוֹ לַשֵּׁנִי שֶׁלֹּא עֵרֵב הֲרֵי זֶה אוֹסֵר שֶׁבְּשָׁעָה שֶׁבִּטְּלוּ לוֹ הַמְעָרְבִין אָסוּר הָיָה » (הלכות עירובין פ״ב ה״ד). ה״ר יהונתן כותב אותו בשם ברייתא, עם טעמו, כמו שמביא הבית יוסף: « כתב ה״ר יהונתן בפירוש המשנה תני בברייתא שאע״פ שחזר השני וביטל רשותו לחבירו אסור להוציא לחצר דכיון שתיכף שביטלו אותן שעירבו לאותן שלא עירבו לא הותר לטלטל בחצר שוב לא מהני ליה ביטול חבירו » (בית יוסף או״ח ש״פ); טעם המקרה של התנאי כתוב כבר בטור: « כיון שלא זכה בחלק שביטלו לו בענין שיוכל לטלטל בחצר אינו יכול לחזור ולהקנותה לחבירו » (טור או״ח ש״פ). הט״ז כותב על דברי המחבר את שני המקרים, כל אחד בטעמו: « כיון שתיכף שביטלו אותן שעירבו לאותן שלא עירבו לא הותר לטלטל בחצר כיון שהיו שנים שוב לא מהני ביטול א' מהם לחבירו ואפי' ע״מ שאמר לא מועיל כיון שלא זכה בחלק שביטלו לו בענין שיוכל לטלטל בחצר אינו יכול להקנו' לחבירו » (ט״ז או״ח ש״פ ס״ק ד); המגן אברהם כותב « כיון דלא אקני ליה שיהא הוא מותר אינו מועיל » (מגן אברהם ש״פ ס״ק ט).

ההבדל: המחבר כותב בפירוש רק את המקרה של התנאי, בלשון „ואפילו”; המשנה ברורה, שכתב אחרי הרב, לומד מ„ואפילו” זה את המקרה בלי תנאי: « ר״ל דלא מיבעי אם לא פירש שיחזור ויבטל לחבירו אלא שביטל סתם לראשון והראשון מעצמו חזר וביטל חלק של עצמו וגם חלק שקיבל דלא מהני » (משנה ברורה ש״פ ס״ק כ״ג). הרב כותב בפירוש, קודם המקרה של התנאי, את המקרה שאחד חוזר ומבטל לחבירו בלי תנאי, כל אחד בטעמו — המקרה הראשון וטעמו כתובים כבר בגמרא וברמב״ם, ואחר כך בדברי ה״ר יהונתן ובט״ז; וטעם השני בטור, שהט״ז מביא את לשונו.

השלכה מעשית: רבים שלא עירבו יכולים לבטל רשותם ליחיד שלא עירב, או לרבים שעירבו, כל אחד לכל אחד; לשנים שלא עירבו, או יותר, אין מבטלים, ואם חזר אחד מהם וביטל לחבירו אינו מועיל, אף אם הראשונים התנו כך (380:7).

4 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן ש״פ · 380, ברגישות החב״דית, לפי סדר סעיפיו. מספרי הסעיפים הם לפי מספור הרב.

לחסיד חב״ד — סימן ש״פ · 380

  • ① ביטול רשות: מי ששכח ולא עירב עם בני החצר אוסר עליהם; הוא מבטל להם את הרשות שיש לו בחצר, ואומר „רשותי מבוטלת לכם” או „רשותי קנויה לכם”, בלי קנין סודר; ויכול לבטל משתחשך, שהרי אינו אלא מסלק עצמו מרשותו; אם הוא דר עם ארבעה או חמשה, הוא צריך לבטל לכל אחד מהם, ולדעת הרב די בנוסח „לכולכם”; ביטול לאחד מהם אינו מועיל, אף אם זה עירב עם השאר. לפירוש שכותב המשנה ברורה ראה פרק 2, נקודה א, והיעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן 380:1.
  • ② ביטול סתם: לא ביטל אלא את החצר; הם והוא מותרים להוציא מבתיהם לחצר; להוציא מביתו לחצר, או להכניס מן החצר לביתו, אסור להם וגם לו; יש מצריכים שינעל ביתו ולא יפתחנו אלא כשרוצה לצאת ולבוא; הרב כותב שהמקל לא הפסיד, כי יש לו על מי לסמוך (להסתייגות המשנה ברורה ראה פרק 2, נקודה ב). הפניה: שולחן ערוך אדמו״ר הזקן 380:2.
  • ③ ביטול גם רשות ביתו: אם ביטל להם גם רשות ביתו, הם והוא מותרים להוציא מביתו לחצר ולהכניס, כמו בבתיהם; אך לא יוציא מביתו לחצר קודם שיוציאו הם מבתיהם לחצר, ולא יכניס מן החצר לביתו קודם שיכניסו הם מן החצר לביתו שלו; הרב מפנה לסימן שפ״א · 381, ושם הוא מונה הכנסה זו בין החזקות (381:1); וזה כשהוא מבטל לרבים, ולא ליחיד (381:4; לשנים בלבד, ראה ⑥). הפניה: שולחן ערוך אדמו״ר הזקן 380:3; 381:4.
  • ④ שכירות: אם רוצה להשכיר רשותו במקום לבטלה, מובאות שתי דעות; הרב כותב שלענין הלכה בדברי סופרים הולכים אחר המקל: השכירות מועילה; ובסימן שפ״ב · 382 הוא כותב שבישראל, שכירות בפחות משוה פרוטה אינה לא שכירות ולא ביטול (382:7). לשכירות בשבת עצמה — הוא מתיר לשכור מן הנכרי אף בשבת (382:8) — ראה פרק 2, נקודה ד, והיעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן 380:4; 382:7, 8.
  • ⑤ רבים שביטלו ליחיד: אם בני החצר שעירבו מבטלים רשותם ליחיד שלא עירב, הוא מוציא מביתו לחצר, ולא מבתיהם, אלא אם כן ביטלו לו בפירוש גם את בתיהם; הם אינם מוציאים אף מביתו לחצר, ואף אם כבר החזיק הוא בחצר; מותרים הם להכניס מן החצר לביתו, אך לא לבתיהם, אף אם ביטלו לו גם אותם; ואינם מוציאים מבתיהם לחצר. להכנסה לביתו ראה פרק 2, נקודה ה, והיעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן 380:5.
  • ⑥ שנים לבד, או רבים שביטלו לרבים: אם רק שנים דרים בחצר ולא עירבו, וביטל אחד מהם לחבירו, דינו כרבים לגבי יחיד, שהיחיד אינו נעשה אורח אלא לגבי רבים, ואין רבים פחותים משנים; וכן רבים שלא עירבו שביטלו רשותם לרבים שעירבו אינם נעשים אורחים אצלם. הרב כותב זאת במפורש: אף היחיד אינו נעשה אורח לגבי יחיד. הביאור הלכה, שכתב אחריו, מביא דעות חולקות, שסעיף ו של הרב אינו הולך בשיטתן (ראה רמה 1, פרק 6); היעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן 380:6.
  • ⑦ למי מבטלים: רבים שלא עירבו יכולים לבטל רשותם ליחיד שלא עירב, או לרבים שעירבו, ובלבד שכל אחד יבטל לכל אחד; לשנים שלא עירבו, או יותר, אין מבטלים, ואם חזר אחד מהם וביטל לחבירו אינו מועיל, אף אם הראשונים התנו כך. הפניה: שולחן ערוך אדמו״ר הזקן 380:7.

⚠ פרק זה מציג את ההנהגה ההלכתית של חב״ד. לכל שאלה ממשית: היעזר ברב שלך.
