דעת הרב

דעת הרב — שולחן ערוך אדמו״ר הזקן על סימן שצ״ו

ארבע אמות שיש לכל אדם בשבת: שורש התחום ואלפים אמה; צורת ארבע האמות; מידתן, מצומצמות או מרווחות; המקום המוקף מחיצות, התל, הקמה הקצורה והעיר הנחשבים כארבע אמות. ארבעת הסעיפים של שולחן ערוך אדמו״ר הזקן, מושווים לשני הסעיפים של המחבר.

מדוע רמה 4 מוקדשת לאדמו״ר הזקן? שולחן ערוך אדמו״ר הזקן אינו פירוש על המחבר: זהו שולחן ערוך בפני עצמו, שנכתב על ידי אדמו״ר הזקן, בחלוקה ובמספור סעיפים משלו. כאן מונה שולחן ערוך הרב ארבעה סעיפים בראש הסימן, ובשולחן ערוך של המחבר שני סעיפים. ספריא מגישה את לשון הרב בארבעה קטעים, והדף מביא וממספר אותם כמות שהם (פרק 1). שתי החלוקות אינן חופפות: סעיף א של הרב מביא את תחילת סעיף א של המחבר, ארבע האמות שיש לכל אדם, ואת דין המקום המוקף, שהמחבר כותב בסעיף ב; סעיפים ב ו־ג של הרב עומדים מול שאר סעיף א ומול הגהת הרמ״א הפותחת שם במילה הגה; סעיף ד של הרב עומד מול סוף סעיף ב (התל והקמה הקצורה), ומוסיף את העיר. בהפניות שבסוגריים שני המספורים באותיות, וההפניה מציינת תמיד את הספר (שו״ע או״ח למחבר, רמ״א או״ח להגהת הרמ״א, שו״ע הרב לרב).

עמוד זה מקבץ, לסימן שצ״ו, את הטקסט המלא של הרב, את מה ששונה בלשונו מדברי המחבר והרמ״א, ואת ההנהגה המעשית העולה מדבריו.

1 · טקסט אדמו״ר הזקן

שלחן ערוך הרב — סימן שצ״ו

הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן שצ״ו — דִּין אַרְבַּע אַמּוֹת שֶׁיֵּשׁ לְכָל אָדָם בְּשַׁבָּת — וּבוֹ ד׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. בכותרת: ד׳ סעיפים; ספריא מגישה את הלשון בארבעה קטעים (בשולחן ערוך של המחבר שני סעיפים, במספור משלו).

סעיף א דִּין אַרְבַּע אַמּוֹת שֶׁיֵּשׁ לְכָל אָדָם בְּשַׁבָּת וּבוֹ ד' סְעִיפִים: "שְׁבוּ אִישׁ תַּחְתָּיו אַל יֵצֵא אִישׁ מִמְּקֹמוֹ בַּיּוֹם הַשְּׁבִיעִי"…

דִּין אַרְבַּע אַמּוֹת שֶׁיֵּשׁ לְכָל אָדָם בְּשַׁבָּת וּבוֹ ד' סְעִיפִים:

"שְׁבוּ אִישׁ תַּחְתָּיו אַל יֵצֵא אִישׁ מִמְּקֹמוֹ בַּיּוֹם הַשְּׁבִיעִי". יֵשׁ אוֹמְרִים שֶׁזּוֹ אַזְהָרָה לְיוֹצֵא חוּץ לַתְּחוּם, וְשִׁעוּר תְּחוּם שֶׁל תּוֹרָה הוּא י״ב מִיל, כְּשִׁעוּר מַחֲנֵה יִשְׂרָאֵל שֶׁהָיְתָה י״ב מִיל עַל י״ב מִיל. וַחֲכָמִים אָסְרוּ לָצֵאת יוֹתֵר מֵאַלְפַּיִם אַמּוֹת, שֶׁהֵן מִיל א'.

וְיֵשׁ אוֹמְרִים שֶׁאֵין לִתְחוּמִין עִקָּר מִן הַתּוֹרָה כְּלָל, אֶלָּא אֲפִלּוּ יוֹתֵר מִי״ב מִיל אֵינוֹ אָסוּר מִן הַתּוֹרָה כְּלָל. וְזֶה שֶׁנֶּאֱמַר "אַל יֵצֵא אִישׁ מִמְּקֹמוֹ" — זוֹ הִיא אַזְהָרָה עַל הוֹצָאָה, שֶׁהִזְהִיר מֹשֶׁה שֶׁלֹּא לָצֵאת עִם הַכְּלִי מֵרְשׁוּת הַיָּחִיד לִרְשׁוּת הָרַבִּים לִלְקוֹט הַמָּן. וּ"שְׁבוּ אִישׁ תַּחְתָּיו" — קִבְּלוּ חֲכָמִים שֶׁזּוֹ אַזְהָרָה שֶׁלֹּא לְטַלְטֵל בִּרְשׁוּת הָרַבִּים אֶלָּא כְּשִׁעוּר תַּחְתָּיו שֶׁל אָדָם, שֶׁהוּא ד' אַמּוֹת, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שמ״ט.

אֶלָּא שֶׁחֲכָמִים סָמְכוּ אִסּוּר הַתְּחוּמִין שֶׁל דִּבְרֵיהֶם לְמִקְרָא זֶה "אַל יֵצֵא אִישׁ מִמְּקֹמוֹ", שֶׁלֹּא יֵצֵא יוֹתֵר מֵאַלְפַּיִם מֵהַמָּקוֹם שֶׁשָּׁבַת בּוֹ, כִּי עַד אַלְפַּיִם נִקְרָא מְקוֹמוֹ, כְּמוֹ שֶׁמָּצִינוּ בְּעָרֵי מִקְלָט, שֶׁנֶּאֱמַר "וְשַׂמְתִּי לְךָ מָקוֹם וְגוֹ'", וְהֵן קוֹלְטוֹת אוֹתוֹ גַּם כְּשֶׁהוּא חוּצָה לָהֶן, שֶׁנֶּאֱמַר "וּמָצָא אוֹתוֹ מִחוּץ לִגְבוּל עִיר מִקְלָטוֹ", וְנֶאֱמַר לְהַלָּן "וּמַדֹּתֶם מִחוּץ לָעִיר אֶת פְּאַת קֵדְמָה אַלְפַּיִם בָּאַמָּה וְגוֹ'". מַה "חוּץ" הָאָמוּר לְהַלָּן — אַלְפַּיִם, אַף "חוּץ" הָאָמוּר כַּאן — אַלְפַּיִם.

וּ"שְׁבוּ אִישׁ תַּחְתָּיו" — כְּשִׁעוּר תַּחְתָּיו. מִכַּאן סָמְכוּ חֲכָמִים שֶׁכָּל אָדָם יֵשׁ לוֹ ד' אַמּוֹת בְּכָל מָקוֹם שֶׁהוּא, אֲפִלּוּ יָצָא חוּץ לַתְּחוּם — יָכוֹל לְהַלֵּךְ שָׁם ד' אַמּוֹת, וְכֵן יָכוֹל לְטַלְטֵל בָּהֶן. וְכֵן כְּשֶׁמּוֹדְדִים אַלְפַּיִם מִמְּקוֹם שְׁבִיתָתוֹ — מוֹדְדִים חוּץ מִד' אַמּוֹת אֵלּוּ, שֶׁהֵן כְּשִׁעוּר תַּחְתָּיו (וְיָכוֹל לְטַלְטֵל בָּהֶן. לְפִיכָךְ) אִם שָׁבַת, דְּהַיְנוּ שֶׁקִּדֵּשׁ עָלָיו הַיּוֹם, בְּמָקוֹם הַמֻּקָּף מְחִצּוֹת בְּעִנְיָן שֶׁמֻּתָּר לְטַלְטֵל בְּכֻלּוֹ — הֲרֵי כֻּלּוֹ נֶחְשָׁב לוֹ כְּד' אַמּוֹת, וְנוֹתְנִים לוֹ אַלְפַּיִם סָבִיב לְהֶקֵּף זֶה, עַל דֶּרֶךְ שֶׁיִּתְבָּאֵר בְּסִמָּן שצ״ט, אֲפִלּוּ יֵשׁ בּוֹ כַּמָּה מִילִין.

וְהוּא שֶׁיְּהֵא מֻקָּף לְדִירָתוֹ מִתְּחִלָּתוֹ, אַף עַל פִּי שֶׁאֵין בּוֹ עַתָּה דִּיּוּרִין וְאֵין מִשְׁתַּמְּשִׁין בּוֹ כְּלָל, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן ת״ח, אוֹ אֲפִלּוּ אֵינוֹ מֻקָּף לְדִירָה אֶלָּא שֶׁאֵין בּוֹ יוֹתֵר מִבֵּית סָאתַיִם. אֲבָל אִם יֵשׁ בּוֹ יוֹתֵר מִבֵּית סָאתַיִם — כְּשֵׁם שֶׁאֵין מְחִצּוֹתֶיהָ מוֹעִילוֹת שֶׁיְּהֵא מֻתָּר לְטַלְטֵל בְּכֻלּוֹ, כָּךְ אֵינָן מוֹעִילוֹת שֶׁיְּהֵא כֻלּוֹ נֶחְשָׁב כְּד' אַמּוֹת לְעִנְיַן תְּחוּמִין:

סעיף ב ד' אַמּוֹת אֵלּוּ שֶׁלְּעִנְיַן תְּחוּמִין…

ד' אַמּוֹת אֵלּוּ שֶׁלְּעִנְיַן תְּחוּמִין — בֵּין ד' אַמּוֹת שֶׁבִּמְקוֹם שְׁבִיתָתוֹ, בֵּין ד' אַמּוֹת שֶׁיֵּשׁ לוֹ כְּשֶׁיָּצָא חוּץ לַתְּחוּם — יֵשׁ אוֹמְרִים שֶׁהֵן כְּד' אַמּוֹת שֶׁבִּרְשׁוּת הָרַבִּים שֶׁלְּעִנְיַן טִלְטוּל, שֶׁהוּא בּוֹרֵר לוֹ ד' אַמּוֹת לְאֵיזוֹ רוּחַ שֶׁיִּרְצֶה, וּכְשֶׁהָלַךְ לְרוּחַ אַחַת ד' אַמּוֹת — שׁוּב אֵינוֹ יָכוֹל לַחֲזוֹר וְלֵילֵךְ לְרוּחַ הַשֵּׁנִית (שֶׁכְּנֶגְדָּהּ) אֲפִלּוּ כָּל שֶׁהוּא אִם הוּא חוּץ לַתְּחוּם, כְּמוֹ שֶׁאֵינוֹ יָכוֹל לְטַלְטֵל חֵפֶץ אֶחָד יוֹתֵר מִד' אַמּוֹת, שֶׁאֵין נוֹתְנִים אֲלַכְסוֹן לְעִנְיַן טִלְטוּל אֶלָּא מִפְּנֵי שֶׁאֵין שָׁם רִבּוּעַ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שמ״ט. וְכֵן בְּמָקוֹם שֶׁשָּׁבַת בּוֹ, אִם הָלַךְ לְרוּחַ אַחַת ד' אַמּוֹת וְאַלְפַּיִם — שׁוּב אֵינוֹ יָכוֹל לֵילֵךְ לְרוּחַ הַשֵּׁנִית (שֶׁכְּנֶגְדָּהּ) אֶלָּא אַלְפַּיִם. וְאִם לֹא נָטַל לוֹ כָּל ד' אַמּוֹת לְרוּחַ אַחַת, יָכוֹל לְהַשְׁלִימָן בְּרוּחַ הַב' (שֶׁכְּנֶגְדָּהּ).

וְיֵשׁ אוֹמְרִים שֶׁד' אַמּוֹת אֵלּוּ שֶׁלְּעִנְיַן תְּחוּמִין אֵינוֹ כְּד' אַמּוֹת שֶׁלְּעִנְיַן טִלְטוּל, אֶלָּא הֵן ד' אַמּוֹת לְכָל רוּחַ, שֶׁהֵן ח' עַל ח' מְרֻבָּעוֹת, בֵּין בְּמָקוֹם שֶׁשָּׁבַת בּוֹ בֵּין כְּשֶׁיָּצָא חוּץ לַתְּחוּם (וּלְפִי סְבָרָא זוֹ כָּל שֶׁכֵּן שֶׁאֵין מוֹדְדִים לוֹ ד' אַמּוֹת עִם אֲלַכְסוֹנָם לְכָל רוּחַ, שֶׁכֵּיוָן שֶׁנּוֹתְנִים לוֹ ח' עַל ח' מְרֻבָּעוֹת — הֲרֵי יֵשׁ בָּהֶן אֲלַכְסוֹן מַמָּשׁ שֶׁל ד' אַמּוֹת לְכָל רוּחַ, דְּהַיְנוּ מֵאֶמְצָעִיתוֹ לְאַרְבַּע הַזָּוִיּוֹת. מַה שֶּׁאֵין כֵּן בְּד' אַמּוֹת שֶׁלְּעִנְיַן טִלְטוּל, שֶׁמּוֹדְדִים לוֹ ד' אַמּוֹת בְּיֹשֶׁר אֶל עֵבֶר פָּנָיו לְאֵיזוֹ רוּחַ שֶׁיִּרְצֶה, וְכֵיוָן שֶׁלֹּא שַׁיָּךְ בָּהֶן אֲלַכְסוֹן מַמָּשׁ — צָרִיךְ לִתֵּן לוֹ הֵן וַאֲלַכְסוֹנָם, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שמ״ט). וְיֵשׁ לִסְמוֹךְ עַל דִּבְרֵיהֶם לְהָקֵל בְּדִבְרֵי סוֹפְרִים:

סעיף ג ד' אַמּוֹת אֵלּוּ שֶׁלְּכָל רוּחַ וְרוּחַ…

ד' אַמּוֹת אֵלּוּ שֶׁלְּכָל רוּחַ וְרוּחַ, שֶׁהֵן שִׁעוּר תַּחְתָּיו שֶׁל אָדָם — מוֹדְדִים אוֹתוֹ לְכָל אָדָם בְּאַמָּה שֶׁלּוֹ. וְאִם הוּא נַנָּס בְּאֵבָרִים, שֶׁגּוּפוֹ גָדוֹל וְאַמָּתוֹ קְטַנָּה — נוֹתְנִים לוֹ ד' אַמּוֹת בֵּינוֹנִית שֶׁל כָּל אָדָם, שֶׁכָּל אַחַת מֵהֶן ו' טְפָחִים, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שמ״ט.

וּכְבָר נִתְבָּאֵר שָׁם שֶׁד' אַמּוֹת אֵלּוּ מוֹדְדִים אוֹתָן מְצֻמְצָמוֹת. וְיֵשׁ אוֹמְרִים שֶׁמּוֹדְדִים אוֹתָן מְרֻוָּחוֹת, וְיֵשׁ לִסְמוֹךְ עַל דִּבְרֵיהֶם בְּד' אַמּוֹת אֵלּוּ שֶׁלְּעִנְיַן תְּחוּמִין, שֶׁעִקָּרָן אֵינָן אֶלָּא מִדִּבְרֵי סוֹפְרִים:

סעיף ד שָׁבַת בְּתֵל גָּבוֹהַּ י' טְפָחִים…

שָׁבַת בְּתֵל גָּבוֹהַּ י' טְפָחִים, אִם אֵין בּוֹ אֶלָּא בֵּית סָאתַיִם — הֲרֵי כֻּלּוֹ נֶחְשָׁב לוֹ כְּד' אַמּוֹת, מִפְּנֵי שֶׁהוּא כְּמֻקָּף מְחִצּוֹת מִלְמַעְלָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שמ״ה, אֶלָּא שֶׁאֵינָן מֻקָּפוֹת לְדִירָה.

וְכֵן אִם שָׁבַת בְּקָמָה קְצוּרָה וְשִׁבֳּלִים גְּבוֹהִים י' טְפָחִים מַקִּיפִים אוֹתָהּ. וְהוּא שֶׁקָּשַׁר הַשִּׁבֳּלִים יַחַד בְּעִנְיָן שֶׁאֵין רוּחַ הַמְּצוּיָה מְנִידָה אוֹתָן, שֶׁכָּל מְחִצָּה שֶׁהָרוּחַ מְנִידָהּ — אֵינָהּ מְחִצָּה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שס״ב.

אֲבָל הַשּׁוֹבֵת בְּעִיר — כָּל הָעִיר נֶחְשֶׁבֶת כְּד' אַמּוֹת, אַף עַל פִּי שֶׁאֵינָהּ מֻקֶּפֶת מְחִצּוֹת כְּהִלְכָתָן, כְּגוֹן שֶׁיֵּשׁ בָּהּ פְּרָצוֹת הַרְבֵּה בְּמִלּוּאָן וּבְיוֹתֵר מֵעֶשֶׂר, דְּהוֹאִיל וּבָתֵּי הָעִיר מֻבְלָעִים בְּתוֹךְ שִׁבְעִים אַמָּה וְשִׁירַיִם שֶׁל זֶה — הֲרֵי כֻּלָּם כְּדִירָה אַחַת, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שצ״ח:

מקור: ספריא — שולחן ערוך הרב, אורח חיים שצ״ו. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה.

2 · לשון הרב ולשון המחבר

לשון הרב מול סעיף המחבר והרמ״א

במה שונה לשון הרב מסעיף המחבר והרמ״א

כל נקודה מעמידה זה לצד זה קטע משולחן ערוך המחבר (במספור שלו, שני סעיפים, המסומנים באותיותיהם), או מהגהת הרמ״א הפותחת במילה הגה בסעיף א, ואת הסעיף המקביל בשולחן ערוך אדמו״ר הזקן (במספור שלו, ארבעה סעיפים), לפי סדר סעיפי הרב, ואינה מציינת אלא מה שעולה מהשוואת שני הלשונות. נקודה שצוינה אינה מיוחדת לרב בשל כך: הדף מציין לפעמים את מקורה; אין הוא מציין את כל המקורות ולא את כל המחלוקות.

נקודה א — שורש התחום ואלפים אמה (שו״ע הרב שצ״ו:א)

שורש התחום

דברי המחבר: בתחילת סעיף א, « שבו איש תחתיו מכאן שכל אדם יש לו ד' אמות בכל מקום אפי' יצא חוץ לתחום » (שו״ע או״ח שצ״ו:א). בסימן שצ״ו אינו כותב מהו שורש התחום.

דברי הרב: « יֵשׁ אוֹמְרִים שֶׁזּוֹ אַזְהָרָה לְיוֹצֵא חוּץ לַתְּחוּם, וְשִׁעוּר תְּחוּם שֶׁל תּוֹרָה הוּא י״ב מִיל, כְּשִׁעוּר מַחֲנֵה יִשְׂרָאֵל שֶׁהָיְתָה י״ב מִיל עַל י״ב מִיל. וַחֲכָמִים אָסְרוּ לָצֵאת יוֹתֵר מֵאַלְפַּיִם אַמּוֹת, שֶׁהֵן מִיל א » (שו״ע הרב שצ״ו:א); ואחר כך « וְיֵשׁ אוֹמְרִים שֶׁאֵין לִתְחוּמִין עִקָּר מִן הַתּוֹרָה כְּלָל, אֶלָּא אֲפִלּוּ יוֹתֵר מִי״ב מִיל אֵינוֹ אָסוּר מִן הַתּוֹרָה כְּלָל » (שו״ע הרב שצ״ו:א).

ההבדל: סעיף א של המחבר לומד מן הפסוק את ארבע האמות שיש לכל אדם, אפילו יצא חוץ לתחום. סעיף א של הרב מביא תחילה שתי דעות בשורש התחום: לדעה אחת, הפסוק הוא אזהרה ליוצא חוץ לתחום, ששיעורו מן התורה י״ב מיל, וחכמים אסרו לצאת יותר מאלפיים אמה; ולדעה השנייה, אין לתחומין עיקר מן התורה, והפסוק הוא אזהרה על הוצאה. אחר כך כותב שחכמים סמכו את איסור התחומין, שהוא מדבריהם, על פסוק זה, ולומד את האלפיים מערי מקלט; ואחר כך, שבארבע אמות אלו יכול להלך ולטלטל, ושהאלפיים נמדדים חוץ מארבע אמות אלו. סעיף א של הרב אינו כותב איזו משתי הדעות עיקר.

השלכה מעשית: לפי לשון הרב, לכל אדם יש ארבע אמות בכל מקום, אפילו יצא חוץ לתחום: יכול להלך בהן ולטלטל בהן; וכשמודדים אלפיים ממקום שביתתו, מודדים חוץ מארבע אמות אלו (שו״ע הרב שצ״ו:א). בין שתי הדעות בשורש התחום הדף אינו מכריע: היעזר ברב שלך. ראה פרק 4, נקודה ①.

נקודה ב — ארבע אמות לרוח אחת או לכל רוח (שו״ע הרב שצ״ו:ב)

צורת ארבע האמות

דברי הרמ״א: בהגהתו הפותחת במילה הגה בסעיף א, « ואלו ד' אמות מודדין לו מרווחת והוא באמצען (טור) ויש אומרים דמותר להלך ארבע אמות לכל צד אבל לטלטל אין לו רק ארבע אמות עם אלכסונן כמו שנתבאר לעיל סי' שמ״ט » (רמ״א או״ח שצ״ו:א). הוא מביא את הדעה הפותחת במילים ויש אומרים, ואינו כותב אם יש לסמוך עליה.

דברי הרב: סעיף ב שלו מביא שתי דעות; השנייה: « וְיֵשׁ אוֹמְרִים שֶׁד' אַמּוֹת אֵלּוּ שֶׁלְּעִנְיַן תְּחוּמִין אֵינוֹ כְּד' אַמּוֹת שֶׁלְּעִנְיַן טִלְטוּל, אֶלָּא הֵן ד' אַמּוֹת לְכָל רוּחַ, שֶׁהֵן ח' עַל ח' מְרֻבָּעוֹת, בֵּין בְּמָקוֹם שֶׁשָּׁבַת בּוֹ בֵּין כְּשֶׁיָּצָא חוּץ לַתְּחוּם » (שו״ע הרב שצ״ו:ב). ומסיים: « וְיֵשׁ לִסְמוֹךְ עַל דִּבְרֵיהֶם לְהָקֵל בְּדִבְרֵי סוֹפְרִים » (שו״ע הרב שצ״ו:ב).

ההבדל: סעיף ב של הרב מפרט גם את הדעה הראשונה: ארבע אמות אלו הן כארבע אמות שברשות הרבים לעניין טלטול; בורר לו ארבע אמות לאיזו רוח שירצה, וכשהלך לרוח אחת ארבע אמות — שוב אינו יכול לילך לרוח שכנגדה, אם הוא חוץ לתחום; במקום ששבת בו, אם הלך לרוח אחת ארבע אמות ואלפיים — אינו יכול לילך לרוח שכנגדה אלא אלפיים; ואם לא נטל כל ארבע האמות לרוח אחת — יכול להשלימן ברוח שכנגדה. ועל הדעה השנייה — ארבע אמות אלו שלעניין תחומין אינן כארבע אמות שלעניין טלטול: ארבע אמות לכל רוח, שמונה על שמונה — כותב הרב שיש לסמוך על דבריהם להקל בדברי סופרים; הגהת הרמ״א אינה כותבת זאת.

אצל המפרשים: במילים « והוא באמצען » (רמ״א או״ח שצ״ו:א) שבהגהת הרמ״א, הבאות לפני הדעה הפותחת במילים ויש אומרים, נחלקו המפרשים: ראה רמה 2.

השלכה מעשית: לפי לשון הרב, יש לסמוך על הדעה שארבע אמות אלו שלעניין תחומין אינן כארבע אמות שלעניין טלטול, והנותנת ארבע אמות לכל רוח, שמונה על שמונה, בין במקום ששבת בו בין כשיצא חוץ לתחום, להקל בדברי סופרים (שו״ע הרב שצ״ו:ב). ראה פרק 4, נקודה ②.

נקודה ג — מרווחות או מצומצמות (שו״ע הרב שצ״ו:ג)

מידת ארבע האמות

דברי המחבר והרמ״א: סעיף א, « ומודדי' לכל אדם באמה שלו ואם היה ננס באיבריו נותנים לו ארבע אמות בינוני' של כל אדם שכל אחת מהן ששה טפחי » (שו״ע או״ח שצ״ו:א); הגהת הרמ״א, הפותחת במילה הגה, « ואלו ד' אמות מודדין לו מרווחת והוא באמצען » (רמ״א או״ח שצ״ו:א).

דברי הרב: אחרי האמה של כל אדם והננס באברים, « וּכְבָר נִתְבָּאֵר שָׁם שֶׁד' אַמּוֹת אֵלּוּ מוֹדְדִים אוֹתָן מְצֻמְצָמוֹת. וְיֵשׁ אוֹמְרִים שֶׁמּוֹדְדִים אוֹתָן מְרֻוָּחוֹת, וְיֵשׁ לִסְמוֹךְ עַל דִּבְרֵיהֶם בְּד' אַמּוֹת אֵלּוּ שֶׁלְּעִנְיַן תְּחוּמִין, שֶׁעִקָּרָן אֵינָן אֶלָּא מִדִּבְרֵי סוֹפְרִים » (שו״ע הרב שצ״ו:ג).

ההבדל: המחבר והרב מודדים שניהם את ארבע האמות לכל אדם באמה שלו, ונותנים לננס באברים ארבע אמות בינוניות של ששה טפחים; הרב מפרש שננס באברים הוא מי שגופו גדול ואמתו קטנה. הגהת הרמ״א כותבת שמודדים אותן מרווחות, ואינה מביאה דעה אחרת במידה זו. הרב כותב שכבר נתבאר בסימן שמ״ט שמודדים אותן מצומצמות, מביא את המרווחות בלשון יש אומרים, וכותב שיש לסמוך על דבריהם בארבע אמות אלו שלעניין תחומין, שעיקרן אינן אלא מדברי סופרים.

אצל המפרשים: במילים הבאות אחרי « מודדין לו מרווחת » (רמ״א או״ח שצ״ו:א) בהגהת הרמ״א, ראה נקודה ב.

השלכה מעשית: לפי לשון הרב, מודדים ארבע אמות אלו לכל אדם באמה שלו; לננס באברים נותנים ארבע אמות בינוניות של ששה טפחים; ובארבע אמות אלו שלעניין תחומין, שעיקרן אינן אלא מדברי סופרים, יש לסמוך על הדעה שמודדים אותן מרווחות (שו״ע הרב שצ״ו:ג). ראה פרק 4, נקודה ③.

נקודה ד — תל וקמה קצורה (שו״ע הרב שצ״ו:ד)

התל והשבלים

דברי המחבר: בסוף סעיף ב, « ואפילו שבת בתל גבוה ובקמה קצורה ושבלי' מקיפו' אותה חשיבי כד' אמות עד סאתים » (שו״ע או״ח שצ״ו:ב).

דברי הרב: « שָׁבַת בְּתֵל גָּבוֹהַּ י' טְפָחִים, אִם אֵין בּוֹ אֶלָּא בֵּית סָאתַיִם — הֲרֵי כֻּלּוֹ נֶחְשָׁב לוֹ כְּד' אַמּוֹת, מִפְּנֵי שֶׁהוּא כְּמֻקָּף מְחִצּוֹת מִלְמַעְלָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שמ״ה, אֶלָּא שֶׁאֵינָן מֻקָּפוֹת לְדִירָה » (שו״ע הרב שצ״ו:ד); ואחר כך « וְכֵן אִם שָׁבַת בְּקָמָה קְצוּרָה וְשִׁבֳּלִים גְּבוֹהִים י' טְפָחִים מַקִּיפִים אוֹתָהּ. וְהוּא שֶׁקָּשַׁר הַשִּׁבֳּלִים יַחַד בְּעִנְיָן שֶׁאֵין רוּחַ הַמְּצוּיָה מְנִידָה אוֹתָן, שֶׁכָּל מְחִצָּה שֶׁהָרוּחַ מְנִידָהּ — אֵינָהּ מְחִצָּה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שס״ב » (שו״ע הרב שצ״ו:ד).

ההבדל: סעיף ב של המחבר כותב « בתל גבוה » (שו״ע או״ח שצ״ו:ב) בלי שיעור, ואת הקמה הקצורה שהשבלים מקיפות אותה בלי תנאי. לשון הרב נותנת לתל ולשבלים גובה עשרה טפחים; נותנת טעם לתל — שהוא כמוקף מחיצות מלמעלה, אלא שאינן מוקפות לדירה; ומתנה בשבלים: שקשר אותן יחד בעניין שאין רוח המצויה מנידה אותן, שכל מחיצה שהרוח מנידה אותה אינה מחיצה.

אצל המפרשים: המשנה ברורה, המאוחר לרב, כותב על התל « עשרה טפחים וה״ה בשבלים » (משנה ברורה שצ״ו ס״ק ט״ו), ועל השבלים « והוא שקשר השבלים יחד בענין שאין רוח מצויה מנידה אותן שכל מחיצה שרוח מניד אותה אינה מחיצה » (משנה ברורה שצ״ו ס״ק ט״ז).

השלכה מעשית: לפי לשון הרב, מי ששבת בתל גבוה עשרה טפחים, אם אין בו אלא בית סאתיים — כולו נחשב לו כארבע אמות; וכן בקמה קצורה ששבלים גבוהים עשרה טפחים מקיפים אותה, והוא שקשר את השבלים יחד בעניין שאין רוח המצויה מנידה אותן (שו״ע הרב שצ״ו:ד). ראה פרק 4, נקודה ④.

נקודה ה — השובת בעיר (שו״ע הרב שצ״ו:ד)

העיר

דברי המחבר: בתחילת סעיף ב, « בכל מקום שקדש היום עליו אם הוא מוקף לדירה אפי' אין בו עתה דיורין חשוב כולו כד' אמו » (שו״ע או״ח שצ״ו:ב).

דברי הרב: סעיף א שלו כותב את דין המקום המוקף לדירה מתחילתו, או שאין בו יותר מבית סאתיים (פרק 1); ובסוף סעיף ד, « אֲבָל הַשּׁוֹבֵת בְּעִיר — כָּל הָעִיר נֶחְשֶׁבֶת כְּד' אַמּוֹת, אַף עַל פִּי שֶׁאֵינָהּ מֻקֶּפֶת מְחִצּוֹת כְּהִלְכָתָן, כְּגוֹן שֶׁיֵּשׁ בָּהּ פְּרָצוֹת הַרְבֵּה בְּמִלּוּאָן וּבְיוֹתֵר מֵעֶשֶׂר, דְּהוֹאִיל וּבָתֵּי הָעִיר מֻבְלָעִים בְּתוֹךְ שִׁבְעִים אַמָּה וְשִׁירַיִם שֶׁל זֶה — הֲרֵי כֻּלָּם כְּדִירָה אַחַת, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שצ״ח » (שו״ע הרב שצ״ו:ד).

ההבדל: סעיף ב של המחבר כותב את המקום המוקף לדירה, אפילו אין בו עתה דיורין. סעיף ד של הרב כותב שכל העיר נחשבת כארבע אמות אף על פי שאינה מוקפת מחיצות כהלכתן — כגון שיש בה פרצות הרבה במלואן ויותר מעשר — משום שבתי העיר מובלעים בתוך שבעים אמה ושירים זה של זה, והרי כולם כדירה אחת; ומפנה לסימן שצ״ח שלו.

השלכה מעשית: לפי לשון הרב, השובת בעיר — כל העיר נחשבת לו כארבע אמות, אף על פי שאינה מוקפת מחיצות כהלכתן, משום שבתי העיר מובלעים בתוך שבעים אמה ושירים זה של זה (שו״ע הרב שצ״ו:ד); הרב מפנה לסימן שצ״ח שלו. ראה פרק 4, נקודה ⑤.

3 · דברי הרבי

שיחות, מאמרים ואגרות הרבי

דברי הרבי על נושאי סימן שצ״ו

אין הפניה מאומתת נכון לעכשיו.
אנו מייחסים לרבי אך ורק מקורות ראשוניים הניתנים לאימות (מכתב, שיחה או פסק מזוהה, עם קישור). בשלב זה לא ניתן היה לאמת מול Sefaria או Chabad.org דבר מקור של הרבי המתייחס ישירות לסימן זה — מוטב שלא להציג דבר מאשר הפניה לא מאומתת.
כיוון חיפוש מועדף לפסקי הרבי לפי סימן: שערי הלכה ומנהג (מהדורת קה״ת). כל הפניה שתאומת תתווסף עם קישורה.
4 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן שצ״ו, ברגישות החב״דית. כל נקודה מפנה לסעיף שלה בשולחן ערוך הרב, לפי סדר הלשון.

לחסיד חב״ד — סימן שצ״ו

  • ① ארבע האמות של כל אדם: לכל אדם יש ארבע אמות בכל מקום, אפילו יצא חוץ לתחום: יכול להלך בהן ולטלטל בהן; האלפיים נמדדים ממקום שביתתו, חוץ מארבע אמות אלו. שבת במקום המוקף מחיצות בעניין שמותר לטלטל בכולו — כולו נחשב לו כארבע אמות, ונותנים לו אלפיים סביב, אפילו יש בו כמה מילין — והוא שיהא מוקף לדירה מתחילתו, אף על פי שאין בו עתה דיורין, או, אם אינו מוקף לדירה, שאין בו יותר מבית סאתיים; מקום שאינו מוקף לדירה ויש בו יותר מבית סאתיים אינו נחשב כארבע אמות (שו״ע הרב שצ״ו:א). בשורש התחום מביא הרב שתי דעות (פרק 2, נקודה א). הפניה: שולחן ערוך אדמו״ר הזקן שצ״ו:א.
  • ② צורת ארבע האמות: הרב מביא שתי דעות — לדעה הראשונה, בורר לו ארבע אמות לרוח אחת; לדעה השנייה, ארבע אמות אלו שלעניין תחומין אינן כארבע אמות שלעניין טלטול: יש לו ארבע אמות לכל רוח, שמונה על שמונה — וכותב שיש לסמוך על השנייה להקל בדברי סופרים, בין במקום ששבת בו בין כשיצא חוץ לתחום (שו״ע הרב שצ״ו:ב). בהגהת הרמ״א ראה פרק 2, נקודה ב. הפניה: שולחן ערוך אדמו״ר הזקן שצ״ו:ב.
  • ③ מידת ארבע האמות: מודדים ארבע אמות אלו לכל אדם באמה שלו; לננס באברים, שגופו גדול ואמתו קטנה, נותנים ארבע אמות בינוניות של ששה טפחים. הרב כבר ביאר בסימן שמ״ט שמודדים אותן מצומצמות; ויש אומרים שמודדים אותן מרווחות, ויש לסמוך על דבריהם בארבע אמות אלו שלעניין תחומין, שעיקרן אינן אלא מדברי סופרים (שו״ע הרב שצ״ו:ג). הפניה: שולחן ערוך אדמו״ר הזקן שצ״ו:ג.
  • ④ התל והקמה הקצורה: מי ששבת בתל גבוה עשרה טפחים, אם אין בו אלא בית סאתיים — כולו נחשב לו כארבע אמות; וכן בקמה קצורה ששבלים גבוהים עשרה טפחים מקיפים אותה, והוא שקשר את השבלים יחד בעניין שאין רוח המצויה מנידה אותן (שו״ע הרב שצ״ו:ד). הפניה: שולחן ערוך אדמו״ר הזקן שצ״ו:ד.
  • ⑤ העיר: השובת בעיר — כל העיר נחשבת לו כארבע אמות, אף על פי שאינה מוקפת מחיצות כהלכתן — כגון שיש בה פרצות הרבה במלואן ויותר מעשר — משום שבתי העיר מובלעים בתוך שבעים אמה ושירים זה של זה (שו״ע הרב שצ״ו:ד); הרב מפנה לסימן שצ״ח שלו. הפניה: שולחן ערוך אדמו״ר הזקן שצ״ו:ד.

⚠ סעיף זה מציג את ההנהגה ההלכתית של חב״ד. לכל שאלה ממשית: להיוועץ ברב שלך.
