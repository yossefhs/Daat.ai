דעת הרב

דעת הרב — שולחן ערוך אדמו״ר הזקן על סימן שע״ה

הדברים השרויים בחצר שאינם אוסרים, בחמשת סעיפי שולחן ערוך אדמו״ר הזקן: המרפסת שלפני העליות והחצר, התל או העמוד שביניהן, הבור וחוליתו, המצבה שלפני המרפסת, הזיזין היוצאים מן הכתלים. בהשוואה לארבעת סעיפי המחבר.

מדוע רמה 4 מוקדשת לאדמו״ר הזקן? שולחן ערוך אדמו״ר הזקן אינו פירוש על המחבר: זהו שולחן ערוך בפני עצמו, שנכתב על ידי אדמו״ר הזקן, בחלוקה ובמספור סעיפים משלו. כאן יש בשולחן ערוך הרב חמישה סעיפים, ובשולחן ערוך של המחבר ארבעה; ההפניות בדף זה הולכות לפי מספור הרב, אלא אם כן נזכר בפירוש השולחן ערוך של המחבר.

דף זה מקבץ, לסימן שע״ה, את הטקסט המלא של הרב, את מה ששונה בלשונו מדברי המחבר, ואת ההנהגה המעשית העולה מדבריו.

שלחן ערוך הרב — סימן שע״ה

הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן שע״ה — מַה הֵם הַדְּבָרִים הַשְּׁרוּיִים בֶּחָצֵר שֶׁאֵינָם אוֹסְרִים — וּבוֹ ה׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. חמישה סעיפים (בשולחן ערוך של המחבר ארבעה, במספור שלו).

סעיף א מַה הֵם הַדְּבָרִים הַשְּׁרוּיִים בֶּחָצֵר שֶׁאֵינָם אוֹסְרִים וּבוֹ ה' סְעִיפִים: מִרְפֶּסֶת אֲרֻכָּה שֶׁלִּפְנֵי הָעֲלִיּוֹת וְעוֹמֶדֶת בֶּחָצֵר וְעוֹלִין לָהּ בְּסֻלָּם, וּבְנֵי הָעֲלִיּוֹת…

מַה הֵם הַדְּבָרִים הַשְּׁרוּיִים בֶּחָצֵר שֶׁאֵינָם אוֹסְרִים וּבוֹ ה' סְעִיפִים:

מִרְפֶּסֶת אֲרֻכָּה שֶׁלִּפְנֵי הָעֲלִיּוֹת וְעוֹמֶדֶת בֶּחָצֵר וְעוֹלִין לָהּ בְּסֻלָּם, וּבְנֵי הָעֲלִיּוֹת יוֹצְאִין לָהּ דֶּרֶךְ פִּתְחֵיהֶם הַפְּתוּחִים לָהּ וְיוֹרְדִין מִמֶּנָּה בְּסֻלָּם לֶחָצֵר, וְעוֹבְרִים לִרְשׁוּת הָרַבִּים — הֲרֵי מִרְפֶּסֶת זוֹ לָעֲלִיּוֹת כְּחָצֵר לְבָתִּים, וְאִם לֹא עֵרְבוּ יַחַד אֲסוּרִים לְהוֹצִיא לָהּ כֵּלִים שֶׁשָּׁבְתוּ בָּעֲלִיּוֹת, שֶׁבְּנֵי הָעֲלִיּוֹת אוֹסְרִים אוֹתָהּ זֶה עַל זֶה, הוֹאִיל וְהִיא מְשַׁמֶּשֶׁת לְכֻלָּם, וְכֻלָּם פְּתוּחִים לָהּ כְּמוֹ הַבָּתִּים לֶחָצֵר.

אֲבָל אֵין צָרִיךְ לְעָרֵב עִם בְּנֵי הֶחָצֵר, שֶׁאַף שֶׁהֵם עֵרְבוּ לִרְשׁוּת הָרַבִּים דֶּרֶךְ הֶחָצֵר אֵינָם אוֹסְרִים אוֹתָהּ עֲלֵיהֶם, הוֹאִיל וְאֵינָן פְּתוּחִים לָהּ אֶלָּא לְמִרְפֶּסֶת, וּסְתָם מִרְפֶּסֶת גָּבוֹהַּ עֲשָׂרָה, וְכָל גָּבוֹהַּ עֲשָׂרָה רְשׁוּת בִּפְנֵי עַצְמוֹ הוּא וּמֻחְלֶקֶת מֵרְשׁוּת שֶׁאֶצְלָהּ.

וּמִכָּל מָקוֹם אִם רָצוּ לְעָרֵב עִם בְּנֵי הֶחָצֵר, כְּדֵי שֶׁגַּם הֵם יִהְיוּ מֻתָּרִים לְהוֹצִיא לֶחָצֵר כֵּלִים שֶׁשָּׁבְתוּ בַּעֲלִיּוֹתֵיהֶם — יְכוֹלִים לְעָרֵב עַל יְדֵי הַסֻּלָּם שֶׁמֵּחָצֵר לְמִרְפֶּסֶת, שֶׁהוּא נִדּוֹן כְּפֶתַח, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שע״ב, וַהֲרֵי הֵן כְּב' חֲצֵרוֹת וּפֶתַח בֵּינֵיהֶן, שֶׁאִם רָצוּ מְעָרְבִים יַחַד וְאִם רָצוּ מְעָרֶבֶת כָּל אַחַת לְעַצְמָהּ. וְאַף שֶׁהַסֻּלָּם תּוֹרַת פֶּתַח עָלָיו, אַף עַל פִּי כֵן אֵין הָעֲלִיּוֹת חֲשׁוּבוֹת כִּפְתוּחוֹת לֶחָצֵר, לְפִי שֶׁהַסֻּלָּם אֵין תּוֹרַת פֶּתַח עָלָיו אֶלָּא לְהָקֵל, שֶׁיְּכוֹלִים לְעָרֵב יַחַד עַל יְדֵי סֻלָּם שֶׁבֵּינֵיהֶן כְּמוֹ עַל יְדֵי פֶּתַח, אֲבָל לֹא לְהַחֲמִיר.

בַּמֶּה דְּבָרִים אֲמוּרִים שֶׁבְּנֵי הַמִּרְפֶּסֶת דְּהַיְנוּ בְּנֵי הָעֲלִיּוֹת אֵינָם אוֹסְרִים בֶּחָצֵר? כְּשֶׁיְּעָרְבוּ הֵם יַחַד, שֶׁהֵם מֻתָּרִים בְּמִרְפֶּסֶת שֶׁלָּהֶם, וַהֲרֵי זוֹ רֶגֶל הַמֻּתֶּרֶת בִּמְקוֹמָהּ שֶׁאֵינָהּ אוֹסֶרֶת שֶׁלֹּא בִמְקוֹמָהּ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שע״ח. אֲבָל אִם לֹא עֵרְבוּ, כֵּיוָן שֶׁהֵם אֲסוּרִים בְּמִרְפֶּסֶת שֶׁלָּהֶם — הֲרֵי זֶה רֶגֶל הָאֲסוּרָה בִּמְקוֹמָהּ וְאוֹסֶרֶת שֶׁלֹּא בִמְקוֹמָהּ עַד שֶׁיְּעָרְבוּ שָׁם, כְּמוֹ שֶׁיִּתְבָּאֵר שָׁם:

סעיף ב אִם עֵרְבוּ בְּנֵי הַמִּרְפֶּסֶת לְעַצְמָן וְלֹא עֵרְבוּ עִם בְּנֵי…

אִם עֵרְבוּ בְּנֵי הַמִּרְפֶּסֶת לְעַצְמָן וְלֹא עֵרְבוּ עִם בְּנֵי הֶחָצֵר, וְיֵשׁ בֶּחָצֵר תֵּל אוֹ עַמּוּד שֶׁהוּא שֶׁל בְּנֵי הֶחָצֵר וְשֶׁל בְּנֵי הַמִּרְפֶּסֶת בְּשֻׁתָּפוּת, וּשְׁנֵיהֶם מִשְׁתַּמְּשִׁים בּוֹ בְּחֹל — אִם הוּא גָבוֹהַּ עֲשָׂרָה מִקַּרְקָעִית הֶחָצֵר וְאֵינוֹ נָמוּךְ עֲשָׂרָה מִמִּרְפֶּסֶת, נוֹתְנִים אוֹתוֹ לִבְנֵי הַמִּרְפֶּסֶת, שֶׁהֵם מֻתָּרִים לְהִשְׁתַּמֵּשׁ בּוֹ בְּשַׁבָּת בְּכֵלִים שֶׁשָּׁבְתוּ בָּעֲלִיּוֹת, וְלֹא בְּנֵי הֶחָצֵר בְּכֵלִים שֶׁשָּׁבְתוּ בַּבַּיִת, לְפִי שֶׁלָּהֶם הוּא תַּשְׁמִישׁוֹ בְּנַחַת, כֵּיוָן שֶׁהוּא תּוֹךְ עֲשָׂרָה לָהֶם, וְלִבְנֵי הֶחָצֵר תַּשְׁמִישׁוֹ בְּקָשֶׁה, שֶׁאֵין דֶּרֶךְ לְהִשְׁתַּמֵּשׁ בּוֹ בְּחָצֵר אֶלָּא לְכַתֵּף עָלָיו מַשּׂוֹי כָּבֵד, וְכֵיוָן שֶׁגָּבוֹהַּ עֲשָׂרָה — אֵינוֹ רָאוּי לְכִתּוּף מַשּׂוֹי אֶלָּא בְּקֹשִׁי. וְכֹל שֶׁלְּזֶה תַּשְׁמִישׁוֹ בְּנַחַת וּלְזֶה תַּשְׁמִישׁוֹ בְּקָשֶׁה — נוֹתְנִים אוֹתוֹ לְזֶה שֶׁבְּנַחַת.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁהוּא סָמוּךְ לַמִּרְפֶּסֶת. אֲבָל אִם הוּא מֻפְלָג מִכְּנֶגְדָּהּ ד' טְפָחִים, כֵּיוָן שֶׁאֵינָהּ יְכוֹלָה לְהִשְׁתַּמֵּשׁ בּוֹ אֶלָּא עַל יְדֵי זְרִיקָה בָּאֲוִיר בְּרֹחַב ד' טְפָחִים — הֲרֵי זֶה גַם כֵּן תַּשְׁמִישׁוֹ בְּקָשֶׁה. וְלָכֵן אוֹסְרִים זֶה עַל זֶה, וּשְׁנֵיהֶם אֲסוּרִים בּוֹ בְּכֵלִים שֶׁשָּׁבְתוּ בַּבַּיִת עַד שֶׁיְּעָרְבוּ.

וְאִם הוּא תּוֹךְ עֲשָׂרָה לִשְׁנֵיהֶם, אוֹ יוֹתֵר מֵעֲשָׂרָה לִשְׁנֵיהֶם — אִם הוּא מֻפְלָג מִכְּנֶגֶד הַמִּרְפֶּסֶת ד', נוֹתְנִים אוֹתוֹ לֶחָצֵר, שֶׁתַּשְׁמִישׁוֹ לָהּ בְּנַחַת יוֹתֵר מִלַּמִּרְפֶּסֶת. וְאִם אֵינוֹ מֻפְלָג ד', שְׁנֵיהֶם אֲסוּרִים.

וְאִם הוּא תּוֹךְ עֲשָׂרָה לֶחָצֵר, וְיוֹתֵר מֵעֲשָׂרָה לַמִּרְפֶּסֶת — נוֹתְנִים אוֹתוֹ לֶחָצֵר בְּכָל עִנְיָן, אַף עַל פִּי שֶׁאֵינוֹ מֻפְלָג ד' מִכְּנֶגֶד הַמִּרְפֶּסֶת:

סעיף ג בּוֹר שֶׁבֶּחָצֵר שֶׁל בְּנֵי הֶחָצֵר וְהַמִּרְפֶּסֶת בְּשֻׁתָּפוּת, וּמַקִּיפָתוֹ…

בּוֹר שֶׁבֶּחָצֵר שֶׁל בְּנֵי הֶחָצֵר וְהַמִּרְפֶּסֶת בְּשֻׁתָּפוּת, וּמַקִּיפָתוֹ חֻלְיָא גָּבוֹהַּ י' וּסְמוּכָה לַמִּרְפֶּסֶת, אַף עַל פִּי שֶׁבְּנֵי הֶחָצֵר מִשְׁתַּמְּשִׁים בּוֹ בְּשִׁלְשׁוּל וּבִזְרִיקָה, שֶׁזּוֹרְקִין דְּלִי לְמַעְלָה מִן הַחֻלְיָא וּמְשַׁלְשְׁלִין אוֹתוֹ לַבּוֹר, וּבְנֵי מִרְפֶּסֶת מִשְׁתַּמְּשִׁים בְּשִׁלְשׁוּל בִּלְבָד — אַף עַל פִּי כֵן אוֹסְרִים זֶה עַל זֶה לְמַלֹּאת בִּדְלִי שֶׁשָּׁבַת בְּבֵיתוֹ אוֹ בָּעֲלִיּוֹת, אוֹ לְהָבִיא הַמַּיִם בְּכֵלִים קַלִּים לְהַצְנִיעָם שָׁמָּה, וְאֵין נוֹתְנִים אוֹתוֹ לִבְנֵי הַמִּרְפֶּסֶת, הוֹאִיל וְגַם לָהֶם תַּשְׁמִישׁוֹ בְּקָשֶׁה עַל יְדֵי שִׁלְשׁוּל י' טְפָחִים. וַאֲפִלּוּ אִם הַבּוֹר מָלֵא מַיִם עַד גֹּבַהּ הַחֻלְיָא סָמוּךְ לַמִּרְפֶּסֶת — אָסוּר, מִשּׁוּם גְּזֵרָה שֶׁמָּא יַחְסְרוּ הַמַּיִם בְּשַׁבָּת וְיִשְׁתַּמְּשׁוּ בּוֹ כְּבַתְּחִלָּה. אֲבָל אִם הָיָה הַבּוֹר מָלֵא בְּדָבָר שֶׁאֵינוֹ נִטָּל בְּשַׁבָּת — הֲרֵי הוּא כְּעַמּוּד אוֹ תֵּל גָּבוֹהַּ עֲשָׂרָה מֵחָצֵר וְתוֹךְ עֲשָׂרָה לַמִּרְפֶּסֶת, שֶׁנּוֹתְנִים אוֹתוֹ לַמִּרְפֶּסֶת אִם אֵינוֹ מֻפְלָג מִכְּנֶגְדָּהּ ד':

סעיף ד הָיְתָה מַצֵּבָה גָּבוֹהַּ ד' טְפָחִים לִפְנֵי הַמִּרְפֶּסֶת…

הָיְתָה מַצֵּבָה גָּבוֹהַּ ד' טְפָחִים לִפְנֵי הַמִּרְפֶּסֶת — אֵין הַמִּרְפֶּסֶת אוֹסֶרֶת עַל בְּנֵי הֶחָצֵר אֲפִלּוּ תֵּל וְעַמּוּד הַסְּמוּכָה לָהּ, שֶׁכֵּיוָן שֶׁעָשׂוּ הֶפְסֵק לִפְנֵי הַמִּרְפֶּסֶת, גִּלּוּ דַעְתָּם שֶׁסִּלְּקוּ עַצְמָן מֵהֶחָצֵר:

סעיף ה זִיזִין הַיּוֹצְאִין מִן הַכְּתָלִים וְהֵם מִבְּנֵי הֶחָצֵר וְהָעֲלִיּוֹת בְּשֻׁתָּפוּת…

זִיזִין הַיּוֹצְאִין מִן הַכְּתָלִים וְהֵם מִבְּנֵי הֶחָצֵר וְהָעֲלִיּוֹת בְּשֻׁתָּפוּת, כֹּל שֶׁהוּא לְמַטָּה מֵעֲשָׂרָה — הֲרֵי זֶה נֶחְשָׁב מֵהֶחָצֵר וּבְנֵי הֶחָצֵר מִשְׁתַּמְּשִׁים בּוֹ, וְכֹל שֶׁהוּא בְּתוֹךְ י' טְפָחִים הָעֶלְיוֹנִים הַסְּמוּכִים לָעֲלִיָּה — אַנְשֵׁי עֲלִיָּה מִשְׁתַּמְּשִׁים בּוֹ. וְהַנִּשְׁאָר בֵּין עֲשָׂרָה הַתַּחְתּוֹנִים עַד תְּחִלַּת עֲשָׂרָה הָעֶלְיוֹנִים מִן הַזִּיזִים הַיּוֹצְאִים — שְׁנֵיהֶם אֲסוּרִים לְהִשְׁתַּמֵּשׁ בָּהֶם בְּכֵלִים שֶׁשָּׁבְתוּ בַּבָּתִּים, אֶלָּא אִם כֵּן עֵרְבוּ יַחַד בְּנֵי הֶחָצֵר עִם בְּנֵי הָעֲלִיּוֹת:

מקור: ספריא — שולחן ערוך הרב, אורח חיים שע״ה. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה. הטקסט מועתק כפי שספריא מגישה אותו. בסעיף א, המהדורה המנוקדת של שולחן ערוך הרב שבספריא גורסת « שֶׁאַף שֶׁהֵם עֵרְבוּ לִרְשׁוּת הָרַבִּים דֶּרֶךְ הֶחָצֵר » (שו״ע הרב שע״ה:א), והמהדורה שאינה מנוקדת « שאף שהם עירבו לרשות הרבים דרך החצר » (שו״ע הרב שע״ה:א); במהדורה המנוקדת יש על מילה זו ציון להערת עורכי קה״ת (הערה ה), שספריא אינה מגישה את לשונה; ובתחילת הסעיף נאמר על אותם אנשים « וְעוֹבְרִים לִרְשׁוּת הָרַבִּים » (שו״ע הרב שע״ה:א). המילה ״ערבו״ הושארה כמות שהיא, ושאלת קריאתה, ומה שכתוב בהערה ה של עורכי קה״ת, מובאת לפני הרב.

2 · לשון הרב ולשון המחבר

לשון הרב מול סעיף המחבר

במה שונה לשון הרב מסעיף המחבר

כל נקודה מעמידה זה לצד זה סעיף מן השולחן ערוך של המחבר (במספור שלו, ארבעה סעיפים) ואת הסעיף המקביל בשולחן ערוך אדמו״ר הזקן (במספור שלו, חמישה סעיפים), לפי סדר סעיפי הרב, ואינה מציינת אלא מה שההשוואה בין הלשונות מראה. כך מקבילות שתי החלוקות (סעיף הרב ← סעיף המחבר): 1 ← 1; 2 ← 2; 3 ← אין סעיף מקביל אצל המחבר; 4 ← 3; 5 ← 4. אין בסימן הגהת רמ״א הפותחת במילה הגה; בסעיף 2 של המחבר יש שני מאמרים מוסגרים באותיות קטנות, עם הפניה לטור; את הראשון מייחס הט״ז לרמ״א (נקודה ב). מה שחסר בלשון המחבר אינו חידוש בשל כך: כמה מן הפרטים שהרב כותב כתובים כבר ברש״י, בטור, בבית יוסף, במגן אברהם או בט״ז; בכל נקודה מצוין היכן הם כתובים, ומה אינו כתוב שם. פרט שלא מצאנו את מקורו אינו מוצג בשל כך כמיוחד לרב.

נקודה א — המרפסת, בני העליות ובני החצר (שו״ע הרב שע״ה:א)

מדוע בני העליות אינם אוסרים על החצר

דברי המחבר: בסעיף 1 הוא כותב « מרפסת שהוא דרך לעליות הפתוחים לה ועומדת בחצר ועולים לה בסולם ובני העליות יורדים ממנה לחצר ועוברים לרשות הרבים אינם אוסרים על בני החצר דסולם תורת פתח עליו והוי כשתי חצרות ופתח ביניהם » (שו״ע או״ח שע״ה:א), ומתנה « ובלבד שיערבו כל בני מרפסת לעצמן כדי שתהא רגל המותרת במקומה » (שו״ע או״ח שע״ה:א). הטעם שהוא נותן לכך שבני העליות אינם אוסרים על החצר הוא הסולם, שתורת פתח עליו, והרי הן כשתי חצרות ופתח ביניהן.

דברי הרב: הוא כותב תחילה « הֲרֵי מִרְפֶּסֶת זוֹ לָעֲלִיּוֹת כְּחָצֵר לְבָתִּים » (שו״ע הרב שע״ה:א), ומכאן הדין « וְאִם לֹא עֵרְבוּ יַחַד אֲסוּרִים לְהוֹצִיא לָהּ כֵּלִים שֶׁשָּׁבְתוּ בָּעֲלִיּוֹת, שֶׁבְּנֵי הָעֲלִיּוֹת אוֹסְרִים אוֹתָהּ זֶה עַל זֶה » (שו״ע הרב שע״ה:א). לענין החצר הוא נותן טעם אחר: « הוֹאִיל וְאֵינָן פְּתוּחִים לָהּ אֶלָּא לְמִרְפֶּסֶת, וּסְתָם מִרְפֶּסֶת גָּבוֹהַּ עֲשָׂרָה, וְכָל גָּבוֹהַּ עֲשָׂרָה רְשׁוּת בִּפְנֵי עַצְמוֹ הוּא וּמֻחְלֶקֶת מֵרְשׁוּת שֶׁאֶצְלָהּ » (שו״ע הרב שע״ה:א). הסולם משמש אצלו רק להתיר עירוב משותף, אם רצו: « יְכוֹלִים לְעָרֵב עַל יְדֵי הַסֻּלָּם שֶׁמֵּחָצֵר לְמִרְפֶּסֶת, שֶׁהוּא נִדּוֹן כְּפֶתַח » (שו״ע הרב שע״ה:א); והוא מגביל זאת: « לְפִי שֶׁהַסֻּלָּם אֵין תּוֹרַת פֶּתַח עָלָיו אֶלָּא לְהָקֵל, שֶׁיְּכוֹלִים לְעָרֵב יַחַד עַל יְדֵי סֻלָּם שֶׁבֵּינֵיהֶן כְּמוֹ עַל יְדֵי פֶּתַח, אֲבָל לֹא לְהַחֲמִיר » (שו״ע הרב שע״ה:א). ובסוף הסעיף התנאי « אֲבָל אִם לֹא עֵרְבוּ, כֵּיוָן שֶׁהֵם אֲסוּרִים בְּמִרְפֶּסֶת שֶׁלָּהֶם — הֲרֵי זֶה רֶגֶל הָאֲסוּרָה בִּמְקוֹמָהּ וְאוֹסֶרֶת שֶׁלֹּא בִמְקוֹמָהּ » (שו״ע הרב שע״ה:א), והוא מפנה לסימן שע״ח, « כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שע״ח » (שו״ע הרב שע״ה:א).

מה שכתוב כבר לפניו: טעם הגובה וגבול הסולם הם דברי רש״י על המשנה, שהבית יוסף מביא: « ואעפ״כ אין אוסרין הואיל וגבוה מרפסת י׳ שלא הלכו בסולם אלא להקל ואמרינן תורת מחיצה עליו ואינן אוסרין » (רש״י עירובין פ״ג ע״ב); והוא נסמך על הגמרא שהוא מביא, « סולם תורת פתח עליו ותורת מחיצה עליו והכל להקל » (רש״י עירובין פ״ג ע״ב). ועל התנאי שבסעיף כותב הבית יוסף « אבל אם לא עירבו בני מרפסת לעצמן ודאי אסרי אבני חצר דרגל האסורה במקומה אוסרת שלא במקומה » (בית יוסף או״ח שע״ה).

ההבדל: המחבר מבסס את הדין על הסולם, העושה את העליות ואת החצר כשתי חצרות ופתח ביניהן; הרב מבסס אותו על כך שהעליות אינן פתוחות אלא למרפסת, ושסתם מרפסת גבוהה עשרה — הגובה הוא טעמו של רש״י —, ואינו נותן לסולם תורת פתח אלא כדי להתיר עירוב משותף — להקל ולא להחמיר. עוד כותב הרב, לפני דין החצר, שהמרפסת לעליות כחצר לבתים, ולכן בני העליות אוסרים אותה זה על זה כל זמן שלא עירבו. תנאי המחבר כבר מזכיר את הרגל המותרת במקומה, « ובלבד שיערבו כל בני מרפסת לעצמן כדי שתהא רגל המותרת במקומה » (שו״ע או״ח שע״ה:א); הרב כותב גם את ההפך — רגל האסורה במקומה, שאוסרת שלא במקומה עד שיערבו —, שכבר כותבו הבית יוסף. הרב כותב « וּסְתָם מִרְפֶּסֶת גָּבוֹהַּ עֲשָׂרָה » (שו״ע הרב שע״ה:א), ואינו אומר בסעיף זה מה דינה של מרפסת נמוכה מזה; אחריו כותב המשנה ברורה « ומיירי כשהמרפסת גבוה י״ט מן החצר דאי לא היה גבוה י״ט הוי ליה עם החצר כחצר אחת » (משנה ברורה שע״ה ס״ק א); דף זה אינו מכריע.

השלכה מעשית: במרפסת סתם, גבוהה עשרה [טפחים], שהעליות פתוחות לה ועולים אליה מן החצר בסולם — אסור לבני העליות להוציא למרפסת כלים ששבתו בעליותיהם כל זמן שלא עירבו ביניהם; אינם צריכים לערב עם בני החצר, ואם רצו — כדי להוציא גם לחצר כלים ששבתו בעליותיהם — יכולים לערב על ידי הסולם; ואם לא עירבו ביניהם, הם אוסרים בחצר עד שיערבו (שו״ע הרב שע״ה:א).

נקודה ב — תל או עמוד שבחצר (שו״ע הרב שע״ה:ב)

למי נותנים את התל או את העמוד

דברי המחבר: בסעיף 2 הוא כותב « אם לא עירבו יחד ויש בחצר תל או עמוד שהוא משותף בין שניהם אם אינם גבוהים י״ט הרי אלו נחשבי׳ בין החצר ובין המרפסת ושניהם אסורים להוציא שם כלים שבבתים » (שו״ע או״ח שע״ה:ב); ואחר כך « ואם הם גבוהים מקרקעית החצר י״ט (ובתוך עשרה למרפסת (טור) והם בתוך ד׳ טפחים למרפסת בני מרפסת מותרים לפי שתשמישו (להם) בנחת מלבני החצר » (שו״ע או״ח שע״ה:ב); ואחר כך « ואם היו רחוקים מהמרפס׳ ד״ט או יותר אע״פ שגבוהים עשרה (מן החצר לתוך עשרה למרפסת) (טור) הרי אלו בכלל החצר והמרפס׳ לפי ששניהם אפשר להשתמש בהם על ידי זריקה לפיכך שניהם אסורים להוציא שם כלים ששבתו בבתים עד שיערבו » (שו״ע או״ח שע״ה:ב). את הראשון משני המאמרים המוסגרים באותיות קטנות עם ההפניה לטור מייחס הט״ז לרמ״א: « ורמ״א הכניס לתוך דברי הרמב״ם מן שיטת הטור דהיינו ובתוך י׳ למרפסת בבבא דגבוהים י׳ » (ט״ז או״ח שע״ה ס״ק ב).

דברי הרב: במקום שהמחבר כותב « אם לא עירבו יחד » (שו״ע או״ח שע״ה:ב), הוא כותב « אִם עֵרְבוּ בְּנֵי הַמִּרְפֶּסֶת לְעַצְמָן וְלֹא עֵרְבוּ עִם בְּנֵי הֶחָצֵר » (שו״ע הרב שע״ה:ב), ומוסיף « וּשְׁנֵיהֶם מִשְׁתַּמְּשִׁים בּוֹ בְּחֹל » (שו״ע הרב שע״ה:ב); ואחר כך « אִם הוּא גָבוֹהַּ עֲשָׂרָה מִקַּרְקָעִית הֶחָצֵר וְאֵינוֹ נָמוּךְ עֲשָׂרָה מִמִּרְפֶּסֶת, נוֹתְנִים אוֹתוֹ לִבְנֵי הַמִּרְפֶּסֶת » (שו״ע הרב שע״ה:ב), עם טעם הקושי לבני החצר, « שֶׁאֵין דֶּרֶךְ לְהִשְׁתַּמֵּשׁ בּוֹ בְּחָצֵר אֶלָּא לְכַתֵּף עָלָיו מַשּׂוֹי כָּבֵד » (שו״ע הרב שע״ה:ב), והכלל « וְכֹל שֶׁלְּזֶה תַּשְׁמִישׁוֹ בְּנַחַת וּלְזֶה תַּשְׁמִישׁוֹ בְּקָשֶׁה — נוֹתְנִים אוֹתוֹ לְזֶה שֶׁבְּנַחַת » (שו״ע הרב שע״ה:ב). אחר כך הוא כותב « אֲבָל אִם הוּא מֻפְלָג מִכְּנֶגְדָּהּ ד׳ טְפָחִים, כֵּיוָן שֶׁאֵינָהּ יְכוֹלָה לְהִשְׁתַּמֵּשׁ בּוֹ אֶלָּא עַל יְדֵי זְרִיקָה בָּאֲוִיר בְּרֹחַב ד׳ טְפָחִים — הֲרֵי זֶה גַם כֵּן תַּשְׁמִישׁוֹ בְּקָשֶׁה » (שו״ע הרב שע״ה:ב), ועוד שני דינים: « וְאִם הוּא תּוֹךְ עֲשָׂרָה לִשְׁנֵיהֶם, אוֹ יוֹתֵר מֵעֲשָׂרָה לִשְׁנֵיהֶם — אִם הוּא מֻפְלָג מִכְּנֶגֶד הַמִּרְפֶּסֶת ד׳, נוֹתְנִים אוֹתוֹ לֶחָצֵר, שֶׁתַּשְׁמִישׁוֹ לָהּ בְּנַחַת יוֹתֵר מִלַּמִּרְפֶּסֶת. וְאִם אֵינוֹ מֻפְלָג ד׳, שְׁנֵיהֶם אֲסוּרִים » (שו״ע הרב שע״ה:ב); וכן « וְאִם הוּא תּוֹךְ עֲשָׂרָה לֶחָצֵר, וְיוֹתֵר מֵעֲשָׂרָה לַמִּרְפֶּסֶת — נוֹתְנִים אוֹתוֹ לֶחָצֵר בְּכָל עִנְיָן, אַף עַל פִּי שֶׁאֵינוֹ מֻפְלָג ד׳ מִכְּנֶגֶד הַמִּרְפֶּסֶת » (שו״ע הרב שע״ה:ב).

מה שכתוב כבר לפניו: שהסעיף מדבר כשבני המרפסת עירבו לעצמם ולא עם בני החצר — כבר מפרש רש״י במשנה: « ולא עירבו אלו עם אלו אבל עירבו אלו לעצמן ואלו לעצמן » (רש״י עירובין פ״ג ע״ב); והרמב״ם כותב כן בהלכה שלפני דין התל: « וְעֵרְבוּ אַנְשֵׁי הֶחָצֵר לְעַצְמָן וְאַנְשֵׁי הַמִּרְפֶּסֶת לְעַצְמָן » (הלכות עירובין פ״ד הט״ו). הכלל הוא מה שהט״ז למד מסעיף זה: « הכלל מסעי׳ זה דכל שלזה תשמישו בנחת ולזה בקשה לא מקרי רק רשות לאותו שתשמישו בנחת » (ט״ז או״ח שע״ה ס״ק ב). אך בהמשך, אחרי שלמד את אותו הכלל בטור, הוא כותב « כל זה שיטת הטור אבל הרמב״ם לא ס״ל כך » (ט״ז או״ח שע״ה ס״ק ב), ולומד מלשון הרמב״ם, שהסעיף מביא אותה מלבד הגהות הרמ״א, ששניהם אוסרים זה על זה גם כשיש לחצר מעלה ולמרפסת ריעותא: « פשיטא שאוסרין זה על זה ואפי׳ יש לחצר מעלה ולמרפסת ריעותא » (ט״ז או״ח שע״ה ס״ק ב); לפי הרמב״ם אין היתר אלא למרפסת, כשיש לה מעלה ולחצר ריעותא (ראו להלן). בתל שאינו גבוה עשרה פותח המגן אברהם « וגם למרפסת אין גבוהים י׳ דהוי לשניהם בנחת » (מגן אברהם שע״ה ס״ק א); הוא מביא « וכתב הב״י דאף ע״פ שרחוק ד׳ מן המרפסת שניהם אסורים » (מגן אברהם שע״ה ס״ק א), וכותב « אבל בטור ובפוסקים משמע דכל פחות מי׳ לאו כלום הוא ונותנין אותו לבני החצר אם הוא רחוק ד׳ מן המרפסת או נמוך ממנו י׳ » (מגן אברהם שע״ה ס״ק א) — וכך כותב הטור, « או אם הוא תוך י׳ לשניהן ורחוק מן המרפסת ד׳ טפחים בני החצר מותרין בו שהוא נוח להם יותר » (טור או״ח שע״ה). והוא ממשיך, בתל השווה לשניהם והסמוך למרפסת, בלשון קרובה ללשון הרב: « ואם הוא לשניהן שוה תוך י׳ או יותר מי׳ והוא בתוך ד׳ למרפסת … שניהם אסורין » (טור או״ח שע״ה). בתל שהוא יותר מעשרה לשניהם ומופלג מן המרפסת כותב הט״ז « ואם היו תרתי לריעות׳ לבעל המרפסת דהיינו למעלה מי׳ ומופלג נ״ל שמותר בעל החצר בהתל » (ט״ז או״ח שע״ה ס״ק ב); והוא מעיר על הרמב״ם, שסעיף המחבר מעתיק את לשונו, « ולא זכר הרמב״ם כלל מענין נמוך התל ממרפסת י׳ או יותר » (ט״ז או״ח שע״ה ס״ק ב).

ההבדל: בתל שאינו גבוה עשרה כותב המחבר ששניהם אסורים, בלי לחלק בין סמוך למרפסת למופלג ממנה, והבית יוסף, שהמגן אברהם מביאו, מפרש כך לשון זו, שהיא לשון הרמב״ם: אף כשרחוק ארבעה, שניהם אסורים; הרב כותב שאם הוא מופלג ארבעה נותנים אותו לחצר, כמו שכותב הטור וכמו שהמגן אברהם קורא בטור ובפוסקים. בתל שהוא תוך עשרה לחצר ויותר מעשרה למרפסת, הדין הראשון שבסעיף המחבר אוסר לשניהם בלי לחלק בגובה המרפסת, וכך קורא הט״ז את לשון הרמב״ם שהסעיף מעתיק: « אבל בנמוך י׳ ממרפסת בני החצר מותרים כדעת הטור ולהרמב״ם אינו כן » (ט״ז או״ח שע״ה ס״ק ב); הרב נותן אותו לחצר בכל ענין, כטור לדברי הט״ז, וכמו שהמגן אברהם קורא בטור ובפוסקים, « או נמוך ממנו י׳ » (מגן אברהם שע״ה ס״ק א). עוד כותב הרב, בין השאר, דינים שאינם כתובים בפירוש בסעיף המחבר: יותר מעשרה לשניהם ומופלג ארבעה — נותנים אותו לחצר, כמו שהט״ז סובר (נ״ל) בשיטת הטור, ואילו לפי הרמב״ם, שהסעיף מביא את לשונו, הוא כותב שכשיש ריעותא לשניהם שניהם אוסרים זה על זה, ואין היתר למרפסת אלא כשיש לה מעלה ולחצר ריעותא — גבוה עשרה ואינו רחוק ארבעה: « כל זה שיטת הטור אבל הרמב״ם לא ס״ל כך » (ט״ז או״ח שע״ה ס״ק ב), « וכן אם לשניהם ריעותא ג״כ אוסרים זה על זה אלא שאם יש למרפסת מעלה ולחצר ריעותא אז יש היתר למרפסת היינו בגובה י׳ ואינם רחוקים ד׳ טפחים » (ט״ז או״ח שע״ה ס״ק ב); יותר מעשרה לשניהם ואינו מופלג ארבעה — שניהם אסורים, כמו שכבר כותב הטור, וכך קורא המשנה ברורה במוסגר הראשון שבסעיף: « דאם היה מרוחק עשרה גם ממרפסת היה לשניהם בקשה ויד שניהן שוין בו לאסור זה על זה » (משנה ברורה שע״ה ס״ק ח). אחריו מפרש המשנה ברורה את הדין הראשון של המחבר כרב, « ודוקא כשהיה התל והעמוד סמוך להמרפסת בתוך ד״ט » (משנה ברורה שע״ה ס״ק ז), ומביא שתי דעות בתל שהוא יותר מעשרה לשניהם ומופלג מן המרפסת: « יש אומרים דכיון שלבעל המרפסת יש שתי ריעותות שנמוכים מעשרה וגם מופלגים ד״ט א״כ תשמיש של אנשי החצר נוח יותר שם ונותנין אותו להם » (משנה ברורה שע״ה ס״ק י״א) וכן « ויש אומרים דכיון שלהחצר נמי גבוה י״ט מקרי לשניהם תשמישו בקשה ואוסרין זה ע״ז » (משנה ברורה שע״ה ס״ק י״א); לשון הרב הולכת כדעה הראשונה. שני הפרטים שהרב כותב על התשמיש — ששניהם משתמשים בו בחול, ושבני החצר אינם משתמשים בו אלא לכתף עליו משוי — אינם כתובים בסעיף המחבר, ולא בטור, בבית יוסף, במגן אברהם או בט״ז על סימן זה (לפי מה שמופיע בספריא); דף זה אינו אומר מנין לקחם הרב.

השלכה מעשית: תל או עמוד שבחצר, של בני החצר ובני המרפסת בשותפות, ששניהם משתמשים בו בחול, כשבני המרפסת עירבו לעצמם ולא עם בני החצר: גבוה עשרה מקרקעית החצר, אינו נמוך עשרה מן המרפסת וסמוך לה — נותנים אותו לבני המרפסת; מופלג ארבעה מכנגד המרפסת — אוסרים זה על זה עד שיערבו; תוך עשרה לשניהם, או יותר מעשרה לשניהם — נותנים אותו לחצר אם הוא מופלג ארבעה מכנגד המרפסת, ושניהם אסורים אם אינו מופלג; תוך עשרה לחצר ויותר מעשרה למרפסת — נותנים אותו לחצר בכל ענין (שו״ע הרב שע״ה:ב).

נקודה ג — בור שבחצר וחוליתו (שו״ע הרב שע״ה:ג)

סעיף שאין לו מקביל בשולחן ערוך של המחבר

דברי המחבר: בשולחן ערוך של המחבר אין סעיף על הבור. הבית יוסף כותב מדוע לא כתב הטור את הבור המלא פירות טבל, שכתבו הרמב״ם: « ורבינו לא חשש לכותבו משום דכיון שהוא מלא דבר שאינו ניטל בשבת ה״ל בכלל תל או עמוד שכתב » (בית יוסף או״ח שע״ה).

דברי הרב: הוא כותב את דין הבור שמקיפתו חוליא גבוהה עשרה וסמוכה למרפסת: « אַף עַל פִּי שֶׁבְּנֵי הֶחָצֵר מִשְׁתַּמְּשִׁים בּוֹ בְּשִׁלְשׁוּל וּבִזְרִיקָה, שֶׁזּוֹרְקִין דְּלִי לְמַעְלָה מִן הַחֻלְיָא וּמְשַׁלְשְׁלִין אוֹתוֹ לַבּוֹר, וּבְנֵי מִרְפֶּסֶת מִשְׁתַּמְּשִׁים בְּשִׁלְשׁוּל בִּלְבָד » (שו״ע הרב שע״ה:ג) — « אַף עַל פִּי כֵן אוֹסְרִים זֶה עַל זֶה לְמַלֹּאת בִּדְלִי שֶׁשָּׁבַת בְּבֵיתוֹ אוֹ בָּעֲלִיּוֹת » (שו״ע הרב שע״ה:ג), « הוֹאִיל וְגַם לָהֶם תַּשְׁמִישׁוֹ בְּקָשֶׁה עַל יְדֵי שִׁלְשׁוּל י׳ טְפָחִים » (שו״ע הרב שע״ה:ג); ואחר כך « וַאֲפִלּוּ אִם הַבּוֹר מָלֵא מַיִם עַד גֹּבַהּ הַחֻלְיָא סָמוּךְ לַמִּרְפֶּסֶת — אָסוּר, מִשּׁוּם גְּזֵרָה שֶׁמָּא יַחְסְרוּ הַמַּיִם בְּשַׁבָּת וְיִשְׁתַּמְּשׁוּ בּוֹ כְּבַתְּחִלָּה » (שו״ע הרב שע״ה:ג); ולבסוף « אֲבָל אִם הָיָה הַבּוֹר מָלֵא בְּדָבָר שֶׁאֵינוֹ נִטָּל בְּשַׁבָּת — הֲרֵי הוּא כְּעַמּוּד אוֹ תֵּל גָּבוֹהַּ עֲשָׂרָה מֵחָצֵר וְתוֹךְ עֲשָׂרָה לַמִּרְפֶּסֶת, שֶׁנּוֹתְנִים אוֹתוֹ לַמִּרְפֶּסֶת אִם אֵינוֹ מֻפְלָג מִכְּנֶגְדָּהּ ד׳ » (שו״ע הרב שע״ה:ג).

מה שכתוב כבר לפניו: הרמב״ם כותב בור מלא פירות טבלים, « בּוֹר שֶׁבֶּחָצֵר אִם הָיָה מָלֵא פֵּרוֹת טְבָלִים שֶׁאָסוּר לְטַלְטְלָן בְּשַׁבָּת וְכַיּוֹצֵא בָּהֶן הֲרֵי הוּא וְחֻלְיָתוֹ כְּסֶלַע אוֹ תֵּל שֶׁבֶּחָצֵר » (הלכות עירובין פ״ד הי״ח), כשהוא גבוה עשרה וסמוך למרפסת, « וְהָיָה סָמוּךְ לַמִּרְפֶּסֶת » (הלכות עירובין פ״ד הי״ח); ובור מלא מים, « אֲבָל אִם הָיָה מָלֵא מַיִם אֵין בְּנֵי חָצֵר וְלֹא בְּנֵי מִרְפֶּסֶת מַכְנִיסִין מִמֶּנּוּ לַבָּתִּים עַד שֶׁיְּעָרְבוּ » (הלכות עירובין פ״ד הי״ח). המגן אברהם מביאו, ונותן טעם לבור מלא מים: « אבל מלא מים אסור שמא ימלא ממנה ותחסר מי׳ » (מגן אברהם שע״ה ס״ק א). שגם לבני המרפסת תשמיש הבור בשלשול — כותב רש״י במשא ומתן של הגמרא: « תשמישתו בשלשול הוא אפילו לבני מרפסת שהרי מקרקעיתו דולין » (רש״י עירובין פ״ד ע״א). ושבני החצר משתמשים בו בזריקת הדלי למעלה מן החוליא — כותב רש״י שם: « שאין יכולין למלאות אלא א״כ זורקין דלי למעלה מן החוליא » (רש״י עירובין פ״ד ע״א). ושכשלזה בשלשול ולזה בזריקה שניהם אסורים — כותב הט״ז להלכה: « ורב סבירא ליה דלא מיקרי נחת אלא במשתמש דרך פתח אבל כאן לא אכפת לן בנחת קצת ושלשול וזריקה שוין הם ואסור והכי קי״ל » (ט״ז או״ח שע״ה ס״ק ב).

ההבדל: הרב עושה מן הבור סעיף, שאינו אצל המחבר; הוא כותב בו ששניהם אוסרים זה על זה, אף על פי שבני החצר משתמשים בו בזריקה ובשלשול ובני המרפסת בשלשול בלבד — הדין שהט״ז כותב להלכה; בבור מלא מים הוא נותן טעם של גזרה, שמא יחסרו המים, וטעם זה נותן גם המגן אברהם בלשונו, במקום שהרמב״ם כותב רק שאין מכניסין ממנו לבתים עד שיערבו. ובבור מלא דבר שאינו ניטל בשבת הוא כותב את התנאי במידתו — שלא יהיה מופלג ארבעה מכנגד המרפסת — במקום שהרמב״ם כותב « וְהָיָה סָמוּךְ לַמִּרְפֶּסֶת » (הלכות עירובין פ״ד הי״ח); אין זה הבדל בדין: המשנה כבר מגדירה את הסמיכות במידה זו, « וְאֵיזוֹ הִיא סְמוּכָה — כׇּל שֶׁאֵינָהּ רְחוֹקָה אַרְבָּעָה טְפָחִים » (עירובין פ״ג ע״ב). אחריו כותב המשנה ברורה את דין הבור בלשון קרובה ללשונו: « משום גזירה שמא יחסרו המים בשבת וישתמשו בו כבתחלה אבל אם היה הבור מלא בדבר שאין ניטל בשבת הרי הוא כעמוד או תל גבוה י׳ מחצר ותוך עשרה למרפסת שנותנין אותו לבני מרפסת [כשאינו מופלג מכנגדה ד״ט וכנ״ל] להשתמש על גבה » (משנה ברורה שע״ה ס״ק י״א).

השלכה מעשית: בור שבחצר, של בני החצר ובני המרפסת בשותפות, שמקיפתו חוליא גבוהה עשרה וסמוכה למרפסת: כשלא עירבו יחד, אוסרים זה על זה למלאות בדלי ששבת בבית או בעליות, או להביא מים בכלים קלים להצניעם שם, אפילו כשהבור מלא מים עד גובה החוליא; ואם הוא מלא דבר שאינו ניטל בשבת — נותנים אותו למרפסת, ובלבד שאינו מופלג ארבעה מכנגדה (שו״ע הרב שע״ה:ג).

נקודה ד — מצבה לפני המרפסת (שו״ע הרב שע״ה:ד)

גבוהה ארבעה, והטעם

דברי המחבר: בסעיף 3 הוא כותב « היתה מצבה ארבע׳ טפחים לפני המרפס׳ אין המרפסת אוסר׳ על בני החצר שהרי נחלקה מהם » (שו״ע או״ח שע״ה:ג), ואינו אומר אם ארבעת הטפחים הם בגובה או ברוחב.

דברי הרב: הוא כותב « הָיְתָה מַצֵּבָה גָּבוֹהַּ ד׳ טְפָחִים לִפְנֵי הַמִּרְפֶּסֶת » (שו״ע הרב שע״ה:ד); הוא מרחיב את הדין, « אֲפִלּוּ תֵּל וְעַמּוּד הַסְּמוּכָה לָהּ » (שו״ע הרב שע״ה:ד); ונותן טעם: « שֶׁכֵּיוָן שֶׁעָשׂוּ הֶפְסֵק לִפְנֵי הַמִּרְפֶּסֶת, גִּלּוּ דַעְתָּם שֶׁסִּלְּקוּ עַצְמָן מֵהֶחָצֵר » (שו״ע הרב שע״ה:ד).

מה שכתוב כבר לפניו: הטעם כבר בגמרא, בדקה שלפני המרפסת: « דְּכֵיוָן דְּעָבֵיד דַּקָּה — אִיסְתַּלּוֹקֵי אִיסְתַּלּוּק לֵיהּ מֵהָכָא » (עירובין ס׳ ע״א), רמה 2, פרק 8; והט״ז נותנו על סעיף זה של המחבר: « דכיון שעשה הפסק לפני המרפסת גילה דעתיה דסליק נפשיה מהחצר » (ט״ז או״ח שע״ה ס״ק ג). המגיד משנה, שהבית יוסף מביאו, כותב שנראה מדברי הרמב״ם שהוא מפרש שהמצבה מתירה גם את הרשויות שנאמר עליהן למעלה שהן אסורות: « ומדבריו נראה שהוא מפרש שהמצבה הזאת מתרת הרשויות שנאמר עליהם למעלה שהם אסורים » (בית יוסף או״ח שע״ה). ולשון הרמב״ם שמופיעה בספריא היא « הָיְתָה מַצֵּבָה רְחָבָה אַרְבָּעָה טְפָחִים לִפְנֵי הַמִּרְפֶּסֶת » (הלכות עירובין פ״ד הט״ז).

ההבדל: המחבר נותן טעם « שהרי נחלקה מהם » (שו״ע או״ח שע״ה:ג); הרב כותב את הטעם שהט״ז נותן לדברי המחבר, והוא טעם הגמרא — בהפסק גילו בני המרפסת דעתם שסילקו עצמם מן החצר — וכותב בגוף הסעיף שהדין חל אפילו על תל ועמוד הסמוכים למרפסת; לפי מה שנראה למגיד משנה, שהבית יוסף מביאו, בדברי הרמב״ם, כבר חל דין המצבה על מה שנאמר למעלה שהוא אסור, ודף זה אינו אומר אם שתי ההרחבות חופפות. הוא כותב שהמצבה גבוהה ארבעה, במקום שהמחבר אינו מפרש אם ארבעת הטפחים בגובה או ברוחב, ולשון הרמב״ם שבספריא כותבת שהיא רחבה ארבעה; אחריו כותב המשנה ברורה על ארבעת הטפחים « בגובה שלה » (משנה ברורה שע״ה ס״ק י״ב), ועל הדין « ר״ל אפילו התל ועמוד הסמוכה לה בתוך ד״ט וגבוה י״ט » (משנה ברורה שע״ה ס״ק י״ג). דף זה אינו אומר אם הרב בא למעט מצבה רחבה ארבעה שאינה גבוהה ארבעה: השאלה מובאת לפני הרב.

השלכה מעשית: מצבה גבוהה ארבעה טפחים לפני המרפסת — אין המרפסת אוסרת על בני החצר, אפילו תל ועמוד הסמוכים לה (שו״ע הרב שע״ה:ד).

נקודה ה — זיזין היוצאין מן הכתלים (שו״ע הרב שע״ה:ה)

של מי הזיזין, ואיזה עירוב מתירם

דברי המחבר: בסעיף 4 הוא כותב « זיזים היוצאים מהכתלים כל שהוא למטה מעשרה טפחים הרי זה נחשב מהחצר ובני החצר משתמשין בו » (שו״ע או״ח שע״ה:ד), ועל הנשאר שביניהם « שניהם אסורים בו ואין משתמשין בהם בכלים ששבתו בבתים אלא אם כן עירבו » (שו״ע או״ח שע״ה:ד), ואינו אומר של מי הזיזין.

דברי הרב: הוא כותב שהם של שניהם, « וְהֵם מִבְּנֵי הֶחָצֵר וְהָעֲלִיּוֹת בְּשֻׁתָּפוּת » (שו״ע הרב שע״ה:ה), ושהעירוב המתיר הוא עירוב בני החצר עם בני העליות: « אֶלָּא אִם כֵּן עֵרְבוּ יַחַד בְּנֵי הֶחָצֵר עִם בְּנֵי הָעֲלִיּוֹת » (שו״ע הרב שע״ה:ה).

מה שכתוב כבר לפניו: בתל ובעמוד כותב כבר המחבר « שהוא משותף בין שניהם » (שו״ע או״ח שע״ה:ב), והבית יוסף מבאר את התנאי: « שאילו היה של א׳ מהם לא היו האחרים אוסרים עליהם » (בית יוסף או״ח שע״ה). בזיזין התנאי אינו כתוב בטור, בבית יוסף, במגן אברהם או בט״ז על סימן זה (לפי מה שמופיע בספריא); דף זה אינו אומר מנין לקחו הרב.

ההבדל: הרב כותב בזיזין את תנאי השותפות, שסעיף המחבר כותב בתל ובעמוד ולא בזיזין, ומפרש מי צריכים לערב יחד. אחריו כותב גם המשנה ברורה « ושהזיזים היו שייכין לבני החצר והעליה בשותפות » (משנה ברורה שע״ה ס״ק י״ד); והוא מוסיף, בשם רש״י, « וגם מיירי שהזיזין היו רחבים ד״ט » (משנה ברורה שע״ה ס״ק י״ד), מה שהרב אינו כותב בסעיף זה.

השלכה מעשית: זיזין היוצאין מן הכתלים, של בני החצר והעליות בשותפות: למטה מעשרה — בני החצר משתמשים בהם; בתוך עשרה טפחים העליונים, הסמוכים לעליה — אנשי העליה משתמשים בהם; ביניהם — שניהם אסורים להשתמש בהם בכלים ששבתו בבתים, אלא אם כן עירבו יחד בני החצר עם בני העליות (שו״ע הרב שע״ה:ה).

3 · דברי הרבי

שיחות, מאמרים ואגרות הרבי

דברי הרבי על נושאי סימן שע״ה

אין הפניה מאומתת נכון לעכשיו.
אנו מייחסים לרבי אך ורק מקורות ראשוניים הניתנים לאימות (מכתב, שיחה או פסק מזוהה, עם קישור). בשלב זה לא ניתן היה לאמת מול Sefaria או Chabad.org דבר מקור של הרבי המתייחס ישירות לסימן זה — מוטב שלא להציג דבר מאשר הפניה לא מאומתת.
כיוון חיפוש מועדף לפסקי הרבי לפי סימן: שערי הלכה ומנהג (מהדורת קה״ת). כל הפניה שתאומת תתווסף עם קישורה.
4 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן שע״ה, ברגישות החב״דית, לפי סדר סעיפיו. מספרי הסעיפים הם של הרב.

לחסיד חב״ד — סימן שע״ה

  • ① המרפסת והחצר: במרפסת סתם, גבוהה עשרה [טפחים], שהעליות פתוחות לה ועולים אליה מן החצר בסולם — אסור לבני העליות להוציא אליה כלים ששבתו בעליותיהם כל זמן שלא עירבו ביניהם; אינם צריכים לערב עם בני החצר, ואם רצו יכולים לערב על ידי הסולם; ואם לא עירבו ביניהם, הם אוסרים בחצר עד שיערבו. במרפסת נמוכה מזה, שהרב אינו עוסק בה בסעיף זה — ראה פרק 2, והיעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן שע״ה:1.
  • ② התל או העמוד: של בני החצר והמרפסת בשותפות, ששניהם משתמשים בו בחול, כשבני המרפסת עירבו לעצמם ולא עם בני החצר: גבוה עשרה מן החצר, אינו נמוך עשרה מן המרפסת וסמוך לה — לבני המרפסת; מופלג ארבעה מכנגדה — אסור לשניהם עד שיערבו; תוך עשרה לשניהם, או יותר מעשרה לשניהם — לחצר אם הוא מופלג ארבעה מכנגד המרפסת, ואסור לשניהם אם אינו מופלג; תוך עשרה לחצר ויותר מעשרה למרפסת — לחצר בכל ענין. ראה פרק 2. הפניה: שולחן ערוך אדמו״ר הזקן שע״ה:2.
  • ③ הבור: בור של שותפות שמקיפתו חוליא גבוהה עשרה וסמוכה למרפסת — כשלא עירבו יחד, אסור לשניהם למלאות ממנו בדלי ששבת בבית או בעליות, או להביא מים בכלים קלים להצניעם שם, אפילו כשהוא מלא מים עד גובה החוליא; מלא דבר שאינו ניטל בשבת — למרפסת, אם אינו מופלג ארבעה מכנגדה. ראה פרק 2. הפניה: שולחן ערוך אדמו״ר הזקן שע״ה:3.
  • ④ המצבה: מצבה גבוהה ארבעה טפחים לפני המרפסת — אין המרפסת אוסרת על בני החצר, אפילו תל ועמוד הסמוכים לה. ראה פרק 2. הפניה: שולחן ערוך אדמו״ר הזקן שע״ה:4.
  • ⑤ הזיזין שבכתלים: של בני החצר והעליות בשותפות: למטה מעשרה — לבני החצר; בעשרה טפחים העליונים — לבני העליה; ביניהם — אסורים בכלים ששבתו בבתים, אלא אם כן עירבו בני החצר עם בני העליות. ראה פרק 2. הפניה: שולחן ערוך אדמו״ר הזקן שע״ה:5.
