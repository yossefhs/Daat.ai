דעת הרב

דעת הרב — שולחן ערוך אדמו״ר הזקן על סימן שע״ו

בור ובאר שבין שתי חצרות, בד׳ סעיפי שולחן ערוך אדמו״ר הזקן: המחיצה שלמעלה מן המים והקורה הרחבה ד׳ טפחים, הבור שבאמצע השביל, שלש חורבות שבין שתי חצרות, בית הכסא שבין שני בתים או שתי חצרות.

מדוע רמה 4 מוקדשת לאדמו״ר הזקן? שולחן ערוך אדמו״ר הזקן אינו פירוש על המחבר: זהו שולחן ערוך בפני עצמו, שנכתב על ידי אדמו״ר הזקן, בחלוקה ובמספור סעיפים משלו. כאן יש בשולחן ערוך הרב ד׳ סעיפים, וגם בשולחן ערוך של המחבר ד׳; ההפניות בעמוד זה הולכות לפי מספור הרב, אלא אם כן נזכר בפירוש השולחן ערוך של המחבר.

עמוד זה מקבץ, לסימן שע״ו, את הטקסט המלא של הרב, את מה ששונה בלשונו מדברי המחבר והרמ״א, ואת ההנהגה המעשית העולה מדבריו.

1 · טקסט אדמו״ר הזקן

שלחן ערוך הרב — סימן שע״ו

הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן שע״ו — בּוֹר וּבְאֵר שֶׁבֵּין שְׁתֵּי חֲצֵרוֹת — וּבוֹ ד׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. ד׳ סעיפים (וגם בשולחן ערוך של המחבר ד׳, במספור שלו).

סעיף א בּוֹר וּבְאֵר שֶׁבֵּין שְׁתֵּי חֲצֵרוֹת וּבוֹ ד' סְעִיפִים: בּוֹר שֶׁבֵּין ב' חֲצֵרוֹת וְאֵין בֵּינֵיהֶן…

בּוֹר וּבְאֵר שֶׁבֵּין שְׁתֵּי חֲצֵרוֹת וּבוֹ ד' סְעִיפִים:

בּוֹר שֶׁבֵּין ב' חֲצֵרוֹת וְאֵין בֵּינֵיהֶן פֶּתַח אוֹ חַלּוֹן שֶׁיּוּכְלוּ לְעָרֵב, אוֹ שֶׁיֵּשׁ בֵּינֵיהֶן וְלֹא עֵרְבוּ — אֵין מְמַלְּאִין מִמֶּנּוּ בְּשַׁבָּת, שֶׁהֲרֵי רְשׁוּת שְׁנֵיהֶן שׁוֹלֶטֶת שָׁם, אֶלָּא אִם כֵּן עָשׂוּ לוֹ מְחִצָּה עֲשָׂרָה טְפָחִים לְמַעְלָה מִן הַמַּיִם לְהֶכֵּר הַפְרָשָׁה בֵּין רְשׁוּת זוֹ לְזוֹ, שֶׁבִּמְחִצָּה זוֹ אוֹמְרִים "גּוּד אַחֵת" לְמַטָּה עַד קַרְקָעִית הַבּוֹר, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שנ״ה, הוֹאִיל וְיֵשׁ בָּהּ הֶכֵּר וְזִכָּרוֹן שֶׁלֹּא יָבֹאוּ לְטַלְטֵל מֵחָצֵר לְחָצֵר בְּמָקוֹם אַחֵר שֶׁלֹּא בִּמְקוֹם מִלּוּי מַיִם, שֶׁבְּמַיִם בִּלְבָד הֵקֵלּוּ לוֹמַר "גּוּד אַחֵת" מִטַּעַם שֶׁנִּתְבָּאֵר שָׁם. אֲבָל כֹּתֶל הֶחָצֵר שֶׁמְּהַלֵּךְ עַל גַּבֵּי הַבּוֹר אֵינוֹ נִדּוֹן מִשּׁוּם מְחִצָּה, לְפִי שֶׁאֵין בּוֹ הֶכֵּר כְּלָל, הוֹאִיל וְלֹא נַעֲשָׂה בִּשְׁבִיל כָּךְ.

וְאִם הַמְּחִצָּה הִיא כֻּלָּהּ בְּתוֹךְ הַמַּיִם — צָרִיךְ שֶׁיְּהֵא טֶפַח יוֹצֵא מִמֶּנָּה לְמַעְלָה מִן הַמַּיִם כְּדֵי שֶׁתְּהֵא הַהַפְרָשָׁה בֵּינֵיהֶן נִכֶּרֶת. וְאִם הִיא כֻּלָּה לְמַעְלָה מִן הַמַּיִם — צָרִיךְ שֶׁיְּהֵא טֶפַח מִמֶּנָּה מְשֻׁקָּע בַּמַּיִם, לִמְנוֹעַ הַדְּלִי מִלֵּילֵךְ וְלַחְצוֹב מַיִם מֵרְשׁוּת חֲבֵרוֹ. וְאַף שֶׁגַּם עַכְשָׁו הוֹלֵךְ הַדְּלִי מִתַּחַת הַמְּחִצָּה לִרְשׁוּת חֲבֵרוֹ — אֵין בְּכָךְ כְּלוּם, שֶׁקַּל הוּא שֶׁהֵקֵלּוּ חֲכָמִים בְּמַיִם שֶׁלֹּא לְהַצְרִיךְ בָּהֶם מְחִצָּה גְמוּרָה לְהַבְדִּיל בֵּין מַיִם לְמַיִם לְגַמְרֵי, אֶלָּא שֶׁיְּהֵא נִרְאֶה כְּמֻבְדָּלִים, וְלֹא יְהֵא נִרְאֶה שֶׁזֶּה דוֹלֶה מֵרְשׁוּת חֲבֵרוֹ אֶלָּא כְּאִלּוּ כָּל אֶחָד דּוֹלֶה מֵרְשׁוּתוֹ.

וְכֵן אִם הִנִּיחוּ עַל פְּנֵי הַבּוֹר קוֹרָה רְחָבָה ד' טְפָחִים — זֶה מְמַלֵּא מִצַּד זֶה וְזֶה מְמַלֵּא מִצַּד זֶה, שֶׁהֵקֵלּוּ בְּמַיִם לוֹמַר "פִּי תִּקְרָה יוֹרֵד וְסוֹתֵם" מִשְׁתֵּי רוּחוֹתֶיהָ, וּכְאִלּוּ פִּי הַתִּקְרָה הוּא מְחִצָּה יוֹרֶדֶת עַד קַרְקָעִית הַבּוֹר. וְכַאן אֵין לָחוּשׁ שֶׁיְּהֵא נִרְאֶה שֶׁזֶּה דוֹלֶה מֵרְשׁוּת חֲבֵרוֹ, לְפִי שֶׁשִּׁעֲרוּ חֲכָמִים שֶׁאֵין דְּלִי מְהַלֵּךְ יוֹתֵר מִד' טְפָחִים, וְאִם כֵּן לֹא יַחְצוֹב מַיִם מִצַּד הַב' שֶׁל הַקּוֹרָה שֶׁהוּא רְשׁוּת חֲבֵרוֹ. וְאַף שֶׁחוֹצֵב מִתַּחַת הַקּוֹרָה, וְשָׁם רְשׁוּת שְׁנֵיהֶם שׁוֹלֶטֶת — אֵין בְּכָךְ כְּלוּם, שֶׁהֵקֵלּוּ חֲכָמִים בְּמַיִם וְלֹא חָשְׁשׁוּ אֶלָּא שֶׁלֹּא יַחְצוֹב הַדְּלִי מִצַּד הַב', שֶׁאָז נִרְאֶה לָעַיִן שֶׁחוֹצֵב מֵרְשׁוּת חֲבֵרוֹ. מַה שֶּׁאֵין כֵּן תַּחַת הַקּוֹרָה אֵינוֹ נִכָּר כָּל כָּךְ שֶׁרְשׁוּת חֲבֵרוֹ שׁוֹלֶטֶת שָׁם, וְאֵין נִרְאֶה כְּחוֹצֵב מֵרְשׁוּת חֲבֵרוֹ. וְכֵן הֵקֵלּוּ בְּמַיִם וְלֹא חָשְׁשׁוּ מַה שֶּׁהַמַּיִם מִתְעָרְבִים וְהוֹלְכִים מִצַּד זֶה שֶׁל הַמְּחִצָּה אוֹ הַקּוֹרָה לְצַד הַב', וְהַלָּה דוֹלֶה אוֹתָם שָׁם.

וְכָל זֶה לְמַלֹּאת הַמַּיִם וּלְהַכְנִיסָם לַבַּיִת, אוֹ אֲפִלּוּ יִשְׁתַּמֵּשׁ בָּהֶם בֶּחָצֵר אֶלָּא שֶׁדּוֹלֶה בִּדְלִי שֶׁשָּׁבַת בַּבַּיִת. אֲבָל אִם שָׁבַת בֶּחָצֵר — אֵין צָרִיךְ שׁוּם תִּקּוּן לַבּוֹר לְמַלֹּאת מִמֶּנּוּ לַחֲצֵרוֹת, שֶׁהֲרֵי כָּל הַחֲצֵרוֹת רְשׁוּת אֶחָד הֵן לְכֵלִים שֶׁשָּׁבְתוּ בְּתוֹכָן, אַף עַל פִּי שֶׁלֹּא עֵרְבוּ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שע״ב:

סעיף ב וְאִם אֵין הַבּוֹר קָרוֹב לְב' הַחֲצֵרוֹת אֶלָּא רָחוֹק מִכָּל אֶחָד ד'…

וְאִם אֵין הַבּוֹר קָרוֹב לְב' הַחֲצֵרוֹת אֶלָּא רָחוֹק מִכָּל אֶחָד ד' טְפָחִים, כְּגוֹן שֶׁיֵּשׁ שְׁבִיל קָטָן בֵּין כָּתְלֵי ב' חֲצֵרוֹת וְהַבּוֹר בְּתוֹכוֹ, אִם אֵין שׁוּם חָצֵר פְּתוּחָה לוֹ בְּפֶתַח גָּמוּר אֶלָּא מְמַלְּאִין מִמֶּנָּה דֶּרֶךְ חַלּוֹנוֹת הַפְּתוּחִים לוֹ מִשְׁתֵּי הַחֲצֵרוֹת, אִם יֵשׁ מִן הַחַלּוֹנוֹת לְמַיִם י' טְפָחִים, שֶׁתַּשְׁמִישׁ מִלּוּי זֶה הוּא תַּשְׁמִישׁ בְּקֹשִׁי — שְׁנֵיהֶם מֻתָּרִים לְמַלֹּאת מִמֶּנּוּ בְּשַׁבָּת, וְאֵינָן אוֹסְרִים זֶה עַל זֶה אֲפִלּוּ הוּא שֶׁל שְׁנֵיהֶם בְּשֻׁתָּפוּת, כֵּיוָן שֶׁאֵינָן מִשְׁתַּמְּשִׁים בּוֹ אֶלָּא בִּזְרִיקַת הַדְּלִי בָּאֲוִיר בְּרֹחַב ד' טְפָחִים, וְאֵין אָדָם אוֹסֵר עַל אֲוִיר חֲבֵרוֹ דֶּרֶךְ שֶׁהוּא בְּרֹחַב ד' טְפָחִים, לְפִי שֶׁאֵין זֶה חָשׁוּב תַּשְׁמִישׁ כְּלָל, וְאֵינוֹ כְּדַאי לֶאֱסוֹר עַל חֲבֵרוֹ אֶלָּא אִם כֵּן הוּא בְּתוֹךְ עֲשָׂרָה, עַיֵּן סִמָּן שע״ה.

אֲבָל אִם חֲצֵרוֹת פְּתוּחוֹת לַשְּׁבִיל — רְשׁוּת שְׁנֵיהֶם שׁוֹלֶטֶת בּוֹ וַהֲרֵי הוּא כְּחָצֵר שֶׁל שְׁנֵיהֶם, וַאֲסוּרִים בּוֹ עַד שֶׁיְּעָרְבוּ.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁהַשְּׁבִיל הוּא רְשׁוּת הַיָּחִיד. אֲבָל אִם הוּא כַּרְמְלִית, כְּגוֹן שֶׁמְּפֻלָּשׁ בִּשְׁנֵי רָאשָׁיו לְכַרְמְלִית — אֲפִלּוּ אִם הַבּוֹר מֻפְלָג מִכָּתְלֵי הַחֲצֵרוֹת ד' טְפָחִים אָסוּר לְמַלֹּאת מִמֶּנּוּ, אֶלָּא אִם כֵּן מַקִּיפָתוֹ חֻלְיָא גָּבוֹהַּ עֲשָׂרָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן ש[נ]״ד:

סעיף ג שְׁתֵּי חֲצֵרוֹת וּבֵינֵיהֶן ג' חֻרְבּוֹת שֶׁאֵין בַּעֲלֵיהֶן דָּרִין בָּהֶן, וּמֻקָּפוֹת…

שְׁתֵּי חֲצֵרוֹת וּבֵינֵיהֶן ג' חֻרְבּוֹת שֶׁאֵין בַּעֲלֵיהֶן דָּרִין בָּהֶן, וּמֻקָּפוֹת מְחִצּוֹת הֵן מִכָּל רוּחוֹתֵיהֶן וַהֲרֵי הֵן רְשׁוּת הַיָּחִיד גְּמוּרָה, אֶלָּא שֶׁהֵן פְּרוּצוֹת זוֹ לְזוֹ (בְּמִלּוּאָן אוֹ בְּיוֹתֵר מֵעֶשֶׂר אַמּוֹת), וְיֵשׁ לְכָל אַחַת מֵהַחֲצֵרוֹת חַלּוֹנוֹת פְּתוּחִים לְחֻרְבָּה שֶׁאֶצְלָהּ — הֲרֵי כָּל אֶחָד מֻתָּר בְּאוֹתָהּ שֶׁאֶצְלָהּ לְהִשְׁתַּמֵּשׁ בָּהּ דֶּרֶךְ חַלּוֹנוֹ עַל יְדֵי שִׁלְשׁוּל וּזְרִיקָה; הוֹאִיל וְאֵין לוֹ בָּהּ תַּשְׁמִישׁ גָּמוּר בְּחֹל, שֶׁאֵין לוֹ פֶּתַח פָּתוּחַ לָהּ אֶלָּא חַלּוֹן, אֵין פִּרְצַת הַמְּחִצָּה אוֹסֶרֶת.

וּמֻתָּר לְהִשְׁתַּמֵּשׁ בְּכֻלָּה, וַחֲבֵרוֹ אֵינוֹ אוֹסְרָהּ עָלָיו. וְאַף עַל פִּי שֶׁגַּם חֲבֵרוֹ מִשְׁתַּמֵּשׁ בָּהּ בְּחֹל, שֶׁזּוֹרֵק דֶּרֶךְ אֲוִיר שֶׁלּוֹ בַּפִּרְצָה עַד תּוֹךְ חֻרְבָּה זוֹ, מִכָּל מָקוֹם כֵּיוָן שֶׁלַּחֲבֵרוֹ אֵין לוֹ בָּהּ אֶלָּא עַל יְדֵי זְרִיקָה בָּאֲוִיר בְּרֹחַב ד' טְפָחִים וְיוֹתֵר — אֵינוֹ יָכוֹל לְאָסְרָהּ עָלָיו, וְנוֹתְנִים אוֹתָהּ לוֹ, שֶׁיֵּשׁ לוֹ בָּהּ תַּשְׁמִישׁ נוֹחַ בְּשִׁלְשׁוּל סָמוּךְ לְחַלּוֹנוֹ. וְאֵין חֲבֵרוֹ יָכוֹל לְהִשְׁתַּמֵּשׁ בָּהּ בְּשַׁבָּת עַל יְדֵי זְרִיקָה בָּאֲוִיר, מִפְּנֵי שֶׁהוּא אוֹסְרָהּ עַל חֲבֵרוֹ.

אֲבָל חֻרְבָּה הָאֶמְצָעִית, כֵּיוָן שֶׁהִיא רְחוֹקָה מִשְּׁנֵיהֶם וְאֵין לִשְׁנֵיהֶם בָּהּ אֶלָּא תַּשְׁמִישׁ עַל יְדֵי זְרִיקָה בָּאֲוִיר בְּרֹחַב ד' טְפָחִים — אֵינָן אוֹסְרִים זֶה עַל זֶה וּשְׁנֵיהֶם מֻתָּרִים בָּהּ בְּשַׁבָּת, אַף עַל פִּי שֶׁלֹּא עֵרְבוּ יַחַד.

וְאִם שְׁלָשְׁתָּן סְמוּכוֹת לַבָּתִּים, כְּגוֹן שֶׁהָאֶמְצָעִית הִיא כְּנֶגֶד הַשְׁתַּיִם כְּג' רַגְלֵי הַקַּנְקַן — כָּל אֶחָד מֻתָּר בְּחֻרְבָּה שֶׁאֶצְלוֹ, וְהַשְּׁלִישִׁית הַקְּרוֹבָה לִשְׁנֵיהֶם שְׁנֵיהֶם אֲסוּרִים בָּהּ, לְפִי שֶׁיֵּשׁ לִשְׁנֵיהֶם בָּהּ תַּשְׁמִישׁ נוֹחַ בְּשִׁלְשׁוּל סָמוּךְ לַחַלּוֹן, וְאוֹסְרִים זֶה עַל זֶה.

וְחֻרְבּוֹת אֵלּוּ, אַף עַל פִּי שֶׁיֵּשׁ לָהֶם בְּעָלִים, מֻתָּרִים בְּנֵי חֲצֵרוֹת הַסְּמוּכוֹת לָהֶן לְהִשְׁתַּמֵּשׁ בָּהֶן וְאֵין הַבְּעָלִים אוֹסְרִים עֲלֵיהֶן, הוֹאִיל וְאֵינָן דָּרִים בָּהֶם, וְדִירָה בְּלֹא בְעָלִים אֵינָהּ אוֹסֶרֶת, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שע״א:

סעיף ד בֵּית הַכִּסֵּא שֶׁבֵּין ב' בָּתִּים וְכָל אֶחָד נִפְנֶה לוֹ מִבֵּיתוֹ…

בֵּית הַכִּסֵּא שֶׁבֵּין ב' בָּתִּים וְכָל אֶחָד נִפְנֶה לוֹ מִבֵּיתוֹ, אוֹ אֲפִלּוּ עוֹמֵד בֵּין ב' חֲצֵרוֹת וְכָל אֶחָד נִפְנֶה לוֹ מֵחֲצֵרוֹ, אִם הוּא שֶׁל שְׁנֵיהֶם בְּשֻׁתָּפוּת וְלֹא עֵרְבוּ יַחַד — אֲסוּרִים לִפָּנוֹת לוֹ לְכַתְּחִלָּה בְּשַׁבָּת, לְפִי שֶׁהוּא כְּחָצֵר שֶׁל שְׁנֵיהֶם, וְהַצּוֹאָה הִיא כְּכֵלִים שֶׁשָּׁבְתוּ בַּבַּיִת, וְאָסוּר לְהוֹצִיאָם אֲפִלּוּ מֵחָצֵר לְחָצֵר עַד שֶׁיְּעָרְבוּ יַחַד אוֹ יַעֲשׂוּ לוֹ תִּקּוּן:

מקור: ספריא — שולחן ערוך הרב, אורח חיים סימן שע״ו. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה. הטקסט מועתק כפי שספריא מגישה אותו: בסוף סעיף ב, בהפניה לסימן שנ״ד, יש אות שהוסיף המהדיר בסוגריים מרובעים.

2 · לשון הרב ולשון המחבר

לשון הרב מול סעיף המחבר והרמ״א

במה שונה לשון הרב מסעיף המחבר והרמ״א

כל נקודה מעמידה זה לצד זה סעיף מן השולחן ערוך של המחבר (במספור שלו, ד׳ סעיפים), עם הגהת הרמ״א כשיש, ואת הסעיף המקביל בשולחן ערוך אדמו״ר הזקן (במספור שלו, ד׳ סעיפים), לפי סדר סעיפי הרב, ומציינת רק מה שההשוואה בין הלשונות מראה. בסימן זה החלוקות מקבילות סעיף כנגד סעיף (סעיף הרב ← סעיף המחבר): א ← א והגהת הרמ״א; ב ← ב והגהת הרמ״א; ג ← ג; ד ← ד. מה שחסר בלשון המחבר אינו חידוש בשל כך: בסימן זה כמה מן הפירושים שבלשון הרב כתובים כבר, בין השאר, בגמרא, ברמב״ם ובהשגות הראב״ד, בטור, בבית יוסף (המביא גם את רש״י ואת הרא״ש), בהגהת הרמ״א בסימן שנ״ה, במגן אברהם או בט״ז, והמשנה ברורה, שכתב אחריו, מביא כמה מהם. בכל נקודה מצוין היכן הם כתובים; ופירוש שלא חיפשנו את מקורו אין מציגים אותו בשל כך כמיוחד לרב.

נקודה א — המחיצה, הקורה, והדלי ששבת בבית (שו״ע הרב שע״ו:א)

המחיצה והדלי

דברי המחבר: בסעיף א הוא כותב « אין ממלאים ממנו בשבת אלא אם כן עשו מחיצה עשרה למעלה מן המים וצריך שיהיה טפח מן המחיצה יורד בתוך המים » (שו״ע או״ח שע״ו:א); ואחר כך את המחיצה שכולה בתוך המים, ואחריה טעם, « כדי שתהי׳ ניכרת רשות זה מרשות זה » (שו״ע או״ח שע״ו:א), ואת הקורה הרחבה ד׳ טפחים, בלי לפרש את טעמה; הרמ״א מוסיף « ודוקא אם בא למלאות בכלים של בית אבל בכלים של חצר לא בעינן שום תיקון דהא חצירות רשות אחת הן » (רמ״א או״ח שע״ו:א), ומפנה לסימן שע״ב.

דברי הרב: על כותל החצר הוא כותב « אֲבָל כֹּתֶל הֶחָצֵר שֶׁמְּהַלֵּךְ עַל גַּבֵּי הַבּוֹר אֵינוֹ נִדּוֹן מִשּׁוּם מְחִצָּה, לְפִי שֶׁאֵין בּוֹ הֶכֵּר כְּלָל, הוֹאִיל וְלֹא נַעֲשָׂה בִּשְׁבִיל כָּךְ » (שו״ע הרב שע״ו:א); הוא מקדים את המחיצה שכולה בתוך המים לזו שכולה למעלה מן המים, ונותן טעם לטפח המשוקע במים: « לִמְנוֹעַ הַדְּלִי מִלֵּילֵךְ וְלַחְצוֹב מַיִם מֵרְשׁוּת חֲבֵרוֹ » (שו״ע הרב שע״ו:א); לקורה: « לְפִי שֶׁשִּׁעֲרוּ חֲכָמִים שֶׁאֵין דְּלִי מְהַלֵּךְ יוֹתֵר מִד׳ טְפָחִים » (שו״ע הרב שע״ו:א); ובמקרה של הגהת הרמ״א: « וְכָל זֶה לְמַלֹּאת הַמַּיִם וּלְהַכְנִיסָם לַבַּיִת, אוֹ אֲפִלּוּ יִשְׁתַּמֵּשׁ בָּהֶם בֶּחָצֵר אֶלָּא שֶׁדּוֹלֶה בִּדְלִי שֶׁשָּׁבַת בַּבַּיִת » (שו״ע הרב שע״ו:א).

מה שכתוב כבר לפניו: הטור כותב, אחרי דעת רש״י, « ובלבד שתהא עשויה לכך להתיר למלאות » (טור או״ח שע״ו), והבית יוסף מפרש שהוא « לאפוקי מר״י דאמר שם באותה משנה לא תהא מחיצה גדולה מן הכותל שביניהם » (בית יוסף או״ח שע״ו); המגן אברהם כותב בס״ק א, על המילים « שתהיה נכרת » (מגן אברהם שע״ו ס״ק א), « ולא ילך הדלי לרשות חבירו והא דלא חיישי׳ להכי בגזוזטרא ברסי׳ שנ״ה » (מגן אברהם שע״ו ס״ק א), ומבאר במה שונה הגזוזטרא שבסימן שנ״ה; הביאור הלכה, שכתב אחרי הרב, כותב על המילים « וצריך שיהיה טפח » (שו״ע או״ח שע״ו:א): « ומה דבסימן שנ״ה גבי גזוזטרא לא מצרכינן לזה עיין בב״י וב״ח ומ״א » (ביאור הלכה שע״ו). בקורה, הגמרא אומרת « הָא קָא אָזֵיל דְּלִי לְאִידַּךְ גִּיסָא, וּמַיְיתֵי? קִים לְהוּ לְרַבָּנַן דְּאֵין דְּלִי מְהַלֵּךְ יוֹתֵר מֵאַרְבָּעָה טְפָחִים » (עירובין פ״ו ע״ב) ו« תַּחַת קוֹרָה מִיהָא הָא עֲרִיבִי מַיָּא! אֶלָּא מִשּׁוּם דְּקַל הוּא שֶׁהֵקֵילּוּ חֲכָמִים בְּמַיִם » (עירובין פ״ו ע״ב); הרמב״ם, « וּכְאִלּוּ הִבְדִּילָה חֵלֶק זֶה מֵחֵלֶק זֶה אַף עַל פִּי שֶׁהַמַּיִם מְעֹרָבִין מִלְּמַטָּה קַל הוּא שֶׁהֵקֵלּוּ חֲכָמִים בְּמַיִם » (הלכות עירובין פ״ג הכ״ב); הבית יוסף, « וטעמא לפי שאנו רואים כאילו קורה זו מחיצה שנמשכה ירדה עד המים וחלק כל רשות לעצמו » (בית יוסף או״ח שע״ו); והמגן אברהם, בס״ק ב, « דלא אזיל דלי יותר מד׳ ואף על גב דעירבו מיא תחת הקורה קל הוא שהקילו חכמים במים » (מגן אברהם שע״ו ס״ק ב). על הגהת הרמ״א, המגן אברהם כותב בס״ק ג, מן הגהות מיימוניות, « ועכ״פ אסור להכניס המים לבית אם לא עשו תיקון » (מגן אברהם שע״ו ס״ק ג). המשנה ברורה, שכתב אחריו, כותב על הכותל « ומחיצת הכותל שעומד ע״ג הבור אין מועיל לזה כלל דבעינן דוקא שתהא עשויה לשם כך ובתוך אוגן הבור » (משנה ברורה שע״ו ס״ק ג).

ההבדל: סעיף המחבר כותב טעם, « כדי שתהי׳ ניכרת רשות זה מרשות זה » (שו״ע או״ח שע״ו:א), בסוף המקרה השני, של המחיצה שכולה בתוך המים, והגהת הרמ״א נותנת את טעמה; לקורה אין הסעיף כותב טעם. הרב מסדר את שני המקרים של המחיצה בסדר הפוך מסדר המחבר, ונותן לכל אחד טעם: לטפח היוצא מן המים, כשהמחיצה כולה בתוך המים, « כְּדֵי שֶׁתְּהֵא הַהַפְרָשָׁה בֵּינֵיהֶן נִכֶּרֶת » (שו״ע הרב שע״ו:א), קרוב ללשון המחבר; לטפח המשוקע במים, כשהיא למעלה מן המים, « לִמְנוֹעַ הַדְּלִי מִלֵּילֵךְ וְלַחְצוֹב מַיִם מֵרְשׁוּת חֲבֵרוֹ » (שו״ע הרב שע״ו:א). המגן אברהם כותב שלא ילך הדלי לרשות חבירו על המילים « שתהיה נכרת » (מגן אברהם שע״ו ס״ק א), שבסוף המקרה השני, והביאור הלכה מביא את השאלה שהוא דן בה על המילים « וצריך שיהיה טפח » (שו״ע או״ח שע״ו:א), שבמקרה הראשון: לאיזה מקרה תולה המגן אברהם טעם זה, והאם הטעם שכותב הסעיף נאמר גם על המקרה הראשון — עמוד זה אינו מכריע. המשנה ברורה, שכתב אחרי הרב, כותב במקרה הראשון « דבעינן רק שלא יכנס הדלי בהדיא לחלק חבירו » (משנה ברורה שע״ו ס״ק ד), ועל המילה ניכרת « ובזה הטפח לבד סגי שלא ילך הדלי לרשות חבירו » (משנה ברורה שע״ו ס״ק ז); גם על איזה טפח הוא מדבר שם, עמוד זה אינו מכריע. הרעיון שלא ילך הדלי לרשות חבירו כתוב כבר במגן אברהם; ההשוואה לרב היא לרעיון זה, ולא למקרה שכל אחד תולה בו אותו. בקורה, הטעמים שנותן הרב — בין השאר, שרואים את פי התקרה [שעושה הקורה] כמחיצה יורדת, שאין דלי מהלך יותר מד׳ טפחים, ושלא חששו לכך שהמים מתערבים — כתובים כבר בגמרא, ברמב״ם, בבית יוסף ובמגן אברהם (לעיל). שכותל החצר אינו נחשב מחיצה — סעיף המחבר אינו כותב זאת בפירוש; הבית יוסף קורא זאת בדברי הטור, המוציאים מדעת רבי יהודה במשנה, והמשנה ברורה, שכתב אחרי הרב, קורא זאת בלשון הסעיף « אלא אם כן עשו מחיצה » (שו״ע או״ח שע״ו:א): « דבעינן דוקא שתהא עשויה לשם כך » (משנה ברורה שע״ו ס״ק ג); וכן על « וכן אם עשו » (שו״ע או״ח שע״ו:א), בקורה: « אבל כשלא נעשה כדי להתיר לא » (משנה ברורה שע״ו ס״ק ח). על הגהת הרמ״א: הרמ״א מצריך תיקון רק במילוי בכלים של בית; המגן אברהם מוסיף שמכל מקום אסור להכניס את המים לבית בלא תיקון; לשון הרב כוללת את שניהם: התיקון נצרך למלא את המים ולהכניסם לבית, או אפילו להשתמש בהם בחצר כשדולה בדלי ששבת בבית; בדלי ששבת בחצר אין צריך שום תיקון למלא ממנו לחצרות.

השלכה מעשית: שתי חצרות שלא עירבו יחד אינן ממלאות בשבת מן הבור שביניהן, להכניס את המים לבית או בדלי ששבת בבית, עד שיעשו מחיצה עשרה טפחים למעלה מן המים (כותל החצר המהלך על גבי הבור אינו נחשב), שטפח ממנה יוצא מן המים אם היא בתוך המים, או משוקע במים אם היא למעלה מהם, או שיניחו על הבור קורה רחבה ד׳ טפחים; בדלי ששבת בחצר ממלאים ממנו לחצרות בלא תיקון (שע״ו:א).

נקודה ב — הבור שבשביל: ההרחקה, החלונות, הפתחים והכרמלית (שו״ע הרב שע״ו:ב)

השביל שבין שתי חצרות

דברי המחבר: בסעיף ב הוא כותב « באר שבאמצע השביל בין שני כותלי חצירות אע״פ שהיא מופלגת מכותל זה ד״ט ומכותל זה ד״ט שניהם ממלאים ממנה ואינם צריכים זיזים על גבן שאין אדם אוסר על חבירו דרך אויר » (שו״ע או״ח שע״ו:ב); הרמ״א מוסיף « ודוקא כשאין חצרות פתוחות בשביל רק בחלונות » (רמ״א או״ח שע״ו:ב), ואחר כך « וי״א דאם אינן מופלגים ארבעה אוסרים אפילו בכה״ג » (רמ״א או״ח שע״ו:ב), ומציין בסוגריים כמקורו את הבית יוסף בשם רש״י.

דברי הרב: הוא מעמיד את המקרה « וְאִם אֵין הַבּוֹר קָרוֹב לְב׳ הַחֲצֵרוֹת אֶלָּא רָחוֹק מִכָּל אֶחָד ד׳ טְפָחִים » (שו״ע הרב שע״ו:ב); הוא מתנה את ההיתר, « אִם יֵשׁ מִן הַחַלּוֹנוֹת לְמַיִם י׳ טְפָחִים, שֶׁתַּשְׁמִישׁ מִלּוּי זֶה הוּא תַּשְׁמִישׁ בְּקֹשִׁי » (שו״ע הרב שע״ו:ב); הוא כותב « וְאֵינָן אוֹסְרִים זֶה עַל זֶה אֲפִלּוּ הוּא שֶׁל שְׁנֵיהֶם בְּשֻׁתָּפוּת » (שו״ע הרב שע״ו:ב), וכן « וְאֵינוֹ כְּדַאי לֶאֱסוֹר עַל חֲבֵרוֹ אֶלָּא אִם כֵּן הוּא בְּתוֹךְ עֲשָׂרָה, עַיֵּן סִמָּן שע״ה » (שו״ע הרב שע״ו:ב); בחצרות הפתוחות לשביל: « אֲבָל אִם חֲצֵרוֹת פְּתוּחוֹת לַשְּׁבִיל — רְשׁוּת שְׁנֵיהֶם שׁוֹלֶטֶת בּוֹ וַהֲרֵי הוּא כְּחָצֵר שֶׁל שְׁנֵיהֶם, וַאֲסוּרִים בּוֹ עַד שֶׁיְּעָרְבוּ » (שו״ע הרב שע״ו:ב); וכל זה רק בשביל שהוא רשות היחיד: « אֲבָל אִם הוּא כַּרְמְלִית, כְּגוֹן שֶׁמְּפֻלָּשׁ בִּשְׁנֵי רָאשָׁיו לְכַרְמְלִית — אֲפִלּוּ אִם הַבּוֹר מֻפְלָג מִכָּתְלֵי הַחֲצֵרוֹת ד׳ טְפָחִים אָסוּר לְמַלֹּאת מִמֶּנּוּ, אֶלָּא אִם כֵּן מַקִּיפָתוֹ חֻלְיָא גָּבוֹהַּ עֲשָׂרָה » (שו״ע הרב שע״ו:ב).

מה שכתוב כבר לפניו: הטור כותב « ואם אין הבור קרוב להם אלא רחוק מכל אחד ד׳ טפחים כגון שיש שביל קטן בין שתי החצירות והבור בתוכו ואין דיורין פתוחין לו אלא ממלאין דרך חלונות » (טור או״ח שע״ו); הבית יוסף מביא את פירוש רש״י, « דאי לאו הכי דמופלג הוי תשמישו בנחת » (בית יוסף או״ח שע״ו), ואחר כך מה שהוא מדייק מדברי הרמב״ם: « ולפ״ז מופלג דנקט שמואל אינו בדוקא כפירש״י אלא לרבותא נקט מופלגת בד׳ וכ״ש בפחות מכאן » (בית יוסף או״ח שע״ו); על לשון המחבר כותב המגן אברהם « וכ״ש בפחו׳ מד׳ (דאז) אף על פי שהחצירו׳ פתוחות לשביל שרי לסברא זו » (מגן אברהם שע״ו ס״ק ד), ועל לשון הרמ״א « אבל כשהחצירות פתוחות לשביל רשות שניהם שולטות בו והוי השביל כחצר שאינ׳ מעורב׳ ואסור » (מגן אברהם שע״ו ס״ק ה); ולשביל שאינו רשות היחיד מביא הבית יוסף מן התוספות פירוש של ר״י, « א״נ כגון שאותו הפסק שבין שתי חצירות שהבור עומד שם אינו רה״י אלא ר״ה או כרמלית ומחיצות הבור גבוהים י׳ דכשממלאים מהבור לחצר מביאין דרך מלמעלה מי׳ דהוי מקום פטור » (בית יוסף או״ח שע״ו). במקום אחר בסימן זה כותב הבית יוסף שהטור כתב את דין השביל לדעת רש״י: « שתשמיש בני החצירות בבור אינו אלא דרך אויר לדעת רש״י שסתם רבינו דבריו לדעתו » (בית יוסף או״ח שע״ו); והט״ז, הכותב בס״ק ג שהדעה שמביא הרמ״א היא דעת רש״י, כותב « דלרש״י שהוא כשיטת הטור לא תמצא שם תרתי לריעותא » (ט״ז או״ח שע״ו ס״ק ג). הראב״ד, בהשגותיו על הרמב״ם (הלכות עירובין פרק ג, הלכה כג), כבר מצריך, בלי לחלק, חוליא גבוהה עשרה, שאז אין אויר רשות הרבים אוסרו, וגם רשות חבירו אינה אוסרתו דרך אויר; והמגיד משנה תמה עליו (רמה 2, פרק 9).

ההבדל: סעיף המחבר מתיר « אע״פ שהיא מופלגת » (שו״ע או״ח שע״ו:ב) ד׳ טפחים, והמגן אברהם קורא בלשון זו שלסברא זו פחות מד׳ מותר כל שכן; הרמ״א מביא את הדעה האוסרת בפחות מד׳ טפחים, אפילו דרך חלונות. הרב כותב את ההיתר בבור הרחוק ד׳ טפחים מכל חצר, בלשון הטור, שהבית יוסף והט״ז תולים אותה בשיטת רש״י — שיטת הדעה שמביא הרמ״א; סעיף ב שלו פותח « וְאִם אֵין הַבּוֹר קָרוֹב לְב׳ הַחֲצֵרוֹת אֶלָּא רָחוֹק » (שו״ע הרב שע״ו:ב), אחרי סעיף א, העוסק בבור שבין שתי חצרות; בבור קרוב יותר אינו כותב בסעיף זה דבר, ועמוד זה אינו מכריע מה דעתו בזה; המשנה ברורה, שכתב אחריו, כותב על הדעה שהביא הרמ״א « ופסקו האחרונים כדעה זו » (משנה ברורה שע״ו ס״ק ט״ו). בחצרות שיש להן פתחים לשביל הרב כותב כמו שכתבו הרמ״א והמגן אברהם בס״ק ה, על הגהת הרמ״א; ובס״ק ד, על לשון הסעיף, המגן אברהם כותב להפך, שלסברא זו מותר אף כשהחצרות פתוחות לשביל (לעיל) — הוא כותב זאת אחרי דבריו על בור בפחות מד׳ טפחים, ועמוד זה אינו מכריע אם כוונתו גם לד׳ טפחים.סעיף המחבר כותב « ואינם צריכים זיזים על גבן » (שו״ע או״ח שע״ו:ב); הרב אינו מזכיר זיזים. בין הדברים שכותב הרב ואינם לא בסעיף המחבר ולא בהגהת הרמ״א יש בין השאר את התנאי של י׳ טפחים מן החלונות למים, שהוא תולה בסימן שע״ה — בסעיף ב של אותו סימן, בשולחן ערוך, בתוספת שבסוגריים בשם הטור, כתוב « ובתוך עשרה למרפסת » (שו״ע או״ח שע״ה:ב) בתל; ועמוד זה אינו אומר שזה אותו תנאי —; ואת השביל שהוא כרמלית, שבו הוא מצריך חוליא גבוהה עשרה — פירוש ר״י שמביא הבית יוסף כבר מדבר בבור בשביל כזה, שמחיצותיו גבוהות עשרה, והראב״ד כבר מצריך חוליא גבוהה עשרה (לעיל).

השלכה מעשית: בור שבשביל שהוא רשות היחיד, רחוק ד׳ טפחים מכל חצר, שממלאים ממנו דרך חלונות שיש מהם למים י׳ טפחים — שתי החצרות ממלאות ממנו בשבת, אפילו הוא [הבור או השביל] של שתיהן בשותפות; אם יש לחצרות פתחים לשביל, הן אסורות בו עד שיערבו; ואם השביל כרמלית, אסור למלא ממנו אפילו כשהוא רחוק ד׳ טפחים, אלא אם כן מקיפתו חוליא גבוהה עשרה (שע״ו:ב). בבור הקרוב פחות מד׳ טפחים אין סעיף זה נותן את הדין: היעזר ברב שלך.

נקודה ג — שלש חורבות, וחורבות שיש להן בעלים (שו״ע הרב שע״ו:ג)

החורבות

דברי המחבר: בסעיף ג הוא כותב « שתי חצרות וביניהם ג׳ חורבות כל א׳ מותר באותה שאצלו להשתמש בה דרך חלונות על ידי זריקה והאמצעי׳ מותרת לשניהם » (שו״ע או״ח שע״ו:ג), ואחר כך את דין שלש החורבות הסמוכות לבתים; הוא אינו אומר אם יש לחורבות בעלים.

דברי הרב: הוא מתאר את החורבות, « שֶׁאֵין בַּעֲלֵיהֶן דָּרִין בָּהֶן, וּמֻקָּפוֹת מְחִצּוֹת הֵן מִכָּל רוּחוֹתֵיהֶן וַהֲרֵי הֵן רְשׁוּת הַיָּחִיד גְּמוּרָה, אֶלָּא שֶׁהֵן פְּרוּצוֹת זוֹ לְזוֹ (בְּמִלּוּאָן אוֹ בְּיוֹתֵר מֵעֶשֶׂר אַמּוֹת) » (שו״ע הרב שע״ו:ג); כל אחד משתמש בזו שאצלו « דֶּרֶךְ חַלּוֹנוֹ עַל יְדֵי שִׁלְשׁוּל וּזְרִיקָה » (שו״ע הרב שע״ו:ג), והוא נותן טעם: « הוֹאִיל וְאֵין לוֹ בָּהּ תַּשְׁמִישׁ גָּמוּר בְּחֹל, שֶׁאֵין לוֹ פֶּתַח פָּתוּחַ לָהּ אֶלָּא חַלּוֹן, אֵין פִּרְצַת הַמְּחִצָּה אוֹסֶרֶת » (שו״ע הרב שע״ו:ג); ובסוף הסעיף הוא כותב « וְחֻרְבּוֹת אֵלּוּ, אַף עַל פִּי שֶׁיֵּשׁ לָהֶם בְּעָלִים, מֻתָּרִים בְּנֵי חֲצֵרוֹת הַסְּמוּכוֹת לָהֶן לְהִשְׁתַּמֵּשׁ בָּהֶן וְאֵין הַבְּעָלִים אוֹסְרִים עֲלֵיהֶן, הוֹאִיל וְאֵינָן דָּרִים בָּהֶם, וְדִירָה בְּלֹא בְעָלִים אֵינָהּ אוֹסֶרֶת » (שו״ע הרב שע״ו:ג), ומפנה לסימן שע״א.

מה שכתוב כבר לפניו: הבית יוסף מביא את רש״י, המתאר את החורבות, « מחיצות פרוצות זו לזו ואין בהם דיורים ואין להם בעלים אלא בני הבתים הללו ויש חלונות בבתים פתוחים לחורבה » (בית יוסף או״ח שע״ו), ואחר כך את הטעמים, « הואיל ואין לו בה תשמיש גמור בחול שאין לו בה פתח פתוח לה אלא חלון אין פירצת המחיצה אוסרתו ומותר להשתמש בכולו » (בית יוסף או״ח שע״ו), וכן « הואיל ויש לחבירו בה תשמיש נוח בשלשול סמוך לחלונו » (בית יוסף או״ח שע״ו); על מה שכתב רש״י בבעלים מקשה הרא״ש « וקשה דהא קי״ל דירה בלא בעלים לא שמה דירה » (בית יוסף או״ח שע״ו), והבית יוסף כותב « ורבינו סתם דבריו כדעת הרא״ש » (בית יוסף או״ח שע״ו); ואכן הטור כותב « ואפי׳ יש לחורבות בעלים אינן אוסרים כיון שאין דרים בהם » (טור או״ח שע״ו).

ההבדל: סעיף המחבר אינו אומר אם יש לחורבות בעלים; הביאור הלכה תמה על כך, « ולפלא על השו״ע שלא הבית דעתו בזה » (ביאור הלכה שע״ו), ומעמיד מצד אחד את רש״י, ומצד שני « אבל בתוספות הביאו בשם ר״י שחולק על רש״י וס״ל דאפילו יש להן בעלים ג״כ מותר לפי שאינה בית דירה וכן הוא ג״כ דעת הרא״ש והרשב״א » (ביאור הלכה שע״ו), מעלה אפשרות שהמחבר אמר זאת בסתמו, « ואפשר מדסתם ולא חילק משמע ג״כ דבכל גווני שרי » (ביאור הלכה שע״ו), ואחר כך כותב « ובלבוש פסק בהדיא דמותר אפילו ביש להן בעלים וכן פסק הגר״ז » (ביאור הלכה שע״ו). הרב כותב בפירוש, כטור וכרא״ש, שבעלים שאינם דרים בהן אינם אוסרים. הוא כותב גם « שִׁלְשׁוּל וּזְרִיקָה » (שו״ע הרב שע״ו:ג) במקום שסעיף המחבר כותב « על ידי זריקה » (שו״ע או״ח שע״ו:ג), ונותן את הטעמים בלשון רש״י, שהמשנה ברורה, שכתב אחריו, מביא אותם בס״ק י״ז וי״ח.

השלכה מעשית: שלש חורבות שאין בעליהן דרין בהן, מוקפות מחיצות ופרוצות זו לזו (בְּמִלּוּאָן אוֹ בְּיוֹתֵר מֵעֶשֶׂר אַמּוֹת), ולכל חצר חלונות לחורבה שאצלה — כל חצר משתמשת דרך חלונה בחורבה שאצלה, בשלשול או בזריקה, ומותרת להשתמש בכולה; החצר השנייה אינה משתמשת בה בשבת בזריקה באויר; החורבה האמצעית, הרחוקה משתיהן, מותרת לשתיהן בלא עירוב; ואם שלשתן עומדות כג׳ רגלי הקנקן, השלישית, הקרובה לשתיהן, אסורה לשתיהן; בעלים שאינם דרים בחורבות אינם אוסרים (שע״ו:ג).

נקודה ד — בית הכסא שבין שני בתים או שתי חצרות (שו״ע הרב שע״ו:ד)

בית הכסא

דברי המחבר: בסעיף ד הוא כותב « בית הכסא שבין שני הבתים ולא עירבו יחד רשות שניהם שולטת בו ואסורים » (שו״ע או״ח שע״ו:ד), ואחריו, בסוגריים, הפניה לסימן שנ״ה.

דברי הרב: הוא מוסיף את המקרה של שתי חצרות ומעמיד תנאי: « אוֹ אֲפִלּוּ עוֹמֵד בֵּין ב׳ חֲצֵרוֹת וְכָל אֶחָד נִפְנֶה לוֹ מֵחֲצֵרוֹ, אִם הוּא שֶׁל שְׁנֵיהֶם בְּשֻׁתָּפוּת וְלֹא עֵרְבוּ יַחַד — אֲסוּרִים לִפָּנוֹת לוֹ לְכַתְּחִלָּה בְּשַׁבָּת » (שו״ע הרב שע״ו:ד); הוא נותן טעם, « וְהַצּוֹאָה הִיא כְּכֵלִים שֶׁשָּׁבְתוּ בַּבַּיִת, וְאָסוּר לְהוֹצִיאָם אֲפִלּוּ מֵחָצֵר לְחָצֵר » (שו״ע הרב שע״ו:ד), וכותב « עַד שֶׁיְּעָרְבוּ יַחַד אוֹ יַעֲשׂוּ לוֹ תִּקּוּן » (שו״ע הרב שע״ו:ד).

מה שכתוב כבר לפניו: המגן אברהם כותב « דדמי לכלים ששבתו בבית » (מגן אברהם שע״ו ס״ק ט); הבית יוסף מביא מהגהות מיימוניות « בית הכסא שבין שני בתים אם יש ביניהם פתח שיכולין לערב יחד מערבין ומותרים בו ואם לאו רשות שניהם שולטת בו ואסורין » (בית יוסף או״ח שע״ו), ונותן טעם: « וטעמא משום דדמי לבור שבין ב׳ חצירות ואין מחיצה תלויה מועלת בבית הכסא כמו בבור » (בית יוסף או״ח שע״ו); והמרדכי, שהוא מביא אחר כך, מביא דעה המתירה: « אך כתב שה״ר יעקב מקינון מתיר משום דצואה מקום פטור הוא » (בית יוסף או״ח שע״ו). הסוגריים שבסעיף ד מפנים לסימן שנ״ה; הגהת הרמ״א שם, בסעיף ג, כותבת « בית הכסא העומד בין שני בתים אסור לפנות שם אם לא עירבו יחד ולא עשו לו תיקון מבע״י כמו שנתבאר סעיף ב׳ דאז שרי דלא הוי אלא כמו חצר שאינה מעורבת כדלקמן סי׳ שנ״ו ועסי׳ שע״ו ס״ד » (רמ״א או״ח שנ״ה:ג), ו« וכל זה דוקא לכתחלה אבל דיעבד ששכחו ולא ערבו מותר דגדול כבוד הבריות » (רמ״א או״ח שנ״ה:ג); סעיף ב של אותו סימן, שאליו היא מפנה, כותב « ואם עשה דף או קנה פחות מג׳ שתפול הצואה עליו קודם שתפול למים אין צריך מחיצות » (שו״ע או״ח שנ״ה:ב), והמשנה ברורה, שכתב אחרי הרב, מבאר הפניה זו: « היינו תיקון דדף או קנה. אבל תיקון דמחיצה תלויה אינו מועיל הכא עד שיורדת עד למטה » (משנה ברורה שנ״ה ס״ק ל״ה) — התיקון שמדובר בו הוא, לדבריו, דף או קנה, ולא המחיצה התלויה שאותו סעיף מתיר בבית הכסא שעל פני המים; גם הבית יוסף כותב שאין מחיצה תלויה מועלת בבית הכסא (לעיל).

ההבדל: סעיף המחבר מדבר בשני בתים ואינו מעמיד תנאי של בעלות; הרב מוסיף בין השאר את שתי החצרות ואת התנאי « אִם הוּא שֶׁל שְׁנֵיהֶם בְּשֻׁתָּפוּת » (שו״ע הרב שע״ו:ד); המילה « לְכַתְּחִלָּה » (שו״ע הרב שע״ו:ד) והתיקון לצד העירוב כתובים כבר בהגהת הרמ״א בסימן שנ״ה, שאליו מפנים הסוגריים שבסעיף ד; וטעם הכלים ששבתו בבית — במגן אברהם. הרב והרמ״א משווים שניהם את המקום לחצר, אבל לא לאותה מסקנה: אצל הרב, « לְפִי שֶׁהוּא כְּחָצֵר שֶׁל שְׁנֵיהֶם » (שו״ע הרב שע״ו:ד) הוא טעם האיסור; אצל הרמ״א, « כמו חצר שאינה מעורבת » (רמ״א או״ח שנ״ה:ג) בא אחרי « דאז שרי » (רמ״א או״ח שנ״ה:ג), לאחר שנעשה התיקון, והמשנה ברורה כותב על ההפניה שאחריו « דשם מבואר דבחצר שאינה מעורבת לכו״ע מותר כחו » (משנה ברורה שנ״ה ס״ק ל״ו). בסימן שנ״ה הרב עצמו כותב במקרה זה: « בֵּית הַכִּסֵּא הָעוֹמֵד בֵּין ב׳ בָּתִּים, אִם לֹא עֵרְבוּ יַחַד — אֲסוּרִים לִפָּנוֹת שָׁם לְדִבְרֵי הַכֹּל, אֲפִלּוּ הָיְתָה שָׁם צוֹאָה מִבְּעוֹד יוֹם, אִם לֹא עַל יְדֵי דַף וְקָנֶה, שֶׁאָז מֻתָּר אַף עַל פִּי שֶׁהִיא בְּתוֹךְ הָעִיר, לְפִי שֶׁבְּחָצֵר שֶׁאֵינָהּ מְעֹרֶבֶת לֹא גָזְרוּ עַל כֹּחוֹ, אַף עַל פִּי שֶׁהִיא בְּתוֹךְ הָעִיר, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שנ״ז » (שו״ע הרב שנ״ה:י״ב); ובסעיף ד של סימן זה הוא כותב « אוֹ יַעֲשׂוּ לוֹ תִּקּוּן » (שו״ע הרב שע״ו:ד) בלי לפרט. מה הדין בדיעבד אין סעיף ד של הרב אומר; הרב כותב זאת בסימן שנ״ה, בסעיף שאחריו: « וְכָל זֶה לְכַתְּחִלָּה. אֲבָל בְּדִיעֲבַד, שֶׁשָּׁכְחוּ וְלֹא עֵרְבוּ — מֻתָּר מִשּׁוּם כְּבוֹד הַבְּרִיּוֹת » (שו״ע הרב שנ״ה:י״ג), כמו הגהת הרמ״א בסימן שנ״ה (לעיל); והמשנה ברורה, שכתב אחרי הרב, מפנה לסימן שנ״ה וכותב « דאם שכח ולא עירב בדיעבד מותר לפנות דגדול כבוד הבריות » (משנה ברורה שע״ו ס״ק כ״ד). בס״ק כ״ג הוא מעמיד את סעיף המחבר במקרה שמקום המושב מיוחד לכל אחד ומקום החפירה של שניהם, וכותב « אבל אם גם מקום המושב היה שייך לשניהם מותר » (משנה ברורה שע״ו ס״ק כ״ג); הרב כותב את תנאי השותפות בלי לחלק בין מקום המושב למקום החפירה, ועמוד זה אינו מכריע כיצד שני הלשונות מתיישבים.

השלכה מעשית: בית הכסא שבין שני בתים, שכל אחד נפנה לו מביתו, או שבין שתי חצרות, שכל אחד נפנה לו מחצרו, והוא של שניהם בשותפות — אם לא עירבו יחד, אין נפנים לו לכתחלה בשבת, עד שיערבו יחד או יעשו לו תיקון (שע״ו:ד); במקרה זה הרב כותב בסימן שנ״ה (סעיף י״ב) דף או קנה, והמשנה ברורה (שנ״ה ס״ק ל״ה) והבית יוסף כותבים שאין מחיצה תלויה מועילה בו. למי ששכח לערב, הרב כותב בסימן שנ״ה (סעיף י״ג) שבדיעבד מותר, משום כבוד הבריות, וכן הגהת הרמ״א בסימן שנ״ה והמשנה ברורה; היעזר ברב שלך.

4 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן שע״ו, ברגישות החב״דית, לפי סדר סעיפיו. מספרי הסעיפים הם לפי מספור הרב.

לחסיד חב״ד — סימן שע״ו

  • ① הבור שבין שתי חצרות: שתי חצרות שלא עירבו יחד אינן ממלאות בשבת מן הבור שביניהן, להכניס את המים לבית או בדלי ששבת בבית, בלא מחיצה עשרה טפחים למעלה מן המים העשויה לכך — שטפח ממנה יוצא מן המים אם היא בתוך המים, או משוקע במים אם היא למעלה מהם —, או בלא קורה רחבה ד׳ טפחים המונחת על הבור; כותל החצר המהלך על גבי הבור אינו נחשב מחיצה; בדלי ששבת בחצר ממלאים לחצרות בלא שום תיקון. הפניה: שולחן ערוך אדמו״ר הזקן 376:1.
  • ② הבור שבשביל: בשביל שהוא רשות היחיד, בור הרחוק ד׳ טפחים מכל חצר, שממלאים ממנו דרך חלונות שיש מהם למים י׳ טפחים, מותר לשתי החצרות, אפילו הוא [הבור או השביל] של שתיהן בשותפות; אם יש לחצרות פתחים לשביל, הן אסורות בו עד שיערבו; ואם השביל כרמלית, אין ממלאים ממנו בלא חוליא גבוהה עשרה סביב הבור. לבור קרוב יותר ראה פרק 2, והיעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן 376:2.
  • ③ שלש החורבות: בין שתי חצרות, שלש חורבות מוקפות מחיצות ופרוצות זו לזו (בְּמִלּוּאָן אוֹ בְּיוֹתֵר מֵעֶשֶׂר אַמּוֹת), ולכל חצר חלונות לחורבה שאצלה: כל חצר משתמשת בחורבה שאצלה דרך חלונה, בשלשול או בזריקה, והחצר השנייה אינה משתמשת בה בשבת בזריקה באויר; האמצעית, הרחוקה משתיהן, מותרת לשתיהן בלא עירוב; כשהן עומדות כג׳ רגלי הקנקן, השלישית, הקרובה לשתיהן, אסורה לשתיהן; בעלים שאינם דרים בחורבות אינם אוסרים: « וְדִירָה בְּלֹא בְעָלִים אֵינָהּ אוֹסֶרֶת » (שו״ע הרב שע״ו:ג). הפניה: שולחן ערוך אדמו״ר הזקן 376:3.
  • ④ בית הכסא המשותף: בית הכסא שבין שני בתים או שתי חצרות, שכל אחד נפנה לו מביתו או מחצרו, שהוא של שניהם בשותפות, אין משתמשים בו לכתחלה בשבת, עד שיערבו או יעשו לו תיקון (במקרה זה דף או קנה, ולא מחיצה תלויה: פרק 2, נקודה ד); למי ששכח לערב, הרב כותב שבדיעבד מותר, משום כבוד הבריות (סימן שנ״ה, סעיף י״ג: פרק 2, נקודה ד); היעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן 376:4.

⚠ פרק זה מציג את ההנהגה ההלכתית של חב״ד. לכל שאלה ממשית: היעזר ברב שלך.
