הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן ש״ע · 370 — דִּינֵי שִׁתּוּף בְּעֵרוּב — וּבוֹ י׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. י׳ סעיפים (בשולחן ערוך של המחבר ח׳, במספור שלו).

סעיף א דִּינֵי שִׁתּוּף בְּעֵרוּב וּבוֹ י' סְעִיפִים: הַדָּר בְּבֵית שַׁעַר, אַף עַל פִּי שֶׁיֵּשׁ לוֹ ד' מְחִצּוֹת…

דִּינֵי שִׁתּוּף בְּעֵרוּב וּבוֹ י' סְעִיפִים:

הַדָּר בְּבֵית שַׁעַר, אַף עַל פִּי שֶׁיֵּשׁ לוֹ ד' מְחִצּוֹת, כֵּיוָן שֶׁרַבִּים עוֹבְרִים דֶּרֶךְ שָׁם לְבָתֵּיהֶם — אֵין עָלָיו תּוֹרַת בַּיִת אֶלָּא תּוֹרַת חָצֵר, וְאֵין צָרִיךְ לְעָרֵב עִם שְׁאָר הַבָּתִּים. וְכֵן הַדָּר בְּאַכְסַדְרָא וּמִרְפֶּסֶת שֶׁבְּחָצֵר אֵינוֹ אוֹסֵר עַל בְּנֵי הֶחָצֵר, שֶׁאֵינָן חֲשׁוּבוֹת דִּירָה. וְכֵן סֻכַּת הֶחָג בֶּחָג אֵינָהּ חֲשׁוּבָה דִירָה לֶאֱסוֹר. אֲבָל הַדָּר בְּבֵית הַתֶּבֶן, בְּבֵית הַבָּקָר, בְּבֵית הָאוֹצָר, בְּבֵית הָעֵצִים — אוֹסֵר עַד שֶׁיְּעָרֵב, לְפִי שֶׁרְאוּיִים לָדוּר בָּהֶם וַחֲשׁוּבִים דִּירָה, אַף עַל פִּי שֶׁעֲשׂוּיִים מִקָּנִים וְכַיּוֹצֵא בָהֶם:

סעיף ב בַּעַל הַבַּיִת שֶׁיֵּשׁ לוֹ הַרְבֵּה בָּתִּים בַּחֲצֵרוֹ…

בַּעַל הַבַּיִת שֶׁיֵּשׁ לוֹ הַרְבֵּה בָּתִּים בַּחֲצֵרוֹ, וְהִשְׂכִּירָן לַאֲחֵרִים אוֹ שֶׁהִשְׁאִילָם (לִזְמַן, בְּעִנְיָן שֶׁאֵינוֹ יָכוֹל לְסַלְּקָם תּוֹךְ הַזְּמַן בְּעַל כָּרְחָם) — הֲרֵי הֵם אוֹסְרִים עָלָיו בֶּחָצֵר כְּאִלּוּ הָיוּ בָּתִּים שֶׁלָּהֶם, עַד שֶׁיְּעָרְבוּ. וְכֵן אָסוּר לְטַלְטֵל מִבֵּיתוֹ לְבָתֵּיהֶם עַד שֶׁיְּעָרְבוּ.

וְאִם יֵשׁ לוֹ בְּכָל אֶחָד מֵהֶם דְּבָרִים שֶׁאֵינָם נִטָּלִים בְּשַׁבָּת מֵחֲמַת אִסּוּר, שֶׁאָסוּר לְטַלְטְלָם אֲפִלּוּ לְצֹרֶךְ גּוּפָם אוֹ מְקוֹמָם, כֵּיוָן שֶׁלֹּא סִלֵּק עַצְמוֹ מִבָּתִּים אֵלּוּ מִכֹּל וְָכֹל אֶלָּא יֵשׁ לוֹ בָּהֶם תְּפִיסַת יָד — נַעֲשָׂה כְּאִלּוּ דָּר בָּהֶם, וְאֵין הַדָּרִים אוֹסְרִים עָלָיו, לְפִי שֶׁנַּעֲשׂוּ כֻלָּם כְּאוֹרְחִים אֶצְלוֹ. וּלְפִיכָךְ גַּם הֵם מֻתָּרִים לְהוֹצִיא מִבָּתֵּיהֶם לֶחָצֵר, אַף עַל פִּי שֶׁלֹּא נָתְנוּ עֵרוּב, כְּמוֹ שֶׁמֻּתָּרִים אוֹרְחִים שֶׁמִּתְאָרְחִים בְּבֵיתוֹ לְהוֹצִיא מִבֵּיתוֹ לֶחָצֵר בְּלִי שֶׁיְּעָרְבוּ עִמּוֹ. אֲבָל אִם אֵין לוֹ בָּהֶם אֶלָּא דְּבָרִים שֶׁמֻּתָּר לְטַלְטֵל לְצֹרֶךְ גּוּפָם אוֹ מְקוֹמָם — אֵין זוֹ חֲשׁוּבָה תְּפִיסַת יָד, כֵּיוָן שֶׁיָּכוֹל לְהוֹצִיאָם מִשָּׁם כְּשֶׁיִּצְטָרֵךְ לָהֶם (אוֹ לִמְקוֹמָם). וְיֵשׁ אוֹמְרִים שֶׁאִם אֵינָן נִטָּלִין מֵחֲמַת כָּבְדָּן, חֲשׁוּבִין תְּפִיסַת יָד אַף עַל פִּי שֶׁמֻּתָּרִים בְּטִלְטוּל (וְיֵשׁ לְהַחֲמִיר בְּזֶה).

וְאֵין תְּפִיסַת יָד מוֹעֶלֶת אֶלָּא בְּבָתִּים שֶׁהֵם שֶׁלּוֹ, בֵּין שֶׁהֵם קְנוּיִים לוֹ בֵּין שֶׁהֵם שְׂכוּרִים אוֹ שְׁאוּלִים אֶצְלוֹ וְחָזַר וְהִשְׂכִּירָם אוֹ הִשְׁאִילָם לַאֲחֵרִים. אֲבָל מִי שֶׁיֵּשׁ לוֹ תְּפִיסַת יָד בְּבַיִת שֶׁאֵינוֹ לֹא קָנוּי וְלֹא שָׂכוּר וְלֹא שָׁאוּל — אֵין זֶה מוֹעִיל כְּלוּם, וְאוֹסֵר עָלָיו בֶּחָצֵר וְכֵן לְטַלְטֵל מִבֵּיתוֹ לְבֵיתָם:

סעיף ג יֵשׁ אוֹמְרִים שֶׁאֵין תְּפִיסַת יָד מוֹעֶלֶת אֶלָּא כְּשֶׁאֵין עִמָּהֶם דִּיּוּרִין אֲחֵרִים בֶּחָצֵר…

יֵשׁ אוֹמְרִים שֶׁאֵין תְּפִיסַת יָד מוֹעֶלֶת אֶלָּא כְּשֶׁאֵין עִמָּהֶם דִּיּוּרִין אֲחֵרִים בֶּחָצֵר, וְאָז אֵין צָרִיךְ לְעָרֵב כְּלָל בֵּינֵיהֶם, אוֹ אֲפִלּוּ יֵשׁ דִּיּוּרִים בֶּחָצֵר אֶלָּא שֶׁאוֹתָן דִּיּוּרִים מְבִיאִים עֵרוּבָן וּמַנִּיחִין אוֹתוֹ בְּבֵיתוֹ שֶׁל בַּעַל הַבַּיִת זֶה שֶׁיֵּשׁ לוֹ תְּפִיסַת יָד, אוֹ בְּאֶחָד מִבָּתִּים אֵלּוּ שֶׁיֵּשׁ לוֹ בָּהֶם תְּפִיסַת יָד, וְאָז אֵין כָּל בָּתִּים אֵלּוּ צְרִיכִים לִתֵּן כְּלָל לָעֵרוּב פַּת, לְפִי שֶׁכָּל בָּתִּים אֵלּוּ הֵם כְּבַיִת אֶחָד, וּבַיִת שֶׁהָעֵרוּב מֻנָּח בּוֹ אֵין צָרִיךְ לִתֵּן אֶת הַפַּת. אֲבָל אִם הָעֵרוּב מֻנָּח בְּאֶחָד מִבָּתֵּי הַדִּיּוּרִין — הֲרֵי כָּל הַבָּתִּים אֵלּוּ צְרִיכִים כָּל אֶחָד לִתֵּן פַּת לָעֵרוּב, שֶׁמִּתּוֹךְ שֶׁבָּתִּים אֵלּוּ הֻזְקְקוּ לְעֵרוּב עַל כָּל פָּנִים בִּשְׁבִיל בַּיִת שֶׁדָּר בּוֹ בַּעַל הַבַּיִת, צְרִיכִים גַּם הֵם לִתֵּן בִּשְׁבִיל בָּתֵּיהֶם, אַף עַל פִּי שֶׁיֵּשׁ לְבַעַל הַבַּיִת בָּהֶם תְּפִיסַת יָד. וַהֲרֵי דִינוֹ עִמָּהֶם כְּדִין אַחִים הַמְקַבְּלִים פְּרָס מֵאֲבִיהֶם שֶׁיִּתְבָּאֵר:

סעיף ד אֶחָד שֶׁשָּׂכַר בִּירָה מִן הַנָּכְרִי וְדָר בָּהּ עִם חֲבֵרוֹ…

אֶחָד שֶׁשָּׂכַר בִּירָה מִן הַנָּכְרִי וְדָר בָּהּ עִם חֲבֵרוֹ, זֶה בְּחֶדֶר זֶה וְזֶה בְּחֶדֶר זֶה, וְיֵשׁ לְהַשּׂוֹכֵר תְּפִיסַת יָד בְּחֶדֶר חֲבֵרוֹ, אִם מִתְּחִלַּת הַשְּׂכִירוּת שֶׁשְּׂכָרָהּ מִן הַנָּכְרִי הִסְכִּימוּ לָדוּר יַחַד וּלְדַעַת כֵּן שְׂכָרָהּ מִן הַנָּכְרִי, אַף עַל פִּי שֶׁהוּא לְבַדּוֹ שְׂכָרָהּ מִן הַנָּכְרִי — הֲרֵי חֲבֵרוֹ שֻׁתָּף עִמּוֹ בָּהּ וְאֵינוֹ יָכוֹל לְסַלְּקוֹ כָּל זְמַן מֶשֶׁךְ הַשְּׂכִירוּת מִן הַנָּכְרִי, שֶׁכֵּיוָן שֶׁלְּדַעַת כֵּן שְׂכָרָהּ מִן הַנָּכְרִי, כְּדֵי שֶׁיָּדוּרוּ בְּיַחַד — הֲרֵי נַעֲשָׂה הוּא שָׁלִיחַ לַחֲבֵרוֹ לִשְׂכּוֹר בַּעֲדוֹ מִן הַנָּכְרִי, וּלְפִיכָךְ אֵין תְּפִיסַת יָד מוֹעֶלֶת לוֹ בְּשֶׁל חֲבֵרוֹ, כֵּיוָן שֶׁאֵינוֹ שֶׁלּוֹ. אֲבָל אִם מִתְּחִלַּת הַשְּׂכִירוּת שְׂכָרָהּ לְעַצְמוֹ, וְאַחַר כָּךְ נִמְלַךְ וְהִשְׂכִּיר לַחֲבֵרוֹ — מוֹעִיל לוֹ תְּפִיסַת יָד בְּשֶׁל חֲבֵרוֹ:

סעיף ה חָמֵשׁ חֲבוּרוֹת שֶׁשָּׁבְתוּ בִּטְרַקְלִין וְחִלְּקוּהוּ בִּמְחִצּוֹת לַחֲמִשָּׁה חֲדָרִים…

חָמֵשׁ חֲבוּרוֹת שֶׁשָּׁבְתוּ בִּטְרַקְלִין וְחִלְּקוּהוּ בִּמְחִצּוֹת לַחֲמִשָּׁה חֲדָרִים, וְכָל אֶחָד מֵהֶם פָּתוּחַ לֶחָצֵר — אִם הַמְּחִצּוֹת מַגִּיעוֹת לַתִּקְרָה, אוֹ בְּתוֹךְ ג' טְפָחִים סָמוּךְ לַתִּקְרָה, הֲרֵי כָּל אֶחָד וְאֶחָד חֶדֶר בִּפְנֵי עַצְמוֹ, אֲפִלּוּ הֵן מְחִצּוֹת עֲרַאי כְּגוֹן שֶׁל יְרִיעוֹת, וּצְרִיכוֹת כָּל אֶחָד וְאֶחָד מֵהַחֲבוּרוֹת לִתֵּן עֵרוּבָן.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁיֵּשׁ עוֹד דִּיּוּרִין בֶּחָצֵר, וְעֵרוּב כָּל הֶחָצֵר מַנִּיחִין אוֹתוֹ בְּאֶחָד מִדִּיּוּרִין הָהֵם אוֹ בְּחָצֵר אַחֶרֶת שֶׁעֵרְבָה עִם זוֹ, שֶׁמִּתּוֹךְ שֶׁטְּרַקְלִין זֶה הֻזְקַק לָעֵרוּב מֵחֲמַת הַדִּיּוּרִין — צְרִיכָה כָּל חֲבוּרָה לִתֵּן עֵרוּב. אֲבָל אִם דִּיּוּרִין הָהֵם נָתְנוּ עֵרוּבָן בְּזֶה הַטְּרַקְלִין, אֵינָם צְרִיכִים לִתֵּן עֵרוּב כְּלָל, שֶׁכֻּלָּם דָּרִים בְּבַיִת זֶה, וּבַיִת שֶׁמַּנִּיחִין בּוֹ הָעֵרוּב אֵינוֹ צָרִיךְ לִתֵּן פַּת לְהָעֵרוּב. וְאַף שֶׁטְּרַקְלִין זֶה מֻחְלָק בִּמְחִצּוֹת לְה' חֲדָרִים, וְהָעֵרוּב הוּא בְּאֶחָד מֵהֶם — אֵין בְּכָךְ כְּלוּם, הוֹאִיל וְאֵינָן אֶלָּא מְחִצּוֹת עֲרַאי. וְכֵן אִם אֵין דִּיּוּרִין אֲחֵרִים כְּלָל בֶּחָצֵר אֵין צָרִיךְ עֵרוּב כְּלָל, שֶׁאֵינָן אוֹסְרִים כְּלָל זֶה עַל זֶה, הוֹאִיל וְכֻלָּם בְּבַיִת אֶחָד שֶׁאֵינוֹ מֻחְלָק אֶלָּא בִּמְחִצּוֹת עֲרַאי.

חִלְּקוּהוּ בִּמְחִצּוֹת שֶׁאֵינָן מַגִּיעוֹת לַתִּקְרָה — אֲפִלּוּ אִם יֵשׁ דִּיּוּרִין בֶּחָצֵר וְעֵרוּב כָּל הֶחָצֵר מַנִּיחִים בְּאֶחָד מִדִּיּוּרִין הָהֵם, אֵין צְרִיכִים כָּל הַחֲבוּרוֹת אֶלָּא לִתֵּן עֵרוּב אֶחָד כְּאִלּוּ אֵין בֵּינֵיהֶן מְחִצּוֹת כְּלָל. וְאִם מִקְצָתָן עָשׂוּ מְחִצּוֹת הַמַּגִּיעוֹת לַתִּקְרָה וּמִקְצָתָן לֹא עָשׂוּ — אוֹתָן שֶׁעָשׂוּ הֵם מְחֻלָּקִים, וְאוֹתָן שֶׁלֹּא עָשׂוּ הֵן כִּמְשֻׁתָּפִין.

וְכָל זֶה כְּשֶׁעָשׂוּ מְחִצּוֹת עֲרַאי לְפִי שָׁעָה. אֲבָל אִם שָׁבְתוּ בַּחֲדָרִים וַעֲלִיּוֹת מַמָּשׁ, וְהֵם פְּתוּחִים לֶחָצֵר — אוֹסְרִים זֶה עַל זֶה אֲפִלּוּ אֵין דִּיּוּרִין אֲחֵרִים בֶּחָצֵר. וּצְרִיכוֹת כָּל חֲבוּרָה וַחֲבוּרָה לִתֵּן עֵרוּב אֲפִלּוּ מַנִּיחִין הָעֵרוּב בְּזֶה הַטְּרַקְלִין.

וְכָל זֶה כְּשֶׁכֻּלָּם פְּתוּחִים לֶחָצֵר. אֲבָל אִם אֵין פָּתוּחַ לֶחָצֵר אֶלָּא הַחִיצוֹן בִּלְבָד, וְכֻלָּם עוֹבְרִים דֶּרֶךְ עָלָיו, אֲפִלּוּ הֵם חֲדָרִים מַמָּשׁ — אֵין צָרִיךְ לִתֵּן עֵרוּב אֶלָּא שְׁנַיִם הַפְּנִימִיִּים, שֶׁהֵם בִּלְבַדָּם חֲשׁוּבִים דִּירוֹת וְאוֹסְרִים בֶּחָצֵר זֶה עַל זֶה וְכֵן לְטַלְטֵל מִזֶּה לְזֶה, אֲבָל כָּל הָאֲחֵרִים חֲשׁוּבִים כְּבֵית שַׁעַר לָהֶם, כְּמוֹ שֶׁיִּתְבָּאֵר.

וְכָל זֶה כְּשֶׁכֻּלָּם שָׁוִין בִּטְרַקְלִין זֶה. אֲבָל בַּעַל הַבַּיִת שֶׁיֵּשׁ לוֹ מְלַמֵּד אוֹ סוֹפֵר בְּבֵיתוֹ, וְכֵן תַּלְמִידִים הַלּוֹמְדִים בִּפְנֵי הָרַב וְדָרִים בְּבֵיתוֹ, אַף עַל פִּי שֶׁיֵּשׁ לְכָל אֶחָד חֶדֶר מְיֻחָד פָּתוּחַ לֶחָצֵר וְגַם לִרְשׁוּת הָרַבִּים, וְאוֹכֵל וְיָשֵׁן בְּחַדְרוֹ — אֵין צָרִיךְ לְעָרֵב בֶּחָצֵר כְּלָל, מִפְּנֵי שֶׁאֵין אוֹסְרִים כְּלָל, לֹא בֶּחָצֵר וְלֹא לְטַלְטֵל מֵחַדְרֵיהֶם לַבַּיִת אוֹ מֵחֶדֶר לְחֶדֶר, אֲפִלּוּ אֵין לוֹ בָּהֶם תְּפִיסַת יָד, לְפִי שֶׁאֵינוֹ מַשְׁאִיל לָהֶם רְשׁוּתוֹ לֶאֱסוֹר עָלָיו וְלֹא לֶאֱסוֹר זֶה עַל זֶה, וַהֲרֵי יָכוֹל לְסַלְּקָם בְּכָל עֵת.

אֲבָל אִם הַחֲדָרִים קְנוּיִים לָהֶם אוֹ שְׂכוּרִים (אוֹ שְׁאוּלִים בְּעִנְיָן שֶׁאֵינוֹ יָכוֹל לְסַלְּקָם תּוֹךְ זְמַן שְׁאִילָתָם — אֵין לוֹמַר שֶׁאֵינוֹ מַשְׁאִיל לָהֶם רְשׁוּתוֹ לֶאֱסוֹר עָלָיו, כֵּיוָן שֶׁכְּבָר הִשְׁאִיל לָהֶם וְאֵין בְּיָדוֹ לְסַלְּקָם, וּבְעַל כָּרְחוֹ הֵם) אוֹסְרִים עָלָיו, אֶלָּא אִם כֵּן יֵשׁ לוֹ בָּהֶם תְּפִיסַת יָד. וַאֲפִלּוּ הֵם מִשְׁתַּמְּשִׁים בַּאֲפִיָּה וּבִשּׁוּל בְּבַיִת גָּדוֹל שֶׁל בַּעַל הַבַּיִת — אֵין זֶה מוֹעִיל כְּלוּם, כֵּיוָן שֶׁאוֹכְלִים בְּחַדְרֵיהֶם, אֶלָּא אִם כֵּן תַּלְמִידִים הַמְקַבְּלִים פְּרָס מֵרַבָּן כְּמוֹ שֶׁיִּתְבָּאֵר:

סעיף ו עִקַּר דִּירַת הָאָדָם הִיא מְקוֹם אֲכִילַת פַּת, וְלֹא מְקוֹם הַשֵּׁנָה וְלִינָה…

עִקַּר דִּירַת הָאָדָם הִיא מְקוֹם אֲכִילַת פַּת, וְלֹא מְקוֹם הַשֵּׁנָה וְלִינָה. לְפִיכָךְ אַנְשֵׁי הֶחָצֵר שֶׁהָיוּ כֻלָּם אוֹכְלִים בְּבַיִת אֶחָד בְּשַׁבָּת, אַף עַל פִּי שֶׁכָּל אֶחָד אוֹכֵל מִשֶּׁלּוֹ עַל שֻׁלְחָן בִּפְנֵי עַצְמוֹ, וְיֵשׁ לְכָל אֶחָד חֶדֶר בִּפְנֵי עַצְמוֹ בֶּחָצֵר שֶׁדָּר שָׁם כָּל הַיּוֹם וְכָל הַלַּיְלָה — אֵינָם צְרִיכִים עֵרוּב, מִפְּנֵי שֶׁהֵם כְּאַנְשֵׁי בַּיִת אֶחָד וְאֵינָן אוֹסְרִים זֶה עַל זֶה.

וְכֵן אִם הֻצְרְכוּ לַעֲשׂוֹת עֵרוּב עִם חָצֵר אַחֵר — עֵרוּב אֶחָד לְכֻלָּם וּפַת אֶחָד בִּלְבָד מוֹלִיכִין לְאוֹתוֹ מָקוֹם שֶׁהֵם מְעָרְבִין עִמּוֹ, וְאֵיזֶה מֵהֶם שֶׁיִּרְצֶה יִתֵּן הַפַּת, וַאֲפִלּוּ בְּנֵי בֵיתוֹ שֶׁל בַּעַל הַבַּיִת יְכוֹלִין לִתֵּן הַפַּת מִשֶּׁלָּהֶם. וְאִם הָיָה הָעֵרוּב בָּא אֶצְלָם — אֵין צָרִיךְ לִתֵּן פַּת כְּלָל, כְּמוֹ הַבַּיִת שֶׁמַּנִּיחִים בּוֹ הָעֵרוּב שֶׁאֵין צָרִיךְ לִתֵּן הַפַּת, שֶׁכָּל אֵלּוּ הַבָּתִּים כְּבַיִת אֶחָד הֵם חֲשׁוּבִים, כֵּיוָן שֶׁאֲכִילַת בַּעֲלֵיהֶם הִיא בְּבַיִת אֶחָד בְּלִי מְחִצָּה מַפְסֶקֶת בֵּינֵיהֶם.

וַאֲפִלּוּ אִם לִפְעָמִים פּוֹרְסִין וִילוֹן לִפְנֵיהֶם לִצְנִיעוּת — אֵינוֹ חָשׁוּב מְחִצָּה לַחֲלוֹק בֵּינֵיהֶם, אֲפִלּוּ אִם מַגִּיעַ לַתִּקְרָה, הוֹאִיל וְאֵינוֹ פָּרוּס שָׁם תָּמִיד כָּל זְמַן שֶׁהֵם בַּבַּיִת, אֶלָּא כְּשֶׁרוֹצִין לַעֲשׂוֹת שָׁם דְּבַר צְנִיעוּת פּוֹרְסִין אוֹתוֹ וְאַחַר כָּךְ מְסַלְּקִין אוֹתוֹ. מַה שֶּׁאֵין כֵּן בַּחֲבוּרוֹת שֶׁשָּׁבְתוּ בִּטְרַקְלִין, שֶׁהַיְרִיעוֹת שֶׁפָּרְסוּ בֵּינֵיהֶן הֵן פְּרוּסוֹת שָׁם כָּל זְמַן שְׁבִיתָתָן שָׁם.

וּמִי שֶׁיֵּשׁ לוֹ שְׁתֵּי בָתִּים בִּשְׁתֵּי חֲצֵרוֹת, בְּאַחַת הוּא אוֹכֵל וּבַשְּׁנִיָּה הוּא דָר כָּל הַיּוֹם וְכָל הַלַּיְלָה — הֲרֵי זֶה אוֹסֵר בְּאוֹתָהּ שֶׁאוֹכֵל אִם אוֹכֵל שָׁם בְּשַׁבָּת, וְצָרִיךְ לְעָרֵב עִמָּהֶם. אֲבָל בַּשְּׁנִיָּה אֵינוֹ אוֹסֵר וְאֵין צָרִיךְ לְעָרֵב וּמֻתָּר לְהִשְׁתַּמֵּשׁ שָׁם בְּשַׁבָּת, לְפִי שֶׁנַּעֲשֶׂה שָׁם כְּאוֹרֵחַ אֶצְלָם:

סעיף ז הָאַחִים שֶׁיֵּשׁ לָהֶם בָּתִּים בַּחֲצַר אֲבִיהֶם וּמְקַבְּלִים פְּרָס מֵאֲבִיהֶם…

הָאַחִים שֶׁיֵּשׁ לָהֶם בָּתִּים בַּחֲצַר אֲבִיהֶם וּמְקַבְּלִים פְּרָס מֵאֲבִיהֶם וְאוֹכְלִים בְּבָתֵּיהֶם, בֵּין שֶׁהַפְּרָס הוּא לֶחֶם בֵּין שֶׁהוּא מָעוֹת לִקְנוֹת בּוֹ לֶחֶם, אִם יֵשׁ עוֹד דִּיּוּרִים בֶּחָצֵר וּמַנִּיחִים עֵרוּב כָּל הֶחָצֵר בְּאֶחָד מִדִּיּוּרִין הָהֵם — צְרִיכִים הָאַחִים עִם אֲבִיהֶם לִתֵּן כָּל אֶחָד עֵרוּבוֹ, שֶׁמִּתּוֹךְ שֶׁבָּתִּים אֵלּוּ שֶׁל הָאָב וּבָנָיו הֻזְקְקוּ לְעֵרוּב, צְרִיכִים לִתֵּן כָּל אֶחָד וְאֶחָד. אֲבָל אִם אֵין עִמָּהֶם דִּיּוּרִין בֶּחָצֵר, אֵין צָרִיךְ עֵרוּב כְּלָל, שֶׁאֵינָן אוֹסְרִים זֶה עַל זֶה, הוֹאִיל וְאוֹכְלִים פְּרַס אֲבִיהֶם — נַעֲשׂוּ כִּבְנֵי בֵיתוֹ. אוֹ אֲפִלּוּ יֵשׁ עִמָּהֶם דִּיּוּרִין, אֶלָּא שֶׁמַּנִּיחִים הָעֵרוּב בְּאֶחָד מִבָּתִּים אֵלּוּ שֶׁל הָאַחִים אוֹ בְּבֵית אֲבִיהֶם — אֵינָן צְרִיכִין לְעָרֵב כְּלָל, לְפִי שֶׁכֻּלָּם כְּבַיִת אֶחָד שֶׁמַּנִּיחִים בּוֹ הָעֵרוּב, שֶׁאֵין צָרִיךְ לִתֵּן אֶת הַפַּת.

וְכֵן בְּנָשִׁים הַמְקַבְּלוֹת פְּרָס מִבַּעֲלֵיהֶן, וּבַעֲבָדִים הַמְקַבְּלִים פְּרָס מֵאֲדוֹנֵיהֶם, וּבְתַלְמִידִים הַמְקַבְּלִים פְּרָס מֵרַבָּן:

סעיף ח עֲשָׂרָה בָתִּים זֶה לִפְנִים מִזֶּה, וְאֵין פָּתוּחַ מֵהֶם לֶחָצֵר אֶלָּא הַחִיצוֹן בִּלְבָד…

עֲשָׂרָה בָתִּים זֶה לִפְנִים מִזֶּה, וְאֵין פָּתוּחַ מֵהֶם לֶחָצֵר אֶלָּא הַחִיצוֹן בִּלְבָד, וְכֻלָּם עוֹבְרִים מִזֶּה לְזֶה וְיוֹצְאִין דֶּרֶךְ הַחִיצוֹן לֶחָצֵר — שְׁנַיִם הַפְּנִימִים לְבַדָּם צְרִיכִים לִתֵּן עֵרוּב, שֶׁהֵם לְבַדָּם תּוֹרַת בַּיִת עֲלֵיהֶם, וְאוֹסְרִים בֶּחָצֵר וְכֵן לְטַלְטֵל מִזֶּה לְזֶה וְכֵן מֵהֶם לְבָתִּים שֶׁלִּפְנֵיהֶם וְלֶחָצֵר. אֲבָל שְׁאָר הַבָּתִּים שֶׁלִּפְנֵיהֶם — תּוֹרַת חָצֵר עֲלֵיהֶם, לְפִי שֶׁכָּל אֶחָד מֵהֶם הוּא בֵּית שַׁעַר לְהַפְּנִימִי מִמֶּנּוּ, וְאִם כֵּן בַּיִת הַשְּׁלִישִׁי מֵהַפְּנִים הוּא בֵּית שַׁעַר לְב' הַפְּנִימִים, וְנִקְרָא בֵּית שַׁעַר שֶׁל רַבִּים, וְאֵינוֹ חָשׁוּב דִּירָה כֵּיוָן שֶׁרַבִּים עוֹבְרִים עָלָיו. אֲבָל בַּיִת הַשֵּׁנִי אֵינוֹ בֵּית שַׁעַר אֶלָּא לְבַיִת אֶחָד הַפְּנִימִי מִמֶּנּוּ, וּבֵית שַׁעַר שֶׁל בַּיִת אֶחָד אֵינוֹ מִתְבַּטֵּל מִתּוֹרַת בַּיִת:

סעיף ט הַמִּתְאָרֵחַ בֶּחָצֵר, אֲפִלּוּ נִתְאָרַח בְּבַיִת בִּפְנֵי עַצְמוֹ…

הַמִּתְאָרֵחַ בֶּחָצֵר, אֲפִלּוּ נִתְאָרַח בְּבַיִת בִּפְנֵי עַצְמוֹ וְאוֹכֵל וְיָשֵׁן שָׁם בְּבֵיתוֹ, אִם לֹא נִתְאָרַח דֶּרֶךְ קֶבַע אֶלָּא לִשְׁלֹשִׁים יוֹם אוֹ פָּחוֹת — אֵינוֹ אוֹסֵר עַל בְּנֵי הֶחָצֵר, וְהוּא וָהֵם מֻתָּרִים לְטַלְטֵל בֶּחָצֵר, בֵּין מִבֵּיתוֹ בֵּין מִבָּתֵּיהֶם. וַאֲפִלּוּ אִם הָאוֹרְחִים רַבִּים וּבַעַל הַבַּיִת אֶחָד — כֻּלָּם טְפֵלִים אֶצְלוֹ וְנַעֲשִׂים כְּאַנְשֵׁי בֵיתוֹ, אֲפִלּוּ אִם הַבָּתִּים שֶׁמִּתְאָרְחִים בָּהֶם אֵינָם שֶׁלּוֹ אֶלָּא שֶׁל נָכְרִים אוֹ שֶׁל יִשְׂרְאֵלִים שֶׁאֵינָם דָּרִים שָׁם.

וְיֵשׁ אוֹמְרִים שֶׁהָאוֹרֵחַ שֶׁמִּתְאָרֵחַ בְּבַיִת אוֹ חֶדֶר בִּפְנֵי עַצְמוֹ אֵינוֹ טָפֵל לְבַעַל הַבַּיִת לִהְיוֹת כְּאַנְשֵׁי בֵיתוֹ, אֶלָּא שֶׁכֹּל שֶׁלֹּא נִתְאָרַח בְּדֶרֶךְ קֶבַע אֵין לוֹ חֵלֶק בֶּחָצֵר, וְאֵינוֹ אוֹסֵר עַל בַּעַל הַבַּיִת לְטַלְטֵל מִבֵּיתוֹ לַחֲצֵרוֹ. אֲבָל לְטַלְטֵל מִבֵּית הָאוֹרֵחַ לֶחָצֵר — אָסוּר בֵּין לוֹ בֵּין לְבַעַל הַבַּיִת, שֶׁכֵּיוָן שֶׁהִשְׂכִּירוֹ לוֹ בַּיִת זֶה לְאֵיזֶה זְמַן, שְׂכִירוּת לְיוֹמוֹ מִמְכָּר הוּא וַהֲרֵי זֶה בַּיִת שֶׁלּוֹ, וּכְשֶׁמּוֹצִיא מִשָּׁם לַחֲצַר בַּעַל הַבַּיִת, הֲרֵי זֶה כְּמוֹצִיא מֵרְשׁוּת לִרְשׁוּת (וְהוּא הַדִּין אִם הִשְׁאִילוּהוּ לוֹ בְּעִנְיָן שֶׁאֵינָם יְכוֹלִים לְסַלְּקוֹ תּוֹךְ מֶשֶׁךְ זְמַן שְׁאֵלָתוֹ. וְאֵינוֹ דוֹמֶה לְאַנְשֵׁי חָצֵר שֶׁשָּׁכַח אֶחָד מֵהֶם וְלֹא עֵרֵב, שֶׁכְּשֶׁמְּבַטֵּל לָהֶם רְשׁוּת בֵּיתוֹ מֻתָּר גַּם הוּא לְהוֹצִיא מִבֵּיתוֹ לֶחָצֵר מִפְּנֵי שֶׁנַּעֲשֶׂה אֶצְלָם כְּאוֹרֵחַ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן ש״פ, לְפִי שֶׁהַמְבַטֵּל סִלֵּק עַצְמוֹ מֵרְשׁוּת בֵּיתוֹ וּנְתָנוֹ לָהֶם. מַה שֶּׁאֵין כֵּן אוֹרֵחַ שֶׁלֹּא בִטֵּל וְלֹא סִלֵּק עַצְמוֹ, וּבְנֵי הֶחָצֵר גַּם כֵּן אֵינָם יְכוֹלִים לְסַלְּקוֹ — הֲרֵי רְשׁוּת בֵּיתוֹ מְיֻחֶדֶת לוֹ וּרְשׁוּת חָצֵר מְיֻחֶדֶת לָהֶם, וְאָסוּר לְהוֹצִיא מֵרְשׁוּת זוֹ לְזוֹ עַד שֶׁיְּעָרְבוּ):

סעיף י אֲבָל לְאַחַר שְׁלֹשִׁים יוֹם אוֹסֵר עַל בְּנֵי הֶחָצֵר…

אֲבָל לְאַחַר שְׁלֹשִׁים יוֹם אוֹסֵר עַל בְּנֵי הֶחָצֵר אִם הוּא בְּעִנְיָן שֶׁאֵינָם יְכוֹלִים לְסַלְּקוֹ תּוֹךְ מֶשֶׁךְ זְמַן שְׂכִירוּתוֹ אוֹ שְׁאֵלָתוֹ.

וּמִכָּל מָקוֹם אֲפִלּוּ אִם קוֹבֵעַ לְיוֹתֵר מִשְּׁלֹשִׁים יוֹם, אֵינוֹ אוֹסֵר עַד לְאַחַר שְׁלֹשִׁים יוֹם, אֲבָל לֹא תּוֹךְ שְׁלֹשִׁים, אֲפִלּוּ אִם רָגִיל לָבֹא לְכַאן וְלִקְבּוֹעַ לְעוֹלָם לְיוֹתֵר מִשְּׁלֹשִׁים. בַּמֶּה דְּבָרִים אֲמוּרִים? בְּאוֹרֵחַ יִשְׂרָאֵל, אֲבָל אַכְסְנַאי נָכְרִי הָרָגִיל לִקְבּוֹעַ — אוֹסֵר מִיָּד, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שפ״ד. וְיֵשׁ אוֹמְרִים שֶׁגַּם יִשְׂרָאֵל הָרָגִיל לִקְבּוֹעַ אוֹסֵר מִיָּד. וּלְעִנְיַן הֲלָכָה, בְּדִבְרֵי סוֹפְרִים הַלֵּךְ אַחַר הַמֵּקֵל.

וְכָל זֶה כְּשֶׁיֵּשׁ בַּעַל הַבַּיִת אֶחָד קָבוּעַ בֶּחָצֵר, שֶׁאָז הָאוֹרְחִים שֶׁבְּבָתִּים אֲחֵרִים שֶׁבְּחָצֵר זוֹ בְּטֵלִים אֶצְלוֹ. אֲבָל אוֹרְחִים לְבַדָּם בְּיַחַד אוֹסְרִים זֶה עַל זֶה מִיָּד אִם יֵשׁ לְכָל אֶחָד חֶדֶר מְיֻחָד לַאֲכִילָה. וְאִם יֵשׁ נָכְרִי דָּר בֶּחָצֵר, צְרִיכִים לִשְׂכּוֹר מִמֶּנּוּ רְשׁוּתוֹ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שפ״ב.

וְכָל זֶה בְּמָקוֹם שֶׁאֵין שָׁם עֵרוּבֵי חֲצֵרוֹת שֶׁל רַבִּים. אֲבָל בְּמָקוֹם שֶׁהַקָּהָל מְעָרְבִים עֵרוּבֵי חֲצֵרוֹת בִּתְחִלַּת שָׁנָה לְכָל הַשָּׁנָה, אֵין הָאוֹרְחִים הַמִּתּוֹסְפִים שָׁם אוֹסְרִים עֲלֵיהֶם וְלֹא זֶה עַל זֶה, מִפְּנֵי שֶׁלֵּב בֵּית דִּין מַתְנֶה עֲלֵיהֶם שֶׁיְּהֵא לָהֶם חֵלֶק בָּעֵרוּב, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שס״ו:

מקור: ספריא — שולחן ערוך הרב, אורח חיים סימן ש״ע · 370. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה. הטקסט מועתק כפי שספריא מגישה אותו: בסעיף ב, בצירוף שפירושו לגמרי, יש וי״ו הנושאת גם קמץ וגם שוא; הדבר הושאר כמות שהוא.

לשון הרב מול סעיף המחבר והרמ״א

במה שונה לשון הרב מסעיף המחבר והרמ״א

כל נקודה מעמידה זה לצד זה סעיף מן השולחן ערוך של המחבר (במספור שלו, ח׳ סעיפים), עם הגהת הרמ״א כשיש, ואת הסעיף המקביל בשולחן ערוך אדמו״ר הזקן (במספור שלו, י׳ סעיפים), לפי סדר סעיפי הרב, ומציינת רק מה שההשוואה בין הלשונות מראה. בעיקרו של דבר החלוקות מקבילות כך (סעיף הרב ← סעיף המחבר): א ← א; ב ← ב; ג ו־ד ← הגהת הרמ״א על סעיף ב; ה ← ג; ו ← ד ותחילת ה; ז ← המשך ה, וסעיף ו; ח ← ז; ט ו־י ← ח והגהת הרמ״א. מה שחסר בלשון המחבר אינו חידוש בשל כך: בסימן זה כמה מן הפירושים שבלשון הרב כתובים כבר בבית יוסף, במגן אברהם או בט״ז. בכל נקודה מצוין היכן הם כתובים, ורמה 2 מביאה מקומות אלו בהרחבה; ופירוש שלא חיפשנו את מקורו אין מציגים אותו בשל כך כמיוחד לרב.

נקודה א — בית שער, סוכת החג, ומה נחשב דירה (שו״ע הרב ש״ע:א)

מה נחשב דירה

דברי המחבר: בסעיף א הוא כותב « הדר בבי' שער אכסדרה ומרפס' שבחצר אינו אוסר על בני החצר שאינם חשובים דירה אבל הדר בבי' התבן בבי' העצים בבי' הבקר ובבי' האוצרו' אוסר » (שו״ע או״ח ש״ע:א), בלי לכתוב טעם לבית התבן, לבית העצים, לבית הבקר ולבית האוצרות.

דברי הרב: לגבי בית שער הוא כותב « הַדָּר בְּבֵית שַׁעַר, אַף עַל פִּי שֶׁיֵּשׁ לוֹ ד' מְחִצּוֹת, כֵּיוָן שֶׁרַבִּים עוֹבְרִים דֶּרֶךְ שָׁם לְבָתֵּיהֶם — אֵין עָלָיו תּוֹרַת בַּיִת אֶלָּא תּוֹרַת חָצֵר » (שו״ע הרב ש״ע:א); הוא מוסיף « וְכֵן סֻכַּת הֶחָג בֶּחָג אֵינָהּ חֲשׁוּבָה דִירָה לֶאֱסוֹר » (שו״ע הרב ש״ע:א); ולגבי בית התבן וכיוצא בו הוא נותן את הטעם: « לְפִי שֶׁרְאוּיִים לָדוּר בָּהֶם וַחֲשׁוּבִים דִּירָה, אַף עַל פִּי שֶׁעֲשׂוּיִים מִקָּנִים וְכַיּוֹצֵא בָהֶם » (שו״ע הרב ש״ע:א).

מה שכתוב כבר לפניו: הבית יוסף מביא את דברי ה״ר יהונתן, על בית שער ואחר כך על הבית העשוי לתבן, « בית שער אע״פ שיש לו ד׳ מחיצות כיון דרבים עוברי׳ לבתיהם דרך שם אין עליה תורת בית אלא תורת חצר אבל אם הניח עירובי חצירות בבית שבחצר שהוא מיוחד לאצור שם תבן ובהמותיו הרי זה עירוב שהרי ראוי לדירה הוא ובית מיקרי והדר שם בבית העשוי לתבן והוא עשוי מענפי אילנות ומקנים כלומר שאינה בית של קבע אפ״ה בית מיקרי לכל דבר » (בית יוסף או״ח ש״ע) (רמה 2, פרק 2); על בית שער שבסעיף כותב המגן אברהם « אף על פי שיש לו ד' מחיצות כיון שעוברים דרך שם לא הוי דירה » (מגן אברהם ש״ע ס״ק א); ועל טעם הסעיף, שאינם חשובים דירה, הוא מוסיף « וה״ה סוכת החג בחג » (מגן אברהם ש״ע ס״ק ב).

ההבדל: המחבר כולל את בית שער, האכסדרה והמרפסת בטעם אחד: שאינם חשובים דירה. לגבי בית שער כותב הרב, בלשון ה״ר יהונתן שמביא הבית יוסף, שרבים עוברים דרכו לבתיהם ושאין עליו תורת בית אלא תורת חצר; המגן אברהם בס״ק א כותב שעוברים דרכו, ואינו אומר ״רבים״. הוא מוסיף את סוכת החג, שהמגן אברהם מוסיף בס״ק ב; וכותב מדוע בית התבן וכיוצא בו אוסרים, אף שהם עשויים מקנים — מה שכותב גם ה״ר יהונתן, ושהמשנה ברורה מביא אחריו (ס״ק ה, רמה 1). על בית שער של יחיד, שהמשנה ברורה מזכיר לצד בית שער של רבים, אין הרב אומר דבר בסעיף א, ודף זה אינו מכריע: המשנה ברורה, שכתב אחריו, אינו מחלק ביניהם, « לא שנא של רבים ולא שנא של יחיד » (משנה ברורה ש״ע ס״ק א); הבית יוסף מבאר את לשון הרמב״ם על בית שער שרבים דורסים בו: « שהרי בבית שער של יחיד ג״כ דורסין בו רבים כשבאים ליכנס לביתו » (בית יוסף או״ח ש״ע), וכך הוא לומד בדברי הטור, « וזה נראה שתופס רבינו עיקר שכתב כאן הדר בבית שער אכסדרה ומרפסת אינו אוסר ולא חילק בין בית שער דרבים לבית שער דיחיד » (בית יוסף או״ח ש״ע), ומסיים « וזהו כמו שכתבתי לדעת הרמב״ם והרי״ף והכי נקטינן » (בית יוסף או״ח ש״ע); המגן אברהם כותב על סעיף ז « אבל בית שער חצר דיחיד אינו אוסר כיון דלא עביד לדירה » (מגן אברהם ש״ע ס״ק י״א); והרב כותב בסעיף ח, על הבית שעוברים דרכו רק לבית פנימי אחד, « וּבֵית שַׁעַר שֶׁל בַּיִת אֶחָד אֵינוֹ מִתְבַּטֵּל מִתּוֹרַת בַּיִת » (שו״ע הרב ש״ע:ח).

השלכה מעשית: הדר בבית שער שרבים עוברים דרכו לבתיהם, אף שיש לו ארבע מחיצות, באכסדרה או במרפסת שבחצר, או בסוכת החג בחג — אינו אוסר על בני החצר; הדר בבית התבן, בבית הבקר, בבית האוצר או בבית העצים — אוסר עד שיערב (370:1). לבית שער של יחיד, שהרב אינו עוסק בו בסעיף א, ראה למעלה, והיעזר ברב שלך.

נקודה ב — תפיסת יד בדברים שאינם ניטלים מחמת כבדן (שו״ע הרב ש״ע:ב)

הכובד והאיסור

דברי המחבר: בסעיף ב, לגבי הבתים שבעל הבית השאילן או השכירן, הוא כותב « ויש לו בכל א' מהם דברים שאינם ניטלים בשב' מחמ' כבדן או מחמ' איסור שהם דברים שאסור לטלטלם אפי' לצורך מקומו אין הדרים בהם אוסרים עליו » (שו״ע או״ח ש״ע:ב): הכובד והאיסור עומדים זה לצד זה.

דברי הרב: הוא קובע את הדין בדברים האסורים בלבד, « וְאִם יֵשׁ לוֹ בְּכָל אֶחָד מֵהֶם דְּבָרִים שֶׁאֵינָם נִטָּלִים בְּשַׁבָּת מֵחֲמַת אִסּוּר, שֶׁאָסוּר לְטַלְטְלָם אֲפִלּוּ לְצֹרֶךְ גּוּפָם אוֹ מְקוֹמָם » (שו״ע הרב ש״ע:ב); הוא כותב שדברים המותרים בטלטול לצורך גופם או מקומם אינם נחשבים, « אֲבָל אִם אֵין לוֹ בָּהֶם אֶלָּא דְּבָרִים שֶׁמֻּתָּר לְטַלְטֵל לְצֹרֶךְ גּוּפָם אוֹ מְקוֹמָם — אֵין זוֹ חֲשׁוּבָה תְּפִיסַת יָד » (שו״ע הרב ש״ע:ב); ואת הכובד הוא מביא כדעה, « וְיֵשׁ אוֹמְרִים שֶׁאִם אֵינָן נִטָּלִין מֵחֲמַת כָּבְדָּן, חֲשׁוּבִין תְּפִיסַת יָד אַף עַל פִּי שֶׁמֻּתָּרִים בְּטִלְטוּל » (שו״ע הרב ש״ע:ב), ומוסיף בסוגריים « וְיֵשׁ לְהַחֲמִיר בְּזֶה » (שו״ע הרב ש״ע:ב).

מה שכתוב כבר לפניו: המחבר עצמו מעמיד את התנאי, « שהם דברים שאסור לטלטלם אפי' לצורך מקומו » (שו״ע או״ח ש״ע:ב); הרמב״ם מזכיר את האיסור בלבד, « בַּמֶּה דְּבָרִים אֲמוּרִים בְּשֶׁהִנִּיחַ שָׁם דָּבָר שֶׁאָסוּר לְטַלְטְלוֹ בְּשַׁבָּת כְּגוֹן טֶבֶל וַעֲשָׁשִׁיּוֹת » (רמב״ם הלכות עירובין פ״ד הי״ד), וממשיך « אֲבָל אִם נִשְׁאַר לוֹ בְּכָל בַּיִת מֵהֶן כֵּלִים שֶׁמֻּתָּר לְטַלְטְלָן הוֹאִיל וְאֶפְשָׁר שֶׁיּוֹצִיאֵם הַיּוֹם וְלֹא יִשָּׁאֵר לוֹ שָׁם תְּפִיסַת יָד הֲרֵי אֵלּוּ אוֹסְרִין עָלָיו » (רמב״ם הלכות עירובין פ״ד הי״ד); הבית יוסף מביא מן המגיד משנה « וכתב ה״ה בפ״ד שנ״ל שכל דבר שראוי לטלטלו לצורך מקומו בכלל דבר שניטל הוא » (בית יוסף או״ח ש״ע); על המלים « מחמ' כבדן » (שו״ע או״ח ש״ע:ב) כותב המגן אברהם « צ״ע דמה גבול יש לכובד זה דמשוי שקשה לא' נוח לב' ומ״מ כ' דוקא דברים המוקצים ע״ש ויש להחמיר » (מגן אברהם ש״ע ס״ק ג) (רמה 2, פרק 3). ולגבי ההשאלה, ספר עבודת הקודש לרשב״א, שמביא המשנה ברורה: « ודוקא כשהשאילן לזמן מרובה אבל אם השאילן לפי שעה אין אוסר עליו בכל ענין » (משנה ברורה ש״ע ס״ק ז) (רמה 1, פרק 3).

ההבדל: סעיף המחבר מזכיר את הכובד ואת האיסור זה לצד זה, « מחמ' כבדן או מחמ' איסור שהם דברים שאסור לטלטלם אפי' לצורך מקומו » (שו״ע או״ח ש״ע:ב); שדבר המותר בטלטול לצורך גופו או מקומו אינו נחשב — כבר כתבו זאת הרמב״ם והמגיד משנה, והמשנה ברורה, שכתב אחריו, לומד זאת ממילים אחרונות אלו: « דאם היה מותר לטלטלן לצורך גופן ומקומן אפשר שיצטרך בעה״ב להשתמש בהם בשבת או להוציאן מביתו מחמת איזה סיבה ולא ישאר לו שם תפיסת יד » (משנה ברורה ש״ע ס״ק ט). ובכובד אין המפרשים לומדים את הסעיף כאחד: המשנה ברורה כותב על הדבר הכבד « אף שאין נעשה מוקצה ע״י הכובד » (משנה ברורה ש״ע ס״ק ח), ושמחמת הטורח לפנותו אין דרך ליטלו בשבת; המגן אברהם מקשה מה גבול יש לכובד כזה, וכותב « ומ״מ כ' דוקא דברים המוקצים ע״ש ויש להחמיר » (מגן אברהם ש״ע ס״ק ג). בלשון הרב עיקר הדין נקבע באיסור בלבד; ושדבר שאינו ניטל מחמת כובדו חשוב תפיסת יד אף על פי שמותר בטלטול — הוא מביא כדעת יש אומרים, וכותב בסוגריים שיש להחמיר בזה. עוד מפרש הרב בסוגריים שההשאלה היא « לִזְמַן, בְּעִנְיָן שֶׁאֵינוֹ יָכוֹל לְסַלְּקָם תּוֹךְ הַזְּמַן בְּעַל כָּרְחָם » (שו״ע הרב ש״ע:ב); ספר עבודת הקודש, שמביא המשנה ברורה, מגביל את ההשאלה האוסרת להשאלה לזמן מרובה (ס״ק ז), והרב — להשאלה שאינו יכול לסלקם בה תוך הזמן.

השלכה מעשית: אם בעל הבית מניח בכל אחד מן הבתים שהשכיר או השאיל דברים שאסור לטלטלם אפילו לצורך גופם או מקומם, הדרים בהם אינם אוסרים עליו; דברים המותרים בטלטול לצורך גופם או מקומם אינם מועילים; ובדברים שאינם ניטלים מחמת כבדן כותב הרב « וְיֵשׁ לְהַחֲמִיר בְּזֶה » (שו״ע הרב ש״ע:ב). תפיסת יד מועילה רק בבית שקנוי לו, שכור לו או שאול לו (370:2).

נקודה ג — מלמד, סופר ותלמידים הדרים בבית בעל הבית (שו״ע הרב ש״ע:ה)

למה אינם אוסרים

דברי המחבר: בסוף סעיף ג הוא כותב « אבל מי שיש לו מלמד או סופר בביתו וכן תלמידים הלומדים בפני הרב ודרים בביתו כל א' בחדרו אפילו יש לכל א' פתח פתוח לחצר ואוכל וישן בחדרו אינם אוסרים » (שו״ע או״ח ש״ע:ג), בלי לכתוב את הטעם.

דברי הרב: הוא נותן לכך טעם, « לְפִי שֶׁאֵינוֹ מַשְׁאִיל לָהֶם רְשׁוּתוֹ לֶאֱסוֹר עָלָיו וְלֹא לֶאֱסוֹר זֶה עַל זֶה, וַהֲרֵי יָכוֹל לְסַלְּקָם בְּכָל עֵת » (שו״ע הרב ש״ע:ה); ואחר כך, בחדרים הקנויים להם או השכורים להם, או השאולים להם באופן שאינו יכול לסלקם, הוא כותב שהם « אוֹסְרִים עָלָיו, אֶלָּא אִם כֵּן יֵשׁ לוֹ בָּהֶם תְּפִיסַת יָד » (שו״ע הרב ש״ע:ה), ומוסיף « וַאֲפִלּוּ הֵם מִשְׁתַּמְּשִׁים בַּאֲפִיָּה וּבִשּׁוּל בְּבַיִת גָּדוֹל שֶׁל בַּעַל הַבַּיִת — אֵין זֶה מוֹעִיל כְּלוּם, כֵּיוָן שֶׁאוֹכְלִים בְּחַדְרֵיהֶם, אֶלָּא אִם כֵּן תַּלְמִידִים הַמְקַבְּלִים פְּרָס מֵרַבָּן » (שו״ע הרב ש״ע:ה).

מה שכתוב כבר לפניו: התוספות נותנים שני טעמים, והשני הוא « ועוד דאין משאיל להם רשותו לאסור עליו ולא זה על זה ומצי לסלוקינהו » (תוספות עירובין ע״ב ע״א), והבית יוסף מביא אותם (רמה 2, פרק 7). הט״ז נותן על סעיף זה שני טעמים, « הטעם כיון שכולם משתמשים בשל בע״ה באפייה ובישול ובכל דבר חשיבי כולהו כאלו אוכלים וישני' במקו' א' ואע״פ שיש להם פתח לצד ר״ה נקראים ע״ש בע״ה » (ט״ז או״ח ש״ע ס״ק ו), ו« ועוד דאין משאיל להם רשותו לאסור עליו כ״כ התוספות » (ט״ז או״ח ש״ע ס״ק ו); ואחר כך הוא מביא « וכתב ב״י שעיקר ההיתר בצירוף טעם השני דוק' ואין לסמוך להתיר משום טעם הראשון במקום שאין שם שייכות טעם השני » (ט״ז או״ח ש״ע ס״ק ו). המגן אברהם נותן את אותם שני הטעמים בס״ק ח, וכותב על חדרים הקנויים להם « ואם מקום חדרים קנוי להם או מושכר להם ומשתמשין כל א' בפ״ע אוסרין כמ״ש ס״ב » (מגן אברהם ש״ע ס״ק ח). הבית יוסף כבר כותב שטעמים אלו נצרכים רק להתיר כשאין לבעל הבית תפיסת יד במקום שהשאיל להם: « ודבר פשוט הוא דלא איצטריכינן להני טעמי אלא כדי להתיר כשאין לב״ה תפיסת יד במקום שהשאיל להם שאם היה לו שם תפיסת יד בלאו הני טעמא הוה שרי » (בית יוסף או״ח ש״ע) (רמה 2, פרק 7).

ההבדל: המחבר כותב את ההיתר בלי טעמו. הט״ז והמגן אברהם כותבים את הטעם השני בלי סיום התוספות, « ומצי לסלוקינהו » (תוספות עירובין ע״ב ע״א), וכך מעיר הביאור הלכה: « ולכאורה יפלא על הט״ז ומ״א שלא הזכירו בטעם השני רק משום דאין משאיל רשותו כדי לאסור עליו ולא סיימו משום דמצי לסלוקינהו » (ביאור הלכה ש״ע); הרב נותן כטעם רק את השני, ועם הסיום הזה: « וַהֲרֵי יָכוֹל לְסַלְּקָם בְּכָל עֵת » (שו״ע הרב ש״ע:ה). ועל הטעם הראשון, השימוש בבית הגדול לאפייה ובישול, הוא כותב שאינו מועיל כלום כשהחדרים קנויים להם, שכורים להם, או שאולים להם באופן שאינו יכול לסלקם, כיון שאוכלים בחדריהם. הביאור הלכה מבין כך את שיטתו, ומזהה אותה עם שיטת הרשב״א והריטב״א: « ודע דדעת הגר״ז דתלוי רק במצי לסלוקינהו והיכי דלא מצי לסלוקינהו משמע מיניה דצריך עירוב אפילו אופין ומבשלין ביחד והוא ממש כשיטת הרשב״א והריטב״א » (ביאור הלכה ש״ע); ומצדו הוא כותב ששני הטעמים נצרכים רק בהשאלה לזמן, « ונראה דעד כאן לא הצריכו תרתי אלא במקום דמשאיל לו לזמן דלא מצי לסלוקינהו בכל שעה » (ביאור הלכה ש״ע), ואחר כך « וכן מבואר בח״א דהיכי דאיכא תרתי טעמים אפילו השאילו לזמן ארוך שרי אבל היכי דמצי לסלוקינהו בכל שעה כגון שלא השאילו לזמן מיוחד לא יכול לאסור עליו אפילו אופה ומבשל בפני עצמו » (ביאור הלכה ש״ע): לפי החיי אדם, ההשאלה מותרת אפילו לזמן ארוך כשיש שני הטעמים; ובמקום שבעל הבית יכול לסלקם בכל עת, ממשיך אותו משפט (אחר הזכרת החיי אדם), אינם אוסרים אף כשכל אחד מבשל בפני עצמו. המשנה ברורה כותב, בלי חילוק זה, « או אפילו בשאול רק שמשתמשין כל אחד בפ״ע בחדרו אוסרין דדוקא בצירוף שני הטעמים ביחד מקילינן. וכ״ז בשלא היה לבעה״ב אצלם תפיסת יד דאל״ה מותר בכל גווני » (משנה ברורה ש״ע ס״ק ל״ג). שני מקרים מראים את ההבדל לעומת המשנה ברורה כפי שהוא כתוב: חדר שאול שבעל הבית יכול לחזור בו בכל עת, וכל אחד משתמש בפני עצמו — סעיף ה של הרב אינו מזכיר במקרה זה את הבישול, ולפי הבנת הביאור הלכה (« וכן משמע מהגר״ז », ביאור הלכה ש״ע) אינם אוסרים לדעתו, והמשנה ברורה כותב שאוסרים (הביאור הלכה עצמו מגביל דרישה זו להשאלה לזמן); חדר שאול לזמן, וכולם אופים ומבשלים בבית הגדול — הרב כותב שאין זה מועיל כלום, והביאור הלכה מביא מן החיי אדם ששני הטעמים יחד מתירים. בחדרים הקנויים להם או השכורים להם כותב המגן אברהם שהם אוסרים כשכל אחד משתמש בפני עצמו (ס״ק ח, המפנה לסעיף ב של המחבר); הרב כותב כן אף כשהם אופים ומבשלים בבית הגדול, וכך כתב כבר הבית יוסף: « ואם שנים או ג' אנשים דרים כל אחד בביתו המושכר או הקנוי לו אע״פ שכל הבתים פתוחים לטרקלין ומקום אפיית' ובישולם ותשמישם ביחד נראה דלכ״ע אוסרים זה ע״ז » (בית יוסף או״ח ש״ע) (רמה 2, פרק 7). הרב מזכיר את היוצא מן הכלל של תפיסת יד, שכבר כתבו הבית יוסף וכותבו גם המשנה ברורה (ראה לעיל); מצרף לחדרים אלו את השאלה שאינו יכול לחזור בה תוך הזמן — ולפי הביאור הלכה זו שיטת הרשב״א והריטב״א עצמה (ראה לעיל); ומוציא מן הכלל תלמידים המקבלים פרס מרבן, כפי שיבאר בסעיף ז: כשאין דיורים אחרים בחצר או שמניחים את עירוב החצר אצלם או בבית רבם. ולעניין הפתח, המחבר כותב « אפילו יש לכל א' פתח פתוח לחצר » (שו״ע או״ח ש״ע:ג), והרב « אַף עַל פִּי שֶׁיֵּשׁ לְכָל אֶחָד חֶדֶר מְיֻחָד פָּתוּחַ לֶחָצֵר וְגַם לִרְשׁוּת הָרַבִּים » (שו״ע הרב ש״ע:ה); הבית יוסף מביא שהנוסחאות חלוקות בזה, « אלא שבתוס' ושאר הספרים כתבו בין יש להם פתח לרשות הרבים בין אין להם פתח לרשות הרבים ובפסקי הרא״ש כתב לחצר במקום לר״ה ורבינו כתב כלשון הרא״ש » (בית יוסף או״ח ש״ע), וכותב שגם לדעת הרא״ש מותר אף כשיש לכל אחד פתח לרשות הרבים: « ונ״ל דלהרא״ש נמי אפי' יש לכל אחד פתח פתוח לר״ה שרי » (בית יוסף או״ח ש״ע).

השלכה מעשית: המלמד, הסופר או התלמידים הדרים בבית בעל הבית אינם אוסרים, אף שכל אחד אוכל וישן בחדרו, ואף שאין לבעל הבית בהם תפיסת יד, כיון שאינו משאיל להם רשותו לאסור עליו, והרי הוא יכול לסלקם בכל עת; אבל אם חדריהם קנויים להם, שכורים להם, או שאולים להם באופן שאינו יכול לסלקם תוך הזמן — הם אוסרים עליו, אלא אם כן יש לו בהם תפיסת יד, ואפילו הם אופים ומבשלים בביתו הגדול — חוץ מתלמידים המקבלים פרס מרבן, כשאין דיורים אחרים בחצר או שמניחים את עירוב החצר אצלם או בבית רבם (370:5, 7). בחדר השאול אין המשנה ברורה (ס״ק ל״ג) כותב חילוק זה, והביאור הלכה מגביל את דרישתו להשאלה לזמן (ראה לעיל): היעזר ברב שלך.

נקודה ד — האורח: שלשים יום, הרגיל לקבוע, ואורחים לבדם (שו״ע הרב ש״ע:י)

האורח

דברי המחבר: בסעיף ח הוא כותב « אם לא נתארח דרך קבע אלא לשלשים יום או פחות אינו אוסר על בני החצר » (שו״ע או״ח ש״ע:ח); והרמ״א מוסיף « ודוקא בדאיכא בעל הבית א' קבוע דאז האורחים בטלים לגביה אבל אורחים ביחד אוסרים זע״ז מיד » (רמ״א או״ח ש״ע:ח); ההגהה מפנה לסוף סימן שצ״א · 391, ואחר כך באה, בסוגריים מרובעים במהדורה המודפסת בספריא (מגיני ארץ), הפניה לסימן שפ״ד · 384 לגבי נכרי המתארח.

דברי הרב: בסעיף י הוא מפרש « וּמִכָּל מָקוֹם אֲפִלּוּ אִם קוֹבֵעַ לְיוֹתֵר מִשְּׁלֹשִׁים יוֹם, אֵינוֹ אוֹסֵר עַד לְאַחַר שְׁלֹשִׁים יוֹם, אֲבָל לֹא תּוֹךְ שְׁלֹשִׁים, אֲפִלּוּ אִם רָגִיל לָבֹא לְכַאן וְלִקְבּוֹעַ לְעוֹלָם לְיוֹתֵר מִשְּׁלֹשִׁים » (שו״ע הרב ש״ע:י); הוא מביא דעה, ואחר כך כותב את ההלכה: « וְיֵשׁ אוֹמְרִים שֶׁגַּם יִשְׂרָאֵל הָרָגִיל לִקְבּוֹעַ אוֹסֵר מִיָּד. וּלְעִנְיַן הֲלָכָה, בְּדִבְרֵי סוֹפְרִים הַלֵּךְ אַחַר הַמֵּקֵל » (שו״ע הרב ש״ע:י); לגבי אורחים לבדם הוא כותב « אֲבָל אוֹרְחִים לְבַדָּם בְּיַחַד אוֹסְרִים זֶה עַל זֶה מִיָּד אִם יֵשׁ לְכָל אֶחָד חֶדֶר מְיֻחָד לַאֲכִילָה » (שו״ע הרב ש״ע:י); ובמקום שהקהל מערבים עירובי חצרות בתחלת השנה לכל השנה הוא כותב « אֵין הָאוֹרְחִים הַמִּתּוֹסְפִים שָׁם אוֹסְרִים עֲלֵיהֶם וְלֹא זֶה עַל זֶה, מִפְּנֵי שֶׁלֵּב בֵּית דִּין מַתְנֶה עֲלֵיהֶם שֶׁיְּהֵא לָהֶם חֵלֶק בָּעֵרוּב » (שו״ע הרב ש״ע:י).

מה שכתוב כבר לפניו: המגן אברהם כותב על אורחים לבדם « ודוקא שיש לכל א' חדר מיוחד לאכילה כמ״ש ס״ג » (מגן אברהם ש״ע ס״ק י״ב), ולומד מדברי הרמ״א « משמע דס״ל לרמ״א דבישראל אפי' רגיל אין אוסר עססי' שצ״א » (מגן אברהם ש״ע ס״ק י״ב) — ההפניה לסוף סימן שצ״א · 391, שם יהודים עוברי דרך שנתאכסנו בחצר אחרת אינם אוסרים, « דאורח אינו אוסר » (שו״ע או״ח שצ״א:ב); הט״ז מדייק מן הסעיף « משמע ביותר מל' יום אוסר » (ט״ז או״ח ש״ע ס״ק י), וכותב « במקום שיש קהל ועשו עירובי חצירות בע״פ כדרכינו אז פטור גם בזה כיון שלב ב״ד מתנה על כל שיתוספו דיורים » (ט״ז או״ח ש״ע ס״ק י).

ההבדל: זה, בין השאר, מה שכתוב בלשון הרב ואינו כתוב במקום הזה בסעיף המחבר ובהגהת הרמ״א (ההגהה מפנה גם לסוף סימן שצ״א · 391, העוסק ביהודים עוברי דרך שנתאכסנו בחצר אחרת ואינם אוסרים משום שאורח אינו אוסר, ואינו מדבר לא במשך השהייה ולא בעירוב הקהל; אליו קושר המגן אברהם את מה שהוא לומד מדברי הרמ״א בישראל הרגיל): שאורח הקובע ליותר משלשים יום אינו אוסר אלא לאחר שלשים יום — הט״ז מדייק מן הסעיף שיותר משלשים יום אוסר, והרב כותב ממתי; והמשנה ברורה, שכתב אחריו, כותב « ומסתברא דאם קבע עצמו לכתחלה יותר משלשים אוסר מיד » (משנה ברורה ש״ע ס״ק נ״ה): במי שקובע עצמו מתחילה ליותר משלשים יום אין שני הספרים מסכימים; שלאחר שלשים יום הוא אוסר « אִם הוּא בְּעִנְיָן שֶׁאֵינָם יְכוֹלִים לְסַלְּקוֹ תּוֹךְ מֶשֶׁךְ זְמַן שְׂכִירוּתוֹ אוֹ שְׁאֵלָתוֹ » (שו״ע הרב ש״ע:י) — המשנה ברורה אומר שיותר משלשים יום הוא אוסר « אפילו אם היה רק בשאילה לבד » (משנה ברורה ש״ע ס״ק נ״ה), בלי תנאי זה, ולא חיפשנו לתנאי זה מקור שקדם לרב; הדעה שגם ישראל הרגיל לקבוע אוסר מיד, וההלכה ללכת אחר המקל בדברי סופרים — המגן אברהם כבר למד מדברי הרמ״א שישראל, אפילו רגיל, אינו אוסר; התנאי של חדר מיוחד לאכילה כשהאורחים לבדם, שהוא דברי המגן אברהם; והפטור במקום שהקהל מערבים בתחלת השנה לכל השנה, שהט״ז כותב אותו בעירוב שנעשה בערב פסח, כמנהג זמנו (« בע״פ כדרכינו », ט״ז או״ח ש״ע ס״ק י).

השלכה מעשית: אורח המתארח לשלשים יום או פחות אינו אוסר; לאחר שלשים יום הוא אוסר אם אין יכולים לסלקו תוך זמן שכירותו או שאלתו, ואפילו קבע ליותר, אינו אוסר קודם לכן (המשנה ברורה כותב בלשון « ומסתברא » (משנה ברורה ש״ע ס״ק נ״ה) שהקובע עצמו מתחילה ליותר משלשים יום אוסר מיד: היעזר ברב שלך); בישראל הרגיל לקבוע הולכים אחר המקל; אכסנאי נכרי הרגיל לקבוע אוסר מיד (סימן שפ״ד · 384); אורחים לבדם, שלכל אחד חדר מיוחד לאכילה, אוסרים זה על זה מיד; ובמקום שהקהל מערבים בתחלת השנה לכל השנה, האורחים המתוספים אינם אוסרים (370:10).

4 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן ש״ע · 370, ברגישות החב״דית, לפי סדר סעיפיו. מספרי הסעיפים הם לפי מספור הרב.

לחסיד חב״ד — סימן ש״ע · 370

  • ① מה נחשב דירה: הדר בבית שער שרבים עוברים דרכו לבתיהם, באכסדרה או במרפסת שבחצר, או בסוכת החג בחג — אינו אוסר על בני החצר; הדר בבית התבן, בבית הבקר, בבית האוצר או בבית העצים — אוסר עד שיערב, אף שהם עשויים מקנים. ראה פרק 2; לבית שער של יחיד, שהרב אינו עוסק בו בסעיף א, היעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן 370:1.
  • ② בתים מושכרים או שאולים, ותפיסת יד: בתים שבעל הבית השכיר, או השאיל לזמן באופן שאינו יכול לסלק את הדרים בהם תוך הזמן, אוסרים עליו עד שיערבו, אלא אם כן הוא מניח בכל אחד מהם דברים שאסור לטלטלם אפילו לצורך גופם או מקומם; בדברים שאינם ניטלים מחמת כבדן כותב הרב « וְיֵשׁ לְהַחֲמִיר בְּזֶה » (שו״ע הרב ש״ע:ב); תפיסת יד מועילה רק בבית שקנוי לו, שכור לו או שאול לו (370:2). אחר כך מביא הרב דעת יש אומרים, שתפיסת יד מועילה רק כשאין דיורים אחרים בחצר, או כשעירובם מונח בביתו או באחד הבתים שיש לו בהם תפיסת יד (370:3); ובבירה ששכר מן הנכרי ודר בה עם חבירו, תפיסת היד מועילה לשוכר רק אם שכרה מתחלה לעצמו (370:4). הפניה: שולחן ערוך אדמו״ר הזקן 370:2-4.
  • ③ טרקלין המחולק במחיצות: מחיצות המגיעות לתקרה או בתוך ג׳ טפחים לתקרה, אפילו של עראי, עושות כל חלק לחדר בפני עצמו, וכל חבורה נותנת עירובה כשעירוב החצר מונח אצל דיורים אחרים, או בחצר אחרת שעירבה עם זו; אם הוא מונח בטרקלין, או שאין דיורים אחרים, אין צריך, הואיל ואינן אלא מחיצות עראי; מחיצות שאינן מגיעות לתקרה — עירוב אחד לכולם; וכל זה, כותב הרב, « וְכָל זֶה כְּשֶׁעָשׂוּ מְחִצּוֹת עֲרַאי לְפִי שָׁעָה » (שו״ע הרב ש״ע:ה): חדרים ממש הפתוחים לחצר אוסרים זה על זה אפילו אין דיורים אחרים, וכל חבורה נותנת עירוב אפילו מניחים את העירוב בטרקלין; אם אין פתוח לחצר אלא החיצון וכולם עוברים דרך עליו, אין נותנים עירוב אלא שנים הפנימיים; ולגבי מלמד, סופר ותלמידים — ראה פרק 2. הפניה: שולחן ערוך אדמו״ר הזקן 370:5.
  • ④ מקום האכילה: אנשי החצר שכולם אוכלים בבית אחד בשבת, אפילו כל אחד משלו ועל שלחן בפני עצמו, אינם צריכים עירוב; עם חצר אחרת מוליכים פת אחד, ואיזה מהם שירצה נותנו, ואפילו בני ביתו של בעל הבית יכולים לתתו משלהם; וילון שפורסים לפעמים לצניעות אינו מחיצה, אפילו מגיע לתקרה; האוכל בחצר אחת ודר בחצר אחרת — אוסר במקום שאוכל, אם אוכל שם בשבת. הפניה: שולחן ערוך אדמו״ר הזקן 370:6.
  • ⑤ הפרס: אחים שיש להם בתים בחצר אביהם ומקבלים פרס מאביהם — לחם, או מעות לקנות בו לחם — ואוכלים בבתיהם, נותנים, הם ואביהם, כל אחד עירובו כשעירוב החצר מונח אצל אחד הדיורים האחרים; ואם אין דיורים אחרים, או שהעירוב מונח באחד מבתי האחים או בבית אביהם, אין צריך; וכן בנשים, בעבדים ובתלמידים המקבלים פרס. הפניה: שולחן ערוך אדמו״ר הזקן 370:7.
  • ⑥ בתים זה לפנים מזה: עשרה בתים זה לפנים מזה, ואין פתוח לחצר אלא החיצון — אין נותנים עירוב אלא שנים הפנימיים. הפניה: שולחן ערוך אדמו״ר הזקן 370:8.
  • ⑦ האורח: אורח המתארח לשלשים יום או פחות אינו אוסר, אפילו בבית בפני עצמו שאוכל וישן בו, ואורחים רבים טפלים לבעל בית אחד; הרב מביא גם את הדעה שהאורח המתארח בבית או בחדר בפני עצמו אינו טפל לבעל הבית, ושלפיה « אֲבָל לְטַלְטֵל מִבֵּית הָאוֹרֵחַ לֶחָצֵר — אָסוּר בֵּין לוֹ בֵּין לְבַעַל הַבַּיִת, שֶׁכֵּיוָן שֶׁהִשְׂכִּירוֹ לוֹ בַּיִת זֶה לְאֵיזֶה זְמַן, שְׂכִירוּת לְיוֹמוֹ מִמְכָּר הוּא וַהֲרֵי זֶה בַּיִת שֶׁלּוֹ » (שו״ע הרב ש״ע:ט); ומוסיף בסוגריים « וְהוּא הַדִּין אִם הִשְׁאִילוּהוּ לוֹ בְּעִנְיָן שֶׁאֵינָם יְכוֹלִים לְסַלְּקוֹ תּוֹךְ מֶשֶׁךְ זְמַן שְׁאֵלָתוֹ » (שו״ע הרב ש״ע:ט) (370:9). להמשך — לאחר שלשים יום, הרגיל לקבוע, אורחים לבדם, והעירוב שנעשה לכל השנה — ראה פרק 2 (370:10). הפניה: שולחן ערוך אדמו״ר הזקן 370:9-10.

⚠ פרק זה מציג את ההנהגה ההלכתית של חב״ד. לכל שאלה ממשית: היעזר ברב שלך.
