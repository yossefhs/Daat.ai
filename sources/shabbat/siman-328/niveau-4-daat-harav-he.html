דעת הרב

דעת הרב — שולחן ערוך אדמו״ר הזקן על סימן שכ״ח

שיטת אדמו״ר הזקן בנ״ה הסעיפים המקודדים את הקורפוס הגדול של חולה בשבת — דרגות החולה (מיחושים, חולה שאין בו סכנה, חולה שיש בו סכנה, פיקוח נפש), היתרים ואיסורים, יישום מעשי לתרופות, בדיקות וטיפולים.

מדוע רמה 4 ייעודית לאדמו״ר הזקן? שולחן ערוך אדמו״ר הזקן אינו פירוש על המחבר — אלא שולחן ערוך עצמאי ושלם, שכתב אדמו״ר הזקן. ייחודו: הוא משלב הלכה + טעמי המצוות + ממד פנימי בחיבור אחד, ומכריע ברמה תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון. עמוד זה מאסף עבור סימן שכ״ח את הטקסט המלא של הרב, חידושו והדברי הרבי המאירים את הסוגיה.

1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן שכ״ח

סימן שכ״ח — דִּין חוֹלֶה בְּשַׁבָּת — וּבוֹ נה סְעִיפִים
מקור: הוצאת קה״ת, כפי שמופיע בספריא. נ״ה סעיפים.

סעיף אדִּין חוֹלֶה בְּשַׁבָּת וּבוֹ נ״ה סְעִיפִים: מִי שֶׁיֵּשׁ לוֹ מִחוּשׁ בְּעָלְמָא…

דִּין חוֹלֶה בְּשַׁבָּת וּבוֹ נ״ה סְעִיפִים:

מִי שֶׁיֵּשׁ לוֹ מִחוּשׁ בְּעָלְמָא שֶׁאֵין בּוֹ חֲשַׁשׁ סַכָּנָה כְּלָל, וְהוּא מִתְחַזֵּק וְהוֹלֵךְ כְּבָרִיא — אָסוּר לַעֲשׂוֹת לוֹ שׁוּם רְפוּאָה בְּשַׁבָּת, אֲפִלּוּ עַל יְדֵי נָכְרִי. וַאֲפִלּוּ דָּבָר שֶׁאֵין בּוֹ מִשּׁוּם סְרַךְ מְלָאכָה אֲפִלּוּ מִדִּבְרֵי סוֹפְרִים, מִכָּל מָקוֹם כֵּיוָן שֶׁעוֹשֶׂה לִרְפוּאָה וְנִכָּר הַדָּבָר — אָסוּר, מִשּׁוּם גְּזֵרָה שֶׁמָּא יִשְׁחוֹק סַמְמָנִים, וְיִתְחַיֵּב מִשּׁוּם טוֹחֵן:

סעיף במִי שֶׁיֵּשׁ לוֹ חֹלִי שֶׁל סַכָּנָה — מִצְוָה לְחַלֵּל עָלָיו אֶת הַשַּׁבָּת…

מִי שֶׁיֵּשׁ לוֹ חֹלִי שֶׁל סַכָּנָה — מִצְוָה לְחַלֵּל עָלָיו אֶת הַשַּׁבָּת, וְהַזָּרִיז — הֲרֵי זֶה מְשֻׁבָּח. וְהַשּׁוֹאֵל — הֲרֵי זֶה שׁוֹפֵךְ דָּמִים. וַאֲפִלּוּ סְפֵק נְפָשׁוֹת דּוֹחֶה אֶת הַשַּׁבָּת, שֶׁנֶּאֱמַר "וָחַי בָּהֶם" — שֶׁיְּהֵא חַי וַדַּאי בַּעֲשִׂיַּת הַמִּצְווֹת. וּמִכָּל מָקוֹם, אֲפִלּוּ בְּוַדַּאי סַכָּנָה — אֵין מְחַלְּלִין אֶלָּא בִּרְפוּאָה יְדוּעָה לַכֹּל אוֹ עַל פִּי מֻמְחֶה:

סעיף גכָּל מַכָּה שֶׁל חָלָל — מְחַלְּלִין עָלֶיהָ אֶת הַשַּׁבָּת…

כָּל מַכָּה שֶׁל חָלָל, דְּהַיְנוּ בְּאֵיבָרִים הַפְּנִימִיִּים מֵהַשִּׁנַּיִם וְלִפְנִים, וְשִׁנַּיִם עַצְמָן בִּכְלָל — מְחַלְּלִין עָלֶיהָ אֶת הַשַּׁבָּת. וְהוּא שֶׁנִּתְקַלְקְלוּ הַשִּׁנַּיִם אוֹ אֶחָד מִשְּׁאָר אֵיבָרִים הַפְּנִימִיִּים מֵחֲמַת מַכָּה, אוֹ שֶׁיֵּשׁ בָּהֶם בּוּעָא וְכַיּוֹצֵא בָזֶה. אֲבָל מִחוּשִׁים בְּעָלְמָא אֵינָן נִקְרָאִים מַכָּה. וְאִם מִצְטַעֵר כָּל כָּךְ עַד שֶׁחָלָה מִמֶּנּוּ כָּל גּוּפוֹ — מֻתָּר לְחַלֵּל עַל יְדֵי נָכְרִי אֲפִלּוּ בִּמְלָאכָה גְמוּרָה:

סעיף דמַכָּה שֶׁל חָלָל אֵינָהּ צְרִיכָה אֹמֶד…

מַכָּה שֶׁל חָלָל אֵינָהּ צְרִיכָה אֹמֶד, אֶלָּא אֲפִלּוּ בִּסְתָם שֶׁאֵין שָׁם בְּקִיאִין וְגַם הַחוֹלֶה אֵינוֹ אוֹמֵר כְּלוּם — עוֹשִׂין לוֹ כָּל מַה שֶׁרְגִילִין לַעֲשׂוֹת לוֹ בְּחֹל. אֲבָל כְּשֶׁיּוֹדְעִים וּמַכִּירִים בְּאוֹתוֹ חֹלִי שֶׁהוּא מַמְתִּין — אָסוּר לְחַלֵּל עָלָיו אַף עַל פִּי שֶׁהִיא מַכָּה שֶׁל חָלָל:

סעיף המַכָּה שֶׁאֵינָהּ שֶׁל חָלָל — נִשְׁאָלִין בְּבָקִי וּבְחוֹלֶה…

מַכָּה שֶׁאֵינָהּ שֶׁל חָלָל — נִשְׁאָלִין בְּבָקִי וּבְחוֹלֶה, אִם יֹאמַר אֶחָד מֵהֶם שֶׁצָּרִיךְ לְחִלּוּל, אוֹ שֶׁיֹּאמַר שֶׁאִם לֹא יְחַלְּלוּ אֶפְשָׁר שֶׁיַּכְבִּיד עָלָיו חָלְיוֹ עַד לִידֵי סַכָּנָה — מְחַלְּלִין, וְאִם לָאו — אֵין מְחַלְּלִין:

סעיף ומַכָּה שֶׁעַל גַּב הַיָּד אוֹ עַל גַּב הָרֶגֶל…

מַכָּה שֶׁעַל גַּב הַיָּד אוֹ עַל גַּב הָרֶגֶל, וְכֵן מִי שֶׁבָּלַע עֲלוּקָה, וְכֵן מִי שֶׁנְּשָׁכוֹ כֶּלֶב שׁוֹטֶה אוֹ אֶחָד מִזּוֹחֲלֵי עָפָר הַמְּמִיתִים, אֲפִלּוּ סָפֵק אִם מֵמִית — הֲרֵי הֵם כְּמַכָּה שֶׁל חָלָל:

סעיף זוְכֵן מְחַלְּלִין עַל כָּל מַכָּה שֶׁנַּעֲשֵׂית מֵחֲמַת בַּרְזֶל…

וְכֵן מְחַלְּלִין עַל כָּל מַכָּה שֶׁנַּעֲשֵׂית מֵחֲמַת בַּרְזֶל, אֲפִלּוּ עַל הַבָּשָׂר מִבַּחוּץ. וְכֵן מְחַלְּלִין עַל שְׁחִין הַבָּא בְּפִי הַטַּבַּעַת, וְעַל סִמְטָא, וְעַל מִי שֶׁיֵּשׁ בּוֹ קַדַּחַת חַם בְּיוֹתֵר, אוֹ קַדַּחַת עִם קְרִירוּת:

סעיף חמִי שֶׁאֲחָזוֹ דָם — מַקִּיזִין אוֹתוֹ…

מִי שֶׁאֲחָזוֹ דָם (וְהוּא בְּעִנְיָן שֶׁיֵּשׁ בּוֹ חֲשַׁשׁ סַכָּנָה) — מַקִּיזִין אוֹתוֹ, וַאֲפִלּוּ הוֹלֵךְ עַל רַגְלָיו, וַאֲפִלּוּ הוּא יוֹם הָרִאשׁוֹן שֶׁאֲחָזוֹ הַדָּם:

סעיף טהַחוֹשֵׁשׁ בְּעֵינָיו…

הַחוֹשֵׁשׁ בְּעֵינָיו, אוֹ בְּעֵינוֹ אַחַת וְיֵשׁ בָּהּ צִיר, אוֹ שֶׁשּׁוֹתֵת מִמֶּנָּה דְּמָעוֹת מֵחֲמַת הַכְּאֵב, אוֹ שֶׁשּׁוֹתֵת דָּם, אוֹ שֶׁיֵּשׁ בָּהּ רִיר תָּמִיד, אוֹ שֶׁהוּא שׂוֹרֵף וְקוֹדֵחַ — מְחַלְּלִין בִּתְחִלַּת הַחֹלִי, מִפְּנֵי שֶׁמְּאוֹר הָעַיִן תָּלוּי בַּלֵּב:

סעיף יכָּל חֹלִי שֶׁהָרוֹפְאִים אוֹמְרִים שֶׁהוּא סַכָּנָה…

כָּל חֹלִי שֶׁהָרוֹפְאִים אוֹמְרִים שֶׁהוּא סַכָּנָה — מְחַלְּלִין עָלָיו אֶת הַשַּׁבָּת. וְאִם רוֹפֵא אֶחָד אוֹמֵר צָרִיךְ וְרוֹפֵא אֶחָד אוֹמֵר אֵין צָרִיךְ — מְחַלְּלִין, שֶׁכָּל סְפֵק נְפָשׁוֹת לְהָקֵל. וַאֲפִלּוּ אֵין שָׁם רוֹפֵא מֻמְחֶה — מְחַלְּלִין עַל פִּיו, כִּי כָּל בְּנִי אָדָם חֲשׁוּבִין מֻמְחִין קְצָת. וְכָל זֶה בְּיִשְׂרָאֵל, אֲבָל סְתָם נָכְרִים שֶׁאֵינָן רוֹפְאִים — אֵין מַחֲזִיקִים אוֹתָם כִּבְקִיאִים:

סעיף יאאִם לֹא רָצָה הַחוֹלֶה לְקַבֵּל הַתְּרוּפָה — כּוֹפִין אוֹתוֹ…

אִם לֹא רָצָה הַחוֹלֶה לְקַבֵּל הַתְּרוּפָה שֶׁאֵינוֹ רוֹצֶה שֶׁיְּחַלְּלוּ עָלָיו שַׁבָּת — כּוֹפִין אוֹתוֹ, שֶׁהִיא חֲסִידוּת שֶׁל שְׁטוּת:

סעיף יבחוֹלֶה אוֹמֵר צָרִיךְ אֲנִי לִתְרוּפָה פְלוֹנִית…

חוֹלֶה אוֹמֵר צָרִיךְ אֲנִי לִתְרוּפָה פְלוֹנִית, וְרוֹפֵא אוֹמֵר אֵין צָרִיךְ — שׁוֹמְעִין לַחוֹלֶה, כִּי "לֵב יוֹדֵעַ מָרַת נַפְשׁוֹ". וְאִם הָרוֹפֵא אוֹמֵר שֶׁאוֹתָהּ הַתְּרוּפָה מַזִּיק לוֹ — שׁוֹמְעִין לָרוֹפֵא:

סעיף יגחוֹלֶה שֶׁיֵּשׁ בּוֹ סַכָּנָה שֶׁאֲמָדוּהוּ בְּיוֹם שַׁבָּת…

חוֹלֶה שֶׁיֵּשׁ בּוֹ סַכָּנָה שֶׁאֲמָדוּהוּ בְּיוֹם שַׁבָּת שֶׁצָּרִיךְ לַעֲשׂוֹת לוֹ רְפוּאָה יְדוּעָה שֶׁיֵּשׁ בָּהּ מְלֶאכֶת חִלּוּל שַׁבָּת שְׁמוֹנָה יָמִים — אֵין אוֹמְרִים נַמְתִּין עַד הַלַּיְלָה, אֶלָּא יַעֲשׂוּ מִיָּד אַף עַל פִּי שֶׁמְּחַלְּלִין עָלָיו שְׁתֵּי שַׁבָּתוֹת. וּמִצְוָה מִן הַמֻּבְחָר לְהִשְׁתַּדֵּל לַעֲשׂוֹת הַכֹּל עַל יְדֵי יִשְׂרְאֵלִים גְּדוֹלִים בְּחָכְמָה, וְלֹא עַל יְדֵי הֶדְיוֹטוֹת וְנָשִׁים, כְּדֵי שֶׁלֹּא תְהֵא שַׁבָּת קַלָּה בְּעֵינֵיהֶם:

סעיף ידכָּל הַזָּרִיז לְחַלֵּל שַׁבָּת בְּדָבָר שֶׁיֵּשׁ בּוֹ סַכָּנָה…

כָּל הַזָּרִיז לְחַלֵּל שַׁבָּת בְּדָבָר שֶׁיֵּשׁ בּוֹ סַכָּנָה — הֲרֵי זֶה מְשֻׁבָּח, אֲפִלּוּ אִם מְתַקֵּן עִמּוֹ דָּבָר אַחֵר, כְּגוֹן שֶׁפֵּרַשׂ מְצוּדָה לְהַעֲלוֹת תִּינוֹק שֶׁנָּפַל לַנָּהָר וְצָד עִמּוֹ דָּגִים:

סעיף טונִנְעֲלָה דֶלֶת בִּפְנֵי תִינוֹק — שׁוֹבֵר הַדֶּלֶת…

נִנְעֲלָה דֶלֶת בִּפְנֵי תִינוֹק — שׁוֹבֵר הַדֶּלֶת וּמוֹצִיאוֹ, שֶׁמָּא יִבָּעֵת הַתִּינוֹק וְיָמוּת:

סעיף טזהָיָה חוֹלֶה שֶׁיֵּשׁ בּוֹ סַכָּנָה צָרִיךְ בָּשָׂר — שׁוֹחֲטִין לוֹ…

הָיָה חוֹלֶה שֶׁיֵּשׁ בּוֹ סַכָּנָה צָרִיךְ בָּשָׂר — שׁוֹחֲטִין לוֹ, וְאֵין אוֹמְרִים נַאֲכִילֶנּוּ נְבֵלָה. וּמִכָּל מָקוֹם, אִם הַחוֹלֶה צָרִיךְ לַאֲכִילָה לְאַלְתַּר וְהַנְּבֵלָה מוּכֶנֶת מִיָּד וְהַשְּׁחוּטָה מִתְאַחֶרֶת לוֹ — מַאֲכִילִין אוֹתוֹ הַנְּבֵלָה:

סעיף יזאֲמָדוּהוּ הָרוֹפְאִים שֶׁצָּרִיךְ גְּרוֹגֶרֶת אַחַת…

אֲמָדוּהוּ הָרוֹפְאִים שֶׁצָּרִיךְ גְּרוֹגֶרֶת אַחַת, וְרָצוּ עֲשָׂרָה וְהֵבִיאוּ לוֹ כָּל אֶחָד גְּרוֹגֶרֶת — כֻּלָּן פְּטוּרִין, וְיֵשׁ לָהֶם שָׂכָר מֵאֵת ה', אֲפִלּוּ הִבְרִיא בָּרִאשׁוֹנָה:

סעיף יחאֲמָדוּהוּ לִשְׁתֵּי גְרוֹגְרוֹת…

אֲמָדוּהוּ לִשְׁתֵּי גְרוֹגְרוֹת, וְלֹא מָצְאוּ אֶלָּא שְׁתֵּי גְרוֹגְרוֹת מְחֻבָּרוֹת כָּל אַחַת בְּעֹקֶץ אֶחָד בִּפְנֵי עַצְמוֹ, וְעוֹד שָׁלֹשׁ גְרוֹגְרוֹת מְחֻבָּרוֹת שְׁלָשְׁתָּן בְּעֹקֶץ אֶחָד — כּוֹרֵת הָעֹקֶץ שֶׁיֵּשׁ בּוֹ שָׁלֹשׁ. וְאִם הַדָּבָר בָּהוּל — אֵין מְדַקְדְּקִים בְּכָךְ:

סעיף יטחוֹלֶה שֶׁנָּפַל מֵחֲמַת חָלְיוֹ לְמִשְׁכָּב וְאֵין בּוֹ סַכָּנָה…

חוֹלֶה שֶׁנָּפַל מֵחֲמַת חָלְיוֹ לְמִשְׁכָּב וְאֵין בּוֹ סַכָּנָה, אוֹ שֶׁיֵּשׁ לוֹ מִחוּשׁ שֶׁמִּצְטַעֵר כָּל כָּךְ עַד שֶׁנֶּחֱלַשׁ מִמֶּנּוּ כָּל גּוּפוֹ — הֲרֵי צְרָכָיו נַעֲשִׂין עַל יְדֵי נָכְרִי אֲפִלּוּ בִּמְלָאכוֹת גְּמוּרוֹת שֶׁל תּוֹרָה. אֲבָל אֵין מְחַלְּלִין עָלָיו שַׁבָּת עַל יְדֵי יִשְׂרָאֵל בְּאִסּוּר שֶׁל תּוֹרָה אֲפִלּוּ יֵשׁ בּוֹ סַכָּנַת אֵבֶר. וּלְחַלֵּל עָלָיו יִשְׂרָאֵל בְּאִסּוּרֵי דִּבְרֵי סוֹפְרִים — מֻתָּר בְּלֹא שׁוּם שִׁנּוּי אִם יֵשׁ בּוֹ סַכָּנַת אֵבֶר, וּבְשִׁנּוּי אִם אֵין בּוֹ סַכָּנַת אֵבֶר:

סעיף כוְאִם לֹא נָפַל לְמִשְׁכָּב…

וְאִם לֹא נָפַל לְמִשְׁכָּב, וְגַם אֵינוֹ מִצְטַעֵר כָּל כָּךְ עַד שֶׁנֶּחֱלַשׁ כָּל גּוּפוֹ (אֲבָל עַל כָּל פָּנִים יֵשׁ לוֹ צַעַר גָּדוֹל) — מֻתָּר לַעֲשׂוֹת לוֹ עַל יְדֵי יִשְׂרָאֵל כָּל שְׁבוּת הַנַּעֲשֶׂה בְּשִׁנּוּי. אֲבָל אִם אֵין לוֹ אֲפִלּוּ מִקְצָת חֹלִי, אֶלָּא מִחוּשׁ בְּעָלְמָא — אָסוּר לַעֲשׂוֹת לוֹ אֲפִלּוּ עַל יְדֵי נָכְרִי שׁוּם דָּבָר הַנִּכָּר שֶׁהוּא לִרְפוּאָה:

סעיף כאכָּל שֶׁאָסוּר לַעֲשׂוֹת עַל יְדֵי יִשְׂרָאֵל — אָסוּר אֲפִלּוּ עַל יְדֵי הַחוֹלֶה בְּעַצְמוֹ…

כָּל שֶׁאָסוּר לַעֲשׂוֹת עַל יְדֵי יִשְׂרָאֵל — אָסוּר אֲפִלּוּ עַל יְדֵי הַחוֹלֶה בְּעַצְמוֹ. אֲבָל כְּשֶׁעוֹשֶׂה לוֹ הַנָּכְרִי — מֻתָּר לַחוֹלֶה לְסַיְּעוֹ קְצָת, שֶׁמְּסַיֵּעַ אֵין בּוֹ מַמָּשׁ:

סעיף כבמֻתָּר לוֹמַר לְנָכְרִי לַעֲשׂוֹת תַּבְשִׁיל לְקָטָן…

מֻתָּר לוֹמַר לְנָכְרִי לַעֲשׂוֹת תַּבְשִׁיל לְקָטָן שֶׁאֵין לוֹ מַה לֶּאֱכוֹל, שֶׁסְּתָם צָרְכֵי קָטָן דִּינָם כְּצָרְכֵי חוֹלֶה שֶׁאֵין בּוֹ סַכָּנָה:

סעיף כגלֹא הִתִּירוּ אֲמִירָה לְנָכְרִי בִּשְׁבִיל חוֹלֶה אֶלָּא כְּשֶׁצָּרִיךְ לָהּ בְּשַׁבָּת…

לֹא הִתִּירוּ אֲמִירָה לְנָכְרִי בִּשְׁבִיל חוֹלֶה אֶלָּא כְּשֶׁצָּרִיךְ לָהּ בְּשַׁבָּת, אֲבָל לֹא לְצֹרֶךְ מוֹצָאֵי שַׁבָּת:

סעיף כדהִקִּיז דָּם וְנִצְטַנֵּן — סַכָּנָה הִיא…

הִקִּיז דָּם וְנִצְטַנֵּן — סַכָּנָה הִיא, וְעוֹשִׂין לוֹ מְדוּרָה בְּשַׁבָּת אֲפִלּוּ בִּתְקוּפַת תַּמּוּז. אֲבָל לְחוֹלֶה אַחֵר, אֵין עוֹשִׂין לוֹ מְדוּרָה אֶלָּא עַל יְדֵי נָכְרִים, אֶלָּא אִם כֵּן אֵין שָׁם בְּגָדִים לְחַמְּמוֹ בָּהֶם:

סעיף כהאֵין נוֹתְנִין יַיִן לְתוֹךְ הָעַיִן…

אֵין נוֹתְנִין יַיִן לְתוֹךְ הָעַיִן, מִפְּנֵי שֶׁנִּכָּר הַדָּבָר שֶׁמִּתְכַּוֵּן לִרְפוּאָה. אֲבָל מֻתָּר לִתְּנוֹ עַל גַּבֵּי הָעַיִן, מִפְּנֵי שֶׁאֵינוֹ נִרְאֶה אֶלָּא כְּרוֹחֵץ. וְהוּא שֶׁאֵינוֹ פּוֹתֵחַ וְסוֹגֵר הָעַיִן:

סעיף כורֹק תָּפֵל…

רֹק תָּפֵל, וְהוּא חָזָק וּמְרַפֵּא — אָסוּר לִתְּנוֹ אֲפִלּוּ עַל גַּבֵּי הָעַיִן, שֶׁאֵין דֶּרֶךְ לִרְחוֹץ בְּרֹק. וְאִם רוֹחֵץ פִּיו בְּמַיִם וְאַחַר כָּךְ מַעֲבִירָם עַל עֵינָיו — מֻתָּר:

סעיף כזשׁוֹרֶה אָדָם קִילוֹרִין מֵעֶרֶב שַׁבָּת…

שׁוֹרֶה אָדָם קִילוֹרִין רַכִּין וּצְלוּלִין מֵעֶרֶב שַׁבָּת וְנוֹתֵן עַל גַּבֵּי הָעַיִן בְּשַׁבָּת לִרְפוּאָה, וְאֵין גּוֹזְרִים מִשּׁוּם שְׁחִיקַת סַמָּנִים, שֶׁכֵּיוָן שֶׁהִצְרִיכוּהוּ לִשְׁרוֹתָן מֵעֶרֶב שַׁבָּת — יֵשׁ לוֹ הֶכֵּר בָּזֶה:

סעיף כחמַעֲבִירִין גֶּלְדֵי מַכָּה וְסָכִין בַּשֶּׁמֶן…

מַעֲבִירִין גֶּלְדֵי מַכָּה וְסָכִין בַּשֶּׁמֶן, בִּמְקוֹמוֹת שֶׁדַּרְכָּן לָסוּךְ אַף שֶׁלֹּא לִרְפוּאָה. אֲבָל לֹא יָסוּךְ בַּחֵלֶב, מִפְּנֵי שֶׁנִּמּוֹחַ. וַאֲפִלּוּ בִּגְמַר מַכָּה — מֻתָּר לְהַעֲבִיר גִּלְדֶיהָ וּלְסוּכָהּ לְתַעֲנוּג בְּעָלְמָא:

סעיף כטמֻתָּר לִתֵּן עַל גַּבֵּי הַמַּכָּה סְפוֹג…

מֻתָּר לִתֵּן עַל גַּבֵּי הַמַּכָּה סְפוֹג וַחֲתִיכוֹת בְּגָדִים יְבֵשִׁים אִם הֵם חֲדָשִׁים, מִפְּנֵי שֶׁאֵינָן לִרְפוּאָה אֶלָּא כְּדֵי שֶׁלֹּא יְסָרְטוּ בְּגָדָיו אֶת הַמַּכָּה. אֲבָל לֹא חֲתִיכוֹת בְּגָדִים יְשָׁנִים שֶׁהֵן מְרַפְּאוֹת:

סעיף לנוֹתְנִין עָלֶה עַל גַּבֵּי הַמַּכָּה בְּשַׁבָּת…

נוֹתְנִין עָלֶה עַל גַּבֵּי הַמַּכָּה בְּשַׁבָּת, שֶׁאֵינוֹ אֶלָּא כִּמְשַׁמְּרָהּ, חוּץ מֵעֲלֵי גְפָנִים שֶׁהֵם לִרְפוּאָה. אֲבָל רְטִיָּה שֶׁהֱסִירָהּ מֵעַל גַּבֵּי הַמַּכָּה בְּמֵזִיד — אָסוּר לְהַחֲזִירָהּ, גְּזֵרָה שֶׁמָּא יְמָרֵחַ. וְאִם הֻחְלְקָה מֵעַל גַּבֵּי הַמַּכָּה מֵאֵלֶיהָ — לֹא גָזְרוּ בָּזֶה, וּמֻתָּר לְהַחֲזִירָהּ:

סעיף לאמַכָּה שֶׁנִּתְרַפְּאָה — מֻתָּר לִתֵּן עָלֶיהָ רְטִיָּה…

מַכָּה שֶׁנִּתְרַפְּאָה — מֻתָּר לִתֵּן עָלֶיהָ רְטִיָּה בַּתְּחִלָּה בְּשַׁבָּת, מִפְּנֵי שֶׁאֵינוֹ אֶלָּא כִּמְשַׁמְּרָהּ:

סעיף לבהַמֵּפִיס מֻרְסָא בְּשַׁבָּת…

הַמֵּפִיס מֻרְסָא בְּשַׁבָּת, אִם כְּדֵי לַעֲשׂוֹת לָהּ פֶּה בִּשְׁבִיל שֶׁיִּכָּנֵס בָּהּ הָאֲוִיר לִרְפוּאָה — הֲרֵי זֶה חַיָּב מִשּׁוּם "מַכֶּה בַּפַּטִּישׁ". וְאִם הֱפִיסָהּ כְּדֵי לְהוֹצִיא מִמֶּנָּה הַלֵּחָה הַמְצַעַרְתּוֹ בִּלְבַד — מֻתָּר:

סעיף לגאֲבָל אָסוּר לָחוּךְ שְׁחִין…

אֲבָל אָסוּר לָחוּךְ שְׁחִין, שֶׁמּוֹצִיא דָם, וְיֵשׁ בּוֹ מִשּׁוּם חוֹבֵל:

סעיף לדנֶקֶב שֶׁבַּמַּכָּה שֶׁכְּבָר נִפְתַּח…

נֶקֶב שֶׁבַּמַּכָּה שֶׁכְּבָר נִפְתַּח וּבָא לְהַרְחִיבוֹ בְּשַׁבָּת, אֲפִלּוּ כָּל שֶׁהוּא — אָסוּר. וְאִם יוֹדֵעַ שֶׁמּוֹצִיא דָם כְּשֶׁמְּקַנְּחָהּ — לֹא יְקַנְחָהּ בְּשַׁבָּת:

סעיף להמִי שֶׁנִּגְּפָה יָדוֹ אוֹ רַגְלוֹ…

מִי שֶׁנִּגְּפָה יָדוֹ אוֹ רַגְלוֹ שֶׁלֹּא מֵחֲמַת בַּרְזֶל — צוֹמְתָהּ בְּיַיִן כְּדֵי לְהַעֲמִיד הַדָּם, אֲבָל לֹא בְחֹמֶץ, מִפְּנֵי שֶׁהוּא חָזָק וְיֵשׁ בּוֹ מִשּׁוּם רְפוּאָה. וְאִם הוּא מְעֻנָּג — אַף הַיַּיִן לוֹ כְּמוֹ הַחֹמֶץ:

סעיף לומִי שֶׁנִּשְׁמַט פֶּרֶק יָדוֹ…

מִי שֶׁנִּשְׁמַט פֶּרֶק יָדוֹ אוֹ רַגְלוֹ — לֹא יְשַׁפְשְׁפֶנּוּ הַרְבֵּה בְּצוֹנֵן, שֶׁזֶּהוּ רְפוּאָתוֹ, אֶלָּא רוֹחֵץ כְּדַרְכּוֹ, וְאִם נִתְרַפֵּא — נִתְרַפֵּא:

סעיף לזצִפֹּרֶן שֶׁפֵּרְשָׁה וְצִיצִין…

צִפֹּרֶן שֶׁפֵּרְשָׁה, וְצִיצִין שֶׁהֵם כְּמִין רְצוּעוֹת דַּקּוֹת שֶׁפֵּרְשׁוּ מֵעוֹר הָאֶצְבַּע סְבִיב הַצִּפֹּרֶן, אִם פֵּרְשׁוּ רֻבָּן — אֵין בָּהֶם מִשּׁוּם גּוֹזֵז מִן הַתּוֹרָה. אֲבָל לְהָסִיר בְּיָד מֻתָּר אֲפִלּוּ לְכַתְּחִלָּה, אִם הֵן מְצַעֲרוֹת אוֹתוֹ:

סעיף לחהַחוֹשֵׁשׁ בְּשִׁנָּיו…

הַחוֹשֵׁשׁ בְּשִׁנָּיו — לֹא יְגַמֵּעַ בָּהֶן חֹמֶץ וְיִפְלֹט, שֶׁנִּכָּר הוּא שֶׁהוּא לִרְפוּאָה, אֲבָל מְגַמֵּעַ וּבוֹלֵעַ:

סעיף לטהַחוֹשֵׁשׁ בִּגְרוֹנוֹ…

הַחוֹשֵׁשׁ בִּגְרוֹנוֹ — לֹא יְעָרְעֶנּוּ בַּשֶּׁמֶן, שֶׁנִּכָּר שֶׁמְּכַוֵּן לִרְפוּאָה, אֲבָל בּוֹלֵעַ הוּא שֶׁמֶן:

סעיף מהַגּוֹנֵחַ מִכְּאֵב לֵב…

הַגּוֹנֵחַ מִכְּאֵב לֵב, שֶׁרְפוּאָתוֹ לִינַק בְּפִיו מִן הַבְּהֵמָה — מֻתָּר לִינַק בְּשַׁבָּת:

סעיף מאלֹא תַקִּיל אִשָּׁה חָלָב מִדַּדֶּיהָ…

לֹא תַקִּיל אִשָּׁה חָלָב מִדַּדֶּיהָ לְתוֹךְ הַכּוֹס אוֹ לְתוֹךְ הַקְּדֵרָה לְהָנִיק אֶת בְּנָהּ. אֲבָל מֻתָּר לְאִשָּׁה לְקַלֵּחַ מֵהֶחָלָב בִּכְדֵי שֶׁיֶּאֱחוֹז הַתִּינוֹק אֶת הַדַּד וְיִנוֹק:

סעיף מבאֵין לוֹעֲסִין מַצְטְכִי…

אֵין לוֹעֲסִין מִין שְׂרָף הַנִּקְרָא מַצְטְכִי, וְלֹא שָׁפִין אֶת הַשִּׁנַּיִם בְּסַם, בִּזְמַן שֶׁמִּתְכַּוֵּן לִרְפוּאָה. אֲבָל מִשּׁוּם רֵיחַ הַפֶּה — מֻתָּר:

סעיף מגכָּל אֳכָלִין וּמַשְׁקִין שֶׁהֵם מַאֲכַל בְּרִיאִים…

כָּל אֳכָלִין וּמַשְׁקִין שֶׁהֵם מַאֲכַל בְּרִיאִים — מֻתָּר לְאָכְלָן וְלִשְׁתּוֹתָן לִרְפוּאָה. וְכָל שֶׁאֵינוֹ מַאֲכַל וּמַשְׁקֵה בְּרִיאִים — אָסוּר לְאָכְלוֹ וְלִשְׁתּוֹתוֹ לִרְפוּאָה, גְּזֵרָה מִשּׁוּם שְׁחִיקַת סַמָּנִים:

סעיף מדאֵין עוֹשִׂים אַפִּיקְטְוִזִין…

אֵין עוֹשִׂים אַפִּיקְטְוִזִין, דְּהַיְנוּ גְּרָמַת קִיא, שֶׁלֹּא לְצֹרֶךְ רְפוּאָה, אֲפִלּוּ בְּחֹל, מִשּׁוּם הֶפְסֵד אֳכָלִין. וְאִם מִצְטַעֵר מֵרֹב הַמַּאֲכָל — בְּחֹל מֻתָּר אֲפִלּוּ בְּסַם, וּבְשַׁבָּת אָסוּר בְּסַם, אֲבָל מֻתָּר לְהַכְנִיס יָדוֹ לְתוֹךְ גְּרוֹנוֹ עַד שֶׁיָּקִיא:

סעיף מההַחוֹשֵׁשׁ בְּמֵעָיו…

הַחוֹשֵׁשׁ בְּמֵעָיו — מֻתָּר לִכְפּוֹת עֲלֵיהֶם כּוֹס שֶׁעֵרוּ מִמֶּנּוּ חַמִּין, אַף עַל פִּי שֶׁעֲדַיִן יֵשׁ בּוֹ הֶבֶל. וְכֵן מֻתָּר לְהַעֲלוֹת אָזְנַיִם וּלְהַעֲלוֹת אֻנְקְלִי:

סעיף מומִי שֶׁנִּשְׁתַּכֵּר…

וְכֵן מִי שֶׁנִּשְׁתַּכֵּר, שֶׁרְפוּאָתוֹ לָסוּךְ כַּפּוֹת יָדָיו וְרַגְלָיו בְּשֶׁמֶן וּמֶלַח — מֻתָּר לְסוּכָם בְּשַׁבָּת:

סעיף מזאָסוּר לְשַׁפְשֵׁף עַל הַגּוּף בְּכֹחַ…

אָסוּר לְשַׁפְשֵׁף עַל הַגּוּף בְּכֹחַ, וְאֵין צָרִיךְ לוֹמַר כְּדֵי שֶׁיִּיגַע וְיַזִּיעַ. וְאָסוּר לִדְחוֹק כְּרֵסוֹ שֶׁל תִּינוֹק כְּדֵי לְהוֹצִיא הָרְעִי:

סעיף מחרוֹחֲצִין לִרְפוּאָה בְּמֵי גְרָר…

רוֹחֲצִין לִרְפוּאָה בְּמֵי גְרָר וּבְמֵי חַמְתָן וּבְמֵי טְבֶרְיָא וּבַמַּיִם הַיָּפִים שֶׁבַּיָּם הַגָּדוֹל, אַף עַל פִּי שֶׁהֵם מְלוּחִים. אֲבָל לֹא בַּמַּיִם הָרָעִים שֶׁבַּיָּם הַגָּדוֹל וְלֹא בְּמֵי מִשְׁרֶה, שֶׁהֵם מְאוּסִים. וְאִם אֵינוֹ שׁוֹהֶה — מֻתָּר, שֶׁאֵינוֹ נִרְאֶה אֶלָּא כְּמֵקֵר:

סעיף מטאֵין רוֹחֲצִין בְּמַיִם הַמְשַׁלְשְׁלִים…

אֵין רוֹחֲצִין בְּמַיִם הַמְשַׁלְשְׁלִים, וְלֹא בְּטִיט שֶׁטּוֹבְעִין בּוֹ. וְאֵין שׁוֹתִין מַשְׁקֶה הַמְשַׁלְשֵׁל, מִפְּנֵי שֶׁכָּל אֵלּוּ צַעַר הֵם, וְכָתוּב "וְקָרָאתָ לַשַּׁבָּת עֹנֶג":

סעיף נלוֹחֲשִׁין עַל נְחָשִׁים וְעַקְרַבִּים…

לוֹחֲשִׁין עַל נְחָשִׁים וְעַקְרַבִּים בִּשְׁבִיל שֶׁלֹּא יַזִּיקוּ, וְאֵין בְּכָךְ מִשּׁוּם צֵידָה:

סעיף נאנוֹתְנִין כְּלִי עַל גַּבֵּי הָעַיִן לְהָקֵר…

נוֹתְנִין כְּלִי עַל גַּבֵּי הָעַיִן לְהָקֵר, וְהוּא שֶׁיְּהֵא כְּלִי הַמֻּתָּר בְּטִלְטוּל:

סעיף נבעֶצֶם שֶׁיָּצָא מִמְּקוֹמוֹ…

עֶצֶם שֶׁיָּצָא מִמְּקוֹמוֹ — אָסוּר לְהַחֲזִירוֹ, מִפְּנֵי שֶׁדּוֹמֶה לְבוֹנֶה. אֲבָל עֶצֶם שֶׁנִּשְׁבַּר — מַחֲזִירִים אוֹתוֹ לִמְקוֹמוֹ, שֶׁאִם לֹא יַחֲזִירוּהוּ בְּשַׁבָּת — יֵשׁ בָּזֶה סַכָּנָה לְאוֹתוֹ אֵבֶר:

סעיף נגאָסוּר לְהַנִּיחַ בֶּגֶד עַל מַכָּה שֶׁיּוֹצֵא מִמֶּנָּה דָּם…

אָסוּר לְהַנִּיחַ בֶּגֶד עַל מַכָּה שֶׁיּוֹצֵא מִמֶּנָּה דָּם, מִפְּנֵי שֶׁהַדָּם יִצְבַּע אוֹתוֹ. אֶלָּא כּוֹרֵךְ עָלֶיהָ קוּרֵי עַכָּבִישׁ וּמְכַסֶּה בָהֶם כָּל הַדָּם וְכָל הַחַבּוּרָה, וְאַחַר כָּךְ כּוֹרֵךְ עָלֶיהָ סְמַרְטוּט:

סעיף נדאָסוּר לִמְצוֹץ בְּפִיו דָּם מֵהַמַּכָּה…

אָסוּר לִמְצוֹץ בְּפִיו דָּם מֵהַמַּכָּה, מִשּׁוּם חוֹבֵל. וְאָסוּר לְהַנִּיחַ עַל הַמַּכָּה דָּבָר הַמּוֹשֵׁךְ דָּם וְלֵחָה, מִשּׁוּם מְפָרֵק:

סעיף נהאָסוּר לָשׂוּם פְּתִילָה בְּפִי הַטַּבַּעַת…

אָסוּר לָשׂוּם פְּתִילָה בְּפִי הַטַּבַּעַת — אֶלָּא אִם כֵּן יָשִׂים אוֹתָהּ בְּשִׁנּוּי, שֶׁיֹּאחֲזֶנָּה בִּשְׁתֵּי אֶצְבְּעוֹתָיו וְיַנִּיחֶנָּה בְּנַחַת. אֲבָל אָסוּר לַעֲשׂוֹת קריסטי״ר אֲפִלוּ עַל יְדֵי שִׁנּוּי, אִם לֹא בְּחוֹלֶה. וְאִם אֶפְשָׁר לַעֲשׂוֹת עַל יְדֵי נָכְרִי — יַעֲשֶׂה עַל יְדֵי נָכְרִי:

מקור: ספריא — שולחן ערוך הרב, אורח חיים שכ״ח.

2 · חידוש אדמו״ר הזקן

חידושים מיוחדים של הרב

חידושים המייחדים את אדמו״ר הזקן בסימן שכ״ח. כל חידוש לפי המבנה: שיטה קלאסית → חידוש הרב → השלכה מעשית.

חידוש א — מבנה ארבע הדרגות (שכ״ח:1-5)

מערכת ארבע דרגות החולה

שיטה קלאסית: —

חידוש הרב: אדמו״ר הזקן בונה מערכת מובנית של ארבע דרגות המאפשרת לסווג כל מקרה. כלי לימודי מרכזי.

השלכה מעשית: מבחן לכל מקרה — לזהות את הדרגה כדי ליישם את ההיתרים הנכונים.

חידוש ב — פיקוח נפש קודם (שכ״ח:31-45)

זריזות וזריזות

שיטה קלאסית: —

חידוש הרב: אדמו״ר הזקן מדגיש: לפיקוח נפש, לא רק שמחללים שבת — צריך לעשות זאת בזריזות. בלי היסוס.

השלכה מעשית: למצבי חירום, לפעול בלי להתלבט. עיכוב למחשבה = חילול נוסף.

3 · דברי הרבי

שיחות, מאמרים ואגרות הרבי

✓ הפניה מאומתת
שיחת הרבי — לא לקבוע ניתוח סמוך לשבת. הרבי מורה שאסור לקבוע ניתוח שאינו דחוף ליום שישי — ואף לא ליום חמישי או רביעי — שכן האשפוז יגרור חילולי שבת: אין להכניס עצמו מדעת למצב שבו ייאלץ לחלל שבת — היבט הנוגע ישירות לסימן שכ״ח (טיפול רפואי סמוך לשבת ובשבת). (ליקוטי שיחות חט״ז, ע' 518 ואילך; שיחת יום ב' דחג השבועות תשל״ח.)
← Chabad.org — Surgery and Medical Procedures Prior to Shabbos (שיחת הרבי)
אין הפניה מאומתת נוספת נכון לעכשיו.
מלבד המקור המאומת לעיל, אנו מייחסים לרבי אך ורק מקורות ראשוניים הניתנים לאימות (מכתב, שיחה או פסק מזוהה, עם קישור); לא ניתן היה לאמת מול Sefaria או Chabad.org דבר מקור נוסף של הרבי המתייחס ישירות לסימן זה.
כיוון חיפוש מועדף לפסקי הרבי לפי סימן: שערי הלכה ומנהג (מהדורת קה״ת). כל הפניה שתאומת תתווסף עם קישורה.
4 · למעשה

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן שכ״ח, ברגישות חב״דית.

לחסיד חב״ד — סימן שכ״ח

  • ① ארבע דרגות: מיחוש / לא חמור / חמור / פיקוח נפש. חידוש א של הרב. הפניה: שולחן ערוך אדמו״ר הזקן שכ״ח:1-5.
  • ② אין תרופות למיחוש קל (גזירה). הפניה: שולחן ערוך אדמו״ר הזקן שכ״ח:6-15.
  • ③ חולה שאין בו סכנה: שבותים מותרים (נכרי, שינוי). הפניה: שולחן ערוך אדמו״ר הזקן שכ״ח:16-30.
  • ④ חולה שיש בו סכנה: הכל מותר בזריזות. חידוש ב של הרב. הפניה: שולחן ערוך אדמו״ר הזקן שכ״ח:31-45.
  • ⑤ הכן תרופות החסיד לפני שבת. הפניה: שולחן ערוך אדמו״ר הזקן שכ״ח:46-55.

⚠ פסקה זו מציגה את ההנהגה ההלכתית של חב״ד. לכל שאלה מעשית: היוועץ ברבך.
