דעת הרב — שולחן ערוך אדמו"ר הזקן על סימן של"ד

שיטת אדמו"ר הזקן על כ"ט סעיפי דיני דליקה בשבת — גזירת ההצלה (שלא להציל יותר מצורך השבת), חריגות (פיקוח נפש, ספר תורה, מזון היום), יישום בני זמננו למכבי-אש.

למה רמה 4 ייחודית לאדמו"ר הזקן? השולחן ערוך של אדמו"ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתב אדמו"ר הזקן. ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחומרה תלמודית ייחודית.

לחב"ד, אדמו"ר הזקן הוא הפוסק האחרון. דף זה מאחד, לסימן של"ד, את הטקסט המלא של הרב, את ייחודו, ואת דברי הרבי המאירים את הנושא.

שולחן ערוך הרב — סימן של"ד

סימן של"ד — דִּינֵי דְלֵקָה בְּשַׁבָּת — וּבוֹ כט סְעִיפִים
מקור: מהדורת קה"ת, כפי שמופיעה בספריא. 29 סעיפים + 0 ערכי קונטרס אחרון.

סעיף אדִּינֵי דְלֵקָה בְּשַׁבָּת — נָפְלָה דְלֵקָה בְּשַׁבָּת...

דִּינֵי דְלֵקָה בְּשַׁבָּת וּבוֹ כ"ט סְעִיפִים:

נָפְלָה דְלֵקָה בְּשַׁבָּת — גָּזְרוּ חֲכָמִים שֶׁלֹּא יַצִּילוּ כְּלוּם מִפָּנֶיהָ, אֲפִלּוּ לְמָקוֹם הַמֻּתָּר בְּטִלְטוּל, וַחֲפָצִים הַמֻּתָּרִים בְּטִלְטוּל, וַאֲפִלּוּ אֹכָלִים וּמַשְׁקִים לֹא יַצִּיל אֶלָּא מַה שֶּׁצָּרִיךְ לוֹ לְאוֹתָהּ שַׁבָּת בִּלְבָד, כְּגוֹן אִם נָפְלָה בְּלֵיל שַׁבָּת קֹדֶם הַסְּעֻדָּה — מַצִּיל מְזוֹן ג' סְעֻדּוֹת, הָרָאוּי לְאָדָם — לְאָדָם, הָרָאוּי לִבְהֵמָה — לִבְהֵמָה, וּבְשַׁחֲרִית — מְזוֹן שְׁתֵּי סְעֻדּוֹת, וּבַמִּנְחָה — מְזוֹן סְעֻדָּה אַחַת, שֶׁאִם יִהְיֶה מֻתָּר לְהַצִּיל כָּל חֲפָצָיו — יֵשׁ לָחוּשׁ שֶׁמֵּחֲמַת טִרְדַּת הַהַצָּלָה שֶׁהוּא בָּהוּל וְנֶחְפָּז לְהַצִּיל יִשְׁכַּח שֶׁהַיּוֹם שַׁבָּת וְיָבֹא לְכַבּוֹת הַדְּלֵקָה:

סעיף בבַּמֶּה דְּבָרִים אֲמוּרִים? בַּבַּיִת שֶׁהַדְּלֵקָה בּוֹ...

בַּמֶּה דְּבָרִים אֲמוּרִים? בַּבַּיִת שֶׁהַדְּלֵקָה בּוֹ, אֲבָל בָּתִּים הַקְּרוֹבִים וִירֵאִים שֶׁתַּגִּיעַ לָהֶם הַדְּלֵקָה — יְכוֹלִים לְהַצִּיל כָּל מַה שֶּׁיִּרְצוּ, לְפִי שֶׁהֵם אֵינָן בְּהוּלִין כָּל כָּךְ שֶׁיִּשְׁכְּחוּ הַשַּׁבָּת מֵחֲמַת טִרְדַּת הַהַצָּלָה.

וַאֲפִלּוּ מָעוֹת אוֹ שְׁאָר דְּבָרִים הַמֻּקְצִים הַחֲשׁוּבִים — מֻתָּר לָהֶם לְהַצִּיל, וְהוּא הַדִּין שֶׁמֻּתָּר לְטַלְטְלָם כְּדֵי לְהַצִּילָם מִפְּנֵי הַלִּסְטִים אוֹ, מִפְּנֵי עַבְדֵי הַמּוֹשֵׁל הַבָּאִים לְחַפֵּשׂ בְּתֵבָתוֹ וְכַיּוֹצֵא בְּעִנְיָן זֶה, לְפִי שֶׁלֹּא גָזְרוּ כְּלָל עַל טִלְטוּל מֻקְצֶה בִּמְקוֹם הֶפְסֵד מְרֻבֶּה, אֶלָּא אִם כֵּן אֶפְשָׁר לְהַצִּיל בְּעִנְיָן אַחֵר.

אֲבָל יֵשׁ אוֹסְרִים טִלְטוּל מֻקְצֶה וַאֲפִלּוּ בִּמְקוֹם הֶפְסֵד מְרֻבֶּה וְאִי אֶפְשָׁר לְהַצִּיל בְּעִנְיָן אַחֵר. וְכֵן עִקָּר.

וְאַף עַל פִּי כֵן יֵשׁ אוֹמְרִים שֶׁמֻּתָּר לְהַצִּיל מָעוֹת וְכַיּוֹצֵא בָהֶם מִפְּנֵי הַדְּלֵקָה אוֹ מִפְּנֵי הַלִּסְטִים, לְפִי שֶׁאָדָם בָּהוּל עַל מָמוֹנוֹ וּבְוַדַּאי יַצִּילֵם מֵהֶפְסֵד, וְאִם לֹא נַתִּיר לוֹ לְהַצִּיל עַל יְדֵי טִלְטוּל מֻקְצֶה יָבֹא לִידֵי אִסּוּר חָמוּר שֶׁיְּכַבֶּה הַדְּלֵקָה.

סעיף גהִצִּיל פַּת נְקִיָּה לְשָׁלֹשׁ סְעֻדּוֹת — לֹא יַחֲזִיר וְיַצִּיל פַּת הַדְרָאָה...

הִצִּיל פַּת נְקִיָּה לְשָׁלֹשׁ סְעֻדּוֹת — לֹא יַחֲזִיר וְיַצִּיל פַּת הַדְרָאָה. אֲבָל אִם הִצִּיל פַּת הַדְרָאָה — יָכוֹל לַחֲזוֹר וּלְהַצִּיל פַּת נְקִיָּה, כִּי יָכוֹל לְהַעֲרִים וְלוֹמַר בְּפַת נְקִיָּה אֲנִי חָפֵץ. וְכֵן יָכוֹל לְהַעֲרִים לְהַצִּיל שָׁלֹשׁ סְעֻדּוֹת מִכָּל מִין. וַאֲפִלּוּ יֵשׁ לוֹ מַה יֹּאכַל בְּמָקוֹם אַחֵר כָּל הַשַּׁבָּת — רַשַּׁאי לְהַצִּיל ג' סְעֻדּוֹת מֵהַדְּלֵקָה.

סעיף דמַצִּילִין מִיּוֹם הַכִּפּוּרִים לְשַׁבָּת...

מַצִּילִין מִיּוֹם הַכִּפּוּרִים לְשַׁבָּת, כְּגוֹן אִם חָל יוֹם הַכִּפּוּרִים בְּעֶרֶב שַׁבָּת. וְאַף שֶׁעַכְשָׁו לְפִי הַקְּבִיעוּת שֶׁבְּיָדֵינוּ אִי אֶפְשָׁר שֶׁיִּהְיֶה יוֹם הַכִּפּוּרִים בְּעֶרֶב שַׁבָּת, מִכָּל מָקוֹם גַּם אִם חָל בְּה' וְיוֹדֵעַ שֶׁלֹּא יִמְצָא לִקְנוֹת בְּעֶרֶב שַׁבָּת — מַצִּילִין בּוֹ ג' סְעֻדּוֹת לְשַׁבָּת. אֲבָל אֵין מַצִּילִין מִשַּׁבָּת לְיוֹם הַכִּפּוּרִים וְיוֹם טוֹב, וְלֹא לְשַׁבָּת הַבָּאָה. אֲבָל בְּיוֹם הַכִּפּוּרִים לְמוֹצָאֵי יוֹם הַכִּפּוּרִים — מַצִּילִין מְזוֹן סְעֻדָּה אַחַת, מִפְּנֵי עַגְמַת נֶפֶשׁ:

סעיף המַצִּילִין לְחוֹלֶה וּלְזָקֵן וּלְרַעַבְתָן כְּבֵינוֹנִי:

מַצִּילִין לְחוֹלֶה וּלְזָקֵן וּלְרַעַבְתָן כְּבֵינוֹנִי:

סעיף ולֹא אָסְרוּ לְהַצִּיל יוֹתֵר מִג' סְעֻדּוֹת אֶלָּא כְּשֶׁמּוֹצִיא וְחוֹזֵר...

לֹא אָסְרוּ לְהַצִּיל יוֹתֵר מִג' סְעֻדּוֹת אֶלָּא כְּשֶׁמּוֹצִיא וְחוֹזֵר וּמוֹצִיא, אֲבָל כְּשֶׁמּוֹצִיא הַכֹּל בְּפַעַם אַחַת — מֻתָּר. וְהוּא שֶׁלֹּא יִהְיוּ בְּכֵלִים הַרְבֵּה, אֶלָּא בִּכְלִי אֶחָד. לְפִיכָךְ, מַצִּילִין סַל מָלֵא כִּכָּרוֹת אַף עַל פִּי שֶׁיֵּשׁ בּוֹ מֵאָה סְעֻדּוֹת, וְעִגּוּל שֶׁל דְּבֵלָה, וְחָבִית שֶׁל יַיִן.

סעיף זכְּשֵׁם שֶׁהוּא אֵינוֹ רַשַּׁאי לְהַצִּיל יוֹתֵר מִג' סְעֻדּוֹת...

כְּשֵׁם שֶׁהוּא אֵינוֹ רַשַּׁאי לְהַצִּיל יוֹתֵר מִג' סְעֻדּוֹת — כָּךְ אֲחֵרִים אֵינָם רַשָּׁאִים לְהַצִּיל לוֹ. אֲבָל מֻתָּר לוֹ לוֹמַר לָהֶם "בֹּאוּ וְהַצִּילוּ לָכֶם" אַף עַל פִּי שֶׁיּוֹדֵעַ בָּהֶם שֶׁיַּחֲזִירוּ לוֹ, וּבִלְבָד שֶׁלֹּא יַצִּילוּ כָּל אֶחָד יוֹתֵר מִג' סְעֻדּוֹת בִּכְלִי אֶחָד.

סעיף חמֻתָּר לוֹ לְהַצִּיל כְּלֵי תַשְׁמִישׁוֹ הַצְּרִיכִים לוֹ הַיּוֹם...

מֻתָּר לוֹ לְהַצִּיל כְּלֵי תַשְׁמִישׁוֹ הַצְּרִיכִים לוֹ הַיּוֹם (לְג' סְעֻדּוֹת הַלָּלוּ), כְּגוֹן כּוֹסוֹת וְקִיתוֹנִיּוֹת:

סעיף טכָּל הַצָּלָה שֶׁאָמַרְנוּ אֵינָהּ אֶלָּא לְחָצֵר אַחֶרֶת הַמְעֹרֶבֶת...

כָּל הַצָּלָה שֶׁאָמַרְנוּ אֵינָהּ אֶלָּא לְחָצֵר אַחֶרֶת הַמְעֹרֶבֶת עִם חָצֵר זוֹ, אֲבָל לֹא לְשֶׁאֵינָהּ מְעֹרֶבֶת, וְאֵין צָרִיךְ לוֹמַר לְכַרְמְלִית, שֶׁלֹּא הִתִּירוּ אִסּוּר דִּבְרֵי סוֹפְרִים בִּשְׁבִיל ג' סְעֻדּוֹת שֶׁל שַׁבָּת, אֶלָּא הַהַצָּלָה בִּלְבַדָּהּ:

סעיף יוְכָל זֶה בִּמְזוֹנוֹת וּכְלֵי תַשְׁמִישׁוֹ הַנִּטָּלִים בְּיָד...

וְכָל זֶה בִּמְזוֹנוֹת וּכְלֵי תַשְׁמִישׁוֹ הַנִּטָּלִים בְּיָד, אֲבָל בְּגָדִים שֶׁיָּכוֹל לְלָבְשָׁם — מֻתָּר לוֹ לִלְבּוֹשׁ וְלַעֲטוֹף כָּל מַה שֶּׁיָּכוֹל לַעֲטוֹף וּמוֹצִיא וּפוֹשֵׁט, וְחוֹזֵר וְלוֹבֵשׁ וּמוֹצִיא, אֲפִלּוּ כָּל הַיּוֹם כֻּלּוֹ.

סעיף יא-יזהצלת בית מעורב, כתבי הקודש, אזכרה, פ"ה אותיות...

סעיפים יא-יז: דין הצלה לבית אחר מעורב; דין כל כתבי הקודש האידנא; דין מגילת אסתר; דין קמיעות; דין ליקוט פ"ה אותיות; דין צירוף תרגום ל-פ"ה; דין הצלה לחצר שאינה מעורבת בכתבי הקודש.
(הטקסטים המלאים מובאים בגרסה הצרפתית והאנגלית, ועיין שו"ע הרב סימן של"ד באתר ספריא.)

סעיף יח-כאתיק הספר עם הספר, אמירה לנכרי, גליונות, אפיקורסים...

סעיף יח: מצילים תיק הספר עם הספר ותיק התפילין עם התפילין אפילו לחצר שאינה מעורבת.
סעיף יט: יש מתירים לומר לנכרי להציל ספרים אפילו דרך רשות הרבים.
סעיף כ: הגליונים שלמעלה ושלמטה — אין מצילין אותם אם נחתכו מן הספר.
סעיף כא: המינים, האדוקים בעבודה זרה, שכתבו להם כתבי הקודש — אין מצילין אותם.

סעיף כבתֵּבָה שֶׁאָחַז בָּהּ הָאוּר — מֻתָּר לִטּוֹל עוֹר שֶׁל גְּדִי...

תֵּבָה שֶׁאָחַז בָּהּ הָאוּר מִצִּדָּהּ הָאֶחָד — מֻתָּר לִטּוֹל עוֹר שֶׁל גְּדִי הַמֻּתָּר בְּטִלְטוּל וְלִפְרוֹס אוֹתוֹ עָלֶיהָ סָמוּךְ לָאוּר, כְּדֵי שֶׁלֹּא יִתְפַּשֵּׁט הָאוּר יוֹתֵר. וּמֻתָּר לַעֲשׂוֹת מְחִצָּה בְּכָל הַכֵּלִים, בֵּין רֵיקָנִים בֵּין מְלֵאִים, לְהַפְסִיק הַדְּלֵקָה. וַאֲפִלּוּ בִּכְלֵי חֶרֶס חֲדָשִׁים מְלֵאִים מַיִם, שֶׁמִּתְבַּקְּעִים וּמְכַבִּים אֶת הַדְּלֵקָה — לֹא גָזְרוּ בִּמְקוֹם הֶפְסֵד עַל גְּרַם כִּבּוּי כָּזֶה.

סעיף כג-כזטלית שאחז בה האור, משקין, נכרי, קטן, "כל המכבה אינו מפסיד"...

סעיף כג: טלית שאחז בה האור מצד אחד — מותר ליתן עליה משקין רחוק קצת מן האור.
סעיף כד: טלית מקופלת שאחז בה האור — פושטה ומתכסה בה.
סעיף כה: אסור לומר לנכרי לכבות הדליקה משום הפסד ממונו.
סעיף כו: נכרי שבא לכבות מעצמו — אין צריך למחות בידו; קטן — צריך למחות.
סעיף כז: בדליקה התירו לומר בפני הנכרי — "כל המכבה אינו מפסיד".

סעיף כחיֵשׁ אוֹמְרִים שֶׁכָּל מַה שֶּׁאָסְרוּ חֲכָמִים בִּדְלֵקָה — לֹא אָסְרוּ אֶלָּא בִּימֵיהֶם...

יֵשׁ אוֹמְרִים שֶׁכָּל מַה שֶּׁאָסְרוּ חֲכָמִים בִּדְלֵקָה — לֹא אָסְרוּ אֶלָּא בִּימֵיהֶם שֶׁלֹּא הָיוּ שְׁרוּיִים בֵּין הַנָּכְרִים, וַאֲפִלּוּ בְּגָלוּתָם הָיוּ לָהֶם מְקוֹמוֹת לְעַצְמָם, אֲבָל בִּזְמַן הַזֶּה אוֹתָם שֶׁשְּׁרוּיִים בֵּין הַנָּכְרִים וְיֵשׁ חֲשַׁשׁ סַכָּנַת נְפָשׁוֹת — מֻתָּר לְכַבּוֹת הַדְּלֵקָה בְּשַׁבָּת, וְכָל הַזָּרִיז לְכַבּוֹת הֲרֵי זֶה מְשֻׁבָּח. וּמִכָּל מָקוֹם, לֹא הֵקֵלּוּ אֶלָּא בְּכִבּוּי הַדְּלֵקָה, אֲבָל מִשּׁוּם הַצָּלַת מָמוֹנוֹ — אָסוּר לְחַלֵּל שַׁבָּת.

סעיף כטגַּחֶלֶת שֶׁל מַתֶּכֶת — לֹא שַׁיָּךְ בָּהּ כִּבּוּי מִן הַתּוֹרָה...

גַּחֶלֶת שֶׁל מַתֶּכֶת — לֹא שַׁיָּךְ בָּהּ כִּבּוּי מִן הַתּוֹרָה, הוֹאִיל וְאֵינָהּ בּוֹעֶרֶת. אֲבָל אָסוּר לְכַבּוֹתָהּ מִדִּבְרֵי סוֹפְרִים. וְאִם הִיא מֻנַּחַת בְּמָקוֹם שֶׁרַבִּים נִזּוֹקִים בָּהּ — מֻתָּר לְכַבּוֹתָהּ. אֲבָל גַּחֶלֶת שֶׁל עֵץ שַׁיָּךְ בָּהּ כִּבּוּי. וְכֵן עִקָּר.

מקור: ספריא — שולחן ערוך הרב, אורח חיים של"ד.

2 · חידוש אדמו"ר הזקן

חידושים מיוחדים של הרב

ייחודיות המבדילים את אדמו"ר הזקן בסימן של"ד. כל חידוש בנוי לפי המבנה: שיטה קלאסית → חידוש הרב → השלכה מעשית.

חידוש א — גזירת הבהלה (של"ד:א-ה)

לא לכבות = הגנה מפני עבירה חמורה יותר

שיטה קלאסית: —

חידוש הרב: אדמו"ר הזקן מבטא: גזירת אי-הכיבוי היא מגנה — למנוע שיחללו שבת בחומרה יתרה בבהלה.

השלכה מעשית: כלי: להכיר את חמש החריגות כדי לא להיכנס לבהלה.

חידוש ב — פיקוח נפש קודם (של"ד:ו-טו)

חיי אדם בראשונה

שיטה קלאסית: —

חידוש הרב: אדמו"ר הזקן מבטא: אם מישהו נמצא בבית, זה פיקוח נפש — לכבות הכל מותר.

השלכה מעשית: כלי: לוודא תחילה שכולם יצאו מהבית.

3 · דברי הרבי

שיחות, מאמרים ואגרות הרבי

⚠ הפניות לאימות — הסעיף בתהליך מקור.
טרם קיימת הפניה מאומתת לדברי הרבי על סימן זה. הציונים הכלליים שהוצגו קודם («שיחות שונות בנושא», «תשובות במקרים מעשיים») לא היו ציטוטים ממקור, והוסרו כדי לא לייחס לרבי דברים שאינם מאומתים.

כיווני חיפוש לאימות מול הכרכים (מהדורת קה"ת), תוך ציון כרך, עמוד ותאריך מדויקים: שערי הלכה ומנהג (המקור הישיר ביותר לפסקי הרבי לפי סימן), ואחריו ליקוטי שיחות, אגרות קודש, היום יום.
4 · למעשה

הלכה למעשה — מנהג חב"ד

נקודות הנהגה הלכתית הנובעות ישירות משו"ע אדמו"ר הזקן סימן של"ד, ברגישות החב"דית.

לחסיד חב"ד — סימן של"ד

  • ① דליקה: לא לכבות (גזירת הבהלה). חידוש א של הרב. מקור: שו"ע אדמו"ר הזקן של"ד:1-5.
  • ② חיי אדם = פיקוח נפש, הכל מותר. חידוש ב של הרב. מקור: שו"ע אדמו"ר הזקן של"ד:6-15.
  • ③ הצלת ספר תורה, מזון שבת. מקור: שו"ע אדמו"ר הזקן של"ד:6-15.
  • ④ לקרוא למכבי-אש (חזקת פיקוח נפש). מקור: שו"ע אדמו"ר הזקן של"ד:23-29.
  • ⑤ להקיף במים = בסדר (לא כיבוי ישיר). מקור: שו"ע אדמו"ר הזקן של"ד:16-22.
