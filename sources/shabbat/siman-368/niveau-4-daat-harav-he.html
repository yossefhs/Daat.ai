דעת הרב

דעת הרב — שולחן ערוך אדמו״ר הזקן על סימן שס״ח

עירוב שנתקלקל או שנתמעט: האם צריך להודיע לבני החצר כדי לתקנו? מה שיעורו, מה דין שיוריו, ומה עושים בפת עירוב שנתעפשה? ארבעת הסעיפים של שולחן ערוך אדמו״ר הזקן, מושווים לחמשת הסעיפים של המחבר ולהגהת הרמ״א.

מדוע רמה 4 מוקדשת לאדמו״ר הזקן? שולחן ערוך אדמו״ר הזקן אינו פירוש על המחבר: זהו שולחן ערוך בפני עצמו, שנכתב על ידי אדמו״ר הזקן, בחלוקה ובמספור סעיפים משלו. כאן יש בשולחן ערוך הרב ארבעה סעיפים, ובשולחן ערוך של המחבר חמישה: סעיף ג של הרב כולל את מה שאומר המחבר בסעיפים ג ו־ד, וסעיף ד של הרב מקביל לסעיף ה של המחבר ולהגהת הרמ״א שאחריו. ההפניות בעמוד זה הולכות לפי מספור הרב, אלא אם כן נזכר בפירוש שולחן ערוך של המחבר.

עמוד זה מקבץ, לסימן שס״ח, את הטקסט המלא של הרב, את מה ששונה בלשונו מדברי המחבר והרמ״א, ואת ההנהגה המעשית העולה מדבריו.

1 · טקסט אדמו״ר הזקן

שלחן ערוך הרב — סימן שס״ח

הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן שס״ח — אִם אַחַר שֶׁעֵרְבוּ נִתְקַלְקֵל הָעֵרוּב — וּבוֹ ד׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. ארבעה סעיפים (בשולחן ערוך של המחבר חמישה, במספור שלו).

סעיף א אִם אַחַר שֶׁעֵרְבוּ נִתְקַלְקֵל הָעֵרוּב וּבוֹ ד' סְעִיפִים: עֵרְבוּ, וְאַחַר כָּךְ נִתְקַלְקֵל…

אִם אַחַר שֶׁעֵרְבוּ נִתְקַלְקֵל הָעֵרוּב וּבוֹ ד' סְעִיפִים:

עֵרְבוּ, וְאַחַר כָּךְ נִתְקַלְקֵל הָעֵרוּב וְנִתְמַעֵט מִשִּׁעוּרוֹ קֹדֶם שֶׁנִּכְנַס הַשַּׁבָּת, וּבָא אֶחָד מִבְּנֵי הֶחָצֵר לְתַקְּנוֹ לְהוֹסִיף עָלָיו עַד כַּשִּׁעוּר — אִם בָּא לְעָרֵב מִמִּין הָרִאשׁוֹן, אֵין צָרִיךְ לְהוֹדִיעַ לָהֶם, אֲפִלּוּ מְעָרֵב עֲלֵיהֶם מִשֶּׁלָּהֶם, וּבִלְבָד שֶׁלֹּא יִטּוֹל פִּתָּם מִבָּתֵּיהֶם שֶׁלֹּא מִדַּעְתָּם, אֶלָּא בְּנֵי הַבַּיִת יִתְּנוּ לוֹ. וַאֲפִלּוּ אִם כָּלָה הָרִאשׁוֹן לְגַמְרֵי, וּבָא לְעָרֵב מֵחָדָשׁ מִשֶּׁלָּהֶם — אֵין צָרִיךְ לְהוֹדִיעָם. וַאֲפִלּוּ מִי שֶׁצָּרִיךְ לְהוֹדִיעַ לוֹ בִּתְחִלַּת הָעֵרוּב וְאֵין מְעָרְבִין בַּתְּחִלָּה שֶׁלֹּא מִדַּעְתּוֹ, כְּגוֹן שֶׁבֵּיתוֹ פָּתוּחַ לִשְׁתֵּי חֲצֵרוֹת, שֶׁאֵין הָעֵרוּב הַזֶּה זְכוּת לוֹ בְּוַדַּאי, וּכְמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שס״ז — אֵין צָרִיךְ לְהוֹדִיעַ לוֹ כְּשֶׁנִּתְקַלְקֵל וְחוֹזֵר וּמְתַקְּנִים אוֹתוֹ, שֶׁכֵּיוָן שֶׁכְּבָר נִתְרַצָּה בּוֹ בַּתְּחִלָּה, מִן הַסְּתָם גַּם עַכְשָׁו יִתְרַצֶּה בּוֹ.

אֲבָל אִם בָּא לְעָרֵב מִמִּין אַחֵר (כְּגוֹן שֶׁמִּתְּחִלָּה הָיָה פַּת חִטִּים וְעַכְשָׁו בָּא לַעֲשׂוֹתוֹ מִפַּת אֹרֶז) — אִם כָּלָה מִין הָרִאשׁוֹן לְגַמְרֵי, צָרִיךְ לְהוֹדִיעַ לְכָל מִי שֶׁצָּרִיךְ לְהוֹדִיעַ לוֹ בִּתְחִלַּת עֵרוּב, שֶׁגַּם עַכְשָׁו הוּא כִּתְחִלַּת עֵרוּב, כֵּיוָן שֶׁכָּלָה מִין הָרִאשׁוֹן לְגַמְרֵי, וְשֶׁמָּא לֹא יִתְרַצּוּ בְּמִין זֶה.

וְהוּא שֶׁמְּעָרֵב מִשֶּׁלָּהֶם. אֲבָל אִם מְזַכֶּה לָהֶם מִשֶּׁלּוֹ — אֵין צָרִיךְ לְהוֹדִיעָם אֲפִלּוּ כָּלָה מִין הָרִאשׁוֹן, לְפִי שֶׁבְּוַדַּאי יִתְרַצּוּ שֶׁיְּעָרֵב עֲלֵיהֶם בְּכָל הַמִּינִים שֶׁיִּרְצֶה, כֵּיוָן שֶׁהֵם אֵין נוֹתְנִים כְּלוּם.

וְאִם לֹא כָלָה מִין הָרִאשׁוֹן לְגַמְרֵי, אֶלָּא נִתְמַעֵט מִשִּׁעוּרוֹ קֹדֶם שֶׁנִּכְנַס הַשַּׁבָּת, וּבָא לְהוֹסִיף עָלָיו — יָכוֹל לְהוֹסִיף אֲפִלּוּ מִמִּין אַחֵר וְאֵין צָרִיךְ לְהוֹדִיעָם, אַף עַל פִּי שֶׁמְּעָרֵב מִשֶּׁלָּהֶם:

סעיף ב וְכָל זֶה אִם בָּא לְהוֹסִיף מֵחֲמַת שֶׁנִּתְקַלְקֵל. אֲבָל אִם בָּא לְהוֹסִיף מֵחֲמַת שֶׁנִּתּוֹסְפוּ…

וְכָל זֶה אִם בָּא לְהוֹסִיף מֵחֲמַת שֶׁנִּתְקַלְקֵל. אֲבָל אִם בָּא לְהוֹסִיף מֵחֲמַת שֶׁנִּתּוֹסְפוּ דִיּוּרִין בֶּחָצֵר, אֲפִלּוּ אִם מְזַכֶּה לָהֶם מִשֶּׁלּוֹ — צָרִיךְ לְהוֹדִיעַ לְכָל מִי מֵהֶם שֶׁצָּרִיךְ לְהוֹדִיעַ לוֹ בִּתְחִלַּת עֵרוּב, דְּהַיְנוּ כָּל מִי שֶׁבֵּיתוֹ פָּתוּחַ לִשְׁתֵּי חֲצֵרוֹת, לְפִי שֶׁהוֹסָפָה זוֹ הִיא תְּחִלַּת עֵרוּב לָהֶם, וּבִתְחִלַּת עֵרוּב אֵין מְעָרְבִין שֶׁלֹּא לָדַעַת לְכָל מִי שֶׁבֵּיתוֹ פָּתוּחַ לִשְׁתֵּי חֲצֵרוֹת אֲפִלּוּ אִם מְזַכֶּה לוֹ מִשֶּׁלּוֹ, כִּי שֶׁמָּא אֵינוֹ חָפֵץ כְּלָל לְעָרֵב בְּחָצֵר זוֹ אֶלָּא בְּאַחֶרֶת:

סעיף ג כַּמָּה הוּא שִׁעוּר הָעֵרוּב בִּתְחִלָּתוֹ? בִּזְמַן שֶׁהֵם י״ח אוֹ פָּחוֹת — שִׁעוּרָן כִּגְרוֹגֶרֶת…

כַּמָּה הוּא שִׁעוּר הָעֵרוּב בִּתְחִלָּתוֹ? בִּזְמַן שֶׁהֵם י״ח אוֹ פָּחוֹת — שִׁעוּרָן כִּגְרוֹגֶרֶת פַּת לְכָל אֶחָד. וְאִם הֵם יוֹתֵר מִי״ח, אֲפִלּוּ הֵם אֶלֶף אוֹ יוֹתֵר — שִׁעוּרוֹ מְזוֹן ב' סְעֻדּוֹת, שֶׁהֵם י״ח גְּרוֹגָרוֹת, שֶׁהֵם כְּשִׁשָּׁה בֵּיצִים בֵּינוֹנִית פַּת. וְיֵשׁ אוֹמְרִים שֶׁב' סְעֻדּוֹת הֵן כִּשְׁמוֹנֶה בֵּיצִים.

וְכָל זֶה בִּתְחִלַּת הָעֵרוּב. אֲבָל שִׁיּוּרֵי עֵרוּב, דְּהַיְנוּ עֵרוּב שֶׁהָיָה בּוֹ כַּשִּׁעוּר וְנִתְמַעֵט אַחַר שֶׁנִּכְנַס שַׁבָּת רִאשׁוֹנָה, וְנִשְׁתַּיֵּר בּוֹ אֲפִלּוּ כָּל שֶׁהוּא — כָּשֵׁר אַף לִשְׁאָר שַׁבָּתוֹת.

וַאֲפִלּוּ אִם הַמְּשֻׁיָּר אֵינוֹ כִּכָּר שָׁלֵם אֶלָּא פְּרוּסָה. וְלֹא אָמְרוּ שֶׁאֵין מְעָרְבִין בִּפְרוּסָה אֶלָּא בִּתְחִלַּת הָעֵרוּב. וַאֲפִלּוּ אִם נִפְרַס קֹדֶם כְּנִיסַת הַשַּׁבָּת — כָּשֵׁר אִם שִׁעוּרוֹ קַיָּם עַד כְּנִיסַת הַשַּׁבָּת, שֶׁהֲרֵי לֹא הִצְרִיכוּ שָׁלֵם אֶלָּא בִּשְׁבִיל שֶׁלֹּא יָבֹאוּ לִידֵי מַחֲלֹקֶת, וְאִם כֵּן דַּיּוֹ כְּשֶׁהוּא שָׁלֵם בִּשְׁעַת הַנָּחָתוֹ. אֲבָל אִם נִתְמַעֵט מִשִּׁעוּרוֹ קֹדֶם כְּנִיסַת הַשַּׁבָּת, אַף אִם הַמְּשֻׁיָּר הוּא כִּכָּר קָטָן שָׁלֵם — פָּסוּל, שֶׁתְּחִלַּת הָעֵרוּב הוּא תְּחִלַּת כְּנִיסַת הַשַּׁבָּת:

סעיף ד אִם נִתְעַפֵּשׁ פַּת הָעֵרוּב וְנִפְסַל מִלֶּאֱכוֹל — הֲרֵי הוּא כְּמוֹ שֶׁכָּלָה לְגַמְרֵי…

אִם נִתְעַפֵּשׁ פַּת הָעֵרוּב וְנִפְסַל מִלֶּאֱכוֹל — הֲרֵי הוּא כְּמוֹ שֶׁכָּלָה לְגַמְרֵי, וְצָרִיךְ לְעָרֵב מֵחָדָשׁ. וְלָכֵן נָהֲגוּ בִּמְדִינוֹת אֵלּוּ לַעֲשׂוֹת הָעֵרוּב בְּעֶרֶב פֶּסַח מֵחַלַּת מַצָּה, שֶׁאֵינָהּ מְמַהֶרֶת לְהִתְעַפֵּשׁ, וְעוֹד שֶׁיְּכוֹלִין לְשָׁמְרָהּ בִּימֵי הַפֶּסַח וִיכוֹלִין לִשְׁמוֹר הָעֵרוּב כָּל הַשָּׁנָה. וְזֶה טוֹב יוֹתֵר מִלְּעָרֵב כָּל עֶרֶב שַׁבָּת, כִּי שֶׁמָּא יִשְׁכְּחוּ מִלְּעָרֵב פַּעַם אַחַת. זֶהוּ טַעַם הַמִּנְהָג. אֲבָל בֶּאֱמֶת יוֹתֵר טוֹב לְעָרֵב כָּל עֶרֶב שַׁבָּת, כִּי רֹב פְּעָמִים מַצָּה שֶׁל עֵרוּב מִתְקַלְקֶלֶת וּמִתְעַפֶּשֶׁת מֵאֹרֶךְ הַזְּמַן וְאֵינָהּ רְאוּיָה לַאֲכִילָה, וַהֲרֵי הִיא כְּמוֹ שֶׁכָּלְתָה לְגַמְרֵי. וְכָל הַמְעָרֵב בְּכָל עֶרֶב שַׁבָּת, יֵשׁ לוֹ לֶאֱכוֹל הָעֵרוּב לְמָחָר, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שצ״ד.

וְהָרוֹצֶה לַעֲשׂוֹת עַל צַד הַיּוֹתֵר טוֹב — יְעָרֵב מִשֶּׁלּוֹ בְּכָל עֶרֶב שַׁבָּת, וִיזַכֵּהוּ לְכָל הַקָּהָל עַל יְדֵי אַחֵר, וּבְזֶה אֵין חֲשָׁשׁ אִם יִתְקַלְקֵל הָעֵרוּב שֶׁבְּבֵית הַכְּנֶסֶת. וְלֹא יְבָרֵךְ עַל שֶׁלּוֹ, כֵּיוָן שֶׁיֵּשׁ עֵרוּב בְּבֵית הַכְּנֶסֶת וְשֶׁמָּא הוּא כָּשֵׁר, וְהַבְּרָכוֹת אֵינָן מְעַכְּבוֹת:

מקור: ספריא — שולחן ערוך הרב, אורח חיים שס״ח. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה.

לשון הרב מול סעיף המחבר והרמ״א

במה שונה לשון הרב מסעיף המחבר והרמ״א

מה שכתוב בלשון הרב ואינו כתוב ואינו משתמע מלשון המחבר והגהת הרמ״א, בסימן זה או בדינים שהמחבר קובע בסימנים שס״ו ושס״ז, כפי שעולה מהשוואת הטקסטים, המובאים זה לצד זה. כשיש מפרשים הקוראים נקודה מסוימת כבר בדברי המחבר, ואחרים חולקים, הדף מציין אותה ומזכיר אותם בשמם, ואינו מכריע אם יש כאן הבדל מן המחבר או פירוש לדבריו. אין בכך גם קביעה שאדמו״ר הזקן הוא הראשון שכתב זאת: מפרשי השולחן ערוך אינם מושווים כאן אחד אחד, וכשאחד מהם כבר כותב נקודה מסוימת, הוא נזכר בשמו; גם המשנה ברורה, המאוחר לרב, נזכר כשהוא כותב אותה נקודה.

התנאי « וּבִלְבָד שֶׁלֹּא יִטּוֹל פִּתָּם מִבָּתֵּיהֶם שֶׁלֹּא מִדַּעְתָּם » (שו״ע הרב שס״ח:א) אינו נמנה כאן: המחבר עצמו קובע אותו בסימן הקודם, « אבל בני החצר אין יכולים ליקח פתו מביתו לערב שלא מדעת אחד מבני הבית » (שו״ע או״ח שס״ז:א), והט״ז מסמיך אותו לסעיף זה (ט״ז שס״ח ס״ק ב). אבל הביאור הלכה כותב: « עיין בט״ז ולא העתקתיו במ״ב משום שהשיגו עליו הרבה אחרונים » (ביאור הלכה שס״ח); אם הרב, הכותב את התנאי בסעיף עצמו, מכריע בכך — אין הדף מכריע. וכן כשהרב כותב שהפטור מלהודיע חל גם על מי שביתו פתוח לשתי חצרות (שו״ע הרב שס״ח:א), הוא מפרש את סעיף א של המחבר, שאינו מחלק בין אדם לאדם.

נקודה א — אפילו מזכה להם משלו (שו״ע הרב שס״ח:ב)

דיורין שנתוספו, אפילו משלו

דברי המחבר: « נתוספו דיורים בחצר אם הבית פתוח לשתי חצרות צריך להודיעם » (שו״ע או״ח שס״ח:ב). אינו אומר אם מערב משלו או משלהם.

דברי הרב: « אֲפִלּוּ אִם מְזַכֶּה לָהֶם מִשֶּׁלּוֹ — צָרִיךְ לְהוֹדִיעַ » (שו״ע הרב שס״ח:ב), משום « שֶׁהוֹסָפָה זוֹ הִיא תְּחִלַּת עֵרוּב לָהֶם » (שו״ע הרב שס״ח:ב).

ההבדל: סעיף ב של המחבר אינו אומר אם החובה להודיע קיימת גם כשמערב משלו; לשון הרב כותבת זאת. המגן אברהם והמשנה ברורה, להלן, כותבים זאת בפירוש סעיף זה של המחבר; רש״י ורבינו יהונתן פוטרים מלהודיע כשמערב משלו (רמה 2, פרק 5), והמשנה ברורה מביא מקילין: אין הדף מכריע אם יש כאן הבדל מן המחבר או פירוש לדבריו.

אצל מפרשי המחבר: המגן אברהם כותב זאת לפניו: « אפי׳ רוצה לערב משלו דשמא לא ניחא להם » (מגן אברהם שס״ח ס״ק א). המשנה ברורה, המאוחר לרב, כותב כך, « ואפילו אם מערב עתה משלו » (משנה ברורה שס״ח ס״ק ט), ומביא את הדעה החולקת: « ויש מקילין שיכול לערב משלו בכל גווני » (משנה ברורה שס״ח ס״ק ט).

השלכה מעשית: לפי לשון הרב, המוסיף על העירוב בשביל דיורין שנתוספו מודיע למי מהם שביתו פתוח לשתי חצרות, אפילו כשמערב משלו (שו״ע הרב שס״ח:ב); ראה פרק 4, נקודה ③.

נקודה ב — שיורי עירוב ועירוב שנפרס קודם השבת (שו״ע הרב שס״ח:ג)

שיורי עירוב, ועירוב שנפרס קודם השבת

דברי המחבר: « נתמעט מכשיעורו אחר שנכנס שבת ראשונ׳ ונשתייר בו אפי׳ כל שהוא כשר אף לשאר שבתות » (שו״ע או״ח שס״ח:ד). ובסימן שס״ו קבע: « אין מערבין בפרוסה אפי׳ היא גדולה הרבה אבל בשלימה אפי׳ קטנה מאוד מערבין ובלבד שיהא בהם כל כך שיהא בהם כשיעור » (שו״ע או״ח שס״ו:ו). בסעיף ד של סימן שס״ח אינו אומר בפירוש אם השיורים יכולים להיות פרוסה.

דברי הרב: באותו סעיף שבו הוא נותן גם את שיעור העירוב, הוא מוסיף « וַאֲפִלּוּ אִם הַמְּשֻׁיָּר אֵינוֹ כִּכָּר שָׁלֵם אֶלָּא פְּרוּסָה » (שו״ע הרב שס״ח:ג), ונותן טעם: « וְלֹא אָמְרוּ שֶׁאֵין מְעָרְבִין בִּפְרוּסָה אֶלָּא בִּתְחִלַּת הָעֵרוּב » (שו״ע הרב שס״ח:ג). ואחר כך מפרש שעירוב שנפרס קודם כניסת השבת כשר אם שיעורו קיים עד כניסת השבת, ואם נתמעט משיעורו קודם כניסת השבת — פסול, אף אם המשויר הוא ככר קטן שלם, « שֶׁתְּחִלַּת הָעֵרוּב הוּא תְּחִלַּת כְּנִיסַת הַשַּׁבָּת » (שו״ע הרב שס״ח:ג).

ההבדל: סעיף ד של המחבר כבר קובע את כניסת השבת הראשונה כגבול. שהשיורים כשרים אף אם אינם אלא פרוסה — אין המחבר אומר זאת בפירוש, ולשון הרב כותבת זאת; הט״ז כבר קורא זאת במילים כל שהוא שבסעיף זה, אבל תשובת רמ״מ שמביא המגן אברהם מפרשת כל שהוא — שלם (להלן). לכן נזכרת נקודה זו כאן, כמו בנקודה א, שבה המגן אברהם כבר כותב את דברי הרב והמשנה ברורה מביא מקילין; אם יש כאן הבדל מן המחבר או פירוש לדבריו — אין הדף מכריע. עוד מוסיפה לשון הרב שעירוב שנפרס קודם כניסת השבת כשר אם שיעורו קיים עד כניסת השבת. וכשהוא כותב שעירוב שנתמעט משיעורו קודם כניסת השבת פסול אף אם נשתייר ממנו ככר קטן שלם, הוא מפרש את מה שכבר עולה מדברי המחבר: השיעור נדרש אף בשלימה (שו״ע או״ח שס״ו:ו), ורק מיעוט שאחר כניסת השבת הראשונה הוכשר (שו״ע או״ח שס״ח:ד); וכך כותב המשנה ברורה: « דאלו נתמעט קודם שקידש עליו היום הו״ל כתחלת עירוב » (משנה ברורה שס״ח ס״ק י״ד).

אצל מפרשי המחבר: הט״ז כבר כותב, לעניין השיורים, « ומ״ש בסי׳ שס״ו שאין מערבין בפרוסה היינו בשבת ראשון » (ט״ז או״ח שס״ח ס״ק ג) — לשונו נוקטת שבת ראשון, ולא, כמשנה ברורה להלן, את שעת הנחת העירוב; אין הוא אומר מה דינה של ככר שנפרסה קודם כניסת השבת הראשונה, והדף אינו מונה את הט״ז עם אחת הדעות שלהלן; והמגן אברהם מביא בשם המהרי״ל, לעניין הפרוסה, « ואם נשבר העירוב אפילו קודם שנכנס שבת ראשונה שרי כיון דבשעת הנחה הי׳ שלם » (מגן אברהם שס״ח ס״ק ג). הוא מביא גם את התשובה שהוא מכנה רמ״מ, « ובתשו׳ רמ״מ כ׳ דאם נשבר אפילו אחר שבת ראשונה טוב לעשות אחר » (מגן אברהם שס״ח ס״ק ג), ובה טעם ראשון, « וטעמו דהא דתנן נשתייר כל שהוא כשר היינו שלם » (מגן אברהם שס״ח ס״ק ג), והמגן אברהם דוחה פירוש זה, « גם מ״ש דכל שהוא היינו שלם הוא דוחק בעיני » (מגן אברהם שס״ח ס״ק ג), ומכריע: « לכן נ״ל כמהרי״ל » (מגן אברהם שס״ח ס״ק ג); ראה רמה 2, פרק 8. גם המשנה ברורה, המאוחר לרב, כותב: « היינו כשהיתה פרוסה בשעה שהניחה לשם עירוב משא״כ הכא שנפרסה אח״כ לית לן בה » (משנה ברורה שס״ח ס״ק י״ז); ולעניין ככר שנפרסה קודם כניסת השבת הוא מביא דעה מחמירה, שלשון הרב אינה מביאה: « ומ״מ יש מחמירין בשנפרסה קודם בין השמשות כיון שיש שהות עדיין להניח ככר אחר שלם » (משנה ברורה שס״ח ס״ק י״ז).

השלכה מעשית: לפי לשון הרב, עירוב שהיה בו כשיעור בכניסת השבת ונתמעט אחר כך כשר לשבתות הבאות כל זמן שנשתייר ממנו משהו, אפילו פרוסה; וככר שנפרסה קודם כניסת השבת כשרה אם שיעורה קיים עד כניסת השבת (שו״ע הרב שס״ח:ג) — ובזה המשנה ברורה מביא מחמירין (ס״ק י״ז, לעיל). אבל אם נתוספו דיורין אחר שנתמעט העירוב משיעורו, צריך להוסיף בשבילם, שלגביהם זו תחילת עירוב (שו״ע הרב שס״ו:י״ב). הסתייגות נוספת של המשנה ברורה בשיור כל שהוא — ראה פרק 4, נקודה ⑤.

נקודה ג — לערב בכל ערב שבת (שו״ע הרב שס״ח:ד)

חלת מצה, או עירוב בכל ערב שבת

דברי הרמ״א: אחרי המנהג לעשות את העירוב מחלת מצה, הגהתו מוסיפה « וזה טוב יותר מלערב כל שבת ושבת שלא ישכחו מלערב » (רמ״א או״ח שס״ח:ה), וכן « מיהו אם רוצים לערב כל ע״ש ולאכול העירוב כל שבת הרשות בידם » (רמ״א או״ח שס״ח:ה). לערב בכל ערב שבת — רשות, וחלת המצה מוצגת כטוב יותר.

דברי הרב: הוא מביא את אותו מנהג ואת אותו טעם, ואחר כך כותב « זֶהוּ טַעַם הַמִּנְהָג » (שו״ע הרב שס״ח:ד), ומיד « אֲבָל בֶּאֱמֶת יוֹתֵר טוֹב לְעָרֵב כָּל עֶרֶב שַׁבָּת » (שו״ע הרב שס״ח:ד), « כִּי רֹב פְּעָמִים מַצָּה שֶׁל עֵרוּב מִתְקַלְקֶלֶת וּמִתְעַפֶּשֶׁת מֵאֹרֶךְ הַזְּמַן » (שו״ע הרב שס״ח:ד). ומוסיף דרך לעשות על צד היותר טוב: « יְעָרֵב מִשֶּׁלּוֹ בְּכָל עֶרֶב שַׁבָּת, וִיזַכֵּהוּ לְכָל הַקָּהָל עַל יְדֵי אַחֵר » (שו״ע הרב שס״ח:ד); וכן « וְלֹא יְבָרֵךְ עַל שֶׁלּוֹ » (שו״ע הרב שס״ח:ד), כיון שיש עירוב בבית הכנסת ושמא הוא כשר.

ההבדל: לרמ״א חלת המצה טובה יותר, והעירוב בכל ערב שבת רשות; לרב, מה שמעדיף את חלת המצה אינו אלא טעם המנהג, « זֶהוּ טַעַם הַמִּנְהָג » (שו״ע הרב שס״ח:ד), אבל באמת העירוב בכל ערב שבת טוב יותר. הדרך שבה יחיד מזכה את עירובו לכל הקהל, בלי ברכה, אינה בהגהת הרמ״א. הט״ז כבר כותב את שתיהן: « ולעד״נ דיותר טוב לערב כל ע״ש » (ט״ז או״ח שס״ח ס״ק ד), וכן « והרוצה לעשות עצהי״ט יערב משלו בכל ע״ש ויזכ׳ ע״י אחר » (ט״ז או״ח שס״ח ס״ק ד). המשנה ברורה, המאוחר לרב, מביא בשם האחרונים את ההעדפה ואת טעמה, « והאחרונים כתבו דיותר טוב לערב כל ע״ש כי רוב פעמים מצה של עירוב מתקלקלת ומתעפשת מאורך הזמן » (משנה ברורה שס״ח ס״ק כ״א), את דרך העשייה, « והרוצה לעשות עצהיו״ט יערב משלו בכל ע״ש ויזכהו לכל הקהל ע״י אחר » (משנה ברורה שס״ח ס״ק כ״א), ואת הדין שאינו מברך, « ולא יברך על שלו כיון שיש עירוב בבהכ״נ ושמא הוא כשר והברכות אין מעכבות » (משנה ברורה שס״ח ס״ק כ״א). הדף אינו קובע מי הם אחרונים אלה.

השלכה מעשית: לפי לשון הרב, טוב יותר לערב בכל ערב שבת, והמערב כך יש לו לאכול את העירוב למחר; והרוצה לעשות על צד היותר טוב מערב משלו בכל ערב שבת, מזכה אותו לכל הקהל על ידי אחר, ואינו מברך על שלו (שו״ע הרב שס״ח:ד).

3 · דברי הרבי

שיחות, מאמרים ואגרות הרבי

דברי הרבי על נושאי סימן שס״ח

אין הפניה מאומתת נכון לעכשיו.
אנו מייחסים לרבי אך ורק מקורות ראשוניים הניתנים לאימות (מכתב, שיחה או פסק מזוהה, עם קישור). בשלב זה לא ניתן היה לאמת מול Sefaria או Chabad.org דבר מקור של הרבי המתייחס ישירות לסימן זה — מוטב שלא להציג דבר מאשר הפניה לא מאומתת.
כיוון חיפוש מועדף לפסקי הרבי לפי סימן: שערי הלכה ומנהג (מהדורת קה״ת). כל הפניה שתאומת תתווסף עם קישורה.
4 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן שס״ח, ברגישות החב״דית. כל נקודה מפנה לסעיף שלה בשולחן ערוך הרב, לפי סדר הלשון.

לחסיד חב״ד — סימן שס״ח

  • ① תיקון העירוב ממין הראשון: העירוב נתקלקל ונתמעט קודם כניסת השבת; המוסיף עליו ממין הראשון אינו צריך להודיע לבני החצר, אפילו מערב משלהם, ובלבד שלא יטול את פתם מבתיהם שלא מדעתם, אלא בני הבית נותנים לו. ואפילו כלה הראשון לגמרי והוא מערב מחדש משלהם — אינו צריך להודיעם. וכך גם במי שביתו פתוח לשתי חצרות (שו״ע הרב שס״ח:א). הפניה: שולחן ערוך אדמו״ר הזקן שס״ח:א.
  • ② תיקון ממין אחר: אם כלה מין הראשון לגמרי והוא מערב משל בני החצר — צריך להודיע לכל מי שצריך להודיע לו בתחילת עירוב; אם מזכה להם משלו — אינו צריך להודיע; ואם לא כלה הראשון אלא נתמעט — מוסיף אפילו ממין אחר בלי להודיע (שו״ע הרב שס״ח:א). הפניה: שולחן ערוך אדמו״ר הזקן שס״ח:א.
  • ③ נתוספו דיורין בחצר: המוסיף על העירוב בשבילם, אפילו משלו, צריך להודיע לכל מי מהם שביתו פתוח לשתי חצרות, שהוספה זו היא תחילת עירוב להם (שו״ע הרב שס״ח:ב). הפניה: שולחן ערוך אדמו״ר הזקן שס״ח:ב.
  • ④ שיעור העירוב בתחילתו: כשהם שמונה עשר או פחות — כגרוגרת פת לכל אחד; יותר משמונה עשר, אפילו אלף — מזון שתי סעודות, שהן שמונה עשרה גרוגרות, כשש ביצים בינוניות פת; והרב מביא גם את הדעה « וְיֵשׁ אוֹמְרִים שֶׁב׳ סְעֻדּוֹת הֵן כִּשְׁמוֹנֶה בֵּיצִים » (שו״ע הרב שס״ח:ג). הפניה: שולחן ערוך אדמו״ר הזקן שס״ח:ג.
  • ⑤ שיורי עירוב: אם היה בו כשיעור בכניסת השבת הראשונה ונתמעט אחר כך — כשר לשאר שבתות כל זמן שנשתייר בו כל שהוא, אפילו פרוסה; ככר שנפרסה קודם כניסת השבת כשרה אם שיעורה קיים עד כניסת השבת, ובזה המשנה ברורה מביא דעה מחמירה: « ומ״מ יש מחמירין בשנפרסה קודם בין השמשות » (משנה ברורה שס״ח ס״ק י״ז); ואם נתמעט משיעורו קודם כניסת השבת — פסול כמות שהוא, אף אם נשתייר ככר קטן שלם (שו״ע הרב שס״ח:ג), ומשלימים אותו עד כשיעור קודם כניסת השבת (שו״ע הרב שס״ח:א; ראה נקודות ① ו־②). ואם נתוספו דיורין אחר שנתמעט העירוב משיעורו, השיור כל שהוא אינו פוטר מלהוסיף בשבילם: « וְאִם נִתּוֹסְפוּ דִיּוּרִין אַחַר שֶׁנִּתְמַעֵט הָעֵרוּב מִשִּׁעוּרוֹ … מִכָּל מָקוֹם צָרִיךְ לְהוֹסִיף עָלָיו מֵחֲמַת הַדִּיּוּרִין שֶׁנִּתּוֹסְפוּ עַתָּה, שֶׁלָּהֶם הוּא תְּחִלַּת עֵרוּב » (שו״ע הרב שס״ו:י״ב); ראה נקודה ③. המשנה ברורה מביא את המגן אברהם, שלדעתו דין זה נאמר כשעשו עירובי חצרות ושיתופי מבואות כל אחד בפני עצמו, ולא כשעושים שיתוף ועירוב ביחד; ומציין שבביאור הלכה יש חולקין, וכותב: « ולכתחלה בודאי צריך ליזהר להוסיף עד כשיעור » (משנה ברורה שס״ח ס״ק ט״ז). הביאור הלכה מסיק: « וע״כ נראה דיש להקל בזה בשעת הדחק » (ביאור הלכה שס״ח). לשון הרב בשס״ח:ג אינה מחלקת בכך; כשעושים עירוב ושיתוף ביחד — אין הדף מכריע, והיעזר ברב שלך. ראה פרק 2. הפניה: שולחן ערוך אדמו״ר הזקן שס״ח:ג.
  • ⑥ פת עירוב שנתעפשה: אם נפסלה מלאכול — הרי היא כאילו כלתה לגמרי, וצריך לערב מחדש. למנהג חלת המצה יש טעם, אבל באמת טוב יותר לערב בכל ערב שבת, והמערב כך יש לו לאכול את העירוב למחר; ועל צד היותר טוב — עירוב משלו בכל ערב שבת, שמזכה לכל הקהל על ידי אחר, בלי ברכה על שלו, כיון שיש עירוב בבית הכנסת ושמא הוא כשר (שו״ע הרב שס״ח:ד). ראה פרק 2. הפניה: שולחן ערוך אדמו״ר הזקן שס״ח:ד.

⚠ סעיף זה מציג את ההנהגה ההלכתית של חב״ד. לכל שאלה ממשית: להיוועץ ברב שלך.
