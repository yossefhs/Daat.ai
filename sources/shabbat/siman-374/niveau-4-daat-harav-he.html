דעת הרב — שולחן ערוך אדמו״ר הזקן על סימן שע״ד

פתח או חלון שעירבו דרכו שתי חצרות, ונסתם בשבת או קודם לה; כותל שנפרץ בין שתי חצרות; חצר קטנה וגג קטן שנפרצו במלואם לגדולים מהם. ארבעת הסעיפים של שולחן ערוך אדמו״ר הזקן, בהשוואה לארבעת הסעיפים של המחבר.

מדוע רמה 4 מוקדשת לאדמו״ר הזקן? שולחן ערוך אדמו״ר הזקן אינו פירוש על המחבר: זהו שולחן ערוך בפני עצמו, שנכתב על ידי אדמו״ר הזקן, בחלוקה ובמספור סעיפים משלו. כאן יש בשולחן ערוך הרב ארבעה סעיפים, וגם בשולחן ערוך של המחבר ארבעה; כל אחד מסעיפים א עד ד של הרב עוסק בדין הסעיף שבאותו מקום אצל המחבר. ההפניות בעמוד זה הולכות לפי מספור הרב, אלא כשהשולחן ערוך של המחבר נזכר בשמו. בעמוד זה שני המספורים כתובים באותיות, ולכן כל הפניה לסעיף נזכרת עם שם החיבור: של המחבר (שו״ע או״ח) או של הרב (שו״ע הרב).

עמוד זה מקבץ, לסימן שע״ד, את הטקסט המלא של הרב, את מה ששונה בלשונו מדברי המחבר ומן הסוגריים שבסעיפו, ואת ההנהגה המעשית העולה מדבריו.

שלחן ערוך הרב — סימן שע״ד

הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן שע״ד — נִסְתַּם בְּשַׁבָּת פֶּתַח אוֹ חַלּוֹן בְּמָקוֹם שֶׁעֵרֵב בּוֹ — וּבוֹ ד׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. ארבעה סעיפים (גם בשולחן ערוך של המחבר ארבעה, במספור שלו).

סעיף א נִסְתַּם בְּשַׁבָּת פֶּתַח אוֹ חַלּוֹן בְּמָקוֹם שֶׁעֵרֵב בּוֹ וּבוֹ ד' סְעִיפִים: ב' בָּתִּים אוֹ ב' חֲצֵרוֹת…

נִסְתַּם בְּשַׁבָּת פֶּתַח אוֹ חַלּוֹן בְּמָקוֹם שֶׁעֵרֵב בּוֹ וּבוֹ ד' סְעִיפִים:

ב' בָּתִּים אוֹ ב' חֲצֵרוֹת שֶׁעֵרְבוּ יַחַד עַל יְדֵי שֶׁהָיָה בֵּינֵיהֶם חַלּוֹן אוֹ פֶּתַח, וְנִסְתַּם בְּשַׁבָּת — מֻתָּרִים לְהִשְׁתַּמֵּשׁ מִזֶּה לְזֶה דֶּרֶךְ גֹּבַהּ כֹּתֶל וְחוֹרָיו, שֶׁשַּׁבָּת כֵּיוָן שֶׁהֻתְּרָה — הֻתְּרָה.

אֲבָל אִם נִסְתַּם מִבְּעוֹד יוֹם, אֲפִלּוּ חָזַר וְנִפְתַּח בְּשַׁבָּת — אֲסוּרִים, לְפִי שֶׁעֲדַיִן לֹא קָנוּ עֵרוּב מֵעוֹלָם, שֶׁקְּנִיַּת הָעֵרוּב הוּא בֵּין הַשְּׁמָשׁוֹת, וְכַאן לֹא הָיָה עֵרוּב בֵּין הַשְּׁמָשׁוֹת כֵּיוָן שֶׁהַפֶּתַח סָתוּם, וּמִשֶּׁחָשֵׁכָה לֹא קָנוּ עֵרוּב, שֶׁאֵין מְעָרְבִין מִשֶּׁתֶּחְשַׁךְ.

וַאֲפִלּוּ עֵרְבוּ כְּבָר לִפְנֵי שַׁבָּת שֶׁעָבְרָה, וְאוֹתוֹ עֵרוּב נִשְׁאָר עֲדַיִן קַיָּם — אֵינָן יְכוֹלִים לִסְמוֹךְ עָלָיו בְּשַׁבָּת זוֹ אַחַר שֶׁנִּפְתַּח הַפֶּתַח, הוֹאִיל וּכְבָר נִתְבַּטֵּל מִבְּעוֹד יוֹם וְאֵינוֹ חוֹזֵר וְנֵעוֹר מִשֶּׁתֶּחְשַׁךְ. אֲבָל אִם מִתְּחִלָּה עֵרְבוּ לְשַׁבָּתוֹת הַרְבֵּה, אֲפִלּוּ עֵרְבוּ לְשָׁנָה אוֹ יוֹתֵר, וְקָנוּ עֵרוּב בְּבֵין הַשְּׁמָשׁוֹת שֶׁל שַׁבָּת רִאשׁוֹנָה, שֶׁהָיָה הַפֶּתַח פָּתוּחַ וְאַחַר כָּךְ נִסְתַּם הַפֶּתַח וְנִפְתַּח בְּשַׁבָּת אַחֶרֶת, אֲפִלּוּ בְּסוֹף הַשָּׁנָה — חוֹזֵר הָעֵרוּב וְנֵעוֹר, וַאֲפִלּוּ סְתָמוֹ בְּמֵזִיד. וְיֵשׁ מִי שֶׁחוֹשֵׁשׁ לְהַחֲמִיר אִם סְתָמוֹ בְּמֵזִיד, הוֹאִיל וְגִלָּה דַעְתּוֹ בִּסְתִימָה זוֹ שֶׁאֵינוֹ חָפֵץ בָּעֵרוּב — אֵינוֹ חוֹזֵר וְנֵעוּר כְּשֶׁנִּפְתַּח. וּלְעִנְיַן הֲלָכָה, בְּדִבְרֵי סוֹפְרִים הַלֵּךְ אַחַר הַמֵּקֵל:

סעיף ב הָיָה כֹּתֶל בֵּין [שְׁתֵּי] חֲצֵרוֹת, וְעֵרְבָה כָּל אַחַת לְעַצְמָהּ…

הָיָה כֹּתֶל בֵּין [שְׁתֵּי] חֲצֵרוֹת, וְעֵרְבָה כָּל אַחַת לְעַצְמָהּ, וְנִפְרַץ הַכֹּתֶל בְּשַׁבָּת — מֻתָּרִים לְטַלְטֵל כָּל אֶחָד בַּחֲצֵרוֹ עַד מְקוֹם הַכֹּתֶל אֲפִלּוּ כֵּלִים שֶׁשָּׁבְתוּ בַּבַּיִת, שֶׁשַּׁבָּת כֵּיוָן שֶׁהֻתְּרָה — הֻתְּרָה.

אֲבָל חָצֵר שֶׁנִּפְרְצָה בְּשַׁבָּת לִרְשׁוּת הָרַבִּים אוֹ לְכַרְמְלִית פִּרְצָה הָאוֹסַרְתָּהּ — אֵין אוֹמְרִים "שַׁבָּת כֵּיוָן שֶׁהֻתְּרָה הֻתְּרָה", כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שס״ב:

סעיף ג חָצֵר קְטַנָּה שֶׁנִּפְרְצָה מִבְּעוֹד יוֹם בְּמִלּוּאָהּ לִגְדוֹלָה…

חָצֵר קְטַנָּה שֶׁנִּפְרְצָה מִבְּעוֹד יוֹם בְּמִלּוּאָהּ לִגְדוֹלָה, וְאֵין בַּפִּרְצָה יוֹתֵר מֵעֶשֶׂר אַמּוֹת — גְּדוֹלָה מֻתֶּרֶת לְהוֹצִיא כֵּלִים שֶׁשָּׁבְתוּ בַּבַּיִת לַחֲצֵרָהּ אִם עֵרְבָה לְעַצְמָהּ, וְאַף עַל פִּי שֶׁלֹּא עֵרְבָה עִם הַקְּטַנָּה, אֵין הַקְּטַנָּה אוֹסֶרֶת עָלֶיהָ, מִפְּנֵי שֶׁפִּרְצָה זוֹ הִיא לַגְּדוֹלָה כְּפֶתַח שֶׁאֵין לָהּ בְּמִלּוּאָהּ, אֶלָּא נִשְׁתַּיְּרוּ בָּהּ גִּפּוּפִים עוֹדְפִים עַל פִּרְצַת הַקְּטַנָּה מִכַּאן וּמִכַּאן. וְהַקְּטַנָּה אֲסוּרָה לְהוֹצִיא כֵּלִים שֶׁשָּׁבְתוּ בַּבַּיִת לַחֲצֵרָה, הוֹאִיל וְנִפְרְצָה בְּמִלּוּאָהּ לְהַגְּדוֹלָה, אֶלָּא אִם כֵּן עֵרְבָה עִם הַגְּדוֹלָה.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁנִּכְנָסִין כָּתְלֵי אֹרֶךְ הַקְּטַנָּה לְתוֹךְ הַגְּדוֹלָה וּמֻפְלָגִין שָׁם ג' טְפָחִים מִכָּתְלֵי אֹרֶךְ הַגְּדוֹלָה, שֶׁאִם לֹא כֵן אַף הַקְּטַנָּה נִתֶּרֶת עַל יְדֵי גִפּוּפֵי הַגְּדוֹלָה, מִשּׁוּם נִרְאֶה מִבְּחוּץ וְשָׁוֶה מִבִּפְנִים, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן ש״ס:

סעיף ד וְכֵן גַּג קָטָן הַמֻּקָּף מַעֲקֶה סְבִיבוֹ גְּבוֹהָה עָלָיו עֲשָׂרָה…

וְכֵן גַּג קָטָן הַמֻּקָּף מַעֲקֶה סְבִיבוֹ גְּבוֹהָה עָלָיו עֲשָׂרָה שֶׁנִּפְרַץ שָׁם בְּמִלּוּאוֹ לְגַג גָּדוֹל הַמֻּקָּף גַּם כֵּן כָּמֹהוּ, וְלֹא עֵרְבוּ הַבָּתִּים יַחַד — קָטָן אָסוּר לְהַעֲלוֹת עָלָיו כֵּלִים שֶׁשָּׁבְתוּ בַּבַּיִת אִם מְחִצּוֹתָיו נִכְנָסוֹת לְתוֹךְ הַגָּדוֹל, הוֹאִיל וְאֵין לוֹ מְחִצָּה לְהַפְסִיק בֵּינוֹ לַגָּדוֹל שֶׁלֹּא עֵרֵב עִמּוֹ, וְגָדוֹל מֻתָּר, מִפְּנֵי שֶׁיֵּשׁ לוֹ גִּפּוּפִים עוֹדְפִים עַל פִּרְצַת הַקָּטָן.

וְכֵן אֲפִלּוּ אִם אֵינָן מֻקָּפִים מַעֲקֶה כְּלָל, אֶלָּא שֶׁמְּחִצּוֹת הַבַּיִת סָבִיב נִכָּרוֹת וְנִרְאוֹת לְמִי שֶׁעוֹמֵד עַל שְׂפַת הַגַּג כְּשֶׁמִּסְתַּכֵּל תַּחַת רַגְלָיו, שֶׁאָז אוֹמְרִים בָּהֶן "גּוּד אַסֵּק" וּכְאִלּוּ הֵן עוֹמְדוֹת לְמַעְלָה עַל שְׂפַת הַגַּג, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שמ״ה. אֲבָל מְחִצּוֹת הַבַּיִת שֶׁבֵּין קָטָן לַגָּדוֹל אֵין אוֹמְרִים בָּהּ "גּוּד אַסֵּק", הוֹאִיל וְאֵינָהּ נִרְאֵית לְמִי שֶׁעוֹמֵד שָׁם מִלְמַעְלָה, שֶׁהֲרֵי הַגַּגִּין מְחֻבָּרִין זֶה לְזֶה וּמְכַסִּים עַל פִּרְצַת הַקְּטַנָּה וְאֵין רוֹאִים אוֹתָהּ, וּמְקוֹם מְחִצָּה זוֹ לְמַעְלָה הִיא כְּפִרְצָה בְּמִלּוּאָהּ מֵהַקָּטָן לְהַגָּדוֹל, וּלְהַגָּדוֹל אֵינָהּ בְּמִלּוּאוֹ, שֶׁהֲרֵי כֹּתֶל בַּיִת הַגָּדוֹל עוֹדֵף מִכַּאן וּמִכַּאן וְנִרְאֶה לְמִי שֶׁעוֹמֵד עַל שְׂפַת הַגַּג. וּלְפִיכָךְ גָּדוֹל מֻתָּר וְקָטָן אָסוּר אִם נִכְנָסִים כָּתְלֵי בֵּיתוֹ לְתוֹךְ הַגָּדוֹל.

אֲבָל אִם גַּם מְחִצּוֹת הַבַּיִת סָבִיב אֵינָן נִכָּרוֹת וְנִרְאוֹת לְמִי שֶׁעוֹמֵד עַל שְׂפַת הַגַּג, כְּגוֹן שֶׁהַגַּג בּוֹלֵט עֲלֵיהֶן — כַּרְמְלִית הוּא, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שמ״ה, אֶלָּא אִם כֵּן חַלּוֹן ד' עַל ד' פָּתוּחַ לוֹ מֵהַבַּיִת (וְאָז הוּא רְשׁוּת הַיָּחִיד כְּמוֹ שֶׁנִּתְבָּאֵר שָׁם). אֲבָל מִכָּל מָקוֹם אִם לֹא עֵרְבוּ יַחַד אַף הַגָּדוֹל אָסוּר בְּכֵלִים שֶׁשָּׁבְתוּ בַּבַּיִת, לְפִי שֶׁאֵין לוֹ גִּפּוּפִים עוֹדְפִים מִכַּאן וּמִכַּאן, שֶׁהֲרֵי כֹּתֶל הַבַּיִת אֵין אוֹמְרִים בּוֹ "גּוּד אַסֵּק", הוֹאִיל וְאֵינוֹ נִכָּר לְמִי שֶׁעוֹמֵד לְמַעְלָה:

מקור: ספריא — שולחן ערוך הרב, אורח חיים שע״ד. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה.

2 · לשון הרב ולשון המחבר

לשון הרב מול סעיף המחבר

במה שונה לשון הרב מסעיף המחבר

כל נקודה מעמידה זה לצד זה סעיף של שולחן ערוך המחבר (במספור שלו, ארבעה סעיפים, המסומנים באותיותיהם) ואת הסעיף שבאותו מקום בשולחן ערוך אדמו״ר הזקן (במספור שלו, גם הוא ארבעה סעיפים), לפי סדר סעיפי הרב, ואינה מציינת אלא מה שעולה מהשוואת שני הטקסטים. אין בכך שנקודה מסוימת מיוחדת לרב: כל נקודה מציינת, בין השאר, היכן נקרא כבר מה שהרב כותב — בטור, בבית יוסף וברש״י שהוא מביא, במגן אברהם ובט״ז, שקדמו לו —, ובמקום הצורך, מה שאינו נקרא שם, וכן מה שכותבים בה המשנה ברורה והביאור הלכה, המאוחרים לרב; וכשהמפרשים קוראים את הנקודה בדברי המחבר עצמם, הדף אומר זאת. את הראשונים שמביא הבית יוסף לא השווינו אחד אחד. בסעיף א של המחבר המלים « ואפילו סתמה במזיד » (שו״ע או״ח שע״ד:א) נדפסו באותיות קטנות, בסוגריים, בלי המלה הגה, ואחריהן מקורן, האגור; הטקסט אינו מזכיר את בעליהן, והביאור הלכה מייחס אותן לרמ״א (נקודה ג).

נקודה א — שימוש מזו לזו, ודעת הרמב״ם (שו״ע הרב שע״ד:א)

פתח שנסתם בשבת

דברי המחבר: « ערבו דרך חלון או פתח שביניהם ונסתם בשבת מותרים להשתמש דרך גובה הכותל וחוריו » (שו״ע או״ח שע״ד:א), בלי לכתוב את הטעם; ובסוף הסעיף הוא מביא את דעת הרמב״ם: « ולהרמב״ם אין מותר לטלטל אלא כל אחת לעצמה אבל מזו לזו לא » (שו״ע או״ח שע״ד:א).

דברי הרב: הוא מפרש את המקרה, « ב׳ בָּתִּים אוֹ ב׳ חֲצֵרוֹת שֶׁעֵרְבוּ יַחַד עַל יְדֵי שֶׁהָיָה בֵּינֵיהֶם חַלּוֹן אוֹ פֶּתַח » (שו״ע הרב שע״ד:א), וכותב: « מֻתָּרִים לְהִשְׁתַּמֵּשׁ מִזֶּה לְזֶה דֶּרֶךְ גֹּבַהּ כֹּתֶל וְחוֹרָיו, שֶׁשַּׁבָּת כֵּיוָן שֶׁהֻתְּרָה — הֻתְּרָה » (שו״ע הרב שע״ד:א). בסעיף א שלו אין הוא מביא את דעת הרמב״ם.

ההבדל: סעיף א של המחבר כותב את הדין בלי טעמו, ומביא אחריו את דעת הרמב״ם, שלדבריו כל חצר מותרת רק לעצמה; לשון הרב כותבת « מִזֶּה לְזֶה » (שו״ע הרב שע״ד:א), נותנת את הטעם, ואינה מביאה את דעת הרמב״ם. הטעם והפירוש שהשימוש הוא מזו לזו אינם מיוחדים לרב: הטור נותן את הטעם, « דשבת כיון שהותרה הותרה » (טור או״ח שע״ד), והבית יוסף כותב שכך פירש הטור: « ופירש רבינו דמותרין לטלטל מזו לזו כמו שהיו » (בית יוסף או״ח שע״ד), ואחר כך מביא את הרמב״ם, « אבל הרמב״ם בפ״ג מה״ע כתב שכל א׳ מותר לעצמה » (בית יוסף או״ח שע״ד). שהרב אינו מביא את דעת הרמב״ם — הדף מציין זאת, ואינו מסיק מזה טעם.

אצל מפרשי המחבר: המגן אברהם כותב על דברי הרמב״ם « דוקא כשהפתח סתום » (מגן אברהם שע״ד ס״ק ב) ו« אבל כשנפתח הפתח חזרו שניהם להיתרן » (מגן אברהם שע״ד ס״ק ב). המשנה ברורה, המאוחר לרב, מפרש את הדעה הראשונה « מזו לזו בשבת זו וכ״ש כל אחת לעצמה » (משנה ברורה שע״ד ס״ק ב), וכותב על דעת הרמב״ם: « ולענין הלכה העיקר כסברא הראשונה כ״כ האחרונים » (משנה ברורה שע״ד ס״ק ו).

השלכה מעשית: לפי לשון הרב, שני בתים או שתי חצרות שעירבו יחד דרך חלון או פתח שנסתם בשבת — מותרים להשתמש מזה לזה דרך גובה הכותל וחוריו (שו״ע הרב שע״ד:א); ראה פרק 4, נקודה ①.

נקודה ב — נסתם מבעוד יום, ועירוב לשבתות הרבה (שו״ע הרב שע״ד:א)

מתי קנו עירוב

דברי המחבר: « ואפי׳ אם עירב לשנה ונסתם הפתח בחול ונפתח בשבת » (שו״ע או״ח שע״ד:א), ואחרי הסוגריים באותיות קטנות (נקודה ג): « חזר העירוב להתירו » (שו״ע או״ח שע״ד:א). אין הוא כותב לא את דין העירוב שנעשה לשבת אחת ולא את הטעם.

דברי הרב: « אֲבָל אִם נִסְתַּם מִבְּעוֹד יוֹם, אֲפִלּוּ חָזַר וְנִפְתַּח בְּשַׁבָּת — אֲסוּרִים » (שו״ע הרב שע״ד:א), « לְפִי שֶׁעֲדַיִן לֹא קָנוּ עֵרוּב מֵעוֹלָם, שֶׁקְּנִיַּת הָעֵרוּב הוּא בֵּין הַשְּׁמָשׁוֹת » (שו״ע הרב שע״ד:א); ואחר כך « וַאֲפִלּוּ עֵרְבוּ כְּבָר לִפְנֵי שַׁבָּת שֶׁעָבְרָה, וְאוֹתוֹ עֵרוּב נִשְׁאָר עֲדַיִן קַיָּם — אֵינָן יְכוֹלִים לִסְמוֹךְ עָלָיו בְּשַׁבָּת זוֹ » (שו״ע הרב שע״ד:א); ובעירוב לשבתות הרבה, « וְקָנוּ עֵרוּב בְּבֵין הַשְּׁמָשׁוֹת שֶׁל שַׁבָּת רִאשׁוֹנָה, שֶׁהָיָה הַפֶּתַח פָּתוּחַ » (שו״ע הרב שע״ד:א), « אֲפִלּוּ בְּסוֹף הַשָּׁנָה — חוֹזֵר הָעֵרוּב וְנֵעוֹר » (שו״ע הרב שע״ד:א).

ההבדל: סעיף א של המחבר כותב את דין העירוב לשנה ואת מסקנתו; לשון הרב כותבת גם את המקרה ההפוך — פתח שנסתם מבעוד יום, עירוב שנעשה לפני שבת שעברה — ואת הטעם של שניהם: העירוב נקנה בבין השמשות. הטור, בדין העירוב לשנה, מעיר « דהשתא לא נכנסה השבת בהיתר » (טור או״ח שע״ד), ומסיק שחזר העירוב להתירו; את המקרה ההפוך ואת הטעם הזה אינו כותב. המגן אברהם, שקדם לרב, כותב כבר שהעירוב נקנה בבין השמשות, בפתח שחזר ונפתח קודם שבת: « דכשחוזר ופותח׳ קודם שבת הרי הם כשותפין בפת ובין השמשו׳ שהוא קונה עירוב הרי יש פתח ביניהם ראוי לערב » (מגן אברהם שע״ד ס״ק א).

אצל מפרשי המחבר: המשנה ברורה, המאוחר לרב, מפרש כך את דין השנה: « כיון שבעת הנחת העירוב שהניח לכמה שבתות או לשנה היה פתח מזו לזו וחל העירוב » (משנה ברורה שע״ד ס״ק ד). הביאור הלכה, המאוחר לרב, פותח את דבריו על « ונסתם בשבת » (שו״ע או״ח שע״ד:א) באותו דין, בלשון קרובה מאוד ללשון הרב, ואינו מזכיר מקור: « אבל אם נסתם מבעוד יום אפילו חזר ונפתח בשבת אסורין לפי שעדיין לא קנו עירוב מעולם שקניית העירוב הוא בה״ש » (ביאור הלכה שע״ד), ו« ועירב לשנה שאני שמערב מתחלה על כל השבתות וקנה העירוב בשבת ראשונה שהיה הפתח פתוח ולכן אפילו אם אח״כ נסתם הפתח ונפתח בשבת אחרת ואפילו בסוף השנה חוזר העירוב וניעור » (ביאור הלכה שע״ד). אחר כך הוא דן בעצם דין העירוב לשנה, « ועכ״פ אין דין זה ברור » (ביאור הלכה שע״ד), ומסיק: « ומ״מ אחרי שהרא״ש והטור והמחבר והאחרונים שאחרי השו״ע העתיקו דין זה בודאי אין לדחותו מהלכה » (ביאור הלכה שע״ד).

השלכה מעשית: לפי לשון הרב, אם נסתם הפתח מבעוד יום — אסורים אפילו חזר ונפתח בשבת, ועירוב שעירבו לפני שבת שעברה אינו מועיל; אבל אם עירבו מתחילה לשבתות הרבה, וקנו עירוב בבין השמשות של שבת ראשונה כשהפתח פתוח — חוזר העירוב וניעור כשנפתח בשבת אחרת, אפילו בסוף השנה (שו״ע הרב שע״ד:א); ראה פרק 4, נקודה ②.

נקודה ג — סתמו במזיד (שו״ע הרב שע״ד:א)

סתמו במזיד

דברי סעיף המחבר: בדין העירוב לשנה, הסוגריים באותיות קטנות « ואפילו סתמה במזיד » (שו״ע או״ח שע״ד:א), ואחריהם מקורם (אגור), בלי דעה חולקת. הטקסט אינו מזכיר את בעליהם; הביאור הלכה מייחס אותם לרמ״א וכותב « את סברת הרמ״א דמתיר בכל גווני » (ביאור הלכה שע״ד).

דברי הרב: « וַאֲפִלּוּ סְתָמוֹ בְּמֵזִיד » (שו״ע הרב שע״ד:א); ואחר כך הוא מביא דעה מחמירה, בלי לנקוב בשמה, עם טעמה, « וְיֵשׁ מִי שֶׁחוֹשֵׁשׁ לְהַחֲמִיר אִם סְתָמוֹ בְּמֵזִיד, הוֹאִיל וְגִלָּה דַעְתּוֹ בִּסְתִימָה זוֹ שֶׁאֵינוֹ חָפֵץ בָּעֵרוּב » (שו״ע הרב שע״ד:א), ומכריע: « וּלְעִנְיַן הֲלָכָה, בְּדִבְרֵי סוֹפְרִים הַלֵּךְ אַחַר הַמֵּקֵל » (שו״ע הרב שע״ד:א).

ההבדל: סעיף א של המחבר כותב את הסוגריים בלי דעה חולקת; לשון הרב כותבת אותם, מביאה את הדעה המחמירה ואת טעמה, וכותבת שלהלכה, בדברי סופרים, הולכים אחר המקל. הטעם הזה כתוב לפניו במגן אברהם, המפנה לתשובות מהרי״ל, על המלים « ואפי׳ סתמה במזיד » (מגן אברהם שע״ד ס״ק א): « כשנפתח בשבת אסור דכיון דסתמו במזיד גלי דעתי׳ שאינו חפץ בעירו׳ וא״כ ב״ה לא קנה עירוב » (מגן אברהם שע״ד ס״ק א); והוא מוסיף שלזה נתכוון האגור, « ולזה נתכוין האגו׳ שכ׳ בשם מהרי״ל » (מגן אברהם שע״ד ס״ק א). הרב אינו נוקב בשם המחמיר, והדף אינו נוקב בשמו במקומו. הט״ז, שקדם לרב, דוחה על אותן מלים את חילוקו של חמיו — חילוק הב״ח, שהמגן אברהם מביא, « והב״ח כתב דבפרץ פצימיו בטל העירוב » (מגן אברהם שע״ד ס״ק א) — וכותב « אלא ודאי דהעיקר תלוי בעשיי׳ העירוב כיון דהיה ראוי באותה שעה » (ט״ז או״ח שע״ד ס״ק א).

אצל מפרשי המחבר: המשנה ברורה, המאוחר לרב, כותב « ודע דמ״א וכמה אחרונים צידדו לאסור בזה כשחזר ונפתח בשבת » (משנה ברורה שע״ד ס״ק ה), ואחר כך « ואין להקל כשסתמו במזיד אלא כשנפתח קודם השבת » (משנה ברורה שע״ד ס״ק ה) — חוץ מ« אכן אם בעת שסתמו היה בדעתו רק לפי שעה ואח״כ לחזור ולפתחו מסתברא דמותר אף כשחזר ונפתח בשבת » (משנה ברורה שע״ד ס״ק ה). הביאור הלכה כותב « ומ״מ לענין סתמו במזיד שהמ״א הביא ממהרי״ל שלא התיר אלא דוקא בשנפתח קודם השבת » (ביאור הלכה שע״ד), ומסיק « מ״מ למעשה נראה שאין להקל » (ביאור הלכה שע״ד). במקרה שכותב הרב הפתח נפתח « בְּשַׁבָּת אַחֶרֶת » (שו״ע הרב שע״ד:א): בפתח שנסתם במזיד ונפתח בשבת, לשון הרב מקילה, והמשנה ברורה והביאור הלכה אינם מקילים; ומי שלא סתמו אלא לפי שעה ובדעתו לחזור ולפתחו — מסתבר למשנה ברורה שמותר.

השלכה מעשית: לפי לשון הרב, עירוב שנעשה לשבתות הרבה חוזר וניעור אפילו אם נסתם הפתח במזיד; הוא מביא דעה מחמירה, וכותב שבדברי סופרים הולכים אחר המקל (שו״ע הרב שע״ד:א). המשנה ברורה והביאור הלכה כותבים שאין להקל כשהפתח, שנסתם במזיד, נפתח בשבת; ומי שלא סתמו אלא לפי שעה ובדעתו לחזור ולפתחו — מסתבר למשנה ברורה שמותר (לעיל): היעזר ברב שלך. ראה פרק 4, נקודה ②.

נקודה ד — כותל שנפרץ בשבת, וחצר שנפרצה לרשות הרבים (שו״ע הרב שע״ד:ב)

עד מקום הכותל

דברי המחבר: « היה כותל בין שתי חצרו׳ ועירבה כל אחת לעצמה ונפרץ הכותל בשבת מותרים לטלטל כל א׳ בחצרו אפי׳ כלי הבי׳ » (שו״ע או״ח שע״ד:ב), ואחר כך « אבל חצר שנפרצה בשבת לרשו׳ הרבים או לכרמלית אסור » (שו״ע או״ח שע״ד:ב). אין הוא כותב טעם לא לזה ולא לזה.

דברי הרב: « מֻתָּרִים לְטַלְטֵל כָּל אֶחָד בַּחֲצֵרוֹ עַד מְקוֹם הַכֹּתֶל אֲפִלּוּ כֵּלִים שֶׁשָּׁבְתוּ בַּבַּיִת, שֶׁשַּׁבָּת כֵּיוָן שֶׁהֻתְּרָה — הֻתְּרָה » (שו״ע הרב שע״ד:ב); ואחר כך « אֲבָל חָצֵר שֶׁנִּפְרְצָה בְּשַׁבָּת לִרְשׁוּת הָרַבִּים אוֹ לְכַרְמְלִית פִּרְצָה הָאוֹסַרְתָּהּ — אֵין אוֹמְרִים » (שו״ע הרב שע״ד:ב) שבת כיון שהותרה הותרה, « כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שס״ב » (שו״ע הרב שע״ד:ב).

ההבדל: סעיף ב של המחבר כותב « אפי׳ כלי הבי׳ » (שו״ע או״ח שע״ד:ב) בלי לומר עד היכן מטלטלים, ואוסר בחצר שנפרצה לרשות הרבים, בלי טעם; לשון הרב כותבת « עַד מְקוֹם הַכֹּתֶל » (שו״ע הרב שע״ד:ב), את טעם ההיתר, ובחצר שנפרצה לרשות הרבים או לכרמלית היא מפרשת « פִּרְצָה הָאוֹסַרְתָּהּ » (שו״ע הרב שע״ד:ב) וכותבת שאין אומרים בה שבת כיון שהותרה הותרה, ומפנה לסימן שס״ב. פרטים אלו כתובים כבר בבית יוסף, המביא את דין שמואל, « זה מטלטל עד עיקר מחיצה כדמעיקרא דאמרינן הואיל והותרה הותרה » (בית יוסף או״ח שע״ד), וכותב בחצר שנפרצה לרשות הרבים « אבל כשנפרץ לרשות אחרת ממש דהיינו לר״ה או לכרמלית לא אמרי׳ ביה הואיל והותרה הותרה » (בית יוסף או״ח שע״ד); ולשיעור הפרצה הוא מפנה לסימן אחר: « ושיעור פירצה זו ליאסר לטלטל בחצר מפניה נתבאר בסימן ש״ס » (בית יוסף או״ח שע״ד).

אצל מפרשי המחבר: הט״ז נותן את טעם החילוק בין שני המקרים, « כיון דלא נפרץ לרשות אחר אלא לחצר משא״כ בסיפא שנפרץ לרשות הרבים או לכרמלית » (ט״ז או״ח שע״ד ס״ק ב), והמגן אברהם את טעם האיסור, « כיון שאין המחיצות קיימו׳ » (מגן אברהם שע״ד ס״ק ג). המשנה ברורה, המאוחר לרב, כותב בחצר שנפרצה לרשות הרבים « בזה לא אמרינן כיון שהותרה הותרה » (משנה ברורה שע״ד ס״ק ט), והביאור הלכה מפרש על « ונפרץ הכותל בשבת » (שו״ע או״ח שע״ד:ב): « היינו במילואו או ביותר מעשר אמות » (ביאור הלכה שע״ד).

השלכה מעשית: לפי לשון הרב, כותל שבין שתי חצרות שעירבה כל אחת לעצמה ונפרץ בשבת — מותרים לטלטל כל אחד בחצרו עד מקום הכותל, אפילו כלים ששבתו בבית; אבל בחצר שנפרצה בשבת לרשות הרבים או לכרמלית פרצה האוסרתה — אין אומרים שבת כיון שהותרה הותרה (שו״ע הרב שע״ד:ב); ראה פרק 4, נקודה ③.

נקודה ה — חצר קטנה שנפרצה לגדולה (שו״ע הרב שע״ד:ג)

למה אין הקטנה אוסרת על הגדולה

דברי המחבר: « גדול׳ מותרת להוציא כלי׳ ששבתו בבית לחצרה אם עירבה לעצמה » (שו״ע או״ח שע״ד:ג), בלי לומר למה אין הקטנה אוסרת עליה; ואחר כך « והיינו כשנכנסין כותלי קטנה לגדולה וכשכתלי קטנה מופלגים ג״ט מכותלי אורך הגדולה » (שו״ע או״ח שע״ד:ג), ולמקרה ההפוך — הטעם: « דאם לא כן היתה קטנה ניתרת ע״י נראה מבחוץ ושוה מבפנים » (שו״ע או״ח שע״ד:ג).

דברי הרב: « וְאַף עַל פִּי שֶׁלֹּא עֵרְבָה עִם הַקְּטַנָּה, אֵין הַקְּטַנָּה אוֹסֶרֶת עָלֶיהָ » (שו״ע הרב שע״ד:ג), « מִפְּנֵי שֶׁפִּרְצָה זוֹ הִיא לַגְּדוֹלָה כְּפֶתַח שֶׁאֵין לָהּ בְּמִלּוּאָהּ, אֶלָּא נִשְׁתַּיְּרוּ בָּהּ גִּפּוּפִים עוֹדְפִים עַל פִּרְצַת הַקְּטַנָּה מִכַּאן וּמִכַּאן » (שו״ע הרב שע״ד:ג); הקטנה אסורה « הוֹאִיל וְנִפְרְצָה בְּמִלּוּאָהּ לְהַגְּדוֹלָה » (שו״ע הרב שע״ד:ג); ואת התנאי הוא כותב במשפט אחד, « כְּשֶׁנִּכְנָסִין כָּתְלֵי אֹרֶךְ הַקְּטַנָּה לְתוֹךְ הַגְּדוֹלָה וּמֻפְלָגִין שָׁם ג׳ טְפָחִים מִכָּתְלֵי אֹרֶךְ הַגְּדוֹלָה » (שו״ע הרב שע״ד:ג), ולמקרה ההפוך — הטעם: « שֶׁאִם לֹא כֵן אַף הַקְּטַנָּה נִתֶּרֶת עַל יְדֵי גִפּוּפֵי הַגְּדוֹלָה, מִשּׁוּם נִרְאֶה מִבְּחוּץ וְשָׁוֶה מִבִּפְנִים » (שו״ע הרב שע״ד:ג), ומפנה לסימן ש״ס · 360.

ההבדל: סעיף ג של המחבר כותב את שני הדינים בלי טעמיהם, ואת התנאי בשני משפטים; לשון הרב כותבת שהקטנה אינה אוסרת על הגדולה אף בלי עירוב ביניהן, נותנת את הטעם — הפרצה לגדולה כפתח —, נותנת את טעם איסור הקטנה, וכותבת את התנאי במשפט אחד. הטעמים אינם מיוחדים לרב: הבית יוסף מביא את המשנה, « גדולה מותרת וקטנה אסורה מפני שהיא כפתחה של גדולה » (בית יוסף או״ח שע״ד), ואת רש״י, « ולא אסרי עלה בני קטנה » (בית יוסף או״ח שע״ד); המגן אברהם נותן את טעם האיסור, « דנפרצה במלואה » (מגן אברהם שע״ד ס״ק ד). למקרה ההפוך נותן הרב אותו טעם שנותן המחבר, « מִשּׁוּם נִרְאֶה מִבְּחוּץ וְשָׁוֶה מִבִּפְנִים » (שו״ע הרב שע״ד:ג), ומפרש שהקטנה ניתרת אז « עַל יְדֵי גִפּוּפֵי הַגְּדוֹלָה » (שו״ע הרב שע״ד:ג), ומפנה לסימן ש״ס · 360. בהפלגת שלשה טפחים מזכיר הבית יוסף לבוד, ומבאר אותו מיד: « ומוקי לה נמי בשכותלי קטנה מופלגים ג׳ טפחים מכותלי אורך הגדולה דאל״כ הוה מישתרי קטנה ע״י לבוד כלומר דהוה אמרינן חזינן כאילו ראשי כניסת של כתלי קטנה סניפין לצד כתלי אורך הגדולה דהדר הויא להו הכניסה להך גיפופי נראים מבחוץ » (בית יוסף או״ח שע״ד) — כלומר: בלי ההפלגה רואים את ראשי כתלי הקטנה כסמוכים לכתלי אורך הגדולה, והבליטה חוזרת להיות גיפופים הנראים מבחוץ. כך מחזיר הבית יוסף את הלבוד לאותו טעם: אין זה טעם בפני עצמו.

אצל מפרשי המחבר: המשנה ברורה, המאוחר לרב, כותב « ואין הקטנה אוסרת עליה דהא הגדולה יש לה גיפופין » (משנה ברורה שע״ד ס״ק י״ג), ובהפלגה, כבית יוסף, « והא דבעינן שלשה טפחים דאל״ה אמרינן לבוד לאורך הגדולה והוי כאלו לא היה מופלג כלל » (משנה ברורה שע״ד ס״ק י״ח). הביאור הלכה כותב על « וכשכותלי קטנה מופלגים ג״ט » (ביאור הלכה שע״ד), בשם הגר״א, « לאו דוקא דאף כותל אחד סגי » (ביאור הלכה שע״ד), ומפרש: « ור״ל דאף אם היה רק כותל אחד מופלג ג״ט והשני היה פחות מזה ג״כ אסור » (ביאור הלכה שע״ד); לשון הרב כותבת, כמחבר, את הכתלים בלשון רבים, והדף אינו אומר מה דעתו כשרק כותל אחד מופלג.

השלכה מעשית: לפי לשון הרב, חצר קטנה שנפרצה מבעוד יום במלואה לגדולה, ואין בפרצה יותר מעשר אמות — הגדולה מותרת להוציא כלים ששבתו בבית לחצרה אם עירבה לעצמה, אף בלי עירוב עם הקטנה; והקטנה אסורה, אלא אם כן עירבה עם הגדולה. זה כשנכנסין כתלי אורך הקטנה לתוך הגדולה ומופלגין שם שלשה טפחים מכתלי אורך הגדולה; ואם לא כן — אף הקטנה ניתרת (שו״ע הרב שע״ד:ג). אם רק כותל אחד מופלג שלושה טפחים, הביאור הלכה, בשם הגר״א, אוסר את הקטנה; מה דעת הרב במקרה זה — הדף אינו אומר (לעיל): היעזר ברב שלך. ראה פרק 4, נקודה ④.

נקודה ו — גג קטן שנפרץ לגג גדול (שו״ע הרב שע״ד:ד)

המעקה, המחיצות הניכרות, והגג הגדול

דברי המחבר: « גג קטן שנפרץ קודם שבת במילואו לגג גדול קטן אסור להעלות עליו כלים ששבתו בבית וגדול מותר והוא שיהו מחיצות הבית ניכרות למי שעומד על הגג » (שו״ע או״ח שע״ד:ד), ואחר כך « אבל אם אינם ניכרות כגון שהגג בולט עליהם הוי כרמלית אלא אם כן פתוח לו מהבית חלון ארבע על ארבע » (שו״ע או״ח שע״ד:ד).

דברי הרב: הוא פותח בגגות המוקפים מעקה, « וְכֵן גַּג קָטָן הַמֻּקָּף מַעֲקֶה סְבִיבוֹ גְּבוֹהָה עָלָיו עֲשָׂרָה » (שו״ע הרב שע״ד:ד), « וְלֹא עֵרְבוּ הַבָּתִּים יַחַד » (שו״ע הרב שע״ד:ד), ואוסר את הקטן « אִם מְחִצּוֹתָיו נִכְנָסוֹת לְתוֹךְ הַגָּדוֹל » (שו״ע הרב שע״ד:ד); ואחר כך « וְכֵן אֲפִלּוּ אִם אֵינָן מֻקָּפִים מַעֲקֶה כְּלָל » (שו״ע הרב שע״ד:ד), כשמחיצות הבית ניכרות משפת הגג; הוא כותב שאין אומרים גוד אסק במחיצה שבין הקטן לגדול, « הוֹאִיל וְאֵינָהּ נִרְאֵית לְמִי שֶׁעוֹמֵד שָׁם מִלְמַעְלָה » (שו״ע הרב שע״ד:ד); ואחרי דין הכרמלית והחלון, « אֲבָל מִכָּל מָקוֹם אִם לֹא עֵרְבוּ יַחַד אַף הַגָּדוֹל אָסוּר בְּכֵלִים שֶׁשָּׁבְתוּ בַּבַּיִת » (שו״ע הרב שע״ד:ד).

ההבדל: סעיף ד של המחבר אינו מדבר במעקה, אינו אומר שהבתים לא עירבו יחד, אינו כותב בגג את התנאי של מחיצות הנכנסות לתוך הגדול — שהוא כותב, בסעיף ג, בחצר —, אינו אומר למה אין המחיצה שבין שני הגגות נחשבת, ואינו כותב את המשפט שהרב כותב אחרי דין החלון: שאם לא עירבו יחד, אף הגדול אסור בכלים ששבתו בבית. לשון הרב כותבת את כל אלה. תנאי המחיצות הניכרות עצמו כתוב בסעיף ד של המחבר, « והוא שיהו מחיצות הבית ניכרות למי שעומד על הגג » (שו״ע או״ח שע״ד:ד); הרב מפרט אותו, « לְמִי שֶׁעוֹמֵד עַל שְׂפַת הַגַּג כְּשֶׁמִּסְתַּכֵּל תַּחַת רַגְלָיו » (שו״ע הרב שע״ד:ד), כמו המשנה ברורה אחריו (להלן). רש״י, שמביא הבית יוסף, מפרש אותו: « נהי דשרינן ליה במחיצות התחתונות משום גוד אסיק מיהו מחיצות ניכרות בעינן שלא יהיה גג בולט חוצה להם » (בית יוסף או״ח שע״ד). רש״י כותב עוד, בגג שאין מחיצותיו ניכרות, « וגדול נמי אסור דאין שם שום מחיצה וכל שאין לו מחיצות אינו רה״י » (בית יוסף או״ח שע״ד): הוא מדבר בגג שאין לו שום מחיצה ואינו רשות היחיד. הרב כותב את משפטו מיד אחרי החלון שעושה את הגג רשות היחיד, « וְאָז הוּא רְשׁוּת הַיָּחִיד כְּמוֹ שֶׁנִּתְבָּאֵר שָׁם » (שו״ע הרב שע״ד:ד), ופותח אותו ב« אֲבָל מִכָּל מָקוֹם » (שו״ע הרב שע״ד:ד); נראה שדברי רש״י עוסקים במקרה אחר (זו קריאת הדף, והיא מובאת להכרעת הרב), והמשנה ברורה, המאוחר לרב, הוא שכותב אותו דין בגג שיש לו חלון, בסעיף קטן כ״ט שלו (להלן). והבית יוסף מציין שהרא״ש מצריך בחלון שיעור ארבעה על ארבעה, « אלא שהרא״ש הצריך שיהיה החלון ד׳ על ד׳ » (בית יוסף או״ח שע״ד). המעקה, ותנאי המחיצות הנכנסות לתוך הגדול, אינם כתובים לא בטור, לא בבית יוסף, לא במגן אברהם, לא בט״ז ולא במשנה ברורה על סימן זה (לפי ערכיהם המופיעים בספריא).

אצל מפרשי המחבר: המשנה ברורה, המאוחר לרב, כותב « ומיירי נמי שלא עירבו בעליהם יחד וגם שאין גג הקטן מחזיק יותר מעשר אמות » (משנה ברורה שע״ד ס״ק כ״ג). הביאור הלכה שואל למה אין הגג הקטן ניתר בגיפופי הגדול משום נראה מבחוץ ושוה מבפנים, כבסעיף ג, ומשיב על פי הריטב״א: « י״ל דכי אמרינן גוד אסיק לגבי גג הגדול עצמו אבל לא להתיר הגג קטן בגיפופי גדול » (ביאור הלכה שע״ד); לשון הרב נותנת את הטעם שהמחיצה שבין שני הגגות אינה נראית מלמעלה. אם שני ההסברים עולים בקנה אחד — הדף אינו אומר. המשנה ברורה מפרש את המחיצות הניכרות, « נראין מחיצות הבית לעומדין על שפת הגג כשמסתכלין תחת רגליהם ואז אמרינן גוד אסיק מחיצתא » (משנה ברורה שע״ד ס״ק כ״ז). בגג הבולט כותב המגן אברהם, על פי התוספות (דף פ״ט), « בתו׳ דף פ״ט משמע דוקא כשבולט ד׳ לר״ה ור״ה מהלכת תחתיו » (מגן אברהם שע״ד ס״ק ה), והמשנה ברורה « דוקא כשבולט ד״ט » (משנה ברורה שע״ד ס״ק כ״ח); לשון הרב אינה כותבת כאן שיעור, ומפנה לסימן שמ״ה, שהדף אינו מביא. על החלון כותב המשנה ברורה « אבל להעלות עליו כלים ששבתו בבית אסור כ״ז שלא עירבו יחד כיון שאין המחיצות ניכרות עליו » (משנה ברורה שע״ד ס״ק כ״ט).

השלכה מעשית: לפי לשון הרב, גג קטן המוקף מעקה גבוה עשרה [טפחים] שנפרץ במלואו לגג גדול המוקף גם כן, ולא עירבו הבתים יחד — אסור להעלות על הקטן כלים ששבתו בבית אם מחיצותיו נכנסות לתוך הגדול, והגדול מותר (המשנה ברורה מוסיף שאין הגג הקטן יותר מעשר אמות, ס״ק כ״ג); וכן בלא מעקה, כשמחיצות הבית ניכרות משפת הגג. אם אינן ניכרות — כרמלית הוא, אלא אם כן חלון ד׳ על ד׳ [טפחים] פתוח לו מהבית; ואף בחלון זה, אם לא עירבו יחד, אף הגדול אסור בכלים ששבתו בבית, לפי שכותל הבית אינו ניכר למי שעומד למעלה (שו״ע הרב שע״ד:ד); ראה פרק 4, נקודה ⑤.

4 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן שע״ד, ברגישות החב״דית. כל נקודה מפנה לסעיף שלה בשולחן ערוך הרב, לפי סדר הלשון.

לחסיד חב״ד — סימן שע״ד

  • ① פתח שנסתם בשבת: שני בתים או שתי חצרות שעירבו יחד דרך חלון או פתח שביניהם, ונסתם בשבת — מותרים להשתמש מזה לזה דרך גובה הכותל וחוריו, ששבת כיון שהותרה הותרה (שו״ע הרב שע״ד:א). ראה פרק 2, נקודה א. הפניה: שולחן ערוך אדמו״ר הזקן שע״ד:א.
  • ② פתח שנסתם קודם השבת: אם נסתם מבעוד יום — אסורים אפילו חזר ונפתח בשבת, ועירוב שעירבו לפני שבת שעברה אינו מועיל. אם עירבו מתחילה לשבתות הרבה, אפילו לשנה או יותר, וקנו עירוב בבין השמשות של שבת ראשונה כשהפתח פתוח — חוזר העירוב וניעור כשנפתח בשבת אחרת, אפילו בסוף השנה, ואפילו סתמו במזיד; במקרה האחרון מביא הרב דעה מחמירה, וכותב שבדברי סופרים הולכים אחר המקל (שו״ע הרב שע״ד:א). בפתח שנסתם במזיד ונפתח בשבת — היעזר ברב שלך (מה שכותבים בזה המשנה ברורה והביאור הלכה — ראה פרק 2, נקודה ג). ראה פרק 2, נקודות ב ו־ג. הפניה: שולחן ערוך אדמו״ר הזקן שע״ד:א.
  • ③ כותל שנפרץ בשבת: בין שתי חצרות שעירבה כל אחת לעצמה — מותרים לטלטל כל אחד בחצרו עד מקום הכותל, אפילו כלים ששבתו בבית. אבל בחצר שנפרצה בשבת לרשות הרבים או לכרמלית פרצה האוסרתה — אין אומרים שבת כיון שהותרה הותרה; הרב מפנה לסימן שס״ב (שו״ע הרב שע״ד:ב). ראה פרק 2, נקודה ד. הפניה: שולחן ערוך אדמו״ר הזקן שע״ד:ב.
  • ④ חצר קטנה שנפרצה לגדולה: מבעוד יום, במלואה, ואין בפרצה יותר מעשר אמות — הגדולה מותרת להוציא כלים ששבתו בבית לחצרה אם עירבה לעצמה, אף בלי עירוב עם הקטנה; והקטנה אסורה, אלא אם כן עירבה עם הגדולה. זה כשנכנסין כתלי אורך הקטנה לתוך הגדולה ומופלגין שם שלשה טפחים מכתלי אורך הגדולה; ואם לא כן — אף הקטנה ניתרת (שו״ע הרב שע״ד:ג). אם רק כותל אחד מופלג שלושה טפחים, הביאור הלכה, בשם הגר״א, אוסר את הקטנה; מה דעת הרב במקרה זה — הדף אינו אומר: היעזר ברב שלך. ראה פרק 2, נקודה ה. הפניה: שולחן ערוך אדמו״ר הזקן שע״ד:ג.
  • ⑤ גג קטן שנפרץ במלואו לגג גדול: כשהגגות מוקפים מעקה גבוה עשרה [טפחים], או בלא מעקה כשמחיצות הבית ניכרות משפת הגג, ולא עירבו הבתים יחד — אסור להעלות על הקטן כלים ששבתו בבית אם מחיצותיו נכנסות לתוך הגדול, והגדול מותר (המשנה ברורה מוסיף שאין הגג הקטן יותר מעשר אמות, ס״ק כ״ג). אם אין המחיצות ניכרות משפת הגג — כרמלית הוא, אלא אם כן חלון ד׳ על ד׳ [טפחים] פתוח לו מהבית; ואף בחלון זה, אם לא עירבו יחד, אף הגדול אסור בכלים ששבתו בבית, לפי שכותל הבית אינו ניכר למי שעומד למעלה (שו״ע הרב שע״ד:ד). ראה פרק 2, נקודה ו. הפניה: שולחן ערוך אדמו״ר הזקן שע״ד:ד.

⚠ פרק זה מציג את ההנהגה ההלכתית של חב״ד. לכל שאלה ממשית: להיוועץ ברב שלך.
