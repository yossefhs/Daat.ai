1 · טקסט אדמו"ר הזקן

שולחן ערוך הרב — סימן שי"ג

סימן שי"ג — טִלְטוּל דֶּלֶת וְחַלּוֹן וְהַמַּנְעוּל — וּבוֹ כ"ה סְעִיפִים

סעיף אפְּקָק הַחַלּוֹן, כְּגוֹן לוּחַ אוֹ שְׁאָר כָּל דָּבָר שֶׁסּוֹתְמִין בּוֹ הַחַלּוֹן…

טִלְטוּל דֶּלֶת וְחַלּוֹן וְהַמַּנְעוּל בְּשַׁבָּת וּבוֹ כ"ה סְעִיפִים:

פְּקָק הַחַלּוֹן, כְּגוֹן לוּחַ אוֹ שְׁאָר כָּל דָּבָר שֶׁסּוֹתְמִין בּוֹ הַחַלּוֹן, אֲפִלּוּ הוּא דָבָר שֶׁאֵין עָלָיו תּוֹרַת כְּלִי כְּלָל, וְגַם לֹא פָקַק בּוֹ מֵעוֹלָם בְּחֹל, וּבָא לִפְקוֹק בּוֹ בַּתְּחִלָּה בְּשַׁבָּת — אִם חָשַׁב עָלָיו מִבְּעוֹד יוֹם לִפְקוֹק בּוֹ — הֲרֵי זֶה מֻתָּר, שֶׁעַל יְדֵי מַחֲשָׁבָה זוֹ יָרְדָה עָלָיו תּוֹרַת כְּלִי וְהֻתַּר בְּטִלְטוּל כִּשְׁאָר כָּל הַכֵּלִים. וַאֲפִלּוּ אֲרֻבָּה שֶׁבַּגַּג מֻתָּר לִפְקוֹק בַּפְּקָק הַמֻּתָּר בְּטִלְטוּל. וְאֵין בָּזֶה מִשּׁוּם עֲשִׂיַּת אֹהֶל בְּשַׁבָּת, מִפְּנֵי שֶׁאֵינוֹ עוֹשֶׂה בָּזֶה אֶלָּא תּוֹסָפוֹת לְאֹהֶל, וְהוֹסָפָה זוֹ עֲרַאי הִיא:

סעיף בוְקָנֶה שֶׁהִתְקִינוֹ לִהְיוֹת נוֹעֵל בּוֹ…

וְקָנֶה שֶׁהִתְקִינוֹ לִהְיוֹת נוֹעֵל בּוֹ, שֶׁתּוֹחֲבוֹ בַּכֹּתֶל אֵצֶל הַדֶּלֶת — אֵין צָרִיךְ שֶׁיִּקְשְׁרֶנּוּ בַּדֶּלֶת מִבְּעוֹד יוֹם, שֶׁאַף שֶׁלֹּא הָיָה קָשׁוּר וְתָלוּי שָׁם מִבְּעוֹד יוֹם וְתוֹחֲבוֹ שָׁם בְּשַׁבָּת — אֵין בּוֹ מִשּׁוּם בּוֹנֶה בְּשַׁבָּת, הוֹאִיל וְאֵינוֹ מְבַטְּלוֹ שָׁם, אֶלָּא עָשׂוּי לְהוֹצִיאוֹ וּלְהַכְנִיסוֹ תָּדִיר. אֲבָל אִם הוּא בְּעִנְיָן שֶׁאָסוּר לִנְעוֹל בּוֹ בְּשַׁבָּת — אָסוּר לְטַלְטְלוֹ כִּי אִם לְצֹרֶךְ גּוּפוֹ אוֹ מְקוֹמוֹ:

סעיף גוְנֶגֶר שֶׁהוּא יָתֵד שֶׁתּוֹחֲבִין אוֹתוֹ בְּחוֹר שֶׁבָּאַסְקֻפָּה…

וְנֶגֶר שֶׁהוּא יָתֵד שֶׁתּוֹחֲבִין אוֹתוֹ בְּחוֹר שֶׁבָּאַסְקֻפָּה לִנְעוֹל הַדֶּלֶת, הוֹאִיל וְתוֹחֲבִין אוֹתוֹ בָּאַסְקֻפָּה לְמַטָּה — דּוֹמֶה הוּא לְבִנְיָן יוֹתֵר מִקָּנֶה שֶׁתּוֹחֲבִין בַּכֹּתֶל מִן הַצַּד, וְאֵין נוֹעֲלִים בּוֹ אֶלָּא צָרִיךְ שֶׁיִּהְיֶה קָשׁוּר שָׁם מִבְּעוֹד יוֹם, שֶׁהוֹאִיל וְהוּא קָשׁוּר וּמְחֻבָּר שָׁם מִבְּעוֹד יוֹם — אֵינוֹ נִרְאֶה כְּבוֹנֶה בְּשַׁבָּת:

סעיף דכֵּיצַד יְהֵא קָשׁוּר? אִם יֵשׁ בְּרֹאשׁוֹ גְּלֻסְטְרָא…

כֵּיצַד יְהֵא קָשׁוּר? אִם יֵשׁ בְּרֹאשׁוֹ גְּלֻסְטְרָא, דְּהַיְנוּ שֶׁרֹאשׁוֹ עָב וְרָאוּי לְבוּכְנָא לִכְתּוֹשׁ וְלִשְׁחוֹק בּוֹ פִּלְפְּלִין, הוֹאִיל וְהוּא דוֹמֶה לִכְלִי — אֵינוֹ נִרְאֶה כָּל כָּךְ כְּבוֹנֶה. וּלְפִיכָךְ אֵין צָרִיךְ שֶׁיְּהֵא הַקֶּשֶׁר אַמִּיץ, אֶלָּא אֲפִלּוּ אִם קָשׁוּר בְּחֶבֶל דַּק — דַּי בְּכָךְ:

סעיף הוְאִם אֵין בְּרֹאשׁוֹ גְּלֻסְטְרָא, אִם הוּא קָשׁוּר בַּדֶּלֶת עַצְמָהּ…

וְאִם אֵין בְּרֹאשׁוֹ גְּלֻסְטְרָא, אִם הוּא קָשׁוּר בַּדֶּלֶת עַצְמָהּ, אֲפִלּוּ קָשׁוּר בְּחֶבֶל דַּק שֶׁאֵינוֹ רָאוּי לִנָּטֵל בּוֹ — דַּי בְּכָךְ. אֲבָל אִם אֵינוֹ קָשׁוּר בַּדֶּלֶת עַצְמָהּ אֶלָּא בַּבְּרִיחַ — צָרִיךְ שֶׁיְּהֵא הַקֶּשֶׁר אַמִּיץ בְּעִנְיָן שֶׁיְּהֵא רָאוּי לִנָּטֵל בּוֹ:

סעיף ווְכָל זֶה כְּשֶׁהָאַסְקֻפָּה הִיא גְבוֹהָה…

וְכָל זֶה כְּשֶׁהָאַסְקֻפָּה הִיא גְבוֹהָה וְאֵין הַחוֹר שֶׁלָּהּ נוֹקֵב עַד לָאָרֶץ, אֲבָל אִם נִפְחַת הַחוֹר מִתַּחְתָּיו וְהִגִּיעַ לָאָרֶץ, בְּעִנְיָן שֶׁכְּשֶׁמַּכְנִיס הַנֶּגֶר בַּחוֹר נוֹקֵב הַנֶּגֶר תַּחְתָּיו בָּאָרֶץ — הֲרֵי זֶה נִרְאֶה כְּבוֹנֶה, וְאָסוּר לִנְעוֹל בּוֹ בְּכָל עִנְיָן:

סעיף זנֶגֶר שֶׁתָּחַב בּוֹ בֵּית יָד בְּאֶמְצִיעוּתוֹ…

נֶגֶר שֶׁתָּחַב בּוֹ בֵּית יָד בְּאֶמְצִיעוּתוֹ, וַהֲרֵי הוּא דוֹמֶה לְמַקֶּבֶת שֶׁהוּא כְּלִי — מֻתָּר לִנְעוֹל בּוֹ אַף אִם הוּא נוֹקֵב בָּאָרֶץ, וְאַף שֶׁאֵינוֹ קָשׁוּר כְּלָל, שֶׁאֵינוֹ נִרְאֶה כְּבוֹנֶה, הוֹאִיל וְנִכָּר בּוֹ שֶׁהוּא כְּלִי:

סעיף חכָּל פֶּתַח שֶׁאֵינוֹ עָשׂוּי לִכְנִיסָה וְלִיצִיאָה תָּדִיר…

כָּל פֶּתַח שֶׁאֵינוֹ עָשׂוּי לִכְנִיסָה וְלִיצִיאָה תָּדִיר אֶלָּא לְעִתִּים רְחוֹקִים — אִם יֵשׁ צִיר בִּדְלָתוֹת אֵלּוּ, אוֹ אֲפִלּוּ אֵין בָּהֶם עַתָּה אֶלָּא שֶׁהָיָה לָהֶן צִיר וְנִשְׁבַּר וּמְקוֹמוֹ נִכָּר — מֻתָּר לִנְעוֹל בָּהֶן בְּשַׁבָּת. אֲבָל אִם אֵין לָהֶן הֶכֵּר צִיר — אָסוּר לִנְעוֹל בָּהֶן בְּשַׁבָּת:

סעיף טוַאֲפִלּוּ כְּשֶׁיֵּשׁ לָהֶן צִיר, לֹא הִתִּירוּ לִנְעוֹל בָּהֶן אֶלָּא כְּשֶׁקְּשָׁרָן…

וַאֲפִלּוּ כְּשֶׁיֵּשׁ לָהֶן צִיר, לֹא הִתִּירוּ לִנְעוֹל בָּהֶן אֶלָּא כְּשֶׁקְּשָׁרָן וּתְלָאָן שָׁם מִבְּעוֹד יוֹם לִנְעוֹל בָּהֶן. וְאִם קְשָׁרוֹ וּקְבָעוֹ שָׁם בְּעִנְיָן שֶׁאַף כְּשֶׁפּוֹתְחָן אֵינָן נִגְרָרוֹת בָּאָרֶץ — מֻתָּר לִנְעוֹל בָּהֶם אַף אִם אֵין לָהֶן הֶכֵּר צִיר:

סעיף יוּפֶתַח הֶעָשׂוּי לִכְנִיסָה וְלִיצִיאָה תָּדִיר — מֻתָּר לְנָעֳלוֹ…

וּפֶתַח הֶעָשׂוּי לִכְנִיסָה וְלִיצִיאָה תָּדִיר — מֻתָּר לְנָעֳלוֹ אֲפִלּוּ בְּדֶלֶת שֶׁאֵין לָהּ הֶכֵּר צִיר וְהִיא נִגְרֶרֶת בָּאָרֶץ כְּשֶׁפּוֹתְחָהּ. לְפִי שֶׁכֵּיוָן שֶׁעֲשׂוּיָה לְהִפָּתֵחַ תָּדִיר — אֵינוֹ עוֹשֶׂה בִּנְעִילָתָהּ מְחִצָּה קְבוּעָה אֶלָּא עֲרַאי, וְאֵין אִסּוּר כְּלָל לַעֲשׂוֹת מְחִצַּת עֲרַאי בְּשַׁבָּת:

סעיף יאוְכָל זֶה בְּדֶלֶת שֶׁיֵּשׁ לָהּ אַסְקֻפָּה הַתַּחְתּוֹנָה…

וְכָל זֶה בְּדֶלֶת שֶׁיֵּשׁ לָהּ אַסְקֻפָּה הַתַּחְתּוֹנָה. אֲבָל אִם אֵין לָהּ אַסְקֻפָּה, וּכְשֶׁנּוֹעֲלִין בָּהּ מוֹרִידִין אוֹתָהּ עַל הָאָרֶץ, וּכְשֶׁפּוֹתְחִין שׁוֹמְטִין אוֹתָהּ וְעוֹקְרִין אוֹתָהּ, הוֹאִיל וְאֵין לָהּ צוּרַת הַפֶּתַח — אֵין נוֹעֲלִין בָּהּ. וְיֵשׁ מַתִּירִין בְּפֶתַח הֶעָשׂוּי לִכְנִיסָה וִיצִיאָה תָּדִיר. וְיֵשׁ לְהָקֵל בְּדִבְרֵי סוֹפְרִים:

סעיף יבוְכֵן דֶּלֶת הֶעָשׂוּי מִלּוּחַ אֶחָד…

וְכֵן דֶּלֶת הֶעָשׂוּי מִלּוּחַ אֶחָד, וּכְשֶׁפּוֹתְחִין שׁוֹמְטִין אוֹתָהּ — אֵין נוֹעֲלִין בָּהּ, אֲפִלּוּ יֵשׁ לָהּ צִיר. וְיֵשׁ מַתִּירִין בְּפֶתַח הֶעָשׂוּי לִכְנִיסָה וִיצִיאָה תָּדִיר:

סעיף יגאֲבָל דֶּלֶת הָעֲשׂוּיָה מִקְּרָשִׁים הַרְבֵּה…

אֲבָל דֶּלֶת הָעֲשׂוּיָה מִקְּרָשִׁים הַרְבֵּה וְיֵשׁ לָהּ אַסְקֻפָּה — מֻתָּר לִנְעוֹל בָּהּ לְדִבְרֵי הַכֹּל בְּפֶתַח הֶעָשׂוּי לִכְנִיסָה וִיצִיאָה תָּדִיר:

סעיף ידכָּל דַּלְתוֹת הַכֵּלִים — אִם יָצָא צִיר הַתַּחְתּוֹן כֻּלּוֹ…

כָּל דַּלְתוֹת הַכֵּלִים, כְּגוֹן שִׁדָּה תֵּבָה וּמִגְדָּל — אִם יָצָא צִיר הַתַּחְתּוֹן כֻּלּוֹ מִמְּקוֹמוֹ — אָסוּר לְהַחֲזִירוֹ, גְּזֵרָה שֶׁמָּא יִתְקַע. אֲבָל אִם יָצָא מִקְצָתוֹ — דּוֹחֲקוֹ עַד שֶׁמַּחֲזִירוֹ לִמְקוֹמוֹ. אֲבָל אִם יָצָא צִיר הָעֶלְיוֹן, אֲפִלּוּ מִקְצָתוֹ — אָסוּר:

סעיף טווַאֲפִלּוּ דֶּלֶת שֶׁאֵין לָהּ צִיר לֹא לְמַעְלָה וְלֹא לְמַטָּה…

וַאֲפִלּוּ דֶּלֶת שֶׁאֵין לָהּ צִיר לֹא לְמַעְלָה וְלֹא לְמַטָּה אֶלָּא בְּאֶמְצָעָהּ — אַף עַל פִּי כֵן אָסוּר לְהַחֲזִיר צִיר זֶה לְחוֹרוֹ בְּשַׁבָּת, גְּזֵרָה מִשּׁוּם צִיר שֶׁאֵינוֹ בְּאֶמְצַע הַדֶּלֶת. אֲבָל בְּיוֹם טוֹב מֻתָּר, מִשּׁוּם שִׂמְחַת יוֹם טוֹב:

סעיף טזוְכָל זֶה לְהַחֲזִיר הַדְּלָתוֹת, אֲבָל לִטּוֹל הַדְּלָתוֹת מֵהַכֵּלִים…

וְכָל זֶה לְהַחֲזִיר הַדְּלָתוֹת, אֲבָל לִטּוֹל הַדְּלָתוֹת מֵהַכֵּלִים — מֻתָּר בְּכָל עִנְיָן, אִם אֵינָן תְּקוּעִין בָּהֶן בְּחֹזֶק, וְאֵין בָּזֶה מִשּׁוּם סוֹתֵר, מִפְּנֵי שֶׁאֵין בִּנְיָן וּסְתִירָה בְּכֵלִים:

סעיף יזאֲבָל דֶּלֶת שֶׁל בַּיִת וְשֶׁל בּוֹר וְדוּת…

אֲבָל דֶּלֶת שֶׁל בַּיִת וְשֶׁל בּוֹר וְדוּת, אוֹ שֶׁל כֵּלִים הַמְחֻבָּרִים בַּקַּרְקַע — אָסוּר בֵּין לִטּוֹל בֵּין לְהַחֲזִיר, שֶׁכָּל דָּבָר הַמְחֻבָּר בַּקַּרְקַע יֵשׁ בּוֹ בִּנְיָן וּסְתִירָה:

סעיף יחוַאֲפִלּוּ כֵּלִים שֶׁאֵינָם מְחֻבָּרִין לַקַּרְקַע…

וַאֲפִלּוּ כֵּלִים שֶׁאֵינָם מְחֻבָּרִין לַקַּרְקַע, אִם הֵם גְּדוֹלִים כָּל כָּךְ שֶׁמַּחֲזִיקִין אַרְבָּעִים סְאָה — הֲרֵי הֵם כְּאֹהֶל, וְיֵשׁ בָּהֶן בִּנְיָן וּסְתִירָה:

סעיף יטלֹא אָמְרוּ שֶׁאֵין בִּנְיָן וּסְתִירָה בְּכֵלִים אֶלָּא בְּבִנְיָן גָּרוּעַ…

לֹא אָמְרוּ שֶׁאֵין בִּנְיָן וּסְתִירָה בְּכֵלִים אֶלָּא בְּבִנְיָן גָּרוּעַ וּסְתִירַת בִּנְיָן גָּרוּעַ. אֲבָל הָעוֹשֶׂה כְּלִי מִתְּחִלָּתוֹ — הֲרֵי זֶה בִּנְיָן גָּמוּר, וְחַיָּב מִשּׁוּם בּוֹנֶה. וְהַשּׁוֹבֵר כְּלִי שָׁלֵם — הֲרֵי זוֹ סְתִירַת בִּנְיָן גָּמוּר. וְכֵן כָּל הַתּוֹקֵעַ עֵץ בְּעֵץ, בֵּין שֶׁתָּקַע בְּמַסְמֵר בֵּין שֶׁתָּקַע בָּעֵץ עַצְמוֹ עַד שֶׁנִּתְאַחֵד — הֲרֵי זֶה תּוֹלֶדֶת בּוֹנֶה וְחַיָּב:

סעיף כוַאֲפִלּוּ בִּכְלִי שֶׁל פְּרָקִים, אִם תָּקַע פְּרָקָיו…

וַאֲפִלּוּ בִּכְלִי שֶׁל פְּרָקִים, אִם תָּקַע פְּרָקָיו, דְּהַיְנוּ שֶׁהִדְּקָם בְּחֹזֶק בְּעִנְיָן שֶׁצָּרִיךְ לָזֶה גְּבוּרָה וְאֻמָּנוּת — הֲרֵי זֶה בִּנְיָן גָּמוּר, וְחַיָּב מִשּׁוּם בּוֹנֶה אִם עָשׂוּי לְהִתְקַיֵּם הַרְבֵּה:

סעיף כאלְפִיכָךְ, כָּל כְּלִי שֶׁל פְּרָקִים שֶׁנִּתְפָּרֵק…

לְפִיכָךְ, כָּל כְּלִי שֶׁל פְּרָקִים שֶׁנִּתְפָּרֵק — אָסְרוּ חֲכָמִים לְהַחֲזִירוֹ בְּשַׁבָּת, אֲפִלּוּ אֵינוֹ מְהַדְּקוֹ אֶלָּא מְעַט, גְּזֵרָה שֶׁמָּא יִתְקַע בְּחֹזֶק. אֲבָל אִם אֵינוֹ מְהַדְּקוֹ כְּלָל אֶלָּא מַנִּיחוֹ רָפוּי לְגַמְרֵי — מֻתָּר. וְהוּא שֶׁדַּרְכּוֹ שֶׁל כְּלִי לִהְיוֹת כָּךְ רָפוּי לְעוֹלָם:

סעיף כבכָּל כְּלִי שֶׁל פְּרָקִים שֶׁהוּא מְהֻדָּק בְּחֹזֶק…

כָּל כְּלִי שֶׁל פְּרָקִים שֶׁהוּא מְהֻדָּק בְּחֹזֶק בְּעִנְיָן שֶׁהָיָה חַיָּב מִשּׁוּם הִדּוּק זֶה — אָסוּר לְפָרְקוֹ בְּשַׁבָּת מִשּׁוּם סוֹתֵר. אֲבָל אִם הוּא מְהֻדָּק קְצָת — אֵין בּוֹ גַם כֵּן מִשּׁוּם סוֹתֵר, וּמֻתָּר לְפָרְקוֹ לְכַתְּחִלָּה:

סעיף כגקוֹרָה מִתִּקְרַת הַבַּיִת שֶׁנִּשְׁבְּרָה…

קוֹרָה מִתִּקְרַת הַבַּיִת שֶׁנִּשְׁבְּרָה — מֻתָּר לְסָמְכָהּ בְּסַפְסָל אוֹ בַּאֲרֻכּוֹת הַמִּטָּה שֶׁהֵם כְּלִי, לֹא כְּדֵי שֶׁתַּעֲלֶה, שֶׁהֲרֵי זֶה בּוֹנֶה, אֶלָּא כְּדֵי שֶׁלֹּא תוֹסִיף לָרֶדֶת יוֹתֵר. וְצָרִיךְ שֶׁיְּהֵא הַכְּלִי רָפוּי קְצָת תַּחַת הַקּוֹרָה:

סעיף כדהַמַּשְׁוֶה פְּנֵי הַקַּרְקַע בַּבַּיִת אוֹ בֶּחָצֵר…

הַמַּשְׁוֶה פְּנֵי הַקַּרְקַע בַּבַּיִת אוֹ בֶּחָצֵר, כְּגוֹן שֶׁהִשְׁפִּיל תֵּל אוֹ מִלָּא גוּמָא אוֹ גַיְא — הֲרֵי זֶה בּוֹנֶה, וְחַיָּב. לְפִיכָךְ, חָצֵר שֶׁנִּתְקַלְקְלָה בְּמֵימֵי גְשָׁמִים — מֵבִיא תֶּבֶן וְזוֹרֶה בָּהּ, שֶׁהַתֶּבֶן אֵינוֹ מְבַטְּלוֹ שָׁם:

סעיף כהכְּשֶׁזּוֹרֶה הַתֶּבֶן בְּחָצֵר שֶׁנִּתְקַלְקְלָה…

כְּשֶׁזּוֹרֶה הַתֶּבֶן בְּחָצֵר שֶׁנִּתְקַלְקְלָה — לֹא יִזְרֶה לֹא בְסַל וְלֹא בְקֻפָּה, אֶלָּא בְּשׁוּלֵי הַקֻּפָּה, שֶׁיְּהַפְּכֶנּוּ וְיָבִיא הַתֶּבֶן עַל שׁוּלֶיהָ, כְּדֵי לְשַׁנּוֹת מִדֶּרֶךְ הַחֹל:

מקור: ספריא — שולחן ערוך הרב, אורח חיים שי"ג.
