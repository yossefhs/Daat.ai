דעת הרב — שולחן ערוך אדמו"ר הזקן על סימן ש"י

שיטת אדמו"ר הזקן על כ' הסעיפים של דיני מוקצה — סיום בלוק המוקצה. מסגרת מושגית: ההכנה כעיקרון הקצאה.

מדוע רמה 4 מוקדשת לאדמו"ר הזקן? שולחן ערוך אדמו"ר הזקן אינו פירוש על המחבר — זהו שולחן ערוך עצמאי ושלם, המשלב הלכה + טעמי המצוות + ממד פנימי.

עבור חב"ד, אדמו"ר הזקן הוא הפוסק האחרון. עמוד זה מקבץ, עבור סימן ש"י, את הטקסט המלא של הרב.

1 · טקסט אדמו"ר הזקן

שולחן ערוך הרב — סימן ש"י

סימן ש"י — דִּינֵי מֻקְצֶה בְּשַׁבָּת — וּבוֹ כ' סְעִיפִים
מקור: מהדורת קה"ת. כ' סעיפים + 0 הערות קונטרס אחרון.

סעיף א דִּינֵי מֻקְצֶה בְּשַׁבָּת וּבוֹ כ' סְעִיפִים: כְּבָר נִתְבָּאֵר בְּסִמָּן ש"ח שֶׁמֻּקְצֶה מֵחֲמַת מִאוּס…

דִּינֵי מֻקְצֶה בְּשַׁבָּת וּבוֹ כ' סְעִיפִים:

כְּבָר נִתְבָּאֵר בְּסִמָּן ש"ח שֶׁמֻּקְצֶה מֵחֲמַת מִאוּס מֻתָּר, לְפִיכָךְ עֵץ שֶׁתּוֹלִין בּוֹ דָּגִים לְיַבְּשָׁם, אַף עַל פִּי שֶׁהוּא מָאוּס — מֻתָּר לְטַלְטְלוֹ:

סעיף ב כָּל דָּבָר הָרָאוּי לַאֲכִילָה בְּשַׁבָּת — אֵין בּוֹ דִּין מֻקְצֶה…

כָּל דָּבָר הָרָאוּי לַאֲכִילָה בְּשַׁבָּת — אֵין בּוֹ דִּין מֻקְצֶה, אֲפִלּוּ אִם הוּא עוֹמֵד לִסְחוֹרָה, כְּגוֹן תְּמָרִים וּשְׁקֵדִים וּשְׁאָר פֵּרוֹת… חוּץ מִגְּרוֹגָרוֹת וְצִמּוּקִים, שֶׁאָסוּר לֶאֱכוֹל מֵהֶם אוֹ לְטַלְטְלָן מִשֶּׁהִנִּיחָם בַּמֻּקְצֶה עַד לְאַחַר שֶׁנִּתְיַבְּשׁוּ לְגַמְרֵי…

סעיף ג בַּמֶּה דְּבָרִים אֲמוּרִים? בִּגְרוֹגָרוֹת וְצִמּוּקִים שֶׁל יִשְׂרָאֵל…

בַּמֶּה דְּבָרִים אֲמוּרִים? בִּגְרוֹגָרוֹת וְצִמּוּקִים שֶׁל יִשְׂרָאֵל, אֲבָל גְּרוֹגָרוֹת וְצִמּוּקִים שֶׁל נָכְרִי — מֻתָּרִין אִם הֵם רְאוּיִין וְאֵינָן רְאוּיִין, שֶׁאֵין דִּין מֻקְצֶה בְּשֶׁל נָכְרִי, לְפִי שֶׁהַכֹּל מוּכָן אֶצְלוֹ.

סעיף ד כָּל דָּבָר שֶׁהָיָה אָסוּר בְּטִלְטוּל בְּבֵין הַשְּׁמָשׁוֹת…

כָּל דָּבָר שֶׁהָיָה אָסוּר בְּטִלְטוּל בְּבֵין הַשְּׁמָשׁוֹת וּבְשַׁבָּת הָלַךְ לוֹ הַדָּבָר הַגּוֹרֵם לוֹ הָאִסּוּר… אַף עַל פִּי כֵן אָסוּר לְטַלְטְלוֹ כָּל הַשַּׁבָּת כֻּלָּהּ. בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁדְּחָאוֹ הָאָדָם בְּיָדַיִם…

סעיף ה וְכֵן דָּבָר שֶׁלֹּא הָיָה רָאוּי בְּבֵין הַשְּׁמָשׁוֹת מֵחֲמַת עַצְמוֹ…

וְכֵן דָּבָר שֶׁלֹּא הָיָה רָאוּי בְּבֵין הַשְּׁמָשׁוֹת מֵחֲמַת עַצְמוֹ וְלֹא מֵחֲמַת אֵיזֶה אִסּוּר וְנַעֲשָׂה רָאוּי מִשֶּׁחָשֵׁכָה, שֶׁהוּא אָסוּר בְּטִלְטוּל כָּל הַשַּׁבָּת כֻּלָּהּ אִם נִדְחָה בִּידֵי אָדָם…

סעיף ו אֲבָל דָּבָר שֶׁהָיָה מֻקְצֶה בְּבֵין הַשְּׁמָשׁוֹת בְּעַל כָּרְחוֹ שֶׁל אָדָם…

אֲבָל דָּבָר שֶׁהָיָה מֻקְצֶה בְּבֵין הַשְּׁמָשׁוֹת בְּעַל כָּרְחוֹ שֶׁל אָדָם — לֹא הֻקְצָה לְכָל הַשַּׁבָּת כֻּלָּהּ, כְּגוֹן דָּבָר שֶׁהָיָה מֻקְצֶה מֵחֲמַת אִסּוּר הַבָּא מֵאֵלָיו…

סעיף ז וְכָל זֶה בְּדָבָר שֶׁהָיָה מֻקְצֶה כָּל בֵּין הַשְּׁמָשׁוֹת…

וְכָל זֶה בְּדָבָר שֶׁהָיָה מֻקְצֶה כָּל בֵּין הַשְּׁמָשׁוֹת, אֲבָל דָּבָר שֶׁהָיָה רָאוּי בְּבֵין הַשְּׁמָשׁוֹת אֲפִלּוּ בְּמִקְצָתוֹ לְבָד וְאַחַר כָּךְ נִתְקַלְקֵל בְּעִנְיָן שֶׁנֶּאֱסַר…

סעיף ח גְּרוֹגָרוֹת וְצִמּוּקִים שֶׁהָיוּ מֻקְצִים מִפְּנֵי שֶׁאֵינָן רְאוּיִין…

גְּרוֹגָרוֹת וְצִמּוּקִים שֶׁהָיוּ מֻקְצִים מִפְּנֵי שֶׁאֵינָן רְאוּיִין, וּכְשֶׁהִגִּיעַ בֵּין הַשְּׁמָשׁוֹת נִתְיַבְּשׁוּ כְּבָר וַהֲרֵי הֵם רְאוּיִים לַאֲכִילָה… מֻתָּרִין בַּאֲכִילָה.

סעיף ט גְּרוֹגָרוֹת וְצִמּוּקִים שֶׁיָּבְשׁוּ קְצָת בְּעִנְיָן שֶׁהֵם רְאוּיִים וְאֵינָן רְאוּיִין…

גְּרוֹגָרוֹת וְצִמּוּקִים שֶׁיָּבְשׁוּ קְצָת בְּעִנְיָן שֶׁהֵם רְאוּיִים וְאֵינָן רְאוּיִין, אִם הִזְמִינָם לַאֲכִילָה מִבְּעוֹד יוֹם — מֻתָּר לֶאֱכוֹל מֵהֶן בְּשַׁבָּת.

סעיף י כָּל דָּבָר שֶׁאָסוּר לְטַלְטְלוֹ — אָסוּר לִתֵּן תַּחְתָּיו כְּלִי כְּדֵי שֶׁיִּפּוֹל לְתוֹכוֹ…

כָּל דָּבָר שֶׁאָסוּר לְטַלְטְלוֹ — אָסוּר לִתֵּן תַּחְתָּיו כְּלִי כְּדֵי שֶׁיִּפּוֹל לְתוֹכוֹ, אֲפִלּוּ אִם דַּעְתּוֹ לְנַעֲרוֹ אַחַר כָּךְ, מִפְּנֵי שֶׁאוֹסֵר הַכְּלִי בְּטִלְטוּל… אֲבָל מֻתָּר לִכְפּוֹת עָלָיו כְּלִי.

סעיף יא כָּל דָּבָר הָאָסוּר בְּטִלְטוּל שֶׁהִנִּיחוּ מִבְּעוֹד יוֹם עַל גַּבֵּי דָבָר הַמֻּתָּר…

כָּל דָּבָר הָאָסוּר בְּטִלְטוּל שֶׁהִנִּיחוּ מִבְּעוֹד יוֹם עַל גַּבֵּי דָבָר הַמֻּתָּר בְּטִלְטוּל עַל דַּעַת שֶׁיִּשָּׁאֵר מֻנָּח שָׁם בְּשַׁבָּת… אַף עַל פִּי כֵן אָסוּר לְטַלְטְלוֹ כָּל הַשַּׁבָּת כֻּלָּהּ.

סעיף יב וְכָל זֶה בְּמִטָּה שֶׁאֵינָהּ מְיֻחֶדֶת לְמָעוֹת…

וְכָל זֶה בְּמִטָּה שֶׁאֵינָהּ מְיֻחֶדֶת לְמָעוֹת, אוֹ אֲפִלּוּ יִחֲדָהּ לְמָעוֹת אֶלָּא שֶׁעֲדַיִן לֹא הִנִּיחַ עָלֶיהָ מָעוֹת מֵעוֹלָם…

סעיף יג וּמִכָּל מָקוֹם, כִּיס שֶׁל מָעוֹת הַתָּפוּר בְּבֶגֶד…

וּמִכָּל מָקוֹם, כִּיס שֶׁל מָעוֹת הַתָּפוּר בְּבֶגֶד, הוֹאִיל וְעִקַּר הַבֶּגֶד עוֹמֵד לִלְבּוֹשׁ, אִם הוֹצִיא הַמָּעוֹת מִשָּׁם — יוּכַל לִלְבּוֹשׁ הַבֶּגֶד.

סעיף יד וַאֲפִלּוּ אִם יֵשׁ מָעוֹת בְּתוֹכוֹ…

וַאֲפִלּוּ אִם יֵשׁ מָעוֹת בְּתוֹכוֹ — אֵין אִסּוּר אֶלָּא לִלְבּוֹשׁ אֶת הַבֶּגֶד, גְּזֵרָה שֶׁמָּא יִשְׁכַּח וְיֵצֵא בּוֹ לִרְשׁוּת הָרַבִּים.

סעיף טו וְכָל זֶה בְּכִיס שֶׁפִּיו בִּלְבָד תָּפוּר בְּחוֹר שֶׁבַּבֶּגֶד…

וְכָל זֶה בְּכִיס שֶׁפִּיו בִּלְבָד תָּפוּר בְּחוֹר שֶׁבַּבֶּגֶד וְהוּא כֻּלּוֹ תָּלוּי, אֲבָל אִם כֻּלּוֹ תָּלוּי לְאָרְכּוֹ בַּבֶּגֶד — נַעֲשָׂה כָּל הַבֶּגֶד בָּסִיס לְמָעוֹת.

סעיף טז כְּבָר נִתְבָּאֵר בְּסִמָּן ש"ט שֶׁכְּלִי שֶׁנַּעֲשָׂה בָּסִיס לְאִסּוּר וּלְהֶתֵּר…

כְּבָר נִתְבָּאֵר בְּסִמָּן ש"ט שֶׁכְּלִי שֶׁנַּעֲשָׂה בָּסִיס לְאִסּוּר וּלְהֶתֵּר — מֻתָּר לְטַלְטְלוֹ. וְהוּא שֶׁדָּבָר הַמֻּתָּר חָשׁוּב מִדָּבָר הָאָסוּר.

סעיף יז וַאֲפִלּוּ אִם דָּבָר הַמֻּתָּר חָשׁוּב יוֹתֵר, אִם אֶפְשָׁר לְנַעֵר הָאִסּוּר…

וַאֲפִלּוּ אִם דָּבָר הַמֻּתָּר חָשׁוּב יוֹתֵר, אִם אֶפְשָׁר לְנַעֵר הָאִסּוּר בִּלְבַדּוֹ — צָרִיךְ לְנַעֵר מִתְּחִלָּה.

סעיף יח וְכָל זֶה כְּשֶׁהָיָה הַהֶתֵּר שָׁם עִם הָאִסּוּר מִבְּעוֹד יוֹם…

וְכָל זֶה כְּשֶׁהָיָה הַהֶתֵּר שָׁם עִם הָאִסּוּר מִבְּעוֹד יוֹם, שֶׁבְּבֵין הַשְּׁמָשׁוֹת נַעֲשָׂה הַכְּלִי בָּסִיס לְאִסּוּר וּלְהֶתֵּר.

סעיף יט אֵין הָאֵפֶר מֻתָּר בְּטִלְטוּל אֶלָּא כְּשֶׁהֻסַּק מִבְּעוֹד יוֹם…

אֵין הָאֵפֶר מֻתָּר בְּטִלְטוּל אֶלָּא כְּשֶׁהֻסַּק מִבְּעוֹד יוֹם, אֲבָל אִם הֻסַּק בְּשַׁבָּת — אָסוּר מִשּׁוּם נוֹלָד.

סעיף כ תֵּבָה שֶׁיֵּשׁ בָּהּ דְּבָרִים הַמֻּתָּרִים בְּטִלְטוּל וּמָעוֹת…

תֵּבָה שֶׁיֵּשׁ בָּהּ דְּבָרִים הַמֻּתָּרִים בְּטִלְטוּל וּמָעוֹת, אִם הַמָּעוֹת אֵינָן עִקָּר — מֻתָּר לְטַלְטְלָהּ כְּמוֹת שֶׁהִיא עַל פִּי הַתְּנָאִים שֶׁנִּתְבָּאֲרוּ:
