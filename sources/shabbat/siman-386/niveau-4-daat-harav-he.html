דעת הרב

דעת הרב — שולחן ערוך אדמו״ר הזקן על סימן שפ״ו

שיתוף מבואות, המשתף את חצרות המבוי: למה אינו בפת בלבד, היכן מניחים אותו, מקומות שמפסיק ביניהם מקום האסור בטלטול, במה משתתפים ושיעורו, אוכל שנדר ממנו, וחצר הפתוחה לשני מבואות. תשעת הסעיפים של שולחן ערוך אדמו״ר הזקן, מושווים לתשעת הסעיפים של המחבר.

מדוע רמה 4 מוקדשת לאדמו״ר הזקן? שולחן ערוך אדמו״ר הזקן אינו פירוש על המחבר: זהו שולחן ערוך בפני עצמו, שנכתב על ידי אדמו״ר הזקן, בחלוקה ובמספור סעיפים משלו. כאן מונה שולחן ערוך הרב תשעה סעיפים בראש הסימן, וגם בשולחן ערוך של המחבר תשעה; ספריא מגישה את לשון הרב בתשעה קטעים, והדף מביא וממספר אותם כמות שהם (פרק 1). אבל החלוקה אינה שווה: סעיף ד של הרב עומד מול סעיף ד של המחבר ומול תחילת סעיף ה שלו (המאכלים שאין משתתפים בהם); סעיף ה של הרב — מול סוף סעיף ה (מים ומלח); סעיף ח של הרב — מול סעיף ח של המחבר ומול תחילת סעיף ט שלו (ככר שאמר עליה שתהא קודש); סעיף ט של הרב — מול המשך סעיף ט (חצר הפתוחה לשני מבואות). סעיפים א, ב, ג, ו ו־ז של הרב עומדים מול סעיפים א, ב, ג, ו ו־ז של המחבר. בהפניות שבסוגריים שני המספורים באותיות, וההפניה מציינת תמיד את הספר (שו״ע או״ח למחבר, רמ״א להגהת הרמ״א, שו״ע הרב לרב).

עמוד זה מקבץ, לסימן שפ״ו · 386, את הטקסט המלא של הרב, את מה ששונה בלשונו מדברי המחבר והרמ״א, ואת ההנהגה המעשית העולה מדבריו.

שלחן ערוך הרב — סימן שפ״ו

הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן שפ״ו — דִּינֵי שִׁתּוּף בְּעֵרוּב — וּבוֹ ט׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. בכותרת: ט׳ סעיפים; ספריא מגישה את הלשון בתשעה קטעים (גם בשולחן ערוך של המחבר תשעה סעיפים, במספור ובחלוקה משלו).

סעיף א דִּינֵי שִׁתּוּף בְּעֵרוּב וּבוֹ ט' סְעִיפִים: מָבוֹי הַמְּתֻקָּן בְּלֶחִי…

דִּינֵי שִׁתּוּף בְּעֵרוּב וּבוֹ ט' סְעִיפִים:

מָבוֹי הַמְּתֻקָּן בְּלֶחִי, וְאֵין צָרִיךְ לוֹמַר בְּצוּרַת פֶּתַח — רְשׁוּת הַיָּחִיד גְּמוּרָה הִיא מִן הַתּוֹרָה, וַאֲפִלּוּ הַמְּתֻקָּן בְּקוֹרָה, שֶׁאֵינָהּ כִּמְחִצָּה אֶלָּא מִשּׁוּם הֶכֵּר בְּעָלְמָא. וּכְשֶׁהַמָּבוֹי פָּתוּחַ לִרְשׁוּת הָרַבִּים אֵינוֹ רְשׁוּת הַיָּחִיד מִן הַתּוֹרָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שמ״ה, מִכָּל מָקוֹם מִן הַתּוֹרָה אֵינוֹ אֶלָּא מְקוֹם פְּטוּר, וּמֻתָּר לְהוֹצִיא וּלְהַכְנִיס מִמֶּנּוּ לַחֲצֵרוֹת אַף עַל פִּי שֶׁהֵן רְשׁוּת הַיָּחִיד גְּמוּרָה. אֶלָּא שֶׁחֲכָמִים אָסְרוּ לְהוֹצִיא וּלְהַכְנִיס מִמֶּנּוּ לַחֲצֵרוֹת, מִפְּנֵי שֶׁהַחֲצֵרוֹת הֵן מְיֻחָדוֹת כָּל אַחַת לִבְעָלֶיהָ וְהַמָּבוֹי מְשֻׁתָּף לְכֻלָּם, וְדוֹמֶה קְצָת לִרְשׁוּת הָרַבִּים, וְאִם יוֹצִיאוּ מֵחֲצֵרוֹת לְמָבוֹי — יֹאמְרוּ שֶׁמֻּתָּר לְהוֹצִיא מֵרְשׁוּת הַיָּחִיד לִרְשׁוּת הָרַבִּים. אֲבָל הִתִּירוּ עַל יְדֵי שִׁתּוּף, שֶׁגּוֹבִין פַּת אוֹ דָּבָר אַחֵר מִמִּינֵי מַאֲכָל אוֹ מַשְׁקֶה מִכָּל חָצֵר וְחָצֵר, וְנוֹתְנִים אוֹתוֹ בְּאַחַת מֵהַחֲצֵרוֹת, וְאָנוּ רוֹאִים כְּאִלּוּ פָּתוּחַ כָּל הַמָּבוֹי לְאוֹתָהּ חָצֵר בִּלְבַדָּהּ, שֶׁכָּל הַחֲצֵרוֹת שֶׁבַּמָּבוֹי נַעֲשׂוּ כְּאַחַת עַל יְדֵי שִׁתּוּף זֶה.

וְאֵין שִׁתּוּף זֶה דּוֹמֶה לְעֵרוּב שֶׁאֵינוֹ אֶלָּא בְּפַת בִּלְבָד, לְפִי שֶׁעֵרוּבֵי חֲצֵרוֹת הוּא מִשּׁוּם דִּירָה, לְעָרֵב דִּירָתָם לַעֲשׂוֹתָן כְּאַחַת, וְדִירָתוֹ שֶׁל אָדָם אֵינָהּ נִמְשֶׁכֶת אֶלָּא אַחַר מְקוֹם פִּתּוֹ. אֲבָל שִׁתּוּף שֶׁבַּמָּבוֹי אֵינוֹ אֶלָּא לְשַׁתֵּף הַחֲצֵרוֹת וְלֹא הַבָּתִּים, וְחָצֵר לָאו בֵּית דִּירָה הִיא. לְפִיכָךְ אֵין צָרִיךְ לְהַנִּיחַ הַשִּׁתּוּף בְּבַיִת הָרָאוּי לְדִירָה כְּמוֹ עֵרוּבֵי חֲצֵרוֹת, אֶלָּא יְכוֹלִים לְהַנִּיחוֹ אֲפִלּוּ בְּבַיִת שֶׁאֵין בּוֹ ד' אַמּוֹת עַל ד' אַמּוֹת, אוֹ בְּבֵית שַׁעַר אַכְסַדְרָה וּמִרְפֶּסֶת, אוֹ אֲפִלּוּ בַּאֲוִיר הֶחָצֵר בְּמָקוֹם הַמִּשְׁתַּמֵּר, כֵּיוָן שֶׁאֵינוֹ אֶלָּא לְשִׁתּוּף הַחֲצֵרוֹת. אֲבָל אֵין מַנִּיחִים אוֹתוֹ בַּאֲוִיר מָבוֹי, מִפְּנֵי שֶׁאֵינוֹ מָקוֹם הַמִּשְׁתַּמֵּר, וְהוּא הַדִּין בְּכָל מָקוֹם שֶׁאֵינוֹ מִשְׁתַּמֵּר:

סעיף ב הַשִּׁתּוּף צָרִיךְ לִהְיוֹת מֻנָּח בְּחָצֵר שֶׁבְּמָבוֹי זֶה…

הַשִּׁתּוּף צָרִיךְ לִהְיוֹת מֻנָּח בְּחָצֵר שֶׁבְּמָבוֹי זֶה, אֲבָל לֹא שֶׁבְּמָבוֹי אַחֵר שֶׁלֹּא נִשְׁתַּתֵּף עִם זֶה. לְפִיכָךְ יִשְׂרְאֵלִים שֶׁדָּרִים בְּב' וְג' מְקוֹמוֹת חֲלוּקִים וּרְחוֹקִים זֶה מִזֶּה, שֶׁמַּפְסִיק בֵּינֵיהֶם מָקוֹם הָאָסוּר בְּטִלְטוּל מִפְּנֵי שֶׁאֵינוֹ מְתֻקָּן בְּלֶחִי אוֹ קוֹרָה אוֹ צוּרַת פֶּתַח, אוֹ מִפְּנֵי שֶׁדָּרִים שָׁם נָכְרִים וְלֹא שָׂכְרוּ מֵהֶם רְשׁוּתָם, וְהַשַּׁמָּשׁ גּוֹבֶה קֶמַח מִכֻּלָּם וְעוֹשֶׂה מִמֶּנּוּ עֵרוּב אֶחָד, וְנוֹתְנוֹ בְּמָקוֹם אֶחָד מֵהֶם — אֵינוֹ מוֹעִיל לָהֶם, אֶלָּא צְרִיכִים לְעָרֵב בְּכָל מָקוֹם וּמָקוֹם בִּפְנֵי עַצְמוֹ (וְאִם עֵרְבוּ בְּמָקוֹם אֶחָד, וּבְמָקוֹם אֶחָד לֹא עֵרְבוּ — יִתְבָּאֵר בְּסִמָּן שצ״ב), שֶׁהֲרֵי אִי אֶפְשָׁר לָהֶם לְהִשְׁתַּתֵּף יַחַד, כֵּיוָן שֶׁאֲסוּרִים לְטַלְטֵל מִזֶּה לְזֶה, וְאִם כֵּן אִי אֶפְשָׁר לְשַׁתֵּף מָקוֹם זֶה וְשֶׁיְּהֵא הַשִּׁתּוּף מֻנָּח בְּמָקוֹם אַחֵר שֶׁאֵינוֹ מְשֻׁתָּף עִמּוֹ:

סעיף ג כְּכָל מִשְׁפְּטֵי עֵרוּבֵי חֲצֵרוֹת וְשִׁעוּרוֹ, וְשֶׁאִשְׁתּוֹ מְעָרֶבֶת…

כְּכָל מִשְׁפְּטֵי עֵרוּבֵי חֲצֵרוֹת וְשִׁעוּרוֹ, וְשֶׁאִשְׁתּוֹ מְעָרֶבֶת בִּשְׁבִילוֹ, וְשֶׁמְּעָרְבִין שֶׁלֹּא לָדַעַת אִם אֵין הַבַּיִת פָּתוּחַ אֶלָּא לְחָצֵר א', וְשֶׁקָּטָן יָכוֹל לִגְבּוֹת אֶת הָעֵרוּב, וְשֶׁיָּכוֹל אֶחָד לִזַכּוֹת לְכֻלָּם, וְשֶׁאִם הָיוּ שֻׁתָּפִים בַּפַּת אֵינָם צְרִיכִים לְעָרֵב — כָּךְ דִּינֵי שִׁתּוּף.

וַאֲפִלּוּ אִם הָיָה מְשֻׁתָּף עִם שְׁכֵנָיו בִּסְחוֹרָה, לְזֶה בְּיַיִן וּלְזֶה בְּשֶׁמֶן — אֵינָם צְרִיכִים לְהִשְׁתַּתֵּף, וְהוּא שֶׁיִּהְיוּ הַשֶּׁמֶן וְהַיַּיִן בִּכְלִי אֶחָד, כְּמוֹ שֶׁעֵרוּבֵי חֲצֵרוֹת צָרִיךְ לִהְיוֹת בִּכְלִי אֶחָד:

סעיף ד מִשְׁתַּתְּפִים בְּכָל מִינֵי מַאֲכָל, וַאֲפִלּוּ ד'…

מִשְׁתַּתְּפִים בְּכָל מִינֵי מַאֲכָל, וַאֲפִלּוּ ד' וְה' מִינֵי מַאֲכָל מִצְטָרְפִין לִמְזוֹן ב' סְעֻדּוֹת, שֶׁהוּא שִׁעוּר הַשִּׁתּוּף בִּתְחִלָּתוֹ אֲפִלּוּ לְאֶלֶף בְּנֵי אָדָם, כְּמוֹ שֶׁנִּתְבָּאֵר בְּעֵרוּבֵי חֲצֵרוֹת.

בְּכָל מִינֵי מַאֲכָל מְעָרְבִין, חוּץ מִגֻּדְגְּדָנִיּוֹת שֶׁהֻקְשׁוּ לְזֶרַע וּתְבָלִין וּפוֹלִין יְבֵשִׁים, וְלֹא בַּעֲלֵי הַבְּצָלִים שֶׁלֹּא גָדְלוּ אֹרֶךְ הַזֶּרֶת, וְלֹא בִּכְמֵהִין וּפִטְרִיּוֹת וְלֹא בְּכַפְנִיּוֹת וְלֹא בַּעֲדָשִׁים חַיִּין וְלֹא בְּחִטִּים וּשְׂעוֹרִים חַיִּין וְלֹא בְּיָרָק שֶׁהוּא בָּשִׁיל וְלֹא בָשִׁיל, מִפְּנֵי שֶׁכָּל אֵלּוּ אֵינָן רְאוּיִים לַאֲכִילָה. אֲבָל בְּיָרָק חַי מִשְׁתַּתְּפִים כְּמוֹ שֶׁיִּתְבָּאֵר, וְהוּא הַדִּין לְפָתוֹת שֶׁלָּנוּ שֶׁנֶּאֱכָלִין גַּם כֵּן חַיִּין, וְאֵינָם גְּרוּעִים מִבָּשָׂר חַי שֶׁמִּשְׁתַּתְּפִים בּוֹ כְּמוֹ שֶׁיִּתְבָּאֵר:

סעיף ה בְּכָל מִינֵי מַשְׁקִים מִשְׁתַּתְּפִים, חוּץ מִן הַמַּיִם…

בְּכָל מִינֵי מַשְׁקִים מִשְׁתַּתְּפִים, חוּץ מִן הַמַּיִם לְבַדָּם וְכֵן מֶלַח לְבַדּוֹ, לְפִי שֶׁאֵינָן זָנִין הַגּוּף כְּלָל, אֲבָל אִם עֵרֵב מַיִם וּמֶלַח בְּיַחַד — מָזוֹן הוּא לְטַבֵּ(ו)ל בּוֹ פִּתּוֹ, וּמִשְׁתַּתְּפִים בָּהֶם:

סעיף ו כָּל דָּבָר שֶׁרְגִילִים לְלַפֵּת בּוֹ אֶת הַפַּת…

כָּל דָּבָר שֶׁרְגִילִים לְלַפֵּת בּוֹ אֶת הַפַּת, שִׁעוּרוֹ כְּדֵי לְלַפֵּת בּוֹ פַּת הַנֶּאֱכָל לְב' סְעֻדּוֹת, שֶׁהֵן כְּשִׁשָּׁה בֵיצִים, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שס״ח. וְכָל דָּבָר שֶׁאֵין מְלַפְּתִין בּוֹ אֶת הַפַּת, שִׁעוּרוֹ כְּדֵי לֶאֱכוֹל מִמֶּנּוּ בְּעַצְמוֹ ב' סְעֻדּוֹת.

וּבָשָׂר חַי אֵינוֹ רָאוּי לְלִפְתָּן, וְצָרִיךְ כְּדֵי לֶאֱכוֹל מִמֶּנּוּ ב' סְעֻדּוֹת. וְהוּא שֶׁיְּהֵא מָלִיחַ, וְהוּא הַדִּין לְדָג מָלִיחַ. אֲבָל בָּשָׂר צָלִי — לִפְתָּן הוּא, וְשִׁעוּרוֹ כְּדֵי לְלַפֵּת בּוֹ ב' סְעֻדּוֹת. וַאֲפִלּוּ אִם יֵשׁ כַּמָּה בְּנֵי אָדָם שֶׁאוֹכְלִים בְּלֹא פַת — בָּטְלָה דַעְתָּם אֵצֶל כָּל אָדָם.

וְחֹמֶץ — שִׁעוּרוֹ כְּדֵי לְטַבֵּל בּוֹ יָרָק מְזוֹן ב' סְעֻדּוֹת, אִם הָיָה אוֹכֵל הַיָּרָק לְבַדּוֹ בְּלֹא פַת.

וְיַיִן מְבֻשָּׁל — לְקִנּוּחַ סְעֻדָּה הוּא בָא (וְלֹא לְטַבֵּל), לְפִיכָךְ שִׁעוּרוֹ כְּדֵי שֶׁמְּבִיאִים מִמֶּנּוּ לְקִנּוּחַ ב' סְעֻדּוֹת. וְכֵן כָּל כַּיּוֹצֵא בָזֶה.

וְיַיִן חַי — שִׁעוּרוֹ ב' רְבִיעִיּוֹת, לִשְׁתּוֹת רְבִיעִית בְּכָל סְעֻדָּה, וְכֵן שִׁעוּר שְׁאָר כָּל הַמַּשְׁקִים:

סעיף ז מִשְׁתַּתְּפִים בִּשְׁנֵי בֵיצִים, אֲפִלּוּ חַיִּין, בִּשְׁנֵי…

מִשְׁתַּתְּפִים בִּשְׁנֵי בֵיצִים, אֲפִלּוּ חַיִּין, בִּשְׁנֵי רִמּוֹנִים, בַּחֲמִשָּׁה אֲפַרְסְקִין (בַּעֲשָׂרָה אֱגוֹזִים וְאֶתְרוֹג אֶחָד), בְּלִטְרָא יָרָק בֵּין חַי בֵּין מְבֻשָּׁל, בְּפוֹלִין לַחִים מְלֹא הַיָּד, בְּתַפּוּחֵי יַעַר מְלֹא הַקַּב (וְהוּא הַדִּין לְתַפּוּחֵי פַרְדֵּס), וְכֵן תְּמָרִים אוֹ גְרוֹגָרוֹת מְלֹא הַקַּב וּדְבֵלָה מִשְׁקַל מָנֶה (וְהוּא לִטְרָא):

סעיף ח מִשְׁתַּתְּפִין אֲפִלּוּ בְּאֹכֶל שֶׁאֵינוֹ רָאוּי לוֹ, אִם…

מִשְׁתַּתְּפִין אֲפִלּוּ בְּאֹכֶל שֶׁאֵינוֹ רָאוּי לוֹ, אִם רָאוּי לְשׁוּם אָדָם מִיִּשְׂרָאֵל, כְּגוֹן לְנָזִיר בְּיַיִן וּלְיִשְׂרָאֵל בִּתְרוּמָה בִּזְמַן שֶׁהָיְתָה נֶאֱכֶלֶת לַכֹּהֲנִים. אֲבָל עַכְשָׁו אֵין תְּרוּמָה טְהוֹרָה רְאוּיָה לְשׁוּם אָדָם, שֶׁכֻּלָּנוּ טְמֵאֵי מֵתִים, שֶׁאֵין לָנוּ אֵפֶר פָּרָה.

וְכֵן הַנּוֹדֵר מֵאֹכֶל אוֹ נִשְׁבַּע שֶׁלֹּא יֹאכְלֶנּוּ מִשְׁתַּתֵּף בּוֹ, כֵּיוָן שֶׁרָאוּי לְיִשְׂרָאֵל אַחֵר. וַאֲפִלּוּ נָדַר וְנִשְׁבַּע שֶׁלֹּא יֵהָנֶה מִמֶּנּוּ לֹא נֶאֱסַר בּוֹ בַּהֲנָאָה זוֹ מַה שֶּׁמִּשְׁתַּתֵּף בּוֹ, לְפִי שֶׁלֹּא נִתְכַּוֵּן אֶלָּא שֶׁלֹּא לֵהָנוֹת מִמֶּנּוּ בְּדֶרֶךְ הֲנָאָתוֹ. וְיֵשׁ אוֹמְרִים שֶׁהַנּוֹדֵר מֵאֹכֶל, אֲפִלּוּ לֹא נָדַר אֶלָּא מֵאֲכִילָתוֹ וְלֹא מֵהֲנָאָתוֹ — אֵינוֹ מִשְׁתַּתֵּף בּוֹ, לְפִי שֶׁהַנּוֹדֵר הוּא שֶׁאוֹסֵר אֶת הַחֵפֶץ עָלָיו, שֶׁאוֹמֵר "קוֹנָם אֹכֶל פְּלוֹנִי עָלַי", דְּהַיְנוּ שֶׁאוֹתוֹ אֹכֶל יְהֵא נֶאֱסַר עָלָיו כְּקָרְבָּן, וּמִתְחַלֵּף הוּא בְּהֶקְדֵּשׁ, שֶׁהַהֶקְדֵּשׁ הוּא גַם כֵּן שֶׁהַחֵפֶץ מִתְקַדֵּשׁ וְנֶאֱסַר עַל הָאָדָם; וּבְהֶקְדֵּשׁ לְדִבְרֵי הַכֹּל אֵין מִשְׁתַּתְּפִים, לְפִי שֶׁאֵינוֹ רָאוּי לְשׁוּם אָדָם. וְיֵשׁ לְהַחֲמִיר כְּדִבְרֵיהֶם.

אָמַר "כִּכָּר זוֹ הַיּוֹם חֹל, וּלְמָחָר תְּהֵא קֹדֶשׁ" אוֹ "קוֹנָם" — מִשְׁתַּתֵּף בָּהּ, שֶׁבֵּין הַשְּׁמָשׁוֹת עֲדַיִן לֹא נִתְקַדְּשָׁה וַדַּאי, וּמִסָּפֵק לֹא חָל עָלֶיהָ קְדֻשָּׁה לְעִנְיַן עֵרוּב וְשִׁתּוּף. וְאַף עַל פִּי שֶׁאִי אֶפְשָׁר לְאָכְלָהּ בֵּין הַשְּׁמָשׁוֹת, שֶׁהִיא שְׁעַת קְנִיַּת הָעֵרוּב וְהַשִּׁתּוּף, מִכָּל מָקוֹם כֵּיוָן שֶׁרְאוּיָה לַאֲכִילָה מִבְּעוֹד יוֹם, וְגַם בְּבֵין הַשְּׁמָשׁוֹת לֹא חָל עָלֶיהָ קְדֻשָּׁה וַדָּאִית עֲדַיִן, מְעָרְבִין לוֹ בָּהּ. אֲבָל אִם אָמַר "הַיּוֹם קֹדֶשׁ וּלְמָחָר חֹל, שֶׁתְּהֵא מְחֻלֶּלֶת עַל מָעוֹת שֶׁיֵּשׁ לִי" — אֵין מְעָרְבִין וּמִשְׁתַּתְּפִים בָּהּ, שֶׁמִּסָּפֵק לֹא נִפְקְעָה קְדֻשָּׁה, וַהֲרֵי אֵינָהּ רְאוּיָה עַד שֶׁתֶּחְשַׁךְ:

סעיף ט חָצֵר הַפְּתוּחָה לְב' מְבוֹאוֹת, וְשִׁתְּפָה עִם…

חָצֵר הַפְּתוּחָה לְב' מְבוֹאוֹת, וְשִׁתְּפָה עִם שְׁנֵיהֶם — מֻתֶּרֶת בִּשְׁנֵיהֶם, לְטַלְטֵל מִבָּתֶּיהָ לְכָל אַחַת מֵהֶם, וּבִלְבָד שֶׁלֹּא תְטַלְטֵל כֵּלִים שֶׁשָּׁבְתוּ בְּאֶחָד מִבָּתֵּי מָבוֹי זֶה לְמָבוֹי הַב' דֶּרֶךְ חֲצֵרָהּ, שֶׁהֲרֵי הַמְּבוֹאוֹת לֹא שִׁתְּפוּ יַחַד. וְאִם נָתְנוּ ב' הַמְּבוֹאוֹת שִׁתּוּפָם בְּחָצֵר זוֹ בִּכְלִי אֶחָד — הֲרֵי כֻּלָּם מְשֻׁתָּפִים יַחַד, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שע״ח, וּמֻתָּר לְטַלְטֵל מִזֶּה לְזֶה.

וְאִם לֹא שִׁתְּפָה עִם שׁוּם אַחַת מֵהֶם, אִם הִיא רְגִילָה בִּשְׁנֵיהֶם לָצֵאת וְלָבֹא בְּחֹל דֶּרֶךְ עֲלֵיהֶם — אוֹסֶרֶת עַל שְׁנֵיהֶם. וְאִם רְגִילָה בְּאַחַת וּבַשֵּׁנִי אֵינָהּ רְגִילָה, זֶה שֶׁרְגִילָה בּוֹ — אוֹסֶרֶת, וְזֶה שֶׁאֵינָהּ רְגִילָה בּוֹ — אֵינָהּ אוֹסֶרֶת, אַף עַל פִּי שֶׁעוֹבֶרֶת עָלָיו בְּשַׁבָּת.

וְאִם שִׁתְּפָה עִם זֶה שֶׁאֵינָהּ רְגִילָה בּוֹ — הֻתַּר הָרָגִיל לְעַצְמוֹ אִם שִׁתְּפוּ בֵּינֵיהֶם, שֶׁכֵּיוָן שֶׁחָצֵר זוֹ שִׁתְּפָה עִם הַשֵּׁנִי, הֲרֵי סִלְּקָה עַצְמָהּ מִזֶּה שֶׁהָיְתָה רְגִילָה בּוֹ וְאֵינָהּ. וְאֵין צָרִיךְ לוֹמַר אִם רְגִילָה בִּשְׁנֵיהֶם וְשִׁתְּפָה עִם אַחַת מֵהֶם, שֶׁהֻתַּר הַשֵּׁנִי לְעַצְמוֹ אִם שִׁתְּפוּ בֵּינֵיהֶם.

וְאִם הַמָּבוֹי שֶׁרְגִילָה בּוֹ שִׁתְּפוּ בֵּינֵיהֶם, וְאוֹתוֹ שֶׁאֵינָהּ רְגִילָה בּוֹ לֹא שִׁתְּפוּ בֵּינֵיהֶם, וְהִיא לֹא שִׁתְּפָה לֹא עִם זֶה וְלֹא עִם זֶה — דּוֹחִין אוֹתָהּ אֵצֶל זֶה שֶׁאֵינָהּ רְגִילָה בּוֹ שֶׁלֹּא נִשְׁתַּתֵּף, וְתִסְתַּלֵּק מִזֶּה שֶׁרְגִילָה בּוֹ שֶׁנִּשְׁתַּתֵּף, שֶׁלֹּא תַעֲבוֹר עָלָיו בְּשַׁבָּת, כְּדֵי שֶׁיְּהֵא מֻתָּר לְעַצְמוֹ. כֵּיוָן שֶׁהִיא אֵינָהּ מַפְסֶדֶת בַּדָּבָר, שֶׁהֲרֵי לֹא שִׁתְּפָה עִמּוֹ וַאֲסוּרָה לְהִשְׁתַּמֵּשׁ בּוֹ, וְיֵשׁ רֶוַח לַאֲחֵרִים, שֶׁעַל יְדֵי זֶה יִהְיוּ מֻתָּרִים לְהִשְׁתַּמֵּשׁ בַּמָּבוֹי — כּוֹפִין אוֹתָהּ עַל מִדַּת סְדוֹם, שֶׁתַּעֲבוֹר בְּשַׁבָּת דֶּרֶךְ מָבוֹי שֶׁלֹּא נִשְׁתַּתֵּף, אַף עַל פִּי שֶׁאֵינָהּ רְגִילָה בּוֹ. וְאֵין צָרִיךְ לוֹמַר אִם רְגִילָה גַּם בּוֹ, שֶׁאֵין לָהּ לַעֲבוֹר בְּשַׁבָּת עַל הַשֵּׁנִי שֶׁנִּשְׁתַּתֵּף וְלֶאֱסוֹר עָלָיו.

וְלֹא אָמְרוּ שֶׁהַעֲבָרָה בְּרֶגֶל לְבָד בְּלִי תַּשְׁמִישׁ אֵינָהּ אוֹסֶרֶת אֶלָּא בב' חֲצֵרוֹת זוֹ לִפְנִים מִזּוֹ, שֶׁאֵין הַפְּנִימִית אוֹסֶרֶת עַל הַחִיצוֹנָה בְּהַעֲבָרַת רַגְלֶיהָ עָלֶיהָ לְבָד, הוֹאִיל וְעִקַּר חָצֵר חִיצוֹנָה הוּא לְתַשְׁמִישׁ בְּנֵי חִיצוֹנָה. אֲבָל הַמָּבוֹי, הוֹאִיל וְתַשְׁמִישׁוֹ שָׁוֶה לְכָל הַחֲצֵרוֹת הָרְגִילוֹת בּוֹ — הֲרֵי הֵן אוֹסְרוֹת זוֹ עַל זוֹ אֲפִלּוּ בְּהַעֲבָרָה בְּרֶגֶל לְבָד בְּשַׁבָּת, אַף עַל פִּי שֶׁמְּסַלֶּקֶת אֶת עַצְמָהּ מִלְּהִשְׁתַּמֵּשׁ בּוֹ מֵחֲמַת אִסּוּר, מִפְּנֵי שֶׁלֹּא עֵרְבוּ יַחַד.

אֲבָל חָצֵר שֶׁסִּלְּקָה עַצְמָהּ מִלְּהִשְׁתַּמֵּשׁ בְּמָבוֹי, וְעָשְׂתָה מַעֲשֶׂה הַמּוֹכִיחַ עַל סִלּוּקָהּ — שֶׁבָּנְתָה אִצְטְבָא גָּבוֹהַּ ד' טְפָחִים עַל פִּתְחָהּ — שׁוּב אֵינָהּ אוֹסֶרֶת עַל בְּנֵי הַמָּבוֹי אַף עַל פִּי שֶׁעוֹבֶרֶת דֶּרֶךְ עֲלֵיהֶם בְּשַׁבָּת, וַאֲפִלּוּ אֵין לָהּ פֶּתַח לְמָבוֹי אַחֵר לַעֲבוֹר דֶּרֶךְ עָלָיו לִרְשׁוּת הָרַבִּים:

מקור: ספריא — שולחן ערוך הרב, אורח חיים שפ״ו. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה.

2 · לשון הרב ולשון המחבר

לשון הרב מול סעיף המחבר והרמ״א

במה שונה לשון הרב מסעיף המחבר והרמ״א

כל נקודה מעמידה זה לצד זה סעיף של שולחן ערוך המחבר (במספור שלו, תשעה סעיפים, המסומנים באותיותיהם, עם הגהות הרמ״א) ואת הסעיף המקביל בשולחן ערוך אדמו״ר הזקן (במספור שלו, גם הוא תשעה סעיפים), לפי סדר סעיפי הרב, ואינה מציינת אלא מה שעולה מהשוואת שני הלשונות; פרק זה אינו מונה את כל ההבדלים שבין שני הלשונות. נקודה שצוינה אינה מיוחדת לרב בשל כך: פעמים רבות היא כתובה כבר לפני הרב, והדף מציין לפעמים היכן, בלי למנות את כל המקורות האלה (ראה גם רמה 2); הוא מציין לפעמים גם את המשנה ברורה, המאוחר לרב, כשהוא כותב אותו דבר, בלי לציין זאת בכל מקום. במקום שהדף מציין שנחלקו המפרשים, הוא אינו מכריע ומפנה לרמה 2.

נקודה א — טעם האיסור, ולמה השיתוף אינו בפת בלבד (שו״ע הרב שפ״ו:א)

טעם האיסור, והפת

דברי המחבר: « אסרו חכמים לטלטל מן החצירו׳ למבוי והתירוהו ע״י שיתוף שגובין פת או דבר אחר ממיני מאכל מכל חצר וחצר ונותנים אותו בא׳ מן החצירו׳ ואנו רואים כאלו פתוח כל המבוי לאותו חצר » (שו״ע או״ח שפ״ו:א), ואחר כך « ומפני שאינו אלא לשתף החצירו׳ יכולים ליתנו באויר החצר או בבית שאין לו ארבע על ארבע וא״צ ליתנו בבית רק שיהי׳ במקום המשתמר הילכך אין נותנין אותו באויר המבוי » (שו״ע או״ח שפ״ו:א). אינו אומר למה אסרו חכמים, ולא למה אין צריך פת.

דברי הרב: אסרו חכמים « מִפְּנֵי שֶׁהַחֲצֵרוֹת הֵן מְיֻחָדוֹת כָּל אַחַת לִבְעָלֶיהָ וְהַמָּבוֹי מְשֻׁתָּף לְכֻלָּם, וְדוֹמֶה קְצָת לִרְשׁוּת הָרַבִּים » (שו״ע הרב שפ״ו:א); ואחר כך « וְאֵין שִׁתּוּף זֶה דּוֹמֶה לְעֵרוּב שֶׁאֵינוֹ אֶלָּא בְּפַת בִּלְבָד, לְפִי שֶׁעֵרוּבֵי חֲצֵרוֹת הוּא מִשּׁוּם דִּירָה, לְעָרֵב דִּירָתָם לַעֲשׂוֹתָן כְּאַחַת, וְדִירָתוֹ שֶׁל אָדָם אֵינָהּ נִמְשֶׁכֶת אֶלָּא אַחַר מְקוֹם פִּתּוֹ » (שו״ע הרב שפ״ו:א), ו« אֲבָל שִׁתּוּף שֶׁבַּמָּבוֹי אֵינוֹ אֶלָּא לְשַׁתֵּף הַחֲצֵרוֹת וְלֹא הַבָּתִּים, וְחָצֵר לָאו בֵּית דִּירָה הִיא » (שו״ע הרב שפ״ו:א).

ההבדל: סעיף א של המחבר מביא את האיסור ואת היתרו על ידי שיתוף, בלי טעם לאיסור; לשון הרב כותבת תחילה מהו המבוי מן התורה, ואחר כך את טעם איסור חכמים, ואומרת בפירוש למה אין השיתוף בפת בלבד, שלא כעירובי חצרות. במקום הנחתו הוא מזכיר גם בית שער, אכסדרה ומרפסת, ומרחיב את הדין של אויר המבוי לכל מקום שאינו משתמר (שו״ע הרב שפ״ו:א).

אצל מפרשי המחבר: המשנה ברורה, המאוחר לרב, כותב את אותו החילוק: « שאני התם דעירוב משום דירה הוא לערב דירתן לעשותן אחת ודירתו של אדם אינו נמשך אלא אחר פתו משא״כ שיתופי מבואות דאינו אלא לשתף רשות החצרות שבמבוי לא רשות הבתים וחצר לאו בית דירה הוא » (משנה ברורה שפ״ו ס״ק ג).

השלכה מעשית: לפי לשון הרב, משתתפים בפת או בדבר אחר ממיני מאכל או משקה, ומניחים אותו באחת מן החצרות, במקום המשתמר; אין מניחים אותו באויר המבוי, ולא במקום שאינו משתמר (שו״ע הרב שפ״ו:א). בהנחתו בבית נחלקו הראשונים (התוספות בשם ר״י, נגד רש״י והרא״ש), כפי שמביאים הבית יוסף והביאור הלכה: ראה רמה 2, פרק 1. ראה פרק 4, נקודה ①.

נקודה ב — מקום האסור בטלטול שמפסיק בין המקומות (שו״ע הרב שפ״ו:ב)

מקומות חלוקים

דברי המחבר: « ישראלים שדרים בג׳ מקומו׳ חלוקים ודלתות כל א׳ מהמקומו׳ נעולים בלילה והשמש גובה קמח מכולם אינו מועיל להם להשתתף » (שו״ע או״ח שפ״ו:ב). דברי הרמ״א: « ואם עירבו במקום אחד אין האחרים שלא עירבו אוסרים עליהם הואיל ואין להם דריסת הרגל זה על זה » (רמ״א או״ח שפ״ו:ב).

דברי הרב: ישראלים שדרים בב׳ וג׳ מקומות חלוקים, « שֶׁמַּפְסִיק בֵּינֵיהֶם מָקוֹם הָאָסוּר בְּטִלְטוּל מִפְּנֵי שֶׁאֵינוֹ מְתֻקָּן בְּלֶחִי אוֹ קוֹרָה אוֹ צוּרַת פֶּתַח, אוֹ מִפְּנֵי שֶׁדָּרִים שָׁם נָכְרִים וְלֹא שָׂכְרוּ מֵהֶם רְשׁוּתָם » (שו״ע הרב שפ״ו:ב); ובדין שבהגהת הרמ״א, « וְאִם עֵרְבוּ בְּמָקוֹם אֶחָד, וּבְמָקוֹם אֶחָד לֹא עֵרְבוּ — יִתְבָּאֵר בְּסִמָּן שצ״ב » (שו״ע הרב שפ״ו:ב).

ההבדל: המחבר מדבר בשלושה מקומות שדלתותיהם נעולות בלילה; לשון הרב מדברת בשנים ושלושה מקומות, נותנת טעם — מקום האסור בטלטול שמפסיק ביניהם — ואינה מזכירה דלתות נעולות. את הדין שבהגהת הרמ״א — עירבו במקום אחד ובמקום אחד לא עירבו — אין הרב מכריע כאן: הוא מפנה לסימן שצ״ב · 392.

אצל מפרשי המחבר: המגן אברהם, שקדם לרב, כותב על « ואין מבואיהן כא׳ »: « דהא אסורין לטלטל מזה לזה » (מגן אברהם שפ״ו ס״ק א). בגדר הדלתות הנעולות שמזכיר המחבר נחלקו המפרשים; המשנה ברורה, המאוחר לרב, מביא את הדעות (משנה ברורה שפ״ו ס״ק ט); ראה רמה 2.

השלכה מעשית: לפי לשון הרב, מקומות שמפסיק ביניהם מקום האסור בטלטול אינם יכולים להשתתף בעירוב אחד המונח באחד מהם: צריכים לערב בכל מקום ומקום (שו״ע הרב שפ״ו:ב). ראה פרק 4, נקודה ②.

נקודה ג — השותפים בפת, ובסחורה ביין ובשמן (שו״ע הרב שפ״ו:ג)

השותפים

בסוגריים, בנוסח שמגישה ספריא: בסעיף ג כתוב « ושאם היו שותפין א״צ לערב » (שו״ע או״ח שפ״ו:ג). דברי הרמ״א: « ואפי׳ היה משותף עם שכיניו בסחורה לזה ביין ולזה בשמן אין צריך להשתתף והוא שיהיה בכלי אחד » (רמ״א או״ח שפ״ו:ג).

דברי הרב: בגוף הסעיף, « וְשֶׁאִם הָיוּ שֻׁתָּפִים בַּפַּת אֵינָם צְרִיכִים לְעָרֵב » (שו״ע הרב שפ״ו:ג), ואחר כך דין היין והשמן, « וְהוּא שֶׁיִּהְיוּ הַשֶּׁמֶן וְהַיַּיִן בִּכְלִי אֶחָד, כְּמוֹ שֶׁעֵרוּבֵי חֲצֵרוֹת צָרִיךְ לִהְיוֹת בִּכְלִי אֶחָד » (שו״ע הרב שפ״ו:ג).

ההבדל: לשון הרב כותבת « בַּפַּת » (שו״ע הרב שפ״ו:ג), שאינו בסוגריים שבסעיף ג; היא כותבת בגוף הסעיף את הדין שהרמ״א כותב בהגהתו, ונותנת לו טעם: כעירובי חצרות, שצריך להיות בכלי אחד.

אצל מפרשי המחבר: בדין היין והשמן נחלקו המפרשים; המשנה ברורה, המאוחר לרב, מביא את הדעות (משנה ברורה שפ״ו ס״ק ט״ז); ראה רמה 2.

השלכה מעשית: לפי לשון הרב, דיני שיתוף כדיני עירובי חצרות; שותפים בפת אינם צריכים לערב; ביין ובשמן הוא מתנה שיהיו בכלי אחד (שו״ע הרב שפ״ו:ג), ודין זה שנוי במחלוקת (לעיל). ראה פרק 4, נקודה ③.

נקודה ד — מה שאינו ראוי לאכילה, ולפתות שלנו (שו״ע הרב שפ״ו:ד)

מה שאינו ראוי לאכילה

דברי המחבר: « ולא בעדשים ולא בחטים ושעורים ולא בירק שהוא בשיל ולא בשיל » (שו״ע או״ח שפ״ו:ה), בלי טעם.

דברי הרב: « וְלֹא בַּעֲדָשִׁים חַיִּין וְלֹא בְּחִטִּים וּשְׂעוֹרִים חַיִּין וְלֹא בְּיָרָק שֶׁהוּא בָּשִׁיל וְלֹא בָשִׁיל » (שו״ע הרב שפ״ו:ד), « מִפְּנֵי שֶׁכָּל אֵלּוּ אֵינָן רְאוּיִים לַאֲכִילָה » (שו״ע הרב שפ״ו:ד); ואחר כך « אֲבָל בְּיָרָק חַי מִשְׁתַּתְּפִים כְּמוֹ שֶׁיִּתְבָּאֵר, וְהוּא הַדִּין לְפָתוֹת שֶׁלָּנוּ שֶׁנֶּאֱכָלִין גַּם כֵּן חַיִּין, וְאֵינָם גְּרוּעִים מִבָּשָׂר חַי שֶׁמִּשְׁתַּתְּפִים בּוֹ כְּמוֹ שֶׁיִּתְבָּאֵר » (שו״ע הרב שפ״ו:ד).

ההבדל: במקום שהמחבר כותב עדשים, חטים ושעורים סתם, לשון הרב כותבת « חַיִּין » (שו״ע הרב שפ״ו:ד); היא נותנת טעם אחד לכל הפסולים, וכותבת שמשתתפים בירק חי וב« לְפָתוֹת שֶׁלָּנוּ » (שו״ע הרב שפ״ו:ד).

אצל מפרשי המחבר: המגן אברהם, שקדם לרב, כותב על העדשים « דאין נאכלין חיין » (מגן אברהם שפ״ו ס״ק ד), ועל הירק שהוא בשיל ולא בשיל « ונ״ל דלפתות דידן מערבין מהן חיין דלא גרע מבשר חי » (מגן אברהם שפ״ו ס״ק ה).

השלכה מעשית: לפי לשון הרב, אין משתתפים במה שהוא מונה כאינו ראוי לאכילה; משתתפים בירק חי, ובלפתות שלנו, הנאכלים חיים (שו״ע הרב שפ״ו:ד). ראה פרק 4, נקודה ④.

נקודה ה — מים ומלח (שו״ע הרב שפ״ו:ה)

מים ומלח

דברי המחבר: « ולא במים לבדם ולא במלח לבדו אבל אם עירבם יחד משתתפין בהם וי״א דדוקא כשנתן לתוכן שמן וי״א שמערבין בתבלין » (שו״ע או״ח שפ״ו:ה).

דברי הרב: « חוּץ מִן הַמַּיִם לְבַדָּם וְכֵן מֶלַח לְבַדּוֹ, לְפִי שֶׁאֵינָן זָנִין הַגּוּף כְּלָל, אֲבָל אִם עֵרֵב מַיִם וּמֶלַח בְּיַחַד — מָזוֹן הוּא לְטַבֵּ(ו)ל בּוֹ פִּתּוֹ, וּמִשְׁתַּתְּפִים בָּהֶם » (שו״ע הרב שפ״ו:ה).

ההבדל: לשון הרב נותנת טעם לפסול (מים ומלח אינם זנין את הגוף) ולהיתר התערובת (לטבל בו פתו); ואינה מביאה לא את הדעה שאין התערובת מועילה אלא כשנתן לתוכה שמן, ולא את הדעה שמערבין בתבלין, שהמחבר מביא ב« יש אומרים ». בסעיף ד מונה הרב את התבלין בין מה שאין מערבין בו (שו״ע הרב שפ״ו:ד), כמו המחבר לפני שתי הדעות.

אצל מפרשי המחבר: המשנה ברורה, המאוחר לרב, דן בשתי הדעות (משנה ברורה שפ״ו ס״ק כ״ט, ל); ראה רמה 2.

השלכה מעשית: לפי לשון הרב, אין משתתפים במים לבדם ולא במלח לבדו, ומשתתפים במים ומלח מעורבים (שו״ע הרב שפ״ו:ה). ראה פרק 4, נקודה ④.

נקודה ו — בשר חי, חומץ ויין מבושל (שו״ע הרב שפ״ו:ו)

בשר חי, חומץ ויין מבושל

דברי המחבר: « ובשר חי לא הוי ליפתן וצריך כדי שיאכל ממנו ב׳ סעודו׳ אבל צלי הוי ליפתן ושיעורו ללפת בו ב׳ סעודו׳ חומץ הוי ליפתן וכן יין מבושל אבל יין חי לא הוי ליפתן ושיעורו שני רביעיו׳ וכן שיעור שאר משקים » (שו״ע או״ח שפ״ו:ו).

דברי הרב: בבשר חי, « וְהוּא שֶׁיְּהֵא מָלִיחַ, וְהוּא הַדִּין לְדָג מָלִיחַ » (שו״ע הרב שפ״ו:ו); בחומץ, « וְחֹמֶץ — שִׁעוּרוֹ כְּדֵי לְטַבֵּל בּוֹ יָרָק מְזוֹן ב׳ סְעֻדּוֹת, אִם הָיָה אוֹכֵל הַיָּרָק לְבַדּוֹ בְּלֹא פַת » (שו״ע הרב שפ״ו:ו); ביין מבושל, « וְיַיִן מְבֻשָּׁל — לְקִנּוּחַ סְעֻדָּה הוּא בָא (וְלֹא לְטַבֵּל), לְפִיכָךְ שִׁעוּרוֹ כְּדֵי שֶׁמְּבִיאִים מִמֶּנּוּ לְקִנּוּחַ ב׳ סְעֻדּוֹת » (שו״ע הרב שפ״ו:ו).

ההבדל: לשון הרב מתנה בבשר חי שיהא מליח, ומרחיבה לדג מליח; במקום שהמחבר כותב בחומץ שהוא ליפתן, הרב נותן את שיעורו: כדי לטבל בו ירק מזון שתי סעודות, הנאכל בלא פת; ובמקום שהמחבר כותב « וכן יין מבושל » (שו״ע או״ח שפ״ו:ו) אחר החומץ, הרב כותב שהיין המבושל לקינוח סעודה הוא בא, ונותן לו שיעור לפי זה. הרב כותב גם ששתי הסעודות « שֶׁהֵן כְּשִׁשָּׁה בֵיצִים » (שו״ע הרב שפ״ו:ו), ומפנה לסימן שס״ח · 368.

אצל מפרשי המחבר: המגן אברהם, שקדם לרב, כותב על החומץ « שיעורו כדי לטבל ירק מזון ב׳ סעודות אם היה אוכלו בלא פת » (מגן אברהם שפ״ו ס״ק ז). בשיעור היין המבושל נחלקו המפרשים; המשנה ברורה, המאוחר לרב, מביא את הדעות (משנה ברורה שפ״ו ס״ק ל״ו); ראה רמה 2.

השלכה מעשית: לפי לשון הרב, משתתפים בבשר חי מליח או בדג מליח כשיעור שתי סעודות; בחומץ כשיעור שהוא נותן; וביין חי כשיעור שתי רביעיות (שו״ע הרב שפ״ו:ו). בשיעור היין המבושל נחלקו (לעיל). ראה פרק 4, נקודה ⑤.

נקודה ז — תרומה בזמן הזה, והנודר מאוכל (שו״ע הרב שפ״ו:ח)

התרומה והנדר

דברי המחבר: « משתתפין אפי׳ באוכל שאינו ראוי לו אם ראוי לשום אדם כגון לנזיר ביין ולישראל בתרומה » (שו״ע או״ח שפ״ו:ח); ואחר דין הנודר או הנשבע שלא יאכל מאכל, « ויש אומרים דהיינו דוקא כשנדר או נשבע שלא יאכלנו אבל אם נדר או נשבע שלא יהנה ממנו אין משתתפין לו בה » (שו״ע או״ח שפ״ו:ח). ואחר כך, בסוגריים, בנוסח שמגישה ספריא: « ואם אמר קונם הנאתו או אכילתו עלי לכ״ע אין משתתף בה » (שו״ע או״ח שפ״ו:ח).

דברי הרב: בתרומה, « בִּזְמַן שֶׁהָיְתָה נֶאֱכֶלֶת לַכֹּהֲנִים. אֲבָל עַכְשָׁו אֵין תְּרוּמָה טְהוֹרָה רְאוּיָה לְשׁוּם אָדָם, שֶׁכֻּלָּנוּ טְמֵאֵי מֵתִים, שֶׁאֵין לָנוּ אֵפֶר פָּרָה » (שו״ע הרב שפ״ו:ח); בנדר, « וַאֲפִלּוּ נָדַר וְנִשְׁבַּע שֶׁלֹּא יֵהָנֶה מִמֶּנּוּ לֹא נֶאֱסַר בּוֹ בַּהֲנָאָה זוֹ מַה שֶּׁמִּשְׁתַּתֵּף בּוֹ, לְפִי שֶׁלֹּא נִתְכַּוֵּן אֶלָּא שֶׁלֹּא לֵהָנוֹת מִמֶּנּוּ בְּדֶרֶךְ הֲנָאָתוֹ » (שו״ע הרב שפ״ו:ח); ואחר כך « וְיֵשׁ אוֹמְרִים שֶׁהַנּוֹדֵר מֵאֹכֶל, אֲפִלּוּ לֹא נָדַר אֶלָּא מֵאֲכִילָתוֹ וְלֹא מֵהֲנָאָתוֹ — אֵינוֹ מִשְׁתַּתֵּף בּוֹ » (שו״ע הרב שפ״ו:ח), עם טעמם — הנודר אוסר את החפץ, ומתחלף הוא בהקדש —, ואחר כך כותב « וְיֵשׁ לְהַחֲמִיר כְּדִבְרֵיהֶם » (שו״ע הרב שפ״ו:ח).

ההבדל: לשון הרב כותבת שדין התרומה אינו נוהג עכשיו, שאין תרומה טהורה ראויה לשום אדם. בנדר, שני הלשונות אינם מעמידים את המחלוקת במקום אחד: בסעיף ח, ה« יש אומרים » שבדברי המחבר פוסל את הנודר או הנשבע שלא יהנה ממנו, והסוגריים שאחריו בנוסח שמגישה ספריא כותבים שב« קונם » לדברי הכל אין משתתף בה; אצל הרב, הדעה הראשונה מתירה אפילו נדר ונשבע שלא יהנה ממנו, וה« יש אומרים » פוסל את הנודר מאוכל אפילו לא נדר אלא מאכילתו; ואחר כך כותב הרב שיש להחמיר כדעה השנייה.

אצל מפרשי המחבר: המגן אברהם, שקדם לרב, כותב על התרומה « ועכשיו אין תרומה טהורה הראויה לשום אדם » (מגן אברהם שפ״ו ס״ק ט); ובנדר הוא מקשה על ה« לכ״ע » שבסוגריים אלו, שהוא מייחס לרמ״א (מגן אברהם שפ״ו ס״ק י). בנדרים אלו נחלקו המפרשים; המשנה ברורה, המאוחר לרב, מביא את הדעות (משנה ברורה שפ״ו ס״ק מ״ט, נ); ראה רמה 2.

השלכה מעשית: לפי לשון הרב, דין התרומה אינו נוהג עכשיו, שאין תרומה טהורה ראויה לשום אדם; ובמאכל שנדר ממנו, אפילו לא נדר אלא מאכילתו, יש להחמיר (שו״ע הרב שפ״ו:ח). ראה פרק 4, נקודה ⑥.

נקודה ח — חצר הפתוחה לשני מבואות (שו״ע הרב שפ״ו:ט)

חצר הפתוחה לשני מבואות

דברי המחבר: « ואסור לטלטל כלים ששבתו במבוי זה למבוי האחר דרך החצר » (שו״ע או״ח שפ״ו:ט), ואחר כך דיני החצר שלא שיתפה עם שום אחד מהם, או עם אחד, עד « כופין אותו על מדת סדום » (שו״ע או״ח שפ״ו:ט).

דברי הרב: « וְאִם נָתְנוּ ב׳ הַמְּבוֹאוֹת שִׁתּוּפָם בְּחָצֵר זוֹ בִּכְלִי אֶחָד — הֲרֵי כֻּלָּם מְשֻׁתָּפִים יַחַד, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִמָּן שע״ח, וּמֻתָּר לְטַלְטֵל מִזֶּה לְזֶה » (שו״ע הרב שפ״ו:ט); ואחר אותם הדינים, שתי פסקאות: « וְלֹא אָמְרוּ שֶׁהַעֲבָרָה בְּרֶגֶל לְבָד בְּלִי תַּשְׁמִישׁ אֵינָהּ אוֹסֶרֶת אֶלָּא בב׳ חֲצֵרוֹת זוֹ לִפְנִים מִזּוֹ, שֶׁאֵין הַפְּנִימִית אוֹסֶרֶת עַל הַחִיצוֹנָה בְּהַעֲבָרַת רַגְלֶיהָ עָלֶיהָ לְבָד, הוֹאִיל וְעִקַּר חָצֵר חִיצוֹנָה הוּא לְתַשְׁמִישׁ בְּנֵי חִיצוֹנָה » (שו״ע הרב שפ״ו:ט), עם הטעם שבמבוי אין הדין כן; ו« אֲבָל חָצֵר שֶׁסִּלְּקָה עַצְמָהּ מִלְּהִשְׁתַּמֵּשׁ בְּמָבוֹי, וְעָשְׂתָה מַעֲשֶׂה הַמּוֹכִיחַ עַל סִלּוּקָהּ — שֶׁבָּנְתָה אִצְטְבָא גָּבוֹהַּ ד׳ טְפָחִים עַל פִּתְחָהּ — שׁוּב אֵינָהּ אוֹסֶרֶת עַל בְּנֵי הַמָּבוֹי אַף עַל פִּי שֶׁעוֹבֶרֶת דֶּרֶךְ עֲלֵיהֶם בְּשַׁבָּת, וַאֲפִלּוּ אֵין לָהּ פֶּתַח לְמָבוֹי אַחֵר לַעֲבוֹר דֶּרֶךְ עָלָיו לִרְשׁוּת הָרַבִּים » (שו״ע הרב שפ״ו:ט).

ההבדל: סעיף ט של המחבר אינו כותב לא את דין שני המבואות שנתנו שיתופם בחצר זו בכלי אחד, לא למה במבוי ההעברה ברגל לבד אוסרת, ולא את דין החצר שבנתה איצטבא גבוה ד׳ טפחים על פתחה; לשון הרב כותבת אותם (שו״ע הרב שפ״ו:ט).

אצל מפרשי המחבר: המגן אברהם, שקדם לרב, כותב על « ואסור לטלטל »: « ואם נתנו שני המבואות שתופן בחצר זו בכלי א׳ ה״ל כולן כא׳ ומותר לטלטל ממבוי זה למבוי זה » (מגן אברהם שפ״ו ס״ק י״ד).

השלכה מעשית: לפי לשון הרב, חצר הפתוחה לשני מבואות ושיתפה עם שניהם מטלטלת מבתיה לכל אחד מהם, אבל אינה מטלטלת דרך חצרה ממבוי למבוי כלים ששבתו בבתי אחד מהם, אלא אם כן נתנו שני המבואות שיתופם בחצר זו בכלי אחד (שו״ע הרב שפ״ו:ט). ראה פרק 4, נקודה ⑦.

3 · דברי הרבי

שיחות, מאמרים ואגרות הרבי

דברי הרבי על נושאי סימן שפ״ו

אין הפניה מאומתת נכון לעכשיו.
אנו מייחסים לרבי אך ורק מקורות ראשוניים הניתנים לאימות (מכתב, שיחה או פסק מזוהה, עם קישור). בשלב זה לא ניתן היה לאמת מול Sefaria או Chabad.org דבר מקור של הרבי המתייחס ישירות לסימן זה — מוטב שלא להציג דבר מאשר הפניה לא מאומתת.
כיוון חיפוש מועדף לפסקי הרבי לפי סימן: שערי הלכה ומנהג (מהדורת קה״ת). כל הפניה שתאומת תתווסף עם קישורה.
4 · למעשה

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן שפ״ו · 386, ברגישות החב״דית. כל נקודה מפנה לסעיף שלה בשולחן ערוך הרב, לפי סדר הלשון.

לחסיד חב״ד — סימן שפ״ו

  • ① השיתוף ומקומו: גובין מכל חצר וחצר פת או דבר אחר ממיני מאכל או משקה, ומניחים אותו באחת מן החצרות, במקום המשתמר — אפילו בבית שאין בו ד׳ אמות על ד׳ אמות, או באויר החצר; אין מניחים אותו באויר המבוי, ולא במקום שאינו משתמר (שו״ע הרב שפ״ו:א). בהנחתו בבית נחלקו הראשונים (פרק 2, נקודה א). הפניה: שולחן ערוך אדמו״ר הזקן שפ״ו:א.
  • ② מקומות חלוקים: ישראלים הדרים במקומות שמפסיק ביניהם מקום האסור בטלטול צריכים לערב בכל מקום ומקום בפני עצמו; עירוב אחד המונח באחד מהם אינו מועיל להם (שו״ע הרב שפ״ו:ב; דין עירבו במקום אחד ובמקום אחד לא עירבו — הרב מפנה לסימן שצ״ב · 392). הפניה: שולחן ערוך אדמו״ר הזקן שפ״ו:ב.
  • ③ דיני עירובי חצרות: דיני שיתוף כדיני עירובי חצרות; שותפים בפת אינם צריכים לערב (שו״ע הרב שפ״ו:ג). בשותפים בסחורה ביין ובשמן הרב מתנה שיהיו בכלי אחד, ובדין זה נחלקו המפרשים (פרק 2, נקודה ג). הפניה: שולחן ערוך אדמו״ר הזקן שפ״ו:ג.
  • ④ במה משתתפים: משתתפים בכל מיני מאכל ומשקה, וכמה מיני מאכל מצטרפים למזון שתי סעודות; לא במה שהרב מונה כאינו ראוי לאכילה; בירק חי ובלפתות שלנו; לא במים לבדם ולא במלח לבדו, אבל במים ומלח מעורבים (שו״ע הרב שפ״ו:ד–ה). הפניה: שולחן ערוך אדמו״ר הזקן שפ״ו:ד, ה.
  • ⑤ השיעורים: הרב נותן את שיעור הדבר שמלפתים בו את הפת, ושל מה שאין מלפתים בו, של בשר חי מליח, של בשר צלי, של חומץ, של יין חי ושל שאר משקים (שו״ע הרב שפ״ו:ו), ושל מה שהוא מונה בסעיף ז (שו״ע הרב שפ״ו:ז). בשיעור היין המבושל נחלקו (פרק 2, נקודה ו). הפניה: שולחן ערוך אדמו״ר הזקן שפ״ו:ו, ז.
  • ⑥ אוכל שאינו ראוי לו: משתתפים באוכל שאינו ראוי לו אם ראוי לישראל אחר; עכשיו, כותב הרב, אין תרומה טהורה ראויה לשום אדם; במאכל שנדר ממנו, אפילו לא נדר אלא מאכילתו, הרב כותב שיש להחמיר, ובנדרים אלו נחלקו המפרשים (פרק 2, נקודה ז) (שו״ע הרב שפ״ו:ח). הפניה: שולחן ערוך אדמו״ר הזקן שפ״ו:ח.
  • ⑦ חצר הפתוחה לשני מבואות: שיתפה עם שניהם — מטלטלת מבתיה לכל אחד מהם, אבל אינה מטלטלת דרך חצרה ממבוי למבוי כלים ששבתו בבתי אחד מהם, אלא אם כן נתנו שניהם שיתופם בחצר זו בכלי אחד; לא שיתפה עם שום אחד מהם — אוסרת על מה שהיא רגילה בו לצאת ולבוא בחול; הרב כותב גם מתי היא מסתלקת ממנו, ומתי כופין אותה לעבור דרך השני (שו״ע הרב שפ״ו:ט). הפניה: שולחן ערוך אדמו״ר הזקן שפ״ו:ט.
