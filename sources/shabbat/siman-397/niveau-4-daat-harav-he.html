דעת הרב

דעת הרב — שולחן ערוך אדמו״ר הזקן על סימן שצ״ז

שביתת היחיד וכליו ומהלך אלפים אמה: מידת הפסיעות למי שקדש עליו היום בבקעה; בהמה שנמסרה לבן או לרועה; שור של רועה ושור של פטם; כלים של האחים, חלוק ששאלו שנים, בהמה וחבית שלקחו בשותפות; השואל כלי, מים ומלח לעיסה, גחלת ושלהבת; בור, נהרות, מעיינות וגשמים; פירות מופקדים ומנות האורחים. עשרים הסעיפים של שולחן ערוך אדמו״ר הזקן; בשולחן ערוך של המחבר שמונה עשר.

מדוע רמה 4 מוקדשת לאדמו״ר הזקן? שולחן ערוך אדמו״ר הזקן אינו פירוש על המחבר: זהו שולחן ערוך בפני עצמו, שנכתב על ידי אדמו״ר הזקן, בחלוקה ובמספור סעיפים משלו. כאן מונה שולחן ערוך הרב עשרים סעיפים בראש הסימן, ובשולחן ערוך של המחבר שמונה עשר. ספריא מגישה את לשון הרב בעשרים קטעים, והדף מביא וממספר אותם כמות שהם (פרק 1). סעיפים א עד ה של הרב עומדים מול סעיפים א עד ה של המחבר; הרב כותב שור של רועה בסעיף ו ושור של פטם בסעיף ז, והמחבר כותב שור של פטם בסעיף ו ושור של רועה בסעיף ז; סעיפים ח עד יד של הרב עומדים מול סעיפים ח עד יד של המחבר; סעיף טו של המחבר (נהרות ומעיינות, ואחר כך מים הבאים מחוץ לתחום) מתחלק אצל הרב בין סעיף טו לסעיף טז; לסעיף יז של הרב (גשמים היורדים ביום טוב) אין סעיף מקביל בסימן שצ״ז של המחבר; סעיפים יח, יט ו־כ של הרב עומדים מול סעיפים טז, יז ו־יח של המחבר. בהפניות שבסוגריים שני המספורים באותיות, וההפניה מציינת תמיד את הספר (שו״ע או״ח למחבר, שו״ע הרב לרב).

עמוד זה מקבץ, לסימן שצ״ז, את הטקסט המלא של הרב, את מה ששונה בלשונו מדברי המחבר, ואת ההנהגה המעשית העולה מדבריו.

1 · טקסט אדמו״ר הזקן

שלחן ערוך הרב — סימן שצ״ז

הטקסט המלא של שולחן ערוך אדמו״ר הזקן (שולחן ערוך הרב)

סימן שצ״ז — דִּין שְׁבִיתַת הַיָּחִיד וְכֵלָיו וּמַהֲלַךְ אַלְפַּיִם אַמָּה — וּבוֹ כ׳ סְעִיפִים
מקור: מהדורת קה״ת, כפי שמופיע בספריא. בכותרת: כ׳ סעיפים; ספריא מגישה את הלשון בעשרים קטעים (בשולחן ערוך של המחבר שמונה עשר סעיפים, במספור משלו).

סעיף א דִּין שְׁבִיתַת הַיָּחִיד וְכֵלָיו וּמַהֲלַךְ אַלְפַּיִם אַמָּה וּבוֹ כ' סְעִיפִים: כָּל אָדָם…

דִּין שְׁבִיתַת הַיָּחִיד וְכֵלָיו וּמַהֲלַךְ אַלְפַּיִם אַמָּה וּבוֹ כ' סְעִיפִים:



כָּל אָדָם יֵשׁ לוֹ חוּץ מִד' אַמּוֹת אוֹ מֵהַמָּקוֹם שֶׁשָּׁבַת בּוֹ אַלְפַּיִם אַמָּה לְכָל רוּחַ, מְרֻבָּעוֹת כְּטַבְלָא מְרֻבַּעַת, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שצ״ט:

סעיף ב קִדֵּשׁ עָלָיו הַיּוֹם בַּבִּקְעָה הַרְחֵק מִן…

קִדֵּשׁ עָלָיו הַיּוֹם בַּבִּקְעָה הַרְחֵק מִן הָעִיר, שֶׁאֵין שָׁם תְּחוּמֵי שַׁבָּת מְסֻמָּנִים וְאֵינוֹ יוֹדֵעַ לְשַׁעֵר תְּחוּם שַׁבָּת — מְהַלֵּךְ אַלְפַּיִם פְּסִיעוֹת בֵּינוֹנִיּוֹת כְּדֶרֶךְ הֲלִיכָתוֹ שֶׁל אָדָם, וְזֶהוּ תְּחוּם שַׁבָּת, כִּי דֶרֶךְ הֲלִיכָתוֹ שֶׁל אָדָם אַמָּה מֵרֹאשׁ גּוּדָל רֶגֶל זוֹ עַד רֹאשׁ גּוּדָל רֶגֶל זוֹ, שֶׁמַּרְחִיק כַּחֲצִי אַמָּה בֵּין רֶגֶל לְרֶגֶל. וּמְקוֹם מַצַּב הָרֶגֶל עִם הַמִּנְעָל הוּא גַם כֵּן כַּחֲצִי אַמָּה:

סעיף ג כְּשֵׁם שֶׁאֵין אָדָם רַשַּׁאי לְהַלֵּךְ בְּשַׁבָּת…

כְּשֵׁם שֶׁאֵין אָדָם רַשַּׁאי לְהַלֵּךְ בְּשַׁבָּת וְיוֹם טוֹב אֶלָּא אַלְפַּיִם אַמָּה לְכָל רוּחַ חוּץ לִמְקוֹם שְׁבִיתָתוֹ, כָּךְ כֵּלָיו וּבְהֶמְתּוֹ אֵין יְכוֹלִים שׁוּם אָדָם לְהוֹלִיכָם חוּץ לְאַלְפַּיִם שֶׁל בַּעֲלֵיהֶם, מִפְּנֵי שֶׁקָּנוּ שְׁבִיתָה בִּמְקוֹם שְׁבִיתַת בַּעֲלֵיהֶם, אַף עַל פִּי שֶׁלֹּא הָיוּ שָׁם כְּשֶׁקִּדֵּשׁ הַיּוֹם. וְאִם עֵרְבוּ בַּעֲלֵיהֶם לְרוּחַ אַחַת — אֵין שׁוּם אָדָם יָכוֹל לְהוֹלִיכָם לְרוּחַ הָאַחֶרֶת אֲפִלּוּ פְּסִיעָה אַחַת בְּמָקוֹם שֶׁאֵין הַבְּעָלִים יְכוֹלִים לְהַלֵּךְ:

סעיף ד הַמּוֹסֵר בְּהֶמְתּוֹ לִבְנוֹ, הֲרֵי הִיא כְּרַגְלֵי…

הַמּוֹסֵר בְּהֶמְתּוֹ לִבְנוֹ, הֲרֵי הִיא כְּרַגְלֵי הָאָב, שֶׁאֵין שׁוּם אָדָם יָכוֹל לְהוֹלִיכָהּ אֶלָּא בְּמָקוֹם שֶׁהָאָב יָכוֹל לְהַלֵּךְ וְלֹא בְּמָקוֹם שֶׁהַבֵּן יָכוֹל לְהַלֵּךְ, אַף עַל פִּי שֶׁמְּסָרָהּ לוֹ מִבְּעוֹד יוֹם, לְפִי שֶׁדֶּרֶךְ בְּנֵי אָדָם לְהַפְקִיד חֶפְצֵיהֶם בְּיַד בְּנֵיהֶם וְאֵין כַּוָּנָתָם לִמְסוֹר בִּרְשׁוּתָם:

סעיף ה אֲבָל הַמּוֹסֵר בְּהֶמְתּוֹ לְרוֹעֶה, אֲפִלּוּ מְסָרָהּ…

אֲבָל הַמּוֹסֵר בְּהֶמְתּוֹ לְרוֹעֶה, אֲפִלּוּ מְסָרָהּ לוֹ בְּיוֹם טוֹב וְלֹא מִבְּעוֹד יוֹם — הֲרֵי הִיא כְּרַגְלֵי הָרוֹעֶה, בֵּין שֶׁהוּא יִשְׂרָאֵל בֵּין שֶׁהוּא נָכְרִי, לְפִי שֶׁמִּן הַסְּתָם גַּם מֵאֶתְמוֹל הָיָה בְּדַעְתּוֹ לְמָסְרָהּ לוֹ הַיּוֹם, וְקָנְתָה שְׁבִיתָה בִּמְקוֹם שְׁבִיתַת הָרוֹעֶה וַאֲפִלּוּ הוּא נָכְרִי, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן ת״א. וְאִם מְסָרָהּ לְב' רוֹעִים — הֲרֵי הִיא כְּרַגְלֵי בְעָלֶיהָ, מִפְּנֵי שֶׁלֹּא קָנָה אֶחָד מֵהֶם. וַאֲפִלּוּ אִם מְסָרָהּ לִשְׁנֵיהֶם מֵעֶרֶב יוֹם טוֹב וְלֹא בֵרֵר דְּבָרָיו, וּבְיוֹם טוֹב בֵּרֵר דְּבָרָיו שֶׁמְּסָרָהּ לְאֶחָד מֵהֶם, וְאֵלָיו נִתְכַּוֵּן מֵעֶרֶב יוֹם טוֹב — אַף עַל פִּי כֵן כֵּיוָן שֶׁבְּשָׁעָה שֶׁקִּדֵּשׁ הַיּוֹם לֹא הָיָה מְבֹרָר עֲדַיִן לְאֵיזֶה מֵהֶם מְסָרָהּ, לֹא קָנוּ שְׁנֵיהֶם, וְנִשְׁאֲרָה בִּרְשׁוּת הַבְּעָלִים וְלָכֵן הִיא כְּרַגְלָיו:

סעיף ו שׁוֹר שֶׁל רוֹעֶה — כְּרַגְלֵי אַנְשֵׁי…

שׁוֹר שֶׁל רוֹעֶה — כְּרַגְלֵי אַנְשֵׁי אוֹתָהּ הָעִיר, שֶׁאֲפִלּוּ עֵרֵב הָרוֹעֶה לְרוּחַ אַחַת אֵינוֹ מְעַכֵּב עַל אֶחָד מִבְּנֵי אוֹתָהּ הָעִיר שֶׁלָּקְחוּ מִמֶּנּוּ בְּיוֹם טוֹב מִלְּהוֹלִיכוֹ אַלְפַּיִם לְרוּחַ הָאַחֶרֶת, מִפְּנֵי שֶׁמֵּאֶתְמוֹל מַעֲמִידוֹ בִּרְשׁוּת בְּנֵי הָעִיר, לְפִי שֶׁרְגִילִים לִקַּח מִמֶּנּוּ. אֲבָל אִם אֶחָד מֵעִיר אַחֶרֶת שֶׁעֵרֵב לְעִיר זוֹ לָקְחוּ מִמֶּנּוּ בְּיוֹם טוֹב — אֵינוֹ יָכוֹל לְהוֹלִיכוֹ לְעִירוֹ:

סעיף ז אֲבָל שׁוֹר שֶׁל פַּטָּם הוּא כְּרַגְלֵי…

אֲבָל שׁוֹר שֶׁל פַּטָּם הוּא כְּרַגְלֵי מִי שֶׁלְּקָחוֹ לְשָׁחֳטוֹ בְּיוֹם טוֹב, אֲפִלּוּ הוּא מֵעִיר אַחֶרֶת. וְכֵן אִם שְׁחָטוֹ בְּעָלָיו בְּיוֹם טוֹב וּמָכַר בְּשָׂרוֹ — הֲרֵי כָּל אֶחָד מֵהַלּוֹקְחִים מוֹלִיךְ מָנָתוֹ לְמָקוֹם שֶׁהוּא הוֹלֵךְ, אֲפִלּוּ הוּא מֵעִיר אַחֶרֶת, מִפְּנֵי שֶׁדַּעַת בְּעָלָיו לְמָכְרוֹ גַּם לַאֲנָשִׁים אֲחֵרִים חוּץ מֵאַנְשֵׁי אוֹתָהּ הָעִיר, מִפְּנֵי שֶׁהוּא מְפֻטָּם וְהַכֹּל שׁוֹמְעִים שִׁמְעוֹ וּבָאִים לִקְנוֹתוֹ:

סעיף ח כֵּלִים הַמְיֻחָדִים לְאֶחָד מֵהָאַחִים שֶׁבַּבַּיִת …

כֵּלִים הַמְיֻחָדִים לְאֶחָד מֵהָאַחִים שֶׁבַּבַּיִת — הֲרֵי הֵם כְּרַגְלָיו, וְשֶׁאֵינָם מְיֻחָדִים — אֵינָם יְכוֹלִים לְהוֹלִיכָם אֶלָּא לְמָקוֹם שֶׁכֻּלָּם יְכוֹלִים לְהַלֵּךְ:

סעיף ט שְׁנַיִם שֶׁשָּׁאֲלוּ מֵעֶרֶב יוֹם טוֹב חָלוּק…

שְׁנַיִם שֶׁשָּׁאֲלוּ מֵעֶרֶב יוֹם טוֹב חָלוּק אֶחָד, זֶה לֵילֵךְ בּוֹ יוֹם טוֹב שַׁחֲרִית וְזֶה לֵילֵךְ בּוֹ עַרְבִית — הֲרֵי הוּא כְּרַגְלֵי שְׁנֵיהֶם, דְּהַיְנוּ שֶׁאֵין יְכוֹלִים לְהוֹלִיכוֹ אֶלָּא לְמָקוֹם שֶׁשְּׁנֵיהֶם יְכוֹלִים לְהַלֵּךְ. וְאִם עֵרֵב זֶה לְסוֹף אַלְפַּיִם לְמִזְרָח, וְזֶה לְסוֹף אַלְפַּיִם לְמַעֲרָב — לֹא יְזִיזוּהוּ מִמְּקוֹם שְׁבִיתָתָם, שֶׁהֲרֵי זֶה שֶׁעֵרֵב לְמִזְרָח אֵינוֹ יָכוֹל לְהַלֵּךְ לְמַעֲרָב מִמְּקוֹם שְׁבִיתָתוֹ אֲפִלּוּ פְּסִיעָה אַחַת, וְזֶה שֶׁעֵרֵב לְמַעֲרָב אֵינוֹ יָכוֹל לֵילֵךְ לְמִזְרָח אֲפִלּוּ פְּסִיעָה אַחַת, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן ת״ח:

סעיף י ב' שֶׁלָּקְחוּ בְּהֵמָה בְּשֻׁתָּפוּת וּשְׁחָטוּהָ בְּיוֹם…

ב' שֶׁלָּקְחוּ בְּהֵמָה בְּשֻׁתָּפוּת וּשְׁחָטוּהָ בְּיוֹם טוֹב, אַף עַל פִּי שֶׁחֲלָקוּהָ וְלָקְחוּ כָּל אֶחָד מָנָתוֹ — הֲרֵי כָּל הַבָּשָׂר שֶׁל זֶה וְזֶה כְּרַגְלֵי שְׁנֵיהֶם, לְפִי שֶׁבֵּין הַשְּׁמָשׁוֹת יָנַק כָּל אֶחָד מֵחֲבֵרוֹ כְּשֶׁהָיְתָה הַבְּהֵמָה בְּחַיֶּיהָ. אֲבָל אִם לָקְחוּ חָבִית שֶׁל יַיִן בְּשֻׁתָּפוּת מֵעֶרֶב יוֹם טוֹב, וַחֲלָקוּהָ בְּיוֹם טוֹב — הֲרֵי חֶלְקוֹ שֶׁל כָּל אֶחָד מֵהֶם כְּרַגְלָיו, לְפִי שֶׁבְּשֶׁל דִּבְרֵיהֶם יֵשׁ בְּרֵרָה, דְּהַיְנוּ שֶׁאָנוּ אוֹמְרִים שֶׁהֻבְרַר הַדָּבָר לְמַפְרֵעַ שֶׁזֶּה הָיָה חֶלְקוֹ הַמַּגִּיעַ לוֹ מֵעֶרֶב יוֹם טוֹב, וְקָנָה שְׁבִיתָה בִּמְקוֹם שְׁבִיתָתוֹ:

סעיף יא הַשּׁוֹאֵל כְּלִי מֵחֲבֵרוֹ מֵעֶרֶב יוֹם טוֹב…

הַשּׁוֹאֵל כְּלִי מֵחֲבֵרוֹ מֵעֶרֶב יוֹם טוֹב, אֲפִלּוּ לֹא לָקְחוּ עַד הַלַּיְלָה, הֲרֵי הוּא כְּרַגְלֵי הַשּׁוֹאֵל, שֶׁכֵּיוָן שֶׁאָמַר לוֹ מִבְּעוֹד יוֹם לְהַשְׁאִילוֹ — הֶעֱמִידוֹ בִּרְשׁוּתוֹ שֶׁיּוֹלִיכֶנּוּ לְמָקוֹם שֶׁיִּרְצֶה. אֲבָל אִם שְׁאָלוֹ מִמֶּנּוּ בְּיוֹם טוֹב — הֲרֵי הוּא כְּרַגְלֵי הַמַּשְׁאִיל. וַאֲפִלּוּ דַּרְכּוֹ לְשָׁאֳלוֹ מִמֶּנּוּ בְּכָל יוֹם טוֹב, אֵין אוֹמְרִים שֶׁמֵּאֶתְמוֹל הֶעֱמִידוֹ בִּרְשׁוּתוֹ, לְפִי שֶׁכֵּיוָן שֶׁלֹּא בָּא לְשָׁאֳלוֹ מִמֶּנּוּ מֵעֶרֶב יוֹם טוֹב, סָבוּר הַמַּשְׁאִיל שֶׁמָּא מְצָאוֹ אָדָם אַחֵר שֶׁהִשְׁאִיל לוֹ וְלֹא יִשְׁאַל מִמֶּנּוּ:

סעיף יב הָאִשָּׁה שֶׁשָּׁאֲלָה מֵחֲבֶרְתָּהּ בְּיוֹם טוֹב מַיִם…

הָאִשָּׁה שֶׁשָּׁאֲלָה מֵחֲבֶרְתָּהּ בְּיוֹם טוֹב מַיִם אוֹ מֶלַח לְעִסָּתָהּ וּתְבָלִין לִקְדֵרָתָהּ — הֲרֵי הָעִסָּה וְהַתַּבְשִׁיל כְּרַגְלֵי שְׁתֵּיהֶן, וְאֵין אוֹמְרִים שֶׁיִבָּטְלוּ הַמַּיִם וּמֶלַח בְּעִסָּה, מִפְּנֵי שֶׁהוּא דָבָר שֶׁיֵּשׁ לוֹ מַתִּירִין. וְאַף עַל פִּי שֶׁדָּבָר שֶׁיֵּשׁ לוֹ מַתִּירִין אֵינוֹ בָטֵל אֶלָּא בְּמִין שֶׁנִּתְעָרֵב בְּמִינוֹ, אֲבָל בְּשֶׁאֵינוֹ מִינוֹ בָּטֵל הוּא אַף עַל פִּי שֶׁיֵּשׁ לוֹ מַתִּירִין, מִכָּל מָקוֹם כֵּיוָן שֶׁהַמַּיִם וּמֶלַח בָּאִים לְתַקֵּן הָעִסָּה, הֲרֵי הֵם עִמָּהּ כְּמִין בְּמִינוֹ לְעִנְיָן זֶה:

סעיף יג לָקַח מֵחֲבֵרוֹ בְּיוֹם טוֹב גַּחֶלֶת, לֹא…

לָקַח מֵחֲבֵרוֹ בְּיוֹם טוֹב גַּחֶלֶת, לֹא יוֹלִיכֶנָּה אֶלָּא כְּרַגְלֵי הַנּוֹתֵן. אֲבָל אִם הִדְלִיק נֵר אוֹ עֵץ שֶׁלּוֹ מִשַּׁלְהֶבֶת חֲבֵרוֹ — הֲרֵי הוּא כְּרַגְלֵי זֶה שֶׁהִדְלִיק:

סעיף יד בּוֹר שֶׁל יָחִיד — הֲרֵי הוּא…

בּוֹר שֶׁל יָחִיד — הֲרֵי הוּא כְּרַגְלֵי בְעָלִים, וְשֶׁל אוֹתָהּ הָעִיר — כְּרַגְלֵי אַנְשֵׁי אוֹתָהּ הָעִיר.

וְאִם מִי שֶׁהוּא מֵעִיר אַחֶרֶת בָּא וּמִלֵּא עִמָּהֶם — אֵינוֹ מוֹלִיכָם אֶלָּא כְּרַגְלֵי אַנְשֵׁי אוֹתָהּ הָעִיר, דְּהַיְנוּ אַלְפַּיִם לְכָל רוּחַ, אִם כֻּלָּם לֹא עֵרְבוּ. וְאִם אֶחָד מֵהָעִיר עֵרֵב לְצָפוֹן וְאֶחָד לְדָרוֹם — זֶה מוֹלִיךְ לְצָפוֹן כְּרַגְלָיו וְזֶה לְדָרוֹם כְּרַגְלָיו, שֶׁהֲרֵי הֵם שֻׁתָּפִים בַּבּוֹר, וְיֵשׁ בְּרֵרָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּחָבִית שֶׁל יַיִן (בִּסְעִיף י). אֲבָל אִם מִי שֶׁהוּא מֵעִיר אַחֶרֶת בָּא וּמִלֵּא מֵהֶם לְעַצְמוֹ — לֹא יְזִיזֵם מִמְּקוֹם שְׁבִיתָתָם אִם זֶה עֵרֵב לְסוֹף אַלְפַּיִם לְצָפוֹן וְזֶה לְסוֹף אַלְפַּיִם לְדָרוֹם, שֶׁהֲרֵי מַיִם אֵלּוּ הֵם לוֹ כְּרַגְלֵי שְׁנֵיהֶן, שֶׁגַּם מִשֶּׁלָּהֶם הוּא מְמַלֵּא. וְאֵין שַׁיָּךְ לוֹמַר בּוֹ "יֵשׁ בְּרֵרָה", כֵּיוָן שֶׁאֵין לוֹ חֵלֶק כְּלָל בְּבוֹר זֶה. אֲבָל אִם אֶחָד מִבְּנֵי הָעִיר מִלֵּא וְנָתַן לוֹ — מוֹלִיכָן כְּרַגְלֵי הַמְמַלֵּא, שֶׁהֲרֵי מֵחֶלְקוֹ מִלֵּא וְנָתַן לוֹ, שֶׁיֵּשׁ בְּרֵרָה.

וּבוֹר שֶׁל הֶפְקֵר — כְּרַגְלֵי הַמְמַלֵּא רִאשׁוֹן, שֶׁאַף שֶׁלֹּא קָנוּ הַמַּיִם שׁוּם שְׁבִיתָה בְּבֵין הַשְּׁמָשׁוֹת, שֶׁחֶפְצֵי הֶפְקֵר אֵינָם קוֹנִים שְׁבִיתָה, מִכָּל מָקוֹם כְּשֶׁבָּא זֶה וּמִלֵּא, וְזָכָה בְּמַה שֶּׁמִּלֵּא מִן הַהֶפְקֵר — הֲרֵי קָנוּ אָז שְׁבִיתָה כִּשְׁבִיתָתוֹ:

סעיף טו וְכָל זֶה בְּמַיִם מְכֻנָּסִים שֶׁהֵם קְווּיִם…

וְכָל זֶה בְּמַיִם מְכֻנָּסִים שֶׁהֵם קְווּיִם וְעוֹמְדִים, כְּגוֹן מֵי גְשָׁמִים. אֲבָל נְהָרוֹת הַמּוֹשְׁכִים, אַף עַל פִּי שֶׁאֵינָם נוֹבְעִים, הוֹאִיל וְהֵם נָדִים וְנָעִים — לֹא קָנוּ שְׁבִיתָה אֲפִלּוּ הֵם שֶׁל יָחִיד. וְכֵן מַעְיָנוֹת הַנּוֹבְעִים, וְאַף עַל פִּי שֶׁאֵין יוֹצְאִין מִמְּקוֹם נְבִיעָתָם לִמָּשֵׁךְ לְהָלְאָה, הוֹאִיל וְאֵינָם נָחִים — לֹא קָנוּ שְׁבִיתָה אֲפִלּוּ הֵם שֶׁל יָחִיד. וּלְפִיכָךְ הֵם כְּרַגְלֵי הַמְמַלֵּא רִאשׁוֹן, שֶׁאָז קָנוּ שְׁבִיתָה כִּשְׁבִיתָתוֹ, כֵּיוָן שֶׁנִּפְסְקוּ אָז מִמְּקוֹם נְבִיעוּתָם אוֹ מִמְּקוֹם הִלּוּכָם:

סעיף טז הָיוּ בָּאִים מִחוּץ לַתְּחוּם לְתוֹךְ הַתְּחוּם…

הָיוּ בָּאִים מִחוּץ לַתְּחוּם לְתוֹךְ הַתְּחוּם — מְמַלְּאִים מֵהֶם בְּשַׁבָּת, וְאֵין צָרִיךְ לוֹמַר בְּיוֹם טוֹב, הוֹאִיל וְלֹא קָנוּ שְׁבִיתָה כְּלָל חוּץ לַתְּחוּם:

סעיף יז גְּשָׁמִים הַיּוֹרְדִים בְּיוֹם טוֹב בְּתוֹךְ תְּחוּם…

גְּשָׁמִים הַיּוֹרְדִים בְּיוֹם טוֹב בְּתוֹךְ תְּחוּם הָעִיר — הֵם כְּרַגְלֵי אַנְשֵׁי אוֹתָהּ הָעִיר, וְכָל שֶׁכֵּן כְּשֶׁיָּרְדוּ לְתוֹךְ הָעִיר עַצְמָהּ, מִפְּנֵי שֶׁלֹּא קָנוּ שְׁבִיתָה בְּאוּקְיָנוּס אוֹ בֶּעָבִים, הוֹאִיל וְהֵם נָדִים וְנָעִים שָׁם:

סעיף יח מִלֵּא מַיִם מִבּוֹר שֶׁל הֶפְקֵר לְצֹרֶךְ…

מִלֵּא מַיִם מִבּוֹר שֶׁל הֶפְקֵר לְצֹרֶךְ חֲבֵרוֹ — הֵם כְּרַגְלֵי הַמְמַלְּאִים. וְאַף עַל פִּי שֶׁבִּשְׁעַת הַמִּלּוּי זָכָה בָהֶם חֲבֵרוֹ, שֶׁהַמַּגְבִּיהַּ מְצִיאָה לַחֲבֵרוֹ קָנָה חֲבֵרוֹ, מִכָּל מָקוֹם זְכִיָּה זוֹ אֵינָהּ אֶלָּא מִשּׁוּם שֶׁאִם הָיָה רוֹצֶה הָיָה זוֹכֶה בָּהֶם לְעַצְמוֹ, יָכוֹל לִזְכּוֹת בָּהֶם גַּם כֵּן לַחֲבֵרוֹ. וְכֵיוָן שֶׁלֹּא זָכָה אֶלָּא מִכֹּחוֹ — הֲרֵי הֵם כְּרַגְלָיו:

סעיף יט מִי שֶׁהָיוּ לוֹ פֵּרוֹת מֻפְקָדִים בְּעִיר…

מִי שֶׁהָיוּ לוֹ פֵּרוֹת מֻפְקָדִים בְּעִיר אַחֶרֶת שֶׁחוּץ לַתְּחוּם, וְעֵרְבוּ בְּנֵי אוֹתָהּ הָעִיר לָבֹא אֶצְלוֹ — לֹא יָבִיאוּ לוֹ מִפֵּרוֹתָיו, שֶׁפֵּרוֹתָיו כָּמֹהוּ.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁהַנִּפְקָד יִחֵד לוֹ קֶרֶן זָוִית, שֶׁאָמַר לוֹ "הַנִּיחֵם בְּזָוִית זוֹ", שֶׁאָז לֹא קִבֵּל עָלָיו שְׁמִירָתָן כְּלָל, רַק כַּוָּנָתוֹ הִיא "הֲרֵי בֵּיתִי לְפָנֶיךָ, הַנַּח לְךָ בְּזָוִיּוֹת זוֹ וּשְׁמוֹר לְךָ" (וְאֵין צָרִיךְ לוֹמַר אִם אָמַר לוֹ בְּפֵרוּשׁ "הִנֵּה הַבַּיִת לְפָנֶיךָ, הַנַּח לְךָ בְּמָקוֹם שֶׁתִּרְצֶה" — שֶׁלֹּא קִבֵּל עָלָיו שְׁמִירָתָן כְּלָל, כְּמוֹ שֶׁיִּתְבָּאֵר בְּחֹשֶׁן מִשְׁפָּט סִמָּן רצ״[א]). אֲבָל אִם כְּשֶׁהִפְקִידָם אֶצְלוֹ קִבְּלָם מִמֶּנּוּ סְתָם — הֲרֵי קִבֵּל עָלָיו שְׁמִירָתָן וַהֲרֵי הֵם כְּרַגְלָיו, כֵּיוָן שֶׁהֵם מְסוּרִים בְּיָדוֹ לְשָׁמְרָן:

סעיף כ מִי שֶׁזִּמֵּן אֶצְלוֹ אוֹרְחִים בְּיוֹם טוֹב…

מִי שֶׁזִּמֵּן אֶצְלוֹ אוֹרְחִים בְּיוֹם טוֹב — לֹא יוֹלִיכוּ בְּיָדָם מָנוֹת לְמָקוֹם שֶׁאֵין בַּעַל הַסְּעֻדָּה יָכוֹל לֵילֵךְ, אֶלָּא אִם כֵּן זִכָּה לָהֶם בְּמָנוֹת אֵלּוּ עַל יְדֵי אַחֵר מֵעֶרֶב יוֹם טוֹב:

מקור: ספריא — שולחן ערוך הרב, אורח חיים שצ״ז. בספריא לא מופיעים ערכי קונטרס אחרון לסימן זה. הטקסט מועתק כפי שספריא מגישה אותו: בסעיף יח המילה הַמְמַלְּאִים בלשון רבים, וסוף אותו סעיף אומר כְּרַגְלָיו; בסעיף יט, בהפניה לחושן משפט, אות אחת בסוגריים מרובעים. הם מובאים כמות שהם.

2 · לשון הרב ולשון המחבר

לשון הרב מול סעיף המחבר

במה שונה לשון הרב מסעיף המחבר

כל נקודה מעמידה זה לצד זה סעיף של שולחן ערוך המחבר (במספור שלו, שמונה עשר סעיפים, המסומנים באותיותיהם) ואת הסעיף המקביל בשולחן ערוך אדמו״ר הזקן (במספור שלו, עשרים סעיפים), לפי סדר סעיפי הרב, ואינה מציינת אלא מה שעולה מהשוואת שני הלשונות. נקודה שצוינה אינה מיוחדת לרב בשל כך: כמה מן הטעמים האלה נמצאים גם אצל מפרשי המחבר, והדף אינו מציין אותם כאן; אין הוא מציין את כל המקורות ולא את כל המחלוקות. במקום שהדף מציין שנחלקו המפרשים בדין, הוא מפנה לרמה 2.

נקודה א — צורת אלפים האמה (שו״ע הרב שצ״ז:א)

אלפים אמה מרובעות

דברי המחבר: « כל אדם יש לו אלפים אמה לכל רוח חוץ מארבע אמותיו או מהמקום ששבת בו » (שו״ע או״ח שצ״ז:א).

דברי הרב: « כָּל אָדָם יֵשׁ לוֹ חוּץ מִד' אַמּוֹת אוֹ מֵהַמָּקוֹם שֶׁשָּׁבַת בּוֹ אַלְפַּיִם אַמָּה לְכָל רוּחַ, מְרֻבָּעוֹת כְּטַבְלָא מְרֻבַּעַת, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן שצ״ט » (שו״ע הרב שצ״ז:א).

ההבדל: סעיף א של המחבר אינו אומר את צורת אלפים האמה; לשון הרב כותבת שהן מרובעות כטבלא מרובעת, ומפנה לסימן שצ״ט.

נקודה ב — מידת הפסיעה (שו״ע הרב שצ״ז:ב)

אלפים פסיעות בינוניות

דברי המחבר: « קדש עליו היום בבקעה ואינו יודע תחום שבת מהלך אלפי' פסיעו' (בינוניות) (טור) שהם תחום שבת » (שו״ע או״ח שצ״ז:ב).

דברי הרב: « קִדֵּשׁ עָלָיו הַיּוֹם בַּבִּקְעָה הַרְחֵק מִן הָעִיר, שֶׁאֵין שָׁם תְּחוּמֵי שַׁבָּת מְסֻמָּנִים וְאֵינוֹ יוֹדֵעַ לְשַׁעֵר תְּחוּם שַׁבָּת — מְהַלֵּךְ אַלְפַּיִם פְּסִיעוֹת בֵּינוֹנִיּוֹת כְּדֶרֶךְ הֲלִיכָתוֹ שֶׁל אָדָם, וְזֶהוּ תְּחוּם שַׁבָּת, כִּי דֶרֶךְ הֲלִיכָתוֹ שֶׁל אָדָם אַמָּה מֵרֹאשׁ גּוּדָל רֶגֶל זוֹ עַד רֹאשׁ גּוּדָל רֶגֶל זוֹ, שֶׁמַּרְחִיק כַּחֲצִי אַמָּה בֵּין רֶגֶל לְרֶגֶל. וּמְקוֹם מַצַּב הָרֶגֶל עִם הַמִּנְעָל הוּא גַם כֵּן כַּחֲצִי אַמָּה » (שו״ע הרב שצ״ז:ב).

ההבדל: בסעיף ב של המחבר המילה בינוניות באה בסוגריים באותיות קטנות, בלי המילה הגה, ואחריה הסוגריים (טור), והסעיף אינו אומר את מידת הפסיעה; לשון הרב כותבת פסיעות בינוניות בגוף הסעיף, מוסיפה שאין שם תחומי שבת מסומנים, ונותנת את המידה: אמה מראש גודל לראש גודל, כחצי אמה בין רגל לרגל וכחצי אמה למקום מצב הרגל עם המנעל.

נקודה ג — טעם שביתת הכלים והבהמה (שו״ע הרב שצ״ז:ג)

שביתת הכלים

דברי המחבר: « כשם שאין אדם רשאי להלך בשבת וביום טוב אלא אלפי' אמה לכל רוח כך כליו ובהמתו אין יכול שום אדם להוליכ' חוץ לאלפים אמה של בעליה' » (שו״ע או״ח שצ״ז:ג); אינו נותן טעם.

דברי הרב: אחרי אותו דין, « מִפְּנֵי שֶׁקָּנוּ שְׁבִיתָה בִּמְקוֹם שְׁבִיתַת בַּעֲלֵיהֶם, אַף עַל פִּי שֶׁלֹּא הָיוּ שָׁם כְּשֶׁקִּדֵּשׁ הַיּוֹם » (שו״ע הרב שצ״ז:ג).

ההבדל: סעיף ג של המחבר כותב את הדין בלי טעם; לשון הרב נותנת טעם: הכלים והבהמה קנו שביתה במקום שביתת בעליהם, אף על פי שלא היו שם כשקידש היום.

נקודה ד — בהמה שנמסרה לבן (שו״ע הרב שצ״ז:ד)

בהמה שנמסרה לבן

דברי המחבר: « המוסר בהמתו לבנו הרי היא כרגלי האב » (שו״ע או״ח שצ״ז:ד), ולא יותר.

דברי הרב: אחרי אותו דין, « שֶׁאֵין שׁוּם אָדָם יָכוֹל לְהוֹלִיכָהּ אֶלָּא בְּמָקוֹם שֶׁהָאָב יָכוֹל לְהַלֵּךְ וְלֹא בְּמָקוֹם שֶׁהַבֵּן יָכוֹל לְהַלֵּךְ, אַף עַל פִּי שֶׁמְּסָרָהּ לוֹ מִבְּעוֹד יוֹם, לְפִי שֶׁדֶּרֶךְ בְּנֵי אָדָם לְהַפְקִיד חֶפְצֵיהֶם בְּיַד בְּנֵיהֶם וְאֵין כַּוָּנָתָם לִמְסוֹר בִּרְשׁוּתָם » (שו״ע הרב שצ״ז:ד).

ההבדל: לשון הרב מוסיפה שכך הדין אף על פי שמסרה לו מבעוד יום, ונותנת טעם: דרך בני אדם להפקיד חפציהם ביד בניהם, ואין כוונתם למסור ברשותם.

אצל המפרשים: בבהמה שנמסרה לבן נחלקו המפרשים: ראה רמה 2.

נקודה ה — בהמה שנמסרה לרועה (שו״ע הרב שצ״ז:ה)

רועה אחד או שנים

דברי המחבר: « מסרה לרועה אפי' נתנה לו בי״ט הרי הוא כרגלי הרועה ואם מסרה לשני רועים הרי הוא כרגלי בעליה מפני שלא קנה א' מהם » (שו״ע או״ח שצ״ז:ה).

דברי הרב: ברועה, « בֵּין שֶׁהוּא יִשְׂרָאֵל בֵּין שֶׁהוּא נָכְרִי, לְפִי שֶׁמִּן הַסְּתָם גַּם מֵאֶתְמוֹל הָיָה בְּדַעְתּוֹ לְמָסְרָהּ לוֹ הַיּוֹם, וְקָנְתָה שְׁבִיתָה בִּמְקוֹם שְׁבִיתַת הָרוֹעֶה וַאֲפִלּוּ הוּא נָכְרִי, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִמָּן ת״א » (שו״ע הרב שצ״ז:ה); ובשני רועים, « וַאֲפִלּוּ אִם מְסָרָהּ לִשְׁנֵיהֶם מֵעֶרֶב יוֹם טוֹב וְלֹא בֵרֵר דְּבָרָיו, וּבְיוֹם טוֹב בֵּרֵר דְּבָרָיו שֶׁמְּסָרָהּ לְאֶחָד מֵהֶם, וְאֵלָיו נִתְכַּוֵּן מֵעֶרֶב יוֹם טוֹב — אַף עַל פִּי כֵן כֵּיוָן שֶׁבְּשָׁעָה שֶׁקִּדֵּשׁ הַיּוֹם לֹא הָיָה מְבֹרָר עֲדַיִן לְאֵיזֶה מֵהֶם מְסָרָהּ, לֹא קָנוּ שְׁנֵיהֶם, וְנִשְׁאֲרָה בִּרְשׁוּת הַבְּעָלִים וְלָכֵן הִיא כְּרַגְלָיו » (שו״ע הרב שצ״ז:ה).

ההבדל: לשון הרב כותבת שדין הרועה הוא בין שהוא ישראל בין שהוא נכרי, ונותנת טעם; ומוסיפה את הדין שמסרה לשניהם מערב יום טוב ולא בירר דבריו, וביום טוב בירר: לא קנו שניהם, והיא כרגלי הבעלים.

אצל המפרשים: בבהמה שנמסרה לרועה, ובזו שנמסרה לשני רועים, נחלקו המפרשים: ראה רמה 2.

נקודה ו — שור של רועה ושור של פטם (שו״ע הרב שצ״ז:ו, ז)

שני השוורים, בסדר הפוך

דברי המחבר: בסעיף ו, « שור של פטם כרגלי מי שלקחו לשחטו ביו״ט » (שו״ע או״ח שצ״ז:ו); בסעיף ז, « שור של רועה כרגלי אנשי אותה העיר » (שו״ע או״ח שצ״ז:ז).

דברי הרב: בסעיף ו, שור של רועה, ואחר כך « אֲבָל אִם אֶחָד מֵעִיר אַחֶרֶת שֶׁעֵרֵב לְעִיר זוֹ לָקְחוּ מִמֶּנּוּ בְּיוֹם טוֹב — אֵינוֹ יָכוֹל לְהוֹלִיכוֹ לְעִירוֹ » (שו״ע הרב שצ״ז:ו); בסעיף ז, « אֲבָל שׁוֹר שֶׁל פַּטָּם הוּא כְּרַגְלֵי מִי שֶׁלְּקָחוֹ לְשָׁחֳטוֹ בְּיוֹם טוֹב, אֲפִלּוּ הוּא מֵעִיר אַחֶרֶת » (שו״ע הרב שצ״ז:ז).

ההבדל: הרב מביא את שני השוורים בסדר הפוך מסדר המחבר. בשור של רועה הוא מוסיף שאפילו עירב הרועה לרוח אחת אינו מעכב על בן העיר שלקחו ממנו ביום טוב מלהוליכו אלפים לרוח האחרת, עם הטעם, ושאחד מעיר אחרת שעירב לעיר זו ולקחו ממנו ביום טוב אינו יכול להוליכו לעירו; בשור של פטם הוא מוסיף שהדין כן אפילו הוא מעיר אחרת, ואת הטעם.

נקודה ז — חלוק ששאלו שנים (שו״ע הרב שצ״ז:ט)

חלוק ששאלו מערב יום טוב

דברי המחבר: « שנים ששאלו חלוק זה לילך בו שחרית וזה לילך בו ערבית לא יוליכו אלא למקום ששניה' יכולים לילך » (שו״ע או״ח שצ״ז:ט).

דברי הרב: « שְׁנַיִם שֶׁשָּׁאֲלוּ מֵעֶרֶב יוֹם טוֹב חָלוּק אֶחָד, זֶה לֵילֵךְ בּוֹ יוֹם טוֹב שַׁחֲרִית וְזֶה לֵילֵךְ בּוֹ עַרְבִית — הֲרֵי הוּא כְּרַגְלֵי שְׁנֵיהֶם, דְּהַיְנוּ שֶׁאֵין יְכוֹלִים לְהוֹלִיכוֹ אֶלָּא לְמָקוֹם שֶׁשְּׁנֵיהֶם יְכוֹלִים לְהַלֵּךְ » (שו״ע הרב שצ״ז:ט).

ההבדל: סעיף ט של המחבר אינו אומר מתי שאלו את החלוק; לשון הרב כותבת ששאלוהו מערב יום טוב. ובדין שעירב זה למזרח וזה למערב, היא נותנת גם את הטעם שלא יזיזוהו, ומפנה לסימן ת״ח.

נקודה ח — בהמה וחבית של שותפים (שו״ע הרב שצ״ז:י)

הבהמה והחבית

דברי המחבר: « ב' שלקחו בהמה בשותפות ושחטוה ביום טוב אע״פ שלקח כל א' מנתו הרי כל הבשר כרגלי שניהם אבל אם לקחו חבית של יין וחלקוה ביום טוב חלקו של כל א' מהם כרגליו » (שו״ע או״ח שצ״ז:י); אינו נותן טעם.

דברי הרב: בבהמה, « לְפִי שֶׁבֵּין הַשְּׁמָשׁוֹת יָנַק כָּל אֶחָד מֵחֲבֵרוֹ כְּשֶׁהָיְתָה הַבְּהֵמָה בְּחַיֶּיהָ » (שו״ע הרב שצ״ז:י); בחבית, « לְפִי שֶׁבְּשֶׁל דִּבְרֵיהֶם יֵשׁ בְּרֵרָה, דְּהַיְנוּ שֶׁאָנוּ אוֹמְרִים שֶׁהֻבְרַר הַדָּבָר לְמַפְרֵעַ שֶׁזֶּה הָיָה חֶלְקוֹ הַמַּגִּיעַ לוֹ מֵעֶרֶב יוֹם טוֹב, וְקָנָה שְׁבִיתָה בִּמְקוֹם שְׁבִיתָתוֹ » (שו״ע הרב שצ״ז:י).

ההבדל: סעיף י של המחבר כותב את שני הדינים בלי טעם; לשון הרב נותנת טעם לכל אחד מהם, ומפרשת שאת החבית לקחו מערב יום טוב.

אצל המפרשים: בבהמה שלקחו בשותפות ושחטוה ביום טוב נחלקו המפרשים: ראה רמה 2.

נקודה ט — השואל כלי (שו״ע הרב שצ״ז:יא)

שאלו מערב יום טוב או ביום טוב

דברי המחבר: « השואל כלי מחבירו מערב יום טוב אפי' לא לקחו עד הלילה הוא כרגלי השואל ואם שאלו ממנו ביום טוב אע״פ שדרכו לשאלו ממנו בכל יום טוב הרי הוא כרגלי המשאיל » (שו״ע או״ח שצ״ז:יא).

דברי הרב: בשאלו מערב יום טוב, « שֶׁכֵּיוָן שֶׁאָמַר לוֹ מִבְּעוֹד יוֹם לְהַשְׁאִילוֹ — הֶעֱמִידוֹ בִּרְשׁוּתוֹ שֶׁיּוֹלִיכֶנּוּ לְמָקוֹם שֶׁיִּרְצֶה » (שו״ע הרב שצ״ז:יא); בשאלו ביום טוב, « וַאֲפִלּוּ דַּרְכּוֹ לְשָׁאֳלוֹ מִמֶּנּוּ בְּכָל יוֹם טוֹב, אֵין אוֹמְרִים שֶׁמֵּאֶתְמוֹל הֶעֱמִידוֹ בִּרְשׁוּתוֹ, לְפִי שֶׁכֵּיוָן שֶׁלֹּא בָּא לְשָׁאֳלוֹ מִמֶּנּוּ מֵעֶרֶב יוֹם טוֹב, סָבוּר הַמַּשְׁאִיל שֶׁמָּא מְצָאוֹ אָדָם אַחֵר שֶׁהִשְׁאִיל לוֹ וְלֹא יִשְׁאַל מִמֶּנּוּ » (שו״ע הרב שצ״ז:יא).

ההבדל: סעיף יא של המחבר כותב את שני הדינים בלי טעם; לשון הרב נותנת טעם לכל אחד מהם: בשאלו מערב יום טוב — כיון שאמר לו מבעוד יום להשאילו, העמידו ברשותו; בשאלו ביום טוב — סבור המשאיל שמא מצא אדם אחר שהשאיל לו.

נקודה י — מים ומלח שבעיסה אינם בטלים (שו״ע הרב שצ״ז:יב)

מדוע אינם בטלים

דברי המחבר: « האשה ששאלה מחברתה מים ומלח לעיסתה ותבלין לקדירתה בי״ט הרי העיסה ותבשיל כרגלי שתיהן » (שו״ע או״ח שצ״ז:יב); אינו נותן טעם.

דברי הרב: אחרי אותו דין, « וְאֵין אוֹמְרִים שֶׁיִבָּטְלוּ הַמַּיִם וּמֶלַח בְּעִסָּה, מִפְּנֵי שֶׁהוּא דָבָר שֶׁיֵּשׁ לוֹ מַתִּירִין. וְאַף עַל פִּי שֶׁדָּבָר שֶׁיֵּשׁ לוֹ מַתִּירִין אֵינוֹ בָטֵל אֶלָּא בְּמִין שֶׁנִּתְעָרֵב בְּמִינוֹ, אֲבָל בְּשֶׁאֵינוֹ מִינוֹ בָּטֵל הוּא אַף עַל פִּי שֶׁיֵּשׁ לוֹ מַתִּירִין, מִכָּל מָקוֹם כֵּיוָן שֶׁהַמַּיִם וּמֶלַח בָּאִים לְתַקֵּן הָעִסָּה, הֲרֵי הֵם עִמָּהּ כְּמִין בְּמִינוֹ לְעִנְיָן זֶה » (שו״ע הרב שצ״ז:יב).

ההבדל: לשון הרב מבארת מדוע אין המים והמלח בטלים בעיסה: דבר שיש לו מתירין הוא, ואף על פי שדבר כזה בטל בשאינו מינו, המים והמלח, שבאים לתקן העיסה, הרי הם עמה כמין במינו לעניין זה.

נקודה יא — הבור (שו״ע הרב שצ״ז:יד)

מי שהוא מעיר אחרת, ובור של הפקר

דברי המחבר: « בור של יחיד הרי הוא כרגלי בעליו ושל אותה העיר כרגלי אנשי אותה העיר ושל הפקר כרגלי הממלא » (שו״ע או״ח שצ״ז:יד).

דברי הרב: בבור של הפקר, « וּבוֹר שֶׁל הֶפְקֵר — כְּרַגְלֵי הַמְמַלֵּא רִאשׁוֹן, שֶׁאַף שֶׁלֹּא קָנוּ הַמַּיִם שׁוּם שְׁבִיתָה בְּבֵין הַשְּׁמָשׁוֹת, שֶׁחֶפְצֵי הֶפְקֵר אֵינָם קוֹנִים שְׁבִיתָה, מִכָּל מָקוֹם כְּשֶׁבָּא זֶה וּמִלֵּא, וְזָכָה בְּמַה שֶּׁמִּלֵּא מִן הַהֶפְקֵר — הֲרֵי קָנוּ אָז שְׁבִיתָה כִּשְׁבִיתָתוֹ » (שו״ע הרב שצ״ז:יד).

ההבדל: בין בור של העיר לבור של הפקר, לשון הרב מוסיפה את דין מי שהוא מעיר אחרת שבא ומילא עמהם, או מילא מהם לעצמו, ואת דינו של מי שאחד מבני העיר מילא ונתן לו; ובבור של הפקר היא כותבת שהוא כרגלי הממלא ראשון, ונותנת טעם.

נקודה יב — נהרות ומעיינות (שו״ע הרב שצ״ז:טו, טז)

מים שאינם נחים

דברי המחבר: בסעיף אחד, « נהרות המושכי' ומעיינו' הנובעים הרי הם כרגלי הממלא היו באים מחוץ לתחום לתוך התחום ממלאים מהם בשבת ואין צריך לומר ביום טוב » (שו״ע או״ח שצ״ז:טו).

דברי הרב: סעיף טו פותח ב« וְכָל זֶה בְּמַיִם מְכֻנָּסִים שֶׁהֵם קְווּיִם וְעוֹמְדִים, כְּגוֹן מֵי גְשָׁמִים » (שו״ע הרב שצ״ז:טו); סעיף טז כותב את דין המים הבאים מחוץ לתחום, ונותן טעם: « הוֹאִיל וְלֹא קָנוּ שְׁבִיתָה כְּלָל חוּץ לַתְּחוּם » (שו״ע הרב שצ״ז:טז).

ההבדל: לשון הרב תולה את דיני הבור במים מכונסים; היא כותבת שנהרות המושכים, אף על פי שאינם נובעים, ומעיינות הנובעים, אף על פי שאין יוצאים ממקום נביעתם, לא קנו שביתה אפילו הם של יחיד, והם כרגלי הממלא ראשון; ונותנת טעמים שסעיף טו של המחבר אינו נותן.

נקודה יג — גשמים היורדים ביום טוב (שו״ע הרב שצ״ז:יז)

הגשמים

דברי המחבר: בסימן שצ״ז של המחבר אין סעיף על גשמים.

דברי הרב: « גְּשָׁמִים הַיּוֹרְדִים בְּיוֹם טוֹב בְּתוֹךְ תְּחוּם הָעִיר — הֵם כְּרַגְלֵי אַנְשֵׁי אוֹתָהּ הָעִיר, וְכָל שֶׁכֵּן כְּשֶׁיָּרְדוּ לְתוֹךְ הָעִיר עַצְמָהּ, מִפְּנֵי שֶׁלֹּא קָנוּ שְׁבִיתָה בְּאוּקְיָנוּס אוֹ בֶּעָבִים, הוֹאִיל וְהֵם נָדִים וְנָעִים שָׁם » (שו״ע הרב שצ״ז:יז).

ההבדל: לשון הרב כותבת בסימן שצ״ז את דין הגשמים היורדים ביום טוב, שסימן שצ״ז של המחבר אינו כותב.

אצל המפרשים: בגשמים שירדו ביום טוב נחלקו המפרשים: ראה רמה 2.

נקודה יד — המילא לצורך חבירו (שו״ע הרב שצ״ז:יח)

לצורך חבירו

דברי המחבר: « מילא מים מבור של הפקר לצורך חבירו הרי הם כרגלי הממלא » (שו״ע או״ח שצ״ז:טז); אינו נותן טעם.

דברי הרב: אחרי אותו דין, « וְאַף עַל פִּי שֶׁבִּשְׁעַת הַמִּלּוּי זָכָה בָהֶם חֲבֵרוֹ, שֶׁהַמַּגְבִּיהַּ מְצִיאָה לַחֲבֵרוֹ קָנָה חֲבֵרוֹ, מִכָּל מָקוֹם זְכִיָּה זוֹ אֵינָהּ אֶלָּא מִשּׁוּם שֶׁאִם הָיָה רוֹצֶה הָיָה זוֹכֶה בָּהֶם לְעַצְמוֹ, יָכוֹל לִזְכּוֹת בָּהֶם גַּם כֵּן לַחֲבֵרוֹ. וְכֵיוָן שֶׁלֹּא זָכָה אֶלָּא מִכֹּחוֹ — הֲרֵי הֵם כְּרַגְלָיו » (שו״ע הרב שצ״ז:יח).

ההבדל: סעיף טז של המחבר כותב את הדין בלי טעם; לשון הרב מבארת מדוע המים כרגלי הממלא, אף על פי שבשעת המילוי זכה בהם חבירו.

אצל המפרשים: במילא מבור של הפקר לצורך חבירו נחלקו המפרשים: ראה רמה 2.

נקודה טו — פירות מופקדים (שו״ע הרב שצ״ז:יט)

קרן זוית, וקבלת שמירה

דברי המחבר: הסעיף פותח ב« מי שהיו לו פירות מופקדים בעיר אחרת רחוקה ממנו » (שו״ע או״ח שצ״ז:יז), ומסיים ב« אבל אם לא ייחד להם הם כרגלי זה שהם מופקדים אצלו » (שו״ע או״ח שצ״ז:יז).

דברי הרב: הסעיף פותח ב« מִי שֶׁהָיוּ לוֹ פֵּרוֹת מֻפְקָדִים בְּעִיר אַחֶרֶת שֶׁחוּץ לַתְּחוּם » (שו״ע הרב שצ״ז:יט), ומסיים ב« אֲבָל אִם כְּשֶׁהִפְקִידָם אֶצְלוֹ קִבְּלָם מִמֶּנּוּ סְתָם — הֲרֵי קִבֵּל עָלָיו שְׁמִירָתָן וַהֲרֵי הֵם כְּרַגְלָיו, כֵּיוָן שֶׁהֵם מְסוּרִים בְּיָדוֹ לְשָׁמְרָן » (שו״ע הרב שצ״ז:יט).

ההבדל: במקום שסעיף יז של המחבר כותב בעיר אחרת רחוקה ממנו, לשון הרב כותבת בעיר אחרת שחוץ לתחום. היא מבארת את קרן הזוית: אז לא קיבל עליו הנפקד שמירתן כלל, ומוסיפה בסוגריים את מי שאמר לו בפירוש להניח במקום שירצה, עם הפניה לחושן משפט; ובדין האחר היא כותבת שקיבלם ממנו סתם, ולכן קיבל עליו שמירתן.

אצל המפרשים: במה תלוי הדין בסעיף זה, ביחוד קרן זוית או בקבלת שמירה, נחלקו המפרשים: ראה רמה 2.

3 · דברי הרבי

שיחות, מאמרים ואגרות הרבי

דברי הרבי על נושאי סימן שצ״ז

אין הפניה מאומתת נכון לעכשיו.
אנו מייחסים לרבי אך ורק מקורות ראשוניים הניתנים לאימות (מכתב, שיחה או פסק מזוהה, עם קישור). בשלב זה לא ניתן היה לאמת מול Sefaria או Chabad.org דבר מקור של הרבי המתייחס ישירות לסימן זה — מוטב שלא להציג דבר מאשר הפניה לא מאומתת.
כיוון חיפוש מועדף לפסקי הרבי לפי סימן: שערי הלכה ומנהג (מהדורת קה״ת). כל הפניה שתאומת תתווסף עם קישורה.
4 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה המעשית לחב״ד

נקודות הנהגה הלכתית הנובעות ישירות משולחן ערוך אדמו״ר הזקן סימן שצ״ז, ברגישות החב״דית. כל נקודה מפנה לסעיף שלה בשולחן ערוך הרב, לפי סדר הלשון.

לחסיד חב״ד — סימן שצ״ז

  • ① אלפים אמה: לכל אדם יש חוץ מארבע אמות או מהמקום ששבת בו אלפים אמה לכל רוח, מרובעות כטבלא מרובעת (שו״ע הרב שצ״ז:א). הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:א.
  • ② בבקעה: מי שקדש עליו היום בבקעה הרחק מן העיר, שאין שם תחומי שבת מסומנים, ואינו יודע לשער תחום שבת — מהלך אלפים פסיעות בינוניות כדרך הליכתו של אדם; פסיעה היא אמה (שו״ע הרב שצ״ז:ב). הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:ב.
  • ③ כלים ובהמה: אין שום אדם מוליכם חוץ לאלפים של בעליהם; ואם עירבו בעליהם לרוח אחת — אין שום אדם מוליכם לרוח האחרת אפילו פסיעה אחת במקום שאין הבעלים יכולים להלך (שו״ע הרב שצ״ז:ג). הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:ג.
  • ④ בהמה שנמסרה לבן או לרועה: לפי לשון הרב, המוסרה לבנו, אף על פי שמסרה לו מבעוד יום — היא כרגלי האב (שו״ע הרב שצ״ז:ד); המוסרה לרועה, אפילו ביום טוב — היא כרגלי הרועה, בין שהוא ישראל בין שהוא נכרי; מסרה לשני רועים — היא כרגלי הבעלים, ואפילו מסרה לשניהם מערב יום טוב ולא בירר דבריו, וביום טוב בירר (שו״ע הרב שצ״ז:ה). בדינים אלה נחלקו המפרשים: ראה פרק 2, נקודות ד ו־ה, ורמה 2; היעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:ד, ה.
  • ⑤ שני השוורים: שור של רועה — כרגלי אנשי אותה העיר; אבל אחד מעיר אחרת שעירב לעיר זו ולקחו ממנו ביום טוב — אינו יכול להוליכו לעירו (שו״ע הרב שצ״ז:ו). שור של פטם — כרגלי מי שלקחו לשחטו ביום טוב, אפילו הוא מעיר אחרת; ואם שחטו בעליו ביום טוב ומכר בשרו — כל אחד מהלוקחים מוליך מנתו למקום שהוא הולך, אפילו הוא מעיר אחרת (שו״ע הרב שצ״ז:ז). הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:ו, ז.
  • ⑥ דבר של כמה בני אדם: כלים המיוחדים לאחד מהאחים — כרגליו; ושאינם מיוחדים — אין מוליכים אותם אלא למקום שכולם יכולים להלך (שו״ע הרב שצ״ז:ח). חלוק ששאלו שנים מערב יום טוב — אין מוליכים אותו אלא למקום ששניהם יכולים להלך; עירב זה לסוף אלפים למזרח וזה לסוף אלפים למערב — לא יזיזוהו ממקום שביתתם (שו״ע הרב שצ״ז:ט). בשר בהמה שלקחו בשותפות ושחטוה ביום טוב — כרגלי שניהם, אף על פי שחלקוה; אבל חבית של יין שלקחו בשותפות מערב יום טוב וחלקוה ביום טוב — חלקו של כל אחד כרגליו (שו״ע הרב שצ״ז:י). בבהמה שלקחו בשותפות נחלקו המפרשים: ראה פרק 2, נקודה ח, ורמה 2; היעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:ח, ט, י.
  • ⑦ מה ששואלים או לוקחים מאחר: השואל כלי מערב יום טוב, אפילו לא לקחו עד הלילה — הוא כרגלי השואל; שאלו ביום טוב — הוא כרגלי המשאיל, ואפילו דרכו לשאלו ממנו בכל יום טוב (שו״ע הרב שצ״ז:יא). האשה ששאלה מחברתה ביום טוב מים או מלח לעיסתה ותבלין לקדרתה — העיסה והתבשיל כרגלי שתיהן (שו״ע הרב שצ״ז:יב). גחלת שלקח מחבירו ביום טוב — לא יוליכנה אלא כרגלי הנותן; נר או עץ שהדליק משלהבת חבירו — כרגלי זה שהדליק (שו״ע הרב שצ״ז:יג). הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:יא, יב, יג.
  • ⑧ המים: בור של יחיד — כרגלי בעליו, ושל אותה העיר — כרגלי אנשי אותה העיר, ושל הפקר — כרגלי הממלא ראשון; ובמי שהוא מעיר אחרת — ראה סעיף יד (שו״ע הרב שצ״ז:יד). וכל זה במים מכונסים; נהרות המושכים ומעיינות הנובעים, אפילו הם של יחיד — כרגלי הממלא ראשון (שו״ע הרב שצ״ז:טו); ואם היו באים מחוץ לתחום לתוך התחום — ממלאים מהם בשבת, ואין צריך לומר ביום טוב (שו״ע הרב שצ״ז:טז). בגשמים היורדים ביום טוב (שו״ע הרב שצ״ז:יז), ובמילא מבור של הפקר לצורך חבירו (שו״ע הרב שצ״ז:יח), נחלקו המפרשים: ראה פרק 2, נקודות יג ו־יד, ורמה 2; היעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:יד, טו, טז, יז, יח.
  • ⑨ פירות מופקדים: מי שהיו לו פירות מופקדים בעיר אחרת שחוץ לתחום, ועירבו בני אותה העיר לבוא אצלו — לא יביאו לו מפירותיו כשהנפקד ייחד לו קרן זוית, שאז לא קיבל עליו שמירתן; אבל אם קיבלם ממנו סתם — קיבל עליו שמירתן, והם כרגליו (שו״ע הרב שצ״ז:יט). במה תלוי דין זה נחלקו המפרשים: ראה פרק 2, נקודה טו, ורמה 2; היעזר ברב שלך. הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:יט.
  • ⑩ מנות האורחים: מי שזימן אצלו אורחים ביום טוב — לא יוליכו בידם מנות למקום שאין בעל הסעודה יכול לילך, אלא אם כן זיכה להם במנות אלו על ידי אחר מערב יום טוב (שו״ע הרב שצ״ז:כ). הפניה: שולחן ערוך אדמו״ר הזקן שצ״ז:כ.
