דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קי״א · דִּין סְמִיכַת גְּאֻלָּה לִתְפִלָּה

דרכו של אדמו״ר הזקן בדִּין סְמִיכַת גְּאֻלָּה לִתְפִלָּה : אֵין לְהַפְסִיק בֵּין גְּאֻלָּה לִתְפִלָּה — אֲפִלּוּ בְּאָמֵן, חוּץ מִ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״ ; בְּשַׁבָּת — יֵשׁ מִי שֶׁאוֹמֵר אֵין צָרִיךְ ; הַשְּׁלִיחַ צִבּוּר וְהַעֲדִיפוּת עַל תְּפִלַּת הַצִּבּוּר — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קי״א — דִּין סְמִיכַת גְּאֻלָּה לִתְפִלָּה. אצל הרב הסימן נפרש לד׳ סְעִיפִים : עִיקָּר הַסְּמִיכָה וְדִין הַהֶפְסֵק, דִּין שַׁבָּת וְיוֹם טוֹב עִם מָשָׁל אוֹהֵב הַמֶּלֶךְ, הַשְּׁלִיחַ צִבּוּר וּ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״, וַעֲדִיפוּת הַסְּמִיכָה עַל תְּפִלַּת הַצִּבּוּר בְּשַׁחֲרִית.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קי״א — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קי״א). אדמו״ר הזקן דן בו בְּדִין סְמִיכַת גְּאֻלָּה לִתְפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קי״א — דִּין סְמִיכַת גְּאֻלָּה לִתְפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קי״א — דִּין סְמִיכַת גְּאֻלָּה לִתְפִלָּה — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.111), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.111.k).

סעיף א צָרִיךְ לִסְמֹךְ גְּאֻלָּה לִתְפִלָּה, וְלֹא יַפְסִיק — חוּץ מִ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״.

דִּין סְמִיכוּת גְּאֻלָּה לִתְפִלָּה וּבוֹ ד׳ סְעִיפִים:(אַף עַל פִּי שֶׁקְּרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ מִצְוָה בִּפְנֵי עַצְמָהּ וּתְפִלָּה מִצְוָה בִּפְנֵי עַצְמָהּ) צָרִיךְ לִסְמֹךְ בִּרְכַּת גְּאֻלָּה לִתְפִלָּה, וְלֹא יַפְסִיק בֵּינֵיהֶם אֲפִלּוּ בְּאָמֵן אַחַר ״גָּאַל יִשְׂרָאֵל״, וְלֹא בְּשׁוּם פָּסוּק שֶׁנּוֹהֲגִין לוֹמַר קֹדֶם תְּפִלַּת י״ח, כְּגוֹן פָּסוּק ״כִּי שֵׁם ה׳ אֶקְרָא״ וְדוֹמָיו, חוּץ מִפָּסוּק ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״, שֶׁהוּא חוֹבָה מִתַּקָּנַת חֲכָמִים וְאֵינוֹ חָשׁוּב הֶפְסֵק, שֶׁכֵּיוָן שֶׁקְּבָעוּהוּ חֲכָמִים בַּתְּפִלָּה — נַעֲשָׂה מִכְּלַל הַתְּפִלָּה וְכִתְפִלָּה אֲרֻכָּה הִיא. וְיֵשׁ אוֹמְרִים שֶׁמֻּתָּר לַעֲנוֹת אָמֵן אַחַר ״גָּאַל יִשְׂרָאֵל״. וְכֵן נוֹהֲגִין, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ס״ו. וּבְמוּסָף וּמִנְחָה שֶׁאֵין שָׁם סְמִיכַת גְּאֻלָּה לִתְפִלָּה — מֻתָּר לוֹמַר פְּסוּקִים קֹדֶם ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״, אֲבָל לֹא אַחֲרָיו מִשּׁוּם הֶפְסֵק בַּתְּפִלָּה, שֶׁהֲרֵי פָּסוּק זֶה מִכְּלַל הַתְּפִלָּה הוּא:

עִיקַּר הַסְּמִיכָה. אַף שֶׁקְּרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ וְהַתְּפִלָּה כָּל אַחַת מִצְוָה בִּפְנֵי עַצְמָהּ — צָרִיךְ לִסְמֹךְ בִּרְכַּת גְּאֻלָּה לִתְפִלָּה וְלֹא יַפְסִיק, אֲפִלּוּ בְּאָמֵן אַחַר ״גָּאַל יִשְׂרָאֵל״, וְלֹא בְּשׁוּם פָּסוּק. חוּץ מִ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״, שֶׁהוּא חוֹבָה מִתַּקָּנַת חֲכָמִים וְאֵינוֹ הֶפְסֵק, כִּי מִכְּלַל הַתְּפִלָּה הוּא (כִּתְפִלָּה אֲרֻכָּה). וְיֵשׁ אוֹמְרִים שֶׁמֻּתָּר לַעֲנוֹת אָמֵן, וְכֵן נוֹהֲגִין (עיין סימן 66). בְּמוּסָף וּמִנְחָה — מֻתָּר פְּסוּקִים קֹדֶם ״אֲדֹנָי שְׂפָתַי״ אֲבָל לֹא אַחֲרָיו.

סעיף ב בְּשַׁבָּת — יֵשׁ מִי שֶׁאוֹמֵר אֵין צָרִיךְ ; מָשָׁל אוֹהֵב הַמֶּלֶךְ.

יֵשׁ מִי שֶׁאוֹמֵר שֶׁבְּשַׁבָּת אֵין צָרִיךְ לִסְמֹךְ גְּאֻלָּה לִתְפִלָּה, מִפְּנֵי שֶׁסְּמִיכָה זוֹ לְמָדוּהָ חֲכָמִים מִמַּה שֶּׁכָּתוּב: ״יִהְיוּ לְרָצוֹן וְגוֹ׳ צוּרִי וְגֹאֲלִי״, וּכְתִיב אַחֲרָיו: ״יַעַנְךָ ה׳ בְּיוֹם צָרָה״, אֲבָל שַׁבָּת לָאו יוֹם צָרָה הוּא. וְיָמִים טוֹבִים דִּינָם כְּחֹל, הוֹאִיל וְהֵם יְמֵי הַדִּין, שֶׁבַּפֶּסַח הָעוֹלָם נִדּוֹן עַל הַתְּבוּאָה, וּבָעֲצֶרֶת עַל פֵּרוֹת הָאִילָן, וּבֶחָג עַל הַמַּיִם. וְאֵין לִסְמֹךְ עַל סְבָרָא זוֹ אִם לֹא בְּמָקוֹם שֶׁצָּרִיךְ לְכָךְ, לְפִי שֶׁלִּמּוּד זֶה אֵינוֹ אֶלָּא סֶמֶךְ בְּעָלְמָא, וְעִקַּר סְמִיכַת גְּאֻלָּה לִתְפִלָּה הוּא מִדִּבְרֵי חֲכָמִים שֶׁדִּמּוּ י״ח בְּרָכוֹת אַחַר בִּרְכַּת גְּאֻלָּה לְאוֹהֲבוֹ שֶׁל מֶלֶךְ שֶׁבָּא וְדָפַק עַל פֶּתַח הַמֶּלֶךְ וְיָצָא הַמֶּלֶךְ לִקְרָאתוֹ, וְאִם רוֹאֵהוּ הַמֶּלֶךְ שֶׁהִפְלִיג וְהָלַךְ לוֹ אַף הוּא מַפְלִיג, וְשׁוּב אֵין מִתְקָרֵב אֵלָיו כְּשֶׁחוֹזֵר וְדוֹפֵק — כָּךְ הוּא בְּמַפְסִיק בֵּין גְּאֻלָּה לִתְפִלָּה:

דִּין שַׁבָּת. יֵשׁ מִי שֶׁאוֹמֵר שֶׁבְּשַׁבָּת אֵין צָרִיךְ לִסְמֹךְ גְּאֻלָּה לִתְפִלָּה, שֶׁלְּמָדוּהָ מִ״יִהְיוּ לְרָצוֹן… צוּרִי וְגֹאֲלִי״ הַסָּמוּךְ לְ״יַעַנְךָ ה׳ בְּיוֹם צָרָה״ — וְשַׁבָּת לָאו יוֹם צָרָה. וְיָמִים טוֹבִים כְּחֹל, שֶׁהֵם יְמֵי הַדִּין (פֶּסַח עַל הַתְּבוּאָה, עֲצֶרֶת עַל פֵּרוֹת הָאִילָן, חַג עַל הַמַּיִם). וְאֵין לִסְמֹךְ עַל סְבָרָא זוֹ אֶלָּא בְּמָקוֹם שֶׁצָּרִיךְ, שֶׁלִּמּוּד זֶה סֶמֶךְ בְּעָלְמָא ; הָעִיקָּר מִדִּבְרֵי חֲכָמִים שֶׁדִּמּוּ הַדָּבָר לְאוֹהֵב הַמֶּלֶךְ הַדּוֹפֵק בַּפֶּתַח — שֶׁאִם הִפְלִיג, אַף הַמֶּלֶךְ מַפְלִיג.

סעיף ג שְׁלִיחַ צִבּוּר — אוֹמֵר ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״ בְּלַחַשׁ.

הַשְּׁלִיחַ צִבּוּר כְּשֶׁמַּתְחִיל י״ח בְּקוֹל — צָרִיךְ לוֹמַר תְּחִלָּה פָּסוּק ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״, הוֹאִיל וְהוּא מִכְּלַל הַתְּפִלָּה. וְטוֹב שֶׁיֹּאמְרֶנּוּ בְּלַחַשׁ. אֲבָל שְׁאָר פְּסוּקִים אֵין צָרִיךְ לוֹמַר. וְאִם רָצָה לוֹמַר — רַשַּׁאי, וְאֵין אִסּוּר כְּלָל לְהַפְסִיק (בְּעִנְיְנֵי הַתְּפִלָּה וּצְרָכֶיהָ) בֵּין תְּפִלַּת לַחַשׁ לִתְפִלָּה שֶׁבְּקוֹל (אֲבָל לְהִתְעַסֵּק בִּדְבָרִים אֲחֵרִים אָסוּר עַד אַחַר נְפִילַת אַפַּיִם, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קל״א):

הַשְּׁלִיחַ צִבּוּר. הַשְּׁלִיחַ צִבּוּר, כְּשֶׁמַּתְחִיל י״ח בְּקוֹל, צָרִיךְ לוֹמַר תְּחִלָּה ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״, הוֹאִיל וְהוּא מִכְּלַל הַתְּפִלָּה, וְטוֹב שֶׁיֹּאמְרֶנּוּ בְּלַחַשׁ. שְׁאָר פְּסוּקִים אֵין צָרִיךְ ; רָצָה — רַשַּׁאי. וְאֵין אִסּוּר לְהַפְסִיק (בְּעִנְיְנֵי הַתְּפִלָּה) בֵּין לַחַשׁ לַחֲזָרָה שֶׁבְּקוֹל, אֲבָל לְהִתְעַסֵּק בִּדְבָרִים אֲחֵרִים אָסוּר עַד אַחַר נְפִילַת אַפַּיִם (עיין סימן 131).

סעיף ד מָצָא צִבּוּר מִתְפַּלְּלִין — סְמִיכַת גְּאֻלָּה לִתְפִלָּה שַׁחֲרִית גְּדוֹלָה.

אִם עַד שֶׁלֹּא קָרָא קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ מָצָא צִבּוּר שֶׁמִּתְפַּלְלִין — לֹא יִתְפַּלֵּל עִמָּהֶם תְּחִלָּה וְלִקְרוֹת אַחַר כָּךְ קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ כְּמוֹ שֶׁעוֹשִׂים בִּפְסוּקֵי דְּזִמְרָה, לְפִי שֶׁסְּמִיכַת גְּאֻלָּה לִתְפִלָּה שַׁחֲרִית גְּדוֹלָה מִתְּפִלָּה עִם הַצִּבּוּר (אֲבָל בְּעַרְבִית תְּפִלַּת הַצִּבּוּר גְּדוֹלָה מִסְּמִיכַת גְּאֻלָּה לִתְפִלָּה, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן רל״ו):

עֲדִיפוּת הַסְּמִיכָה. אִם עַד שֶׁלֹּא קָרָא ק״ש וּבִרְכוֹתֶיהָ מָצָא צִבּוּר מִתְפַּלְּלִין — לֹא יִתְפַּלֵּל עִמָּהֶם תְּחִלָּה וְלִקְרוֹת אַחַר כָּךְ (כְּמוֹ בִּפְסוּקֵי דְּזִמְרָה), שֶׁסְּמִיכַת גְּאֻלָּה לִתְפִלָּה בְּשַׁחֲרִית גְּדוֹלָה מִתְּפִלָּה עִם הַצִּבּוּר. אֲבָל בְּעַרְבִית — תְּפִלַּת הַצִּבּוּר גְּדוֹלָה מִסְּמִיכַת גְּאֻלָּה לִתְפִלָּה (עיין סימן 236).

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קי״א (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין סְמִיכַת גְּאֻלָּה לִתְפִלָּה

היסוד : דִּין סְמִיכַת גְּאֻלָּה לִתְפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה עִיקָּר הַסְּמִיכָה וְדִין הַהֶפְסֵק (עִם יוֹצֵא הַכְּלָל ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״). אחר כך דִּין שַׁבָּת וְיוֹם טוֹב וְמָשָׁל אוֹהֵב הַמֶּלֶךְ. אחר כך הַשְּׁלִיחַ צִבּוּר. ולבסוף עֲדִיפוּת הַסְּמִיכָה עַל תְּפִלַּת הַצִּבּוּר בְּשַׁחֲרִית.

עִיקָּר הַסְּמִיכָה — אֵין הֶפְסֵק

עִיקָּר הַסְּמִיכָה — אֵין הֶפְסֵק

היסוד. צָרִיךְ לִסְמֹךְ גְּאֻלָּה לִתְפִלָּה ; לֹא יַפְסִיק — אֲפִלּוּ בְּאָמֵן אַחַר ״גָּאַל יִשְׂרָאֵל״ וְלֹא בְּשׁוּם פָּסוּק. חוּץ מִ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״ שֶׁהוּא מִכְּלַל הַתְּפִלָּה (סעיף א).

המסקנה. יֵשׁ אוֹמְרִים שֶׁמֻּתָּר אָמֵן אַחַר ״גָּאַל יִשְׂרָאֵל״ — וְכֵן נוֹהֲגִין. עיין סימן 66 (עניית אמן).

שַׁבָּת וְיוֹם טוֹב — מָשָׁל אוֹהֵב הַמֶּלֶךְ

שַׁבָּת וְיוֹם טוֹב — מָשָׁל אוֹהֵב הַמֶּלֶךְ

היסוד. יֵשׁ מִי שֶׁאוֹמֵר שֶׁבְּשַׁבָּת אֵין צָרִיךְ (שַׁבָּת לָאו יוֹם צָרָה) ; יָמִים טוֹבִים כְּחֹל (יְמֵי הַדִּין). וְאֵין לִסְמֹךְ עַל סְבָרָא זוֹ אֶלָּא בְּמָקוֹם שֶׁצָּרִיךְ (סעיף ב).

המסקנה. הָעִיקָּר מִדִּבְרֵי חֲכָמִים (מָשָׁל אוֹהֵב הַמֶּלֶךְ). עיין סימן 110 (תפילת הדרך ובקצרה).

הַשְּׁלִיחַ צִבּוּר

הַשְּׁלִיחַ צִבּוּר

היסוד. הַשְּׁלִיחַ צִבּוּר כְּשֶׁמַּתְחִיל י״ח בְּקוֹל — אוֹמֵר תְּחִלָּה ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״, וְטוֹב בְּלַחַשׁ ; שְׁאָר פְּסוּקִים אֵין צָרִיךְ (סעיף ג).

המסקנה. אֵין אִסּוּר לְהַפְסִיק בְּעִנְיְנֵי הַתְּפִלָּה בֵּין לַחַשׁ לַחֲזָרָה. עיין סימן 131 (נפילת אפים).

עֲדִיפוּת הַסְּמִיכָה בְּשַׁחֲרִית

עֲדִיפוּת הַסְּמִיכָה בְּשַׁחֲרִית

היסוד. מָצָא צִבּוּר מִתְפַּלְּלִין — לֹא יִתְפַּלֵּל עִמָּהֶם תְּחִלָּה, שֶׁסְּמִיכַת גְּאֻלָּה לִתְפִלָּה שַׁחֲרִית גְּדוֹלָה מִתְּפִלָּה עִם הַצִּבּוּר (סעיף ד).

המסקנה. בְּעַרְבִית — תְּפִלַּת הַצִּבּוּר גְּדוֹלָה. עיין סימן 236 (ברכות ק״ש של ערבית).

הפניה ללימוד. היקף הסימן — אֵין הֶפְסֵק, דִּין שַׁבָּת וַעֲדִיפוּת הַסְּמִיכָה — נוגע להנהגת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין סְמִיכַת גְּאֻלָּה לִתְפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קי״א. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
אֵין הֶפְסֵק וְ״אֲדֹנָי שְׂפָתַי״ או״ח קי״א — צָרִיךְ לִסְמֹךְ גְּאֻלָּה לִתְפִלָּה ; לֹא יַפְסִיק אֲפִלּוּ בְּאָמֵן, חוּץ מֵ״ה׳ שְׂפָתַי תִּפְתָּח״. מ״ב קי״א — ״אֲדֹנָי שְׂפָתַי״ אֵינוֹ הֶפְסֵק, וּגְדֶר הָאָמֵן. ערוה״ש קי״א — טַעַם הַסְּמִיכָה וְדִין הַהַפְסָקוֹת. דרך אדמו״ר הזקן — לֹא יַפְסִיק אֲפִלּוּ בְּאָמֵן וְלֹא בְּשׁוּם פָּסוּק ; ״אֲדֹנָי שְׂפָתַי״ מִכְּלַל הַתְּפִלָּה, כִּתְפִלָּה אֲרֻכָּה (סעיף א).
שַׁבָּת וְיוֹם טוֹב או״ח קי״א — וְיֵשׁ אוֹמְרִים בְּשַׁבָּת אֵין צָרִיךְ ; וְטוֹב לְהַחְמִיר. מ״ב קי״א — גֶּדֶר ״יוֹם צָרָה״ וְיָמִים טוֹבִים יְמֵי הַדִּין. ערוה״ש קי״א — הַהַכְרָעָה בְּשַׁבָּת וְיוֹם טוֹב. דרך אדמו״ר הזקן — שַׁבָּת לָאו יוֹם צָרָה ; יָמִים טוֹבִים כְּחֹל (יְמֵי הַדִּין) ; וְאֵין לִסְמֹךְ עַל סְבָרָא זוֹ אֶלָּא בְּמָקוֹם שֶׁצָּרִיךְ (סעיף ב).
הַשְּׁלִיחַ צִבּוּר או״ח קי״א — הַחַזָּן כְּשֶׁמַּתְחִיל י״ח בְּקוֹל רָם חוֹזֵר וְאוֹמֵר ״ה׳ שְׂפָתַי תִּפְתָּח״. מ״ב קי״א — שֶׁיֹּאמְרֶנּוּ בְּלַחַשׁ וְדִין שְׁאָר פְּסוּקִים. ערוה״ש קי״א — הַפְסָקָה בֵּין לַחַשׁ לַחֲזָרָה. דרך אדמו״ר הזקן — אוֹמֵר תְּחִלָּה ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״ וְטוֹב בְּלַחַשׁ ; אֵין אִסּוּר לְהַפְסִיק בְּעִנְיְנֵי הַתְּפִלָּה עַד נְפִילַת אַפַּיִם (סעיף ג).
עֲדִיפוּת הַסְּמִיכָה או״ח קי״א — מָצָא צִבּוּר מִתְפַּלְּלִין לֹא יִתְפַּלֵּל עִמָּהֶם, אֶלָּא קוֹרֵא ק״ש וְאַחַר כָּךְ יִתְפַּלֵּל. מ״ב קי״א — סְמִיכַת גְּאֻלָּה לִתְפִלָּה עֲדִיפָא מִתְּפִלָּה בְּצִבּוּר בְּשַׁחֲרִית. ערוה״ש קי״א — חִלּוּק שַׁחֲרִית וְעַרְבִית בָּעֲדִיפוּת. דרך אדמו״ר הזקן — שַׁחֲרִית: סְמִיכָה גְּדוֹלָה מִתְּפִלָּה עִם הַצִּבּוּר ; עַרְבִית: תְּפִלַּת הַצִּבּוּר גְּדוֹלָה (סעיף ד).

הטבלה נערכה מטקסט המחבר (אורח חיים קי״א, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע א, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ד׳-ט׳, ירושלמי ברכות). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — אֵין הֶפְסֵק בֵּין גְּאֻלָּה לִתְפִלָּה

קו א — אֵין הֶפְסֵק בֵּין גְּאֻלָּה לִתְפִלָּה

יסוד הסימן : צָרִיךְ לִסְמֹךְ ; לֹא יַפְסִיק אֲפִלּוּ בְּאָמֵן, חוּץ מִ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״ (סעיף א).

קריאת הרב : ״אֲדֹנָי שְׂפָתַי״ אֵינוֹ הֶפְסֵק, שֶׁמִּכְּלַל הַתְּפִלָּה הוּא (כִּתְפִלָּה אֲרֻכָּה). עיין סימן 66.

מסקנה למעשה : אֵין אָמֵן וְלֹא פְּסוּקִים ; בְּמוּסָף/מִנְחָה — פְּסוּקִים מֻתָּרִים קֹדֶם ״אֲדֹנָי שְׂפָתַי״, לֹא אַחֲרָיו.

קו ב — שַׁבָּת וְטַעַם הַסְּמִיכָה

קו ב — שַׁבָּת וְטַעַם הַסְּמִיכָה

יסוד הסימן : יֵשׁ מִי שֶׁאוֹמֵר שֶׁבְּשַׁבָּת אֵין צָרִיךְ ; יָמִים טוֹבִים כְּחֹל (סעיף ב).

קריאת הרב : לִמּוּד זֶה סֶמֶךְ בְּעָלְמָא ; הָעִיקָּר מִדִּבְרֵי חֲכָמִים (מָשָׁל אוֹהֵב הַמֶּלֶךְ). עיין סימן 112.

מסקנה למעשה : אֵין לִסְמֹךְ עַל סְבָרַת שַׁבָּת אֶלָּא בְּמָקוֹם שֶׁצָּרִיךְ.

קו ג — עֲדִיפוּת הַסְּמִיכָה בְּשַׁחֲרִית

קו ג — עֲדִיפוּת הַסְּמִיכָה בְּשַׁחֲרִית

יסוד הסימן : מָצָא צִבּוּר מִתְפַּלְּלִין — לֹא יִתְפַּלֵּל עִמָּהֶם תְּחִלָּה ; סְמִיכָה שַׁחֲרִית גְּדוֹלָה (סעיף ד).

קריאת הרב : בְּעַרְבִית — תְּפִלַּת הַצִּבּוּר גְּדוֹלָה מִסְּמִיכַת גְּאֻלָּה לִתְפִלָּה. עיין סימן 236.

מסקנה למעשה : בְּשַׁחֲרִית — קוֹרֵא תְּחִלָּה ק״ש וּבִרְכוֹתֶיהָ לִסְמֹךְ, אַף עַל חֶשְׁבּוֹן תְּפִלָּה בְּצִבּוּר.

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קי״א ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קי״א (דִּין סְמִיכַת גְּאֻלָּה לִתְפִלָּה)

  • ① אֵין הֶפְסֵק. לִסְמֹךְ גְּאֻלָּה לִתְפִלָּה ; לֹא לְהַפְסִיק — אֲפִלּוּ בְּאָמֵן וְלֹא בְּשׁוּם פָּסוּק. יסוד : שוע״ר קי״א סעיף א.
  • ② ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״. הַיּוֹצֵא הַיָּחִיד : פָּסוּק זֶה, שֶׁמִּכְּלַל הַתְּפִלָּה (כִּתְפִלָּה אֲרֻכָּה). בְּמוּסָף/מִנְחָה — פְּסוּקִים מֻתָּרִים קֹדֶם, לֹא אַחֲרָיו. יסוד : שוע״ר קי״א סעיף א.
  • ③ שַׁבָּת וְיוֹם טוֹב. יֵשׁ מִי שֶׁאוֹמֵר שֶׁבְּשַׁבָּת אֵין צָרִיךְ ; יָמִים טוֹבִים כְּחֹל. אֵין סוֹמְכִין אֶלָּא בְּמָקוֹם שֶׁצָּרִיךְ. יסוד : שוע״ר קי״א סעיף ב.
  • ④ שְׁלִיחַ צִבּוּר. כְּשֶׁמַּתְחִיל י״ח בְּקוֹל — אוֹמֵר תְּחִלָּה ״אֲדֹנָי שְׂפָתַי תִּפְתָּח״, וְטוֹב בְּלַחַשׁ ; שְׁאָר פְּסוּקִים אֵין צָרִיךְ. יסוד : שוע״ר קי״א סעיף ג.
  • ⑤ עֲדִיפוּת בְּשַׁחֲרִית. מָצָא צִבּוּר מִתְפַּלְּלִין קֹדֶם ק״ש — לֹא יִתְפַּלֵּל עִמָּהֶם ; הַסְּמִיכָה בְּשַׁחֲרִית גְּדוֹלָה (בְּעַרְבִית — לְהֵפֶךְ). יסוד : שוע״ר קי״א סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קי״א (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — אֵין הֶפְסֵק, דִּין שַׁבָּת, הָעֲדִיפוּת — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
