דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קכ״ז · דִּין מוֹדִים דְּרַבָּנָן

דרכו של אדמו״ר הזקן בדִּין מוֹדִים דְּרַבָּנָן : הַשְּׁחִיָּה עִם הַשַּׁ״ץ וְהַחֲתִימָה ״בָּרוּךְ אֵל הַהוֹדָאוֹת״ בְּלֹא הַזְכָּרַת הַשֵּׁם ; הַבִּרְכַּת כֹּהֲנִים עַל יְדֵי הַשַּׁ״ץ וְ״כֵּן יְהִי רָצוֹן״ ; נֻסְחֵי ״שִׂים שָׁלוֹם״ וּ״שָׁלוֹם רָב״ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קכ״ז — דִּין מוֹדִים דְּרַבָּנָן. אצל הרב הסימן נפרש לג׳ סְעִיפִים : הַשְּׁחִיָּה עִם הַשַּׁ״ץ בְּ״מוֹדִים״ וְהַחֲתִימָה, הַבִּרְכַּת כֹּהֲנִים עַל יְדֵי הַשַּׁ״ץ עִם ״כֵּן יְהִי רָצוֹן״, וְנֻסְחֵי ״שִׂים שָׁלוֹם״ / ״שָׁלוֹם רָב״.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קכ״ז — ג׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קכ״ז). אדמו״ר הזקן דן בו בְּדִין מוֹדִים דְּרַבָּנָן. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קכ״ז — דִּין מוֹדִים דְּרַבָּנָן

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קכ״ז — דִּין מוֹדִים דְּרַבָּנָן — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.127), ואחריו תרגומנו. הסימן כולל ג׳ סעיפים (סעיף k → ‎.127.k).

סעיף א כְּשֶׁמַּגִּיעַ הַשַּׁ״ץ לְ״מוֹדִים״ — שׁוֹחִין ; חוֹתְמִין ״בָּרוּךְ אֵל הַהוֹדָאוֹת״.

דִּין מוֹדִים דְּרַבָּנָן וּבוֹ ג׳ סְעִיפִים:כְּשֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לְ״מוֹדִים״ — שׁוֹחִין עִמּוֹ הַצִּבּוּר. וַאֲפִלּוּ יָחִיד הַמִּתְפַּלֵּל י״ח — צָרִיךְ לָשׁוּחַ עִמּוֹ אִם הוּא עוֹמֵד בְּאֶמְצַע בְּרָכָה כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ק״ט, כְּדֵי שֶׁלֹּא יְהֵא נִרְאֶה כְּכוֹפֵר בְּמִי שֶׁחֲבֵרָיו מִשְׁתַּחֲוִים לוֹ. וְלֹא דַּי בִּשְׁחִיָּה מוּעֶטֶת, אֶלָּא כְּשִׁעוּר שְׁאָר שְׁחִיּוֹת שֶׁנִּתְבָּאֵר בְּסִימָן קי״ג, וּבִלְבַד שֶׁלֹּא יִשְׁחוּ יוֹתֵר מִדַּאי כְּמוֹ שֶׁנִּתְבָּאֵר שָׁם. וְאוֹמְרִים הַצִּבּוּר ״מוֹדִים״ שֶׁהוּא תִּקּוּן הַרְבֵּה חֲכָמִים, וְנִקְרָא ״מוֹדִים דְּרַבָּנָן״. וְיֵשׁ אוֹמְרִים שֶׁחוֹתְמִין ״בָּרוּךְ אַתָּה ה׳ אֵל הַהוֹדָאוֹת״. וְיֵשׁ אוֹמְרִים שֶׁלֹּא לַחְתֹּם. לְפִיכָךְ נוֹהֲגִין לַחְתֹּם בְּלֹא הַזְכָּרַת הַשֵּׁם, וְאוֹמְרִים ״בָּרוּךְ אֵל הַהוֹדָאוֹת״. וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁצָּרִיךְ לִשְׁחוֹת גַּם בַּסּוֹף בְּ״עַל שֶׁאָנוּ מוֹדִים לָךְ״. וְטוֹב לָחֹש לִדְבָרָיו. וּבִמְדִינוֹת אֵלּוּ הַמִּנְהָג לוֹמַר הַכֹּל בִּשְׁחִיָּה אַחַת מִתְּחִלָּה וְעַד סוֹף, אַף עַל פִּי שֶׁיֵּשׁ בּוֹ הַזְכָּרָה — אֵין לְהַקְפִּיד מִשּׁוּם ״ה׳ זֹקֵף כְּפוּפִים״ אֶלָּא בְּסוֹף הַבְּרָכָה אוֹ בִּתְחִלָּתָהּ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן קי״ג:

מוֹדִים דְּרַבָּנָן. כְּשֶׁמַּגִּיעַ הַשַּׁ״ץ לְ״מוֹדִים״ — שׁוֹחִין עִמּוֹ הַצִּבּוּר, וַאֲפִלּוּ יָחִיד הָעוֹמֵד בְּאֶמְצַע בְּרָכָה (סימן ק״ט), כְּדֵי שֶׁלֹּא יְהֵא נִרְאֶה כְּכוֹפֵר. לֹא שְׁחִיָּה מוּעֶטֶת אֶלָּא כְּשִׁעוּר שְׁאָר שְׁחִיּוֹת (סימן קי״ג), וְלֹא יוֹתֵר מִדַּאי. אוֹמְרִים ״מוֹדִים דְּרַבָּנָן״, וְחוֹתְמִין בְּלֹא הַזְכָּרַת הַשֵּׁם ״בָּרוּךְ אֵל הַהוֹדָאוֹת״. וְטוֹב לָחֹש לִשְׁחוֹת גַּם בַּסּוֹף. בִּמְדִינוֹת אֵלּוּ — הַכֹּל בִּשְׁחִיָּה אַחַת.

סעיף ב אֵין כֹּהֲנִים — הַשַּׁ״ץ אוֹמֵר בִּרְכַּת כֹּהֲנִים ; ״כֵּן יְהִי רָצוֹן״ בִּמְקוֹם אָמֵן.

יֵשׁ מְקוֹמוֹת שֶׁהַכֹּהֲנִים נוֹשְׂאִים כַּפֵּיהֶם בְּכָל יוֹם, וְאֵין אוֹמְרִים שָׁם ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ בָּרְכֵנוּ כו׳״, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ח. וְאִם אֵין שָׁם כֹּהֲנִים — צָרִיךְ הַשְּׁלִיחַ צִבּוּר לוֹמַר בִּרְכַּת כֹּהֲנִים, שֶׁהִיא ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ בָּרְכֵנוּ כו׳״ עַד ״וַאֲנִי אֲבָרֲכֵם״. וּבִמְדִינוֹת אֵלּוּ אֵין אוֹמְרִים רַק עַד ״שָׁלוֹם״. וַאֲפִלּוּ כְּשֶׁהַכֹּהֲנִים נוֹשְׂאִים כַּפֵּיהֶם אוֹמֵר אוֹתָהּ הַשְּׁלִיחַ צִבּוּר בִּמְדִינוֹת אֵלּוּ, כְּמוֹ שֶׁיִּתְבָּאֵר שָׁם. וּבִזְמַן שֶׁאֵינוֹ רָאוּי לִנְשִׂיאוּת כַּפַּיִם — אֵין אוֹמְרִים אוֹתָהּ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן קכ״א. וְאֵין הַצִּבּוּר נוֹהֲגִים לַעֲנוֹת אָמֵן אַחַר בִּרְכַּת כֹּהֲנִים שֶׁאוֹמֵר הַשְּׁלִיחַ צִבּוּר, לְפִי שֶׁאֵין צָרִיךְ לַעֲנוֹת אָמֵן אֶלָּא כְּשֶׁשּׁוֹמְעִין מִפִּי כֹּהֵן הַמְבָרֵךְ. אֶלָּא נוֹהֲגִין לוֹמַר ״כֵּן יְהִי רָצוֹן״ בִּמְקוֹם אָמֵן. יֵשׁ נוֹהֲגִין לוֹמַר כֵּן בְּסוֹף הַבְּרָכָה בִּלְבַד. וְיֵשׁ נוֹהֲגִין לוֹמַר כֵּן אַחַר כָּל פָּסוּק. וְהַנּוֹהֲגִין לוֹמַר אַחַר כָּל פָּסוּק ״יְהִי רָצוֹן״ וּבְסוֹף פָּסוּק ג׳ ״כֵּן יְהִי רָצוֹן״ — טוֹעִים הֵם, שֶׁ״יְּהִי רָצוֹן״ בְּלֹא מִלַּת ״כֵּן״ — אֵין מַשְׁמָעוּת אֵיזֶה דָּבָר ״יְהִי רָצוֹן״:

בִּרְכַּת כֹּהֲנִים עַל יְדֵי הַשַּׁ״ץ. בִּמְקוֹמוֹת שֶׁהַכֹּהֲנִים נוֹשְׂאִים כַּפֵּיהֶם בְּכָל יוֹם אֵין אוֹמְרִים ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״ (סימן קכ״ח). אֵין שָׁם כֹּהֲנִים — הַשַּׁ״ץ אוֹמֵר בִּרְכַּת כֹּהֲנִים עַד ״וַאֲנִי אֲבָרֲכֵם״ ; בִּמְדִינוֹת אֵלּוּ רַק עַד ״שָׁלוֹם״. וּבִזְמַן שֶׁאֵינוֹ רָאוּי לִנְשִׂיאוּת כַּפַּיִם — אֵין אוֹמְרִים אוֹתָהּ (סימן קכ״א). הַצִּבּוּר אֵין עוֹנִים אָמֵן אַחַר הַשַּׁ״ץ, אֶלָּא ״כֵּן יְהִי רָצוֹן״ בִּמְקוֹם אָמֵן. ״יְהִי רָצוֹן״ בְּלֹא מִלַּת ״כֵּן״ — טָעוּת, שֶׁאֵין מַשְׁמָעוּת.

סעיף ג ״שִׂים שָׁלוֹם״ בְּשַׁחֲרִית ; ״שָׁלוֹם רָב״ בְּעַרְבִית וּבְמִנְחָה.

נָהֲגוּ בִּמְדִינוֹת אֵלּוּ לוֹמַר בְּשַׁחֲרִית ״שִׂים שָׁלוֹם״. וְכֵן בְּכָל זְמַן שֶׁאוֹמֵר ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ כו׳״. אֲבָל בְּעַרְבִית מַתְחִילִין ״שָׁלוֹם רָב״. וְכֵן בְּמִנְחָה אֲפִלּוּ הַשְּׁלִיחַ צִבּוּר, כְּשֶׁאֵין אוֹמְרִים ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ כו׳״. וְיֵשׁ מַתְחִילִין ״שִׂים שָׁלוֹם״ בְּמִנְחָה שֶׁל שַׁבָּת, הוֹאִיל וְאוֹמְרִים בּוֹ ״כִּי בְאוֹר פָּנֶיךָ נָתַתָּ לָּנוּ״, שֶׁהִיא הַתּוֹרָה שֶׁקּוֹרִין בְּמִנְחָה בְּשַׁבָּת:

שִׂים שָׁלוֹם וְשָׁלוֹם רָב. נָהֲגוּ בִּמְדִינוֹת אֵלּוּ לוֹמַר בְּשַׁחֲרִית ״שִׂים שָׁלוֹם״, וְכֵן בְּכָל זְמַן שֶׁאוֹמֵר ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״ ; אֲבָל בְּעַרְבִית מַתְחִילִין ״שָׁלוֹם רָב״, וְכֵן בְּמִנְחָה — אֲפִלּוּ הַשַּׁ״ץ — כְּשֶׁאֵין אוֹמְרִים ״אֱלֹהֵינוּ״. וְיֵשׁ מַתְחִילִין ״שִׂים שָׁלוֹם״ בְּמִנְחָה שֶׁל שַׁבָּת, הוֹאִיל וְאוֹמְרִים בּוֹ ״כִּי בְאוֹר פָּנֶיךָ נָתַתָּ לָּנוּ״ — הַתּוֹרָה שֶׁקּוֹרִין בְּמִנְחָה בְּשַׁבָּת.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קכ״ז (מהדורת קה״ת, ג׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

היסוד — דִּין מוֹדִים דְּרַבָּנָן

היסוד : דִּין מוֹדִים דְּרַבָּנָן

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים. תחילה הַשְּׁחִיָּה בְּמוֹדִים וְהַחֲתִימָה ״בָּרוּךְ אֵל הַהוֹדָאוֹת״. אחר כך הַבִּרְכַּת כֹּהֲנִים עַל יְדֵי הַשַּׁ״ץ וְ״כֵּן יְהִי רָצוֹן״. ולבסוף נֻסְחֵי הַשָּׁלוֹם — ״שִׂים שָׁלוֹם״ וּ״שָׁלוֹם רָב״.

מוֹדִים דְּרַבָּנָן וְהַשְּׁחִיָּה

מוֹדִים דְּרַבָּנָן וְהַשְּׁחִיָּה

היסוד. כְּשֶׁמַּגִּיעַ הַשַּׁ״ץ לְ״מוֹדִים״ שׁוֹחִין עִמּוֹ הַצִּבּוּר — וַאֲפִלּוּ יָחִיד בְּאֶמְצַע בְּרָכָה ; כְּשִׁעוּר שְׁאָר שְׁחִיּוֹת, וְלֹא יוֹתֵר מִדַּאי (סעיף א).

המסקנה. חוֹתְמִין בְּלֹא הַזְכָּרַת הַשֵּׁם ״בָּרוּךְ אֵל הַהוֹדָאוֹת״, וְטוֹב לָחֹש לִשְׁחוֹת גַּם בַּסּוֹף. עיין סימן 113 (שיעור הכריעות).

בִּרְכַּת כֹּהֲנִים עַל יְדֵי הַשַּׁ״ץ

בִּרְכַּת כֹּהֲנִים עַל יְדֵי הַשַּׁ״ץ

היסוד. אֵין שָׁם כֹּהֲנִים — הַשַּׁ״ץ אוֹמֵר ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״ עַד ״וַאֲנִי אֲבָרֲכֵם״ ; בִּמְדִינוֹת אֵלּוּ עַד ״שָׁלוֹם״ (סעיף ב).

המסקנה. אֵין עוֹנִים אָמֵן, אֶלָּא ״כֵּן יְהִי רָצוֹן״ — מִלַּת ״כֵּן״ מְעַכֶּבֶת אֶת הַמַּשְׁמָעוּת. עיין סימן 128 (נשיאת כפים).

שִׂים שָׁלוֹם וְשָׁלוֹם רָב

שִׂים שָׁלוֹם וְשָׁלוֹם רָב

היסוד. בְּשַׁחֲרִית — ״שִׂים שָׁלוֹם״ ; בְּעַרְבִית וּבְמִנְחָה — ״שָׁלוֹם רָב״ (סעיף ג).

המסקנה. וְיֵשׁ מַתְחִילִין ״שִׂים שָׁלוֹם״ בְּמִנְחָה שֶׁל שַׁבָּת, מִשּׁוּם ״כִּי בְאוֹר פָּנֶיךָ״. עיין סימן 126 (חזרת הש״ץ).

הפניה ללימוד. היקף הסימן — הַשְּׁחִיָּה, הַבִּרְכַּת כֹּהֲנִים וְנֻסְחֵי הַשָּׁלוֹם — נוגע להנהגת חֲזָרַת הַשַּׁ״ץ היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין מוֹדִים דְּרַבָּנָן — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קכ״ז. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מג׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
מוֹדִים דְּרַבָּנָן וְהַשְּׁחִיָּה או״ח קכ״ז — שׁוֹחִין עִם הַשַּׁ״ץ, וְחוֹתֵם ״בָּרוּךְ אֵל הַהוֹדָאוֹת״ בְּלֹא הַזְכָּרַת הַשֵּׁם. מ״ב קכ״ז — שִׁעוּר הַשְּׁחִיָּה וְהַמִּנְהָג בִּשְׁחִיָּה אַחַת. ערוה״ש קכ״ז — נֻסַּח מוֹדִים דְּרַבָּנָן וְגֶדֶר הַחֲתִימָה. דרך אדמו״ר הזקן — שׁוֹחִין וְאַף יָחִיד בְּאֶמְצַע ; כְּשִׁעוּר שְׁאָר שְׁחִיּוֹת, לֹא יוֹתֵר מִדַּאי ; חוֹתְמִין בְּלֹא שֵׁם ; הַכֹּל בִּשְׁחִיָּה אַחַת (סעיף א).
בִּרְכַּת כֹּהֲנִים עַל יְדֵי הַשַּׁ״ץ או״ח קכ״ז — אֵין כֹּהֲנִים הַשַּׁ״ץ אוֹמֵר ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״ ; אֵין עוֹנִין אָמֵן. מ״ב קכ״ז — ״כֵּן יְהִי רָצוֹן״ בִּמְקוֹם אָמֵן וְגִדְרוֹ. ערוה״ש קכ״ז — דִּין עֲנִיַּת הַצִּבּוּר וְהַנֻּסָּח. דרך אדמו״ר הזקן — עַד ״וַאֲנִי אֲבָרֲכֵם״ (בִּמְדִינוֹת אֵלּוּ עַד ״שָׁלוֹם״) ; ״כֵּן יְהִי רָצוֹן״ ; מִלַּת ״כֵּן״ מְעַכֶּבֶת הַמַּשְׁמָעוּת (סעיף ב).
שִׂים שָׁלוֹם וְשָׁלוֹם רָב או״ח קכ״ז — שִׂים שָׁלוֹם בְּשַׁחֲרִית ; מִנְחָה שֶׁל שַׁבָּת. מ״ב קכ״ז — זְמַנֵּי ״שִׂים שָׁלוֹם״ וְ״שָׁלוֹם רָב״. ערוה״ש קכ״ז — מִנְהַג הַנֻּסָּח בַּתְּפִלּוֹת. דרך אדמו״ר הזקן — ״שִׂים שָׁלוֹם״ בְּשַׁחֲרִית וּכְשֶׁאוֹמֵר אֱלֹהֵינוּ ; ״שָׁלוֹם רָב״ בְּעַרְבִית וּמִנְחָה ; מִנְחַת שַׁבָּת ״כִּי בְאוֹר פָּנֶיךָ״ (סעיף ג).

הטבלה נערכה מטקסט המחבר (אורח חיים קכ״ז, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפלה ט, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד סוטה מ׳, ברכות כ״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַשְּׁחִיָּה בְּמוֹדִים

קו א — הַשְּׁחִיָּה בְּמוֹדִים

יסוד הסימן : שׁוֹחִין עִם הַשַּׁ״ץ בְּ״מוֹדִים״, כְּשִׁעוּר שְׁאָר שְׁחִיּוֹת וְלֹא יוֹתֵר מִדַּאי (סעיף א).

קריאת הרב : אֲפִלּוּ יָחִיד בְּאֶמְצַע בְּרָכָה שׁוֹחֶה, כְּדֵי שֶׁלֹּא יְהֵא נִרְאֶה כְּכוֹפֵר. עיין סימן 113.

מסקנה למעשה : חוֹתְמִין בְּלֹא הַזְכָּרַת הַשֵּׁם ״בָּרוּךְ אֵל הַהוֹדָאוֹת״, וְהַמִּנְהָג הַכֹּל בִּשְׁחִיָּה אַחַת.

קו ב — בִּרְכַּת כֹּהֲנִים בְּפִי הַשַּׁ״ץ

קו ב — בִּרְכַּת כֹּהֲנִים בְּפִי הַשַּׁ״ץ

יסוד הסימן : אֵין שָׁם כֹּהֲנִים — הַשַּׁ״ץ אוֹמֵר ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״ (בִּמְדִינוֹת אֵלּוּ עַד ״שָׁלוֹם״) (סעיף ב).

קריאת הרב : אֵין עוֹנִים אָמֵן אֶלָּא כְּשֶׁשּׁוֹמְעִין מִפִּי כֹּהֵן — כָּאן ״כֵּן יְהִי רָצוֹן״. עיין סימן 128.

מסקנה למעשה : מִלַּת ״כֵּן״ נִצְרֶכֶת ; ״יְהִי רָצוֹן״ לְבַדּוֹ אֵין מַשְׁמָעוּת.

קו ג — נֻסְחֵי הַשָּׁלוֹם

קו ג — נֻסְחֵי הַשָּׁלוֹם

יסוד הסימן : ״שִׂים שָׁלוֹם״ בְּשַׁחֲרִית ; ״שָׁלוֹם רָב״ בְּעַרְבִית וּבְמִנְחָה (סעיף ג).

קריאת הרב : הַנֻּסָּח נִקְבָּע לְפִי אֲמִירַת ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״. עיין סימן 126.

מסקנה למעשה : בְּמִנְחָה שֶׁל שַׁבָּת, יֵשׁ מַתְחִילִין ״שִׂים שָׁלוֹם״ מִשּׁוּם ״כִּי בְאוֹר פָּנֶיךָ״.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קכ״ז ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קכ״ז (דִּין מוֹדִים דְּרַבָּנָן)

  • ① שְׁחִיָּה. כְּשֶׁמַּגִּיעַ הַשַּׁ״ץ לְ״מוֹדִים״ — שׁוֹחִין עִמּוֹ, כְּשִׁעוּר שְׁאָר שְׁחִיּוֹת, וְלֹא יוֹתֵר מִדַּאי. יסוד : שוע״ר קכ״ז סעיף א.
  • ② חֲתִימָה. אוֹמְרִים ״מוֹדִים דְּרַבָּנָן״ וְחוֹתְמִין בְּלֹא הַזְכָּרַת הַשֵּׁם ״בָּרוּךְ אֵל הַהוֹדָאוֹת״ ; הַמִּנְהָג הַכֹּל בִּשְׁחִיָּה אַחַת. יסוד : שוע״ר קכ״ז סעיף א.
  • ③ בִּרְכַּת כֹּהֲנִים. אֵין שָׁם כֹּהֲנִים — הַשַּׁ״ץ אוֹמֵר ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״ ; בִּמְדִינוֹת אֵלּוּ עַד ״שָׁלוֹם״. יסוד : שוע״ר קכ״ז סעיף ב.
  • ④ עֲנִיָּה. אֵין עוֹנִים אָמֵן אַחַר הַשַּׁ״ץ — אֶלָּא ״כֵּן יְהִי רָצוֹן״, עִם מִלַּת ״כֵּן״ הַנִּצְרֶכֶת. יסוד : שוע״ר קכ״ז סעיף ב.
  • ⑤ שָׁלוֹם. ״שִׂים שָׁלוֹם״ בְּשַׁחֲרִית ; ״שָׁלוֹם רָב״ בְּעַרְבִית וּבְמִנְחָה ; מִנְחַת שַׁבָּת — יֵשׁ מַתְחִילִין ״שִׂים שָׁלוֹם״. יסוד : שוע״ר קכ״ז סעיף ג.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קכ״ז (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַשְּׁחִיָּה, הַבִּרְכַּת כֹּהֲנִים, נֻסְחֵי הַשָּׁלוֹם — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
