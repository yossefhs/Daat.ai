דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן נ׳ · טַעַם אֲמִירַת מִשְׁנַת אֵיזֶהוּ מְקוֹמָן

דרכו של אדמו״ר הזקן בטַעַם אֲמִירַת מִשְׁנַת אֵיזֶהוּ מְקוֹמָן: אַחַר פָּרָשַׁת הַתָּמִיד קָבְעוּ לוֹמַר פֶּרֶק אֵיזֶהוּ מְקוֹמָן וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל, כְּדֵי שֶׁיִּזְכֶּה כָּל אָדָם בְּכָל יוֹם לְמִקְרָא מִשְׁנָה וְתַלְמוּד — שֶׁהַמִּדְרָשׁ כְּתַלְמוּד ; הַלִּמּוּד אֵינוֹ עוֹלֶה אֶלָּא לְמִי שֶׁמֵּבִין ; וְאוֹמְרִים אֵיזֶהוּ מְקוֹמָן אַף בְּבֵית הָאָבֵל, שֶׁהוּא סֵדֶר הַיּוֹם — ברגישות חב״ד.

למה רמה 4 מיוחדת לאדמו״ר הזקן? השולחן ערוך של אדמו״ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתבו אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. דף זה מאסף את שיטת הרב על סימן נ׳ — טַעַם אֲמִירַת מִשְׁנַת אֵיזֶהוּ מְקוֹמָן. הסימן כולל אצל הרב שְׁלֹשָׁה סְעִיפִים, שבהם אדמו״ר הזקן פורש: הַטַּעַם לוֹמַר פֶּרֶק אֵיזֶהוּ מְקוֹמָן (משנה, זבחים) וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל אַחַר פָּרָשַׁת הַתָּמִיד, כְּדֵי לִזְכּוֹת בְּכָל יוֹם לְמִקְרָא מִשְׁנָה וְתַלְמוּד (שֶׁהַמִּדְרָשׁ כְּתַלְמוּד) ; שֶׁהַלִּמּוּד אֵינוֹ עוֹלֶה אֶלָּא לְמִי שֶׁמֵּבִין ; וְשֶׁאוֹמְרִים אֵיזֶהוּ מְקוֹמָן אַף בְּבֵית הָאָבֵל, שֶׁהוּא סֵדֶר הַיּוֹם וְלֹא תַּלְמוּד תּוֹרָה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בדף זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן נ׳ — שְׁלֹשָׁה סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנסרקה בספריא (Shulchan Arukh HaRav, Orach Chayim נ׳). אדמו״ר הזקן דן בטַעַם לומר אחר פָּרָשַׁת הַתָּמִיד את משנת אֵיזֶהוּ מְקוֹמָן וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל — מִקְרָא מִשְׁנָה וְתַלְמוּד בְּכָל יוֹם, שֶׁהַמִּדְרָשׁ כְּתַלְמוּד — בתנאי הַהֲבָנָה (מִי שֶׁמֵּבִין) ובאמירה בְּבֵית הָאָבֵל (סֵדֶר הַיּוֹם). הביאורים הם שלנו. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן נ׳ — טַעַם אֲמִירַת מִשְׁנַת אֵיזֶהוּ מְקוֹמָן

הטקסט המלא של שולחן ערוך אדמו״ר הזקן

שולחן ערוך הרב, סימן נ׳ — טַעַם אֲמִירַת מִשְׁנַת אֵיזֶהוּ מְקוֹמָן — וּבוֹ שְׁלֹשָׁה סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.50), ולאחריו ביאור קצר. הסימן כולל שְׁלֹשָׁה סְעִיפִים.

סעיף א קָבְעוּ לִשְׁנוֹת אַחַר פָּרָשַׁת הַתָּמִיד פֶּרֶק אֵיזֶהוּ מְקוֹמָן וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל, כְּדֵי שֶׁיִּזְכֶּה כָּל אָדָם לִלְמֹד מִקְרָא מִשְׁנָה וְתַלְמוּד.

טַעַם לָמָּה אוֹמְרִים מִשְׁנַת ״אֵיזֶהוּ מְקוֹמָן״ וּבוֹ ג׳ סְעִיפִים:קָבְעוּ לִשְׁנוֹת אַחַר פָּרָשַׁת הַתָּמִיד פֶּרֶק ״אֵיזֶהוּ מְקוֹמָן״ וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל, כְּדֵי שֶׁיִּזְכֶּה כָּל אָדָם לִלְמֹד בְּכָל יוֹם מִקְרָא מִשְׁנָה וְתַלְמוּד, דִּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל הִיא בִּמְקוֹם תַּלְמוּד שֶׁהַמִּדְרָשׁ כְּתַלְמוּד. וּמִפְּנֵי שֶׁהִיא תְּחִלַּת תּוֹרַת כֹּהֲנִים, שֶׁהוּא רֹאשׁ לְכָל פָּרָשַׁת הַקָּרְבָּנוֹת — קְבָעוּהָ אֵצֶל הַקָּרְבָּנוֹת. וְכֵן מִשְׁנַת ״אֵיזֶהוּ מְקוֹמָן״ קְבָעוּהָ אַחַר הַתָּמִיד, מִשּׁוּם שֶׁנֶּאֱמַר: ״וּבְכָל מָקוֹם מֻקְטָר מֻגָּשׁ לִשְׁמִי״, וְכִי בְּכָל מָקוֹם מַקְטִירִין וּמַגִּישִׁין? אֶלָּא אֵלּוּ תַּלְמִידֵי חֲכָמִים שֶׁעוֹסְקִין בְּהִלְכוֹת עֲבוֹדָה בְּכָל יוֹם בְּכָל מָקוֹם שֶׁהֵם — מַעֲלֶה אֲנִי עֲלֵיהֶם כְּאִלּוּ מַקְטִירִים וּמַגִּישִׁים לִשְׁמִי. וְעוֹד, מִפְּנֵי שֶׁבְּפֶרֶק זֶה אֵין בּוֹ מַחֲלֹקֶת וְהוּא מִשְׁנָה בְּרוּרָה לְמֹשֶׁה מִסִּינַי:

טַעַם אֲמִירַת מִשְׁנַת אֵיזֶהוּ מְקוֹמָן. קָבְעוּ חֲכָמִים לוֹמַר, אַחַר פָּרָשַׁת הַתָּמִיד, אֶת פֶּרֶק אֵיזֶהוּ מְקוֹמָן (משנה, זבחים) וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל, כְּדֵי שֶׁיִּזְכֶּה כָּל אָדָם לִלְמֹד בְּכָל יוֹם מִקְרָא מִשְׁנָה וְתַלְמוּד — שֶׁהַבְּרַיְתָא דְּרַבִּי יִשְׁמָעֵאל הִיא בִּמְקוֹם תַּלְמוּד, שֶׁהַמִּדְרָשׁ כְּתַלְמוּד. וּלְפִי שֶׁהִיא תְּחִלַּת תּוֹרַת כֹּהֲנִים, רֹאשׁ לְכָל פָּרָשַׁת הַקָּרְבָּנוֹת — קְבָעוּהָ אֵצֶל הַקָּרְבָּנוֹת. וּמִשְׁנַת אֵיזֶהוּ מְקוֹמָן נִקְבְּעָה אַחַר הַתָּמִיד, עַל פִּי וּבְכָל מָקוֹם מֻקְטָר מֻגָּשׁ לִשְׁמִי : תַּלְמִידֵי חֲכָמִים הָעוֹסְקִים בְּהִלְכוֹת עֲבוֹדָה בְּכָל יוֹם, בְּכָל מָקוֹם, נֶחְשָׁבִים כְּאִלּוּ מַקְטִירִים וּמַגִּישִׁים לִשְׁמוֹ. וְעוֹד, שֶׁפֶּרֶק זֶה אֵין בּוֹ מַחֲלֹקֶת, מִשְׁנָה בְּרוּרָה לְמֹשֶׁה מִסִּינַי.

סעיף ב אֵין קְרִיאַת פֶּרֶק זֶה עוֹלָה לְלִמּוּד אֶלָּא לְמִי שֶׁמֵּבִין ; מִי שֶׁאֵינוֹ מֵבִין צָרִיךְ לִלְמֹד וּלְהָבִין.

אֵין קְרִיאַת פֶּרֶק זֶה וּבְרַיְתָא זוֹ עוֹלָה לְלִמּוּד מִשְׁנָה וְתַלְמוּד אֶלָּא לְמִי שֶׁמֵּבִין, אֲבָל מִי שֶׁאֵינוֹ מֵבִין — צָרִיךְ לִלְמֹד וּלְהָבִין, שֶׁאִם לֹא כֵן אֵינוֹ נֶחְשָׁב לְלִמּוּד, שֶׁאַף בִּתְפִלָּה מוּטָב לְהִתְפַּלֵּל בְּלָשׁוֹן שֶׁמֵּבִין, אֶלָּא שֶׁיֵּשׁ לוֹמַר שֶׁאַף שֶׁהוּא אֵינוֹ מֵבִין הַקָּדוֹשׁ בָּרוּךְ הוּא יוֹדֵעַ כַּוָּנָתוֹ וּמֵבִין, אֲבָל אִם אוֹמֵר הַמִּשְׁנָה וְאֵינוֹ מֵבִין — אֵינוֹ נִקְרָא לִמּוּד כְּלָל:

אֵין עוֹלֶה לְלִמּוּד אֶלָּא לְמִי שֶׁמֵּבִין. קְרִיאַת פֶּרֶק זֶה וּבְרַיְתָא זוֹ אֵינָהּ עוֹלָה לְלִמּוּד מִשְׁנָה וְתַלְמוּד אֶלָּא לְמִי שֶׁמֵּבִין. מִי שֶׁאֵינוֹ מֵבִין — צָרִיךְ לִלְמֹד וּלְהָבִין, שֶׁאִם לֹא כֵן אֵינוֹ נֶחְשָׁב לְלִמּוּד, שֶׁאַף בַּתְּפִלָּה מוּטָב לְהִתְפַּלֵּל בְּלָשׁוֹן שֶׁמֵּבִין. וְאַף שֶׁיֵּשׁ לוֹמַר שֶׁגַּם כְּשֶׁאֵינוֹ מֵבִין הַקָּדוֹשׁ בָּרוּךְ הוּא יוֹדֵעַ כַּוָּנָתוֹ וּמֵבִין — מִכָּל מָקוֹם אִם אוֹמֵר הַמִּשְׁנָה וְאֵינוֹ מֵבִין, אֵינוֹ נִקְרָא לִמּוּד כְּלָל.

סעיף ג יֵשׁ מְקוֹמוֹת שֶׁאֵין אוֹמְרִים אֵיזֶהוּ מְקוֹמָן בְּבֵית הָאָבֵל — וְאֵין זֶה נָכוֹן, שֶׁכָּל שֶׁהוּא סֵדֶר הַיּוֹם אֵין בּוֹ מִשּׁוּם תַּלְמוּד תּוֹרָה.

יֵשׁ מְקוֹמוֹת שֶׁאֵין אוֹמְרִים ״אֵיזֶהוּ מְקוֹמָן״ בְּבֵית הָאָבֵל, מִפְּנֵי שֶׁאָסוּר בְּתַלְמוּד תּוֹרָה. וְאֵין זֶה נָכוֹן, שֶׁכָּל שֶׁהוּא סֵדֶר הַיּוֹם — אֵין בּוֹ מִשּׁוּם תַּלְמוּד תּוֹרָה לָאָבֵל, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תקנ״ד וּבְיוֹרֶה דֵּעָה סִימָן שפ״ד:

אֲמִירָה בְּבֵית הָאָבֵל. יֵשׁ מְקוֹמוֹת שֶׁאֵין אוֹמְרִים אֵיזֶהוּ מְקוֹמָן בְּבֵית הָאָבֵל, מִפְּנֵי שֶׁהָאָבֵל אָסוּר בְּתַלְמוּד תּוֹרָה. וְאֵין זֶה נָכוֹן : כָּל שֶׁהוּא סֵדֶר הַיּוֹם — אֵין בּוֹ מִשּׁוּם תַּלְמוּד תּוֹרָה לָאָבֵל, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תקנ״ד וּבְיוֹרֶה דֵּעָה סִימָן שפ״ד.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim נ׳ (מהדורת קה״ת, שְׁלֹשָׁה סְעִיפִים). ביאור: DAAT. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.

היסוד — מִקְרָא מִשְׁנָה וְתַלְמוּד וְהַמִּדְרָשׁ כְּתַלְמוּד

היסוד: לימוד בכל יום והמדרש כתלמוד

קריאת אדמו״ר הזקן על סימן זה בשלושה מהלכים — שלושת הסעיפים. תחילה הַטַּעַם לַמִּנְהָג: לוֹמַר בְּכָל יוֹם, אַחַר פָּרָשַׁת הַתָּמִיד, אֶת מִשְׁנַת אֵיזֶהוּ מְקוֹמָן וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל, כְּדֵי לִלְמֹד מִקְרָא מִשְׁנָה וְתַלְמוּד — שֶׁהַמִּדְרָשׁ כְּתַלְמוּד. אַחַר כָּךְ הַתְּנַאי: אֵין הַלִּמּוּד עוֹלֶה אֶלָּא לְמִי שֶׁמֵּבִין. וּלְבַסּוֹף דִּין הַבֵּית הָאָבֵל: אוֹמְרִים בּוֹ אֵיזֶהוּ מְקוֹמָן, שֶׁהוּא סֵדֶר הַיּוֹם וְלֹא תַּלְמוּד תּוֹרָה.

מִקְרָא מִשְׁנָה וְתַלְמוּד בְּכָל יוֹם — הַמִּדְרָשׁ כְּתַלְמוּד

טעם המנהג: ללמוד בכל יום מקרא, משנה ותלמוד

היסוד. קָבְעוּ חֲכָמִים לוֹמַר, אַחַר פָּרָשַׁת הַתָּמִיד, אֶת פֶּרֶק אֵיזֶהוּ מְקוֹמָן וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל, כְּדֵי שֶׁיִּזְכֶּה כָּל אָדָם בְּכָל יוֹם לְמִקְרָא (פְּסוּקִים), מִשְׁנָה (אֵיזֶהוּ מְקוֹמָן) וְתַלְמוּד — שֶׁהַבְּרַיְתָא דְּרַבִּי יִשְׁמָעֵאל בִּמְקוֹם תַּלְמוּד, שֶׁהַמִּדְרָשׁ כְּתַלְמוּד.

התוצאה. אֵיזֶהוּ מְקוֹמָן נִבְחַר כְּפֶרֶק שֶׁאֵין בּוֹ מַחֲלֹקֶת (מִשְׁנָה בְּרוּרָה לְמֹשֶׁה מִסִּינַי), וְנִקְבַּע אַחַר הַתָּמִיד עַל פִּי וּבְכָל מָקוֹם מֻקְטָר מֻגָּשׁ לִשְׁמִי. עַל פָּרָשַׁת הַקָּרְבָּנוֹת עַיֵּן סימן מ״ח · 48.

אֵין עוֹלֶה לְלִמּוּד אֶלָּא לְמִי שֶׁמֵּבִין

התנאי: אינו נחשב לימוד אלא למי שמבין

העיקרון. קְרִיאַת הַפֶּרֶק וְהַבְּרַיְתָא אֵינָהּ עוֹלָה לְלִמּוּד מִשְׁנָה וְתַלְמוּד אֶלָּא לְמִי שֶׁמֵּבִין. מִי שֶׁאֵינוֹ מֵבִין — צָרִיךְ לִלְמֹד וּלְהָבִין, שֶׁאִם לֹא כֵן אֵינוֹ נֶחְשָׁב לְלִמּוּד.

ההרחבה. אַף בַּתְּפִלָּה מוּטָב לְהִתְפַּלֵּל בְּלָשׁוֹן שֶׁמֵּבִין ; וְאִם אוֹמֵר הַמִּשְׁנָה וְאֵינוֹ מֵבִין, אֵינוֹ נִקְרָא לִמּוּד כְּלָל. עַל בִּרְכַּת הַתּוֹרָה וְהַלִּמּוּד עַיֵּן סימן מ״ז · 47.

אֲמִירָה בְּבֵית הָאָבֵל — סֵדֶר הַיּוֹם

אמירה בבית האבל: שהוא סדר היום

העיקרון. יֵשׁ שֶׁאֵין אוֹמְרִים אֵיזֶהוּ מְקוֹמָן בְּבֵית הָאָבֵל, מִפְּנֵי שֶׁהָאָבֵל אָסוּר בְּתַלְמוּד תּוֹרָה. וְהָרַב דּוֹחֶה מִנְהָג זֶה: וְאֵין זֶה נָכוֹן.

ההנהגה. כָּל שֶׁהוּא סֵדֶר הַיּוֹם אֵין בּוֹ מִשּׁוּם תַּלְמוּד תּוֹרָה לָאָבֵל — כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תקנ״ד וּבְיוֹרֶה דֵּעָה סִימָן שפ״ד. עַיֵּן גַּם סימן נ״א · 51.

הפניה ללימוד. משקל היסוד שהונח כאן — מִקְרָא מִשְׁנָה וְתַלְמוּד בְּכָל יוֹם, הַמִּדְרָשׁ כְּתַלְמוּד, תְּנַאי הַהֲבָנָה וְסֵדֶר הַיּוֹם — נוגע גם בהנהגה היומיומית (אמירת אֵיזֶהוּ מְקוֹמָן בכל בוקר) וגם ברגישות חב״ד על תַּלְמוּד תּוֹרָה יומיומי. דף זה מציג זאת ברמת העיקרון המתועד בטקסט של הרב, בלי להוסיף דבר מומצא. ללימוד מעמיק ולהנהגה אישית — יש ללמוד עם רב של חב״ד ובתוך המקורות עצמם.

טַעַם אֲמִירַת מִשְׁנַת אֵיזֶהוּ מְקוֹמָן — בעיקרון · ללימוד, פנה לרב של חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרכו של אדמו״ר הזקן עם המחבר, הרמ״א (רמ״א) והמשנה ברורה (משנה ברורה) בנקודות המפתח של סימן נ׳. שורת שולחן ערוך הרב (שולחן ערוך הרב) מובלטת — היא חושפת את רגישות הרב כפי שהיא עולה משלושת סעיפיו. הרב מפרש את הטעם ואת הפרט במקום שהמחבר מסתפק באמירה.

נושא מחבר רמ״א משנה ברורה דרך אדמו״ר הזקן (שולחן ערוך הרב)
אֲמִירַת אֵיזֶהוּ מְקוֹמָן — מִקְרָא מִשְׁנָה וְתַלְמוּד או״ח נ׳ — אוֹמְרִים אֵיזֶהוּ מְקוֹמָן וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל אַחַר פָּרָשַׁת הַתָּמִיד. — מ״ב נ׳ / קידושין ל׳ ע״א — מִקְרָא מִשְׁנָה וְתַלְמוּד בְּכָל יוֹם. דרך אדמו״ר הזקן — כְּדֵי שֶׁיִּזְכֶּה כָּל אָדָם לְמִקְרָא מִשְׁנָה וְתַלְמוּד ; דִּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל בִּמְקוֹם תַּלְמוּד שֶׁהַמִּדְרָשׁ כְּתַלְמוּד (סעיף א).
לִמּוּד בַּהֲבָנָה — מִי שֶׁמֵּבִין או״ח נ׳ — סְתַם אֲמִירַת הַפֶּרֶק וְהַבְּרַיְתָא. — מ״ב נ׳ — יָבִין פֵּרוּשׁ הַדְּבָרִים. דרך אדמו״ר הזקן — אֵין עוֹלֶה לְלִמּוּד אֶלָּא לְמִי שֶׁמֵּבִין ; אוֹמֵר וְאֵינוֹ מֵבִין אֵינוֹ נִקְרָא לִמּוּד כְּלָל (סעיף ב).
אֲמִירָה בְּבֵית הָאָבֵל — סֵדֶר הַיּוֹם או״ח נ׳ — סְתָם אֲמִירַת אֵיזֶהוּ מְקוֹמָן. — מ״ב נ׳ / יו״ד שפ״ד — דִּינֵי תַּלְמוּד תּוֹרָה לָאָבֵל. דרך אדמו״ר הזקן — אוֹמְרִים אֵיזֶהוּ מְקוֹמָן בְּבֵית הָאָבֵל, שֶׁכָּל סֵדֶר הַיּוֹם אֵין בּוֹ מִשּׁוּם תַּלְמוּד תּוֹרָה לָאָבֵל (סעיף ג).

הטבלה נערכה מטקסט המחבר (אורח חיים נ׳, סָעִיף אֶחָד, מקור ספריא) ומנושאי הכלים הניתנים לאימות (בית יוסף, טור, רמב״ם — הלכות תלמוד תורה, רא״ש, אבודרהם, כל בו, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד קידושין ל׳ ע״א, זבחים מ״ז ע״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (שְׁלֹשָׁה סְעִיפִים) המובא ב§1. ייחוסי סעיף קטן שאינם ודאיים נותרו « על דרך הסברה ».

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה — מעבר ליישום הפשוט של הכלל. כל אחת בנויה: עיקרון הסימן → קריאת הרב → תוצאה מעשית. כל קו נשען על הטקסט האמיתי של הרב (§1) ועל שלושת סעיפיו.

קו א — מִקְרָא מִשְׁנָה וְתַלְמוּד וְהַמִּדְרָשׁ כְּתַלְמוּד

הלימוד היומיומי והמדרש כתלמוד

עיקרון הסימן: לוֹמַר אֵיזֶהוּ מְקוֹמָן וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל אַחַר פָּרָשַׁת הַתָּמִיד (סעיף א).

קריאת הרב: כְּדֵי שֶׁיִּזְכֶּה כָּל אָדָם לְמִקְרָא מִשְׁנָה וְתַלְמוּד בְּכָל יוֹם ; הַבְּרַיְתָא בִּמְקוֹם תַּלְמוּד, שֶׁהַמִּדְרָשׁ כְּתַלְמוּד.

תוצאה מעשית: לוֹמַר בְּכָל בּוֹקֶר אֶת הַמִּשְׁנָה וְהַבְּרַיְתָא, בְּכַוָּנַת הַלִּמּוּד הַיּוֹמִי.

קו ב — לִמּוּד בַּהֲבָנָה

לימוד המצריך הבנה

עיקרון הסימן: אֵינוֹ עוֹלֶה לְלִמּוּד אֶלָּא לְמִי שֶׁמֵּבִין (סעיף ב).

קריאת הרב: מִי שֶׁאֵינוֹ מֵבִין צָרִיךְ לִלְמֹד וּלְהָבִין ; אוֹמֵר וְאֵינוֹ מֵבִין — אֵינוֹ נִקְרָא לִמּוּד כְּלָל.

תוצאה מעשית: לִלְמֹד אֶת פֵּרוּשׁ אֵיזֶהוּ מְקוֹמָן וְהַי״ג מִדּוֹת, וְלֹא לוֹמַר בְּעָלְמָא.

קו ג — הָאָבֵל וְסֵדֶר הַיּוֹם

האבל וסדר היום

עיקרון הסימן: דִּין בֵּית הָאָבֵל (סעיף ג).

קריאת הרב: אוֹמְרִים אֵיזֶהוּ מְקוֹמָן בְּבֵית הָאָבֵל, שֶׁכָּל שֶׁהוּא סֵדֶר הַיּוֹם אֵין בּוֹ מִשּׁוּם תַּלְמוּד תּוֹרָה לָאָבֵל ; עַיֵּן סִימָן תקנ״ד וּבְיוֹרֶה דֵּעָה סִימָן שפ״ד.

תוצאה מעשית: לוֹמַר אֶת סֵדֶר הַיּוֹם אַף בְּאֵבֶל. עַיֵּן סימן מ״ח · 48 וְסימן נ״א · 51.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן נ׳ ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, בהפניה לרב לפרטי המקרים (אמירת המשנה והברייתא בכל יום, הבנת הנאמר, אמירה בְּבֵית הָאָבֵל).

לחסיד חב״ד — סימן נ׳ (טַעַם אֲמִירַת מִשְׁנַת אֵיזֶהוּ מְקוֹמָן)

  • ① בכל יום, המשנה והברייתא. לוֹמַר בְּכָל בּוֹקֶר, אַחַר פָּרָשַׁת הַתָּמִיד, אֶת פֶּרֶק אֵיזֶהוּ מְקוֹמָן וּבְרַיְתָא דְּרַבִּי יִשְׁמָעֵאל — כְּדֵי לִזְכּוֹת בְּמִקְרָא מִשְׁנָה וְתַלְמוּד בְּכָל יוֹם. יסוד: שוע״ר נ׳ סעיף א.
  • ② להבין את הנאמר. הַלִּמּוּד אֵינוֹ עוֹלֶה אֶלָּא לְמִי שֶׁמֵּבִין ; מִי שֶׁאֵינוֹ מֵבִין צָרִיךְ לִלְמֹד וּלְהָבִין — שֶׁאִם לֹא כֵן אֵינוֹ נִקְרָא לִמּוּד כְּלָל. יסוד: שוע״ר נ׳ סעיף ב.
  • ③ בְּבֵית הָאָבֵל. אוֹמְרִים אֵיזֶהוּ מְקוֹמָן אַף בְּבֵית הָאָבֵל — שֶׁהוּא סֵדֶר הַיּוֹם, וְאֵין בּוֹ מִשּׁוּם תַּלְמוּד תּוֹרָה לָאָבֵל. יסוד: שוע״ר נ׳ סעיף ג.
  • ④ טעם הבחירה. אֵיזֶהוּ מְקוֹמָן נִבְחַר כְּפֶרֶק שֶׁאֵין בּוֹ מַחֲלֹקֶת (מִשְׁנָה בְּרוּרָה לְמֹשֶׁה מִסִּינַי) ; וְהַבְּרַיְתָא בִּמְקוֹם תַּלְמוּד, שֶׁהַמִּדְרָשׁ כְּתַלְמוּד. יסוד: שוע״ר נ׳ סעיף א.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן נ׳ (הטקסט האמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַלִּמּוּד בַּהֲבָנָה, אֲמִירָה בְּבֵית הָאָבֵל — יש להתייעץ עם רב, ולחב״ד עם רב של חב״ד.
