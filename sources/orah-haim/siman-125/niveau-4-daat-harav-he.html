דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קכ״ה · דִּינֵי קְדֻשָּׁה

דרכו של אדמו״ר הזקן בדִּינֵי קְדֻשָּׁה : עֶצֶם מִצְוַת הַקְּדֻשָּׁה — הַשְּׁלִיחַ צִבּוּר אוֹמֵר ״נַקְדִּישָׁךְ״, הַצִּבּוּר שׁוֹתְקִין וְעוֹנִין ״קָדוֹשׁ״ ; אִסּוּר דִּבּוּר בְּאֶמְצַע קְדֻשָּׁה ; מִנְהֲגֵי נְשִׂיאַת עֵינַיִם וְגוּף ; וְהַשּׁוֹמֵעַ קְדֻשָּׁה — חוֹזֵר וְעוֹנֶה עִמָּהֶם — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קכ״ה — דִּינֵי קְדֻשָּׁה. אצל הרב הסימן נפרש לד׳ סְעִיפִים : עֶצֶם מִצְוַת הַקְּדֻשָּׁה עִם דִּין הַיָּחִיד וְהָעֲשָׂרָה, אִסּוּר דִּבּוּר בְּאֶמְצַע קְדֻשָּׁה, מִנְהֲגֵי נְשִׂיאַת עֵינַיִם וְנִעְנוּעַ הַגּוּף, וְדִין הַשּׁוֹמֵעַ קְדֻשָּׁה וְנִכְנַס לְבֵית הַכְּנֶסֶת.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קכ״ה — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קכ״ה). אדמו״ר הזקן דן בו בְּדִינֵי קְדֻשָּׁה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קכ״ה — דִּינֵי קְדֻשָּׁה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קכ״ה — דִּינֵי קְדֻשָּׁה — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.125), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.125.k).

סעיף א הַשְּׁלִיחַ צִבּוּר אוֹמֵר ״נַקְדִּישָׁךְ״ וְהַצִּבּוּר שׁוֹתְקִין וְעוֹנִין ״קָדוֹשׁ״ ; דִּין הַיָּחִיד וְהָעֲשָׂרָה.

דִּינֵי קְדֻשָּׁה וּבוֹ ד׳ סְעִיפִים:כְּשֵׁם שֶׁמִּצְוַת קַדִּישׁ הִיא שֶׁהַשְּׁלִיחַ צִבּוּר אוֹמֵר ״יִתְגַּדַּל וְיִתְקַדַּשׁ כו׳״ בִּשְׁבִיל כָּל הַצִּבּוּר וְהֵם עוֹנִים אַחֲרָיו ״אָמֵן יְהֵא״ וּמְכַוְּנִים — כָּךְ הִיא מִצְוַת הַקְּדֻשָּׁה לְכַתְּחִלָּה שֶׁיֹּאמַר הַשְּׁלִיחַ צִבּוּר ״נַקְדִּישָׁךְ״ אוֹ ״נְקַדֵּשׁ״ וְהַצִּבּוּר שׁוֹתְקִין וּמְכַוְּנִים לְמַה שֶּׁשְּׁלִיחַ צִבּוּר אוֹמֵר עַד שֶׁמַּגִּיעַ לִקְדֻשָּׁה וְאָז עוֹנִין הַצִּבּוּר ״קָדוֹשׁ״, וְכֵן בִּ״לְעֻמָּתָם״ [וּבְ]״וּבְדִבְרֵי קָדְשְׁךָ״. וַאֲפִלּוּ אִם יֵשׁ ט׳ וְיוֹתֵר הַשּׁוֹמְעִים וּמְכַוְּנִים לַשְּׁלִיחַ צִבּוּר — (אֵין לַיָּחִיד לוֹמַר עִם הַשְּׁלִיחַ צִבּוּר ״נַקְדִּישָׁךְ״ אוֹ ״נְקַדֵּשׁ״, שֶׁכֵּיוָן שֶׁקָּבְעוּ נֻסָּח זֶה בִּקְדֻשָּׁה — יֵשׁ לִנְהֹג בּוֹ דִּין קְדֻשָּׁה כְּמוֹ בְּ״קָדוֹשׁ״ וּ״בָרוּךְ״, וְאֵין אוֹמְרִים דָּבָר שֶׁבִּקְדֻשָּׁה בְּפָחוֹת מֵעֲשָׂרָה, וְלִשְׁלִיחַ צִבּוּר יֵשׁ ט׳ הַשּׁוֹמְעִים וּמְכַוְּנִים לוֹ, וְלוֹ מִי שׁוֹמֵעַ?) וְאִם רָצָה לוֹמַר מִלָּה בְּמִלָּה עִם הַשְּׁלִיחַ צִבּוּר בְּלַחַשׁ — אֵין אִסּוּר בַּדָּבָר, שֶׁכֵּיוָן שֶׁהֵם אוֹמְרִים כָּל מִלָּה וּמִלָּה יַחְדָּו — אֲמִירַת שְׁנֵיהֶם נֶחְשֶׁבֶת כְּאַחַת, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קפ״ג. אֶלָּא שֶׁאֵין לַעֲשׂוֹת כֵּן לְכַתְּחִלָּה אֶלָּא מִדֹּחַק, וּכְמוֹ שֶׁיִּתְבָּאֵר שָׁם, וּכְמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ק״ט. וְכָל זֶה בְּיָחִיד, אֲבָל אִם כָּל הַצִּבּוּר נָהֲגוּ לוֹמַר ״נַקְדִּישָׁךְ״ עִם הַשְּׁלִיחַ צִבּוּר, אַף עַל פִּי שֶׁאֵין אוֹמְרִים בְּלַחַשׁ מִלָּה בְּמִלָּה עִם הַשְּׁלִיחַ צִבּוּר — אֵין לִמְחוֹת, כֵּיוָן שֶׁמַּקְדִּישִׁין בַּעֲשָׂרָה. וְאַף עַל פִּי שֶׁאֵינוֹ נִקְרָא צִבּוּר אֶלָּא כְּשֶׁאֶחָד אוֹמֵר וְט׳ שׁוֹמְעִין וְעוֹנִין, אֲבָל כְּשֶׁכָּל אֶחָד אוֹמֵר לְעַצְמוֹ כִּיחִידִים הֵם כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תקצ״ד, מִכָּל מָקוֹם בְּדָבָר שֶׁבִּקְדֻשָּׁה — אֵין צָרִיךְ לְכָךְ, שֶׁלֹּא אָמְרוּ אֶלָּא: אֵין אוֹמְרִים דָּבָר שֶׁבִּקְדֻשָּׁה בְּפָחוֹת מֵעֲשָׂרָה, אֲבָל עֲשָׂרָה רַשָּׁאִים כֻּלָּם לוֹמַר ״נְקַדֵּשׁ״, כְּמוֹ שֶׁאוֹמְרִים ״קָדוֹשׁ״ וּ״בָרוּךְ״ שֶׁהוּא עִקַּר הַקְּדֻשָּׁה, וְאַף עַל פִּי כֵן אוֹמְרִים כָּל הַצִּבּוּר אֲפִלּוּ שֶׁלֹּא בְּדֶרֶךְ עֲנִיָּה אַחַר הַשְּׁלִיחַ צִבּוּר, כְּגוֹן קְדֻשַּׁת ״וּבָא לְצִיּוֹן״ לְדִבְרֵי הָאוֹמְרִים שֶׁאֵינָהּ נֶאֱמֶרֶת אֶלָּא בַּעֲשָׂרָה. אֶלָּא שֶׁבִּקְדֻשַּׁת י״ח יֵשׁ לִנְהֹג לְכַתְּחִלָּה כְּמוֹ בְּקַדִּישׁ, שֶׁ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״ הוּא עֲנִיָּה שֶׁעוֹנִין הַצִּבּוּר אַחַר ״יִתְגַּדַּל״ שֶׁאָמַר הַשְּׁלִיחַ צִבּוּר, כְּמוֹ עֲנִיַּת אָמֵן אַחַר הַבְּרָכוֹת, וְכֵן בִּקְדֻשָּׁה יֵשׁ לַעֲנוֹת ״קָדוֹשׁ״ וּ״בָרוּךְ״ אַחַר ״נַקְדִּישָׁךְ״ אוֹ ״נְקַדֵּשׁ״ שֶׁאָמַר הַשְּׁלִיחַ צִבּוּר, אֶלָּא אִם כֵּן מִדֹּחַק שֶׁמֻּכְרָח לוֹמַר עִם הַשְּׁלִיחַ צִבּוּר כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ק״ט. וְיֵשׁ נוֹהֲגִין כֵּן מִטַּעַם הַיָּדוּעַ לָהֶם אֲפִלּוּ שֶׁלֹּא מִדֹּחַק, לוֹמַר כָּל נֻסַּח הַקְּדֻשָּׁה עִם הַשְּׁלִיחַ צִבּוּר מִלָּה בְּמִלָּה בְּלַחַשׁ וְתֵבַת ״נַקְדִּישָׁךְ וְנַעֲרִיצָךְ״ וְ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״ בְּקוֹל רָם:

עֶצֶם מִצְוַת הַקְּדֻשָּׁה. כְּשֵׁם שֶׁמִּצְוַת קַדִּישׁ הִיא שֶׁהַשְּׁלִיחַ צִבּוּר אוֹמֵר ״יִתְגַּדַּל״ בִּשְׁבִיל הַצִּבּוּר וְהֵם עוֹנִים ״אָמֵן יְהֵא״ — כָּךְ מִצְוַת הַקְּדֻשָּׁה לְכַתְּחִלָּה שֶׁיֹּאמַר הַשְּׁלִיחַ צִבּוּר ״נַקְדִּישָׁךְ״ אוֹ ״נְקַדֵּשׁ״ וְהַצִּבּוּר שׁוֹתְקִין וּמְכַוְּנִים, וְאָז עוֹנִין ״קָדוֹשׁ״. תמצית : יָחִיד אֵין לוֹ לוֹמַר ״נַקְדִּישָׁךְ״ עִם הַשְּׁלִיחַ צִבּוּר (דָּבָר שֶׁבִּקְדֻשָּׁה בַּעֲשָׂרָה), אֲבָל מִלָּה בְּמִלָּה בְּלַחַשׁ מֻתָּר, שֶׁאֲמִירַת שְׁנֵיהֶם נֶחְשֶׁבֶת כְּאַחַת (עיין סימן קפ״ג, סימן ק״ט) — לְכַתְּחִלָּה רַק מִדֹּחַק. וְאִם כָּל הַצִּבּוּר נָהֲגוּ לוֹמַר ״נַקְדִּישָׁךְ״ — אֵין לִמְחוֹת, שֶׁעֲשָׂרָה רַשָּׁאִים כֻּלָּם ״נְקַדֵּשׁ״ כְּמוֹ ״קָדוֹשׁ״ וּ״בָרוּךְ״ שֶׁהוּא עִקַּר הַקְּדֻשָּׁה.

סעיף ב אָסוּר לְדַבֵּר בְּאֶמְצַע קְדֻשָּׁה ; אַף לִלְמֹד ; אַף בְּיֵשׁ מִנְיָן בִּלְעָדוֹ.

אָסוּר לְדַבֵּר בְּאֶמְצַע קְדֻשָּׁה. וְכָל הַמְדַבֵּר — עָלָיו הַכָּתוּב אוֹמֵר: ״וְלֹא אֹתִי קָרָאתָ יַעֲקֹב וְגו׳״, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן נ״ו. וַאֲפִלּוּ יֵשׁ מִנְיָן בִּלְעָדוֹ — אָסוּר. וַאֲפִלּוּ לִלְמֹד, אִם לֹא בְּשָׁעָה שֶׁהַשְּׁלִיחַ צִבּוּר מְנַגֵּן בִּשְׁעַת הַנִּגּוּן, וְלֹא בְּשָׁעָה שֶׁמְּחַתֵּךְ הָאוֹתִיּוֹת. וְהַנֻּסָּח שֶׁמּוֹסִיפִין בְּשַׁבָּת — אֵינוֹ מִכְּלַל קְדֻשָּׁה. וּמִכָּל מָקוֹם, אָסוּר לְסַפֵּר בֵּין ״קָדוֹשׁ״ לְ״בָרוּךְ״, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן נ״ו. וְטוֹב שֶׁלֹּא לְסַפֵּר עַד אַחַר אָמֵן שֶׁל ״הָאֵל הַקָּדוֹשׁ״. וְיִרְאֶה הַשְּׁלִיחַ צִבּוּר לְסַיֵּם ״לְעֻמָּתָם כו׳ וּבְדִבְרֵי כו׳״ קֹדֶם שֶׁיַּתְחִילוּ הַקָּהָל ״בָּרוּךְ״ וְ״יִמְלֹךְ״:

אִסּוּר דִּבּוּר בְּאֶמְצַע קְדֻשָּׁה. אָסוּר לְדַבֵּר בְּאֶמְצַע קְדֻשָּׁה, וְעַל הַמְדַבֵּר הַכָּתוּב אוֹמֵר ״וְלֹא אֹתִי קָרָאתָ יַעֲקֹב״ (עיין סימן נ״ו). תמצית : אֲפִלּוּ יֵשׁ מִנְיָן בִּלְעָדוֹ — אָסוּר ; אֲפִלּוּ לִלְמֹד, חוּץ מִשְּׁעַת הַנִּגּוּן וְלֹא בְּשָׁעָה שֶׁמְּחַתֵּךְ הָאוֹתִיּוֹת. הַנֻּסָּח שֶׁמּוֹסִיפִין בְּשַׁבָּת אֵינוֹ מִכְּלַל קְדֻשָּׁה ; אָסוּר לְסַפֵּר בֵּין ״קָדוֹשׁ״ לְ״בָרוּךְ״, וְטוֹב עַד אַחַר ״הָאֵל הַקָּדוֹשׁ״. וְיִרְאֶה הַשְּׁלִיחַ צִבּוּר לְסַיֵּם ״לְעֻמָּתָם כו׳ וּבְדִבְרֵי כו׳״ קֹדֶם שֶׁיַּתְחִילוּ הַקָּהָל ״בָּרוּךְ״ וְ״יִמְלֹךְ״.

סעיף ג נְשִׂיאַת עֵינַיִם לַמָּרוֹם וְנִעְנוּעַ הַגּוּף בְּ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״.

וְנוֹהֲגִין לִשָּׂא הָעֵינַיִם לַמָּרוֹם בְּשָׁעָה שֶׁאוֹמְרִים ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״. וְטוֹב שֶׁתִּהְיֶינָה סְגוּרוֹת. וְכֵן נוֹהֲגִין שֶׁמְּנַעַנְעִין גּוּפָן וְנוֹשְׂאִין אוֹתוֹ מִן הָאָרֶץ, וְסֶמֶךְ לַמִּנְהָג מִסֵּפֶר הֵיכָלוֹת: בְּרוּכִים אַתֶּם לַה׳ שָׁמַיִם וְיוֹרְדֵי מֶרְכָּבָה אִם תֹּאמְרוּ וְתַגִּידוּ לְבָנַי מַה שֶּׁאֲנִי עוֹשֶׂה בְּשָׁעָה שֶׁמַּקְדִּישִׁין וְאוֹמְרִים ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״, וְלַמְּדוּ אוֹתָן שֶׁיִּהְיוּ עֵינֵיהֶם נְשׂוּאוֹת לַמָּרוֹם לְבֵית תְּפִלָּתָם וְנוֹשְׂאִים עַצְמָם לְמַעְלָה, כִּי אֵין לִי הֲנָאָה בָּעוֹלָם כְּאוֹתָהּ שָׁעָה שֶׁעֵינֵיהֶם נְשׂוּאוֹת בְּעֵינַי וְעֵינַי בְּעֵינֵיהֶם, וּבְאוֹתָהּ שָׁעָה אֲנִי אוֹחֵז בְּכִסֵּא כְּבוֹדִי בִּדְמוּת יַעֲקֹב וּמְחַבְּקָהּ וּמְנַשְּׁקָהּ וּמַזְכִּיר גָּלוּתָם וּמְמַהֵר גְּאֻלָּתָם. וְאָמְרוּ בַּמִּדְרָשׁ: ״וּבִשְׁתַּיִם יְעוֹפֵף״, מִכָּאן תִּקְּנוּ לָעוּף אָדָם עַל רַגְלָיו בְּשָׁעָה שֶׁאוֹמְרִים ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״, וְלֹא כְּאוֹתָם שֶׁדּוֹלְגִים וְקוֹפְצִים. וְיֵשׁ מִי שֶׁקַּבָּלָה בְּיָדוֹ שֶׁיָּרִים גּוּפוֹ וַעֲקֵבָיו לְמַעְלָה בְּ״בָרוּךְ״ וְ״יִמְלֹךְ״ כְּמוֹ בְּ״קָדוֹשׁ״. וְיֵשׁ מִי שֶׁהָיָה נוֹהֵג מִטַּעַם הַיָּדוּעַ לוֹ לִכְרֹעַ בְּ״בָרוּךְ״ וְ״יִמְלֹךְ״ וְלִזְקֹף בַּשֵּׁם. וְאֵין בָּזֶה מִשּׁוּם אִסּוּר הִשְׁתַּחֲוָיָה בְּמָקוֹם שֶׁלֹּא תִּקְּנוּ חֲכָמִים, כְּמוֹ שֶׁאֵין אִסּוּר בְּכוֹרֵעַ בְּאֶמְצַע בִּרְכוֹת י״ח, מִטַּעַם שֶׁנִּתְבָּאֵר בְּסִימָן קי״ג. וְיִזָּהֵר לְכַוֵּן רַגְלָיו בְּשָׁעָה שֶׁאוֹמֵר קְדֻשָּׁה עִם שְׁלִיחַ צִבּוּר, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן צ״ה:

מִנְהֲגֵי נְשִׂיאַת עֵינַיִם וְגוּף. נוֹהֲגִין לִשָּׂא הָעֵינַיִם לַמָּרוֹם בְּ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״ (וְטוֹב שֶׁתִּהְיֶינָה סְגוּרוֹת), וּלְנַעֲנֵעַ הַגּוּף וּלְנָשְׂאוֹ מִן הָאָרֶץ. תמצית : סֶמֶךְ מִסֵּפֶר הֵיכָלוֹת (״… בְּעֵינַי וְעֵינַי בְּעֵינֵיהֶם …״), וְהַמִּדְרָשׁ ״וּבִשְׁתַּיִם יְעוֹפֵף״ — לָעוּף עַל רַגְלָיו, לֹא כְּדוֹלְגִים וְקוֹפְצִים. יֵשׁ שֶׁקַּבָּלָה בְּיָדָם לְהָרִים גּוּף וַעֲקֵבַיִם בְּ״בָרוּךְ״ וְ״יִמְלֹךְ״ כְּמוֹ בְּ״קָדוֹשׁ״ ; אֵין בָּזֶה אִסּוּר הִשְׁתַּחֲוָיָה (עיין סימן קי״ג). וְיִזָּהֵר לְכַוֵּן רַגְלָיו בַּקְּדֻשָּׁה עִם הַשְּׁלִיחַ צִבּוּר (עיין סימן צ״ה).

סעיף ד הַשּׁוֹמֵעַ קְדֻשָּׁה וְנִכְנַס וּמָצָא צִבּוּר אוֹמְרִים — חוֹזֵר וְעוֹנֶה עִמָּהֶם.

הַשּׁוֹמֵעַ קְדֻשָּׁה בְּמָקוֹם אֶחָד וְנִכְנַס לְבֵית הַכְּנֶסֶת וּמָצָא צִבּוּר אוֹמְרִים קְדֻשָּׁה — חוֹזֵר וְעוֹנֶה עִמָּהֶם:

הַשּׁוֹמֵעַ קְדֻשָּׁה וְנִכְנַס. הַשּׁוֹמֵעַ קְדֻשָּׁה בְּמָקוֹם אֶחָד, וְנִכְנַס לְבֵית הַכְּנֶסֶת וּמָצָא צִבּוּר אוֹמְרִים קְדֻשָּׁה — חוֹזֵר וְעוֹנֶה עִמָּהֶם.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קכ״ה (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּינֵי קְדֻשָּׁה

היסוד : דִּינֵי קְדֻשָּׁה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה עֶצֶם מִצְוַת הַקְּדֻשָּׁה וְדִין הַיָּחִיד וְהָעֲשָׂרָה. אחר כך אִסּוּר דִּבּוּר בְּאֶמְצַע קְדֻשָּׁה. אחר כך מִנְהֲגֵי נְשִׂיאַת עֵינַיִם וְנִעְנוּעַ הַגּוּף. ולבסוף דִּין הַשּׁוֹמֵעַ קְדֻשָּׁה וְנִכְנַס.

עֶצֶם מִצְוַת הַקְּדֻשָּׁה — שׁוֹתְקִין וְעוֹנִין ״קָדוֹשׁ״

עֶצֶם מִצְוַת הַקְּדֻשָּׁה — שׁוֹתְקִין וְעוֹנִין ״קָדוֹשׁ״

היסוד. כְּמוֹ הַקַּדִּישׁ : הַשְּׁלִיחַ צִבּוּר אוֹמֵר ״נַקְדִּישָׁךְ״ אוֹ ״נְקַדֵּשׁ״, הַצִּבּוּר שׁוֹתְקִין וּמְכַוְּנִים וְאָז עוֹנִין ״קָדוֹשׁ״ (סעיף א).

המסקנה. הַקְּדֻשָּׁה בְּעִקָּרָהּ עֲנִיָּה ; מִנְהַג חַב״ד לְאָמְרָהּ מִלָּה בְּמִלָּה בְּלַחַשׁ עִם הַשְּׁלִיחַ צִבּוּר. עיין סימן 55 (דָּבָר שֶׁבִּקְדֻשָּׁה).

יָחִיד וּבַעֲשָׂרָה — דָּבָר שֶׁבִּקְדֻשָּׁה

יָחִיד וּבַעֲשָׂרָה — דָּבָר שֶׁבִּקְדֻשָּׁה

היסוד. אֵין אוֹמְרִים דָּבָר שֶׁבִּקְדֻשָּׁה בְּפָחוֹת מֵעֲשָׂרָה ; אֲבָל בְּלַחַשׁ מִלָּה בְּמִלָּה עִם הַשְּׁלִיחַ צִבּוּר אֵין אִסּוּר — אֲמִירַת שְׁנֵיהֶם נֶחְשֶׁבֶת כְּאַחַת (סעיף א).

המסקנה. עֲשָׂרָה רַשָּׁאִים כֻּלָּם לוֹמַר ״נְקַדֵּשׁ״ כְּמוֹ ״קָדוֹשׁ״ וּ״בָרוּךְ״. עיין סימן 109 (אֲמִירָה עִם הַשְּׁלִיחַ צִבּוּר).

אִסּוּר דִּבּוּר בְּאֶמְצַע קְדֻשָּׁה

אִסּוּר דִּבּוּר בְּאֶמְצַע קְדֻשָּׁה

היסוד. אָסוּר לְדַבֵּר בְּאֶמְצַע קְדֻשָּׁה — ״וְלֹא אֹתִי קָרָאתָ יַעֲקֹב״ ; אֲפִלּוּ יֵשׁ מִנְיָן בִּלְעָדוֹ, אֲפִלּוּ לִלְמֹד (סעיף ב).

המסקנה. אָסוּר לְסַפֵּר בֵּין ״קָדוֹשׁ״ לְ״בָרוּךְ״, וְטוֹב עַד אַחַר ״הָאֵל הַקָּדוֹשׁ״. עיין סימן 56 (אִסּוּר הֶפְסֵק).

מִנְהֲגֵי הַגּוּף וְהַשּׁוֹמֵעַ קְדֻשָּׁה

מִנְהֲגֵי הַגּוּף וְהַשּׁוֹמֵעַ קְדֻשָּׁה

היסוד. נְשִׂיאַת עֵינַיִם לַמָּרוֹם וְנִעְנוּעַ הַגּוּף בְּ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״ (סֵפֶר הֵיכָלוֹת, ״וּבִשְׁתַּיִם יְעוֹפֵף״) ; לְכַוֵּן רַגְלָיו (סעיף ג).

המסקנה. הַשּׁוֹמֵעַ קְדֻשָּׁה בְּמָקוֹם אֶחָד וְנִכְנַס — חוֹזֵר וְעוֹנֶה עִמָּהֶם (סעיף ד). עיין סימן 113 (כְּרִיעָה וּזְקִיפָה).

הפניה ללימוד. היקף הסימן — עֲנִיַּת ״קָדוֹשׁ״, אִסּוּר דִּבּוּר וּמִנְהֲגֵי הַגּוּף — נוגע להנהגת התפילה בציבור היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קכ״ה. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
עֶצֶם הַקְּדֻשָּׁה וַאֲמִירַת יָחִיד או״ח קכ״ה — הַצִּבּוּר שׁוֹתְקִין וּמְכַוְּנִים לַשְּׁלִיחַ צִבּוּר עַד ״קָדוֹשׁ״. מ״ב קכ״ה — פְּרָטֵי הַשְּׁתִיקָה וְהָעֲנִיָּה ״קָדוֹשׁ״ ״בָרוּךְ״. ערוה״ש קכ״ה — גֶּדֶר דָּבָר שֶׁבִּקְדֻשָּׁה בַּעֲשָׂרָה. דרך אדמו״ר הזקן — שׁוֹתְקִין וְעוֹנִין ״קָדוֹשׁ״ ; יָחִיד לֹא עִם הַשְּׁלִיחַ צִבּוּר, אֲבָל מִלָּה בְּמִלָּה בְּלַחַשׁ מֻתָּר, וַעֲשָׂרָה רַשָּׁאִים כֻּלָּם ״נְקַדֵּשׁ״ (סעיף א).
אֲמִירָה עִם הַשְּׁלִיחַ צִבּוּר או״ח קכ״ה — עִקַּר הַדִּין שֶׁיִּשְׁתֹּק וִיכַוֵּן. מ״ב קכ״ה — הַנּוֹהֲגִין לוֹמַר עִם הַשְּׁלִיחַ צִבּוּר. ערוה״ש קכ״ה — טַעַם הַשְּׁתִיקָה וּסְמַךְ הַמִּנְהָג. דרך אדמו״ר הזקן — אֲמִירַת שְׁנֵיהֶם נֶחְשֶׁבֶת כְּאַחַת ; לְכַתְּחִלָּה מִדֹּחַק, וְיֵשׁ נוֹהֲגִין מִטַּעַם הַיָּדוּעַ לוֹמַר כָּל הַנֻּסָּח בְּלַחַשׁ (סעיף א).
אִסּוּר דִּבּוּר בַּקְּדֻשָּׁה או״ח קכ״ה — אֵין לְדַבֵּר בְּאֶמְצַע הַקְּדֻשָּׁה [רמ״א]. מ״ב קכ״ה — פְּרָטֵי הַהֶפְסֵק וְהַלִּמּוּד. ערוה״ש קכ״ה — גֶּדֶר הַשְּׁתִיקָה בַּקְּדֻשָּׁה. דרך אדמו״ר הזקן — אָסוּר לְדַבֵּר (״וְלֹא אֹתִי קָרָאתָ״), אֲפִלּוּ מִנְיָן בִּלְעָדוֹ, אֲפִלּוּ לִלְמֹד ; אָסוּר בֵּין ״קָדוֹשׁ״ לְ״בָרוּךְ״ (סעיף ב).
מִנְהֲגֵי הַגּוּף וְהַשּׁוֹמֵעַ או״ח קכ״ה — לְכַוֵּן רַגְלָיו ; נְשִׂיאַת עֵינַיִם וְנִעְנוּעַ הַגּוּף [רמ״א]. מ״ב קכ״ה — טַעַם נְשִׂיאַת הָעֵינַיִם וְהָרְמַת הַגּוּף. ערוה״ש קכ״ה — הַשּׁוֹמֵעַ קְדֻשָּׁה חוֹזֵר וְעוֹנֶה. דרך אדמו״ר הזקן — נְשִׂיאַת עֵינַיִם וְנִעְנוּעַ הַגּוּף (סֵפֶר הֵיכָלוֹת, ״וּבִשְׁתַּיִם יְעוֹפֵף״) ; הַשּׁוֹמֵעַ קְדֻשָּׁה וְנִכְנַס — חוֹזֵר וְעוֹנֶה עִמָּהֶם (סעיפים ג-ד).

הטבלה נערכה מטקסט המחבר והרמ״א (אורח חיים קכ״ה, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה ט, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; ספר היכלות, תשובת הרשב״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — עֶצֶם הַקְּדֻשָּׁה וּבַעֲשָׂרָה

קו א — עֶצֶם הַקְּדֻשָּׁה וּבַעֲשָׂרָה

יסוד הסימן : הַצִּבּוּר שׁוֹתְקִין וְעוֹנִין ״קָדוֹשׁ״ ; אֵין דָּבָר שֶׁבִּקְדֻשָּׁה בְּפָחוֹת מֵעֲשָׂרָה (סעיף א).

קריאת הרב : אֲמִירַת שְׁנֵיהֶם כְּאַחַת — מֻתָּר לוֹמַר בְּלַחַשׁ מִלָּה בְּמִלָּה עִם הַשְּׁלִיחַ צִבּוּר, וַעֲשָׂרָה רַשָּׁאִים כֻּלָּם ״נְקַדֵּשׁ״. עיין סימן 55.

מסקנה למעשה : מִנְהַג חַב״ד לוֹמַר כָּל הַקְּדֻשָּׁה בְּלַחַשׁ, וְ״נַקְדִּישָׁךְ״ וְ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״ בְּקוֹל רָם.

קו ב — אִסּוּר הֶפְסֵק בַּקְּדֻשָּׁה

קו ב — אִסּוּר הֶפְסֵק בַּקְּדֻשָּׁה

יסוד הסימן : אָסוּר לְדַבֵּר בְּאֶמְצַע קְדֻשָּׁה, אֲפִלּוּ יֵשׁ מִנְיָן בִּלְעָדוֹ, אֲפִלּוּ לִלְמֹד (סעיף ב).

קריאת הרב : אָסוּר לְסַפֵּר בֵּין ״קָדוֹשׁ״ לְ״בָרוּךְ״, וְטוֹב עַד אַחַר ״הָאֵל הַקָּדוֹשׁ״. עיין סימן 56.

מסקנה למעשה : לֹא לְדַבֵּר וְלֹא לִלְמֹד בַּקְּדֻשָּׁה, חוּץ מִשְּׁעַת הַנִּגּוּן שֶׁל הַשְּׁלִיחַ צִבּוּר.

קו ג — מִנְהֲגֵי הַגּוּף וְהַחֲזָרָה

קו ג — מִנְהֲגֵי הַגּוּף וְהַחֲזָרָה

יסוד הסימן : נְשִׂיאַת עֵינַיִם לַמָּרוֹם וְנִעְנוּעַ הַגּוּף בְּ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״ ; לְכַוֵּן רַגְלָיו (סעיף ג).

קריאת הרב : סֶמֶךְ מִסֵּפֶר הֵיכָלוֹת וּמֵהַמִּדְרָשׁ ״וּבִשְׁתַּיִם יְעוֹפֵף״ — לָעוּף עַל רַגְלָיו, לֹא בְּדִלּוּג. עיין סימן 113.

מסקנה למעשה : הַשּׁוֹמֵעַ קְדֻשָּׁה וְנִכְנַס — חוֹזֵר וְעוֹנֶה עִמָּהֶם (סעיף ד).

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קכ״ה ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קכ״ה (דִּינֵי קְדֻשָּׁה)

  • ① עֶצֶם הַמִּצְוָה. הַשְּׁלִיחַ צִבּוּר אוֹמֵר ״נַקְדִּישָׁךְ״ / ״נְקַדֵּשׁ״, הַצִּבּוּר שׁוֹתְקִין וּמְכַוְּנִים וְאָז עוֹנִין ״קָדוֹשׁ״. יסוד : שוע״ר קכ״ה סעיף א.
  • ② יָחִיד וּבַעֲשָׂרָה. יָחִיד אֵין לוֹ לוֹמַר ״נַקְדִּישָׁךְ״ עִם הַשְּׁלִיחַ צִבּוּר ; אֲבָל בְּלַחַשׁ מִלָּה בְּמִלָּה — מֻתָּר, וַעֲשָׂרָה רַשָּׁאִים כֻּלָּם. יסוד : שוע״ר קכ״ה סעיף א.
  • ③ אִסּוּר דִּבּוּר. אָסוּר לְדַבֵּר בְּאֶמְצַע קְדֻשָּׁה — אַף בְּיֵשׁ מִנְיָן בִּלְעָדוֹ, אַף לִלְמֹד ; אָסוּר בֵּין ״קָדוֹשׁ״ לְ״בָרוּךְ״. יסוד : שוע״ר קכ״ה סעיף ב.
  • ④ מִנְהֲגֵי הַגּוּף. נְשִׂיאַת עֵינַיִם לַמָּרוֹם וְנִעְנוּעַ הַגּוּף בְּ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״ ; לְכַוֵּן רַגְלָיו. יסוד : שוע״ר קכ״ה סעיף ג.
  • ⑤ הַשּׁוֹמֵעַ קְדֻשָּׁה. שָׁמַע קְדֻשָּׁה וְנִכְנַס וּמָצָא צִבּוּר אוֹמְרִים — חוֹזֵר וְעוֹנֶה עִמָּהֶם. יסוד : שוע״ר קכ״ה סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קכ״ה (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — עֲנִיַּת ״קָדוֹשׁ״, אִסּוּר דִּבּוּר, מִנְהֲגֵי הַגּוּף — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
