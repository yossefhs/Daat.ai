דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״פ · דִּינֵי פֵּרוּרֵי הַפַּת

דרכו של אדמו״ר הזקן בדִּינֵי פֵּרוּרֵי הַפַּת : שִׁיּוּר פַּת עַל הַשֻּׁלְחָן ; הַמַּפָּה וְהַלֶּחֶם עַד בִּרְכַּת הַמָּזוֹן ; פַּת שְׁלֵמָה וַעֲבוֹדָה זָרָה ; כִּבּוּד מִפְּנֵי הַפֵּרוּרִים ; אִבּוּד הַפֵּרוּרִים ; וְכִּסּוּי הַסַּכִּין — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״פ — דִּינֵי פֵּרוּרֵי הַפַּת, הנפרש אצל הרב לו׳ סְעִיפִים : מִשִּׁיּוּר פַּת עַל הַשֻּׁלְחָן שֶׁהוּא סִימַן בְּרָכָה, לְאִי הֲסָרַת הַמַּפָּה וְהַלֶּחֶם עַד בִּרְכַּת הַמָּזוֹן, לְאִי הֲבָאַת פַּת שְׁלֵמָה עַל שֻׁלְחָן רֵיק (כְּעוֹרֵךְ שֻׁלְחָן לַעֲבוֹדָה זָרָה), לְכִבּוּד הַקַּרְקַע מִן הַפֵּרוּרִים קֹדֶם מַיִם אַחֲרוֹנִים, לְאִסּוּר זְרִיקַת הַפֵּרוּרִים בִּמְקוֹם דְּרִיסָה, וְעַד מִנְהַג כִּסּוּי הַסַּכִּין בִּשְׁעַת בִּרְכַּת הַמָּזוֹן.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״פ — ו׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״פ). אדמו״ר הזקן דן בו בְּדִינֵי פֵּרוּרֵי הַפַּת. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״פ — דִּינֵי פֵּרוּרֵי הַפַּת

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״פ — דִּינֵי פֵּרוּרֵי הַפַּת — וּבוֹ ו׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.180). הסימן כולל ו׳ סעיפים (סעיף k → ‎.180.k).

סעיף א מִי שֶׁאֵינוֹ מְשַׁיֵּר פַּת — אֵין רוֹאֶה סִימַן בְּרָכָה.

דִּין פֵּרוּרֵי הַפַּת, וּבוֹ ו׳ סְעִיפִים:כָּל מִי שֶׁאֵינוֹ מְשַׁיֵּר פַּת עַל שֻׁלְחָנוֹ — אֵין רוֹאֶה סִימַן בְּרָכָה לְעוֹלָם, שֶׁנֶּאֱמַר "אֵין שָׂרִיד לְאָכְלוֹ, עַל כֵּן לֹא יָחִיל טוּבוֹ", אֲבָל בִּבְרָכָה מַהוּ אוֹמֵר? "אָכֹל וְהוֹתֵר":

סעיף ב שֶׁלֹּא לְהָסִיר הַמַּפָּה וְהַלֶּחֶם עַד אַחַר בִּרְכַּת הַמָּזוֹן.

וְנָכוֹן שֶׁלֹּא לְהָסִיר הַמַּפָּה וְהַלֶּחֶם עַד אַחַר בִּרְכַּת הַמָּזוֹן, כְּדֵי שֶׁיְּהֵא נִכָּר שֶׁמְּבָרֵךְ ה׳ עַל חַסְדּוֹ וְטוּבוֹ הַגָּדוֹל, שֶׁהֵכִין מָזוֹן לְכָל בְּרִיּוֹתָיו:

סעיף ג לֹא יָבִיא פַּת שְׁלֵמָה — כְּעוֹרֵךְ שֻׁלְחָן לַעֲבוֹדָה זָרָה.

אִם לֹא נִשְׁאַר לוֹ לֶחֶם עַל הַשֻּׁלְחָן — לֹא יָבִיא פַּת שְׁלֵמָה וְיִתְּנֶנָּה עַל הַשֻּׁלְחָן, מִפְּנֵי שֶׁנִּרְאֶה כְּעוֹרֵךְ שֻׁלְחָן לַעֲבוֹדָה זָרָה, כְּמוֹ שֶׁאָמַר הַכָּתוּב: "הַעֹרְכִים לַגַּד שֻׁלְחָן וְגוֹ׳". אֲבָל אִם הָיְתָה שְׁלֵמָה מֻנַּחַת כְּבָר עַל הַשֻּׁלְחָן, כְּמוֹ שֶׁבְּשַׁבָּת בּוֹצְעִים עַל לֶחֶם מִשְׁנֶה — טוֹב שֶׁלֹּא לַהֲסִירָהּ עַד אַחַר בִּרְכַּת הַמָּזוֹן:

סעיף ד כִּבּוּד הַקַּרְקַע מִפְּנֵי הַפֵּרוּרִים קֹדֶם מַיִם אַחֲרוֹנִים.

בִּימֵי חַכְמֵי הַתַּלְמוּד שֶׁהָיוּ מְסַלְּקִים אֶת הַשֻּׁלְחָנוֹת הַקְּטַנּוֹת שֶׁלִּפְנֵי כָּל אֶחָד וְאֶחָד קֹדֶם מַיִם אַחֲרוֹנִים, וְהָיוּ נוֹטְלִים מַיִם אַחֲרוֹנִים בִּמְקוֹם הַשֻּׁלְחָנוֹת — הָיוּ צְרִיכִים לְכַבֵּד קַרְקַע הַבַּיִת בִּמְקוֹם הַשֻּׁלְחָנוֹת קֹדֶם הַנְּטִילָה, שֶׁלֹּא יִשָּׁאֲרוּ שָׁם פֵּרוּרֵי פַּת וְיִמָּאֲסוּ בְּמֵי הַנְּטִילָה. וְאַף עַל פִּי שֶׁמֻּתָּר לְאַבֵּד פֵּרוּרִים שֶׁאֵין בָּהֶם כְּזַיִת — חוֹשְׁשִׁים שֶׁמָּא יִהְיֶה הַשַּׁמָּשׁ עַם הָאָרֶץ, שֶׁמֻּתָּר לְהִשְׁתַּמֵּשׁ בְּשַׁמָּשׁ עַם הָאָרֶץ, וְיַנִּיחַ עַל הַשֻּׁלְחָן גַּם פֵּרוּרִים שֶׁיֵּשׁ בָּהֶם כְּזַיִת כְּשֶׁמֵּסִיר הַלֶּחֶם מֵעַל הַשֻּׁלְחָן, וְאַחַר כָּךְ יִפְּלוּ מִמֶּנּוּ כְּשֶׁמְּסַלְּקִין אוֹתָן מִשָּׁם וְיִמָּאֲסוּ בְּמֵי הַנְּטִילָה, לְכָךְ יְכַבֵּד תְּחִלָּה.
אֲבָל אָנוּ שֶׁאֵין אָנוּ מְסַלְּקִין הַשֻּׁלְחָן, וְאָנוּ נוֹטְלִים הַיָּדַיִם בְּמָקוֹם שֶׁאֵין שָׁם פֵּרוּרִים — אֵין צָרִיךְ לְכַבֵּד תְּחִלָּה:

סעיף ה לֹא יִזְרֹק פֵּרוּרִים בִּמְקוֹם דְּרִיסַת רַגְלַיִם.

אַף עַל פִּי שֶׁמֻּתָּר לְאַבֵּד פֵּרוּרִים שֶׁאֵין בָּהֶם כְּזַיִת בַּיָּדַיִם — לֹא יִזְרְקֵם בִּמְקוֹם דְּרִיסַת רַגְלֵי בְּנֵי אָדָם, מִפְּנֵי שֶׁקָּשֶׁה לַעֲנִיּוּת כְּשֶׁדּוֹרְסִים עַל פֵּרוּרֵי לֶחֶם, וְלֹא הִתִּירוּ אֶלָּא לְאַבְּדָם, כְּגוֹן לְהַשְׁלִיכָם לַמַּיִם וְכַיּוֹצֵא בָּזֶה:

סעיף ו נוֹהֲגִים לְכַסּוֹת הַסַּכִּין בִּשְׁעַת בִּרְכַּת הַמָּזוֹן.

נוֹהֲגִים לְכַסּוֹת הַסַּכִּין בִּשְׁעַת בִּרְכַּת הַמָּזוֹן, כִּי הַשֻּׁלְחָן דּוֹמֶה לְמִזְבֵּחַ, וּבַמִּזְבֵּחַ נֶאֱמַר: "לֹא תָנִיף עֲלֵיהֶם בַּרְזֶל", לְפִי שֶׁהַבַּרְזֶל מְקַצֵּר יָמָיו שֶׁל אָדָם וְהַמִּזְבֵּחַ מַאֲרִיךְ יְמֵי הָאָדָם, וְאֵינוֹ בְּדִין שֶׁיּוּנַף הַמְקַצֵּר עַל הַמַּאֲרִיךְ, וְגַם הַשֻּׁלְחָן מַאֲרִיךְ יָמָיו שֶׁל אָדָם וּמְכַפֵּר עֲוֹנוֹתָיו בְּהַכְנָסַת אוֹרְחִים, שֶׁגָּדוֹל כֹּחָהּ שֶׁל לְגִימָה שֶׁמַּשְׁרָה שְׁכִינָה.
וְנָהֲגוּ בִּמְקוֹמוֹת הַרְבֵּה שֶׁלֹּא לְכַסּוֹתוֹ בְּשַׁבָּת וְיוֹם טוֹב, כִּי בְּחֹל מְכַסִּין אוֹתוֹ מִפְּנֵי שֶׁהוּא כֹּחוֹ שֶׁל עֵשָׂו, וּבְשַׁבָּת וְיוֹם טוֹב אֵין שָׂטָן וּפֶגַע רַע. וּמִנְהָגָן שֶׁל יִשְׂרָאֵל תּוֹרָה הִיא:

2 · למעשה

הלכה למעשה — מנהג חב״ד

הנהגה מעשית לחב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״פ ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״פ (דִּינֵי פֵּרוּרֵי הַפַּת)

  • ① סעיף א. הַמְשַׁיֵּר פַּת עַל שֻׁלְחָנוֹ בְּסוֹף הַסְּעוּדָה — מִי שֶׁאֵינוֹ מְשַׁיֵּר אֵין רוֹאֶה סִימַן בְּרָכָה לְעוֹלָם; הַכָּתוּב מַנְגִּיד "אֵין שָׂרִיד לְאָכְלוֹ" לְ"אָכֹל וְהוֹתֵר".
  • ② סעיף ב. אֵין לְהָסִיר הַמַּפָּה וְהַלֶּחֶם עַד אַחַר בִּרְכַּת הַמָּזוֹן, כְּדֵי שֶׁיְּהֵא נִכָּר שֶׁמְּבָרֵךְ ה׳ הַמֵּכִין מָזוֹן לְכָל בְּרִיּוֹתָיו.
  • ③ סעיף ג. לֹא נִשְׁאַר לֶחֶם — לֹא יָבִיא פַּת שְׁלֵמָה לַשֻּׁלְחָן (כְּעוֹרֵךְ שֻׁלְחָן לְ"גַד", עֲבוֹדָה זָרָה); אֲבָל שְׁלֵמָה שֶׁכְּבָר מֻנַּחַת (לֶחֶם מִשְׁנֶה בְּשַׁבָּת) — טוֹב שֶׁלֹּא לַהֲסִירָהּ עַד בִּרְכַּת הַמָּזוֹן.
  • ④ סעיף ד. בִּימֵי הַתַּלְמוּד כִּבְּדוּ הַקַּרְקַע מִן הַפֵּרוּרִים קֹדֶם מַיִם אַחֲרוֹנִים (חֲשַׁשׁ שַׁמָּשׁ עַם הָאָרֶץ הַמַּנִּיחַ כְּזַיִת); אֲבָל אָנוּ שֶׁאֵין מְסַלְּקִין הַשֻּׁלְחָן וְנוֹטְלִים בְּמָקוֹם בְּלֹא פֵּרוּרִים — אֵין צָרִיךְ לְכַבֵּד תְּחִלָּה.
  • ⑤ סעיף ה. מֻתָּר לְאַבֵּד בַּיָּד פֵּרוּרִים שֶׁאֵין בָּהֶם כְּזַיִת, אַךְ לֹא לְזָרְקָם בִּמְקוֹם דְּרִיסַת רַגְלַיִם (קָשֶׁה לַעֲנִיּוּת); רַק לְאַבְּדָם, כְּגוֹן לְהַשְׁלִיכָם לַמַּיִם.
  • ⑥ סעיף ו. נוֹהֲגִים לְכַסּוֹת הַסַּכִּין בְּבִרְכַּת הַמָּזוֹן (הַשֻּׁלְחָן כְּמִזְבֵּחַ, "לֹא תָנִיף עֲלֵיהֶם בַּרְזֶל"; הַשֻּׁלְחָן מַאֲרִיךְ יָמִים וּמְכַפֵּר בְּהַכְנָסַת אוֹרְחִים); וּבִמְקוֹמוֹת הַרְבֵּה אֵין מְכַסִּים בְּשַׁבָּת וְיוֹם טוֹב (הַסַּכִּין כֹּחוֹ שֶׁל עֵשָׂו, שֶׁאֵינוֹ בְּשַׁבָּת) — וּמִנְהָגָן שֶׁל יִשְׂרָאֵל תּוֹרָה הִיא.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן ק״פ (הטקסט האמיתי מובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — שיור פת על השולחן, הסרת המפה והלחם, הפירורים וכיסוי הסכין — יש לשאול את רבך, ולחב״ד רב חב״די.
