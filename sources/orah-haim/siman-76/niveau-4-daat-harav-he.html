דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ע״ו · דִּין לְהִזָּהֵר מִצּוֹאָה בִּשְׁעַת קְרִיאַת שְׁמַע

דרכו של אדמו״ר הזקן בדִּין לְהִזָּהֵר מִצּוֹאָה בִּשְׁעַת קְרִיאַת שְׁמַע : צוֹאָה בַּעֲשָׁשִׁית וּבַגֻּמָּא — כִּסּוּי מַתִּיר ; צוֹאָה עוֹבֶרֶת וּפִי חֲזִיר — מַפְסִיק ; צוֹאָה בִּמְקוֹמָהּ — אָסוּר לְדִבְרֵי הַכֹּל ; הַסָּפֵק וְהַמֵּי רַגְלַיִם — חֲזָקוֹת וּמִדְּרַבָּנָן — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ע״ו — דִּין לְהִזָּהֵר מִצּוֹאָה בִּשְׁעַת קְרִיאַת שְׁמַע. אצל הרב הסימן נפרש לי״א סְעִיפִים : הַכִּסּוּי הַמַּתִּיר (עֲשָׁשִׁית, גֻּמָּא, בֶּגֶד), הַצּוֹאָה עוֹבֶרֶת וּפִי חֲזִיר, הַצּוֹאָה עַל בְּשָׂרוֹ וּבִמְקוֹמָהּ, הַבִּטּוּל בְּרֹק וְחֳלִי הַטְּחוֹרִים, וְהַסָּפֵק עִם הַחֲזָקוֹת וְקֻלַּת הַמֵּי רַגְלַיִם.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ע״ו — י״א סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ע״ו). אדמו״ר הזקן דן בו בְּדִין לְהִזָּהֵר מִצּוֹאָה בִּשְׁעַת קְרִיאַת שְׁמַע. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ע״ו — דִּין לְהִזָּהֵר מִצּוֹאָה בִּשְׁעַת קְרִיאַת שְׁמַע

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ע״ו — דִּין לְהִזָּהֵר מִצּוֹאָה בִּשְׁעַת קְרִיאַת שְׁמַע — וּבוֹ י״א סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.76), ואחריו תרגומנו. הסימן כולל י״א סעיפים (סעיף k → ‎.76.k).

סעיף א צוֹאָה בַּעֲשָׁשִׁית — מֻתָּר לִקְרוֹת כְּנֶגְדָּהּ, שֶׁהִיא מְכֻסָּה.

לְהִזָּהֵר מִצּוֹאָה בִּשְׁעַת קְרִיאַת שְׁמַע וּבוֹ י״א סְעִיפִים:צוֹאָה בַּעֲשָׁשִׁית — מֻתָּר לִקְרוֹת כְּנֶגְדָּהּ, אַף עַל פִּי שֶׁרוֹאֶה אוֹתָהּ דֶּרֶךְ דָּפְנוֹתֶיהָ, לְפִי שֶׁבְּצּוֹאָה תָּלָה הַכָּתוּב בְּכִסּוּי, שֶׁנֶּאֱמַר: ״וְשַׁבְתָּ וְכִסִּיתָ אֶת צֵאָתֶךָ״, וַהֲרֵי הִיא מְכֻסָּה:

צוֹאָה בַּעֲשָׁשִׁית. הָרוֹאֶה צוֹאָה דֶּרֶךְ דָּפְנוֹת הָעֲשָׁשִׁית — מֻתָּר לִקְרוֹת כְּנֶגְדָּהּ, שֶׁלֹּא תָּלָה הַכָּתוּב אֶלָּא בְּכִסּוּי (״וְכִסִּיתָ אֶת צֵאָתֶךָ״), וַהֲרֵי מְכֻסָּה.

סעיף ב צוֹאָה בַּגֻמָּא — מַנִּיחַ סַנְדָּל וְקוֹרֵא ; וְעַל בֶּגֶד מִבִּפְנִים מֻתָּר.

צוֹאָה בַּגֻמָּא — מַנִּיחַ סַנְדָּל עַל גַּבֵּי הַגֻּמָּא וְקוֹרֵא, שֶׁעַל יְדֵי כֵן חֲשׁוּבָה כִּמְכֻסָּה. וְהוּא שֶׁאֵין רֵיחַ רַע מַגִּיעַ אֵלָיו, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ע״ט. וְאֵין אוֹמְרִים שֶׁהַסַּנְדָּל בָּטֵל לְגַבֵּי רַגְלוֹ וְאֵינוֹ חָשׁוּב כִּסּוּי, כְּמוֹ שֶׁהָרֶגֶל עַצְמָהּ אֵינָהּ מוֹעֶלֶת לְכַסּוֹת צוֹאָה שֶׁתַּחְתֶּיהָ בַּגֻּמָּא, מִשּׁוּם דְּאֵין זֶה ״מַחֲנֶיךָ קָדוֹשׁ״ כֵּיוָן שֶׁאֵין הֶפְסֵק כִּסּוּי בֵּין גּוּף הָאָדָם לַצּוֹאָה שֶׁבַּגֻּמָּא, אֶלָּא הַסַּנְדָּל הוּא חָשׁוּב כִּסּוּי וְהֶפְסֵק בֵּינֵיהֶם. וּמִכָּל מָקוֹם, בָּטֵל הוּא לְגַבֵּי רַגְלוֹ לְעִנְיָן שֶׁצָּרִיךְ שֶׁלֹּא יִהְיֶה נוֹגֵעַ בַּצּוֹאָה שֶׁבַּגֻּמָא, שֶׁאִם הוּא נוֹגֵעַ בָּהּ — יֵשׁ לְהִסְתַּפֵּק וְלוֹמַר שֶׁהֲרֵי זֶה כְּאִלּוּ נוֹגֵעַ בְּרַגְלוֹ, מִפְּנֵי שֶׁהַסַּנְדָּל הוּא טָפֵל וּמְשַׁמֵּשׁ לְרַגְלוֹ בִּנְגִיעָה זוֹ, שֶׁהָרֶגֶל נוֹגֵעַ בַּצּוֹאָה עַל יְדֵי הַסַּנְדָּל, שֶׁהִיא מַעֲמִידָה אֶת הַסַּנְדָּל עַל הַצּוֹאָה, וַחֲשׁוּבָה כְּאִלּוּ הִיא עַצְמָהּ נוֹגַעַת בַּצּוֹאָה. וְאַף לְהַמַּתִּירִים בְּצוֹאָה עַל בְּשָׂרוֹ וּמְכֻסָּה בְּבִגְדוֹ — אֵין הַסַּנְדָּל נֶחְשָׁב כִּסּוּי לַצּוֹאָה שֶׁעָלָיו, וְאֵין זֶה ״מַחֲנֶיךָ קָדוֹשׁ״. וּסְפֵקָא שֶׁל תּוֹרָה לְחֻמְרָא. אֲבָל צוֹאָה שֶׁעַל בְּגָדָיו מִבִּפְנִים, כֵּיוָן שֶׁהוּא מְכֻסֶּה בִּבְגָדָיו — הֲרֵי זֶה ״מַחֲנֶיךָ קָדוֹשׁ״. וְאֵין אוֹמְרִים שֶׁהַבֶּגֶד בָּטֵל לְגַבֵּי גּוּפוֹ וּכְאִלּוּ גּוּפוֹ נוֹגֵעַ בַּצּוֹאָה עַל יְדֵי הַבֶּגֶד, כֵּיוָן שֶׁאֵין הַבֶּגֶד מְשַׁמֵּשׁ לַגּוּף בִּנְגִיעָה זוֹ שֶׁהַבֶּגֶד נוֹגֵעַ בַּצּוֹאָה, כֵּיוָן שֶׁאֵין רֵיחַ רַע מַגִּיעַ אֵלָיו — מֻתָּר לִקְרוֹת וּלְהִתְפַּלֵּל כְּשֶׁהוּא לָבוּשׁ בִּבְגָדִים אֵלֶּה. וּמִכָּל מָקוֹם מִדַּת חֲסִידוּת לְהַחֲלִיפָם בִּשְׁעַת הַתְּפִלָּה, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן (צ״א) [צ״ח]:

צוֹאָה בַּגֻמָּא. מַנִּיחַ סַנְדָּל עַל הַגֻּמָּא וְקוֹרֵא — חֲשׁוּבָה כִּמְכֻסָּה, וּבִלְבַד שֶׁאֵין רֵיחַ רַע וְאֵין הַסַּנְדָּל נוֹגֵעַ בָּהּ. וְאֵין הַסַּנְדָּל בָּטֵל לָרֶגֶל אֶלָּא כִּסּוּי מַפְסִיק. וְצוֹאָה עַל בֶּגֶד מִבִּפְנִים וּמְכֻסָּה — ״מַחֲנֶיךָ קָדוֹשׁ״, מֻתָּר ; וּמִדַּת חֲסִידוּת לְהַחֲלִיף בִּתְפִלָּה (סימן צ״ח).

סעיף ג צוֹאָה עוֹבֶרֶת וּפִי חֲזִיר — צָרִיךְ לְהַפְסִיק.

צוֹאָה אֲפִלּוּ הִיא עוֹבֶרֶת, כְּגוֹן שֶׁצָּפָה עַל פְּנֵי הַמַּיִם, אוֹ שֶׁאֶחָד נוֹשֵׂא גְּרָף שֶׁל רְעִי וּמַעֲבִירוֹ לִפְנֵי הַקּוֹרֵא אוֹ הַמִּתְפַּלֵל — צָרִיךְ לְהַפְסִיק עַד שֶׁתַּעֲבֹר מִכְּנֶגֶד פָּנָיו ד׳ אַמּוֹת לִצְדָדִין אוֹ לְפָנָיו כִּמְלֹא עֵינָיו, כְּמוֹ בְּצוֹאָה מֻנַּחַת כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ע״ט. לְפִיכָךְ פִּי חֲזִיר אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע כְּנֶגְדּוֹ אַף עַל פִּי שֶׁהוּא עוֹבֵר, מִפְּנֵי שֶׁהוּא כְּצוֹאָה עוֹבֶרֶת. וַאֲפִלּוּ הוּא עוֹלֶה מִן הַנָּהָר — אֵין הָרְחִיצָה מוֹעֶלֶת לוֹ, מִפְּנֵי שֶׁהוּא כִּגְרָף שֶׁל רְעִי שֶׁאָסוּר לִקְרוֹת כְּנֶגְדּוֹ אַף עַל פִּי שֶׁאֵין בּוֹ רְעִי, הוֹאִיל וְהוּא מְיֻחָד לְכָךְ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן פ״ז:

צוֹאָה עוֹבֶרֶת. צוֹאָה עוֹבֶרֶת (צָפָה עַל הַמַּיִם אוֹ גְּרָף שֶׁל רְעִי) — מַפְסִיק עַד שֶׁתַּעֲבֹר (ד׳ אַמּוֹת אוֹ כִּמְלֹא עֵינָיו, סימן ע״ט). וּפִי חֲזִיר כְּצוֹאָה עוֹבֶרֶת, וַאֲפִלּוּ עוֹלֶה מִן הַנָּהָר אֵין רְחִיצָה מוֹעֶלֶת, כִּגְרָף שֶׁל רְעִי (סימן פ״ז).

סעיף ד צוֹאָה עַל בְּשָׂרוֹ מְכֻסָּה — נָכוֹן כְּדִבְרֵי הָאוֹסְרִים ; וְשִׁכְבַת זֶרַע.

הָיְתָה צוֹאָה עַל בְּשָׂרוֹ וּמְכֻסָּה בִּבְגָדָיו אֲפִלּוּ הַבֶּגֶד נוֹגֵעַ בָּהּ, וְכֵן אִם הִכְנִיס יָדוֹ לְבֵית הַכִּסֵּא וְאֵינוֹ מֵרִיחַ רֵיחַ רַע, וּכְגוֹן שֶׁיֵּשׁ מְחִצָּה בֵּינוֹ לְבֵין מְחִצּוֹת בֵּית הַכִּסֵּא וְהִכְנִיס יָדוֹ לְבֵית הַכִּסֵּא דֶּרֶךְ חֹר שֶׁבַּמְּחִיצוֹת, דְאִם לֹא כֵּן אָסוּר מִשּׁוּם בֵּית הַכִּסֵּא עַצְמוֹ אַף עַל פִּי שֶׁיֵּשׁ לוֹ מְחִיצוֹת כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן פ״ג, אֲבָל עַכְשָׁו שֶׁאֵין לֶאֱסֹר אֶלָּא מִשּׁוּם הַיָּד בִּלְבַדָּהּ שֶׁהִיא בְּבֵית הַכִּסֵּא, אוֹ מִשּׁוּם אוֹתוֹ מָקוֹם בִּבְשָׂרוֹ שֶׁהַצּוֹאָה מֻנַּחַת עָלָיו תַּחַת בְּגָדָיו, שֶׁאוֹתוֹ מָקוֹם אֵין אֲנִי קוֹרֵא בּוֹ ״וְהָיָה מַחֲנֶיךָ קָדוֹשׁ וְגו׳״ — (ו)יֵשׁ אוֹמְרִים שֶׁאֵין לֶאֱסֹר בִּשְׁבִילָם לִקְרוֹת וּלְהִתְפַּלֵּל, שֶׁנֶּאֱמַר: ״כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ״, הַפֶּה וְהַחֹטֶם הֵם בִּכְלַל הַהִלּוּל וְלֹא שְׁאָר אֵיבָרִים. וְאַף עַל פִּי שֶׁאֵינָם נְקִיִּים — מֻתָּר, וְיֵשׁ אוֹסְרִים, מִשּׁוּם שֶׁנֶּאֱמַר: ״כָּל עַצְמֹתַי תֹּאמַרְנָה ה׳ מִי כָמוֹךָ״. וְיֵשׁ אוֹמְרִים שֶׁלֹּא הִתִּירוּ הַמַּתִּירִים בְּצוֹאָה שֶׁעַל בְּשָׂרוֹ אֶלָּא בְּמָקוֹם שֶׁהִיא נִכְסֵית מֵאֵלֶיהָ בְּלֹא מַלְבּוּשׁ, כְּגוֹן בֵּין אַצִּילֵי יָדָיו. וּלְעִנְיַן מַעֲשֶׂה נָכוֹן לַעֲשׂוֹת כְּדִבְרֵי הָאוֹסְרִים בְּכָל עִנְיָן. וַאֲפִלּוּ בְּשִׁכְבַת זֶרַע שֶׁעַל בְּשָׂרוֹ — נָכוֹן לְהַחֲמִיר בָּהּ כְּמוֹ בְּצוֹאָה אַף עַל פִּי שֶׁהִיא מְכֻסָּה בְּבִגְדוֹ, אַף עַל פִּי שֶׁמֵּעִקָּר הַדִּין אֵינָהּ כְּצוֹאָה כְּלָל. וּמִכָּל מָקוֹם אֵין צָרִיךְ לִרְחֹץ בְּמַיִם אֶלָּא אִם כֵּן הִיא יְבֵשָׁה כְּבָר, אֲבָל אִם עֲדַיִן הִיא לַחָה מְקַנְּחָהּ בְּמַפָּה וְדַיּוֹ. בַּמֶּה דְּבָרִים אֲמוּרִים כְּשֶׁהִיא עַל בְּשָׂרוֹ, אֲבָל עַל בִּגְדוֹ וּמְכֻסָּה שֶׁמֻּתָּר אֲפִלּוּ בְּצוֹאָה כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל — כָּל שֶׁכֵּן בְּשִׁכְבַת זֶרַע שֶׁאֵינוֹ כְּצוֹאָה כְּלָל מִן הַדִּין. וּמִכָּל מָקוֹם מִקְצָת חֲסִידִים הָרִאשׁוֹנִים הָיוּ מְשַׁמְּשִׁים מִטּוֹתֵיהֶם עֲרֻמִּים, שֶׁלֹּא תִּדְבַּק שִׁכְבַת זֶרַע בַּחֲלוּקֵיהֶן, אַף עַל פִּי שֶׁהוּא מְכֻסֶּה מִלְמַעְלָה. וְכֵן הָיוּ מַחֲמִירִים עַל עַצְמָן שֶׁלֹּא יִהְיֶה עַל בִּגְדֵיהֶם אֲפִלּוּ טִפֵּי מֵימֵי רַגְלַיִם, וְכָל שֶׁכֵּן טִפּוֹת צוֹאָה:

צוֹאָה עַל בְּשָׂרוֹ. מְכֻסָּה בְּבִגְדוֹ אוֹ הִכְנִיס יָדוֹ לְבֵית הַכִּסֵּא בְּלֹא רֵיחַ — יֵשׁ מַתִּירִים (״כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ״) וְיֵשׁ אוֹסְרִים (״כָּל עַצְמֹתַי תֹּאמַרְנָה״) ; לְמַעֲשֶׂה נָכוֹן כְּדִבְרֵי הָאוֹסְרִים בְּכָל עִנְיָן. וְאַף שִׁכְבַת זֶרַע נָכוֹן לְהַחֲמִיר, אַף שֶׁאֵינָהּ כְּצוֹאָה מֵעִיקַּר הַדִּין ; לַחָה מְקַנְּחָהּ בְּמַפָּה, יְבֵשָׁה רוֹחֵץ.

סעיף ה יָדַיִם מְטֻנָּפוֹת וְלִכְלוּךְ יָבֵשׁ — מֻתָּר לִקְרוֹת.

מִי שֶׁיָּדָיו מְטֻנָּפוֹת מִקִּנּוּחַ בֵּית הַכִּסֵּא, אִם אֵין בָּהֶם מַמָּשׁוּת צוֹאָה אֶלָּא לִכְלוּךְ בְּעָלְמָא וְהִיא יְבֵשָׁה וְהָלַךְ רֵיחָהּ — מֻתָּר לִקְרוֹת אַף עַל פִּי שֶׁהִיא מְגֻלָּה עַל יָדָיו:

יָדַיִם מְטֻנָּפוֹת. אֵין בָּהֶן מַמָּשׁוּת צוֹאָה אֶלָּא לִכְלוּךְ יָבֵשׁ שֶׁהָלַךְ רֵיחוֹ — מֻתָּר לִקְרוֹת אַף שֶׁמְּגֻלֶּה עַל יָדָיו.

סעיף ו צוֹאָה בִּמְקוֹמָהּ בְּפִי הַטַּבַּעַת — אָסוּר לְדִבְרֵי הַכֹּל.

צוֹאָה בִּמְקוֹמָהּ הַיְנוּ בְּנֶקֶב פִּי הַטַּבַּעַת אֲפִלּוּ הִיא מְכֻסָּה — אָסוּר לִקְרוֹת לְדִבְרֵי הַכֹּל, שֶׁאַף הַמַּתִּירִים בְּצוֹאָה מְכֻסָּה עַל בְּשָׂרוֹ — מוֹדִים בְּצוֹאָה בִּמְקוֹמָהּ, מִפְּנֵי שֶׁזֻּהֲמָתָהּ מְרֻבָּה, שֶׁחַמָּה הִיא וְלֹא יָצְאָה לָאֲוִיר וְאֵין רֵיחָהּ מִתְפַּזֵּר. וְהוּא שֶׁאִם הָיָה עוֹמֵד הָיְתָה נִרְאֵית, וַאֲפִלּוּ לֹא הָיְתָה נִרְאֵית כְּשֶׁהוּא עוֹמֵד אֶלָּא כְּשֶׁהוּא יוֹשֵׁב. אֲבָל אִם לֹא הָיְתָה נִרְאֵית כְּלָל — מֻתָּר, מִפְּנֵי שֶׁלֹּא נִתְּנָה תּוֹרָה לְמַלְאֲכֵי הַשָּׁרֵת. וְכָל זֶה לְאוֹתוֹ אָדָם עַצְמוֹ, אֲבָל אַחֵר מֻתָּר לִקְרוֹת אֶצְלוֹ כְּשֶׁהוּא מְכֻסֶּה בִּבְגָדָיו:

צוֹאָה בִּמְקוֹמָהּ. בְּנֶקֶב פִּי הַטַּבַּעַת, אֲפִלּוּ מְכֻסָּה — אָסוּר לְדִבְרֵי הַכֹּל, שֶׁזֻּהֲמָתָהּ מְרֻבָּה ; וְהוּא שֶׁנִּרְאֵית בַּעֲמִידָה אוֹ בִּישִׁיבָה, אֲבָל אֵינָהּ נִרְאֵית כְּלָל — מֻתָּר. וְכָל זֶה לְעַצְמוֹ, אֲבָל אַחֵר מֻתָּר אֶצְלוֹ.

סעיף ז חֳלִי הַטְּחוֹרִים וְלֵחָה סְרוּחָה — דִּינָהּ.

מִי שֶׁיֵּשׁ לוֹ חֳלִי הַטְּחוֹרִים וְזָב מִמֶּנּוּ דָּם תָּמִיד, וְאַגַּב הַדָּם יוֹצֵא מִמֶּנּוּ לֵחָה סְרוּחָה וּמְעֻפֶּשֶׁת וְיֵשׁ לָהּ רֵיחַ רַע — אָסוּר בְּכָל דָּבָר שֶׁבִּקְדֻשָּׁה כָּל זְמַן שְׁלֵחָה סְרוּחָה שׁוֹתֵת מִמֶּנּוּ. וְאִם אֵין שָׁם רֵיחַ רַע, אִם הַדָּם אוֹ הַלֵּחָה יוֹצֵא בְּדֶרֶךְ דְּחִיָּה בְּסֵרוּגִין — מִן הַמֵּעַיִם הֵם, וְיֵשׁ לוֹ תַּקָּנָה לִרְחֹץ מְקוֹם הַזֻּהֲמָא. וְאִם שׁוֹתֵת וְיוֹרֶדֶת תָּמִיד מֵעַצְמָהּ שֶׁלֹּא בְּדֶרֶךְ דְּחִיָּה — מִפִּי הַטַּבַּעַת הִיא, וְאֵין צָרִיךְ תַּקָּנָה דַהֲוֵי כְּדַם מַכָּתוֹ:

חֳלִי הַטְּחוֹרִים. לֵחָה סְרוּחָה בַּעֲלַת רֵיחַ רַע שׁוֹתֶתֶת — אָסוּר בְּכָל שֶׁבִּקְדֻשָּׁה. בְּלֹא רֵיחַ : בְּדֶרֶךְ דְּחִיָּה בְּסֵרוּגִין מִן הַמֵּעַיִם וְיֵשׁ תַּקָּנָה לִרְחֹץ ; שׁוֹתֶתֶת מֵעַצְמָהּ מִפִּי הַטַּבַּעַת וְאֵין צָרִיךְ, כְּדַם מַכָּתוֹ.

סעיף ח מְעַט צוֹאָה — מְבַטְּלָהּ בְּרֹק עָבֶה לְפִי שָׁעָה.

הָיָה לְפָנָיו מְעַט צוֹאָה — יָכוֹל לְבַטְּלָהּ בְּרֹק שֶׁיָּרֹק בָּהּ וְיִקְרָא. וְהוּא שֶׁיִּהְיֶה הָרֹק עָבֶה, שֶׁלֹּא תְּהֵא הַצּוֹאָה נִרְאֵית. וְכֵן מַיִם שֶׁעַל גַּבֵּי צוֹאָה אֵינָם חֲשׁוּבִים כִּסּוּי, אֶלָּא אִם כֵּן הֵם עֲכוּרִים שֶׁאֵין הַצּוֹאָה נִרְאֵית מִתּוֹכָם. וְאֵין בִּטּוּל הָרֹק מוֹעִיל אֶלָּא לְפִי שָׁעָה, אֲבָל אִם לֹא יִקְרָא מִיָּד וְהָרֹק נִמּוֹחַ וְנִבְלָע בָּהּ — לֹא בְּטֵלָה:

בִּטּוּל בְּרֹק. מְעַט צוֹאָה — מְבַטְּלָהּ בְּרֹק עָבֶה שֶׁלֹּא תֵּרָאֶה וְקוֹרֵא ; וְכֵן מַיִם עֲכוּרִים כִּסּוּי. וְאֵין הַבִּטּוּל אֶלָּא לְפִי שָׁעָה, נִמּוֹחַ הָרֹק — לֹא בְּטֵלָה.

סעיף ט סְפֵק צוֹאָה בַּבַּיִת אוֹ בָּאַשְׁפָּה ; וּסְפֵק מֵי רַגְלַיִם מֻתָּר.

סָפֵק אִם צוֹאָה בַּבַּיִת — מֻתָּר לִקְרוֹת, שֶׁחֶזְקַת בַּיִת שֶׁאֵין בּוֹ צוֹאָה. סָפֵק אִם צוֹאָה בָּאַשְׁפָּה אֲפִלּוּ אִם הוּא בַּבַּיִת — אָסוּר, שֶׁחֶזְקַת אַשְׁפָּה שֶׁיֵּשׁ בָּהּ צוֹאָה. וַאֲפִלּוּ בְּמָקוֹם [אַחֵר] שֶׁסָּפֵק אִם יֵשׁ שָׁם צוֹאָה, כְּגוֹן לִפְנֵי הַבָּתִּים שֶׁדֶּרֶךְ לִהְיוֹת שָׁם צוֹאָה — אָסוּר, וְכָל שֶׁכֵּן בֶּחָצֵר, אֶלָּא אִם כֵּן יָדוּעַ שֶׁאֵין דֶּרֶךְ לִהְיוֹת שָׁם צוֹאָה. אֲבָל סְפֵק מֵי רַגְלַיִם — אֲפִלּוּ בָּאַשְׁפָּה מֻתָּר, מִפְּנֵי שֶׁלֹּא אָסְרָה תּוֹרָה לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם אֶלָּא כְּנֶגֶד עַמּוּד שֶׁל קִלּוּחַ, אֲבָל לְאַחַר שֶׁנָּפְלוּ לָאָרֶץ מֻתָּר לִקְרוֹת אֶצְלָן מִן הַתּוֹרָה אַף עַל פִּי שֶׁהֵם מְגֻלִּים, שֶׁנֶּאֱמַר: ״וְיָד תִּהְיֶה לְךָ מִחוּץ לַמַּחֲנֶה וְיָצָאתָ שָּׁמָּה חוּץ״. וְלֹא הִצְרִיכָן כִּסּוּי כְּמוֹ בְּצוֹאָה, שֶׁנֶּאֱמַר בָּהּ: ״וְכִסִּיתָ אֶת צֵאָתֶךָ״, אֶלָּא שֶׁחֲכָמִים גָּזְרוּ עֲלֵיהֶן גְּזֵירָה מִשּׁוּם צוֹאָה, וְלֹא גָּזְרוּ אֶלָּא בְּוַדָּאָן — אֲבָל לֹא בִּסְפֵקָן:

סְפֵק צוֹאָה. בַּבַּיִת — מֻתָּר (חֶזְקַת בַּיִת) ; בָּאַשְׁפָּה וּבְמָקוֹם הָרָגִיל — אָסוּר. אֲבָל סְפֵק מֵי רַגְלַיִם — אֲפִלּוּ בָּאַשְׁפָּה מֻתָּר, שֶׁלֹּא אָסְרָה תּוֹרָה אֶלָּא עַמּוּד שֶׁל קִלּוּחַ, וּלְאַחַר שֶׁנָּפְלוּ אֵינוֹ אֶלָּא מִדְּרַבָּנָן וּבִסְפֵקָן לֹא גָּזְרוּ.

סעיף י סָפֵק צוֹאַת אָדָם אוֹ כְּלָבִים — תּוֹלִין בַּמָּצוּי.

מָצָא צוֹאָה בֵּין בַּבַּיִת בֵּין בָּאַשְׁפָּה, וְסָפֵק אִם הִיא צוֹאַת אָדָם אוֹ שֶׁל כְּלָבִים — תּוֹלִין בַּמָּצוּי, בֵּין לְהָקֵל בֵּין לְהַחֲמִיר, שֶׁאִם מְצוּיִין בָּהּ תִּינוֹקוֹת יוֹתֵר מִכְּלָבִים — אֲפִלּוּ בְּבַיִת אָסוּר, וְאִם אֲפִלּוּ בָּאַשְׁפָּה מְצוּיִין כְּלָבִים יוֹתֵר — תּוֹלִין בַּכְּלָבִים:

סָפֵק צוֹאַת אָדָם. מָצָא צוֹאָה וְסָפֵק אִם אָדָם אוֹ כְּלָבִים — תּוֹלִין בַּמָּצוּי, בֵּין לְהָקֵל בֵּין לְהַחֲמִיר : רֹב תִּינוֹקוֹת — אָסוּר אַף בַּבַּיִת ; רֹב כְּלָבִים — תּוֹלִין בַּכְּלָבִים אַף בָּאַשְׁפָּה.

סעיף י״א קָרָא וּמָצָא צוֹאָה — חוֹזֵר ; מֵי רַגְלַיִם — אֵין צָרִיךְ.

קָרָא קְרִיאַת שְׁמַע אוֹ הִתְפַּלֵּל בְּמָקוֹם שֶׁרָאוּי לְהִסְתַּפֵּק בּוֹ בְּצוֹאָה וּמְצָאָהּ שָׁם אַחַר כָּךְ — צָרִיךְ לַחֲזֹר וְלִקְרוֹת וּלְהִתְפַּלֵּל, לְפִי שֶׁפָּשַׁע בַּמֶּה שֶׁלֹּא בָּדַק הַמָּקוֹם קֹדֶם שֶׁקָּרָא אוֹ הִתְפַּלֵּל, וַהֲרֵינִי קוֹרֵא בּוֹ ״וְזֶבַח רְשָׁעִים תּוֹעֵבָה״ (וְאִם בֵּרֵךְ בִּרְכַּת הַמָּזוֹן אוֹ שְׁאָר בְּרָכוֹת — יִתְבָּאֵר בְּסִימָן קפ״ה). אֲבָל אִם אֵין הַמָּקוֹם רָאוּי לְהִסְתַּפֵּק בּוֹ בְּצוֹאָה — אֵין צָרִיךְ לַחֲזֹר וְלִקְרוֹת וּלְהִתְפַּלֵּל. וְאִם מָצָא מֵי רַגְלַיִם אֲפִלּוּ בִּמְקוֹם שֶׁרָאוּי לְהִסְתַּפֵּק בָּהֶם — אֵין צָרִיךְ לַחֲזֹר וְלִקְרוֹת וּלְהִתְפַּלֵּל, הוֹאִיל וְאֵין אִסּוּרָן אֶלָּא מִדִּבְרֵי סוֹפְרִים:

קָרָא וּמָצָא. קָרָא בְּמָקוֹם הָרָאוּי לְהִסְתַּפֵּק וּמָצָא צוֹאָה — חוֹזֵר, שֶׁפָּשַׁע שֶׁלֹּא בָּדַק (״וְזֶבַח רְשָׁעִים תּוֹעֵבָה״) ; אֵין הַמָּקוֹם רָאוּי — אֵין צָרִיךְ. וּמֵי רַגְלַיִם אֲפִלּוּ בְּמָקוֹם רָאוּי — אֵין צָרִיךְ, שֶׁאֵין אִסּוּרָן אֶלָּא מִדִּבְרֵי סוֹפְרִים.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ע״ו (מהדורת קה״ת, י״א סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין לְהִזָּהֵר מִצּוֹאָה בִּשְׁעַת קְרִיאַת שְׁמַע

היסוד : הַזְּהִירוּת מִצּוֹאָה בִּקְרִיאַת שְׁמַע

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַכִּסּוּי הַמַּתִּיר — עֲשָׁשִׁית, גֻּמָּא, בֶּגֶד. אחר כך הַצּוֹאָה עוֹבֶרֶת וּפִי חֲזִיר. אחר כך הַצּוֹאָה עַל הַגּוּף וּבִמְקוֹמָהּ. ולבסוף הַסָּפֵק, הַחֲזָקוֹת וְקֻלַּת הַמֵּי רַגְלַיִם.

כִּסּוּי מַתִּיר — וְכִסִּיתָ אֶת צֵאָתֶךָ

כִּסּוּי מַתִּיר — וְכִסִּיתָ אֶת צֵאָתֶךָ

היסוד. תָּלָה הַכָּתוּב בְּכִסּוּי : צוֹאָה בַּעֲשָׁשִׁית מְכֻסָּה — מֻתָּר ; בַּגֻּמָּא בְּסַנְדָּל — כִּמְכֻסָּה ; עַל בֶּגֶד מִבִּפְנִים — ״מַחֲנֶיךָ קָדוֹשׁ״ (סעיפים א-ב).

המסקנה. כִּסּוּי מַפְסִיק מַתִּיר, וּבִלְבַד שֶׁאֵין רֵיחַ רַע. עיין סימן 74 (כיסוי ערווה).

צוֹאָה עוֹבֶרֶת וּפִי חֲזִיר

צוֹאָה עוֹבֶרֶת וּפִי חֲזִיר

היסוד. צוֹאָה עוֹבֶרֶת (צָפָה אוֹ גְּרָף שֶׁל רְעִי) — מַפְסִיק עַד שֶׁתַּעֲבֹר ; פִּי חֲזִיר כְּצוֹאָה עוֹבֶרֶת, אֵין רְחִיצָה מוֹעֶלֶת (סעיף ג).

המסקנה. הַמְיֻחָד לַזֻּהֲמָא (כִּגְרָף) אוֹסֵר אַף רֵיקָן. עיין סימן 77 (מי רגלים).

צוֹאָה עַל הַגּוּף וּבִמְקוֹמָהּ

צוֹאָה עַל הַגּוּף וּבִמְקוֹמָהּ

היסוד. צוֹאָה עַל בְּשָׂרוֹ מְכֻסָּה — יֵשׁ מַתִּירִים וְיֵשׁ אוֹסְרִים, וְנָכוֹן כְּהָאוֹסְרִים ; בִּמְקוֹמָהּ (פִּי הַטַּבַּעַת) — אָסוּר לְדִבְרֵי הַכֹּל (סעיפים ד, ו).

המסקנה. הָרַב מַחְמִיר עַל הַגּוּף, וְאַף בְּשִׁכְבַת זֶרַע. עיין סימן 75 (ערווה).

סָפֵק וַחֲזָקוֹת — מֵי רַגְלַיִם

סָפֵק וַחֲזָקוֹת — מֵי רַגְלַיִם

היסוד. חֶזְקַת בַּיִת נָקִי מַתִּיר, חֶזְקַת אַשְׁפָּה אוֹסֵר ; תּוֹלִין בַּמָּצוּי ; סְפֵק מֵי רַגְלַיִם מֻתָּר, שֶׁאֵין אִסּוּרָן אֶלָּא מִדְּרַבָּנָן (סעיפים ט-י״א).

המסקנה. הַחֲזָקוֹת וְדַרְגַּת הָאִסּוּר (דְּאוֹרַיְתָא / דְּרַבָּנָן) מַכְרִיעוֹת אֶת הַסָּפֵק. עיין סימן 77 (מי רגלים).

הפניה ללימוד. היקף הסימן — הַכִּסּוּי, הַצּוֹאָה בִּמְקוֹמָהּ וְהַחֲזָקוֹת — נוגע להכנת ק״ש ותפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין לְהִזָּהֵר מִצּוֹאָה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ע״ו. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מי״א סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
כִּסּוּי הַצּוֹאָה או״ח ע״ו — צוֹאָה בַּעֲשָׁשִׁית וּבַגֻּמָּא בְּסַנְדָּל — מֻתָּר, שֶׁמְּכֻסָּה. מ״ב ע״ו — גֶּדֶר הַכִּסּוּי וּתְנַאי שֶׁאֵין רֵיחַ רַע. ערוה״ש ע״ו — דִּין עֲשָׁשִׁית וְכִסּוּי הַצּוֹאָה. דרך אדמו״ר הזקן — תָּלָה בְּכִסּוּי (״וְכִסִּיתָ אֶת צֵאָתֶךָ״) ; סַנְדָּל כִּסּוּי מַפְסִיק, וְעַל בֶּגֶד מִבִּפְנִים ״מַחֲנֶיךָ קָדוֹשׁ״ (סעיפים א-ב).
צוֹאָה עַל הַגּוּף וּבִמְקוֹמָהּ או״ח ע״ו — עַל בְּשָׂרוֹ מְכֻסָּה יֵשׁ מַתִּירִים וְיֵשׁ אוֹסְרִים ; בִּמְקוֹמָהּ אָסוּר לַכֹּל. מ״ב ע״ו — פְּרָטֵי הַמַּחְמִיר וְדִין שִׁכְבַת זֶרַע. ערוה״ש ע״ו — גֶּדֶר צוֹאָה בִּמְקוֹמָהּ וְזֻהֲמָתָהּ. דרך אדמו״ר הזקן — נָכוֹן כְּדִבְרֵי הָאוֹסְרִים בְּכָל עִנְיָן, וְאַף בְּשִׁכְבַת זֶרַע לְהַחֲמִיר ; בִּמְקוֹמָהּ אָסוּר לְדִבְרֵי הַכֹּל (סעיפים ד, ו).
בִּטּוּל בְּרֹק וְחֳלִי הַטְּחוֹרִים או״ח ע״ו — מְעַט צוֹאָה מְבַטְּלָהּ בְּרֹק לְפִי שָׁעָה. מ״ב ע״ו — דִּין הָרֹק הֶעָבֶה וּמַיִם עֲכוּרִים. ערוה״ש ע״ו — דִּין בַּעַל טְחוֹרִים וְלֵחָה סְרוּחָה. דרך אדמו״ר הזקן — רֹק עָבֶה מְבַטֵּל לְפִי שָׁעָה, נִמּוֹחַ לֹא בְּטֵלָה ; בַּעַל טְחוֹרִים בְּדֶרֶךְ דְּחִיָּה יֵשׁ תַּקָּנָה, שׁוֹתֵת כְּדַם מַכָּתוֹ (סעיפים ז-ח).
סָפֵק צוֹאָה וּמֵי רַגְלַיִם או״ח ע״ו — סָפֵק בַּבַּיִת מֻתָּר, בָּאַשְׁפָּה אָסוּר ; סְפֵק מֵי רַגְלַיִם מֻתָּר. מ״ב ע״ו — ״תּוֹלִין בַּמָּצוּי״ וּגְדַר הַחֲזָקוֹת. ערוה״ש ע״ו — הַהַכְרָעָה בַּסָּפֵק וְקָרָא וּמָצָא. דרך אדמו״ר הזקן — חֶזְקַת בַּיִת מַתִּיר, חֶזְקַת אַשְׁפָּה אוֹסֵר, תּוֹלִין בַּמָּצוּי ; מֵי רַגְלַיִם מִדְּרַבָּנָן וּבִסְפֵקָן לֹא גָּזְרוּ (סעיפים ט-י״א).

הטבלה נערכה מטקסט המחבר (אורח חיים ע״ו, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (י״א סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — כִּסּוּי מַתִּיר

קו א — כִּסּוּי מַתִּיר

יסוד הסימן : תָּלָה הַכָּתוּב בְּכִסּוּי ; עֲשָׁשִׁית, גֻּמָּא בְּסַנְדָּל, בֶּגֶד מִבִּפְנִים — מֻתָּר (סעיפים א-ב).

קריאת הרב : כִּסּוּי הַמַּפְסִיק דַּי, כָּל שֶׁאֵין רֵיחַ רַע. עיין סימן 74.

מסקנה למעשה : לְכַסּוֹת אֶת הַמָּקוֹר אוֹ לְהַפְסִיק, וְלִבְדֹּק שֶׁאֵין רֵיחַ.

קו ב — חֻמְרָא בְּצוֹאָה שֶׁעַל הַגּוּף

קו ב — חֻמְרָא בְּצוֹאָה שֶׁעַל הַגּוּף

יסוד הסימן : עַל בְּשָׂרוֹ מְכֻסָּה — נָכוֹן כְּדִבְרֵי הָאוֹסְרִים ; בִּמְקוֹמָהּ — אָסוּר לְדִבְרֵי הַכֹּל (סעיפים ד, ו).

קריאת הרב : הָרַב מַכְרִיעַ לְחֻמְרָא, וְאַף בְּשִׁכְבַת זֶרַע. עיין סימן 75.

מסקנה למעשה : לְנַקּוֹת אֶת הַגּוּף ; אֵין לִסְמֹךְ עַל הַבֶּגֶד כְּשֶׁהַצּוֹאָה עַל הַבָּשָׂר.

קו ג — הֲקֵלָה בַּסָּפֵק וּבְמֵי רַגְלַיִם

קו ג — הֲקֵלָה בַּסָּפֵק וּבְמֵי רַגְלַיִם

יסוד הסימן : חֶזְקַת בַּיִת מַתִּיר ; תּוֹלִין בַּמָּצוּי ; סְפֵק מֵי רַגְלַיִם מֻתָּר (סעיפים ט-י״א).

קריאת הרב : דַּרְגַּת הָאִסּוּר (דְּאוֹרַיְתָא אוֹ דְּרַבָּנָן) מְמַתֶּנֶת אֶת חֻמְרַת הַסָּפֵק. עיין סימן 77.

מסקנה למעשה : לִסְמֹךְ עַל חֶזְקַת הַמָּקוֹם, בְּלִי לְהַרְבּוֹת חֻמְרוֹת בְּמָקוֹם שֶׁאֵין אִסּוּר אֶלָּא מִדְּרַבָּנָן.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ע״ו ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ע״ו (דִּין לְהִזָּהֵר מִצּוֹאָה בִּשְׁעַת קְרִיאַת שְׁמַע)

  • ① כִּסּוּי. לִבְדֹּק שֶׁהַצּוֹאָה מְכֻסָּה (עֲשָׁשִׁית, סַנְדָּל עַל גֻּמָּא, אוֹ תַּחַת הַבֶּגֶד) וְאֵין רֵיחַ רַע — וְאָז מֻתָּר לִקְרוֹת וּלְהִתְפַּלֵּל. יסוד : שוע״ר ע״ו סעיפים א-ב.
  • ② צוֹאָה עוֹבֶרֶת. צוֹאָה עוֹבֶרֶת (אוֹ פִּי חֲזִיר) — מַפְסִיק עַד שֶׁתַּעֲבֹר (ד׳ אַמּוֹת אוֹ כִּמְלֹא עֵינָיו). יסוד : שוע״ר ע״ו סעיף ג.
  • ③ עַל הַגּוּף. צוֹאָה עַל הַבָּשָׂר, אַף מְכֻסָּה — נָכוֹן לְהַחֲמִיר (כְּהָאוֹסְרִים), וְכֵן בְּשִׁכְבַת זֶרַע. יסוד : שוע״ר ע״ו סעיף ד.
  • ④ בִּמְקוֹמָהּ. צוֹאָה בְּפִי הַטַּבַּעַת, אַף מְכֻסָּה — אָסוּר לְדִבְרֵי הַכֹּל אִם הָיְתָה נִרְאֵית ; אֵינָהּ נִרְאֵית כְּלָל — מֻתָּר. יסוד : שוע״ר ע״ו סעיף ו.
  • ⑤ סָפֵק וּמֵי רַגְלַיִם. סְפֵק צוֹאָה : חֶזְקַת בַּיִת מַתִּיר, חֶזְקַת אַשְׁפָּה אוֹסֵר, תּוֹלִין בַּמָּצוּי ; סְפֵק מֵי רַגְלַיִם — מֻתָּר (מִדְּרַבָּנָן). יסוד : שוע״ר ע״ו סעיפים ט-י״א.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ע״ו (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַכִּסּוּי, הַצּוֹאָה בִּמְקוֹמָהּ, הַחֲזָקוֹת — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
