דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ס״ד · דִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע

דרכו של אדמו״ר הזקן בדִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע : קְרָאָהּ לְמַפְרֵעַ — לֹא יָצָא, וְהִקְדִּים פָּרָשָׁה לַחֲבֶרְתָּהּ — יָצָא ; דִּלֵּג פָּסוּק — חוֹזֵר לְרֹאשׁ הַפָּסוּק אוֹ הַפָּרָשָׁה ; הַסָּפֵק בֵּין הַפָּרָשִׁיּוֹת וּבְ״וּכְתַבְתָּם״ — חֲזָקָה וְרְגִילוּת — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ס״ד — דִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע. אצל הרב הסימן נפרש לד׳ סְעִיפִים : הַסֵּדֶר וְהַקְּרָאָהּ לְמַפְרֵעַ, הַדִּלֵּג פָּסוּק וְהַחוֹזֵר בְּטָעוּת, הַסָּפֵק בֵּין הַפָּרָשִׁיּוֹת, וְהַסָּפֵק בְּ״וּכְתַבְתָּם״ עִם חֲזָקָה דִּרְגִילוּת (סִרְכֵיהּ נָקַט וְאָתָא).

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ס״ד — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ס״ד). אדמו״ר הזקן דן בו בְּדִין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ס״ד — דִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ס״ד — דִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.64), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.64.k).

סעיף א קְרָאָהּ לְמַפְרֵעַ — לֹא יָצָא ; הִקְדִּים פָּרָשָׁה לַחֲבֶרְתָּהּ — יָצָא.

דִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע וּבוֹ ד׳ סְעִיפִים:קְרָאָהּ לְמַפְרֵעַ — לֹא יָצָא, שֶׁנֶּאֱמַר: ״וְהָיוּ״ — בַּהֲוָיָתָן יְהוּ. בַּמֶּה דְּבָרִים אֲמוּרִים? בְּסֵדֶר הַפְּסוּקִים, אֲבָל אִם הִקְדִּים פָּרָשָׁה לַחֲבֶרְתָּהּ אַף עַל פִּי שֶׁאֵינוֹ רַשַּׁאי — יָצָא, לְפִי שֶׁאֵינָהּ סְמוּכָה לָהּ בַּתּוֹרָה:

קְרָאָהּ לְמַפְרֵעַ. הַקּוֹרֵא לְמַפְרֵעַ — לֹא יָצָא, ״וְהָיוּ״ בַּהֲוָיָתָן יְהוּ. בַּמֶּה דְּבָרִים אֲמוּרִים ? בְּסֵדֶר הַפְּסוּקִים ; אֲבָל הִקְדִּים פָּרָשָׁה לַחֲבֶרְתָּהּ — אַף שֶׁאֵינוֹ רַשַּׁאי — יָצָא, לְפִי שֶׁאֵינָהּ סְמוּכָה לָהּ בַּתּוֹרָה.

סעיף ב טָעָה בָּהּ — חוֹזֵר לְרֹאשׁ הַפָּסוּק ; אֵינוֹ יוֹדֵעַ הֵיכָן — לְרֹאשׁ הַפָּרָשָׁה.

קָרָא פָּרָשָׁה וְטָעָה בָּהּ, אִם יוֹדֵעַ הֵיכָן טָעָה, כְּגוֹן שֶׁקָּרָא כָּל הַפָּרָשָׁה וְדִלֵּג פָּסוּק אֶחָד בָּאֶמְצַע — חוֹזֵר לְרֹאשׁ אוֹתוֹ הַפָּסוּק וְגוֹמֵר הַפָּרָשָׁה, שֶׁאִם יִקְרָא הַפָּסוּק בַּמָּקוֹם שֶׁנִּזְכַּר נִמְצָא קוֹרְאָהּ לְמַפְרֵעַ (וַאֲפִלּוּ דִּלֵּג תֵּבָה אַחַת אוֹ אוֹת אַחַת, אוֹ שֶׁשִּׁנָּה נְקֻדָּה אַחַת עַל דֶּרֶךְ שֶׁיִּתְבָּאֵר בְּסִימָן קמ״ב — צָרִיךְ לַחֲזֹר לְרֹאשׁ הַפָּסוּק שֶׁטָּעָה בּוֹ). וְאִם אֵינוֹ יוֹדֵעַ הֵיכָן טָעָה — צָרִיךְ לַחֲזֹר לְרֹאשׁ הַפָּסוּק מִמָּקוֹם שֶׁמִּסְתַּפֵּק בּוֹ בְּטָעוּת. כְּגוֹן, אִם יָדוּעַ לוֹ שֶׁאָמַר פָּסוּק ״שְׁמַע״ בְּכַוָּנָה — חוֹזֵר לִ״וְאָהַבְתָּ״, וְאִם אֵינוֹ יוֹדֵעַ שֶׁאָמַר ״שְׁמַע״ בְּכַוָּנָה — חוֹזֵר לִ״שְׁמַע״:

דִּלֵּג פָּסוּק. יוֹדֵעַ הֵיכָן טָעָה (דִּלֵּג פָּסוּק בָּאֶמְצַע) — חוֹזֵר לְרֹאשׁ אוֹתוֹ הַפָּסוּק ; אֲפִלּוּ תֵּבָה אוֹ אוֹת אוֹ נְקֻדָּה (סִימָן קמ״ב). אֵינוֹ יוֹדֵעַ הֵיכָן — חוֹזֵר לְרֹאשׁ הַפָּסוּק מִמְּקוֹם הַסָּפֵק : יָדַע שֶׁאָמַר ״שְׁמַע״ בְּכַוָּנָה → לִ״וְאָהַבְתָּ״ ; לֹא יָדַע → לִ״שְׁמַע״.

סעיף ג סָפֵק בֵּין פָּרָשָׁה לְפָרָשָׁה — חוֹזֵר וּמַתְחִיל ״וְהָיָה אִם שָׁמֹעַ״.

טָעָה בֵּין פָּרָשָׁה לְפָרָשָׁה, שֶׁהוּא יוֹדֵעַ שֶׁסִּיֵּם פָּרָשָׁה וְאֵינוֹ יוֹדֵעַ אִם רִאשׁוֹנָה וְאִם שְׁנִיָּה — חוֹזֵר לְסִיּוּם פָּרָשָׁה רִאשׁוֹנָה וְיַתְחִיל ״וְהָיָה אִם שָׁמֹעַ״:

סָפֵק בֵּין פָּרָשָׁה לְפָרָשָׁה. יוֹדֵעַ שֶׁסִּיֵּם פָּרָשָׁה וְאֵינוֹ יוֹדֵעַ אִם רִאשׁוֹנָה אִם שְׁנִיָּה — חוֹזֵר לְסִיּוּם פָּרָשָׁה רִאשׁוֹנָה וּמַתְחִיל ״וְהָיָה אִם שָׁמֹעַ״.

סעיף ד סָפֵק בְּ״וּכְתַבְתָּם״ — לָרִאשׁוֹנָה ; הִתְחִיל ״לְמַעַן יִרְבּוּ״ — אֵין לָחוּשׁ.

הָיָה עוֹמֵד בְּ״וּכְתַבְתָּם״, וְאֵינוֹ יוֹדֵעַ אִם בְּ״וּכְתַבְתָּם״ שֶׁבְּפָרָשָׁה רִאשׁוֹנָה אִם בְּ״וּכְתַבְתָּם״ שֶׁבְּפָרָשָׁה שְׁנִיָּה — חוֹזֵר לְ״וּכְתַבְתָּם״ שֶׁבְּפָרָשָׁה רִאשׁוֹנָה. בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁלֹּא הִתְחִיל ״לְמַעַן יִרְבּוּ יְמֵיכֶם״, אֲבָל אִם הִתְחִיל ״לְמַעַן יִרְבּוּ יְמֵיכֶם״ — אֵין לוֹ לָחוּשׁ שֶׁמָּא הִתְחִיל כֵּן אַחַר ״וּכְתַבְתָּם״ שֶׁבָּרִאשׁוֹנָה, לְפִי שֶׁמִּן הַסְּתָם קָרָא כְּפִי רְגִילֻתוֹ שֶׁהֻרְגַּל שֶׁלֹּא לוֹמַר ״לְמַעַן יִרְבּוּ״ עַד סִיּוּם פָּרָשָׁה שְׁנִיָּה. וּמִכָּל שֶׁכֵּן אִם מָצָא עַצְמוֹ בְּתֵבָה אַחֶרֶת בְּאֶמְצַע פָּרָשָׁה — חֲזָקָה שֶׁאָמַר כָּל מַה שֶּׁלְּמַעְלָה מִמֶּנָּה:

סָפֵק בְּ״וּכְתַבְתָּם״. עוֹמֵד בְּ״וּכְתַבְתָּם״ וְאֵינוֹ יוֹדֵעַ אִם שֶׁל רִאשׁוֹנָה אִם שֶׁל שְׁנִיָּה — חוֹזֵר לְשֶׁבָּרִאשׁוֹנָה. וְאִם כְּבָר הִתְחִיל ״לְמַעַן יִרְבּוּ״ — אֵין לוֹ לָחוּשׁ, שֶׁמִּן הַסְּתָם קָרָא כְּפִי רְגִילוּתוֹ (חֲזָקָה), וְכָל שֶׁכֵּן בְּתֵבָה אַחֶרֶת בָּאֶמְצַע — חֲזָקָה שֶׁאָמַר כָּל שֶׁלְּמַעְלָה מִמֶּנָּה.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ס״ד (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע

היסוד : דִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַסֵּדֶר הַפְּסוּקִים וְהַקְּרָאָהּ לְמַפְרֵעַ. אחר כך הַדִּלֵּג פָּסוּק וְהַחוֹזֵר בְּטָעוּת. אחר כך הַסָּפֵק בֵּין הַפָּרָשִׁיּוֹת. ולבסוף הַסָּפֵק בְּ״וּכְתַבְתָּם״ וְהַחֲזָקָה דִּרְגִילוּת.

סֵדֶר הַפְּסוּקִים — בַּהֲוָיָתָן יְהוּ

סֵדֶר הַפְּסוּקִים — בַּהֲוָיָתָן יְהוּ

היסוד. לִקְרוֹת אֶת הַק״ש בְּסֵדֶר ; לְמַפְרֵעַ — לֹא יָצָא (״וְהָיוּ״ בַּהֲוָיָתָן יְהוּ) ; אֲבָל הִקְדִּים פָּרָשָׁה שְׁלֵמָה לַחֲבֶרְתָּהּ — יָצָא, שֶׁאֵינָן סְמוּכוֹת בַּתּוֹרָה (סעיף א).

המסקנה. הַסֵּדֶר מְעַכֵּב בְּתוֹךְ הַפָּרָשָׁה, לֹא בֵּין הַפָּרָשִׁיּוֹת. עיין סימן 61 (דקדוק וכוונה).

דִּלֵּג פָּסוּק — חוֹזֵר לְרֹאשׁ הַפָּסוּק

דִּלֵּג פָּסוּק — חוֹזֵר לְרֹאשׁ הַפָּסוּק

היסוד. דִּלֵּג פָּסוּק בָּאֶמְצַע — חוֹזֵר לְרֹאשׁ אוֹתוֹ הַפָּסוּק וְגוֹמֵר (שֶׁלֹּא יִקְרָא לְמַפְרֵעַ) ; אֲפִלּוּ תֵּבָה אוֹ אוֹת אוֹ נְקֻדָּה (סימן קמ״ב) (סעיף ב).

המסקנה. אֵינוֹ יוֹדֵעַ הֵיכָן — חוֹזֵר לְרֹאשׁ הַפָּסוּק מִמְּקוֹם הַסָּפֵק. עיין סימן 65 (קראה סירוגין).

סָפֵק בֵּין הַפָּרָשִׁיּוֹת

סָפֵק בֵּין הַפָּרָשִׁיּוֹת

היסוד. יוֹדֵעַ שֶׁסִּיֵּם פָּרָשָׁה וְאֵינוֹ יוֹדֵעַ אִם רִאשׁוֹנָה אִם שְׁנִיָּה — חוֹזֵר לְסִיּוּם פָּרָשָׁה רִאשׁוֹנָה, וּמַתְחִיל ״וְהָיָה אִם שָׁמֹעַ״ (סעיף ג).

המסקנה. חוֹזְרִים לַוַּדַּאי הַקָּרוֹב, בְּלִי לַחֲזֹר יוֹתֵר מִן הַצֹּרֶךְ. עיין סימן 63 (כוונה בשכיבה).

סָפֵק בְּ״וּכְתַבְתָּם״ — חֲזָקָה וְרְגִילוּת

סָפֵק בְּ״וּכְתַבְתָּם״ — חֲזָקָה וְרְגִילוּת

היסוד. עוֹמֵד בְּ״וּכְתַבְתָּם״ וְאֵינוֹ יוֹדֵעַ אִם רִאשׁוֹנָה אִם שְׁנִיָּה — חוֹזֵר לָרִאשׁוֹנָה ; אֲבָל הִתְחִיל ״לְמַעַן יִרְבּוּ״ — אֵין לוֹ לָחוּשׁ (סעיף ד).

המסקנה. חֲזָקָה דִּרְגִילוּת (סִרְכֵיהּ נָקַט וְאָתָא) : מִן הַסְּתָם קָרָא כְּפִי רְגִילוּתוֹ. עיין סימן 67 (ספק אם קרא ק״ש).

הפניה ללימוד. היקף הסימן — הַסֵּדֶר, הַחוֹזֵר בְּטָעוּת וְהַחֲזָקָה דִּרְגִילוּת — נוגע להנהגת ק״ש היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ס״ד. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
קְרָאָהּ לְמַפְרֵעַ וְהַסֵּדֶר או״ח ס״ד — קְרָאָהּ לְמַפְרֵעַ לֹא יָצָא ; הִקְדִּים פָּרָשָׁה לַחֲבֶרְתָּהּ יָצָא. מ״ב ס״ד — סֵדֶר הַפְּסוּקִים מְעַכֵּב, וְטַעַם ״בַּהֲוָיָתָן יְהוּ״. ערוה״ש ס״ד — גֶּדֶר הַלְּמַפְרֵעַ וּפָרָשָׁה לַחֲבֶרְתָּהּ. דרך אדמו״ר הזקן — לְמַפְרֵעַ לֹא יָצָא (״וְהָיוּ״ בַּהֲוָיָתָן) ; הִקְדִּים פָּרָשָׁה — יָצָא, שֶׁאֵינָהּ סְמוּכָה (סעיף א).
דִּלֵּג פָּסוּק וְחוֹזֵר בְּטָעוּת או״ח ס״ד — יוֹדֵעַ הֵיכָן טָעָה חוֹזֵר לְרֹאשׁ הַפָּסוּק ; אֵינוֹ יוֹדֵעַ — לְרֹאשׁ הַפָּרָשָׁה. מ״ב ס״ד — פְּרָטֵי הַחֲזָרָה וּתְנַאי הַסָּפֵק. ערוה״ש ס״ד — דִּין הַמְדַלֵּג תֵּבָה אוֹ אוֹת. דרך אדמו״ר הזקן — חוֹזֵר לְרֹאשׁ אוֹתוֹ הַפָּסוּק ; אֲפִלּוּ תֵּבָה אוֹ אוֹת אוֹ נְקֻדָּה (סִימָן קמ״ב) ; אֵינוֹ יוֹדֵעַ — מִמְּקוֹם הַסָּפֵק (סעיף ב).
סָפֵק בֵּין פָּרָשָׁה לְפָרָשָׁה או״ח ס״ד — חוֹזֵר לְסִיּוּם פָּרָשָׁה רִאשׁוֹנָה וּמַתְחִיל ״וְהָיָה אִם שָׁמֹעַ״. מ״ב ס״ד — מִדֵּי סְפֵק חוֹזֵר לַוַּדַּאי הַקָּרוֹב. ערוה״ש ס״ד — הַהַכְרָעָה בַּסָּפֵק שֶׁבֵּין הַפָּרָשִׁיּוֹת. דרך אדמו״ר הזקן — יוֹדֵעַ שֶׁסִּיֵּם וְאֵינוֹ יוֹדֵעַ אִם רִאשׁוֹנָה — לְסִיּוּם רִאשׁוֹנָה, ״וְהָיָה אִם שָׁמֹעַ״ (סעיף ג).
סָפֵק בְּ״וּכְתַבְתָּם״ וַחֲזָקָה או״ח ס״ד — סָפֵק בְּ״וּכְתַבְתָּם״ חוֹזֵר לָרִאשׁוֹנָה ; הִתְחִיל ״לְמַעַן יִרְבּוּ״ אֵין לָחוּשׁ. מ״ב ס״ד — ״סִרְכֵיהּ נָקַט וְאָתָא״, חֲזָקָה דִּרְגִילוּת. ערוה״ש ס״ד — גֶּדֶר הַחֲזָקָה בַּהֶרְגֵּל. דרך אדמו״ר הזקן — חוֹזֵר לְ״וּכְתַבְתָּם״ שֶׁבָּרִאשׁוֹנָה ; הִתְחִיל ״לְמַעַן יִרְבּוּ״ — אֵין לָחוּשׁ, מִן הַסְּתָם כְּפִי רְגִילוּתוֹ, וְכָל שֶׁכֵּן בְּתֵבָה אַחֶרֶת — חֲזָקָה (סעיף ד).

הטבלה נערכה מטקסט המחבר (אורח חיים ס״ד, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ב, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ט״ז, מגילה י״ז). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — סֵדֶר הַפְּסוּקִים

קו א — סֵדֶר הַפְּסוּקִים

יסוד הסימן : קְרָאָהּ לְמַפְרֵעַ לֹא יָצָא ; פָּרָשָׁה לַחֲבֶרְתָּהּ — יָצָא (סעיף א).

קריאת הרב : הַסֵּדֶר מְעַכֵּב בְּתוֹךְ הַפָּרָשָׁה (בַּהֲוָיָתָן יְהוּ), לֹא בֵּין פָּרָשִׁיּוֹת שֶׁאֵינָן סְמוּכוֹת. עיין סימן 61.

מסקנה למעשה : לִקְרוֹת בְּסֵדֶר ; הַקְדָּמַת פָּרָשָׁה שֶׁכְּבָר נַעֲשְׂתָה אֵינָהּ מְחַיֶּבֶת לַחֲזֹר.

קו ב — הַחוֹזֵר בְּטָעוּת

קו ב — הַחוֹזֵר בְּטָעוּת

יסוד הסימן : דִּלֵּג פָּסוּק — לְרֹאשׁ הַפָּסוּק ; אֵינוֹ יוֹדֵעַ הֵיכָן — לְרֹאשׁ הַפָּרָשָׁה אוֹ מִמְּקוֹם הַסָּפֵק (סעיפים ב-ג).

קריאת הרב : חוֹזְרִים לַוַּדַּאי הַקָּרוֹב, אַף פַּעַם לֹא יוֹתֵר מִן הַצֹּרֶךְ. עיין סימן 65.

מסקנה למעשה : בְּסָפֵק עַל כַּוָּנַת הַפָּסוּק הָרִאשׁוֹן — חוֹזֵר לִ״וְאָהַבְתָּ״ אוֹ לִ״שְׁמַע״.

קו ג — חֲזָקָה וְהֶרְגֵּל

קו ג — חֲזָקָה וְהֶרְגֵּל

יסוד הסימן : סָפֵק בְּ״וּכְתַבְתָּם״ ; הִתְחִיל ״לְמַעַן יִרְבּוּ״ — אֵין לָחוּשׁ (סעיף ד).

קריאת הרב : חֲזָקָה דִּרְגִילוּת (סִרְכֵיהּ נָקַט וְאָתָא) : מִן הַסְּתָם קָרָא כְּפִי רְגִילוּתוֹ. עיין סימן 67 (ספק אם קרא).

מסקנה למעשה : לִסְמֹךְ עַל הֶרְגֵּל הַקְּרִיאָה הַקָּבוּעַ, בְּלִי לַחֲזֹר לְלֹא צֹרֶךְ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ס״ד ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ס״ד (דִּין הַטּוֹעֶה בִּקְרִיאַת שְׁמַע)

  • ① סֵדֶר. לִקְרוֹת אֶת הַק״ש בְּסֵדֶר ; לְמַפְרֵעַ — לֹא יָצָא, וְחוֹזֵר. הִקְדִּים פָּרָשָׁה שְׁלֵמָה — בְּדִיעֲבַד יָצָא (שֶׁאֵינָן סְמוּכוֹת). יסוד : שוע״ר ס״ד סעיף א.
  • ② דִּלֵּג פָּסוּק. דִּלֵּג פָּסוּק וְיוֹדֵעַ הֵיכָן — חוֹזֵר לְרֹאשׁ אוֹתוֹ הַפָּסוּק וְגוֹמֵר ; אֲפִלּוּ תֵּבָה אוֹ אוֹת אוֹ נְקֻדָּה. יסוד : שוע״ר ס״ד סעיף ב.
  • ③ אֵינוֹ יוֹדֵעַ הֵיכָן. אֵינוֹ יוֹדֵעַ הֵיכָן טָעָה — חוֹזֵר מִמְּקוֹם הַסָּפֵק : ״שְׁמַע״ בְּכַוָּנָה → ״וְאָהַבְתָּ״ ; לָאו → ״שְׁמַע״. יסוד : שוע״ר ס״ד סעיפים ב-ג.
  • ④ סָפֵק בֵּין הַפָּרָשִׁיּוֹת. סָפֵק אֵיזוֹ פָּרָשָׁה סִיֵּם — חוֹזֵר לְסִיּוּם פָּרָשָׁה רִאשׁוֹנָה, ״וְהָיָה אִם שָׁמֹעַ״. יסוד : שוע״ר ס״ד סעיף ג.
  • ⑤ סָפֵק בְּ״וּכְתַבְתָּם״. סָפֵק בְּ״וּכְתַבְתָּם״ — לָרִאשׁוֹנָה ; אֲבָל כְּבָר בְּ״לְמַעַן יִרְבּוּ״ — אֵין לָחוּשׁ (חֲזָקָה דִּרְגִילוּת). יסוד : שוע״ר ס״ד סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ס״ד (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַסֵּדֶר, הַחוֹזֵר בְּטָעוּת, הַחֲזָקָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
