דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קכ״ג · דִּינֵי הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה

דרכו של אדמו״ר הזקן בְּדִּינֵי הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה : פּוֹסֵעַ ג׳ פְּסִיעוֹת בִּכְרִיעָה אַחַת, וְנוֹתֵן שָׁלוֹם לִשְׂמֹאל וְלִימִין ; הַמּוֹסִיף — יוּהֲרָא ; אֵימָתַי חוֹזֵר לִמְקוֹמוֹ — עַד קְדֻשָּׁה אוֹ קוֹל רָם ; שִׁעוּר הַפְּסִיעוֹת וְהַשְּׁלִיחַ צִבּוּר בַּחֲזָרָה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קכ״ג — דִּינֵי הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה. אצל הרב הסימן נפרש לְח׳ סְעִיפִים : הַפְּסִיעוֹת וְנְתִינַת שָׁלוֹם, אִסּוּר יוּהֲרָא, אֵימָתַי חוֹזֵר לִמְקוֹמוֹ (עַד קְדֻשָּׁה / קוֹל רָם / ד׳ אַמּוֹת), הַו׳ פְּסִיעוֹת, יָחִיד וַעֲקִירַת רֶגֶל שְׂמֹאל, שִׁעוּר הַפְּסִיעוֹת, וְהַשְּׁלִיחַ צִבּוּר בַּחֲזָרָה עִם ״יִהְיוּ לְרָצוֹן״.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קכ״ג — ח׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קכ״ג). אדמו״ר הזקן דן בו בְּדִינֵי הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קכ״ג — דִּינֵי הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קכ״ג — דִּינֵי הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה — וּבוֹ ח׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.123), ואחריו תרגומנו. הסימן כולל ח׳ סעיפים (סעיף k → ‎.123.k).

סעיף א הַפּוֹסֵעַ ג׳ פְּסִיעוֹת בִּכְרִיעָה אַחַת וְנוֹתֵן שָׁלוֹם לִשְׂמֹאל וְלִימִין.

דִּינֵי הַכְּרִיעוֹת בְּסִיּוּם י״ח בְּרָכוֹת וּבוֹ ח׳ סְעִיפִים:הַמִּתְפַּלֵּל צָרִיךְ שֶׁיִּפְסַע ג׳ פְּסִיעוֹת לַאֲחוֹרָיו כְּדֶרֶךְ שֶׁנִּפְטָרִים מִלִּפְנֵי הַמֶּלֶךְ וְאַחַר כָּךְ יִתֵּן שָׁלוֹם, וְאִם לֹא עָשָׂה כֵן — כְּאִלּוּ לֹא הִתְפַּלֵּל. וְנוֹתֵן שָׁלוֹם תְּחִלָּה לִשְׂמֹאלוֹ שֶׁהִיא יְמִין הַשְּׁכִינָה הַשְּׁרוּיָה נֶגְדּוֹ בְּצֵאתוֹ מִן הַתְּפִלָּה, וְאַחַר כָּךְ לִימִינוֹ שֶׁהוּא שְׂמֹאל הַשְּׁכִינָה, דְּהַיְנוּ שֶׁכְּשֶׁאוֹמֵר ״עוֹשֶׂה שָׁלוֹם בִּמְרוֹמָיו״ — יִטֶּה בְּרֹאשׁוֹ לְצַד שְׂמֹאלוֹ, וּכְשֶׁאוֹמֵר ״הוּא יַעֲשֶׂה שָׁלוֹם עָלֵינוּ״ — יִטֶּה בְּרֹאשׁוֹ לְצַד יְמִינוֹ, וְלֹא יִתֵּן שָׁלוֹם בְּשָׁעָה שֶׁפּוֹסֵעַ אֶלָּא לְאַחַר כָּךְ. וְקֹדֶם שֶׁפּוֹסֵעַ יֵשׁ לוֹ לִכְרֹעַ, וְיִפְסַע הַג׳ פְּסִיעוֹת בִּכְרִיעָה אַחַת, וּבְעוֹדוֹ כּוֹרֵעַ קֹדֶם שֶׁיִּזְקֹף — יִתֵּן שָׁלוֹם לִשְׂמֹאלוֹ וְלִימִינוֹ. וְהֻרְגְּלוּ לְהִשְׁתַּחֲווֹת אַחַר כָּךְ לִפְנֵי ה׳, כְּעֶבֶד הַנִּפְטָר מֵרַבּוֹ. וְנָהֲגוּ לוֹמַר אַחַר כָּךְ ״יְהִי רָצוֹן … שֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ כו׳״, כִּי הַתְּפִלָּה בִּמְקוֹם הָעֲבוֹדָה וְלָכֵן אָנוּ מְבַקְשִׁים עַל הַמִּקְדָּשׁ שֶׁנּוּכַל לַעֲשׂוֹת עֲבוֹדָה מַמָּשׁ:

הַפְּסִיעוֹת וְהַשָּׁלוֹם. הַמִּתְפַּלֵּל פּוֹסֵעַ ג׳ פְּסִיעוֹת לַאֲחוֹרָיו כִּפְטִירָה מִלִּפְנֵי הַמֶּלֶךְ, וְאַחַר כָּךְ נוֹתֵן שָׁלוֹם ; לֹא עָשָׂה כֵן — כְּאִלּוּ לֹא הִתְפַּלֵּל. תְּחִלָּה לִשְׂמֹאלוֹ (יְמִין הַשְּׁכִינָה) וְאַחַר כָּךְ לִימִינוֹ : בְּ״עוֹשֶׂה שָׁלוֹם״ מַטֶּה רֹאשׁוֹ לִשְׂמֹאל, בְּ״הוּא יַעֲשֶׂה שָׁלוֹם״ לְיָמִין. כּוֹרֵעַ קֹדֶם, פּוֹסֵעַ בִּכְרִיעָה אַחַת, וּבְעוֹדוֹ כּוֹרֵעַ נוֹתֵן שָׁלוֹם. וּמִשְׁתַּחֲוֶה אַחַר כָּךְ כְּעֶבֶד הַנִּפְטָר מֵרַבּוֹ, וְאוֹמֵר ״יְהִי רָצוֹן שֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ״, שֶׁהַתְּפִלָּה בִּמְקוֹם הָעֲבוֹדָה.

סעיף ב הַמּוֹסִיף עַל ג׳ פְּסִיעוֹת — יוּהֲרָא.

מִי שֶׁמּוֹסִיף עַל ג׳ פְּסִיעוֹת — יוּהֲרָא הִיא, שֶׁנִּרְאֶה חוֹלֵק כָּבוֹד לַשְּׁכִינָה יוֹתֵר מִשְּׁאָר בְּנֵי אָדָם:

לֹא לְהוֹסִיף. הַמּוֹסִיף עַל ג׳ פְּסִיעוֹת — יוּהֲרָא, שֶׁנִּרְאֶה חוֹלֵק כָּבוֹד לַשְּׁכִינָה יוֹתֵר מִשְּׁאָר בְּנֵי אָדָם.

סעיף ג אֵימָתַי חוֹזֵר לִמְקוֹמוֹ — עַד קְדֻשָּׁה, קוֹל רָם אוֹ ד׳ אַמּוֹת.

בְּמָקוֹם שֶׁכָּלוּ הַג׳ פְּסִיעוֹת — יַעֲמֹד, וְלֹא יַחֲזֹר לִמְקוֹמוֹ עַד שֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לִקְדֻשָּׁה, שֶׁאִם חוֹזֵר מִיָּד — דּוֹמֶה לְתַלְמִיד שֶׁנִּפְטַר מֵרַבּוֹ וּפָסַע לַאֲחוֹרָיו וְחוֹזֵר מִיָּד, שֶׁסּוֹפוֹ מוֹכִיחַ עַל תְּחִלָּתוֹ שֶׁלֹּא פָּסַע לַאֲחוֹרָיו כְּדֵי לִפָּטֵר מִמֶּנּוּ, אֲבָל כְּשֶׁמַּגִּיעַ שְׁלִיחַ צִבּוּר לִקְדֻשָּׁה — נִרְאֶה לַכֹּל שֶׁחוֹזֵר בִּשְׁבִיל הַקְּדֻשָּׁה. וְיֵשׁ אוֹמְרִים שֶׁמִּשֶּׁהִתְחִיל שְׁלִיחַ צִבּוּר לְהִתְפַּלֵּל בְּקוֹל רָם — מֻתָּר לַחֲזֹר לִמְקוֹמוֹ, מִפְּנֵי שֶׁנִּרְאֶה לַכֹּל שֶׁחוֹזֵר בִּשְׁבִיל לְכַוֵּן לְמַה שֶּׁיֹּאמַר הַשְּׁלִיחַ צִבּוּר. וְהַשְּׁלִיחַ צִבּוּר עַצְמוֹ יָכוֹל לַחֲזֹר מִיָּד מִן הַדִּין, שֶׁנִּרְאֶה לַכֹּל שֶׁעוֹשֶׂה כֵן כְּדֵי לְהוֹצִיא רַבִּים יְדֵי חוֹבָתָן, אֶלָּא שֶׁנָּהֲגוּ לַעֲמֹד כְּדֵי הִלּוּךְ ד׳ אַמּוֹת. וְכֵן יָחִיד הַמִּתְפַּלֵּל דַּי לוֹ שֶׁיַּעֲמֹד כְּדֵי הִלּוּךְ ד׳ אַמּוֹת לְפִי סְבָרָא זוֹ. אֲבָל לְפִי סְבָרָא הָרִאשׁוֹנָה יֵשׁ אוֹמְרִים שֶׁגַּם יָחִיד הַמִּתְפַּלֵּל צָרִיךְ לְהַמְתִּין כְּשִׁעוּר שֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לִקְדֻשָּׁה, אִם לֹא שֶׁצָּרִיךְ לַחֲזֹר וּלְהִתְפַּלֵּל שֵׁנִית, כְּגוֹן לְתַשְׁלוּמִין, שֶׁאָז לְדִבְרֵי הַכֹּל דַּי שֶׁיַּמְתִּין כְּדֵי הִלּוּךְ ד׳ אַמּוֹת מִטַּעַם שֶׁנִּתְבָּאֵר בְּסִימָן ק״ה, וְלֹא יוֹתֵר, מִפְּנֵי שֶׁנִּרְאֶה לַכֹּל שֶׁחוֹזֵר לִמְקוֹמוֹ כְּדֵי לַחֲזֹר וּלְהִתְפַּלֵּל. וְכֵן אִם חוֹזֵר לִמְקוֹמוֹ כְּדֵי לִפֹּל עַל פָּנָיו מְיֻשָּׁב, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קל״א. וּלְעִנְיַן מַעֲשֶׂה, בְּמָקוֹם שֶׁאֶפְשָׁר בְּקַל — יֵשׁ לָחֹש לַסְּבָרָא הָרִאשׁוֹנָה. וְכֵן נוֹהֲגִין. אֲבָל בְּמָקוֹם שֶׁיֵּשׁ דֹּחַק בְּבֵית הַכְּנֶסֶת וְיָבוֹאוּ לִידֵי מַחֲלֹקֶת — יָכוֹל לַחֲזֹר לִמְקוֹמוֹ מִיָּד שֶׁפָּתַח הַשְּׁלִיחַ צִבּוּר בְּקוֹל רָם. וּכְשֶׁאוֹמְרִים פִּיּוּטִים קֹדֶם קְדֻשָּׁה — לְדִבְרֵי הַכֹּל אֵין צָרִיךְ לַעֲמֹד עַד קְדֻשָּׁה, אֶלָּא מִיָּד שֶׁפָּתַח שְׁלִיחַ צִבּוּר בְּקוֹל רָם — חוֹזֵר לִמְקוֹמוֹ:

אֵימָתַי חוֹזֵר. בְּמָקוֹם שֶׁכָּלוּ הַפְּסִיעוֹת — עוֹמֵד וְלֹא חוֹזֵר עַד שֶׁיַּגִּיעַ הַשַּׁ״ץ לִקְדֻשָּׁה (חוֹזֵר מִיָּד — סוֹפוֹ מוֹכִיחַ עַל תְּחִלָּתוֹ). וְיֵשׁ אוֹמְרִים : מִשֶּׁהִתְחִיל הַשַּׁ״ץ בְּקוֹל רָם — חוֹזֵר. הַשַּׁ״ץ עַצְמוֹ חוֹזֵר מִיָּד מִן הַדִּין, וְנָהֲגוּ כְּדֵי הִלּוּךְ ד׳ אַמּוֹת ; וְכֵן הַיָּחִיד. לְמַעֲשֶׂה — בְּקַל לָחֹש לָרִאשׁוֹנָה, וּבִמְקוֹם דֹּחַק וּמַחֲלֹקֶת חוֹזֵר מִיָּד בְּקוֹל רָם ; וּבְפִיּוּטִים חוֹזֵר מִיָּד.

סעיף ד לַעֲמֹד בִּמְקוֹם הַפְּסִיעוֹת אוֹ לִפְסֹעַ ו׳ פְּסִיעוֹת.

אִם הוּא רוֹצֶה עוֹמֵד בַּמָּקוֹם שֶׁפָּסַע וְאֵינוֹ חוֹזֵר לִמְקוֹמוֹ. וְיֵשׁ אוֹמְרִים שֶׁיֵּשׁ לִפְסֹעַ ו׳ פְּסִיעוֹת, ג׳ לַאֲחוֹרָיו וְג׳ בְּשׁוּבוֹ לִמְקוֹמוֹ, שֶׁיֵּשׁ לוֹ לַחֲזֹר לִמְקוֹמוֹ בְּג׳ פְּסִיעוֹת לְפָנָיו. וּמִטַּעַם זֶה קְצָת מַקְפִּידִין שֶׁלֹּא יַעֲבֹר אָדָם לִפְנֵיהֶם בְּעוֹד שֶׁהֵם עוֹמְדִין בְּמָקוֹם שֶׁפָּסַע שֶׁלֹּא יַפְסִיק זֶה בֵּין ו׳ פְּסִיעוֹתֵיהֶם, אֲבָל מִכָּל מָקוֹם עַל יְדֵי קְפִידָתָם טוֹעִים, שֶׁכְּשֶׁרוֹאִים מִי שֶׁרוֹצֶה לַעֲבֹר לִפְנֵיהֶם — הֵם מְמַהֲרִים לַחֲזֹר לִמְקוֹמָן קֹדֶם שִׁעוּר שֶׁנִּתְבָּאֵר:

לַעֲמֹד אוֹ ו׳ פְּסִיעוֹת. אִם רוֹצֶה — עוֹמֵד בִּמְקוֹם שֶׁפָּסַע וְאֵינוֹ חוֹזֵר. וְיֵשׁ אוֹמְרִים ו׳ פְּסִיעוֹת : ג׳ לַאֲחוֹרָיו וְג׳ בְּשׁוּבוֹ. וּמִשּׁוּם כָּךְ מַקְפִּידִין שֶׁלֹּא יַעֲבֹר אָדָם לִפְנֵיהֶם — אֲבָל בַּקְּפִידָה טוֹעִים, שֶׁמְּמַהֲרִים לַחֲזֹר קֹדֶם הַשִּׁעוּר.

סעיף ה יָחִיד שֶׁסִּיֵּם קֹדֶם הַצִּבּוּר ; עוֹקֵר רֶגֶל שְׂמֹאל תְּחִלָּה.

יָחִיד שֶׁמִּתְפַּלֵּל בְּצִבּוּר וְסִיֵּם תְּפִלָּתוֹ קֹדֶם לַצִּבּוּר — אֵין לוֹ רְשׁוּת לְהַחֲזִיר פָּנָיו לַצִּבּוּר עַד שֶׁיְּסַיֵּם שְׁלִיחַ צִבּוּר תְּפִלָּתוֹ, שֶׁאָז רַשַּׁאי לְהַחֲזִיר פָּנָיו, אֲבָל לֹא לַחֲזֹר לִמְקוֹמוֹ עַד שֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לִקְדֻשָּׁה בְּמָקוֹם שֶׁאֶפְשָׁר כְּמוֹ שֶׁנִּתְבָּאֵר. וְיֵשׁ מַחֲמִירִים לַעֲמֹד בַּמָּקוֹם שֶׁפָּסַע וּלְכַוֵּן שָׁם רַגְלָיו כְּמוֹ בַּתְּפִלָּה עַד שֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לִקְדֻשָּׁה. וְטוֹב לָחֹש לְדִבְרֵיהֶם. וּכְשֶׁפּוֹסֵעַ — עוֹקֵר רֶגֶל שְׂמֹאל תְּחִלָּה, לְפִי שֶׁדֶּרֶךְ הֲלִיכַת אָדָם בִּסְתָם הִיא לַעֲקֹר רֶגֶל יָמִין תְּחִלָּה, לָכֵן עוֹקֵר כָּאן רֶגֶל שְׂמֹאל תְּחִלָּה לְהַרְאוֹת כְּאִלּוּ כָּבֵד עָלָיו לִפָּטֵר מִלִּפְנֵי הַמָּקוֹם. וּמִטַּעַם זֶה אִטֵּר רֶגֶל עוֹקֵר תְּחִלָּה שְׂמֹאלוֹ שֶׁהוּא יְמִין כָּל אָדָם:

יָחִיד וְרֶגֶל שְׂמֹאל. יָחִיד שֶׁסִּיֵּם קֹדֶם הַצִּבּוּר — אֵין לוֹ לְהַחֲזִיר פָּנָיו עַד שֶׁיְּסַיֵּם הַשַּׁ״ץ, וְלֹא לַחֲזֹר לִמְקוֹמוֹ עַד קְדֻשָּׁה. וְיֵשׁ מַחֲמִירִים לְכַוֵּן רַגְלָיו כְּמוֹ בַּתְּפִלָּה, וְטוֹב לָחֹש. וּכְשֶׁפּוֹסֵעַ — עוֹקֵר רֶגֶל שְׂמֹאל תְּחִלָּה, וְאִטֵּר — שְׂמֹאלוֹ שֶׁהוּא יְמִין כָּל אָדָם.

סעיף ו שִׁעוּר הַפְּסִיעוֹת — גּוּדָל בְּצַד עָקֵב, וְלֹא גַּסּוֹת יוֹתֵר.

שִׁעוּר פְּסִיעוֹת אֵלּוּ, לְכָל הַפָּחוֹת הוּא כְּדֵי שֶׁיִּתֵּן גוּדָל בְּצַד עָקֵב, וַאֲפִלּוּ אִם הַמָּקוֹם צַר, לְפִי שֶׁפָּחוֹת מִזֶּה אֵינָהּ נִקְרֵאת פְּסִיעָה כְּלָל. וּלְכַתְּחִלָּה לֹא יַפְסִיעַ פְּסִיעוֹת גַּסּוֹת יוֹתֵר מִזֶּה, מִפְּנֵי שֶׁנִּרְאֶה כְּרָץ מִלִּפְנֵי הַמֶּלֶךְ:

שִׁעוּר הַפְּסִיעוֹת. שִׁעוּרָן לְכָל הַפָּחוֹת כְּדֵי גוּדָל בְּצַד עָקֵב, וַאֲפִלּוּ בְּמָקוֹם צַר, שֶׁפָּחוֹת מִזֶּה אֵינָהּ פְּסִיעָה. וּלְכַתְּחִלָּה לֹא גַּסּוֹת יוֹתֵר, שֶׁנִּרְאֶה כְּרָץ מִלִּפְנֵי הַמֶּלֶךְ.

סעיף ז הַשְּׁלִיחַ צִבּוּר פּוֹסֵעַ בְּלַחַשׁ ; בְּקוֹל רָם סוֹמֵךְ עַל הַקַּדִּישׁ.

גַּם הַשְּׁלִיחַ צִבּוּר צָרִיךְ לִפְסֹעַ ג׳ פְּסִיעוֹת כְּשֶׁמִּתְפַּלֵּל בְּלַחַשׁ. אֲבָל כְּשֶׁחוֹזֵר הַתְּפִלָּה בְּקוֹל רָם — אֵין צָרִיךְ לַחֲזֹר וְלִפְסֹעַ ג׳ פְּסִיעוֹת, מִפְּנֵי שֶׁסּוֹמֵךְ עַל הַפְּסִיעוֹת שֶׁבְּסוֹף הַקַּדִּישׁ שָׁלֵם שֶׁתִּקְּנוּהוּ חֲכָמִים לִשְׁלִיחַ צִבּוּר אַחַר כָּל תְּפִלָּה, אֶלָּא שֶׁמַּפְסִיקִים בִּקְרִיאַת סֵפֶר תּוֹרָה וְסֵדֶר קְדֻשָּׁה וְהַלֵּל וְ״אָבִינוּ מַלְכֵּנוּ״ וְכָל כַּיּוֹצֵא בָּהֶן שֶׁכֻּלָּן לְסֵדֶר הַתְּפִלָּה בָּאִים, וְקַדִּישׁ שֶׁלְּאַחֲרֵיהֶם חוֹזֵר עִקָּר עַל תְּפִלַּת י״ח. וְאִם בָּא לִפְסֹעַ גַּם אַחַר תְּפִלַּת י״ח — אֵין לִמְחוֹת בְּיָדוֹ. וְאִם לֹא הִתְפַּלֵּל בְּלַחַשׁ רַק בְּקוֹל רָם — צָרִיךְ לִפְסֹעַ ג׳ פְּסִיעוֹת אַחַר תְּפִלָּתוֹ שֶׁבְּקוֹל רָם מִלְּבַד הַפְּסִיעוֹת שֶׁאַחַר הַקַּדִּישׁ:

שַׁ״ץ בְּלַחַשׁ. גַּם הַשַּׁ״ץ פּוֹסֵעַ ג׳ פְּסִיעוֹת בְּלַחַשׁ ; וּבַחֲזָרָה בְּקוֹל רָם אֵין צָרִיךְ, שֶׁסּוֹמֵךְ עַל פְּסִיעוֹת הַקַּדִּישׁ שָׁלֵם. בָּא לִפְסֹעַ גַּם אַחַר כָּךְ — אֵין לִמְחוֹת. הִתְפַּלֵּל רַק בְּקוֹל רָם — פּוֹסֵעַ אַחַר תְּפִלָּתוֹ מִלְּבַד פְּסִיעוֹת הַקַּדִּישׁ.

סעיף ח חַזָּרַת הַשַּׁ״ץ — אֵין צָרִיךְ ״יִהְיוּ לְרָצוֹן״, סוֹמֵךְ עַל ״תִּתְקַבֵּל״.

כְּשֶׁהַשְּׁלִיחַ צִבּוּר חוֹזֵר הַתְּפִלָּה — אֵין צָרִיךְ לוֹמַר ״יִהְיוּ לְרָצוֹן״ בְּסוֹף הַתְּפִלָּה, מִפְּנֵי שֶׁסּוֹמֵךְ עַל ״תִּתְקַבֵּל צְלוֹתְהוֹן״ שֶׁבְּסוֹף הַקַּדִּישׁ. וְיֵשׁ אוֹמְרִים שֶׁצָּרִיךְ לוֹמַר ״יִהְיוּ לְרָצוֹן״ בְּסוֹף הַתְּפִלָּה. וְנוֹהֲגִין כַּסְּבָרָא הָרִאשׁוֹנָה. וְאַף עַל פִּי כֵן טוֹב לָחֹש לִסְבָרָא אַחֲרוֹנָה:

יִהְיוּ לְרָצוֹן. בַּחֲזָרַת הַשַּׁ״ץ — אֵין צָרִיךְ ״יִהְיוּ לְרָצוֹן״ בַּסּוֹף, שֶׁסּוֹמֵךְ עַל ״תִּתְקַבֵּל צְלוֹתְהוֹן״ שֶׁבַּקַּדִּישׁ. וְיֵשׁ אוֹמְרִים שֶׁצָּרִיךְ, וְנוֹהֲגִין כָּרִאשׁוֹנָה, וְטוֹב לָחֹש לָאַחֲרוֹנָה.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קכ״ג (מהדורת קה״ת, ח׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּינֵי הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה

היסוד : הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַפְּסִיעָה וְהַכְּרִיעָה וְנְתִינַת שָׁלוֹם. אחר כך אִסּוּר יוּהֲרָא. אחר כך אֵימָתַי חוֹזֵר לִמְקוֹמוֹ (עַד קְדֻשָּׁה / קוֹל רָם / ד׳ אַמּוֹת) עִם שִׁעוּר הַפְּסִיעוֹת. ולבסוף הַשְּׁלִיחַ צִבּוּר בַּחֲזָרָה עִם ״יִהְיוּ לְרָצוֹן״.

הַפְּסִיעָה וְהַכְּרִיעָה — כִּפְטִירָה מִן הַמֶּלֶךְ

הַפְּסִיעָה וְהַכְּרִיעָה — כִּפְטִירָה מִן הַמֶּלֶךְ

היסוד. ג׳ פְּסִיעוֹת בִּכְרִיעָה אַחַת, וְאַחַר כָּךְ נְתִינַת שָׁלוֹם לִשְׂמֹאל וְלִימִין ; לֹא עָשָׂה כֵן — כְּאִלּוּ לֹא הִתְפַּלֵּל (סעיף א).

המסקנה. כָּל הַמַּעֲשֶׂה מְבַטֵּא פְּטִירָה מִלִּפְנֵי הַמֶּלֶךְ, וְהַתְּפִלָּה בִּמְקוֹם עֲבוֹדָה. עיין סימן צ״ח · 98 (כוונת התפלה).

אִסּוּר יוּהֲרָא — לֹא לְהוֹסִיף

אִסּוּר יוּהֲרָא — לֹא לְהוֹסִיף

היסוד. הַמּוֹסִיף עַל ג׳ פְּסִיעוֹת — יוּהֲרָא, שֶׁנִּרְאֶה חוֹלֵק כָּבוֹד לַשְּׁכִינָה יוֹתֵר מִשְּׁאָר בְּנֵי אָדָם (סעיף ב).

המסקנה. עוֹמְדִים עַל הַשִּׁעוּר הַקָּבוּעַ. עיין סימן קכ״ב · 122 (סדר סיום התפלה).

אֵימָתַי חוֹזֵר לִמְקוֹמוֹ

אֵימָתַי חוֹזֵר לִמְקוֹמוֹ

היסוד. עוֹמֵד בִּמְקוֹם הַפְּסִיעוֹת עַד שֶׁיַּגִּיעַ שַׁ״ץ לִקְדֻשָּׁה (וְיֵשׁ : מִשֶּׁפָּתַח בְּקוֹל רָם) ; שַׁ״ץ וְיָחִיד — כְּדֵי הִלּוּךְ ד׳ אַמּוֹת ; שִׁעוּר הַפְּסִיעָה — גּוּדָל בְּצַד עָקֵב (סעיפים ג-ד-ה-ו).

המסקנה. מַכְרִיעִים לְפִי הַמָּקוֹם : בְּקַל — לָרִאשׁוֹנָה, וּבִמְקוֹם דֹּחַק — מִיָּד בְּקוֹל רָם. עיין סימן ק״ד · 104 (הפסק בתפלה).

הַשְּׁלִיחַ צִבּוּר בַּחֲזָרָה

הַשְּׁלִיחַ צִבּוּר בַּחֲזָרָה

היסוד. פּוֹסֵעַ בְּלַחַשׁ ; בַּחֲזָרָה סוֹמֵךְ עַל פְּסִיעוֹת הַקַּדִּישׁ ; וְאֵין צָרִיךְ ״יִהְיוּ לְרָצוֹן״, סוֹמֵךְ עַל ״תִּתְקַבֵּל צְלוֹתְהוֹן״ (סעיפים ז-ח).

המסקנה. וְנוֹהֲגִין כַּסְּבָרָא הָרִאשׁוֹנָה, וְטוֹב לָחֹש לָאַחֲרוֹנָה. עיין סימן קכ״ד · 124 (חזרת הש״ץ).

הפניה ללימוד. היקף הסימן — הַפְּסִיעוֹת, נְתִינַת שָׁלוֹם וְאֵימָתַי חוֹזֵר לִמְקוֹמוֹ — נוגע להנהגת התפלה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּינֵי הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קכ״ג. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מח׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הַפְּסִיעוֹת וְנְתִינַת שָׁלוֹם או״ח קכ״ג — כּוֹרֵעַ וּפוֹסֵעַ ג׳ פְּסִיעוֹת בִּכְרִיעָה אַחַת ; ״עוֹשֶׂה שָׁלוֹם״ לִשְׂמֹאל וְ״הוּא יַעֲשֶׂה״ לְיָמִין. מ״ב קכ״ג — פְּרָטֵי הַכְּרִיעָה וְהַשָּׁלוֹם וְנֻסַּח ״יְהִי רָצוֹן״. ערוה״ש קכ״ג — טַעַם הַפְּסִיעוֹת כִּפְטִירָה מִן הַמֶּלֶךְ. דרך אדמו״ר הזקן — ג׳ פְּסִיעוֹת בִּכְרִיעָה אַחַת ; שָׁלוֹם לִשְׂמֹאל (יְמִין הַשְּׁכִינָה) וְלִימִין ; לֹא עָשָׂה — כְּאִלּוּ לֹא הִתְפַּלֵּל (סעיף א).
הַמּוֹסִיף וְיוּהֲרָא או״ח קכ״ג — מִי שֶׁמּוֹסִיף עַל ג׳ פְּסִיעוֹת הֲוֵי יוּהֲרָא. מ״ב קכ״ג — גֶּדֶר הַיּוּהֲרָא בַּהוֹסָפָה. ערוה״ש קכ״ג — הַטַּעַם שֶׁלֹּא לְהוֹסִיף. דרך אדמו״ר הזקן — יוּהֲרָא הִיא, שֶׁנִּרְאֶה חוֹלֵק כָּבוֹד לַשְּׁכִינָה יוֹתֵר מִשְּׁאָר בְּנֵי אָדָם (סעיף ב).
אֵימָתַי חוֹזֵר לִמְקוֹמוֹ או״ח קכ״ג — לֹא יַחֲזֹר עַד שֶׁיַּגִּיעַ שַׁ״ץ לִקְדֻשָּׁה, וְלַפָּחוֹת עַד קוֹל רָם ; יָחִיד כְּדֵי ד׳ אַמּוֹת. מ״ב קכ״ג — פְּרָטֵי הַהַמְתָּנָה וְהַשִּׁעוּר. ערוה״ש קכ״ג — מַחֲלֹקֶת הַסְּבָרוֹת בַּחֲזָרָה לַמָּקוֹם. דרך אדמו״ר הזקן — עַד קְדֻשָּׁה ; וְיֵשׁ מִקּוֹל רָם ; שַׁ״ץ וְיָחִיד כְּדֵי ד׳ אַמּוֹת ; לְמַעֲשֶׂה בְּקַל לָרִאשׁוֹנָה, וּבְדֹחַק מִיָּד (סעיפים ג-ה).
שַׁ״ץ בַּחֲזָרָה וְ״יִהְיוּ לְרָצוֹן״ או״ח קכ״ג — שַׁ״ץ פּוֹסֵעַ בְּלַחַשׁ ; בְּקוֹל רָם אֵין צָרִיךְ ; וְאֵין אוֹמֵר ״יִהְיוּ לְרָצוֹן״. מ״ב קכ״ג — סְמִיכָה עַל פְּסִיעוֹת הַקַּדִּישׁ. ערוה״ש קכ״ג — גֶּדֶר סְמִיכַת הַשַּׁ״ץ עַל הַקַּדִּישׁ. דרך אדמו״ר הזקן — סוֹמֵךְ עַל ״תִּתְקַבֵּל צְלוֹתְהוֹן״ ; וְנוֹהֲגִין כָּרִאשׁוֹנָה, וְטוֹב לָחֹש לָאַחֲרוֹנָה (סעיפים ז-ח).

הטבלה נערכה מטקסט המחבר (אורח חיים קכ״ג, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפלה, רא״ש, מגן אברהם, ט״ז, אליה רבה, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד יומא נ״ג, ברכות). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ח׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַפְּסִיעָה כִּפְטִירָה מִן הַמֶּלֶךְ

קו א — הַפְּסִיעָה כִּפְטִירָה מִן הַמֶּלֶךְ

יסוד הסימן : ג׳ פְּסִיעוֹת בִּכְרִיעָה אַחַת, שָׁלוֹם לִשְׂמֹאל וְלִימִין ; שִׁעוּר — גּוּדָל בְּצַד עָקֵב (סעיפים א, ו).

קריאת הרב : כָּל הַמַּעֲשֶׂה מְבַטֵּא פְּטִירָה מִלִּפְנֵי הַמֶּלֶךְ — וּמִשּׁוּם כָּךְ עוֹקֵר רֶגֶל שְׂמֹאל תְּחִלָּה. עיין סימן צ״ח · 98.

מסקנה למעשה : לִפְסֹעַ בְּשִׁעוּר, לֹא קְצָרוֹת מִדַּי (פָּחוֹת מִפְּסִיעָה) וְלֹא כְּרָץ מִלִּפְנֵי הַמֶּלֶךְ.

קו ב — אֵימָתַי חוֹזֵר לִמְקוֹמוֹ

קו ב — אֵימָתַי חוֹזֵר לִמְקוֹמוֹ

יסוד הסימן : עוֹמֵד בִּמְקוֹם הַפְּסִיעוֹת עַד קְדֻשָּׁה (וְיֵשׁ : מִקּוֹל רָם) ; שַׁ״ץ וְיָחִיד — כְּדֵי ד׳ אַמּוֹת (סעיפים ג-ד-ה).

קריאת הרב : מַכְרִיעִים לְפִי הַמָּקוֹם — בְּקַל לָרִאשׁוֹנָה, וּבִמְקוֹם דֹּחַק וּמַחֲלֹקֶת מִיָּד בְּקוֹל רָם. עיין סימן ק״ד · 104.

מסקנה למעשה : לֹא לַחֲזֹר מִיָּד (סוֹפוֹ מוֹכִיחַ), אַךְ גַּם לֹא לִגְרֹם מַחֲלֹקֶת בְּמָקוֹם צַר.

קו ג — הַשְּׁלִיחַ צִבּוּר בַּחֲזָרָה

קו ג — הַשְּׁלִיחַ צִבּוּר בַּחֲזָרָה

יסוד הסימן : פּוֹסֵעַ בְּלַחַשׁ ; בַּחֲזָרָה סוֹמֵךְ עַל הַקַּדִּישׁ ; אֵין צָרִיךְ ״יִהְיוּ לְרָצוֹן״ (סעיפים ז-ח).

קריאת הרב : מִבְנֵה הַתְּפִלָּה בְּצִבּוּר נִשְׁעָן עַל הַקַּדִּישׁ שָׁלֵם וְעַל ״תִּתְקַבֵּל צְלוֹתְהוֹן״. עיין סימן קכ״ד · 124.

מסקנה למעשה : וְנוֹהֲגִין כָּרִאשׁוֹנָה, וְטוֹב לָחֹש לָאַחֲרוֹנָה — לוֹמַר ״יִהְיוּ לְרָצוֹן״ בְּלַחַשׁ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קכ״ג ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קכ״ג (דִּינֵי הַכְּרִיעוֹת בְּסִיּוּם הַתְּפִלָּה)

  • ① הַפְּסִיעוֹת. לִפְסֹעַ ג׳ פְּסִיעוֹת בִּכְרִיעָה אַחַת, וְלָתֵת שָׁלוֹם לִשְׂמֹאל (״עוֹשֶׂה שָׁלוֹם״) וְאַחַר כָּךְ לְיָמִין (״הוּא יַעֲשֶׂה שָׁלוֹם״) בְּעוֹדוֹ כּוֹרֵעַ. יסוד : שוע״ר קכ״ג סעיף א.
  • ② לֹא לְהוֹסִיף. אֵין לְהוֹסִיף עַל ג׳ פְּסִיעוֹת — יוּהֲרָא. יסוד : שוע״ר קכ״ג סעיף ב.
  • ③ אֵימָתַי חוֹזֵר. עוֹמֵד בִּמְקוֹם הַפְּסִיעוֹת וְחוֹזֵר עַד שֶׁיַּגִּיעַ שַׁ״ץ לִקְדֻשָּׁה (לְכָל הַפָּחוֹת מִקּוֹל רָם) ; וּבְדֹחַק — מִיָּד בְּקוֹל רָם. יסוד : שוע״ר קכ״ג סעיפים ג-ד.
  • ④ רֶגֶל שְׂמֹאל. פּוֹסֵעַ וְעוֹקֵר רֶגֶל שְׂמֹאל תְּחִלָּה (וְאִטֵּר — שְׂמֹאלוֹ שֶׁלּוֹ) ; יָחִיד שֶׁסִּיֵּם קֹדֶם הַצִּבּוּר אֵינוֹ מַחֲזִיר פָּנָיו עַד שֶׁיְּסַיֵּם הַשַּׁ״ץ. יסוד : שוע״ר קכ״ג סעיף ה.
  • ⑤ שִׁעוּר. כָּל פְּסִיעָה לְכָל הַפָּחוֹת גּוּדָל בְּצַד עָקֵב, וּלְכַתְּחִלָּה לֹא גַּסָּה יוֹתֵר (כְּרָץ מִלִּפְנֵי הַמֶּלֶךְ). יסוד : שוע״ר קכ״ג סעיף ו.
  • ⑥ שַׁ״ץ. הַשַּׁ״ץ פּוֹסֵעַ ג׳ פְּסִיעוֹת בְּלַחַשׁ ; וּבַחֲזָרָה בְּקוֹל רָם סוֹמֵךְ עַל פְּסִיעוֹת הַקַּדִּישׁ. יסוד : שוע״ר קכ״ג סעיף ז.
  • ⑦ יִהְיוּ לְרָצוֹן. בַּחֲזָרָה אֵין הַשַּׁ״ץ צָרִיךְ ״יִהְיוּ לְרָצוֹן״ (סוֹמֵךְ עַל ״תִּתְקַבֵּל״) ; וְטוֹב לָחֹש לַסְּבָרָא הַמַּצְרִיכָה. יסוד : שוע״ר קכ״ג סעיף ח.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קכ״ג (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַפְּסִיעוֹת, אֵימָתַי חוֹזֵר, הַשַּׁ״ץ בַּחֲזָרָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
