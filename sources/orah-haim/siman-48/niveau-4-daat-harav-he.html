דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן מ״ח · אֲמִירַת פָּרָשַׁת הַתָּמִיד וּפְסוּקֵי הַקָּרְבָּנוֹת

דרכו של אדמו״ר הזקן באֲמִירַת פָּרָשַׁת הַתָּמִיד: קְרִיאַת פָּרָשַׁת הַתָּמִיד בְּכָל יוֹם קֹדֶם הַתְּפִלָּה — כְּאִלּוּ הִקְרִיבוּ ; סֵדֶר הַמַּעֲרָכָה וְרִבּוֹן הָעוֹלָם ; בְּצִבּוּר וּכְקוֹרֵא בַּתּוֹרָה ; פְּסוּקֵי מוּסַף שַׁבָּת (אֲבָל לֹא בְּר״ח וְיו״ט) ; וּמִנְהַג לְהִתְנוֹעֵעַ — ברגישות חב״ד.

למה רמה 4 מיוחדת לאדמו״ר הזקן? השולחן ערוך של אדמו״ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתבו אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. דף זה מאסף את שיטת הרב על סימן מ״ח — אֲמִירַת פָּרָשַׁת הַתָּמִיד וּפְסוּקֵי הַקָּרְבָּנוֹת. הסימן כולל אצל הרב ג׳ סְעִיפִים, שבהם אדמו״ר הזקן פורש: מִנְהַג קְרִיאַת פָּרָשַׁת הַתָּמִיד בְּכָל יוֹם קֹדֶם הַתְּפִלָּה, כְּאִלּוּ הִקְרִיבוּ קָרְבַּן תָּמִיד ; סֵדֶר הַמַּעֲרָכָה וְרִבּוֹן הָעוֹלָם, וּמִי שֶׁאִי אֶפְשָׁר לוֹ בְּצִבּוּר ; פְּסוּקֵי מוּסַף שַׁבָּת וְטַעַם שֶׁאֵין אוֹמְרִים בְּר״ח וְיו״ט ; וּמִנְהַג הַמְדַקְדְּקִים לְהִתְנוֹעֵעַ בַּתּוֹרָה וּבַתְּפִלָּה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בדף זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן מ״ח — ג׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנסרקה בספריא (Shulchan Arukh HaRav, Orach Chayim מ״ח). אדמו״ר הזקן דן באמירת פרשת התמיד כנגד הקרבן, בפסוקי מוסף שבת, ובמנהג להתנועע. הביאורים הם שלנו. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן מ״ח — אֲמִירַת פָּרָשַׁת הַתָּמִיד וּפְסוּקֵי הַקָּרְבָּנוֹת

הטקסט המלא של שולחן ערוך אדמו״ר הזקן

שולחן ערוך הרב, סימן מ״ח — אֲמִירַת פָּרָשַׁת הַתָּמִיד וּפְסוּקֵי הַקָּרְבָּנוֹת — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.48), ולאחריו ביאור קצר. הסימן כולל ג׳ סְעִיפִים.

סעיף א פָּרָשַׁת הַתָּמִיד בְּכָל יוֹם קֹדֶם הַתְּפִלָּה ; סֵדֶר הַמַּעֲרָכָה וְרִבּוֹן הָעוֹלָם ; בְּצִבּוּר וּכְקוֹרֵא בַּתּוֹרָה.

אוֹמְרִים פָּרָשַׁת הַתָּמִיד וּפְסוּקֵי קָרְבַּן שַׁבָּת אוֹמְרִים אֵצֶל פָּרָשַׁת הַתָּמִיד וּבוֹ ג׳ סְעִיפִים:נָהֲגוּ כָּל יִשְׂרָאֵל וְקָבְעוּ חוֹבָה עַל עַצְמָן לִקְרוֹת ״פָּרָשַׁת הַתָּמִיד״ בְּכָל יוֹם קֹדֶם הַתְּפִלָּה, שֶׁעַל יְדֵי הַקְּרִיאָה זוֹ — מַעֲלֶה הַקָּדוֹשׁ בָּרוּךְ הוּא עֲלֵיהֶם כְּאִלּוּ הִקְרִיבוּ קָרְבַּן תָּמִיד בְּמוֹעֲדוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן א׳. וְטוֹב לוֹמַר גַּם פָּרָשַׁת הַקְּטֹרֶת אַחַר פָּרָשַׁת הַתָּמִיד, כְּמוֹ שֶׁנִּתְבָּאֵר שָׁם. וְיֵשׁ נוֹהֲגִין לוֹמַר ״אַבַּיֵּי הֲוָה מְסַדֵּר סֵדֶר הַמַּעֲרָכָה וְכו׳״ וְ״רִבּוֹן הָעוֹלָם וְכו׳״. וְאִם אִי אֶפְשָׁר לוֹמַר הַכֹּל בְּצִבּוּר מִפְּנֵי שֶׁהַצִּבּוּר אֵין אוֹמְרִים אֶלָּא פָּרָשַׁת הַתָּמִיד בִּלְבַד — יָכוֹל לוֹמַר הַכֹּל בְּבֵיתוֹ מִשֶּׁיֵּאוֹר הַיּוֹם, וְלַחֲזֹר וְלִקְרוֹת פָּרָשַׁת הַתָּמִיד עִם הַצִּבּוּר. וְאֵין לָחֹש שֶׁיִּהְיֶה נִרְאֶה כְּמַקְרִיב ב׳ תְּמִידִין וְעוֹבֵר בְּ״בַל תּוֹסִיף״, לְפִי שֶׁבִּקְרִיאָה אֵין בָּהּ מִשּׁוּם ״בַּל תּוֹסִיף״. וְעַל צַד הַיּוֹתֵר טוֹב — יְכַוֵּן בַּפַּעַם הַב׳ שֶׁהוּא כְּקוֹרֵא בַּתּוֹרָה. וְטוֹב לוֹמַר פָּרָשַׁת הַקָּרְבָּנוֹת מְעֻמָּד — דֻּגְמַת הַקְרָבָתָן שֶׁהָיָה מְעֻמָּד:

אמירת פרשת התמיד בכל יום. נָהֲגוּ כָּל יִשְׂרָאֵל וְקָבְעוּ חוֹבָה עַל עַצְמָן לִקְרוֹת פָּרָשַׁת הַתָּמִיד בְּכָל יוֹם קֹדֶם הַתְּפִלָּה — שֶׁעַל יְדֵי הַקְּרִיאָה מַעֲלֶה עֲלֵיהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא כְּאִלּוּ הִקְרִיבוּ קָרְבַּן תָּמִיד בְּמוֹעֲדוֹ (עי׳ סימן א). וְטוֹב לוֹמַר גַּם פָּרָשַׁת הַקְּטֹרֶת, וְיֵשׁ נוֹהֲגִין סֵדֶר הַמַּעֲרָכָה וְרִבּוֹן הָעוֹלָם. מִי שֶׁאִי אֶפְשָׁר לוֹ בְּצִבּוּר — אוֹמֵר הַכֹּל בְּבֵיתוֹ מִשֶּׁיֵּאוֹר הַיּוֹם וְחוֹזֵר וְקוֹרֵא עִם הַצִּבּוּר, וְאֵין חֲשָׁשׁ שֶׁל בַּל תּוֹסִיף, וְעַל צַד הַיּוֹתֵר טוֹב יְכַוֵּן בַּפַּעַם הַשְּׁנִיָּה כְּקוֹרֵא בַּתּוֹרָה. וְטוֹב לוֹמַר הַקָּרְבָּנוֹת מְעֻמָּד.

סעיף ב בְּשַׁבָּת פְּסוּקֵי מוּסָף אַחַר הַתָּמִיד ; לֹא בְּרֹאשׁ חֹדֶשׁ וְיוֹם טוֹב ; מִנְהַג רֹאשׁ חֹדֶשׁ לְפַרְסֵם.

בְּשַׁבָּת נוֹהֲגִין לוֹמַר פְּסוּקֵי קָרְבַּן מוּסַף שַׁבָּת אַחַר פָּרָשַׁת הַתָּמִיד. וְאַף עַל פִּי שֶׁעֲדַיִן לֹא הִתְפַּלְלוּ שַׁחֲרִית וְאָסוּר לְהַקְדִּים מוּסָף לְתָמִיד שֶׁל שַׁחַר, מִכָּל מָקוֹם הֲרֵי אָמְרוּ כְּבָר פָּרָשַׁת הַתָּמִיד, וְאִם תֵּחָשֵׁב אֲמִירַת הַפְּסוּקִים שֶׁל מוּסָף לְהַקְרָבָה — גַּם אֲמִירַת פָּרָשַׁת הַתָּמִיד נֶחְשְׁבָה לְהַקְרָבָה. אֲבָל בְּרֹאשׁ חֹדֶשׁ וְיוֹם טוֹב — אֵין צָרִיךְ לְהַזְכִּיר פְּסוּקֵי קָרְבַּן מוּסָף אֵצֶל פָּרָשַׁת תָּמִיד, מִפְּנֵי שֶׁקּוֹרִין בַּתּוֹרָה בִּפְסוּקֵי מוּסָף. וּבִמְדִינוֹת אֵלּוּ נוֹהֲגִין שֶׁמַּזְכִּירִין גַּם בְּרֹאשׁ חֹדֶשׁ, כְּדֵי לְפַרְסֵם שֶׁהוּא רֹאשׁ חֹדֶשׁ:

פסוקי מוסף בשבת. בְּשַׁבָּת נוֹהֲגִין לוֹמַר פְּסוּקֵי קָרְבַּן מוּסַף אַחַר פָּרָשַׁת הַתָּמִיד. וְאַף שֶׁאָסוּר לְהַקְדִּים מוּסָף לְתָמִיד שֶׁל שַׁחַר — כֵּיוָן שֶׁכְּבָר אָמְרוּ פָּרָשַׁת הַתָּמִיד, אִם אֲמִירָה נֶחְשֶׁבֶת כְּהַקְרָבָה גַּם הַתָּמִיד בִּכְלָל. אֲבָל בְּרֹאשׁ חֹדֶשׁ וְיוֹם טוֹב אֵין צָרִיךְ, מִפְּנֵי שֶׁקּוֹרִין בַּתּוֹרָה בִּפְסוּקֵי מוּסָף ; וּבִמְדִינוֹת אֵלּוּ מַזְכִּירִין גַּם בְּרֹאשׁ חֹדֶשׁ כְּדֵי לְפַרְסֵם שֶׁהוּא רֹאשׁ חֹדֶשׁ.

סעיף ג נָהֲגוּ הַמְדַקְדְּקִים לְהִתְנוֹעֵעַ בַּתּוֹרָה וּבַתְּפִלָּה ; רֶתֶת וְכָל עַצְמוֹתַי ; וּבִלְבַד שֶׁיְּכַוֵּן.

נָהֲגוּ הַמְדַקְדְּקִים לְהִתְנוֹעֵעַ בְּשָׁעָה שֶׁקּוֹרִין בַּתּוֹרָה, דֻּגְמַת הַתּוֹרָה שֶׁנִּתְּנָה בְּרֶתֶת כְּמוֹ שֶׁכָּתוּב: ״וַיַּרְא הָעָם וַיָּנֻעוּ וְגו׳״. וְכֵן עוֹשִׂין בְּשָׁעָה שֶׁמִּתְפַּלְלִין, עַל שֵׁם ״כָּל עַצְמוֹתַי תֹּאמַרְנָה ה׳ מִי כָמוֹךָ״. וְיֵשׁ מִי שֶׁאוֹמְרִים שֶׁאֵין לְהִתְנוֹעֵעַ אֶלָּא בִּפְסוּקֵי דְּזִמְרָה שֶׁהֵם שֶׁבַח וְהַלֵּל אֲבָל לֹא בִּתְפִלַּת י״ח, רַק בַּחֲתִימַת הַבְּרָכוֹת נוֹהֲגִין קְצָת לְהִתְנוֹעֵעַ בְּנַחַת, לְקַיֵּם ״מִפְּנֵי שְׁמִי נִחַת הוּא״. וּלְעִנְיַן מַעֲשֶׂה, הֵיאַךְ שֶׁיַּעֲשֶׂה — יַעֲשֶׂה, וּבִלְבַד שֶׁיְּכַוֵּן:

להתנועע. נָהֲגוּ הַמְדַקְדְּקִים לְהִתְנוֹעֵעַ בִּשְׁעַת קְרִיאַת הַתּוֹרָה — דֻּגְמַת מַתַּן תּוֹרָה בְּרֶתֶת (וַיַּרְא הָעָם וַיָּנֻעוּ) — וְכֵן בַּתְּפִלָּה, עַל שֵׁם כָּל עַצְמוֹתַי תֹּאמַרְנָה. וְיֵשׁ אוֹמְרִים שֶׁאֵין לְהִתְנוֹעֵעַ אֶלָּא בִּפְסוּקֵי דְּזִמְרָה וְלֹא בִּתְפִלַּת י״ח, רַק בַּחֲתִימַת הַבְּרָכוֹת בְּנַחַת. וּלְמַעֲשֶׂה — כְּפִי שֶׁיַּעֲשֶׂה יַעֲשֶׂה, וּבִלְבַד שֶׁיְּכַוֵּן.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim מ״ח (מהדורת קה״ת, ג׳ סְעִיפִים). ביאור: DAAT. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.

2 · היסוד

היסוד — אֲמִירַת הַקָּרְבָּנוֹת כְּנֶגֶד הָעֲבוֹדָה

היסוד: אמירת הקרבנות כנגד העבודה

שיטת אדמו״ר הזקן בסימן זה בשלושה מהלכים. תחילה אֲמִירַת פָּרָשַׁת הַתָּמִיד: קָבְעוּ יִשְׂרָאֵל חוֹבָה לִקְרוֹתָהּ בְּכָל יוֹם קֹדֶם הַתְּפִלָּה, וּמַעֲלֶה עֲלֵיהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא כְּאִלּוּ הִקְרִיבוּ קָרְבַּן תָּמִיד. אחר כך היחס בְּצִבּוּר / כְּקוֹרֵא בַּתּוֹרָה: מִי שֶׁאִי אֶפְשָׁר לוֹ בְּצִבּוּר אוֹמֵר בְּבֵיתוֹ בְּלֹא חֲשָׁשׁ בַּל תּוֹסִיף, וּמְכַוֵּן בַּשְּׁנִיָּה כְּקוֹרֵא בַּתּוֹרָה. וּלְבַסּוֹף פְּסוּקֵי מוּסָף וּמִנְהַג לְהִתְנוֹעֵעַ: פְּסוּקֵי מוּסָף בְּשַׁבָּת, וְטַעַם שֶׁאֵין בְּר״ח וְיו״ט, וְנִעְנוּעַ הַמְדַקְדְּקִים.

אֲמִירַת פָּרָשַׁת הַתָּמִיד כְּנֶגֶד הַקָּרְבָּן

קריאת פרשת התמיד כקרבן

היסוד. נָהֲגוּ כָּל יִשְׂרָאֵל וְקָבְעוּ חוֹבָה לִקְרוֹת פָּרָשַׁת הַתָּמִיד בְּכָל יוֹם קֹדֶם הַתְּפִלָּה — כְּאִלּוּ הִקְרִיבוּ קָרְבַּן תָּמִיד בְּמוֹעֲדוֹ. וְאוֹמְרִים גַּם פָּרָשַׁת הַקְּטֹרֶת, וְיֵשׁ נוֹהֲגִין סֵדֶר הַמַּעֲרָכָה וְרִבּוֹן הָעוֹלָם.

המסקנה. וְטוֹב לוֹמַר הַקָּרְבָּנוֹת מְעֻמָּד, דֻּגְמַת הַקְרָבָתָן. עַל עִנְיַן אֲמִירַת הַקָּרְבָּנוֹת כְּנֶגֶד הָעֲבוֹדָה עי׳ סימן א.

בְּצִבּוּר וּכְקוֹרֵא בַּתּוֹרָה

בציבור וכקורא בתורה

העיקרון. מִי שֶׁאִי אֶפְשָׁר לוֹ בְּצִבּוּר — שֶׁהַצִּבּוּר אֵין אוֹמְרִים אֶלָּא פָּרָשַׁת הַתָּמִיד — אוֹמֵר הַכֹּל בְּבֵיתוֹ מִשֶּׁיֵּאוֹר הַיּוֹם, וְחוֹזֵר וְקוֹרֵא עִם הַצִּבּוּר.

ההרחבה. וְאֵין חֲשָׁשׁ שֶׁל שְׁנֵי תְּמִידִין וְשֶׁל בַּל תּוֹסִיף, שֶׁבִּקְרִיאָה אֵין בָּהּ בַּל תּוֹסִיף ; וְעַל צַד הַיּוֹתֵר טוֹב יְכַוֵּן בַּשְּׁנִיָּה כְּקוֹרֵא בַּתּוֹרָה. עַל סֵדֶר בִּרְכוֹת הַשַּׁחַר שֶׁאֶצְלוֹ קְרִיאָה זוֹ עי׳ סימן מ״ו.

פְּסוּקֵי מוּסָף וּלְהִתְנוֹעֵעַ

פסוקי מוסף ולהתנועע

העיקרון. בְּשַׁבָּת פְּסוּקֵי מוּסָף אֵצֶל פָּרָשַׁת הַתָּמִיד ; אֲבָל לֹא בְּר״ח וְיו״ט, מִפְּנֵי שֶׁקּוֹרִין בַּתּוֹרָה בִּפְסוּקֵי מוּסָף — וּבִמְדִינוֹת אֵלּוּ מַזְכִּירִין גַּם בְּר״ח לְפַרְסֵם.

ההנהגה. נָהֲגוּ הַמְדַקְדְּקִים לְהִתְנוֹעֵעַ בַּתּוֹרָה (רֶתֶת) וּבַתְּפִלָּה (כָּל עַצְמוֹתַי). עַל בִּרְכַּת הַתּוֹרָה שֶׁקֹּדֶם קְרִיאָה זוֹ עי׳ סימן מ״ז.

הפניה ללימוד. יְסוֹד הַסִּימָן — אֲמִירַת פָּרָשַׁת הַתָּמִיד כְּנֶגֶד הַקָּרְבָּן, הַיַּחַס בְּצִבּוּר / כְּקוֹרֵא בַּתּוֹרָה וּמִנְהַג לְהִתְנוֹעֵעַ — נוֹגֵעַ גַּם לַהֲנָהָגָה הַיּוֹמִית וְגַם לִרְגִישׁוּת חב״ד בִּקְדֻשַּׁת סֵדֶר הַבֹּקֶר. הַדַּף מַצִּיג זֹאת בְּמַדְרֵגַת הָעִקָּרוֹן הַמְּתֹעָד בְּטֶקְסְט הָרַב, בְּלֹא הוֹסָפָה. לְלִימּוּד מַעֲמִיק וּלְהַנְהָגָה אִישִׁית — לִלְמֹד עִם רַב חב״ד וּבַמְּקוֹרוֹת עַצְמָם.

אֲמִירַת פָּרָשַׁת הַתָּמִיד — בְּמַדְרֵגַת הָעִקָּרוֹן · לְלִימּוּד, רַב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרכו של אדמו״ר הזקן עם המחבר, הרמ״א ומשנה ברורה בנקודות המרכזיות של סימן מ״ח. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מג׳ סעיפיו. הרב מפרש את הטעם ואת הפרט במקום שהמחבר מסתפק בַּמַּשְׁמָע.

נושא מחבר רמ״א משנה ברורה דרך אדמו״ר הזקן (שולחן ערוך הרב)
אֲמִירַת פָּרָשַׁת הַתָּמִיד או״ח מ״ח — אוֹמְרִים פָּרָשַׁת הַתָּמִיד (בְּעִקָּר הַגָּהַת רמ״א). רמ״א מ״ח — וְטוֹב לוֹמַר סֵדֶר הַמַּעֲרָכָה וְרִבּוֹן הָעוֹלָם. מ״ב מ״ח / מנחות ק״י ע״א — וּנְשַׁלְּמָה פָרִים שְׂפָתֵינוּ. דרך אדמו״ר הזקן — כְּאִלּוּ הִקְרִיבוּ ; בְּצִבּוּר וּכְקוֹרֵא בַּתּוֹרָה ; מְעֻמָּד (סעיף א).
פְּסוּקֵי מוּסָף בְּשַׁבָּת או״ח מ״ח — פְּסוּקֵי קָרְבַּן שַׁבָּת אֵצֶל פָּרָשַׁת הַתָּמִיד. רמ״א מ״ח — לֹא בְּרֹאשׁ חֹדֶשׁ וְיוֹם טוֹב. מ״ב מ״ח — הַטַּעַם שֶׁקּוֹרִין בַּתּוֹרָה בִּפְסוּקֵי מוּסָף. דרך אדמו״ר הזקן — אֲמִירָה כְּהַקְרָבָה ; מִנְהַג ר״ח לְפַרְסֵם (סעיף ב).
לְהִתְנוֹעֵעַ בַּתּוֹרָה וּבַתְּפִלָּה או״ח מ״ח — נוֹהֲגִין לְהִתְנוֹעֵעַ בִּשְׁעַת הַלִּימּוּד. רמ״א מ״ח — עַל שֵׁם כָּל עַצְמוֹתַי תֹּאמַרְנָה. מ״ב מ״ח — רֶתֶת ; וּמֵבִיא דַּעַת הַשְּׁלָה. דרך אדמו״ר הזקן — וי״א רַק בִּפְסוּקֵי דְּזִמְרָה, וּבַחֲתִימָה בְּנַחַת ; וּבִלְבַד שֶׁיְּכַוֵּן (סעיף ג).

הטבלה נערכה מטקסט המחבר (אורח חיים מ״ח, מקור ספריא) ומנושאי הכלים הניתנים לאימות (בית יוסף, טור, רמב״ם הלכות תמידין / תפילה ז, רא״ש, אבודרהם, כל בו, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד מנחות ק״י ע״א, תענית כ״ז ע״ב, מגילה ל״א ע״ב). שורת אדמו״ר הזקן נסמכת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סְעִיפִים) שהובא ב§1. ייחוסים לא ודאיים הושארו « על דרך הסברה ».

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה — מעבר ליישום הפשוט של הכלל. כל אחת בנויה: עקרון הסימן → קריאת הרב → מסקנה למעשה. כל קו נסמך על הטקסט האמיתי של הרב (§1).

קו א — אֲמִירָה כְּנֶגֶד הַקָּרְבָּן

אמירה כקרבן

עקרון הסימן: קְרִיאַת פָּרָשַׁת הַתָּמִיד בְּכָל יוֹם קֹדֶם הַתְּפִלָּה (סעיף א).

קריאת הרב: נָהֲגוּ כָּל יִשְׂרָאֵל וְקָבְעוּ חוֹבָה ; הַקְּרִיאָה כְּאִלּוּ הִקְרִיבוּ, עִם סֵדֶר הַמַּעֲרָכָה וְרִבּוֹן הָעוֹלָם.

מסקנה למעשה: לוֹמַר פָּרָשַׁת הַתָּמִיד בְּכָל בֹּקֶר, מְעֻמָּד, בְּכַוָּנָה.

קו ב — פְּסוּקֵי מוּסָף בְּשַׁבָּת

פסוקי מוסף בשבת

עקרון הסימן: פְּסוּקֵי מוּסָף אֵצֶל פָּרָשַׁת הַתָּמִיד (סעיף ב).

קריאת הרב: אֲמִירַת פְּסוּקֵי מוּסָף כְּהַקְרָבָה ; אֵין בְּר״ח וְיו״ט מִשּׁוּם שֶׁקּוֹרִין בַּתּוֹרָה, זוּלַת מִנְהַג ר״ח לְפַרְסֵם.

מסקנה למעשה: לְהוֹסִיף פְּסוּקֵי מוּסָף בְּשַׁבָּת ; לִנְהֹג כְּמִנְהַג הַמָּקוֹם בְּר״ח.

קו ג — לְהִתְנוֹעֵעַ בְּלִימּוּד וּבִתְפִלָּה

להתנועע בלימוד ובתפילה

עקרון הסימן: מִנְהַג לְהִתְנוֹעֵעַ (סעיף ג).

קריאת הרב: רֶתֶת בַּתּוֹרָה וְכָל עַצְמוֹתַי בַּתְּפִלָּה ; וי״א רַק בִּפְסוּקֵי דְּזִמְרָה, וּבַחֲתִימָה בְּנַחַת. עי׳ סימן מ״ט.

מסקנה למעשה: הֵיאַךְ שֶׁיַּעֲשֶׂה יַעֲשֶׂה — וּבִלְבַד שֶׁיְּכַוֵּן.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן מ״ח ברגישות אדמו״ר הזקן וחב״ד — במדרגת העיקרון, ומפנות לרב לפרטי המקרים (אֲמִירַת פָּרָשַׁת הַתָּמִיד, סֵדֶר הַמַּעֲרָכָה, בְּצִבּוּר, פְּסוּקֵי מוּסָף, וּלְהִתְנוֹעֵעַ).

לחסיד חב״ד — סימן מ״ח (אֲמִירַת פָּרָשַׁת הַתָּמִיד)

  • ① פרשת התמיד. לוֹמַר פָּרָשַׁת הַתָּמִיד בְּכָל יוֹם קֹדֶם הַתְּפִלָּה — כְּאִלּוּ הִקְרִיבוּ קָרְבַּן תָּמִיד. יסוד: שוע״ר מ״ח סעיף א.
  • ② סדר המערכה. לוֹמַר גַּם פָּרָשַׁת הַקְּטֹרֶת, וּכְמִנְהָג סֵדֶר הַמַּעֲרָכָה וְרִבּוֹן הָעוֹלָם, הַקָּרְבָּנוֹת מְעֻמָּד. יסוד: שוע״ר מ״ח סעיף א.
  • ③ בציבור. מִי שֶׁאִי אֶפְשָׁר לוֹ בְּצִבּוּר אוֹמֵר בְּבֵיתוֹ וְחוֹזֵר עִם הַצִּבּוּר, וּמְכַוֵּן בַּשְּׁנִיָּה כְּקוֹרֵא בַּתּוֹרָה. יסוד: שוע״ר מ״ח סעיף א.
  • ④ פסוקי מוסף. בְּשַׁבָּת מוֹסִיפִין פְּסוּקֵי מוּסָף ; לֹא בְּר״ח/יו״ט (קוֹרִין בַּתּוֹרָה), אַךְ מִנְהַג לְהַזְכִּיר בְּר״ח לְפַרְסֵם. יסוד: שוע״ר מ״ח סעיף ב.
  • ⑤ להתנועע. מִנְהַג לְהִתְנוֹעֵעַ בַּתּוֹרָה וּבַתְּפִלָּה — וּבִלְבַד שֶׁיְּכַוֵּן. יסוד: שוע״ר מ״ח סעיף ג.

⚠ פרק זה מציג הנהגת חב״ד המבוססת על שולחן ערוך אדמו״ר הזקן, סימן מ״ח (טקסט אמיתי שהובא ב§1). אין הוא מחליף הכרעה רבנית למקרה פרטי. לכל שאלה מעשית — אֲמִירַת פָּרָשַׁת הַתָּמִיד, פְּסוּקֵי מוּסָף, הַנִּעְנוּעַ — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחב״ד רַב חב״ד.
