דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״ז · הַמְסֻפָּק אִם הִתְפַּלֵּל וְדִין תְּפִלַּת נְדָבָה

דרכו של אדמו״ר הזקן בדִּין הַמְסֻפָּק אִם הִתְפַּלֵּל וּתְפִלַּת נְדָבָה : מְסֻפָּק אִם הִתְפַּלֵּל — חוֹזֵר וּמַתְנֶה ; בָּרִי לוֹ שֶׁהִתְפַּלֵּל — נְדָבָה עַל יְדֵי חִדּוּשׁ, חוּץ מִמּוּסָף וּבְשַׁבָּת וְיוֹם טוֹב ; הִתְחִיל וְנִזְכַּר — פּוֹסֵק אֲפִלּוּ בְּאֶמְצַע בְּרָכָה ; הַצִּבּוּר אֵינוֹ מִתְפַּלֵּל נְדָבָה, וְהָרוֹצֶה — צָרִיךְ שֶׁיּוּכַל לְכַוֵּן — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״ז — הַמְסֻפָּק אִם הִתְפַּלֵּל וְדִין תְּפִלַּת נְדָבָה. אצל הרב הסימן נפרש לה׳ סְעִיפִים : הַמְסֻפָּק אִם הִתְפַּלֵּל וְהַחוֹזֵר וּמַתְנֶה, דִּין תְּפִלַּת נְדָבָה וְטַעַם הַחִדּוּשׁ, הַפּוֹסֵק בְּטָעוּת, גֶּדֶר הַחִדּוּשׁ בַּבְּרָכוֹת הָאֶמְצָעִיּוֹת, דִּין הַצִּבּוּר, וּלְבַסּוֹף הַצֹּרֶךְ בַּכַּוָּנָה מֵרֹאשׁ וְעַד סוֹף.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״ז — ה׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״ז). אדמו״ר הזקן דן בו בְּדִין הַמְסֻפָּק אִם הִתְפַּלֵּל וְדִין תְּפִלַּת נְדָבָה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״ז — הַמְסֻפָּק אִם הִתְפַּלֵּל וְדִין תְּפִלַּת נְדָבָה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״ז — הַמְסֻפָּק אִם הִתְפַּלֵּל וְדִין תְּפִלַּת נְדָבָה — וּבוֹ ה׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.107), ואחריו תרגומנו. הסימן כולל ה׳ סעיפים (סעיף k → ‎.107.k).

סעיף א מְסֻפָּק אִם הִתְפַּלֵּל — חוֹזֵר וּמַתְנֶה ; בָּרִי לוֹ — נְדָבָה בְּחִדּוּשׁ, חוּץ מִמּוּסָף וְשַׁבָּת.

הַמְסֻפָּק אִם הִתְפַּלֵּל וְדִין תְּפִלַּת נְדָבָה וּבוֹ ה׳ סְעִיפִים:אִם הוּא מְסֻפָּק אִם הִתְפַּלֵּל י״ח אִם לָאו — צָרִיךְ לַחֲזֹר וּלְהִתְפַּלֵּל, וּמַתְנֶה וְאוֹמֵר: אִם לֹא הִתְפַּלַּלְתִּי — תְּהֵא לְחוֹבָתִי, וְאִם הִתְפַּלַּלְתִּי — תְּהֵא לְנִדְבָתִי. וְאַף עַל פִּי שֶׁכָּל סָפֵק מִדִּבְרֵי סוֹפְרִים לְהָקֵל — הִצְרִיכוּ לַחֲזֹר וּלְהִתְפַּלֵּל, מִפְּנֵי שֶׁהַלְוַאי שֶׁיִּתְפַּלֵּל אָדָם כָּל הַיּוֹם בְּתוֹרַת נְדָבָה. אֶלָּא שֶׁאִם בָּרִי לוֹ שֶׁהִתְפַּלֵּל כְּבָר וְחוֹזֵר וּמִתְפַּלֵּל בְּתוֹרַת נְדָבָה — צָרִיךְ לְחַדֵּשׁ אֵיזֶה דָּבָר בִּתְפִלָּתוֹ כְּדֵי לְהוֹדִיעַ שֶׁהוּא נְדָבָה, וְלֹא יְהֵא כְּמַקְרִיב ב׳ תְּמִידִין וְעוֹבֵר מִשּׁוּם ״בַּל תּוֹסִיף״, שֶׁתְּפִלַּת חוֹבָה הִיא בִּמְקוֹם עוֹלַת הַתָּמִיד וּתְפִלַּת נְדָבָה הִיא בִּמְקוֹם עוֹלַת נְדָבָה. אֲבָל אִם מְסֻפָּק אִם הִתְפַּלֵּל אִם לָאו — אֵין צָרִיךְ לְחַדֵּשׁ שׁוּם דָּבָר בִּתְפִלָּתוֹ הַב׳, שֶׁכְּשֶׁחוֹזֵר וּמִתְפַּלֵּל עַל הַסָּפֵק — אֵין לְךָ חִדּוּשׁ גָּדוֹל מִזֶּה. וְעַל יְדֵי חִדּוּשׁ חוֹזֵר וּמִתְפַּלֵּל בִּנְדָבָה כָּל הַפְּעָמִים שֶׁיִּרְצֶה, חוּץ מִתְּפִלַּת מוּסָף שֶׁאֵין מִתְפַּלְלִין אוֹתָהּ בִּנְדָבָה אֲפִלּוּ מוּסַף רֹאשׁ חֹדֶשׁ, כֵּיוָן שֶׁאוֹמֵר ״וּמוּסָפִים כְּהִלְכָתָם״ וְהַמּוּסָפִים אֵינָן קְרֵבִין אֶלָּא מִשֶּׁל צִבּוּר וְאֵין הַיָּחִיד יָכוֹל לְהִתְנַדֵּב אוֹתָם. וּבְשַׁבָּת וְיוֹם טוֹב אֵינָן מִתְפַּלְלִין נְדָבָה כְּלָל, לְפִי שֶׁאֵין נְדָרִים וּנְדָבוֹת קְרֵבִין בָּהֶם:

מְסֻפָּק אִם הִתְפַּלֵּל. מִי שֶׁמְּסֻפָּק אִם הִתְפַּלֵּל י״ח אִם לָאו — חוֹזֵר וּמִתְפַּלֵּל וּמַתְנֶה (״לֹא הִתְפַּלַּלְתִּי — לְחוֹבָתִי, הִתְפַּלַּלְתִּי — לְנִדְבָתִי״). אַף שֶׁכָּל סָפֵק דְּרַבָּנָן לְהָקֵל — כָּאן הִצְרִיכוּ לַחֲזֹר, שֶׁהַלְוַאי שֶׁיִּתְפַּלֵּל כָּל הַיּוֹם בְּתוֹרַת נְדָבָה. וְאִם בָּרִי לוֹ שֶׁהִתְפַּלֵּל — צָרִיךְ חִדּוּשׁ לְהוֹדִיעַ שֶׁהוּא נְדָבָה, שֶׁלֹּא יְהֵא כְּמַקְרִיב ב׳ תְּמִידִין (בַּל תּוֹסִיף). עַל הַסָּפֵק — אֵין צָרִיךְ חִדּוּשׁ. נְדָבָה כָּל הַפְּעָמִים שֶׁיִּרְצֶה, חוּץ מִמּוּסָף (מִשֶּׁל צִבּוּר) וְלֹא בְּשַׁבָּת וְיוֹם טוֹב.

סעיף ב הִתְחִיל לְשֵׁם חוֹבָה וְנִזְכַּר שֶׁכְּבָר הִתְפַּלֵּל — פּוֹסֵק אֲפִלּוּ בְּאֶמְצַע בְּרָכָה.

אִם הִתְחִיל לְהִתְפַּלֵּל לְשֵׁם חוֹבָה, שֶׁהָיָה סוֹבֵר שֶׁלֹּא הִתְפַּלֵּל עֲדַיִן וְנִזְכַּר בְּתוֹךְ הַתְּפִלָּה שֶׁכְּבָר הִתְפַּלֵּל — צָרִיךְ לִפְסֹק אֲפִלּוּ בְּאֶמְצַע בְּרָכָה. וַאֲפִלּוּ יָכוֹל לְחַדֵּשׁ בָּהּ דָּבָר — אֵינוֹ יָכוֹל לְגָמְרָהּ לְשֵׁם נְדָבָה, כֵּיוָן שֶׁכְּבָר הִתְחִיל בָּהּ לְשֵׁם חוֹבָה, וְאִי אֶפְשָׁר לְהָבִיא קָרְבָּן שֶׁחֶצְיוֹ חוֹבָה וְחֶצְיוֹ נְדָבָה:

הִתְחִיל וְנִזְכַּר. הִתְחִיל לְשֵׁם חוֹבָה, שֶׁסָּבַר שֶׁעֲדַיִן לֹא הִתְפַּלֵּל, וְנִזְכַּר בְּתוֹךְ הַתְּפִלָּה שֶׁכְּבָר הִתְפַּלֵּל — צָרִיךְ לִפְסֹק אֲפִלּוּ בְּאֶמְצַע בְּרָכָה. וַאֲפִלּוּ יָכוֹל לְחַדֵּשׁ בָּהּ — אֵינוֹ יָכוֹל לְגָמְרָהּ לְשֵׁם נְדָבָה, שֶׁכְּבָר הִתְחִיל לְשֵׁם חוֹבָה, וְאִי אֶפְשָׁר קָרְבָּן חֶצְיוֹ חוֹבָה וְחֶצְיוֹ נְדָבָה.

סעיף ג גֶּדֶר הַחִדּוּשׁ — דָּבָר בְּכָל בְּרָכָה אֶמְצָעִית ; חִדֵּשׁ בְּאַחַת — דַּיּוֹ.

חִדּוּשׁ זֶה שֶׁאָמַרְנוּ הוּא, שֶׁיְּחַדֵּשׁ אֵיזֶה דָּבָר בְּכָל בְּרָכָה מִן הָאֶמְצָעִית מֵעֵין הַבְּרָכָה. וְאִם חִדֵּשׁ אֲפִלּוּ בִּבְרָכָה אַחַת — דַּיּוֹ, כְּדֵי לְהוֹדִיעַ שֶׁהוּא נְדָבָה וְלֹא חוֹבָה:

גֶּדֶר הַחִדּוּשׁ. הַחִדּוּשׁ שֶׁאָמַרְנוּ : שֶׁיְּחַדֵּשׁ אֵיזֶה דָּבָר בְּכָל בְּרָכָה אֶמְצָעִית, מֵעֵין הַבְּרָכָה. וְאִם חִדֵּשׁ אֲפִלּוּ בִּבְרָכָה אַחַת — דַּיּוֹ, לְהוֹדִיעַ שֶׁהוּא נְדָבָה וְלֹא חוֹבָה.

סעיף ד אֵין צִבּוּר מִתְפַּלְלִים נְדָבָה ; אֲבָל יָחִיד — אֲפִלּוּ בְּצִבּוּר.

אֵין צִבּוּר מִתְפַּלְלִים תְּפִלַּת נְדָבָה, לְפִי שֶׁאֵין צִבּוּר מְבִיאִין קָרְבַּן נְדָבָה אֶלָּא לְקֵיץ הַמִּזְבֵּחַ כְּשֶׁהַמִּזְבֵּחַ בָּטֵל וְזֶהוּ דָּבָר שֶׁאֵינוֹ מָצוּי, לָכֵן אֵין לְהִתְפַּלֵּל תְּפִלַּת י״ח כְּנֶגֶד זֶה. אֲבָל יָחִיד יָכוֹל לְהִתְפַּלֵּל תְּפִלַּת נְדָבָה אֲפִלּוּ בְּצִבּוּר:

צִבּוּר וְיָחִיד. אֵין צִבּוּר מִתְפַּלְלִים תְּפִלַּת נְדָבָה, שֶׁאֵין צִבּוּר מְבִיאִין קָרְבַּן נְדָבָה אֶלָּא לְקֵיץ הַמִּזְבֵּחַ כְּשֶׁהַמִּזְבֵּחַ בָּטֵל — דָּבָר שֶׁאֵינוֹ מָצוּי ; לָכֵן אֵין מִתְפַּלְלִים י״ח כְּנֶגֶד זֶה. אֲבָל יָחִיד יָכוֹל לְהִתְפַּלֵּל נְדָבָה אֲפִלּוּ בְּצִבּוּר.

סעיף ה הָרוֹצֶה לְהִתְפַּלֵּל נְדָבָה — מַכִּיר עַצְמוֹ שֶׁיּוּכַל לְכַוֵּן מֵרֹאשׁ וְעַד סוֹף.

הָרוֹצֶה לְהִתְפַּלֵּל תְּפִלַּת נְדָבָה — צָרִיךְ שֶׁיְּהֵא מַכִּיר אֶת עַצְמוֹ זָרִיז וְזָהִיר וְאוֹמֵד בְּדַעְתּוֹ שֶׁיּוּכַל לְכַוֵּן בִּתְפִלָּתוֹ מֵרֹאשׁ וְעַד סוֹף, אֲבָל אִם אֵינוֹ יָכוֹל לְכַוֵּן יָפֶה — אָנוּ קוֹרִין בּוֹ ״לָמָּה לִּי רֹב זִבְחֵיכֶם״, וְהַלְוַאי שֶׁיּוּכַל לְכַוֵּן בְּג׳ תְּפִלּוֹת הַקְּבוּעִים לְיוֹם:

הַצֹּרֶךְ בַּכַּוָּנָה. הָרוֹצֶה לְהִתְפַּלֵּל נְדָבָה — צָרִיךְ שֶׁיַּכִּיר אֶת עַצְמוֹ זָרִיז וְזָהִיר, וְאוֹמֵד בְּדַעְתּוֹ שֶׁיּוּכַל לְכַוֵּן מֵרֹאשׁ וְעַד סוֹף ; וְאִם לָאו — קוֹרִין בּוֹ ״לָמָּה לִּי רֹב זִבְחֵיכֶם״, וְהַלְוַאי שֶׁיּוּכַל לְכַוֵּן בְּג׳ תְּפִלּוֹת הַקְּבוּעוֹת לַיּוֹם.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק״ז (מהדורת קה״ת, ה׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — הַמְסֻפָּק אִם הִתְפַּלֵּל וְדִין תְּפִלַּת נְדָבָה

היסוד : הַמְסֻפָּק אִם הִתְפַּלֵּל וּתְפִלַּת נְדָבָה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַסָּפֵק אִם הִתְפַּלֵּל וְהַחוֹזֵר וּמַתְנֶה. אחר כך דִּין תְּפִלַּת נְדָבָה עִם הַחִדּוּשׁ. אחר כך הַפּוֹסֵק בְּטָעוּת (הִתְחִיל וְנִזְכַּר). ולבסוף דִּין הַצִּבּוּר, הַיָּחִיד וְחוֹבַת הַכַּוָּנָה.

סָפֵק אִם הִתְפַּלֵּל — חוֹזֵר וּמַתְנֶה

סָפֵק אִם הִתְפַּלֵּל — חוֹזֵר וּמַתְנֶה

היסוד. מְסֻפָּק אִם הִתְפַּלֵּל — חוֹזֵר וּמַתְנֶה (״לֹא הִתְפַּלַּלְתִּי — לְחוֹבָתִי, הִתְפַּלַּלְתִּי — לְנִדְבָתִי״) ; הִצְרִיכוּ לַחֲזֹר אַף שֶׁסָּפֵק דְּרַבָּנָן לְהָקֵל (סעיף א).

המסקנה. הַלְוַאי שֶׁיִּתְפַּלֵּל כָּל הַיּוֹם בְּתוֹרַת נְדָבָה ; עַל הַסָּפֵק אֵין צָרִיךְ חִדּוּשׁ. עיין סימן ק״ח · 108 (תשלומין).

תְּפִלַּת נְדָבָה וְהַחִדּוּשׁ

תְּפִלַּת נְדָבָה וְהַחִדּוּשׁ

היסוד. בָּרִי לוֹ שֶׁהִתְפַּלֵּל וּמִתְפַּלֵּל נְדָבָה — צָרִיךְ חִדּוּשׁ (דָּבָר מֵעֵין הַבְּרָכָה בְּכָל אֶמְצָעִית, אֲפִלּוּ בְּאַחַת דַּיּוֹ), שֶׁלֹּא יְהֵא כְּמַקְרִיב ב׳ תְּמִידִין (בַּל תּוֹסִיף) (סעיפים א, ג).

המסקנה. נְדָבָה כָּל הַפְּעָמִים שֶׁיִּרְצֶה, חוּץ מִמּוּסָף וְלֹא בְּשַׁבָּת וְיוֹם טוֹב. עיין סימן ק״ו · 106 (מי חייב בתפילה).

הַפּוֹסֵק בְּטָעוּת

הַפּוֹסֵק בְּטָעוּת

היסוד. הִתְחִיל לְשֵׁם חוֹבָה וְנִזְכַּר שֶׁכְּבָר הִתְפַּלֵּל — צָרִיךְ לִפְסֹק אֲפִלּוּ בְּאֶמְצַע בְּרָכָה, וְאֵינוֹ יָכוֹל לְגָמְרָהּ לְשֵׁם נְדָבָה (סעיף ב).

המסקנה. אִי אֶפְשָׁר קָרְבָּן חֶצְיוֹ חוֹבָה וְחֶצְיוֹ נְדָבָה : כַּוָּנַת הַהַתְחָלָה קוֹבַעַת אֶת מַהוּת הַתְּפִלָּה. עיין סימן ק״ו · 106.

צִבּוּר, יָחִיד וְכַוָּנָה

צִבּוּר, יָחִיד וְכַוָּנָה

היסוד. אֵין צִבּוּר מִתְפַּלְלִין נְדָבָה (אֵין צִבּוּר מֵבִיא נְדָבָה אֶלָּא לְקֵיץ הַמִּזְבֵּחַ) ; הַיָּחִיד יָכוֹל אֲפִלּוּ בְּצִבּוּר (סעיף ד).

המסקנה. נְדָבָה צְרִיכָה שֶׁיַּכִּיר עַצְמוֹ זָרִיז וְזָהִיר וְשֶׁיּוּכַל לְכַוֵּן מֵרֹאשׁ וְעַד סוֹף, וְאִם לָאו ״לָמָּה לִּי רֹב זִבְחֵיכֶם״ (סעיף ה). עיין סימן ק״ח · 108.

הפניה ללימוד. היקף הסימן — הַחוֹזֵר וּמַתְנֶה, דִּין תְּפִלַּת נְדָבָה וְחוֹבַת הַכַּוָּנָה — נוגע להנהגת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

הַמְסֻפָּק אִם הִתְפַּלֵּל וְדִין תְּפִלַּת נְדָבָה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק״ז. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מה׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
מְסֻפָּק אִם הִתְפַּלֵּל — חוֹזֵר וּמַתְנֶה או״ח ק״ז — מְסֻפָּק אִם הִתְפַּלֵּל חוֹזֵר וּמִתְפַּלֵּל וְאֵינוֹ צָרִיךְ לְחַדֵּשׁ. מ״ב ק״ז — גֶּדֶר הַסָּפֵק בִּתְפִלָּה וְעִנְיַן הַתְּנַאי. ערוה״ש ק״ז — יְסוֹד הַחֲזָרָה בַּסָּפֵק דְּרַבָּנָן. דרך אדמו״ר הזקן — חוֹזֵר וּמַתְנֶה (חוֹבָה/נְדָבָה), הַלְוַאי שֶׁיִּתְפַּלֵּל כָּל הַיּוֹם בְּתוֹרַת נְדָבָה ; עַל הַסָּפֵק אֵין צָרִיךְ חִדּוּשׁ (סעיף א).
נְדָבָה וְהַחִדּוּשׁ ; מוּסָף וְשַׁבָּת או״ח ק״ז — בָּרִי לוֹ עַל יְדֵי חִדּוּשׁ ; חוּץ מִמּוּסָף וּבְשַׁבָּת וְיוֹם טוֹב. מ״ב ק״ז — פְּרָטֵי הַחִדּוּשׁ וְטַעַם ״בַּל תּוֹסִיף״. ערוה״ש ק״ז — דִּין נְדָבָה בַּזְּמַן הַזֶּה. דרך אדמו״ר הזקן — חִדּוּשׁ מֵעֵין הַבְּרָכָה בְּכָל אֶמְצָעִית, אֲפִלּוּ בְּאַחַת דַּיּוֹ ; אֵין נְדָבָה בְּמוּסָף (מִשֶּׁל צִבּוּר) וְלֹא בְּשַׁבָּת (סעיפים א, ג).
הִתְחִיל וְנִזְכַּר — פּוֹסֵק או״ח ק״ז — הִתְחִיל עַל דַּעַת שֶׁלֹּא הִתְפַּלֵּל וְנִזְכַּר — פּוֹסֵק אֲפִלּוּ בְּאֶמְצַע בְּרָכָה. מ״ב ק״ז — גֶּדֶר הַהַתְחָלָה לְשֵׁם חוֹבָה. ערוה״ש ק״ז — טַעַם הַפְּסִיקָה בְּאֶמְצַע. דרך אדמו״ר הזקן — פּוֹסֵק אֲפִלּוּ בְּאֶמְצַע בְּרָכָה ; אֵין לְגָמְרָהּ נְדָבָה, אִי אֶפְשָׁר קָרְבָּן חֶצְיוֹ חוֹבָה וְחֶצְיוֹ נְדָבָה (סעיף ב).
צִבּוּר, יָחִיד וְכַוָּנָה או״ח ק״ז — אֵין צִבּוּר מִתְפַּלְלִין נְדָבָה ; הָרוֹצֶה צָרִיךְ לְכַוֵּן מֵרֹאשׁ וְעַד סוֹף. מ״ב ק״ז — חֲשַׁשׁ הַכַּוָּנָה בַּזְּמַן הַזֶּה. ערוה״ש ק״ז — מִדּוּעַ אֵין נוֹהֲגִין נְדָבָה כַּיּוֹם. דרך אדמו״ר הזקן — אֵין צִבּוּר מִתְפַּלֵּל נְדָבָה, יָחִיד אֲפִלּוּ בְּצִבּוּר ; צָרִיךְ אוֹמֵד שֶׁיּוּכַל לְכַוֵּן, וְאִם לָאו ״לָמָּה לִּי רֹב זִבְחֵיכֶם״ (סעיפים ד-ה).

הטבלה נערכה מטקסט המחבר (אורח חיים ק״ז, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה א & י, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״א, עירובין ס״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ה׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַתְּנַאי בַּסָּפֵק

קו א — הַתְּנַאי בַּסָּפֵק

יסוד הסימן : מְסֻפָּק אִם הִתְפַּלֵּל — חוֹזֵר וּמַתְנֶה, וְעַל הַסָּפֵק אֵין צָרִיךְ חִדּוּשׁ (סעיף א).

קריאת הרב : הַתְּנַאי הוֹפֵךְ אֶת הַחֲזָרָה לִנְדָבָה אִם כְּבָר הִתְפַּלֵּל — הַלְוַאי שֶׁיִּתְפַּלֵּל כָּל הַיּוֹם בְּתוֹרַת נְדָבָה. עיין סימן ק״ח · 108.

מסקנה למעשה : בַּסָּפֵק חוֹזֵר עִם הַתְּנַאי, בְּלִי לְהוֹסִיף דָּבָר.

קו ב — גֶּדֶר הַנְּדָבָה

קו ב — גֶּדֶר הַנְּדָבָה

יסוד הסימן : בָּרִי לוֹ — נְדָבָה עַל יְדֵי חִדּוּשׁ (מֵעֵין הַבְּרָכָה, אֲפִלּוּ בְּאַחַת דַּיּוֹ) ; חוּץ מִמּוּסָף וְשַׁבָּת (סעיפים א, ג).

קריאת הרב : הַחִדּוּשׁ מוֹדִיעַ שֶׁהוּא נְדָבָה וּמַרְחִיק בַּל תּוֹסִיף (כְּמַקְרִיב ב׳ תְּמִידִין). עיין סימן ק״ו · 106.

מסקנה למעשה : לְהִתְפַּלֵּל נְדָבָה — לְחַדֵּשׁ דָּבָר מֵעֵין הַבְּרָכָה ; לְעוֹלָם לֹא נְדָבָה בְּמוּסָף וְלֹא בְּשַׁבָּת וְיוֹם טוֹב.

קו ג — הַכַּוָּנָה וְהַזְּהִירוּת

קו ג — הַכַּוָּנָה וְהַזְּהִירוּת

יסוד הסימן : אֵין צִבּוּר מִתְפַּלֵּל נְדָבָה ; הָרוֹצֶה — מַכִּיר עַצְמוֹ זָרִיז וְזָהִיר שֶׁיּוּכַל לְכַוֵּן (סעיפים ד-ה).

קריאת הרב : בְּלִי כַּוָּנָה מֵרֹאשׁ וְעַד סוֹף — ״לָמָּה לִּי רֹב זִבְחֵיכֶם״ ; וְהַלְוַאי שֶׁיְּכַוֵּן כְּבָר בְּג׳ תְּפִלּוֹת הַיּוֹם. עיין סימן ק״ח · 108 (ספק אם התפלל).

מסקנה למעשה : הַנְּדָבָה אֵינָהּ לַכֹּל — וּמִכָּאן מִעוּט הַנְּהִיגָה בָּהּ בִּזְמַנֵּנוּ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״ז ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״ז (הַמְסֻפָּק אִם הִתְפַּלֵּל וְדִין תְּפִלַּת נְדָבָה)

  • ① סָפֵק. מְסֻפָּק אִם הִתְפַּלֵּל — חוֹזֵר וּמַתְנֶה (״לֹא הִתְפַּלַּלְתִּי — לְחוֹבָתִי, הִתְפַּלַּלְתִּי — לְנִדְבָתִי״) ; עַל הַסָּפֵק אֵין צָרִיךְ חִדּוּשׁ. יסוד : שוע״ר ק״ז סעיף א.
  • ② בָּרִי + חִדּוּשׁ. בָּרִי לוֹ שֶׁהִתְפַּלֵּל וּמִתְפַּלֵּל נְדָבָה — חִדּוּשׁ מֵעֵין הַבְּרָכָה בִּבְרָכָה אֶמְצָעִית (אֲפִלּוּ בְּאַחַת דַּיּוֹ). יסוד : שוע״ר ק״ז סעיפים א, ג.
  • ③ מוּסָף וְשַׁבָּת. אֵין נְדָבָה בְּמוּסָף (אֲפִלּוּ רֹאשׁ חֹדֶשׁ), וְלֹא בְּשַׁבָּת / יוֹם טוֹב (אֵין נְדָרִים וּנְדָבוֹת). יסוד : שוע״ר ק״ז סעיף א.
  • ④ הִתְחִיל וְנִזְכַּר. הִתְחִיל לְשֵׁם חוֹבָה וְנִזְכַּר שֶׁהִתְפַּלֵּל — פּוֹסֵק אֲפִלּוּ בְּאֶמְצַע בְּרָכָה ; אֵין לַהֲפֹךְ לִנְדָבָה. יסוד : שוע״ר ק״ז סעיף ב.
  • ⑤ צִבּוּר / כַּוָּנָה. אֵין צִבּוּר מִתְפַּלֵּל נְדָבָה ; הַיָּחִיד כֵּן, אֲפִלּוּ בְּצִבּוּר, אַךְ רַק אִם בָּטוּחַ שֶׁיְּכַוֵּן מֵרֹאשׁ וְעַד סוֹף. יסוד : שוע״ר ק״ז סעיפים ד-ה.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק״ז (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַחוֹזֵר וּמַתְנֶה, הַחִדּוּשׁ, הַכַּוָּנָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
