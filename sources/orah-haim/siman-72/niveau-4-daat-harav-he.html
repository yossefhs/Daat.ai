דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ע״ב · דִּין נוֹשְׂאֵי הַמִּטָּה וְהַמְלַוִּין

דרכו של אדמו״ר הזקן בדִּין נוֹשְׂאֵי הַמִּטָּה וְהַמְלַוִּין : נוֹשְׂאֵי הַמִּטָּה — פְּטוּרִים, וּשְׁאָר הַמְלַוִּין — חַיָּבִים ; אֵין מוֹצִיאִין אֶת הַמֵּת סָמוּךְ לִזְמַן ק״ש ; הָעוֹסְקִים בְּהֶסְפֵּד — נִשְׁמָטִין אֶחָד אֶחָד ; וּשׁוּרַת הַמְנַחֲמִים — הַפְּנִימִים פָּטוּר וְהַחִיצוֹנִים חַיָּבִים — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ע״ב — דִּין נוֹשְׂאֵי הַמִּטָּה וְהַמְלַוִּין. אצל הרב הסימן נפרש לה׳ סְעִיפִים : נוֹשְׂאֵי הַמִּטָּה הַפְּטוּרִים (עוֹסֵק בְּמִצְוָה פָּטוּר) וּשְׁאָר הַמְלַוִּין הַחַיָּבִים, אִסּוּר הוֹצָאַת הַמֵּת סָמוּךְ לִזְמַן ק״ש, הַהֶסְפֵּד וְהַאֲנִינוּת, סֵדֶר הַתַּנְחוּמִין, וְשׁוּרַת הַמְנַחֲמִים עִם הַחִלּוּק בֵּין פְּנִימִים לַחִיצוֹנִים.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ע״ב — ה׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ע״ב). אדמו״ר הזקן דן בו בְּדִין נוֹשְׂאֵי הַמִּטָּה וְהַמְנַחֲמִים וְהַמְלַוִּין. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ע״ב — דִּין נוֹשְׂאֵי הַמִּטָּה וְהַמְלַוִּין

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ע״ב — דִּין נוֹשְׂאֵי הַמִּטָּה וְהַמְלַוִּין — וּבוֹ ה׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.72), ואחריו תרגומנו. הסימן כולל ה׳ סעיפים (סעיף k → ‎.72.k).

סעיף א נוֹשְׂאֵי הַמִּטָּה וְחִלּוּפֵיהֶם — פְּטוּרִים ; שְׁאָר הַמְלַוִּין — חַיָּבִים.

דִּין נוֹשְׂאֵי הַמִּטָּה וְהַמְנַחֲמִים וְהַמְלַוִּין וּבוֹ ה׳ סְעִיפִים:נוֹשְׂאֵי הַמִּטָּה וְחִלּוּפֵיהֶם וְחִלּוּפֵי חִלּוּפֵיהֶם, בֵּין אוֹתָם שֶׁלִּפְנֵי הַמִּטָּה בֵּין אוֹתָם שֶׁלְּאַחֲרֶיהָ, מֵאַחַר שֶׁיֵּשׁ לַמִּטָּה צֹרֶךְ בָּהֶם לָשֵׂאת אוֹתָהּ, כְּגוֹן בְּעִיר שֶׁיֵּשׁ בָּהּ חֲבוּרָה הַמְזֻמֶּנֶת לְכָךְ, וְכָל בְּנֵי הַחֲבוּרָה מַחֲלִיפִין לָשֵׂאת לְפִי שֶׁכֻּלָּם חֲפֵצִים לִזְכּוֹת בּוֹ — כֻּלָּם פְּטוּרִים מִקְּרִיאַת שְׁמַע (לְפִי שֶׁהֵם טְרוּדִים טִרְדַּת מִצְוָה). אֲבָל שְׁאָר הַמְלַוִּין אֶת הַמֵּת שֶׁאֵין לַמִּטָּה צֹרֶךְ בָּהֶן — חַיָּבִים בִּקְרִיאַת שְׁמַע (וְאַף עַל פִּי שֶׁהַלְוָיַת הַמֵּת הוּא בִּכְלַל גְּמִילוּת חֶסֶד, וּגְמִילוּת חֶסֶד הוּא מִן הַתּוֹרָה, מִכָּל מָקוֹם הֵם אֵינָם טְרוּדִים כְּלָל וִיכוֹלִין לִקְרוֹת פָּסוּק רִאשׁוֹן בְּכַוָּנָה וּבַעֲמִידָה וְהַשְּׁאָר כְּשֶׁהֵם מְהַלְּכִין), אֲבָל פְּטוּרִים מֵהַתְּפִלָּה כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ק״ו:

נוֹשְׂאֵי הַמִּטָּה. נוֹשְׂאֵי הַמִּטָּה וְחִלּוּפֵיהֶם וְחִלּוּפֵי חִלּוּפֵיהֶם, בֵּין שֶׁלִּפְנֵי הַמִּטָּה בֵּין שֶׁלְּאַחֲרֶיהָ, שֶׁיֵּשׁ לַמִּטָּה צֹרֶךְ בָּהֶם — פְּטוּרִים מִק״ש (טְרוּדִים טִרְדַּת מִצְוָה). אֲבָל שְׁאָר הַמְלַוִּין שֶׁאֵין לַמִּטָּה צֹרֶךְ בָּהֶם — חַיָּבִים : אַף שֶׁהַלְוָיָה בִּכְלַל גְּמִילוּת חֶסֶד, אֵינָם טְרוּדִים וִיכוֹלִין לִקְרוֹת פָּסוּק רִאשׁוֹן בְּכַוָּנָה וּבַעֲמִידָה וְהַשְּׁאָר כְּשֶׁמְּהַלְּכִין ; פְּטוּרִים מֵהַתְּפִלָּה (סימן ק״ו).

סעיף ב אֵין מוֹצִיאִין אֶת הַמֵּת סָמוּךְ לִזְמַן ק״ש ; הִתְחִילוּ — אֵין מַפְסִיקִין.

אֵין מוֹצִיאִין אֶת הַמֵּת סָמוּךְ לְהַתְחָלַת זְמַן קְרִיאַת שְׁמַע, בָּעִנְיָן שֶׁאֵין שְׁהוּת לְהוֹצִיאוֹ וּלְקָבְרוֹ קֹדֶם שֶׁיַּגִּיעַ זְמַן קְרִיאַת שְׁמַע, שֶׁמָּא יִתְעַכְּבוּ בִּקְבוּרָתוֹ עַד שֶׁיַּעֲבֹר זְמַנָּהּ. וְאִם הִתְחִילוּ לְהוֹצִיאוֹ (אֲפִלּוּ לְאַחַר שֶׁהִגִּיעַ זְמַן קְרִיאַת שְׁמַע) — אֵין מַפְסִיקִין כְּדֵי לִקְרוֹת, אֶלָּא אוֹתָם שֶׁאֵין לַמִּטָּה צֹרֶךְ בָּהֶם בִּלְבַד. בַּמֶּה דְּבָרִים אַמוּרִים? בִּקְרִיאַת שְׁמַע שֶׁל שַׁחֲרִית, אֲבָל סָמוּךְ לִזְמַן קְרִיאַת שְׁמַע שֶׁל עַרְבִית — מֻתָּר לְהוֹצִיאוֹ אַף עַל פִּי שֶׁאֵין שְׁהוּת לְקָבְרוֹ קֹדֶם זְמַן קְרִיאַת שְׁמַע, לְפִי שֶׁאִם לֹא יִקְבְּרֶנּוּ הַיּוֹם — יַעֲבֹר עַל ״לֹא תָלִין״ (וְאַף אִם יוּכַל בַּלַּיְלָה לְקָבְרוֹ — יַעֲבֹר עַל מִּצְוַת עֲשֵׂה ״כִּי קָבוֹר תִּקְבְּרֶנּוּ בַּיּוֹם הַהוּא״ אִם לֹא יִקְבְּרֶנּוּ קֹדֶם שְׁקִיעַת הַחַמָּה). וּמִכָּל מָקוֹם אִם הוּא מִפְּלַג הַמִּנְחָה וּלְמַעְלָה שֶׁיֵּשׁ אוֹמְרִים שֶׁהוּא זְמַן קְרִיאַת שְׁמַע כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן רל״ה — טוֹב לָחוּשׁ לְדִבְרֵיהֶם וְלִקְרוֹת קְרִיאַת שְׁמַע תְּחִלָּה וּלְקָבְרוֹ אַחַר כָּךְ (כֵּיוָן שֶׁרֹב הָעוֹלָם נוֹהֲגִים כִּסְבָרָא זוֹ, אַף עַל פִּי שֶׁהָעִקָּר כְּדִבְרֵי הַחֹלקִים כְּמוֹ שֶׁיִּתְבָּאֵר שָׁם):

אֵין מוֹצִיאִין אֶת הַמֵּת. אֵין מוֹצִיאִין סָמוּךְ לְהַתְחָלַת זְמַן ק״ש כְּשֶׁאֵין שְׁהוּת לְהוֹצִיאוֹ וּלְקָבְרוֹ קֹדֶם הַזְּמַן, שֶׁמָּא יִתְעַכְּבוּ. הִתְחִילוּ (אֲפִלּוּ אַחַר שֶׁהִגִּיעַ הַזְּמַן) — אֵין מַפְסִיקִין, אֶלָּא אוֹתָם שֶׁאֵין לַמִּטָּה צֹרֶךְ בָּהֶם. בְּמֶה דְּבָרִים ? בְּשַׁחֲרִית ; אֲבָל סָמוּךְ לְעַרְבִית — מֻתָּר לְהוֹצִיאוֹ, שֶׁאִם לֹא יִקְבְּרֶנּוּ הַיּוֹם יַעֲבֹר עַל ״לֹא תָלִין״. וּמִפְּלַג הַמִּנְחָה — יֵשׁ אוֹמְרִים שֶׁהוּא זְמַן ק״ש (סימן רל״ה), וְטוֹב לָחוּשׁ לִקְרוֹת תְּחִלָּה וּלְקָבְרוֹ אַחַר כָּךְ.

סעיף ג הָעוֹסְקִים בְּהֶסְפֵּד — נִשְׁמָטִין אֶחָד אֶחָד וְקוֹרִין ; הָאוֹנֵן דּוֹמֵם.

הָעָם הָעוֹסְקִים בְּהֶסְפֵּד בִּזְמַן שֶׁהַמֵּת מֻטָּל לִפְנֵיהֶם — נִשְׁמָטִין אֶחָד אֶחָד וְקוֹרִין וּמִתְפַּלְלִים, שֶׁאִי אֶפְשָׁר שֶׁיִּקְרְאוּ כֻּלָּם כָּאן אֵצֶל הַמֵּת מִשּׁוּם ״לֹעֵג לָרָשׁ״. אֵין הַמֵּת מֻטָּל לִפְנֵיהֶם אֶלָּא בְּבַיִת אַחֵר — הֵם קוֹרְאִין וּמִתְפַּלְלִין, וְהָאוֹנֵן יוֹשֵׁב וְדוֹמֵם וְאֵין עוֹנֶה אַחֲרֵיהֶם כְּלוּם, מִטַּעַם שֶׁנִּתְבָּאֵר בְּסִימָן ע״א:

הָעוֹסְקִים בְּהֶסְפֵּד. הָעוֹסְקִים בְּהֶסְפֵּד בִּזְמַן שֶׁהַמֵּת מֻטָּל לִפְנֵיהֶם — נִשְׁמָטִין אֶחָד אֶחָד וְקוֹרִין וּמִתְפַּלְלִים, שֶׁאִי אֶפְשָׁר לִקְרוֹת כֻּלָּם אֵצֶל הַמֵּת מִשּׁוּם ״לֹעֵג לָרָשׁ״. אֵין הַמֵּת לִפְנֵיהֶם אֶלָּא בְּבַיִת אַחֵר — קוֹרְאִין וּמִתְפַּלְלִין בִּמְקוֹמָם, וְהָאוֹנֵן יוֹשֵׁב וְדוֹמֵם וְאֵין עוֹנֶה אַחֲרֵיהֶם, מִטַּעַם שֶׁנִּתְבָּאֵר בְּסימן ע״א.

סעיף ד חָזְרוּ לְקַבֵּל תַּנְחוּמִין — אִם יְכוֹלִים לִגְמֹר פָּסוּק קֹדֶם הַשּׁוּרָה — יַתְחִילוּ.

קָבְרוּ אֶת הַמֵּת וְחָזְרוּ הָאֲבֵלִים לְקַבֵּל תַּנְחוּמִין וְכָל הָעָם הוֹלְכִים אַחֲרֵיהֶם מִמְּקוֹם הַקֶּבֶר לַמָּקוֹם שֶׁעוֹמְדִים שָׁם הָאֲבֵלִים לַעֲשׂוֹת שׁוּרָה לְקַבֵּל תַּנְחוּמִין, אִם יְכוֹלִים הָעָם לְהַתְחִיל וְלִגְמֹר אֲפִלּוּ פֶּרֶק אֶחָד וַאֲפִלּוּ פָּסוּק אֶחָד עַד שֶׁלֹּא יַגִּיעוּ לַשּׁוּרָה — יַתְחִילוּ וְיִקְרְאוּ כָּל מַה שֶּׁיּוּכְלוּ. וְאִם לָאו — לֹא יַתְחִילוּ אִם יֵשׁ שְׁהוּת לִקְרֹא אַחַר כָּךְ. אֲבָל אִם יַעֲבֹר זְמַן קְרִיאַת שְׁמַע — יִקְרְאוּ קְרִיאַת שְׁמַע תְּחִלָּה, כֵּיוָן שֶׁעֲדַיִן לֹא הִתְחִילוּ בְּמִצְוַת תַּנְחוּמִין:

חָזְרוּ לְקַבֵּל תַּנְחוּמִין. קָבְרוּ אֶת הַמֵּת וְחָזְרוּ הָאֲבֵלִים לְקַבֵּל תַּנְחוּמִין וְכָל הָעָם הוֹלְכִים אַחֲרֵיהֶם לַעֲשׂוֹת שׁוּרָה : אִם יְכוֹלִים לְהַתְחִיל וְלִגְמֹר פֶּרֶק אוֹ אֲפִלּוּ פָּסוּק קֹדֶם שֶׁיַּגִּיעוּ לַשּׁוּרָה — יַתְחִילוּ וְיִקְרְאוּ כָּל שֶׁיּוּכְלוּ. וְאִם לָאו — לֹא יַתְחִילוּ, אִם יֵשׁ שְׁהוּת אַחַר כָּךְ. אֲבָל אִם יַעֲבֹר הַזְּמַן — קוֹרְאִין ק״ש תְּחִלָּה, שֶׁעֲדַיִן לֹא הִתְחִילוּ בְּמִצְוַת תַּנְחוּמִין.

סעיף ה הָעוֹמְדִים בַּשּׁוּרָה — הַפְּנִימִים הָרוֹאִים פְּנֵי הָאָבֵל פְּטוּרִים, הַחִיצוֹנִים חַיָּבִים.

הָעוֹמְדִים בַּשּׁוּרָה לְנַחֲמוֹ, הַפְּנִימִיִּם הָרוֹאִים פְּנֵי הָאָבֵל — פְּטוּרִים, וְהַחִיצוֹנִים שֶׁאֵינָן רוֹאִים פְּנֵי הָאָבֵל — חַיָּבִים:

הָעוֹמְדִים בַּשּׁוּרָה. הָעוֹמְדִים בַּשּׁוּרָה לְנַחֲמוֹ : הַפְּנִימִים הָרוֹאִים פְּנֵי הָאָבֵל, שֶׁמְּקַיְּמִים בּוֹ נֶחָמָה — פְּטוּרִים ; וְהַחִיצוֹנִים שֶׁאֵינָם רוֹאִים פְּנֵי הָאָבֵל — חַיָּבִים בְּק״ש.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ע״ב (מהדורת קה״ת, ה׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין נוֹשְׂאֵי הַמִּטָּה וְהַמְלַוִּין

היסוד : נוֹשְׂאֵי הַמִּטָּה, הַמְלַוִּין וְק״ש

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה יְסוֹד עוֹסֵק בְּמִצְוָה פָּטוּר מִן הַמִּצְוָה — נוֹשְׂאֵי הַמִּטָּה פְּטוּרִים, שְׁאָר הַמְלַוִּין חַיָּבִים. אחר כך אִסּוּר הוֹצָאַת הַמֵּת סָמוּךְ לִזְמַן ק״ש. אחר כך הַהֶסְפֵּד וְהַאֲנִינוּת. ולבסוף סֵדֶר הַתַּנְחוּמִין וְשׁוּרַת הַמְנַחֲמִים (פְּנִימִים / חִיצוֹנִים).

עוֹסֵק בְּמִצְוָה פָּטוּר — נוֹשְׂאֵי הַמִּטָּה

עוֹסֵק בְּמִצְוָה פָּטוּר — נוֹשְׂאֵי הַמִּטָּה

היסוד. נוֹשְׂאֵי הַמִּטָּה וְחִלּוּפֵיהֶם שֶׁיֵּשׁ לַמִּטָּה צֹרֶךְ בָּהֶם — פְּטוּרִים מִק״ש (טְרוּדִים טִרְדַּת מִצְוָה) ; שְׁאָר הַמְלַוִּין — חַיָּבִים (סעיף א).

המסקנה. זֶהוּ יְסוֹד עוֹסֵק בְּמִצְוָה פָּטוּר מִן הַמִּצְוָה : רַק הַטִּרְדָּה הַמַּמָּשִׁית פּוֹטֶרֶת. עיין סימן ע״א · 71 (מי שמתו מוטל לפניו).

אֵין מוֹצִיאִין סָמוּךְ לַזְּמַן

אֵין מוֹצִיאִין סָמוּךְ לַזְּמַן

היסוד. אֵין מוֹצִיאִין אֶת הַמֵּת סָמוּךְ לִזְמַן ק״ש דְּשַׁחֲרִית, שֶׁמָּא יִתְעַכְּבוּ ; הִתְחִילוּ — אֵין מַפְסִיקִין (סעיף ב).

המסקנה. הָרַב מְחַלֵּק בֵּין שַׁחֲרִית לְעַרְבִית (מֻתָּר, מִשּׁוּם ״לֹא תָלִין״) וְחוֹשֵׁשׁ לִפְלַג הַמִּנְחָה. עיין סימן ע׳ · 70 (מי חייב בק״ש).

הֶסְפֵּד וְאֲנִינוּת — נִשְׁמָטִין אֶחָד אֶחָד

הֶסְפֵּד וְאֲנִינוּת — נִשְׁמָטִין אֶחָד אֶחָד

היסוד. בַּהֶסְפֵּד, כְּשֶׁהַמֵּת מֻטָּל לִפְנֵיהֶם — נִשְׁמָטִין אֶחָד אֶחָד וְקוֹרִין (מִשּׁוּם ״לֹעֵג לָרָשׁ״) ; אֵינוֹ לִפְנֵיהֶם — קוֹרְאִין, וְהָאוֹנֵן דּוֹמֵם (סעיף ג).

המסקנה. כְּבוֹד הַמֵּת (״לֹעֵג לָרָשׁ״) וּמַצַּב הָאוֹנֵן קוֹבְעִים אֶת הַהַנְהָגָה. עיין סימן ע״א · 71 (דין האונן).

תַּנְחוּמִין וְשׁוּרַת הַמְנַחֲמִים

תַּנְחוּמִין וְשׁוּרַת הַמְנַחֲמִים

היסוד. קֹדֶם הַשּׁוּרָה — אִם יְכוֹלִים לִגְמֹר פָּסוּק, יַתְחִילוּ ; וְאִם לָאו — לֹא (אֶלָּא אִם יַעֲבֹר הַזְּמַן) (סעיף ד). בַּשּׁוּרָה : פְּנִימִים הָרוֹאִים — פְּטוּרִים, חִיצוֹנִים — חַיָּבִים (סעיף ה).

המסקנה. רַק הַמְקַיֵּם נֶחָמָה בְּפֹעַל (רוֹאֶה פְּנֵי הָאָבֵל) פָּטוּר. עיין סימן ע״ג · 73.

הפניה ללימוד. היקף הסימן — עוֹסֵק בְּמִצְוָה, שְׁעַת הַהוֹצָאָה, הַהֶסְפֵּד וְהַשּׁוּרָה — נוגע למצבי אבלות וקהילה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין נוֹשְׂאֵי הַמִּטָּה וְהַמְלַוִּין — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ע״ב. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מה׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
נוֹשְׂאֵי הַמִּטָּה וְהַמְלַוִּין או״ח ע״ב — נוֹשְׂאֵי הַמִּטָּה וְחִלּוּפֵיהֶן פְּטוּרִים ; שְׁאָר הַמְלַוִּין שֶׁאֵין לַמִּטָּה צֹרֶךְ בָּהֶם חַיָּבִים. מ״ב ע״ב — גֶּדֶר ״עוֹסֵק בְּמִצְוָה״ וְקוֹרֵא פָּסוּק רִאשׁוֹן בַּעֲמִידָה. ערוה״ש ע״ב — טַעַם הַפְּטוֹר וְחִיּוּב הַמְלַוִּין. דרך אדמו״ר הזקן — נוֹשְׂאֵי הַמִּטָּה וְחִלּוּפֵיהֶם פְּטוּרִים (טְרוּדִים טִרְדַּת מִצְוָה) ; שְׁאָר הַמְלַוִּין חַיָּבִים, פָּסוּק רִאשׁוֹן בַּעֲמִידָה, פְּטוּרִים מֵהַתְּפִלָּה (סעיף א).
הוֹצָאַת הַמֵּת סָמוּךְ לַזְּמַן או״ח ע״ב — אֵין מוֹצִיאִין סָמוּךְ לְק״ש ; הִתְחִילוּ — אֵין מַפְסִיקִין. מ״ב ע״ב — חִלּוּק שַׁחֲרִית וְעַרְבִית וְטַעַם ״לֹא תָלִין״. ערוה״ש ע״ב — פְּרָטֵי הַזְּמַן וְהַקְּבוּרָה. דרך אדמו״ר הזקן — סָמוּךְ לְשַׁחֲרִית אֵין מוֹצִיאִין ; לְעַרְבִית מֻתָּר (״לֹא תָלִין״) ; מִפְּלַג הַמִּנְחָה טוֹב לָחוּשׁ לִקְרוֹת תְּחִלָּה (סימן רל״ה) (סעיף ב).
הֶסְפֵּד וְאֲנִינוּת או״ח ע״ב — הָעוֹסְקִים בְּהֶסְפֵּד נִשְׁמָטִין אֶחָד אֶחָד ; הָאוֹנֵן יוֹשֵׁב וְדוֹמֵם. מ״ב ע״ב — טַעַם ״לֹעֵג לָרָשׁ״ וּמַצַּב הָאוֹנֵן. ערוה״ש ע״ב — דִּין הָעוֹסֵק לִפְנֵי הַמֵּת. דרך אדמו״ר הזקן — מֻטָּל לִפְנֵיהֶם נִשְׁמָטִין אֶחָד אֶחָד (״לֹעֵג לָרָשׁ״) ; בְּבַיִת אַחֵר קוֹרְאִין, וְהָאוֹנֵן דּוֹמֵם (סימן ע״א) (סעיף ג).
תַּנְחוּמִין וְשׁוּרַת הַמְנַחֲמִים או״ח ע״ב — אִם יְכוֹלִים לִגְמֹר פָּסוּק קֹדֶם הַשּׁוּרָה יַתְחִילוּ ; פְּנִימִים פְּטוּרִים, חִיצוֹנִים חַיָּבִים. מ״ב ע״ב — גֶּדֶר ״רוֹאֶה פְּנֵי הָאָבֵל״ וְהַנֶּחָמָה. ערוה״ש ע״ב — הַחִלּוּק בֵּין פְּנִימִים לַחִיצוֹנִים. דרך אדמו״ר הזקן — קֹדֶם הַשּׁוּרָה, פֶּרֶק אוֹ פָּסוּק — יַתְחִילוּ ; יַעֲבֹר הַזְּמַן — ק״ש תְּחִלָּה ; פְּנִימִים הָרוֹאִים פְּטוּרִים, חִיצוֹנִים חַיָּבִים (סעיפים ד-ה).

הטבלה נערכה מטקסט המחבר (אורח חיים ע״ב, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ד, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות י״ז-י״ט). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ה׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — עוֹסֵק בְּמִצְוָה פָּטוּר

קו א — עוֹסֵק בְּמִצְוָה פָּטוּר

יסוד הסימן : נוֹשְׂאֵי הַמִּטָּה וְחִלּוּפֵיהֶם — פְּטוּרִים ; שְׁאָר הַמְלַוִּין — חַיָּבִים (סעיף א).

קריאת הרב : רַק הַטִּרְדָּה הַמַּמָּשִׁית (טְרוּדִים טִרְדַּת מִצְוָה) פּוֹטֶרֶת ; הַמְלַוֶּה נִשְׁאָר חַיָּב, פָּסוּק רִאשׁוֹן בַּעֲמִידָה. עיין סימן ע״א · 71.

מסקנה למעשה : הַנּוֹשֵׂא אוֹ הַמַּחֲלִיף פָּטוּר ; הַהוֹלֵךְ בְּלֹא צֹרֶךְ קוֹרֵא ק״ש.

קו ב — סָמוּךְ לַזְּמַן וְהַהוֹצָאָה

קו ב — סָמוּךְ לַזְּמַן וְהַהוֹצָאָה

יסוד הסימן : אֵין מוֹצִיאִין סָמוּךְ לְשַׁחֲרִית ; הִתְחִילוּ — אֵין מַפְסִיקִין ; עַרְבִית — מֻתָּר (סעיף ב).

קריאת הרב : אִזּוּן בֵּין זְמַן ק״ש לְ״לֹא תָלִין״ ; חֲשַׁשׁ לִפְלַג הַמִּנְחָה (סימן רל״ה). עיין סימן ע׳ · 70.

מסקנה למעשה : אִם יַעֲבֹר הַזְּמַן — קוֹרֵא ק״ש קֹדֶם הַקְּבוּרָה.

קו ג — הַפְּנִימִים וְהַחִיצוֹנִים

קו ג — הַפְּנִימִים וְהַחִיצוֹנִים

יסוד הסימן : הַפְּנִימִים הָרוֹאִים פְּנֵי הָאָבֵל — פְּטוּרִים ; הַחִיצוֹנִים — חַיָּבִים (סעיף ה).

קריאת הרב : הַמַּדָּד הוּא הַנֶּחָמָה בְּפֹעַל (רוֹאֶה פְּנֵי הָאָבֵל) ; בַּשּׁוּרָה, פֶּרֶק אוֹ פָּסוּק קֹדֶם. עיין סימן ע״ג · 73.

מסקנה למעשה : מִי שֶׁאֵינוֹ רוֹאֶה אֶת הָאָבֵל קוֹרֵא ק״ש בְּתוֹרוֹ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ע״ב ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ע״ב (דִּין נוֹשְׂאֵי הַמִּטָּה וְהַמְלַוִּין)

  • ① נוֹשְׂאֵי הַמִּטָּה. הַנּוֹשֵׂא אֶת הַמִּטָּה, חִלּוּפוֹ, אוֹ מִי שֶׁיִּצְטָרְכוּ לוֹ — פָּטוּר מִקְּרִיאַת שְׁמַע (טְרוּדִים טִרְדַּת מִצְוָה). יסוד : שוע״ר ע״ב סעיף א.
  • ② שְׁאָר הַמְלַוִּין. הַמְלַוֶּה שֶׁאֵין לַמִּטָּה צֹרֶךְ בּוֹ — חַיָּב : פָּסוּק רִאשׁוֹן בְּכַוָּנָה וּבַעֲמִידָה, וְהַשְּׁאָר בַּהֲלִיכָה ; פָּטוּר מֵהַתְּפִלָּה. יסוד : שוע״ר ע״ב סעיף א.
  • ③ הוֹצָאַת הַמֵּת. אֵין מוֹצִיאִין סָמוּךְ לִזְמַן ק״ש דְּשַׁחֲרִית ; הִתְחִילוּ — אֵין מַפְסִיקִין. בָּעֶרֶב — מֻתָּר (״לֹא תָלִין״). יסוד : שוע״ר ע״ב סעיף ב.
  • ④ הֶסְפֵּד וְאֲנִינוּת. בַּהֶסְפֵּד לִפְנֵי הַמֵּת — נִשְׁמָטִין אֶחָד אֶחָד וְקוֹרִין ; הָאוֹנֵן יוֹשֵׁב וְדוֹמֵם. יסוד : שוע״ר ע״ב סעיף ג.
  • ⑤ שׁוּרַת הַמְנַחֲמִים. קֹדֶם הַשּׁוּרָה — אִם יְכוֹלִים לִגְמֹר פָּסוּק, יַתְחִילוּ ; בַּשּׁוּרָה — פְּנִימִים הָרוֹאִים פְּטוּרִים, חִיצוֹנִים חַיָּבִים. יסוד : שוע״ר ע״ב סעיפים ד-ה.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ע״ב (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — עוֹסֵק בְּמִצְוָה, הַהוֹצָאָה, הַהֶסְפֵּד, הַשּׁוּרָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
