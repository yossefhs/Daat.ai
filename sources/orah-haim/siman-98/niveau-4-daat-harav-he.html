דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן צ״ח · כַּוָּנַת הַתְּפִלָּה

דרכו של אדמו״ר הזקן בכַּוָּנַת הַתְּפִלָּה : פֵּרוּשׁ הַמִּלּוֹת וְשְׁכִינָה כְּנֶגְדּוֹ ; הַמָּקוֹם וְהַשָּׁעָה הַמְבַטְּלִים כַּוָּנָה ; תַּחֲנוּנִים לֹא קֶבַע ; תְּפִלָּה בִּמְקוֹם קָרְבָּן וְהָעֲנָוָה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן צ״ח — צָרִיךְ שֶׁיִּהְיֶה לוֹ כַּוָּנָה בִּתְפִלָּתוֹ. אצל הרב הסימן נפרש לה׳ סְעִיפִים : הַמַּחֲשָׁבָה וְהַפֵּרוּשׁ הַמִּלּוֹת, הַמָּקוֹם וְהַשָּׁעָה הַמְבַטְּלִים כַּוָּנָה, הַתַּחֲנוּנִים לֹא קֶבַע, הַתְּפִלָּה בִּמְקוֹם קָרְבָּן, וְהָעֲנָוָה — מִי אֲנִי דַּל וְנִבְזֶה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן צ״ח — ה׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim צ״ח). אדמו״ר הזקן דן בו בְּכַוָּנַת הַתְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן צ״ח — צָרִיךְ שֶׁיִּהְיֶה לוֹ כַּוָּנָה בִּתְפִלָּתוֹ

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן צ״ח — כַּוָּנַת הַתְּפִלָּה — וּבוֹ ה׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.98), ואחריו תרגומנו. הסימן כולל ה׳ סעיפים (סעיף k → ‎.98.k).

סעיף א מַחֲשַׁבְתּוֹ כֵּיצַד — יְכַוֵּן פֵּרוּשׁ הַמִּלּוֹת כְּאִלּוּ שְׁכִינָה כְּנֶגְדּוֹ.

צָרִיךְ שֶׁיִּהְיֶה לוֹ כַּוָּנָה בִּתְפִלָּתוֹ וּבוֹ ה׳ סְעִיפִים:מַחֲשַׁבְתּוֹ כֵּיצַד? הַמִּתְפַּלֵּל צָרִיךְ שֶׁיְּכַוֵּן בְּלִבּוֹ פֵּרוּשׁ הַמִּלּוֹת שֶׁמּוֹצִיא בִּשְׂפָתָיו, שֶׁנֶּאֱמַר: ״תָּכִין לִבָּם תַּקְשִׁיב אָזְנֶךָ״. וְצָרִיךְ שֶׁיִּרְאֶה עַצְמוֹ כְּאִלּוּ שְׁכִינָה שְׁרוּיָה כְּנֶגְדוֹ. וְיָעִיר הַכַּוָּנָה וְיָסִיר כָּל הַמַּחֲשָׁבוֹת הַטּוֹרְדוֹת אוֹתוֹ, עַד שֶׁתִּשָּׁאֵר מַחֲשַׁבְתּוֹ וְכַוָּנָתוֹ זַכָּה בִּתְפִלָּתוֹ. וְיַחֲשֹׁב כִּי אִלּוּ הָיָה מְדַבֵּר לִפְנֵי מֶלֶךְ בָּשָׂר וָדָם הָיָה מְסַדֵּר דְּבָרָיו וּמְכַוֵּן בָּהֶם יָפֶה לְבַל יִכָּשֵׁל — קַל וָחֹמֶר לִפְנֵי מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁצָּרִיךְ לְכַוֵּן לְפָנָיו אַף מַחֲשַׁבְתּוֹ, כִּי לְפָנָיו הַמַּחֲשָׁבָה כְּדִבּוּר, כִּי כָּל הַמַּחֲשָׁבוֹת הוּא חוֹקֵר. וְכָךְ הָיוּ עוֹשִׂים חֲסִידִים וְאַנְשֵׁי מַעֲשֶׂה שֶׁהָיוּ מִתְבּוֹדְדִים וּמְכַוְּנִים בִּתְפִלָּתָם עַד שֶׁהָיוּ מַגִּיעִים לְהִתְפַּשְּׁטוּת הַגַּשְׁמִיּוּת וּלְהִתְגַּבְּרוּת רוּחַ הַשִּׂכְלִיּוּת עַד שֶׁהָיוּ מַגִּיעִים קָרוֹב לְמַעֲלַת הַנְּבוּאָה. וְאִם תָּבוֹא לוֹ מַחֲשָׁבָה אַחֶרֶת בְּתוֹךְ הַתְּפִלָּה — יִשְׁתֹּק עַד שֶׁתִּתְבַּטֵּל הַמַּחֲשָׁבָה. וְאִם אֵינוֹ יָכוֹל לְכַוֵּן פֵּרוּשׁ הַמִּלּוֹת — עַל כָּל פָּנִים צָרִיךְ שֶׁיַּחֲשֹׁב בִּשְׁעַת הַתְּפִלָּה בִּדְבָרִים הַמַּכְנִיעִים אֶת הַלֵּב וּמְכַוְּנִים לִבּוֹ לְאָבִיו שֶׁבַּשָּׁמַיִם. וְלֹא יַחְשֹׁב בִּדְבָרִים שֶׁיֵּשׁ בָּהֶם קַלּוּת רֹאשׁ. וְקֹדֶם הַתְּפִלָּה יַחֲשֹׁב מֵרוֹמְמוּת הָאֵל יִתְבָּרַךְ וְשִׁפְלוּת הָאָדָם, וְיָסִיר כָּל תַּעֲנוּגֵי אָדָם מִלִּבּוֹ. וְאֵין לְנַשֵּׁק בָּנָיו הַקְּטַנִּים בְּבֵית הַכְּנֶסֶת, כְּדֵי לִקְבֹּעַ בְּלִבּוֹ שֶׁאֵין אַהֲבָה כְּאַהֲבַת הַמָּקוֹם בָּרוּךְ הוּא. וִילָדִים הַמְבַלְבְּלִים אֲבוֹתֵיהֶם בַּתְּפִלָּה — אֵין לַהֲבִיאָם כְּלָל לְבֵית הַכְּנֶסֶת:

מַחֲשַׁבְתּוֹ כֵּיצַד ? הַמִּתְפַּלֵּל יְכַוֵּן בְּלִבּוֹ פֵּרוּשׁ הַמִּלּוֹת (״תָּכִין לִבָּם תַּקְשִׁיב אָזְנֶךָ״), וְיִרְאֶה עַצְמוֹ כְּאִלּוּ שְׁכִינָה כְּנֶגְדּוֹ, וְיָסִיר הַמַּחֲשָׁבוֹת הַטּוֹרְדוֹת עַד שֶׁתִּשָּׁאֵר מַחֲשַׁבְתּוֹ זַכָּה. כְּמוֹ לִפְנֵי מֶלֶךְ בָּשָׂר וָדָם — קַל וָחֹמֶר לִפְנֵי מֶלֶךְ מַלְכֵי הַמְּלָכִים, שֶׁהַמַּחֲשָׁבָה כְּדִבּוּר. וְכָךְ חֲסִידִים וְאַנְשֵׁי מַעֲשֶׂה עַד לְהִתְפַּשְּׁטוּת הַגַּשְׁמִיּוּת. מַחֲשָׁבָה אַחֶרֶת — יִשְׁתֹּק. קֹדֶם הַתְּפִלָּה — רוֹמְמוּת הָאֵל וְשִׁפְלוּת הָאָדָם. וְאֵין לְנַשֵּׁק בָּנָיו הַקְּטַנִּים בְּבֵית הַכְּנֶסֶת, וִילָדִים הַמְבַלְבְּלִים — אֵין לַהֲבִיאָם.

סעיף ב לֹא בְּמָקוֹם וְלֹא בְּשָׁעָה הַמְבַטְּלִים אֶת כַּוָּנָתוֹ.

לֹא יִתְפַּלֵּל לֹא בְּמָקוֹם שֶׁיֵּשׁ בּוֹ דָּבָר שֶׁמְּבַטֵּל כַּוָּנָתוֹ, כְּגוֹן בְּמָקוֹם שֶׁיֵּשׁ שֵׁכָר חָדָשׁ שֶׁמֵּרִיחַ, וְלֹא בְּשָׁעָה שֶׁמְּבַטֶּלֶת כַּוָּנָתוֹ, שֶׁאֵין דַּעְתּוֹ מְיֻשֶּׁבֶת עָלָיו מֵאֵיזֶה צַעַר אוֹ כַּעַס אוֹ מִטֹּרַח הַדֶּרֶךְ. וְעַכְשָׁו אֵין אָנוּ נִזְהָרִין בְּכָל זֶה, מִפְּנֵי שֶׁאֵין אָנוּ מְכַוְּנִים כָּל כָּךְ בַּתְּפִלָּה:

מָקוֹם וּשְׁעָה. לֹא יִתְפַּלֵּל בְּמָקוֹם שֶׁמְּבַטֵּל כַּוָּנָתוֹ (כְּגוֹן שֵׁכָר חָדָשׁ שֶׁמֵּרִיחַ), וְלֹא בְּשָׁעָה שֶׁמְּבַטֶּלֶת כַּוָּנָתוֹ — שֶׁאֵין דַּעְתּוֹ מְיֻשֶּׁבֶת מֵחֲמַת צַעַר, כַּעַס אוֹ טֹרַח הַדֶּרֶךְ. וְעַכְשָׁו אֵין אָנוּ נִזְהָרִין בְּכָל זֶה, שֶׁאֵין אָנוּ מְכַוְּנִים כָּל כָּךְ בַּתְּפִלָּה.

סעיף ג אַל יַעֲשֶׂה תְּפִלָּתוֹ קֶבַע — רַחֲמִים וְתַחֲנוּנִים.

אַל יַעֲשֶׂה תְּפִלָּתוֹ קֶבַע אֶלָּא רַחֲמִים וְתַחֲנוּנִים לִפְנֵי הַמָּקוֹם בָּרוּךְ הוּא. דְּהַיְנוּ, שֶׁלֹּא יִתְפַּלֵּל מִפְּנֵי הַחִיּוּב כְּדֵי לָצֵאת יְדֵי חוֹבָתוֹ, שֶׁאָז תְּפִלָּתוֹ דּוֹמָה עָלָיו כְּמַשּׂאוֹי וּמְבַקֵּשׁ לִפָּטֵר מִמֶּנָּה, אֶלָּא כְּמִי שֶׁצָּרִיךְ דָּבָר וּבָא לְבַקֵּשׁ מִלִּפְנֵי הַמֶּלֶךְ. וְגַם יֹאמְרֶנָּה בִּלְשׁוֹן תַּחֲנוּנִים, כְּרָשׁ הַמְבַקֵּשׁ בַּפֶּתַח:

תַּחֲנוּנִים לֹא קֶבַע. אַל יַעֲשֶׂה תְּפִלָּתוֹ קֶבַע אֶלָּא רַחֲמִים וְתַחֲנוּנִים לִפְנֵי הַמָּקוֹם ; שֶׁלֹּא יִתְפַּלֵּל מִפְּנֵי הַחִיּוּב לָצֵאת יְדֵי חוֹבָתוֹ — שֶׁאָז תְּפִלָּתוֹ כְּמַשּׂאוֹי — אֶלָּא כְּמִי שֶׁצָּרִיךְ דָּבָר וּבָא לְבַקֵּשׁ מִלִּפְנֵי הַמֶּלֶךְ. וְיֹאמְרֶנָּה בִּלְשׁוֹן תַּחֲנוּנִים, כְּרָשׁ הַמְבַקֵּשׁ בַּפֶּתַח.

סעיף ד הַתְּפִלָּה בִּמְקוֹם קָרְבָּן — מְעֻמָּד, קְבִיעוּת מָקוֹם, בְּלִי חֲצִיצָה.

הַתְּפִלָּה הִיא בִּמְקוֹם קָרְבָּן. וְלָכֵן צָרִיךְ לִזָּהֵר שֶׁתְּהֵא דֻּגְמַת הַקָּרְבָּן בְּכַוָּנָה, וְשֶׁלֹּא יְעָרֵב בָּהּ מַחֲשָׁבָה אַחֶרֶת כְּמוֹ מַחֲשָׁבָה שֶׁפּוֹסֶלֶת בַּקֳּדָשִׁים, וּתְהֵא מְעֻמָּד כְּמוֹ עֲבוֹדָה, וּקְבִיעוּת מָקוֹם כְּמוֹ הַקָּרְבָּנוֹת שֶׁכָּל אֶחָד קָבוּעַ מְקוֹמוֹ לִשְׁחִיטָתוֹ וּמַתַּן דָּמוֹ, וְשֶׁלֹּא יָחֹץ דָּבָר בֵּינוֹ לַקִּיר כְּמוֹ הַקָּרְבָּן שֶׁחֲצִיצָה פּוֹסֶלֶת בֵּינוֹ לַכְּלִי. וְרָאוּי שֶׁיִּהְיֶה לוֹ מַלְבּוּשִׁים נָאִים מְיֻחָדִים לִתְפִלָּה כְּמוֹ בִּגְדֵי כְּהֻנָּה, אֶלָּא שֶׁאֵין כָּל אָדָם יָכוֹל לְבַזְבֵּז עַל זֶה. וּמִכָּל מָקוֹם, טוֹב שֶׁיִּהְיוּ לוֹ מִכְנָסַיִם מְיֻחָדִים לִתְפִלָּה, מִשּׁוּם נְקִיּוּת (אַף עַל פִּי שֶׁמִּן הַדִּין אֵין צָרִיךְ נְקִיּוּת לִתְפִלָּה יוֹתֵר מִתַּלְמוּד תּוֹרָה, שֶׁאַף עַל פִּי שֶׁהַמִּכְנָסַיִם מְלֻכְלָכִים מִבִּפְנִים — מֻתָּר, כְּמוֹ שֶׁנִּתְבָּאֵר סִימָן ע״ו):

בִּמְקוֹם קָרְבָּן. הַתְּפִלָּה בִּמְקוֹם קָרְבָּן : דֻּגְמַת הַקָּרְבָּן בְּכַוָּנָה, בְּלִי מַחֲשָׁבָה אַחֶרֶת (כְּמַחֲשָׁבָה שֶׁפּוֹסֶלֶת בַּקֳּדָשִׁים), מְעֻמָּד כְּמוֹ עֲבוֹדָה, קְבִיעוּת מָקוֹם כְּמוֹ הַקָּרְבָּנוֹת, וְשֶׁלֹּא יָחֹץ דָּבָר בֵּינוֹ לַקִּיר (כַּחֲצִיצָה שֶׁפּוֹסֶלֶת). וְרָאוּי מַלְבּוּשִׁים נָאִים כְּמוֹ בִּגְדֵי כְּהֻנָּה — וְאֵין הַכֹּל יְכוֹלִים ; וּמִכָּל מָקוֹם טוֹב מִכְנָסַיִם מְיֻחָדִים מִשּׁוּם נְקִיּוּת (אַף שֶׁמִּן הַדִּין אֵין צָרִיךְ יוֹתֵר מִתַּלְמוּד תּוֹרָה, כְּמוֹ שֶׁנִּתְבָּאֵר סִימָן ע״ו).

סעיף ה אַל יַחְשֹׁב רָאוּי אֲנִי — אֶלָּא בְּחַסְדּוֹ ; מִי אֲנִי דַּל וְנִבְזֶה.

אַל יַחֲשֹׁב רָאוּי אֲנִי שֶׁיַּעֲשֶׂה הַקָּדוֹשׁ בָּרוּךְ הוּא בַּקָּשָׁתִי כֵּיוָן שֶׁכִּוַּנְתִּי בַּתְּפִלָּה, כִּי אַדְּרַבָּה זֶהוּ מַזְכִּיר עֲוֹנוֹתָיו שֶׁל אָדָם, שֶׁעַל יְדֵי כָּךְ מְפַשְׁפְּשִׁים בְּמַעֲשָׂיו לוֹמַר בָּטוּחַ הוּא בִּזְכֻיּוֹתָיו, אֶלָּא יַחֲשֹׁב שֶׁיַּעֲשֶׂה הַקָּדוֹשׁ בָּרוּךְ הוּא בְּחַסְדּוֹ, וְיֹאמַר בְּלִבּוֹ: מִי אֲנִי דַּל וְנִבְזֶה בָּא לְבַקֵּשׁ מֵאֵת מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא אִם לֹא מֵרֹב חֲסָדָיו שֶׁהוּא מִתְנַהֵג בָּהֶם עִם בְּרִיּוֹתָיו:

הָעֲנָוָה. אַל יַחְשֹׁב ״רָאוּי אֲנִי״ שֶׁיַּעֲשֶׂה הקב״ה בַּקָּשָׁתִי כֵּיוָן שֶׁכִּוַּנְתִּי — כִּי אַדְּרַבָּה זֶהוּ מַזְכִּיר עֲוֹנוֹתָיו (מְפַשְׁפְּשִׁים בְּמַעֲשָׂיו). אֶלָּא יַחֲשֹׁב שֶׁיַּעֲשֶׂה בְּחַסְדּוֹ, וְיֹאמַר בְּלִבּוֹ : מִי אֲנִי דַּל וְנִבְזֶה בָּא לְבַקֵּשׁ מֵאֵת מֶלֶךְ מַלְכֵי הַמְּלָכִים אִם לֹא מֵרֹב חֲסָדָיו עִם בְּרִיּוֹתָיו.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim צ״ח (מהדורת קה״ת, ה׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — כַּוָּנַת הַתְּפִלָּה

היסוד : כַּוָּנַת הַתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַפֵּרוּשׁ הַמִּלּוֹת וְהַשְּׁכִינָה כְּנֶגְדּוֹ. אחר כך הַמָּקוֹם וְהַשָּׁעָה הַמְבַטְּלִים כַּוָּנָה. אחר כך הַתַּחֲנוּנִים לֹא קֶבַע. ולבסוף הַתְּפִלָּה בִּמְקוֹם קָרְבָּן וְהָעֲנָוָה — מִי אֲנִי דַּל וְנִבְזֶה.

פֵּרוּשׁ הַמִּלּוֹת — שְׁכִינָה כְּנֶגְדּוֹ

פֵּרוּשׁ הַמִּלּוֹת — שְׁכִינָה כְּנֶגְדּוֹ

היסוד. יְכַוֵּן בְּלִבּוֹ פֵּרוּשׁ הַמִּלּוֹת ; יִרְאֶה עַצְמוֹ כְּאִלּוּ שְׁכִינָה כְּנֶגְדּוֹ, וְיָסִיר הַמַּחֲשָׁבוֹת הַטּוֹרְדוֹת עַד שֶׁתִּשָּׁאֵר מַחֲשַׁבְתּוֹ זַכָּה (סעיף א).

המסקנה. לִפְנֵי הַמֶּלֶךְ הַמַּחֲשָׁבָה כְּדִבּוּר ; מַחֲשָׁבָה זָרָה — יִשְׁתֹּק. עיין סימן 95 (כוונת הגוף בתפילה).

מָקוֹם וּשְׁעָה הַמְבַטְּלִים כַּוָּנָה

מָקוֹם וּשְׁעָה הַמְבַטְּלִים כַּוָּנָה

היסוד. לֹא יִתְפַּלֵּל בְּמָקוֹם (שֵׁכָר חָדָשׁ שֶׁמֵּרִיחַ) וְלֹא בְּשָׁעָה (צַעַר, כַּעַס, טֹרַח הַדֶּרֶךְ) הַמְבַטְּלִים כַּוָּנָתוֹ (סעיף ב).

המסקנה. וְעַכְשָׁו אֵין אָנוּ נִזְהָרִין בְּכָל זֶה — שֶׁאֵין הַכַּוָּנָה כְּשֶׁהָיְתָה. עיין סימן 96 (הסחת הדעת בתפילה).

תַּחֲנוּנִים לֹא קֶבַע

תַּחֲנוּנִים לֹא קֶבַע

היסוד. אַל יַעֲשֶׂה תְּפִלָּתוֹ קֶבַע אֶלָּא רַחֲמִים וְתַחֲנוּנִים ; לֹא מִפְּנֵי הַחִיּוּב (כְּמַשּׂאוֹי) אֶלָּא כְּמִי שֶׁצָּרִיךְ דָּבָר (סעיף ג).

המסקנה. יֹאמְרֶנָּה בִּלְשׁוֹן תַּחֲנוּנִים, כְּרָשׁ הַמְבַקֵּשׁ בַּפֶּתַח. עיין סימן 97 (דרך ארץ בתפילה).

תְּפִלָּה בִּמְקוֹם קָרְבָּן וְהָעֲנָוָה

תְּפִלָּה בִּמְקוֹם קָרְבָּן וְהָעֲנָוָה

היסוד. הַתְּפִלָּה בִּמְקוֹם קָרְבָּן — מְעֻמָּד, קְבִיעוּת מָקוֹם, בְּלִי חֲצִיצָה בֵּינוֹ לַקִּיר (סעיף ד).

המסקנה. אַל יַחְשֹׁב רָאוּי אֲנִי אֶלָּא בְּחַסְדּוֹ — מִי אֲנִי דַּל וְנִבְזֶה (סעיף ה). עיין סימן 90 (קביעת מקום בבית הכנסת).

הפניה ללימוד. היקף הסימן — הַפֵּרוּשׁ הַמִּלּוֹת, הַתַּחֲנוּנִים וְהָעֲנָוָה — נוגע ללב העֲבוֹדָה שֶׁבַּלֵּב. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

כַּוָּנַת הַתְּפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן צ״ח. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מה׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
פֵּרוּשׁ הַמִּלּוֹת וּשְׁכִינָה כְּנֶגְדּוֹ או״ח צ״ח — יְכַוֵּן פֵּרוּשׁ הַמִּלּוֹת ; כְּאִלּוּ שְׁכִינָה כְּנֶגְדּוֹ ; חֲסִידִים וְאַנְשֵׁי מַעֲשֶׂה. מ״ב צ״ח — עִנְיַן הַכַּוָּנָה וּפְרָטֶיהָ, וְ״תָּכִין לִבָּם״. ערוה״ש צ״ח — גֶּדֶר כַּוָּנַת הַלֵּב וַהֲסָרַת הַטְּרָדוֹת. דרך אדמו״ר הזקן — יְכַוֵּן פֵּרוּשׁ הַמִּלּוֹת ; הַמַּחֲשָׁבָה כְּדִבּוּר ; מַחֲשָׁבָה אַחֶרֶת — יִשְׁתֹּק ; וִילָדִים הַמְבַלְבְּלִים אֵין לַהֲבִיאָם (סעיף א).
מָקוֹם וּשְׁעָה הַמְבַטְּלִים או״ח צ״ח — לֹא בְּמָקוֹם וְלֹא בְּשָׁעָה הַמְבַטְּלִים כַּוָּנָה ; וְעַכְשָׁו אֵין נִזְהָרִין. מ״ב צ״ח — פְּרָטֵי הַמָּקוֹם וְהַשָּׁעָה שֶׁמְּבַטְּלִים. ערוה״ש צ״ח — יִשּׁוּב הַדַּעַת קֹדֶם הַתְּפִלָּה. דרך אדמו״ר הזקן — שֵׁכָר חָדָשׁ שֶׁמֵּרִיחַ ; צַעַר, כַּעַס, טֹרַח הַדֶּרֶךְ ; וְעַכְשָׁו אֵין אָנוּ נִזְהָרִין (סעיף ב).
תַּחֲנוּנִים לֹא קֶבַע או״ח צ״ח — דֶּרֶךְ תַּחֲנוּנִים, כְּרָשׁ הַמְבַקֵּשׁ בַּפֶּתַח, בְּנַחַת. מ״ב צ״ח — שֶׁלֹּא תְּהֵא עָלָיו כְּמַשָּׂא וּמְבַקֵּשׁ לִפָּטֵר. ערוה״ש צ״ד-צ״ח — עִנְיַן קֶבַע וְתַחֲנוּנִים בַּתְּפִלָּה. דרך אדמו״ר הזקן — לֹא מִפְּנֵי הַחִיּוּב (כְּמַשּׂאוֹי) אֶלָּא כְּמִי שֶׁצָּרִיךְ דָּבָר ; בִּלְשׁוֹן תַּחֲנוּנִים כְּרָשׁ בַּפֶּתַח (סעיף ג).
בִּמְקוֹם קָרְבָּן וְהָעֲנָוָה או״ח צ״ח — תְּפִלָּה בִּמְקוֹם קָרְבָּן ; אַל יַחְשֹׁב רָאוּי הוּא אֶלָּא בְּחַסְדּוֹ. מ״ב צ״ח — מְעֻמָּד, קְבִיעוּת מָקוֹם וּמַלְבּוּשִׁים נָאִים. ערוה״ש צ״ח — דִּמְיוֹן הַתְּפִלָּה לָעֲבוֹדָה וְהָעֲנָוָה. דרך אדמו״ר הזקן — מְעֻמָּד, קְבִיעוּת מָקוֹם, בְּלִי חֲצִיצָה לַקִּיר ; מִכְנָסַיִם מִשּׁוּם נְקִיּוּת ; מִי אֲנִי דַּל וְנִבְזֶה (סעיפים ד-ה).

הטבלה נערכה מטקסט המחבר (אורח חיים צ״ח, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפלה ד, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ל״א, ברכות ל״ד). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ה׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — כַּוָּנַת הַלֵּב

קו א — כַּוָּנַת הַלֵּב

יסוד הסימן : יְכַוֵּן פֵּרוּשׁ הַמִּלּוֹת ; כְּאִלּוּ שְׁכִינָה כְּנֶגְדּוֹ ; מַחֲשָׁבָה אַחֶרֶת — יִשְׁתֹּק (סעיף א).

קריאת הרב : לִפְנֵי הַמֶּלֶךְ הַמַּחֲשָׁבָה כְּדִבּוּר — הַכַּוָּנָה בַּמַּחֲשָׁבָה עַצְמָהּ, לֹא בַּשְּׂפָתַיִם בִּלְבַד. עיין סימן 95.

מסקנה למעשה : לְכַוֵּן תְּחִלָּה פֵּרוּשׁ הַמִּלּוֹת ; וְאִם לָאו — לַחֲשֹׁב בִּדְבָרִים הַמַּכְנִיעִים אֶת הַלֵּב.

קו ב — תַּחֲנוּנִים לֹא קֶבַע

קו ב — תַּחֲנוּנִים לֹא קֶבַע

יסוד הסימן : אַל יַעֲשֶׂה תְּפִלָּתוֹ קֶבַע ; לֹא מִפְּנֵי הַחִיּוּב אֶלָּא כְּמִי שֶׁצָּרִיךְ דָּבָר (סעיף ג).

קריאת הרב : הַתְּפִלָּה אֵינָהּ מַשָּׂא לִפָּטֵר מִמֶּנּוּ אֶלָּא בַּקָּשָׁה חַיָּה מִלִּפְנֵי הַמֶּלֶךְ. עיין סימן 97.

מסקנה למעשה : לוֹמַר הַתְּפִלָּה בִּלְשׁוֹן תַּחֲנוּנִים, כְּרָשׁ הַמְבַקֵּשׁ בַּפֶּתַח.

קו ג — בִּמְקוֹם קָרְבָּן וְהָעֲנָוָה

קו ג — בִּמְקוֹם קָרְבָּן וְהָעֲנָוָה

יסוד הסימן : הַתְּפִלָּה בִּמְקוֹם קָרְבָּן — מְעֻמָּד, קְבִיעוּת מָקוֹם, בְּלִי חֲצִיצָה ; אַל יַחְשֹׁב רָאוּי אֲנִי (סעיפים ד-ה).

קריאת הרב : כַּוָּנַת הַתְּפִלָּה כְּכַוָּנַת הַקָּרְבָּן ; הַהֵעָנוּת בָּאָה בְּחַסְדּוֹ, לֹא בִּזְכוּת. עיין סימן 90.

מסקנה למעשה : לַעֲמֹד מְעֻמָּד בְּמָקוֹם קָבוּעַ, וְלוֹמַר בְּלִבּוֹ מִי אֲנִי דַּל וְנִבְזֶה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן צ״ח ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן צ״ח (כַּוָּנַת הַתְּפִלָּה)

  • ① פֵּרוּשׁ הַמִּלּוֹת. לְכַוֵּן הַלֵּב עַל פֵּרוּשׁ הַמִּלּוֹת ; לִרְאוֹת עַצְמוֹ כְּאִלּוּ שְׁכִינָה כְּנֶגְדּוֹ ; מַחֲשָׁבָה זָרָה — יִשְׁתֹּק עַד שֶׁתִּתְבַּטֵּל. יסוד : שוע״ר צ״ח סעיף א.
  • ② מָקוֹם וּשְׁעָה. לְהִמָּנַע מִמָּקוֹם (שֵׁכָר חָדָשׁ שֶׁמֵּרִיחַ) וּמִשָּׁעָה (צַעַר, כַּעַס, טֹרַח הַדֶּרֶךְ) הַמְבַטְּלִים כַּוָּנָה. יסוד : שוע״ר צ״ח סעיף ב.
  • ③ תַּחֲנוּנִים לֹא קֶבַע. לֹא לְהִתְפַּלֵּל כְּמַשּׂאוֹי לָצֵאת יְדֵי חוֹבָה, אֶלָּא בִּלְשׁוֹן תַּחֲנוּנִים, כְּרָשׁ הַמְבַקֵּשׁ בַּפֶּתַח. יסוד : שוע״ר צ״ח סעיף ג.
  • ④ בִּמְקוֹם קָרְבָּן. לְהִתְפַּלֵּל מְעֻמָּד, בְּמָקוֹם קָבוּעַ, בְּלִי חֲצִיצָה בֵּינוֹ לַקִּיר ; מִכְנָסַיִם מְיֻחָדִים מִשּׁוּם נְקִיּוּת. יסוד : שוע״ר צ״ח סעיף ד.
  • ⑤ הָעֲנָוָה. לֹא לִסְמֹךְ עַל זְכוּתוֹ (רָאוּי אֲנִי) אֶלָּא עַל הַחֶסֶד ; לוֹמַר בְּלִבּוֹ מִי אֲנִי דַּל וְנִבְזֶה. יסוד : שוע״ר צ״ח סעיף ה.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן צ״ח (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַפֵּרוּשׁ הַמִּלּוֹת, הַתַּחֲנוּנִים, הָעֲנָוָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
