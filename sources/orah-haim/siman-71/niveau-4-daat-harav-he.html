דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ע״א · דִּין אָבֵל וְהָעוֹסְקִים בַּמֵּת

דרכו של אדמו״ר הזקן בדִּין אָבֵל וְהָעוֹסְקִים בַּמֵּת : הָאוֹנֵן — פָּטוּר מִקְּרִיאַת שְׁמַע וּתְפִלָּה עַד הַקְּבוּרָה ; בְּשַׁבָּת אֵין אֲנִינוּת — חַיָּב ; הַמְשַׁמֵּר וְהַחוֹפֵר — עוֹסֵק בְּמִצְוָה, פָּטוּר ; וְאָסוּר לִקְרוֹת תּוֹךְ ד׳ אַמּוֹת שֶׁל מֵת — לֹעֵג לָרָשׁ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ע״א — דִּין אָבֵל וְהָעוֹסְקִים בַּמֵּת. אצל הרב הסימן נפרש לו׳ סְעִיפִים : הָאֲנִינוּת הַפּוֹטֶרֶת אֶת הָאוֹנֵן, הַהֶבְדֵּל חֹל / שַׁבָּת / יוֹם טוֹב, הָעוֹסֵק בְּמִצְוָה (הַמְשַׁמֵּר וְהַחוֹפֵר), וְהַלֹעֵג לָרָשׁ בְּתוֹךְ ד׳ אַמּוֹת שֶׁל מֵת.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ע״א — ו׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ע״א). אדמו״ר הזקן דן בו בְּדִין אָבֵל וְהָעוֹסְקִים בַּמֵּת. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ע״א — דִּין אָבֵל וְהָעוֹסְקִים בַּמֵּת

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ע״א — דִּין אָבֵל וְהָעוֹסְקִים בַּמֵּת — וּבוֹ ו׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.71), ואחריו תרגומנו. הסימן כולל ו׳ סעיפים (סעיף k → ‎.71.k).

סעיף א אָבֵל וְאוֹנֵן פָּטוּר מִקְּרִיאַת שְׁמַע וּתְפִלָּה עַד הַקְּבוּרָה.

אָבֵל וְהָעוֹסְקִים בַּמֵּת פְּטוּרִים מִקְּרִיאַת שְׁמַע וּבוֹ ו׳ סְעִיפִים:מִי שֶׁמֵּת לוֹ מֵת שֶׁחַיָּב לְהִתְאַבֵּל עָלָיו, אֲפִלּוּ אֵינוֹ מֻטָּל עָלָיו לְקָבְרוֹ, כְּגוֹן שֶׁמֵּתָה אֲחוֹתוֹ נְשׂוּאָה שֶׁבַּעְלָהּ חַיָּב בִּקְבוּרָתָהּ, אוֹ אֲפִלּוּ יֵשׁ לוֹ מִי שֶׁיִּשְׁתַּדֵּל בִּשְׁבִילוֹ לַעֲסֹק בְּצָרְכֵי הַקְּבוּרָה, אוֹ אֲפִלּוּ כְּבָר עָסַק כָּל צְרָכּוֹ — פָּטוּר מִקְּרִיאַת שְׁמַע וּתְפִלָּה וּמִכָּל הַבְּרָכוֹת וַאֲפִלּוּ מֵעֲנִיַּת אָמֵן וּמִכָּל הַמִּצְווֹת הָאֲמוּרוֹת בַּתּוֹרָה. וַאֲפִלּוּ אִם רוֹצֶה לְהַחֲמִיר עַל עַצְמוֹ וְלִקְרוֹת, אוֹ אֲפִלּוּ לַעֲנוֹת אָמֵן שֶׁאֵינוֹ מִתְבַּטֵּל בִּשְׁבִילוֹ מִצָּרְכֵי הַמֵּת — אָסוּר מִדִּבְרֵי סוֹפְרִים מִפְּנֵי כְּבוֹדוֹ שֶׁל מֵת, שֶׁיִּהְיֶה לִבּוֹ פָּנוּי לְחַשֵּׁב בְּצָרְכֵי הַמֵּת וְיַחְשֹׁב בָּהֶם תָּמִיד. וַאֲפִלּוּ הוּא בְּעִיר אַחֶרֶת וְנוֹדַע לוֹ קֹדֶם הַקְּבוּרָה — אָסוּר, שֶׁלֹּא חִלְּקוּ חֲכָמִים. וְיֵשׁ מַתִּירִים אִם יֵשׁ לוֹ מִי שֶׁיִּתְעַסֵּק בִּשְׁבִילוֹ, וְאֵין צָרִיךְ לוֹמַר אִם הוּא בְּעִיר אַחֶרֶת שֶׁאֵינוֹ יָכוֹל לְהִתְעַסֵּק. וְנוֹהֲגִין כִּסְבָרָא הָרִאשׁוֹנָה (וּלְדִבְרֵי הַכֹּל בֵּן שֶׁהוּא אוֹנֵן עַל אָבִיו — יָכוֹל לֵילֵךְ לְבֵית הַכְּנֶסֶת וְלוֹמַר קַדִּישׁ עַל אָבִיו, כִּי זֶהוּ כְּבוֹדוֹ שֶׁל אָבִיו). וְכָל זֶה קֹדֶם הַקְּבוּרָה, אֲבָל מִיָּד שֶׁמַּתְחִילִין לְהַשְׁלִיךְ עָפָר עַל הַמֵּת — חַיָּב בִּקְרִיאַת שְׁמַע אִם לֹא עָבַר זְמַנָּהּ. וּתְפִלַּת י״ח בִּלְבַדָּהּ יָכוֹל לְהִתְפַּלֵּל אֲפִלּוּ אַחַר זְמַנָּהּ שֶׁהוּא שְׁלִישׁ הַיּוֹם, אִם לֹא עָבַר עֲדַיִן חֲצוֹת הַיּוֹם. אֲבָל אַחַר כָּךְ לֹא יִתְפַּלֵּל מִנְחָה שְׁתַּיִם לְתַשְׁלוּמִין שֶׁל שַׁחֲרִית, כֵּיוָן שֶׁהָיָה פָּטוּר בְּשַׁחֲרִית. וּמִטַּעַם זֶה לֹא יֹאמַר כָּל בִּרְכוֹת הַשַּׁחַר, אֲפִלּוּ נִקְבַּר קֹדֶם שֶׁעָבַר זְמַן קְרִיאַת שְׁמַע וּתְפִלָּה, כֵּיוָן שֶׁבְּשָׁעָה שֶׁחָל חִיּוּב הַבְּרָכוֹת דְּהַיְנוּ כְּשֶׁקָּם מִמִּטָּתוֹ הָיָה פָּטוּר מֵהֶן — אֵין צָרִיךְ לְהַשְׁלִימָן אַחַר כָּךְ. וְאִם מְבָרֵךְ — בְּרָכָה לְבַטָּלָה הִיא:

הָאוֹנֵן. מִי שֶׁמֵּת לוֹ מֵת שֶׁחַיָּב לְהִתְאַבֵּל עָלָיו — אֲפִלּוּ אֵין הַקְּבוּרָה מֻטֶּלֶת עָלָיו, אֲפִלּוּ יֵשׁ מִי שֶׁמִּתְעַסֵּק בִּשְׁבִילוֹ, אֲפִלּוּ כְּבָר עָסַק — פָּטוּר מִק״ש וּתְפִלָּה וּמִכָּל הַבְּרָכוֹת וַאֲפִלּוּ מֵעֲנִיַּת אָמֵן וּמִכָּל הַמִּצְווֹת. וְאַף אִם רוֹצֶה לְהַחֲמִיר — אָסוּר מִדִּבְרֵי סוֹפְרִים מִפְּנֵי כְּבוֹד הַמֵּת, שֶׁיִּהְיֶה לִבּוֹ פָּנוּי לְצָרְכֵי הַמֵּת. וְנוֹהֲגִין כִּסְבָרָא הָרִאשׁוֹנָה. וְכָל זֶה קֹדֶם הַקְּבוּרָה ; מִשֶּׁמַּתְחִילִין לְהַשְׁלִיךְ עָפָר — חַיָּב בְּק״ש אִם לֹא עָבַר זְמַנָּהּ. וּתְפִלַּת י״ח לְבַדָּהּ יָכוֹל לְהִתְפַּלֵּל אַחַר זְמַנָּהּ עַד חֲצוֹת. וְאֵינוֹ אוֹמֵר בִּרְכוֹת הַשַּׁחַר שֶׁבְּשָׁעַת חִיּוּבָן הָיָה פָּטוּר, וְאִם בֵּרַךְ — בְּרָכָה לְבַטָּלָה.

סעיף ב בְּשַׁבָּת אֵין אֲנִינוּת — חַיָּב כָּל הַיּוֹם ; מַחְשִׁיךְ עַל הַתְּחוּם — פָּטוּר לְעֵת עֶרֶב.

וְכָל זֶה בְּחֹל, אֲבָל בְּשַׁבָּת כֵּיוָן שֶׁאִי אֶפְשָׁר לַעֲסֹק בְּצָרְכֵי הַמֵּת בִּקְבוּרָתוֹ וְאֵין אֲנִינוּת נוֹהֶגֶת כְּלָל — חַיָּב בִּקְרִיאַת שְׁמַע וּתְפִלָּה וּבְכָל הַמִּצְווֹת הָאֲמוּרוֹת בַּתּוֹרָה כָּל הַיּוֹם. וּלְעֵת עֶרֶב, אִם הוּא מַחְשִׁיךְ עַל הַתְּחוּם לְהִתְעַסֵּק בְּצָרְכֵי הַקְּבוּרָה — פָּטוּר. אֲבָל אִם אֵינוֹ מַחְשִׁיךְ עַל הַתְּחוּם — חַיָּב גַּם לְעֵת עֶרֶב בְּכָל הַמִּצְווֹת. וְיֵשׁ לוֹ גַּם כֵּן לִקְרוֹת קְרִיאַת שְׁמַע שֶׁל עַרְבִית מִבְּעוֹד יוֹם אַף עַל פִּי שֶׁאֵין הַקָּהָל קוֹרִין בְּמוֹצָאֵי שַׁבָּת עַד הַלַּיְלָה, מִכָּל מָקוֹם זֶה שֶׁיִּהְיֶה פָּטוּר בַּלַּיְלָה — לֹא יְבַטֵּל מִמֶּנּוּ עֹל מַלְכוּת שָׁמַיִם כָּל שֶׁיָכוֹל לִקְרוֹת קֹדֶם הַלַּיְלָה, שֶׁהֲרֵי רֹב הַקָּהָל קוֹרִין כָּל יְמוֹת הַשָּׁבוּעַ מִבְּעוֹד יוֹם, אִם כֵּן יָכוֹל זֶה לִסְמֹךְ עַל זֶה בְּאוֹתוֹ שַׁבָּת. וְאֵין צָרִיךְ לוֹמַר שֶׁיֵּשׁ לוֹ לְהִתְפַּלֵּל תְּפִלַּת עַרְבִית שֶׁל מוֹצָאֵי שַׁבָּת מִבְּעוֹד יוֹם מִפְּלַג הַמִנְחָה וּלְמַעְלָה, שֶׁהוּא לַיְלָה גָּמוּר לְעִנְיַן תְּפִלָּה לְר׳ יְהוּדָה. וְאַף מִי שֶׁנָּהַג לְעוֹלָם כַּחֲכָמִים — יָכוֹל לִסְמֹךְ עַל דִּבְרֵי רַבִּי יְהוּדָה אֲפִלּוּ בְּמוֹצָאֵי שַׁבָּת בִּמְקוֹם אֵיזֶה אֹנֶס שֶׁאִי אֶפְשָׁר לוֹ לְהִתְפַּלֵּל בַּלַּיְלָה, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן רצ״ג:

שַׁבָּת. כָּל זֶה בְּחֹל ; אֲבָל בְּשַׁבָּת, כֵּיוָן שֶׁאִי אֶפְשָׁר לַעֲסֹק בַּקְּבוּרָה וְאֵין אֲנִינוּת נוֹהֶגֶת כְּלָל — חַיָּב בְּק״ש, תְּפִלָּה וְכָל הַמִּצְווֹת כָּל הַיּוֹם. וּלְעֵת עֶרֶב : מַחְשִׁיךְ עַל הַתְּחוּם לְצָרְכֵי קְבוּרָה — פָּטוּר ; אֵינוֹ מַחְשִׁיךְ — חַיָּב גַּם לְעֵת עֶרֶב. וְקוֹרֵא ק״ש שֶׁל עַרְבִית מִבְּעוֹד יוֹם שֶׁלֹּא יְבַטֵּל עֹל מַלְכוּת שָׁמַיִם, וּמִתְפַּלֵּל עַרְבִית שֶׁל מוֹצָאֵי שַׁבָּת מִפְּלַג הַמִּנְחָה כְּרַבִּי יְהוּדָה (כְּמוֹ בְּסִימָן רצ״ג).

סעיף ג יוֹם טוֹב שֵׁנִי כְּחֹל ; יוֹם טוֹב רִאשׁוֹן — לְפִי הַקְּבוּרָה.

יוֹם טוֹב שֵׁנִי — דִּינוֹ כְּחֹל, שֶׁהֲרֵי יָכוֹל לְקָבְרוֹ הַיּוֹם כְּמוֹ בְּחֹל. וְיוֹם טוֹב רִאשׁוֹן אִם רוֹצֶה לְקָבְרוֹ בּוֹ בַּיּוֹם עַל יְדֵי נָכְרִים — דִּינוֹ כְּחֹל, שֶׁהֲרֵי צָרִיךְ לְהַמְצִיא לָהֶם אָרוֹן וְתַכְרִיכִין. וְאִם אֵינוֹ רוֹצֶה לְקָבְרוֹ בּוֹ בַּיּוֹם — דִּינוֹ כְּמוֹ בְּשַׁבָּת. וְדִין לֵיל יוֹם טוֹב בְּמָקוֹם שֶׁאֵין קוֹבְרִין מֵת בַּלַּיְלָה, וְכֵן בַּיּוֹם בְּמָקוֹם שֶׁאִי אֶפְשָׁר לְקָבְרוֹ מֵחֲמַת אֵיזֶה אֹנֶס — יִתְבָּאֵר בְּסִימָן תקמ״ח:

יוֹם טוֹב. יוֹם טוֹב שֵׁנִי — דִּינוֹ כְּחֹל, שֶׁיָּכוֹל לְקָבְרוֹ הַיּוֹם. יוֹם טוֹב רִאשׁוֹן : רוֹצֶה לְקָבְרוֹ הַיּוֹם עַל יְדֵי נָכְרִים — כְּחֹל (שֶׁצָּרִיךְ אָרוֹן וְתַכְרִיכִין) ; אֵינוֹ רוֹצֶה לְקָבְרוֹ הַיּוֹם — כְּשַׁבָּת. וְדִין לֵיל יוֹם טוֹב וְאֹנֶס — בְּסִימָן תקמ״ח.

סעיף ד הַמְשַׁמֵּר אֶת הַמֵּת פָּטוּר — עוֹסֵק בְּמִצְוָה ; שְׁנַיִם — זֶה מְשַׁמֵּר וְזֶה קוֹרֵא.

הַמְשַׁמֵּר אֶת הַמֵּת אַף עַל פִּי שֶׁאֵינוֹ מֵתוֹ — פָּטוּר מִקְּרִיאַת שְׁמַע וּמִתְּפִלָּה וּמִכָּל מִּצְווֹת הָאֲמוּרוֹת בַּתּוֹרָה, מִפְּנֵי שֶׁהוּא עוֹסֵק בְּמִצְוָה, וְכָל הָעוֹסֵק בְּמִצְוָה — לֹא הִטְרִיחַתּוּ תּוֹרָה לִטְרֹחַ בְּמִּצְווֹת אֲחֵרוֹת אַף עַל פִּי שֶׁיָּכוֹל, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ל״ח. הָיוּ שְׁנַיִם מְשַׁמְּרִים — זֶה מְשַׁמֵּר וְזֶה קוֹרֵא, וְאַחַר כָּךְ מְשַׁמֵּר זֶה וְקוֹרֵא זֶה:

הַמְשַׁמֵּר. הַמְשַׁמֵּר אֶת הַמֵּת, אַף שֶׁאֵינוֹ מֵתוֹ — פָּטוּר מִק״ש, תְּפִלָּה וְכָל הַמִּצְווֹת, מִפְּנֵי שֶׁהוּא עוֹסֵק בְּמִצְוָה, וְכָל הָעוֹסֵק בְּמִצְוָה לֹא הִטְרִיחַתּוּ תּוֹרָה בְּמִצְווֹת אֲחֵרוֹת אַף שֶׁיָּכוֹל (סִימָן ל״ח). שְׁנַיִם מְשַׁמְּרִים — זֶה מְשַׁמֵּר וְזֶה קוֹרֵא, וְחוֹזְרִים.

סעיף ה הַחוֹפֵר קֶבֶר פָּטוּר ; נוֹסָפִין — נִשְׁמָטִין וְקוֹרִין וְחוֹזְרִין.

וְכֵן הַחוֹפֵר קֶבֶר לְמֵת פָּטוּר מִכֻּלָּן אֲפִלּוּ בִּשְׁעַת נוּחוֹ, כְּמוֹ ש(יִ)[נִ]תְבָּאֵר שָׁם. הָיוּ חוֹפְרִים שְׁנַיִם אוֹ יוֹתֵר כָּל הַצְּרִיכִים לְצָרְכֵי הַחֲפִירָה בְּבַת אַחַת — פְּטוּרִין. וְאִם יֵשׁ נוֹסָפִין — נִשְׁמָטִין וְקוֹרִין וְחוֹזְרִין אֵלּוּ הַנּוֹסָפִים וּמִתְעַסְּקִים, וְהָאֲחֵרִים נִשְׁמָטִין כְּמִסְפַּר הַנּוֹסָפִין וְקוֹרִין וְחוֹזְרִין לְהִתְעַסֵּק, וְנִשְׁמָטִים אֲחֵרִים עַד שֶׁנִּמְצָא שֶׁקָּרְאוּ כֻּלָּם, וְלֹא נִתְבַּטֵּל שׁוּם דָּבָר מֵעֵסֶק הַקֶּבֶר כְּלָל (אֲפִלּוּ רֶגַע):

הַחוֹפֵר. וְכֵן הַחוֹפֵר קֶבֶר לְמֵת — פָּטוּר מִכֻּלָּן אֲפִלּוּ בִּשְׁעַת נוּחוֹ. הָיוּ שְׁנַיִם אוֹ יוֹתֵר, כָּל הַצְּרִיכִים בְּבַת אַחַת — פְּטוּרִין. יֵשׁ נוֹסָפִין — נִשְׁמָטִין וְקוֹרִין וְחוֹזְרִין, וְהָאֲחֵרִים נִשְׁמָטִין כְּמִסְפָּרָם וְקוֹרִין, עַד שֶׁקָּרְאוּ כֻּלָּם, בְּלִי שֶׁיִּתְבַּטֵּל עֵסֶק הַקֶּבֶר כְּלָל (אֲפִלּוּ רֶגַע).

סעיף ו אָסוּר לִקְרוֹת תּוֹךְ ד׳ אַמּוֹת שֶׁל מֵת אוֹ קֶבֶר — לֹעֵג לָרָשׁ.

אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע תּוֹךְ ד׳ אַמּוֹת שֶׁל מֵת אוֹ תּוֹךְ ד׳ אַמּוֹת שֶׁל קֶבֶר, מִשּׁוּם ״לֹעֵג לָרָשׁ חֵרֵף עֹשֵׂהוּ״. וְאִם קָרָא — לֹא יָצָא, וְצָרִיךְ לַחֲזֹר וְלִקְרוֹת, שֶׁקְּנָסוּהוּ חֲכָמִים עַל שֶׁעָבַר עַל דִּבְרֵיהֶם, וַאֲפִלּוּ הָיָה שׁוֹגֵג. וּבֵית הַקְּבָרוֹת הַמֻּקָּף מְחִצָּה — מֻתָּר לִקְרוֹת חוּץ לַמְּחִצָּה, אֲפִלּוּ סָמוּךְ לָהּ וּבְתוֹךְ ד׳ אַמּוֹת לַקְּבָרוֹת. אֲבָל בְּתוֹךְ הַמְּחִצָּה — יֵשׁ לִמְנֹעַ אֲפִלּוּ חוּץ לְד׳ אַמּוֹת, שֶׁמָּא יִתְקָרֵב לְד׳ אַמּוֹת שֶׁל אֵיזֶה קֶבֶר בְּלִי דַּעַת. וְאִם קָרָא — יָצָא, כֵּיוָן שֶׁאֵינוֹ אִסּוּר בָּרוּר אֶלָּא מִשּׁוּם חֲשָׁשׁ לְבַד:

לֹעֵג לָרָשׁ. אָסוּר לִקְרוֹת ק״ש תּוֹךְ ד׳ אַמּוֹת שֶׁל מֵת אוֹ שֶׁל קֶבֶר, מִשּׁוּם ״לֹעֵג לָרָשׁ חֵרֵף עֹשֵׂהוּ״. קָרָא — לֹא יָצָא וְצָרִיךְ לַחֲזֹר, שֶׁקְּנָסוּהוּ חֲכָמִים אֲפִלּוּ שׁוֹגֵג. בֵּית הַקְּבָרוֹת הַמֻּקָּף מְחִצָּה — מֻתָּר חוּץ לַמְּחִצָּה, אֲפִלּוּ סָמוּךְ וּבְתוֹךְ ד׳ אַמּוֹת ; אֲבָל בְּתוֹךְ הַמְּחִצָּה יֵשׁ לִמְנֹעַ, וְאִם קָרָא — יָצָא (חֲשָׁשׁ בִּלְבַד).

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ע״א (מהדורת קה״ת, ו׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין אָבֵל וְהָעוֹסְקִים בַּמֵּת

היסוד : דִּין אָבֵל וְהָעוֹסְקִים בַּמֵּת

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הָאֲנִינוּת הַפּוֹטֶרֶת אֶת הָאוֹנֵן. אחר כך הַהֶבְדֵּל חֹל / שַׁבָּת / יוֹם טוֹב. אחר כך הָעוֹסֵק בְּמִצְוָה (הַמְשַׁמֵּר וְהַחוֹפֵר). ולבסוף הַלֹעֵג לָרָשׁ וְד׳ אַמּוֹת.

אֲנִינוּת — הָאוֹנֵן פָּטוּר

אֲנִינוּת — הָאוֹנֵן פָּטוּר

היסוד. הָאוֹנֵן (קֹדֶם קְבוּרָה) פָּטוּר מִק״ש, תְּפִלָּה, בְּרָכוֹת וַאֲפִלּוּ מֵעֲנִיַּת אָמֵן וּמִכָּל הַמִּצְווֹת ; וְאַף אִם רוֹצֶה לְהַחֲמִיר — אָסוּר מִפְּנֵי כְּבוֹד הַמֵּת (סעיף א).

המסקנה. הַפְּטוֹר קֹדֶם הַקְּבוּרָה ; מִשֶּׁמַּתְחִילִין לְהַשְׁלִיךְ עָפָר — חַיָּב בְּק״ש. עיין סימן 70 (הַפְּטוּרִים מִק״ש).

חֹל, שַׁבָּת וְיוֹם טוֹב

חֹל, שַׁבָּת וְיוֹם טוֹב

היסוד. בְּשַׁבָּת אֵין אֲנִינוּת נוֹהֶגֶת — חַיָּב כָּל הַיּוֹם ; מַחְשִׁיךְ עַל הַתְּחוּם — פָּטוּר לְעֵת עֶרֶב (סעיף ב). יוֹם טוֹב שֵׁנִי כְּחֹל ; רִאשׁוֹן לְפִי הַקְּבוּרָה (סעיף ג).

המסקנה. בְּשַׁבָּת קוֹרֵא וּמִתְפַּלֵּל כְּרָגִיל, וְקוֹרֵא אַף ק״ש שֶׁל עַרְבִית מִבְּעוֹד יוֹם. עיין סימן 72 (נוֹשְׂאֵי הַמִּטָּה).

הָעוֹסֵק בְּמִצְוָה — מְשַׁמֵּר וְחוֹפֵר

הָעוֹסֵק בְּמִצְוָה — מְשַׁמֵּר וְחוֹפֵר

היסוד. הַמְשַׁמֵּר וְהַחוֹפֵר — פָּטוּר, שֶׁכָּל הָעוֹסֵק בְּמִצְוָה לֹא הִטְרִיחַתּוּ תּוֹרָה (סימן ל״ח) (סעיפים ד-ה).

המסקנה. בִּרְבִּים — נִשְׁמָטִין וְקוֹרִין בְּלִי בִּטּוּל הַמְּלָאכָה. עיין סימן 38 (הָעוֹסֵק בְּמִצְוָה).

לֹעֵג לָרָשׁ — ד׳ אַמּוֹת

לֹעֵג לָרָשׁ — ד׳ אַמּוֹת

היסוד. אָסוּר לִקְרוֹת תּוֹךְ ד׳ אַמּוֹת שֶׁל מֵת אוֹ קֶבֶר, מִשּׁוּם ״לֹעֵג לָרָשׁ״ ; קָרָא — לֹא יָצָא (סעיף ו).

המסקנה. חוּץ לַמְּחִיצָה — מֻתָּר ; בְּתוֹךְ הַמְּחִיצָה יֵשׁ לִמְנֹעַ, וְאִם קָרָא — יָצָא (חֲשָׁשׁ בִּלְבַד). עיין סימן 72.

הפניה ללימוד. היקף הסימן — הָאֲנִינוּת, הָעוֹסֵק בְּמִצְוָה וְהַלֹעֵג לָרָשׁ — נוגע לרגעים כבדים ומעשיים. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין אָבֵל וְהָעוֹסְקִים בַּמֵּת — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ע״א. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מו׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
אֲנִינוּת — הָאוֹנֵן פָּטוּר או״ח ע״א — אָבֵל וְהָעוֹסְקִים בַּמֵּת פְּטוּרִים ; רוֹצֶה לְהַחֲמִיר אֵינוֹ רַשַּׁאי. מ״ב ע״א — טַעַם הַפְּטוֹר מִשּׁוּם כְּבוֹד הַמֵּת, וְדִין בִּרְכוֹת הַשַּׁחַר. ערוה״ש ע״א — גֶּדֶר הָאֲנִינוּת קֹדֶם הַקְּבוּרָה. דרך אדמו״ר הזקן — פָּטוּר מִכֹּל אֲפִלּוּ מֵעֲנִיַּת אָמֵן ; אֲפִלּוּ יֵשׁ מִי שֶׁמִּתְעַסֵּק ; מִשֶּׁמַּתְחִילִין לְהַשְׁלִיךְ עָפָר — חַיָּב (סעיף א).
חֹל, שַׁבָּת וְיוֹם טוֹב או״ח ע״א — בְּחֹל פָּטוּר ; בְּשַׁבָּת חַיָּב ; מַחְשִׁיךְ עַל הַתְּחוּם פָּטוּר ; י״ט שֵׁנִי כְּחֹל. מ״ב ע״א — פְּרָטֵי הַמַּחְשִׁיךְ עַל הַתְּחוּם וּלְעֵת עֶרֶב. ערוה״ש ע״א — דִּין הָאֲנִינוּת בְּשַׁבָּת וּבְיוֹם טוֹב. דרך אדמו״ר הזקן — בְּשַׁבָּת אֵין אֲנִינוּת נוֹהֶגֶת — חַיָּב כָּל הַיּוֹם ; קוֹרֵא ק״ש שֶׁל עַרְבִית מִבְּעוֹד יוֹם ; י״ט רִאשׁוֹן לְפִי הַקְּבוּרָה (סעיפים ב-ג).
הָעוֹסֵק בְּמִצְוָה או״ח ע״א — הַמְשַׁמֵּר וְהַחוֹפֵר פָּטוּר ; שְׁנַיִם זֶה מְשַׁמֵּר וְזֶה קוֹרֵא. מ״ב ע״א — גֶּדֶר הָעוֹסֵק בְּמִצְוָה פָּטוּר מִן הַמִּצְוָה. ערוה״ש ע״א — סֵדֶר הַנִּשְׁמָטִין וְקוֹרִין בַּחֲפִירָה. דרך אדמו״ר הזקן — כָּל הָעוֹסֵק בְּמִצְוָה לֹא הִטְרִיחַתּוּ תּוֹרָה (סִימָן ל״ח) ; הַנּוֹסָפִין נִשְׁמָטִין וְקוֹרִין בְּלִי בִּטּוּל הַחֲפִירָה (סעיפים ד-ה).
לֹעֵג לָרָשׁ וּבֵית הַקְּבָרוֹת או״ח ע״א — אָסוּר תּוֹךְ ד׳ אַמּוֹת שֶׁל מֵת אוֹ בְּבֵית הַקְּבָרוֹת ; קָרָא לֹא יָצָא. מ״ב ע״א — טַעַם ״לֹעֵג לָרָשׁ״ וְדִין הַמְּחִיצָה. ערוה״ש ע״א — שִׁעוּר ד׳ אַמּוֹת וּמְחִיצַת בֵּית הַקְּבָרוֹת. דרך אדמו״ר הזקן — ״לֹעֵג לָרָשׁ חֵרֵף עֹשֵׂהוּ״ ; קָרָא — לֹא יָצָא וְקָנְסוּהוּ ; חוּץ לַמְּחִיצָה מֻתָּר, בְּתוֹכָהּ יֵשׁ לִמְנֹעַ (סעיף ו).

הטבלה נערכה מטקסט המחבר (אורח חיים ע״א, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ד, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות י״ז-י״ח, ויורה דעה שמ״א / שס״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ו׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — אֲנִינוּת פּוֹטֶרֶת

קו א — אֲנִינוּת פּוֹטֶרֶת

יסוד הסימן : אָבֵל וְאוֹנֵן פָּטוּר עַד הַקְּבוּרָה (סעיף א).

קריאת הרב : הַפְּטוֹר מִפְּנֵי כְּבוֹד הַמֵּת, שֶׁיִּהְיֶה לִבּוֹ פָּנוּי ; אֵין מַחְמִירִין לִקְרוֹת. עיין סימן 70.

מסקנה למעשה : הָאוֹנֵן אֵינוֹ קוֹרֵא ק״ש וְאֵינוֹ אוֹמֵר בְּרָכוֹת ; מִשֶּׁמַּשְׁלִיכִין עָפָר — חַיָּב.

קו ב — הָעוֹסֵק בְּמִצְוָה

קו ב — הָעוֹסֵק בְּמִצְוָה

יסוד הסימן : הַמְשַׁמֵּר וְהַחוֹפֵר פָּטוּר (סעיפים ד-ה).

קריאת הרב : כָּל הָעוֹסֵק בְּמִצְוָה פָּטוּר מִן הַמִּצְוָה (סימן ל״ח) ; שְׁנַיִם — נִשְׁמָטִין וְקוֹרִין בְּלִי בִּטּוּל. עיין סימן 38.

מסקנה למעשה : הַטָּרוּד בַּמֵּת אוֹ בַּקֶּבֶר פָּטוּר, אַךְ מְסַדְּרִים תּוֹרוֹת לִקְרוֹת.

קו ג — כְּבוֹד הַמֵּת וְלֹעֵג לָרָשׁ

קו ג — כְּבוֹד הַמֵּת וְלֹעֵג לָרָשׁ

יסוד הסימן : אָסוּר תּוֹךְ ד׳ אַמּוֹת שֶׁל מֵת אוֹ קֶבֶר (סעיף ו).

קריאת הרב : מִשּׁוּם ״לֹעֵג לָרָשׁ חֵרֵף עֹשֵׂהוּ״ ; חוּץ לַמְּחִיצָה מֻתָּר, בְּתוֹכָהּ יֵשׁ לִמְנֹעַ. עיין סימן 72.

מסקנה למעשה : אֵין קוֹרְאִין ק״ש וְאֵין אוֹמְרִים בְּרָכוֹת סָמוּךְ לַמֵּת אוֹ בְּבֵית הַקְּבָרוֹת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ע״א ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ע״א (דִּין אָבֵל וְהָעוֹסְקִים בַּמֵּת)

  • ① אֲנִינוּת. הָאוֹנֵן (קֹדֶם קְבוּרָה) פָּטוּר מִק״ש, תְּפִלָּה, בְּרָכוֹת וַאֲפִלּוּ מֵעֲנִיַּת אָמֵן ; וְאֵינוֹ מַחְמִיר לִקְרוֹת. יסוד : שוע״ר ע״א סעיף א.
  • ② מִשֶּׁמַּתְחִילִין הַקְּבוּרָה. מִשֶּׁמַּשְׁלִיכִין עָפָר עַל הַמֵּת — חַיָּב בְּק״ש אִם לֹא עָבַר זְמַנָּהּ ; וְאֵינוֹ מַשְׁלִים בִּרְכוֹת הַשַּׁחַר. יסוד : שוע״ר ע״א סעיף א.
  • ③ שַׁבָּת. בְּשַׁבָּת אֵין אֲנִינוּת — חַיָּב כָּל הַיּוֹם ; מַחְשִׁיךְ עַל הַתְּחוּם — פָּטוּר לְעֵת עֶרֶב. יסוד : שוע״ר ע״א סעיף ב.
  • ④ יוֹם טוֹב. יוֹם טוֹב שֵׁנִי כְּחֹל ; יוֹם טוֹב רִאשׁוֹן — לְפִי אִם קוֹבֵר הַיּוֹם (עַל יְדֵי נָכְרִים). יסוד : שוע״ר ע״א סעיף ג.
  • ⑤ עוֹסֵק בְּמִצְוָה. הַמְשַׁמֵּר וְהַחוֹפֵר פְּטוּרִין (עוסק במצוה) ; בִּרְבִּים — נִשְׁמָטִין וְקוֹרִין בְּלִי בִּטּוּל הַמְּלָאכָה. יסוד : שוע״ר ע״א סעיפים ד-ה.
  • ⑥ ד׳ אַמּוֹת. אֵין קוֹרְאִין ק״ש תּוֹךְ ד׳ אַמּוֹת שֶׁל מֵת אוֹ קֶבֶר (לֹעֵג לָרָשׁ) ; קָרָא — לֹא יָצָא. יסוד : שוע״ר ע״א סעיף ו.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ע״א (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הָאֲנִינוּת, הָעוֹסֵק בְּמִצְוָה, הַלֹעֵג לָרָשׁ — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
