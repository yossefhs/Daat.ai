דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קכ״ט · בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם

דרכו של אדמו״ר הזקן בבְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם : שַׁחֲרִית, מוּסָף וּנְעִילָה — כֵּן ; מִנְחָה — לֹא, מִפְּנֵי חֲשַׁשׁ שִׁכְרוּת ; גְּזֵרַת מִנְחַת תַּעֲנִית שֶׁיֵּשׁ בָּהּ נְעִילָה אַטּוּ מִנְחָה שֶׁל שְׁאָר יָמִים ; וְכֹהֵן שֶׁעָבַר וְעָלָה — אֵין מוֹרִידִין אוֹתוֹ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קכ״ט — בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם. אצל הרב הסימן נפרש לב׳ סְעִיפִים : זְמַן נְשִׂיאַת כַּפַּיִם (שַׁחֲרִית, מוּסָף וּנְעִילָה, לֹא בְּמִנְחָה מִשּׁוּם שִׁכְרוּת) עִם גְּזֵרַת מִנְחַת תַּעֲנִית, וְאַחַר כָּךְ דִּין הַכֹּהֵן שֶׁעָבַר וְעָלָה לַדּוּכָן בְּמִנְחַת יוֹם הַכִּפּוּרִים וּמִנְהַג ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קכ״ט — ב׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קכ״ט). אדמו״ר הזקן דן בו בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קכ״ט — בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קכ״ט — בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם — וּבוֹ ב׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.129), ואחריו תרגומנו. הסימן כולל ב׳ סעיפים (סעיף k → ‎.129.k).

סעיף א אֵין נְשִׂיאַת כַּפַּיִם אֶלָּא בְּשַׁחֲרִית מוּסָף וּנְעִילָה ; לֹא בְּמִנְחָה מִשּׁוּם שִׁכְרוּת.

בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם וּבוֹ ב׳ סְעִיפִים:אֵין נְשִׂיאַת כַּפַּיִם אֶלָּא בְּשַׁחֲרִית וּמוּסָף וּנְעִילָה בְּיוֹם שֶׁיֵּשׁ בּוֹ נְעִילָה כְּגוֹן בְּיוֹם הַכִּפּוּרִים, אֲבָל לֹא בְּמִנְחָה, מִפְּנֵי שֶׁהַשִּׁכְרוּת מְצוּיָה בְּאוֹתָהּ שָׁעָה וְשֶׁמָּא יִהְיֶה הַכֹּהֵן שִׁכּוֹר. וְגָזְרוּ בְּמִנְחָה שֶׁל יוֹם הַכִּפּוּרִים וּשְׁאָר תַּעֲנִית צִבּוּר שֶׁיֵּשׁ בָּהֶן נְעִילָה כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תקע״ט אַף עַל פִּי שֶׁאֵין בָּהֶם חֲשַׁשׁ שִׁכְרוּת, גְּזֵרָה מִשּׁוּם מִנְחָה שֶׁל שְׁאָר יָמִים, לְפִי שֶׁבְּיוֹם שֶׁיֵּשׁ בּוֹ נְעִילָה מִתְפַּלְלִים מִנְחָה בְּעוֹד הַיּוֹם גָּדוֹל כְּדֵי לְהַתְחִיל נְעִילָה קֹדֶם שְׁקִיעַת הַחַמָּה וּמִתְחַלֶּפֶת הִיא בְּמִנְחָה שֶׁל שְׁאָר יָמִים, שֶׁהָיוּ נוֹהֲגִים לְהִתְפַּלֵּל בְּעוֹד הַיּוֹם גָּדוֹל מִפְּנֵי אוֹתָם שֶׁרוֹצִים לֶאֱכֹל אַחַר כָּךְ, שֶׁאָסוּר לֶאֱכֹל קֹדֶם תְּפִלַּת הַמִּנְחָה כְּמוֹ שֶׁיִּתְבָּאֵר סִימָן רל״ב. אֲבָל בְּתַעֲנִית שֶׁאֵין בָּהֶם נְעִילָה, הוֹאִיל וּתְפִלַּת הַמִּנְחָה הוּא סָמוּךְ לִשְׁקִיעַת הַחַמָּה — הִיא דּוֹמָה לִתְפִלַּת נְעִילָה וְאֵינָהּ מִתְחַלֶּפֶת בְּמִנְחָה שֶׁל שְׁאָר הַיָּמִים, לְפִיכָךְ יֵשׁ בָּהּ נְשִׂיאַת כַּפַּיִם:

זְמַן נְשִׂיאַת כַּפַּיִם. אֵין נְשִׂיאַת כַּפַּיִם אֶלָּא בְּשַׁחֲרִית, מוּסָף וּנְעִילָה בְּיוֹם שֶׁיֵּשׁ בּוֹ נְעִילָה (כְּיוֹם הַכִּפּוּרִים) ; אֲבָל לֹא בְּמִנְחָה, מִפְּנֵי שֶׁהַשִּׁכְרוּת מְצוּיָה בְּאוֹתָהּ שָׁעָה וְשֶׁמָּא יִהְיֶה הַכֹּהֵן שִׁכּוֹר. וְגָזְרוּ בְּמִנְחָה שֶׁל יוֹם הַכִּפּוּרִים וּשְׁאָר תַּעֲנִית צִבּוּר שֶׁיֵּשׁ בָּהֶן נְעִילָה (סִימָן תקע״ט) אַף שֶׁאֵין בָּהֶם חֲשַׁשׁ שִׁכְרוּת — גְּזֵרָה מִשּׁוּם מִנְחָה שֶׁל שְׁאָר יָמִים, שֶׁמִּתְפַּלְלִים בְּעוֹד הַיּוֹם גָּדוֹל וּמִתְחַלֶּפֶת בָּהּ. אֲבָל תַּעֲנִית שֶׁאֵין בָּהּ נְעִילָה — מִנְחָתָהּ סָמוּךְ לִשְׁקִיעָה, דּוֹמָה לִנְעִילָה וְאֵינָהּ מִתְחַלֶּפֶת, לְפִיכָךְ יֵשׁ בָּהּ נְשִׂיאַת כַּפַּיִם.

סעיף ב כֹּהֵן שֶׁעָבַר וְעָלָה בְּמִנְחַת יוֹם הַכִּפּוּרִים — אֵין מוֹרִידִין אוֹתוֹ.

כֹּהֵן שֶׁעָבַר וְעָלָה לַדּוּכָן בְּיוֹם הַכִּפּוּרִים בְּמִנְחָה, כֵּיוָן שֶׁהַדָּבָר יָדוּעַ שֶׁאֵין שָׁם שִׁכְרוּת — הֲרֵי זֶה נוֹשֵׂא כַּפָּיו וְאֵין מוֹרִידִין אוֹתוֹ, מִפְּנֵי הַחֲשָׁד שֶׁלֹּא יֹאמְרוּ פָּסוּל הוּא וּלְכָךְ הוֹרִידוּהוּ. וְלָכֵן נוֹהֲגִים בִּמְדִינוֹת אֵלּוּ לוֹמַר ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ בָּרְכֵנוּ כו׳״ בְּיוֹם הַכִּפּוּרִים בְּמִנְחָה, אַף עַל פִּי שֶׁאֵין אוֹמְרִים אֶלָּא בִּזְמַן הָרָאוּי לִנְשִׂיאַת כַּפַּיִם, מִכָּל מָקוֹם הוֹאִיל וְאִם עָלָה לֹא יֵרֵד וְנוֹשֵׂא כַּפָּיו — נִקְרָא קְצָת רָאוּי לִנְשִׂיאַת כַּפַּיִם:

עָבַר וְעָלָה. כֹּהֵן שֶׁעָבַר וְעָלָה לַדּוּכָן בְּמִנְחַת יוֹם הַכִּפּוּרִים, כֵּיוָן שֶׁיָּדוּעַ שֶׁאֵין שָׁם שִׁכְרוּת — נוֹשֵׂא כַּפָּיו וְאֵין מוֹרִידִין אוֹתוֹ, מִפְּנֵי הַחֲשָׁד שֶׁלֹּא יֹאמְרוּ פָּסוּל הוּא וּלְכָךְ הוֹרִידוּהוּ. וְלָכֵן נוֹהֲגִים בִּמְדִינוֹת אֵלּוּ לוֹמַר ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ בָּרְכֵנוּ כו׳״ בְּמִנְחַת יוֹם הַכִּפּוּרִים — אַף שֶׁאֵין אוֹמְרִים אֶלָּא בִּזְמַן הָרָאוּי לִנְשִׂיאַת כַּפַּיִם, מִכָּל מָקוֹם הוֹאִיל וְאִם עָלָה לֹא יֵרֵד וְנוֹשֵׂא כַּפָּיו — נִקְרָא קְצָת רָאוּי לִנְשִׂיאַת כַּפַּיִם.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קכ״ט (מהדורת קה״ת, ב׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם

היסוד : בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים. תחילה זְמַן נְשִׂיאַת כַּפַּיִם (שַׁחֲרִית, מוּסָף וּנְעִילָה) וְטַעַם הוֹצָאַת הַמִּנְחָה (חֲשַׁשׁ שִׁכְרוּת). אחר כך גְּזֵרַת מִנְחַת תַּעֲנִית שֶׁיֵּשׁ בָּהּ נְעִילָה אַטּוּ מנחה של שאר ימים, עִם חִלּוּק הַתַּעֲנִית שֶׁאֵין בָּהּ נְעִילָה. ולבסוף דִּין הַכֹּהֵן שֶׁעָבַר וְעָלָה וְהַחֲשָׁד הַמְיַסֶּדֶת אֶת מִנְהַג ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״.

זְמַן נְשִׂיאַת כַּפַּיִם — שַׁחֲרִית מוּסָף וּנְעִילָה

זְמַן נְשִׂיאַת כַּפַּיִם — שַׁחֲרִית מוּסָף וּנְעִילָה

היסוד. נְשִׂיאַת כַּפַּיִם רַק בְּשַׁחֲרִית, מוּסָף וּנְעִילָה (בְּיוֹם שֶׁיֵּשׁ בּוֹ נְעִילָה) ; לֹא בְּמִנְחָה, מִפְּנֵי שֶׁהַשִּׁכְרוּת מְצוּיָה בְּאוֹתָהּ שָׁעָה — שֶׁמָּא יִהְיֶה הַכֹּהֵן שִׁכּוֹר (סעיף א).

המסקנה. הַטַּעַם אֵינוֹ בַּתְּפִלָּה עַצְמָהּ אֶלָּא בַּשָּׁעָה : מַה שֶּׁמּוֹצִיא אֶת מִנְחָה הוּא חֲשַׁשׁ הַשִּׁכְרוּת שֶׁל אַחַר הַצָּהֳרַיִם. עיין סימן קכ״ח · 128 (דיני נשיאת כפים).

גְּזֵרַת מִנְחַת תַּעֲנִית שֶׁיֵּשׁ בָּהּ נְעִילָה

גְּזֵרַת מִנְחַת תַּעֲנִית שֶׁיֵּשׁ בָּהּ נְעִילָה

היסוד. גָּזְרוּ בְּמִנְחָה שֶׁל יוֹם הַכִּפּוּרִים וּתַעֲנִית צִבּוּר שֶׁיֵּשׁ בָּהֶן נְעִילָה — אַף שֶׁאֵין בָּהֶם חֲשַׁשׁ שִׁכְרוּת — גְּזֵרָה מִשּׁוּם מִנְחָה שֶׁל שְׁאָר יָמִים, שֶׁמִּתְפַּלְלִים בְּעוֹד הַיּוֹם גָּדוֹל וּמִתְחַלֶּפֶת בַּמִּנְחָה הָרְגִילָה (סעיף א).

המסקנה. אֲבָל תַּעֲנִית שֶׁאֵין בָּהּ נְעִילָה — מִנְחָתָהּ סָמוּךְ לִשְׁקִיעָה, דּוֹמָה לִנְעִילָה וְאֵינָהּ מִתְחַלֶּפֶת, לְפִיכָךְ יֵשׁ בָּהּ נְשִׂיאַת כַּפַּיִם. עיין סימן קכ״ח · 128 (זמן וכשרות הכהן).

עָבַר וְעָלָה — ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״

עָבַר וְעָלָה — ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״

היסוד. כֹּהֵן שֶׁעָבַר וְעָלָה לַדּוּכָן בְּמִנְחַת יוֹם הַכִּפּוּרִים — כֵּיוָן שֶׁיָּדוּעַ שֶׁאֵין שָׁם שִׁכְרוּת — נוֹשֵׂא כַּפָּיו וְאֵין מוֹרִידִין אוֹתוֹ, מִפְּנֵי הַחֲשָׁד (סעיף ב).

המסקנה. הוֹאִיל וְאִם עָלָה לֹא יֵרֵד — נִקְרָא קְצָת רָאוּי לִנְשִׂיאַת כַּפַּיִם ; לְכָךְ אוֹמְרִים ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ בָּרְכֵנוּ״. עיין סימן ק״ל · 130 (המשך דיני הכהנים).

הפניה ללימוד. היקף הסימן — זְמַן נְשִׂיאַת כַּפַּיִם, גְּזֵרַת מִנְחָה וְדִין הַכֹּהֵן שֶׁעָבַר וְעָלָה — נוגע להנהגת הַכֹּהֵן בַּתְּפִלּוֹת הַשּׁוֹנוֹת. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קכ״ט. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מב׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
זְמַן נְשִׂיאַת כַּפַּיִם וְטַעַם הַשִּׁכְרוּת או״ח קכ״ט — נְשִׂיאַת כַּפַּיִם בְּשַׁחֲרִית מוּסָף וּנְעִילָה, לֹא בְּמִנְחָה מִשּׁוּם שֶׁכִּיחָא שִׁכְרוּת. מ״ב קכ״ט — גֶּדֶר חֲשַׁשׁ הַשִּׁכְרוּת שֶׁבְּשָׁעַת הַמִּנְחָה. ערוה״ש קכ״ט — טַעַם מְנִיעַת נְשִׂיאַת כַּפַּיִם בְּמִנְחָה. דרך אדמו״ר הזקן — אֵין נְשִׂיאַת כַּפַּיִם אֶלָּא בְּשַׁחֲרִית וּמוּסָף וּנְעִילָה, אֲבָל לֹא בְּמִנְחָה מִפְּנֵי שֶׁהַשִּׁכְרוּת מְצוּיָה בְּאוֹתָהּ שָׁעָה (סעיף א).
גְּזֵרַת מִנְחַת תַּעֲנִית וְהַחִלּוּק או״ח קכ״ט — גָּזְרוּ בְּמִנְחָה שֶׁל תַּעֲנִית אַטּוּ מִנְחָה שֶׁל שְׁאָר יָמִים ; תַּעֲנִית שֶׁאֵין בָּהּ נְעִילָה יֵשׁ בָּהּ נְשִׂיאַת כַּפַּיִם. מ״ב קכ״ט — פְּרָטֵי הַתַּעֲנִיּוֹת שֶׁיֵּשׁ וְשֶׁאֵין בָּהֶן נְעִילָה. ערוה״ש קכ״ט — הַהַכְרָעָה בְּמִנְחַת תַּעֲנִית לְעִנְיַן נְשִׂיאַת כַּפַּיִם. דרך אדמו״ר הזקן — גְּזֵרָה מִשּׁוּם מִנְחָה שֶׁל שְׁאָר יָמִים (מִתְפַּלְלִים בְּעוֹד הַיּוֹם גָּדוֹל) ; אֲבָל תַּעֲנִית שֶׁאֵין בָּהּ נְעִילָה סָמוּךְ לִשְׁקִיעָה — יֵשׁ בָּהּ נְשִׂיאַת כַּפַּיִם (סעיף א).
עָבַר וְעָלָה וּמִנְהַג ״אֱלֹהֵינוּ״ או״ח קכ״ט — עָבַר וְעָלָה בְּמִנְחַת יוֹם הַכִּפּוּרִים אֵין מוֹרִידִין אוֹתוֹ ; רמ״א — אוֹמְרִים ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״. מ״ב קכ״ט — טַעַם הַחֲשָׁד שֶׁלֹּא יֹאמְרוּ פָּסוּל הוּא. ערוה״ש קכ״ט — גֶּדֶר ״קְצָת רָאוּי״ לִנְשִׂיאַת כַּפַּיִם. דרך אדמו״ר הזקן — אֵין מוֹרִידִין אוֹתוֹ מִפְּנֵי הַחֲשָׁד ; וְלָכֵן אוֹמְרִים ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ בָּרְכֵנוּ״, שֶׁנִּקְרָא קְצָת רָאוּי (סעיף ב).

הטבלה נערכה מטקסט המחבר (אורח חיים קכ״ט, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה טו ו-יד, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד תענית כ״ו, מגילה כ״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ב׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — זְמַן נְשִׂיאַת כַּפַּיִם

קו א — זְמַן נְשִׂיאַת כַּפַּיִם

יסוד הסימן : נְשִׂיאַת כַּפַּיִם בְּשַׁחֲרִית מוּסָף וּנְעִילָה ; לֹא בְּמִנְחָה (סעיף א).

קריאת הרב : הַטַּעַם הוּא חֲשַׁשׁ הַשִּׁכְרוּת שֶׁל שְׁעַת הַמִּנְחָה, לֹא טֶבַע הַתְּפִלָּה. עיין סימן קכ״ח · 128.

מסקנה למעשה : בְּשַׁחֲרִית וּמוּסָף (וּנְעִילָה) — נְשִׂיאַת כַּפַּיִם ; בְּמִנְחַת יוֹם רָגִיל — לֹא.

קו ב — גְּזֵרָה וְחִלּוּק הַתַּעֲנִיּוֹת

קו ב — גְּזֵרָה וְחִלּוּק הַתַּעֲנִיּוֹת

יסוד הסימן : מִנְחַת תַּעֲנִית שֶׁיֵּשׁ בָּהּ נְעִילָה — נֶאֶסְרָה אַטּוּ מנחה של שאר ימים ; שֶׁאֵין בָּהּ נְעִילָה — יֵשׁ בָּהּ נְשִׂיאַת כַּפַּיִם (סעיף א).

קריאת הרב : הַמַּפְתֵּחַ הוּא שְׁעַת הַתְּפִלָּה : בְּעוֹד הַיּוֹם גָּדוֹל (מִתְחַלֶּפֶת) מוּל סָמוּךְ לִשְׁקִיעָה (דּוֹמָה לִנְעִילָה). עיין סימן קכ״ח · 128.

מסקנה למעשה : אֵין נְשִׂיאַת כַּפַּיִם בְּמִנְחַת יוֹם הַכִּפּוּרִים ; יֵשׁ בְּמִנְחַת תַּעֲנִית שֶׁאֵין בָּהּ נְעִילָה, סָמוּךְ לִשְׁקִיעָה.

קו ג — חֲשָׁד וּמִנְהַג ״אֱלֹהֵינוּ״

קו ג — חֲשָׁד וּמִנְהַג ״אֱלֹהֵינוּ״

יסוד הסימן : עָבַר וְעָלָה בְּמִנְחַת יוֹם הַכִּפּוּרִים — אֵין מוֹרִידִין אוֹתוֹ מִפְּנֵי הַחֲשָׁד (סעיף ב).

קריאת הרב : הוֹאִיל וְאִם עָלָה לֹא יֵרֵד — נִקְרָא קְצָת רָאוּי, וְזֶה מְיַסֵּד אֶת מִנְהַג אֲמִירַת ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ״. עיין סימן ק״ל · 130.

מסקנה למעשה : בְּמִנְחַת יוֹם הַכִּפּוּרִים אוֹמֵר הַשְּׁלִיחַ צִבּוּר ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ בָּרְכֵנוּ״.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קכ״ט ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קכ״ט (בְּאֵיזֶה תְּפִלּוֹת נוֹשְׂאִים כַּפַּיִם)

  • ① שַׁחֲרִית וּמוּסָף. נְשִׂיאַת כַּפַּיִם בְּשַׁחֲרִית וּמוּסָף ; וּבִנְעִילָה בְּיוֹם שֶׁיֵּשׁ בּוֹ נְעִילָה (יוֹם הַכִּפּוּרִים). יסוד : שוע״ר קכ״ט סעיף א.
  • ② מִנְחָה שֶׁל שְׁאָר יָמִים. אֵין נְשִׂיאַת כַּפַּיִם בְּמִנְחַת יוֹם רָגִיל — מִפְּנֵי שֶׁהַשִּׁכְרוּת מְצוּיָה בְּאוֹתָהּ שָׁעָה. יסוד : שוע״ר קכ״ט סעיף א.
  • ③ מִנְחַת יוֹם הַכִּפּוּרִים. אֵין נְשִׂיאַת כַּפַּיִם בְּמִנְחַת יוֹם הַכִּפּוּרִים וּתַעֲנִית שֶׁיֵּשׁ בָּהּ נְעִילָה — גְּזֵרָה אַטּוּ מנחה של שאר ימים. יסוד : שוע״ר קכ״ט סעיף א.
  • ④ תַּעֲנִית שֶׁאֵין בָּהּ נְעִילָה. מִנְחָתָהּ (סָמוּךְ לִשְׁקִיעָה) דּוֹמָה לִנְעִילָה — יֵשׁ בָּהּ נְשִׂיאַת כַּפַּיִם. יסוד : שוע״ר קכ״ט סעיף א.
  • ⑤ עָבַר וְעָלָה וְ״אֱלֹהֵינוּ״. כֹּהֵן שֶׁעָלָה בְּמִנְחַת יוֹם הַכִּפּוּרִים — אֵין מוֹרִידִין אוֹתוֹ (חֲשָׁד) ; וְאוֹמְרִים ״אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ בָּרְכֵנוּ״ (קְצָת רָאוּי). יסוד : שוע״ר קכ״ט סעיף ב.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קכ״ט (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — זְמַן נְשִׂיאַת כַּפַּיִם, גְּזֵרַת מִנְחָה, דִּין הַכֹּהֵן שֶׁעָבַר וְעָלָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
