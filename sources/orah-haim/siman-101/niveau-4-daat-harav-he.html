דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״א · כַּוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה

דרכו של אדמו״ר הזקן בכַּוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה : לְכַוֵּן בְּכָל הַבְּרָכוֹת — לְפָחוֹת בְּ״אָבוֹת״ ; מְחַתֵּךְ בִּשְׂפָתָיו וּמַשְׁמִיעַ לְאָזְנָיו בְּלַחַשׁ ; הַגְבָּהַת הַקּוֹל וּשְׁלִיחַ צִבּוּר ; הִרְהוּר אֵינוֹ כְּדִבּוּר ; וּלְשׁוֹן הַתְּפִלָּה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״א — כַּוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה. אצל הרב הסימן נפרש לה׳ סְעִיפִים : הַכַּוָּנָה בַּבְּרָכוֹת, הַמְחַתֵּךְ בִּשְׂפָתָיו וְלַחַשׁ, הַגְבָּהַת הַקּוֹל וְהַשְּׁלִיחַ צִבּוּר, הִרְהוּר אֵינוֹ כְּדִבּוּר, וּלְשׁוֹן הַתְּפִלָּה (יָחִיד וְצִבּוּר).

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״א — ה׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״א). אדמו״ר הזקן דן בו בְּכַוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״א — כַּוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״א — כַּוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה — וּבוֹ ה׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.101), ואחריו תרגומנו. הסימן כולל ה׳ סעיפים (סעיף k → ‎.101.k).

סעיף א כָּל הַמִּתְפַּלֵּל צָרִיךְ לְכַוֵּן ; לְפָחוֹת בְּ״אָבוֹת״.

שֶׁצָּרִיךְ לְכַוֵּן בְּכָל הַבְּרָכוֹת וְשֶׁיָּכוֹל לְהִתְפַּלֵּל בְּכָל לָשׁוֹן וּבוֹ ה׳ סְעִיפִים:הַמִּתְפַּלֵּל צָרִיךְ שֶׁיְּכַוֵּן בְּכָל הַבְּרָכוֹת. וְאִם אֵינוֹ יָכוֹל לְכַוֵּן בְּכֻלָּן — לְפָחוֹת יְכַוֵּן בְּ״אָבוֹת״. וּלְעוֹלָם יָמֹד אָדָם אֶת עַצְמוֹ, אִם יָכוֹל לְכַוֵּן לִבּוֹ בְּ״אָבוֹת״ — יִתְפַּלֵּל, וְאִם לָאו — אַל יִתְפַּלֵּל עַד שֶׁתִּתְיַשֵּׁב דַּעְתּוֹ וְיוּכַל לְכַוֵּן בְּ״אָבוֹת״ לְפָחוֹת. וְאִם לֹא כִּוֵּן בְּ״אָבוֹת״ אַף עַל פִּי שֶׁכִּוֵּן בְּכָל הַשְּׁאָר — צָרִיךְ לַחֲזֹר וּלְהִתְפַּלֵּל. וְעַכְשָׁו אֵין חוֹזְרִין בִּשְׁבִיל חֶסְרוֹן כַּוָּנָה, שֶׁאַף בַּחֲזָרָה קָרוֹב הוּא שֶׁלֹּא יְכַוֵּן, אִם כֵּן לָמָּה יַחֲזֹר. אֲבָל קְרִיאַת שְׁמַע שֶׁעִקַּר מִצְוַת כַּוָּנָתָהּ אֵינָהּ אֶלָּא פָּסוּק רִאשׁוֹן — חוֹזְרִין, שֶׁבְּקַל יוּכַל לְכַוֵּן. וְכֵן פָּסוּק ״פּוֹתֵחַ אֶת יָדֶךָ״ שֶׁבִּפְסוּקֵי דְּזִמְרָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן נ״א (וס״א) [וס״ג]. וְיַרְגִּיל אָדָם אֶת עַצְמוֹ שֶׁיְּכַוֵּן לְפָחוֹת בַּחֲתִימַת כָּל בְּרָכָה, שֶׁיֵּשׁ בָּהֶן קי״ג תֵּבוֹת כְּמוֹ שֶׁיֵּשׁ בִּתְפִלַּת חַנָּה, וְקי״ג פְּעָמִים לֵב בְּה׳ חֻמְשֵׁי תּוֹרָה, לוֹמַר שֶׁצָּרִיךְ כַּוָּנַת הַלֵּב:

כַּוָּנָה בַּבְּרָכוֹת. הַמִּתְפַּלֵּל צָרִיךְ לְכַוֵּן בְּכָל הַבְּרָכוֹת ; אֵינוֹ יָכוֹל בְּכֻלָּן — לְפָחוֹת בְּ״אָבוֹת״. יָמֹד אָדָם אֶת עַצְמוֹ : יָכוֹל לְכַוֵּן בְּ״אָבוֹת״ — יִתְפַּלֵּל, וְאִם לָאו — יַמְתִּין עַד שֶׁתִּתְיַשֵּׁב דַּעְתּוֹ. לֹא כִּוֵּן בְּ״אָבוֹת״ — מִן הַדִּין חוֹזֵר, אֲבָל עַכְשָׁו אֵין חוֹזְרִין בִּשְׁבִיל חֶסְרוֹן כַּוָּנָה. ק״ש שֶׁעִקַּר כַּוָּנָתָהּ פָּסוּק רִאשׁוֹן — חוֹזְרִין (וְכֵן ״פּוֹתֵחַ אֶת יָדֶךָ״, סִימָן נ״א / ס״ג). יַרְגִּיל לְכַוֵּן בַּחֲתִימָה — קי״ג תֵּבוֹת כְּמוֹ תְּפִלַּת חַנָּה, וְקי״ג פְּעָמִים ״לֵב״ בַּתּוֹרָה : כַּוָּנַת הַלֵּב.

סעיף ב מְחַתֵּךְ בִּשְׂפָתָיו וּמַשְׁמִיעַ לְאָזְנָיו בְּלַחַשׁ.

לֹא יִתְפַּלֵּל בְּלִבּוֹ לְבַד, אֶלָּא מְחַתֵּךְ הַדְּבָרִים בִּשְׂפָתָיו וּמַשְׁמִיעַ לְאָזְנָיו בְּלַחַשׁ. וְלֹא יַשְׁמִיעַ קוֹלוֹ (לְאַחֵר הָעוֹמֵד אֶצְלוֹ חוּץ לְד׳ אַמּוֹתָיו), שֶׁנֶּאֱמַר: ״רַק שְׂפָתֶיהָ נָעוֹת וְקוֹלָהּ לֹא יִשָּׁמֵעַ״. וְכָל הַמַּשְׁמִיעַ קוֹלוֹ בִּתְפִלָּתוֹ — הֲרֵי זֶה מִקְּטַנֵּי אֲמָנָה, כְּאִלּוּ אֵין הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵעַ תְּפִלַּת לַחַשׁ (וַחֲכָמִים תִּקְּנוּ תְּפִלָּה בְּלַחַשׁ לְגַמְרֵי שֶׁלֹּא לְהַשְׁמִיעַ אֶלָּא לְאָזְנָיו בִּלְבַד, כְּדֵי שֶׁלֹּא לְבַיֵּשׁ עוֹבְרֵי עֲבֵרָה שֶׁהֵם מִתְוַדִּים עַל עֲוֹנוֹתֵיהֶם בַּתְּפִלָּה):

מְחַתֵּךְ בִּשְׂפָתָיו. לֹא בְּלִבּוֹ לְבַד, אֶלָּא מְחַתֵּךְ הַדְּבָרִים בִּשְׂפָתָיו וּמַשְׁמִיעַ לְאָזְנָיו בְּלַחַשׁ, וְלֹא יַשְׁמִיעַ קוֹלוֹ — ״רַק שְׂפָתֶיהָ נָעוֹת וְקוֹלָהּ לֹא יִשָּׁמֵעַ״. הַמַּשְׁמִיעַ קוֹלוֹ — מִקְּטַנֵּי אֲמָנָה, כְּאִלּוּ אֵין הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵעַ תְּפִלַּת לַחַשׁ.

סעיף ג אֵינוֹ יָכוֹל לְכַוֵּן בְּלַחַשׁ — מֻתָּר לְהַגְבִּיהַּ ; בְּצִבּוּר אָסוּר.

וְאִם אֵינוֹ יָכוֹל לְכַוֵּן בְּלַחַשׁ, אוֹ שֶׁאֵינוֹ יָכוֹל לְכַוֵּן כָּל כָּךְ כְּמוֹ בְּקוֹל רָם — מֻתָּר לְהַגְבִּיהַּ קוֹלוֹ כְּדֵי לְכַוֵּן יָפֶה. וְהוּא שֶׁמִּתְפַּלֵּל בֵּינוֹ לְבֵין עַצְמוֹ, אֲבָל בְּצִבּוּר אָסוּר לְהַגְבִּיהַּ קוֹלוֹ, מִפְּנֵי שֶׁמַּטְרִיד הַצִּבּוּר. אֶלָּא אִם אֵינוֹ יָכוֹל לְכַוֵּן בְּלַחַשׁ — יֵלֵךְ וְיִתְפַּלֵּל בְּבֵיתוֹ בְּקוֹל. וַאֲפִלּוּ אִם יָכוֹל לְכַוֵּן בְּלַחַשׁ כְּמוֹ בְּקוֹל, אִם מַשְׁמִיעַ קוֹלוֹ כְּשֶׁמִּתְפַּלֵּל בְּבֵיתוֹ כְּדֵי שֶׁיִּלְמְדוּ מִמֶּנּוּ בְּנֵי בֵּיתוֹ נֻסַּח הַתְּפִלָּה — מֻתָּר. וְלָכֵן נוֹהֲגִין שֶׁבְּרֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר מַגְבִּיהִין קוֹלָם, כְּדֵי שֶׁיְּלַמְּדוּ זֶה אֶת זֶה, מִפְּנֵי שֶׁאֵין רְגִילִין בִּתְפִלּוֹת אֵלּוּ. וְעוֹד כְּדֵי לְעוֹרֵר הַכַּוָּנָה בְּיוֹתֵר, מִפְּנֵי שֶׁהֵם יְמֵי תְּשׁוּבָה, וְאֵין חוֹשְׁשִׁים שֶׁיַּטְרִידוּ וְיַטְעוּ זֶה אֶת זֶה כֵּיוָן שֶׁסִּדּוּרִים וּמַחֲזוֹרִים בְּיָדָם. וְאַף עַל פִּי כֵן יוֹתֵר טוֹב לְהִתְפַּלֵּל בְּלַחַשׁ אִם יָכוֹל לְכַוֵּן. וְעַל כָּל פָּנִים יִזָּהֲרוּ שֶׁלֹּא לְהַגְבִּיהַּ קוֹלָם יוֹתֵר מִדַּאי, שֶׁכָּל הַמַּגְבִּיהַּ קוֹלוֹ הַרְבֵּה — הֲרֵי זֶה מִנְּבִיאֵי הַשֶּׁקֶר, שֶׁנֶּאֱמַר בָּהֶם: ״וַיִּקְרְאוּ בְּקוֹל גָּדוֹל״. וּמִכָּל מָקוֹם, שְׁלִיחַ צִבּוּר הַמַּגְבִּיהַּ קוֹלוֹ כְּדֵי לְעוֹרֵר הַכַּוָּנָה וּלְהַשְׁמִיעַ יָפֶה לַבְּרִיּוֹת — יָפֶה הוּא עוֹשֶׂה, שֶׁעִקַּר תְּפִלַּת הַשְּׁלִיחַ צִבּוּר נִתְקְנָה בְּקוֹל רָם, אֲבָל אוֹתָם שֶׁעוֹשִׂים כֵּן לְהַרְאוֹת קוֹלָם — עוֹשִׂים שֶׁלֹּא כַּהֹגֶן:

הַגְבָּהַת הַקּוֹל. אֵינוֹ יָכוֹל לְכַוֵּן בְּלַחַשׁ — מֻתָּר לְהַגְבִּיהַּ קוֹלוֹ, וְהוּא בֵּינוֹ לְבֵין עַצְמוֹ ; בְּצִבּוּר אָסוּר מִפְּנֵי טִרְדַּת הַצִּבּוּר. בְּרֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר נוֹהֲגִין לְהַגְבִּיהַּ — לְלַמֵּד זֶה אֶת זֶה וּלְעוֹרֵר הַכַּוָּנָה — וּמִכָּל מָקוֹם טוֹב יוֹתֵר בְּלַחַשׁ. לֹא יַגְבִּיהַּ יוֹתֵר מִדַּאי — מִנְּבִיאֵי הַשֶּׁקֶר. שְׁלִיחַ צִבּוּר הַמַּגְבִּיהַּ לְעוֹרֵר כַּוָּנָה — יָפֶה עוֹשֶׂה.

סעיף ד הִתְפַּלֵּל בְּלִבּוֹ וְלֹא הוֹצִיא בִּשְׂפָתָיו — לֹא יָצָא.

הַמִּתְפַּלֵּל בְּלַחַשׁ וְלֹא הִשְׁמִיעַ לְאָזְנָיו — יָצָא. אֲבָל אִם הִתְפַּלֵּל בְּלִבּוֹ וְלֹא הוֹצִיא בִּשְׂפָתָיו — לֹא יָצָא, שֶׁהִרְהוּר אֵינוֹ כְּדִבּוּר:

הִרְהוּר אֵינוֹ כְּדִבּוּר. הִתְפַּלֵּל בְּלַחַשׁ וְלֹא הִשְׁמִיעַ לְאָזְנָיו — יָצָא ; הִתְפַּלֵּל בְּלִבּוֹ וְלֹא הוֹצִיא בִּשְׂפָתָיו — לֹא יָצָא, שֶׁהִרְהוּר אֵינוֹ כְּדִבּוּר.

סעיף ה בְּצִבּוּר בְּכָל לָשׁוֹן ; בְּיָחִיד בִּלְשׁוֹן הַקֹּדֶשׁ.

יָכוֹל לְהִתְפַּלֵּל בְּכָל לָשׁוֹן שֶׁיִּרְצֶה כְּשֶׁמִּתְפַּלֵּל בְּצִבּוּר. אֲבָל בְּיָחִיד לֹא יִתְפַּלֵּל אֶלָּא בִּלְשׁוֹן הַקֹּדֶשׁ, לְפִי שֶׁאֵין מַלְאֲכֵי הַשָּׁרֵת מַכִּירִין בִּשְׁאָר לְשׁוֹנוֹת, וְהַיָּחִיד צָרִיךְ לְמַלְאָךְ מֵלִיץ לְקַבֵּל תְּפִלָּתוֹ. אֲבָל צִבּוּר אֵין צָרִיךְ לְמֵלִיץ, שֶׁנֶּאֱמַר: ״הֵן אֵל כַּבִּיר לֹא יִמְאָס״. וְהַנָּשִׁים שֶׁנָּהֲגוּ לְהִתְפַּלֵּל בְּלַעַ״ז תְּפִלַּת י״ח שֶׁהֵן חַיָּבוֹת בָּהֶן כָּאֲנָשִׁים כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן (ק״י) [ק״ו] — יֵשׁ שֶׁלִּמְּדוּ עֲלֵיהֶן זְכוּת לוֹמַר, שֶׁלֹּא מָנְעוּ לַיָּחִיד לְהִתְפַּלֵּל בִּשְׁאָר לְשׁוֹנוֹת אֶלָּא כְּשֶׁשּׁוֹאֵל צְרָכָיו, כְּגוֹן שֶׁמִּתְפַּלֵּל עַל הַחוֹלֶה אוֹ עַל שׁוּם צַעַר שֶׁיֵּשׁ לוֹ בְּבֵיתוֹ, אֲבָל תְּפִלָּה הַקָּבוּעַ לְצִבּוּר — אֲפִלּוּ יָחִיד יָכוֹל לְאָמְרָהּ בְּכָל לָשׁוֹן. וְיֵשׁ שֶׁלִּמְּדוּ עֲלֵיהֶן זְכוּת אַחֵר לוֹמַר, שֶׁלֹּא מָנְעוּ לְהַיָּחִיד מִלְּהִתְפַּלֵּל בִּשְׁאָר לְשׁוֹנוֹת אֶלָּא מִלָּשׁוֹן אֲרַמִּי בִּלְבַד, שֶׁהוּא מְגֻנֶּה בְּעֵינֵי מַלְאֲכֵי הַשָּׁרֵת וְאֵינָם נִזְקָקִין לוֹ, אֲבָל מַכִּירִין וְיוֹדְעִים בְּכָל הַלְּשׁוֹנוֹת, אֲפִלּוּ מַחְשְׁבוֹת לִבּוֹ שֶׁל אָדָם. וְכָל זֶה לְלַמֵּד זְכוּת עַל הַנָּשִׁים שֶׁחַיָּבוֹת בִּתְפִלָּה, לָמָּה מִתְפַּלְלוֹת בְּלַעַ״ז וְאֵינָן לוֹמְדוֹת אֶת הַתְּפִלָּה בִּלְשׁוֹן הַקֹּדֶשׁ עִם פֵּרוּשׁ הַמִּלּוֹת, אֲבָל עַם הָאָרֶץ שֶׁאֵינוֹ מֵבִין בִּלְשׁוֹן הַקֹּדֶשׁ אַף עַל פִּי שֶׁיּוּכַל לְהִתְפַּלֵּל בִּלְשׁוֹן הַקֹּדֶשׁ — מוּטָב שֶׁיִּתְפַּלֵּל בְּלָשׁוֹן שֶׁמֵּבִין אֲפִלּוּ בְּיָחִיד כְּדֵי שֶׁיְּכַוֵּן לִבּוֹ מִמַּה שֶּׁיִּתְפַּלֵּל בִּלְשׁוֹן הַקֹּדֶשׁ בְּלֹא כַּוָּנַת הַלֵּב, שֶׁתְּפִלָּה בְּלֹא כַּוָּנָה — אֵינָהּ כְּלוּם:

לְשׁוֹן הַתְּפִלָּה. בְּצִבּוּר — בְּכָל לָשׁוֹן ; בְּיָחִיד — בִּלְשׁוֹן הַקֹּדֶשׁ, שֶׁאֵין מַלְאֲכֵי הַשָּׁרֵת מַכִּירִין שְׁאָר לְשׁוֹנוֹת וְהַיָּחִיד צָרִיךְ מֵלִיץ. לְלַמֵּד זְכוּת עַל הַנָּשִׁים (תְּפִלַּת י״ח, סִימָן ק״ו) : יֵשׁ אוֹמְרִים שֶׁלֹּא מָנְעוּ אֶלָּא בִּשְׁאֵלַת צְרָכָיו, וְיֵשׁ אוֹמְרִים אֶלָּא לָשׁוֹן אֲרַמִּי בִּלְבַד. וְעַם הָאָרֶץ — מוּטָב שֶׁיִּתְפַּלֵּל בְּלָשׁוֹן שֶׁמֵּבִין, שֶׁתְּפִלָּה בְּלֹא כַּוָּנָה אֵינָהּ כְּלוּם.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק״א (מהדורת קה״ת, ה׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — כַּוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה

היסוד : כַּוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב חמישה צירים. תחילה הַכַּוָּנָה בְּכָל הַבְּרָכוֹת (לְפָחוֹת בְּ״אָבוֹת״). אחר כך הַמְחַתֵּךְ בִּשְׂפָתָיו וְלַחַשׁ. אחר כך הַגְבָּהַת הַקּוֹל וְהַשְּׁלִיחַ צִבּוּר. אחר כך הִרְהוּר אֵינוֹ כְּדִבּוּר. ולבסוף לְשׁוֹן הַתְּפִלָּה בֵּין יָחִיד לְצִבּוּר.

כַּוָּנָה בַּבְּרָכוֹת — לְפָחוֹת בְּ״אָבוֹת״

כַּוָּנָה בַּבְּרָכוֹת — לְפָחוֹת בְּ״אָבוֹת״

היסוד. הַמִּתְפַּלֵּל צָרִיךְ לְכַוֵּן בְּכָל הַבְּרָכוֹת ; אֵינוֹ יָכוֹל — לְפָחוֹת בְּ״אָבוֹת״ (סעיף א).

המסקנה. לֹא כִּוֵּן בְּ״אָבוֹת״ — בַּזְּמַן הַזֶּה אֵין חוֹזְרִין. עיין סימן 98 (כוונה בתפילה).

מְחַתֵּךְ בִּשְׂפָתָיו וּמַשְׁמִיעַ בְּלַחַשׁ

מְחַתֵּךְ בִּשְׂפָתָיו וּמַשְׁמִיעַ בְּלַחַשׁ

היסוד. לֹא בְּלִבּוֹ לְבַד — מְחַתֵּךְ בִּשְׂפָתָיו וּמַשְׁמִיעַ לְאָזְנָיו בְּלַחַשׁ, וְלֹא יַשְׁמִיעַ קוֹלוֹ (״רַק שְׂפָתֶיהָ נָעוֹת״) (סעיף ב).

המסקנה. הַמַּשְׁמִיעַ קוֹלוֹ מִקְּטַנֵּי אֲמָנָה. עיין סימן 104 (הפסק בתפילה).

הַגְבָּהַת הַקּוֹל וּשְׁלִיחַ צִבּוּר

הַגְבָּהַת הַקּוֹל וּשְׁלִיחַ צִבּוּר

היסוד. אֵינוֹ יָכוֹל לְכַוֵּן בְּלַחַשׁ — מֻתָּר לְהַגְבִּיהַּ בֵּינוֹ לְבֵין עַצְמוֹ ; בְּצִבּוּר אָסוּר ; רֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר יוֹצְאִים מִן הַכְּלָל (סעיף ג).

המסקנה. לֹא לְהַגְבִּיהַּ יוֹתֵר מִדַּאי (מִנְּבִיאֵי הַשֶּׁקֶר) ; שְׁלִיחַ צִבּוּר הַמְעוֹרֵר כַּוָּנָה יָפֶה עוֹשֶׂה. עיין סימן 53 (שליח ציבור).

הִרְהוּר אֵינוֹ כְּדִבּוּר

היסוד. הִתְפַּלֵּל בְּלַחַשׁ בְּלֹא הַשְׁמָעָה — יָצָא ; בְּלִבּוֹ בְּלֹא שְׂפָתָיו — לֹא יָצָא (סעיף ד).

המסקנה. הִרְהוּר אֵינוֹ כְּדִבּוּר : הַתְּפִלָּה צְרִיכָה חִתּוּךְ שְׂפָתַיִם. עיין סימן 62 (הוצאה בשפתיים).

לְשׁוֹן הַתְּפִלָּה — יָחִיד וְצִבּוּר

לְשׁוֹן הַתְּפִלָּה — יָחִיד וְצִבּוּר

היסוד. בְּצִבּוּר בְּכָל לָשׁוֹן ; בְּיָחִיד בִּלְשׁוֹן הַקֹּדֶשׁ (הַמֵּלִיץ) ; זְכוּת לַנָּשִׁים וְלַעַם הָאָרֶץ (סעיף ה).

המסקנה. מוּטָב לָשׁוֹן שֶׁמֵּבִין מִלְּשׁוֹן הַקֹּדֶשׁ בְּלֹא כַּוָּנָה, שֶׁתְּפִלָּה בְּלֹא כַּוָּנָה אֵינָהּ כְּלוּם. עיין סימן 106 (חיוב הנשים).

הפניה ללימוד. היקף הסימן — הַכַּוָּנָה, הַלַּחַשׁ וְלְשׁוֹן הַתְּפִלָּה — נוגע להנהגת התְּפִלָּה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

כַּוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק״א. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מה׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
כַּוָּנָה בַּבְּרָכוֹת או״ח ק״א — צָרִיךְ לְכַוֵּן בְּכָל הַבְּרָכוֹת ; לְפָחוֹת בְּ״אָבוֹת״, וְעַכְשָׁו אֵין חוֹזְרִין. מ״ב ק״א — פְּרָטֵי הַכַּוָּנָה וְטַעַם ״אֵין חוֹזְרִין״ בַּזְּמַן הַזֶּה. ערוה״ש ק״א — גֶּדֶר הַכַּוָּנָה הַמְעַכֶּבֶת בְּ״אָבוֹת״. דרך אדמו״ר הזקן — לְכַוֵּן בְּכֻלָּן, לְפָחוֹת בְּ״אָבוֹת״ ; לֹא כִּוֵּן בְּ״אָבוֹת״ מִן הַדִּין חוֹזֵר, וְעַכְשָׁו אֵין חוֹזְרִין ; יַרְגִּיל בַּחֲתִימָה (קי״ג תֵּבוֹת) (סעיף א).
מְחַתֵּךְ בִּשְׂפָתָיו וְלַחַשׁ או״ח ק״א — מְחַתֵּךְ בִּשְׂפָתָיו וּמַשְׁמִיעַ לְאָזְנָיו בְּלַחַשׁ, וְלֹא יַשְׁמִיעַ קוֹלוֹ. מ״ב ק״א — שִׁעוּר הַלַּחַשׁ וְ״קוֹלָהּ לֹא יִשָּׁמֵעַ״. ערוה״ש ק״א — גֶּדֶר הַשְׁמָעָה לָאֹזֶן. דרך אדמו״ר הזקן — מְחַתֵּךְ בִּשְׂפָתָיו וּמַשְׁמִיעַ לְאָזְנָיו בְּלַחַשׁ ; הַמַּשְׁמִיעַ קוֹלוֹ מִקְּטַנֵּי אֲמָנָה (סעיף ב).
הַגְבָּהַת הַקּוֹל וּשְׁלִיחַ צִבּוּר או״ח ק״א — אֵינוֹ יָכוֹל לְכַוֵּן בְּלַחַשׁ מֻתָּר לְהַגְבִּיהַּ בֵּינוֹ לְבֵין עַצְמוֹ ; בְּצִבּוּר אָסוּר ; בְּר״ה וְיוֹם כִּפּוּר מֻתָּר. מ״ב ק״א — שֶׁלֹּא לְהַגְבִּיהַּ יוֹתֵר מִדַּאי, וְדִין הַצִּבּוּר. ערוה״ש ק״א — דִּין הַשְּׁלִיחַ צִבּוּר בְּקוֹל רָם. דרך אדמו״ר הזקן — מֻתָּר לְהַגְבִּיהַּ לְכַוֵּן וְהוּא בֵּינוֹ לְבֵין עַצְמוֹ ; בְּצִבּוּר אָסוּר ; ר״ה וְיוֹם כִּפּוּר נוֹהֲגִין ; שָׁ״ץ יָפֶה עוֹשֶׂה, וְלֹא לְהַרְאוֹת קוֹלוֹ (סעיף ג).
הִרְהוּר אֵינוֹ כְּדִבּוּר או״ח ק״א — מְחַתֵּךְ בִּשְׂפָתָיו דַּוְקָא ; הַרְהוּר בַּלֵּב אֵינוֹ מוֹצִיא. מ״ב ק״א — הִרְהוּר לָאו כְּדִבּוּר בִּתְפִלָּה. ערוה״ש ק״א — גֶּדֶר יְצִיאָה בְּלַחַשׁ בְּלֹא הַשְׁמָעָה. דרך אדמו״ר הזקן — הִתְפַּלֵּל בְּלַחַשׁ בְּלֹא הַשְׁמָעָה — יָצָא ; בְּלִבּוֹ בְּלֹא שְׂפָתָיו — לֹא יָצָא, הִרְהוּר אֵינוֹ כְּדִבּוּר (סעיף ד).
לְשׁוֹן הַתְּפִלָּה או״ח ק״א — בְּצִבּוּר בְּכָל לָשׁוֹן ; בְּיָחִיד בִּלְשׁוֹן הַקֹּדֶשׁ ; מַחְלֹקֶת בִּשְׁאֵלַת צְרָכָיו וּלְשׁוֹן אֲרַמִּי. מ״ב ק״א — פְּרָטֵי הַיָּחִיד וְהַצִּבּוּר וְהַלַּעַ״ז. ערוה״ש ק״א — טַעַם מַלְאֲכֵי הַשָּׁרֵת וְהַמֵּלִיץ. דרך אדמו״ר הזקן — צִבּוּר בְּכָל לָשׁוֹן, יָחִיד בִּלְשׁוֹן הַקֹּדֶשׁ (הַמֵּלִיץ) ; זְכוּת לַנָּשִׁים ; עַם הָאָרֶץ — מוּטָב בְּלָשׁוֹן שֶׁמֵּבִין, שֶׁתְּפִלָּה בְּלֹא כַּוָּנָה אֵינָהּ כְּלוּם (סעיף ה).

הטבלה נערכה מטקסט המחבר (אורח חיים ק״א, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ט״ו, ל״א ; סוטה ל״ב-ל״ג). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ה׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — כַּוָּנַת הַלֵּב

קו א — כַּוָּנַת הַלֵּב

יסוד הסימן : לְכַוֵּן בְּכָל הַבְּרָכוֹת, לְפָחוֹת בְּ״אָבוֹת״ (סעיף א).

קריאת הרב : הַכַּוָּנָה הִיא לֵב הַתְּפִלָּה ; יָמֹד אָדָם אֶת עַצְמוֹ, וְיַרְגִּיל בַּחֲתִימָה. עיין סימן 98.

מסקנה למעשה : לְכַוֵּן לְפָחוֹת בְּ״אָבוֹת״ ; בַּזְּמַן הַזֶּה אֵין חוֹזְרִין בִּשְׁבִיל חֶסְרוֹן כַּוָּנָה.

קו ב — קוֹל וְלַחַשׁ

קו ב — קוֹל וְלַחַשׁ

יסוד הסימן : מְחַתֵּךְ בִּשְׂפָתָיו וּמַשְׁמִיעַ לְאָזְנָיו בְּלַחַשׁ ; הִרְהוּר אֵינוֹ כְּדִבּוּר (סעיפים ב-ד).

קריאת הרב : אִזּוּן בֵּין הַלַּחַשׁ לַכַּוָּנָה ; מֻתָּר לְהַגְבִּיהַּ בֵּינוֹ לְבֵין עַצְמוֹ. עיין סימן 104.

מסקנה למעשה : לְחַתֵּךְ בַּשְּׂפָתַיִם וּלְהַשְׁמִיעַ לְעַצְמוֹ ; שֶׁלֹּא לְהַטְרִיד הַצִּבּוּר.

קו ג — לְשׁוֹן הַתְּפִלָּה

קו ג — לְשׁוֹן הַתְּפִלָּה

יסוד הסימן : בְּצִבּוּר בְּכָל לָשׁוֹן ; בְּיָחִיד בִּלְשׁוֹן הַקֹּדֶשׁ (סעיף ה).

קריאת הרב : לְעַם הָאָרֶץ הַכַּוָּנָה קוֹדֶמֶת לַלָּשׁוֹן — תְּפִלָּה בְּלֹא כַּוָּנָה אֵינָהּ כְּלוּם. עיין סימן 106.

מסקנה למעשה : מִי שֶׁאֵינוֹ מֵבִין לְשׁוֹן הַקֹּדֶשׁ — מוּטָב שֶׁיִּתְפַּלֵּל בְּלָשׁוֹן שֶׁמֵּבִין.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״א ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״א (כַּוָּנָה בַּבְּרָכוֹת וְלָשׁוֹן הַתְּפִלָּה)

  • ① כַּוָּנָה. לְכַוֵּן בְּכָל הַבְּרָכוֹת ; לְפָחוֹת בְּ״אָבוֹת״. בַּזְּמַן הַזֶּה אֵין חוֹזְרִין בִּשְׁבִיל חֶסְרוֹן כַּוָּנָה. יסוד : שוע״ר ק״א סעיף א.
  • ② לַחַשׁ. מְחַתֵּךְ בִּשְׂפָתָיו וּמַשְׁמִיעַ לְאָזְנָיו בְּלַחַשׁ, וְלֹא יַשְׁמִיעַ קוֹלוֹ לַאֲחֵרִים. יסוד : שוע״ר ק״א סעיף ב.
  • ③ הַגְבָּהַת הַקּוֹל. אֵינוֹ יָכוֹל לְכַוֵּן בְּלַחַשׁ — מֻתָּר לְהַגְבִּיהַּ בֵּינוֹ לְבֵין עַצְמוֹ ; בְּצִבּוּר אָסוּר (חוּץ מֵרֹאשׁ הַשָּׁנָה / יוֹם כִּפּוּר). יסוד : שוע״ר ק״א סעיף ג.
  • ④ הִרְהוּר. הִתְפַּלֵּל בְּלִבּוֹ בְּלֹא חִתּוּךְ שְׂפָתַיִם — לֹא יָצָא, שֶׁהִרְהוּר אֵינוֹ כְּדִבּוּר. יסוד : שוע״ר ק״א סעיף ד.
  • ⑤ לָשׁוֹן. בְּצִבּוּר בְּכָל לָשׁוֹן ; בְּיָחִיד בִּלְשׁוֹן הַקֹּדֶשׁ ; מִי שֶׁאֵינוֹ מֵבִין — מוּטָב בְּלָשׁוֹן שֶׁמֵּבִין, שֶׁתְּפִלָּה בְּלֹא כַּוָּנָה אֵינָהּ כְּלוּם. יסוד : שוע״ר ק״א סעיף ה.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק״א (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַכַּוָּנָה, הַלַּחַשׁ, הַלָּשׁוֹן — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
