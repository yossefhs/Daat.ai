דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן צ״ז · הַנְהָגַת הַגּוּף בִּתְפִלַּת הָעֲמִידָה

דרכו של אדמו״ר הזקן בהַנְהָגַת הַגּוּף בַּתְּפִלָּה : לֹא יְגַהֵק וְלֹא יְפַהֵק, הָרָק בִּתְפִלָּתוֹ — כְּאִלּוּ רָק בִּפְנֵי הַמֶּלֶךְ ; כִּנָּה עוֹקַצְתּוֹ, נִשְׁמַט טַלִּיתוֹ וְנוֹשֵׂא מַשּׂאוֹי — הַכֹּל נִמְדָּד בְּיְסוֹד אֶחָד : כְּאִלּוּ עוֹמֵד לִפְנֵי הַמֶּלֶךְ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן צ״ז — הַנְהָגַת הַגּוּף בַּתְּפִלָּה. אצל הרב הסימן נפרש לה׳ סְעִיפִים : לֹא יְגַהֵק וְלֹא יְפַהֵק, הָרְקִיקָה, הַכִּנָּה עוֹקַצְתּוֹ, הַנִּשְׁמַט טַלִּיתוֹ וְהַנּוֹשֵׂא מַשּׂאוֹי — הַכֹּל בְּיְסוֹד אֶחָד : כְּאִלּוּ עוֹמֵד לִפְנֵי הַמֶּלֶךְ.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן צ״ז — ה׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim צ״ז). אדמו״ר הזקן דן בו בְּהַנְהָגַת הַגּוּף בַּתְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן צ״ז — הַנְהָגַת הַגּוּף בַּתְּפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן צ״ז — הַנְהָגַת הַגּוּף בַּתְּפִלָּה — וּבוֹ ה׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.97), ואחריו תרגומנו. הסימן כולל ה׳ סעיפים (סעיף k → ‎.97.k).

סעיף א לֹא יְגַהֵק וְלֹא יְפַהֵק בִּשְׁעַת הַתְּפִלָּה ; מִתּוֹךְ אֹנֶס — יָדוֹ עַל פִּיו.

שֶׁלֹּא יְגַהֵק וְלֹא יְפַהֵק בִּשְׁעַת הַתְּפִלָּה וּבוֹ ה׳ סְעִיפִים:לֹא יְגַהֵק, דְּהַיְנוּ נְפִיחָה שֶׁאָדָם מוֹצִיא מִגּוּפוֹ לְפִיו מִתּוֹךְ שָׂבְעוֹ. וְלֹא יְפַהֵק, דְּהַיְנוּ שֶׁפּוֹתֵחַ פִּיו כְּאָדָם הָרוֹצֶה לִישֹׁן אוֹ שֶׁעָמַד מִשֵּׁנָה. וְאִם גִּהֵק אוֹ פִּהֵק לִרְצוֹנוֹ — הֲרֵי זֶה מִגַּסֵּי הָרוּחַ. אִם צָרִיךְ לְפַהֵק מִתּוֹךְ אֹנֶס — יַנִּיחַ יָדוֹ עַל פִּיו שֶׁלֹּא תֵּרָאֶה פְּתִיחָתוֹ. וְלֹא יַנִּיחַ יָדוֹ עַל סַנְטֵרוֹ, דְּהַיְנוּ לְחִי הַתַּחְתּוֹן, בִּשְׁעַת הַתְּפִלָּה, מִפְּנֵי שֶׁהוּא דֶּרֶךְ גַּסּוּת הָרוּחַ. וְחַזָּנִים הָעוֹשִׂים כֵּן בִּשְׁעַת הַזֶּמֶר — מֻתָּר, לְפִי שֶׁאֵינָם מִתְכַּוְּנִים לְגַסּוּת אֶלָּא לְהַנְעִים אֶת הַקּוֹל. אֲבָל שֶׁלֹּא בִּשְׁעַת הַזֶּמֶר — אָסוּר:

לֹא יְגַהֵק וְלֹא יְפַהֵק. אֵין מְגַהֵק (נְפִיחָה מִתּוֹךְ שֹׂבַע) וְאֵין מְפַהֵק (פּוֹתֵחַ פִּיו כְּרוֹצֶה לִישֹׁן) בִּשְׁעַת הַתְּפִלָּה ; הָעוֹשֶׂה לִרְצוֹנוֹ — מִגַּסֵּי הָרוּחַ. מִתּוֹךְ אֹנֶס — יַנִּיחַ יָדוֹ עַל פִּיו, וְלֹא עַל סַנְטֵרוֹ (דֶּרֶךְ גַּסּוּת). חַזָּנִים בִּשְׁעַת הַזֶּמֶר — מֻתָּר, שֶׁכַּוָּנָתָם לְהַנְעִים הַקּוֹל.

סעיף ב הָרָק בִּתְפִלָּתוֹ כְּאִלּוּ רָק בִּפְנֵי הַמֶּלֶךְ ; מַבְלִיעוֹ בִּכְסוּתוֹ.

הָרָק בִּתְפִלָּתוֹ — כְּאִלּוּ רָק בִּפְנֵי הַמֶּלֶךְ. לְפִיכָךְ, אָסוּר לָרֹק לִרְצוֹנוֹ שֶׁלֹּא מֵאֹנֶס אֲפִלּוּ עוֹמֵד בְּתַחֲנוּנִים שֶׁלְּאַחַר הַתְּפִלָּה, שֶׁכָּל זְמַן שֶׁלֹּא פָּסַע — הֲרֵי זֶה עוֹמֵד לִפְנֵי הַמֶּלֶךְ. אֲבָל אִם צָרִיךְ לָרֹק — מֻתָּר, כְּדֵי שֶׁלֹּא יִצְטַעֵר בַּתְּפִלָּה וְנִמְצָא טָרוּד. וּמִכָּל מָקוֹם, לֹא יָרֹק עַל גַּבֵּי קַרְקַע אֲפִלּוּ יֵשׁ שָׁם גֶּמִי אוֹ תֶּבֶן שֶׁיְּכַסּוּ הָרֹק, אֶלָּא מַבְלִיעוֹ בִּכְסוּתוֹ בְּעִנְיָן שֶׁלֹּא יְהֵא נִרְאֶה, כְּגוֹן בַּבֶּגֶד הַתַּחְתּוֹן, אֲבָל בַּבֶּגֶד הָעֶלְיוֹן שֶׁהָרֹק נִרְאֶה — אָסוּר. וּבִמְקוֹמוֹת שֶׁלּוֹבְשִׁין בַּחֲגוֹרָה חֲתִיכַת בֶּגֶד (שֶׁקּוֹרִין פציילט) — אֶפְשָׁר שֶׁמֻּתָּר לְהַבְלִיעַ בּוֹ אַף עַל פִּי שֶׁהָרֹק נִרְאֶה, שֶׁכֵּיוָן שֶׁהוּא עָשׂוּי לְכָךְ — אֵין בּוֹ מִשּׁוּם מִאוּס. וְאִם הוּא אִסְטְנִיס שֶׁמָּאוּס עָלָיו לְהַבְלִיעוֹ בִּכְסוּתוֹ, אוֹ שֶׁמָּאוּס עָלָיו לְהַשְׁהוֹתוֹ בְּפִיו עַד שֶׁיִּבְלָעֶנּוּ (אוֹ שֶׁהוּא כְּסוּת נָאֶה וְאִם נַצְרִיכֶנּוּ שֶׁיִּבְלָעֶנּוּ יַטְרִידֶנּוּ גַּם כֵּן) — זוֹרְקוֹ לַאֲחוֹרָיו, אֲבָל לֹא לְפָנָיו, מִפְּנֵי שֶׁהַמִּתְפַּלֵּל שְׁכִינָה לְפָנָיו. וּצְדָדִין הֵם כִּלְפָנָיו לְעִנְיָן זֶה. וְאִם אִי אֶפְשָׁר לוֹ לְזָרְקוֹ לַאֲחוֹרָיו — זוֹרְקוֹ לִשְׂמֹאלוֹ, אֲבָל לֹא לִימִינוֹ, מִשּׁוּם שֶׁנֶּאֱמַר: ״ה׳ צִלְּךָ עַל יַד יְמִינֶךָ״. וְאִם אִי אֶפְשָׁר לוֹ לָרֹק כִּי אִם לְפָנָיו — מֻתָּר כְּדֵי שֶׁלֹּא יִטָּרֵד:

הָרָק בִּתְפִלָּתוֹ. כְּאִלּוּ רָק בִּפְנֵי הַמֶּלֶךְ ; אָסוּר לָרֹק לִרְצוֹנוֹ, וְאִם צָרִיךְ — מַבְלִיעוֹ בִּכְסוּתוֹ (בֶּגֶד הַתַּחְתּוֹן). אִסְטְנִיס — זוֹרְקוֹ לַאֲחוֹרָיו, לֹא לְפָנָיו (שְׁכִינָה לְפָנָיו) ; וְאִם אִי אֶפְשָׁר — לִשְׂמֹאלוֹ, לֹא לִימִינוֹ.

סעיף ג כִּנָּה עוֹקַצְתּוֹ — יְמַשְׁמֵשׁ בִּבְגָדָיו, וְלֹא יְסִירֶנָּה בְּיָדוֹ.

אִם כִּנָּה עוֹקַצְתּוֹ — יְמַשְׁמֵשׁ בִּבְגָדָיו לַהֲסִירָהּ שֶׁלֹּא תִּתְבַּטֵּל כַּוָּנָתוֹ, אֲבָל לֹא יְסִירֶנָּה בְּיָדוֹ אֲפִלּוּ הִיא בְּמָקוֹם שֶׁאֵין שָׁם מִלְמוּלֵי זֵעָה (שֶׁאֵין צָרִיךְ נְטִילַת יָדַיִם לִתְפִלָּה, כְּמוֹ שֶׁאֵין צָרִיךְ אִם נָגַע בִּשְׁאָר בַּעֲלֵי חַיִּים טְמֵאִים), מִכָּל מָקוֹם מְאוּסָה הִיא וּגְנַאי הוּא אֲפִלּוּ אִם יִרְאֶנָּה אָדָם — קַל וָחֹמֶר כְּשֶׁעוֹמֵד בַּתְּפִלָּה לִפְנֵי מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא. אֲבָל שֶׁלֹּא בִּשְׁעַת תְּפִלַּת י״ח — יָכוֹל לִטֹּל כִּנָּה וּלְזָרְקָהּ בְּבֵית הַכְּנֶסֶת, כְּמוֹ שֶׁמֻּתָּר לָרֹק בְּבֵית הַכְּנֶסֶת. (וַאֲפִלּוּ נָגַע בְּמִלְמוּלֵי זֵעָה) — דַּי בְּנִקְיוֹן עָפָר לְהַזְכָּרַת הַשֵּׁם אוֹ לְדִבְרֵי תּוֹרָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ד׳ וצ״ב. אוֹ יִטְּלֶנָּה עַל יְדֵי בֶּגֶד, לְפִי שֶׁמִּשּׁוּם רוּחַ רָעָה שֶׁשּׁוֹרָה עַל הַיָּדַיִם הַנּוֹגְעִים בַּכִּנָּה — אֵין נִקָּיוֹן מוֹעִיל אֶלָּא נְטִילָה בְּמַיִם, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ד׳:

כִּנָּה עוֹקַצְתּוֹ. יְמַשְׁמֵשׁ בִּבְגָדָיו לַהֲסִירָהּ שֶׁלֹּא תִּתְבַּטֵּל כַּוָּנָתוֹ, וְלֹא יְסִירֶנָּה בְּיָדוֹ (מְאוּסָה וּגְנַאי — קַל וָחֹמֶר לִפְנֵי מֶלֶךְ מַלְכֵי הַמְּלָכִים). שֶׁלֹּא בַּתְּפִלָּה — יָכוֹל לִטֹּל וּלְזָרְקָהּ בְּבֵית הַכְּנֶסֶת.

סעיף ד נִשְׁמַט טַלִּיתוֹ — מַחֲזִירוֹ, אֲפִלּוּ נָפַל רֻבּוֹ ; נָפַל כֻּלּוֹ — הֶפְסֵק.

אִם נִשְׁמַט טַלִּיתוֹ מִמְּקוֹמוֹ — יָכוֹל לְמַשְׁמֵשׁ בּוֹ וּלְהַחֲזִירוֹ, אֲפִלּוּ נָפַל רֻבּוֹ. אֲבָל נָפַל כֻּלּוֹ — אֵינוֹ יָכוֹל לַחֲזֹר וּלְהִתְעַטֵּף בּוֹ, מִשּׁוּם הֶפְסֵק (אֶלָּא אִם כֵּן שֶׁהוּא טָרוּד עַל יְדֵי זֶה וְאֵינוֹ יָכוֹל לְכַוֵּן):

נִשְׁמַט טַלִּיתוֹ. יָכוֹל לְמַשְׁמֵשׁ בּוֹ וּלְהַחֲזִירוֹ, אֲפִלּוּ נָפַל רֻבּוֹ ; נָפַל כֻּלּוֹ — אֵינוֹ חוֹזֵר וּמִתְעַטֵּף, מִשּׁוּם הֶפְסֵק (אֶלָּא אִם טָרוּד וְאֵינוֹ יָכוֹל לְכַוֵּן).

סעיף ה נוֹשֵׂא מַשּׂאוֹי — פָּחוֹת מִד׳ קַבִּין מַפְשִׁילוֹ לַאֲחוֹרָיו ; ד׳ קַבִּין — עַל הַקַּרְקַע.

הַנּוֹשֵׂא מַשּׂאוֹי עַל כְּתֵפָיו וְהִגִּיעַ זְמַן הַתְּפִלָּה, פָּחוֹת מִד׳ קַבִּין — מַפְשִׁילוֹ לַאֲחוֹרָיו וּמִתְפַּלֵּל, וְלֹא יֹאחֲזֶנּוּ בְּיָדוֹ עַל כְּתֵפָיו (כְּדֵי שֶׁלֹּא יִהְיֶה טָרוּד בִּתְפִלָּתוֹ בַּאֲחִיזַת הַמַּשּׂאוֹי עַל כְּתֵפָיו בְּיָדוֹ וְלֹא יוּכַל לְכַוֵּן כָּרָאוּי, וְעוֹד לִפְנֵי מֶלֶךְ בָּשָׂר וָדָם אֵין עוֹמְדִין כֵּן — קַל וָחֹמֶר לִפְנֵי מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא). ד׳ קַבִּין — לֹא יַפְשִׁילֶנּוּ לַאֲחוֹרָיו, מִפְּנֵי שֶׁכָּבְדּוֹ מוֹנְעוֹ מִלְּכַוֵּן, אֶלָּא מַנִּיחוֹ עַל גַּבֵּי קַרְקַע עַד לְאַחַר הַתְּפִלָּה:

נוֹשֵׂא מַשּׂאוֹי. הִגִּיעַ זְמַן תְּפִלָּה : פָּחוֹת מִד׳ קַבִּין — מַפְשִׁילוֹ לַאֲחוֹרָיו וּמִתְפַּלֵּל, וְלֹא יֹאחֲזֶנּוּ בְּיָדוֹ. ד׳ קַבִּין — כָּבְדּוֹ מוֹנְעוֹ מִלְּכַוֵּן, אֶלָּא מַנִּיחוֹ עַל גַּבֵּי קַרְקַע עַד לְאַחַר הַתְּפִלָּה.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim צ״ז (מהדורת קה״ת, ה׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — הַנְהָגַת הַגּוּף בַּתְּפִלָּה

היסוד : הַנְהָגַת הַגּוּף בַּתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב הנהגת הגוף לפני המלך. תחילה הַגִּהוּק וְהַפִּהוּק. אחר כך הָרְקִיקָה. אחר כך הַכִּנָּה. ולבסוף הַטַּלִּית וְהַמַּשּׂאוֹי — הַרְחָקַת הַטִּרְדָּה וְהַהֶפְסֵק.

גִּהוּק וּפִהוּק — הַרְחָקַת גַּסּוּת הָרוּחַ

גִּהוּק וּפִהוּק — הַרְחָקַת גַּסּוּת הָרוּחַ

היסוד. לֹא יְגַהֵק וְלֹא יְפַהֵק בִּשְׁעַת הַתְּפִלָּה ; לִרְצוֹנוֹ — מִגַּסֵּי הָרוּחַ. אֹנֶס — יָדוֹ עַל פִּיו, וְלֹא עַל סַנְטֵרוֹ (סעיף א).

המסקנה. כָּל הַנְהָגַת הַגּוּף רְחוֹקָה מִגַּסּוּת ; חַזָּנִים בִּשְׁעַת הַזֶּמֶר מֻתָּר. עיין סימן 95.

רְקִיקָה — כְּאִלּוּ רָק בִּפְנֵי הַמֶּלֶךְ

רְקִיקָה — כְּאִלּוּ רָק בִּפְנֵי הַמֶּלֶךְ

היסוד. אָסוּר לָרֹק לִרְצוֹנוֹ ; אִם צָרִיךְ — מַבְלִיעוֹ בִּכְסוּתוֹ, בֶּגֶד הַתַּחְתּוֹן ; אִסְטְנִיס — לַאֲחוֹרָיו לֹא לְפָנָיו, וְלִשְׂמֹאלוֹ לֹא לִימִינוֹ (סעיף ב).

המסקנה. הַצְּנִיעוּת נוֹבַעַת מִ״שְׁכִינָה לְפָנָיו״ ; שׁוֹמְרִים עַל כְּבוֹד הַמֶּלֶךְ. עיין סימן 96.

כִּנָּה — מִאוּס וּגְנַאי

כִּנָּה — מִאוּס וּגְנַאי

היסוד. כִּנָּה עוֹקַצְתּוֹ — יְמַשְׁמֵשׁ בִּבְגָדָיו לַהֲסִירָהּ, וְלֹא יְסִירֶנָּה בְּיָדוֹ, מִשּׁוּם מְאוּסָה וּגְנַאי (סעיף ג).

המסקנה. שׁוֹמְרִים עַל הַכַּוָּנָה בְּלִי מַעֲשֶׂה גָּנַאי ; שֶׁלֹּא בַּתְּפִלָּה — נוֹטֵל וּמַשְׁלִיךְ בְּבֵית הַכְּנֶסֶת. עיין סימן 96.

טַלִּית וּמַשּׂאוֹי — הֶפְסֵק וְטִרְדָּה

טַלִּית וּמַשּׂאוֹי — הֶפְסֵק וְטִרְדָּה

היסוד. נִשְׁמַט טַלִּיתוֹ — מַחֲזִירוֹ אֲפִלּוּ נָפַל רֻבּוֹ ; נָפַל כֻּלּוֹ — הֶפְסֵק (סעיף ד). מַשּׂאוֹי : פָּחוֹת מִד׳ קַבִּין מַפְשִׁילוֹ לַאֲחוֹרָיו ; ד׳ קַבִּין — עַל הַקַּרְקַע (סעיף ה).

המסקנה. מְמַעֲטִים טִרְדָּה וְשׁוֹמְרִים כַּוָּנָה בְּלִי לַעֲשׂוֹת הֶפְסֵק. עיין סימן 95.

הפניה ללימוד. היקף הסימן — הָרְקִיקָה, הַפִּהוּק, הַכִּנָּה, הַטַּלִּית וְהַמַּשּׂאוֹי — נוגע להנהגת הגוף בשעת התפילה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

הַנְהָגַת הַגּוּף בַּתְּפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן צ״ז. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מה׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
גִּהוּק וּפִהוּק או״ח צ״ז — לֹא יְגַהֵק וְלֹא יְפַהֵק ; מִתּוֹךְ אֹנֶס יַנִּיחַ יָדוֹ עַל פִּיו. מ״ב צ״ז — פִּרְטֵי הַגִּהוּק וְהַפִּהוּק וְדֶרֶךְ גַּסּוּת הָרוּחַ. ערוה״ש צ״ז — גֶּדֶר גַּסּוּת הָרוּחַ בַּתְּפִלָּה. דרך אדמו״ר הזקן — לִרְצוֹנוֹ מִגַּסֵּי הָרוּחַ ; אֹנֶס — יָדוֹ עַל פִּיו, וְלֹא עַל סַנְטֵרוֹ ; חַזָּנִים בִּשְׁעַת הַזֶּמֶר מֻתָּר (סעיף א).
רְקִיקָה או״ח צ״ז — אָסוּר לָרֹק ; מַבְלִיעוֹ בִּכְסוּתוֹ, וְאִם אִי אֶפְשָׁר — לַאֲחוֹרָיו אוֹ לִשְׂמֹאלוֹ. מ״ב צ״ז — דִּינֵי הָרְקִיקָה וְהַצְּדָדִים. ערוה״ש צ״ז — הַהַבְלָעָה בַּכְּסוּת וְהַזְּרִיקָה. דרך אדמו״ר הזקן — כְּאִלּוּ רָק בִּפְנֵי הַמֶּלֶךְ ; מַבְלִיעוֹ בִּכְסוּתוֹ ; אִסְטְנִיס — לַאֲחוֹרָיו לֹא לְפָנָיו ; לִשְׂמֹאלוֹ לֹא לִימִינוֹ (סעיף ב).
כִּנָּה עוֹקַצְתּוֹ או״ח צ״ז — יְמַשְׁמֵשׁ בִּבְגָדָיו, וְלֹא יְסִירֶנָּה בְּיָדוֹ. מ״ב צ״ז — נְטִילַת הַכִּנָּה בַּתְּפִלָּה וְשֶׁלֹּא בָּהּ. ערוה״ש צ״ז — גֶּדֶר הַמִּאוּס וְהַכַּוָּנָה. דרך אדמו״ר הזקן — יְמַשְׁמֵשׁ בִּבְגָדָיו לַהֲסִירָהּ, וְלֹא בְּיָדוֹ (מְאוּסָה וּגְנַאי) ; שֶׁלֹּא בַּתְּפִלָּה — נוֹטֵל וּמַשְׁלִיךְ (סעיף ג).
נִשְׁמַט טַלִּיתוֹ או״ח צ״ז — נִשְׁמַט טַלִּיתוֹ מַחֲזִירוֹ ; נָפַל כֻּלּוֹ — הֶפְסֵק. מ״ב צ״ז — גֶּדֶר הַהֶפְסֵק בַּעֲטִיפָה. ערוה״ש צ״ז — הַחֲזָרַת הַטַּלִּית בַּתְּפִלָּה. דרך אדמו״ר הזקן — מַחֲזִירוֹ אֲפִלּוּ נָפַל רֻבּוֹ ; נָפַל כֻּלּוֹ — אֵינוֹ חוֹזֵר, מִשּׁוּם הֶפְסֵק (סעיף ד).
נוֹשֵׂא מַשּׂאוֹי או״ח צ״ז — פָּחוֹת מִד׳ קַבִּין מַפְשִׁילוֹ לַאֲחוֹרָיו ; ד׳ קַבִּין עַל הַקַּרְקַע. מ״ב צ״ז — שִׁעוּר ד׳ קַבִּין וְהַטִּרְדָּה. ערוה״ש צ״ז — גֶּדֶר הַמַּשּׂאוֹי בַּתְּפִלָּה. דרך אדמו״ר הזקן — פָּחוֹת מִד׳ קַבִּין מַפְשִׁילוֹ לַאֲחוֹרָיו, לֹא יֹאחֲזֶנּוּ בְּיָדוֹ ; ד׳ קַבִּין — עַל גַּבֵּי קַרְקַע (סעיף ה).

הטבלה נערכה מטקסט המחבר (אורח חיים צ״ז, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רי״ף ברכות, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ד). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ה׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — עוֹמֵד לִפְנֵי הַמֶּלֶךְ

קו א — עוֹמֵד לִפְנֵי הַמֶּלֶךְ

יסוד הסימן : הָרָק בִּתְפִלָּתוֹ כְּאִלּוּ רָק בִּפְנֵי הַמֶּלֶךְ ; לִפְנֵי מֶלֶךְ מַלְכֵי הַמְּלָכִים (סעיפים ב-ה).

קריאת הרב : צִיר הַסִּימָן הוּא הַהַכָּרָה שֶׁעוֹמֵד לִפְנֵי הַמֶּלֶךְ, הַמּוֹשֶׁלֶת בְּכָל תְּנוּעָה. עיין סימן 95.

מסקנה למעשה : לִמְדֹּד הַנְהָגַת הַגּוּף לְמַה שֶּׁגְּנַאי לִפְנֵי מֶלֶךְ בָּשָׂר וָדָם.

קו ב — הַרְחָקַת גַּסּוּת וּמִאוּס

קו ב — הַרְחָקַת גַּסּוּת וּמִאוּס

יסוד הסימן : גִּהוּק, פִּהוּק, סַנְטֵרוֹ וְכִנָּה — גַּסּוּת הָרוּחַ וּמִאוּס (סעיפים א, ג).

קריאת הרב : מַרְחִיקִים כָּל דָּבָר גַּס אוֹ מָאוּס, אֲבָל בְּלִי מַעֲשֵׂה יָד גָּנַאי. עיין סימן 96.

מסקנה למעשה : לְכַסּוֹת פִּהוּק אָנוּס בַּיָּד ; לְהָסִיר הַכִּנָּה עַל יְדֵי בֶּגֶד, לֹא בַּיָּד.

קו ג — שְׁמִירַת הַכַּוָּנָה מִן הַטִּרְדָּה

קו ג — שְׁמִירַת הַכַּוָּנָה מִן הַטִּרְדָּה

יסוד הסימן : טַלִּית וּמַשּׂאוֹי — שֶׁלֹּא יִהְיֶה טָרוּד וְלֹא יַעֲשֶׂה הֶפְסֵק (סעיפים ד-ה).

קריאת הרב : מַרְחִיקִים הַטִּרְדָּה כְּדֵי לְכַוֵּן, בְּלִי לַעֲשׂוֹת הֶפְסֵק. עיין סימן 98.

מסקנה למעשה : מַשָּׂא כָּבֵד (ד׳ קַבִּין) מַנִּיחוֹ עַל הַקַּרְקַע ; טַלִּית שֶׁנָּפַל כֻּלּוֹ מַחֲזִיר בְּמִדָּה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן צ״ז ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן צ״ז (הַנְהָגַת הַגּוּף בַּתְּפִלָּה)

  • ① גִּהוּק וּפִהוּק. לֹא יְגַהֵק וְלֹא יְפַהֵק לִרְצוֹנוֹ ; אָנוּס — יָדוֹ עַל פִּיו, וְלֹא עַל הַסַּנְטֵר. יסוד : שוע״ר צ״ז סעיף א.
  • ② רְקִיקָה. לֹא יָרֹק לִרְצוֹנוֹ ; אִם צָרִיךְ — מַבְלִיעוֹ בִּכְסוּתוֹ ; וְאִם לָאו לַאֲחוֹרָיו, אַחַר כָּךְ לִשְׂמֹאלוֹ, לֹא לִימִינוֹ וְלֹא לְפָנָיו. יסוד : שוע״ר צ״ז סעיף ב.
  • ③ כִּנָּה. כִּנָּה עוֹקַצְתּוֹ — יְמַשְׁמֵשׁ בִּבְגָדָיו, וְלֹא בְּיָדוֹ. שֶׁלֹּא בַּתְּפִלָּה — נוֹטֵל וּמַשְׁלִיךְ. יסוד : שוע״ר צ״ז סעיף ג.
  • ④ טַלִּית. נִשְׁמַט טַלִּיתוֹ — מַחֲזִירוֹ אֲפִלּוּ נָפַל רֻבּוֹ ; נָפַל כֻּלּוֹ — הֶפְסֵק, אֵינוֹ חוֹזֵר. יסוד : שוע״ר צ״ז סעיף ד.
  • ⑤ מַשּׂאוֹי. מַשָּׂא עַל כָּתֵף : פָּחוֹת מִד׳ קַבִּין מַפְשִׁילוֹ לַאֲחוֹרָיו ; ד׳ קַבִּין — מַנִּיחוֹ עַל הַקַּרְקַע. יסוד : שוע״ר צ״ז סעיף ה.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן צ״ז (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הָרְקִיקָה, הַכִּנָּה, הַמַּשּׂאוֹי — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
