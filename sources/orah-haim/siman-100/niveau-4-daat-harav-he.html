דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק׳ · תְּפִלַּת הַמּוֹעֵד צָרִיךְ לְסַדֵּר

דרכו של אדמו״ר הזקן בצֹרֶךְ לְהַסְדִּיר תְּפִלַּת הַמּוֹעֵד : תְּפִלָּה שֶׁל מוֹעֲדוֹת וְשֶׁל רָאשֵׁי חֳדָשִׁים — אֵין אָדָם רָגִיל בָּהֶן, לְפִיכָךְ צָרִיךְ לְהַסְדִּיר קֹדֶם שֶׁיִּתְפַּלֵּל, שֶׁתְּהֵא שְׁגוּרָה בְּפִיו ; וְעַכְשָׁו נוֹהֲגִים לְהָקֵל — מִתּוֹךְ הַסֵּפֶר אֵין צָרִיךְ ; וּמִכָּל מָקוֹם תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ — צָרִיךְ לְהַסְדִּיר, בִּפְרָט בְּפִיּוּטִים — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק׳ — תְּפִלַּת הַמּוֹעֵד צָרִיךְ לְסַדֵּר. אצל הרב הסימן הוא סְעִיף אֶחָד : תְּפִלָּה שֶׁל מוֹעֲדוֹת וְרָאשֵׁי חֳדָשִׁים — אֵין אָדָם רָגִיל בָּהֶן, לְפִיכָךְ צָרִיךְ הַסְדָּרָה תְּחִלָּה שֶׁתְּהֵא שְׁגוּרָה בְּפִיו — עִם דִּקְדּוּק הַזְּמַן (עַכְשָׁו נוֹהֲגִים לְהָקֵל, מִתּוֹךְ הַסֵּפֶר) וְהַסְּיָג שֶׁל תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ, בִּפְרָט בְּפִיּוּטִים.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק׳ — סְעִיף אֶחָד מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק׳). אדמו״ר הזקן דן בו בְּצֹרֶךְ לְהַסְדִּיר תְּפִלַּת הַמּוֹעֵד וְרֹאשׁ חֹדֶשׁ. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק׳ — תְּפִלַּת הַמּוֹעֵד צָרִיךְ לְסַדֵּר

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק׳ — תְּפִלַּת הַמּוֹעֵד צָרִיךְ לְסַדֵּר — וּבוֹ סְעִיף אֶחָד
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.100), ואחריו תרגומנו. הסימן כולל סעיף אחד (סְעִיף אֶחָד → ‎.100.1).

סעיף א צָרִיךְ לְהַסְדִּיר תְּפִלַּת הַמּוֹעֵד וְרֹאשׁ חֹדֶשׁ — שְׁגוּרָה בְּפִיו ; מִתּוֹךְ הַסֵּפֶר — אֵין צָרִיךְ.

תְּפִלַּת הַמּוֹעֵד צָרִיךְ לְסַדֵּר וּבוֹ סְעִיף אֶחָד:תְּפִלָּה שֶׁל מוֹעֲדוֹת וְשֶׁל רָאשֵׁי חֳדָשִׁים אֵין אָדָם רָגִיל בָּהֶן כָּל כָּךְ, מִפְּנֵי שֶׁאֵינוֹ מִתְפַּלֵּל אוֹתָן תָּדִיר אֶלָּא לִפְרָקִים, לְפִיכָךְ צָרִיךְ לְהַסְדִּיר אוֹתָן מִתְּחִלָּה קֹדֶם שֶׁיִּתְפַּלֵּל, כְּדֵי שֶׁתְּהֵא תְּפִלָּתוֹ שְׁגוּרָה בְּפִיו. וְעַכְשָׁו נוֹהֲגִים לְהָקֵל בָּזֶה. וְיֵשׁ שֶׁלִּמְּדוּ זְכוּת עֲלֵיהֶם לוֹמַר, שֶׁלֹּא הִצְרִיכוּ לְהַסְדִּיר אֶלָּא כְּשֶׁמִּתְפַּלְלִין בְּעַל פֶּה, אֲבָל כְּשֶׁמִּתְפַּלְלִין מִתּוֹךְ הַסֵּפֶר — אֵין צָרִיךְ לְהַסְדִּיר, שֶׁהֲרֵי רוֹאֶה מַה שֶּׁמִּתְפַּלֵּל. וּמִכָּל מָקוֹם, תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ הַדְּבָרִים וַדַּאי צָרִיךְ לִלְמֹד וּלְהַסְדִּיר תְּחִלָּה, שֶׁאֵין סֵפֶר מוֹעִיל לוֹ בָּזֶה, בִּפְרָט בְּפִיּוּטִים שֶׁחָמוּר פֵּרוּשָׁם:

תְּפִלַּת הַמּוֹעֵד צָרִיךְ לְסַדֵּר. תְּפִלָּה שֶׁל מוֹעֲדוֹת וְרָאשֵׁי חֳדָשִׁים — אֵין אָדָם רָגִיל בָּהֶן כָּל כָּךְ, שֶׁאֵינוֹ מִתְפַּלֵּל אוֹתָן תָּדִיר אֶלָּא לִפְרָקִים ; לְפִיכָךְ צָרִיךְ לְהַסְדִּיר אוֹתָן מִתְּחִלָּה, קֹדֶם שֶׁיִּתְפַּלֵּל, כְּדֵי שֶׁתְּהֵא תְּפִלָּתוֹ שְׁגוּרָה בְּפִיו. וְעַכְשָׁו נוֹהֲגִים לְהָקֵל בָּזֶה, וְיֵשׁ שֶׁלִּמְּדוּ זְכוּת : לֹא הִצְרִיכוּ הַסְדָּרָה אֶלָּא כְּשֶׁמִּתְפַּלְלִין בְּעַל פֶּה, אֲבָל מִתּוֹךְ הַסֵּפֶר — אֵין צָרִיךְ לְהַסְדִּיר, שֶׁהֲרֵי רוֹאֶה מַה שֶּׁמִּתְפַּלֵּל. וּמִכָּל מָקוֹם, תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ הַדְּבָרִים וַדַּאי צָרִיךְ לִלְמֹד וּלְהַסְדִּיר תְּחִלָּה, שֶׁאֵין סֵפֶר מוֹעִיל לוֹ בָּזֶה, בִּפְרָט בְּפִיּוּטִים שֶׁחָמוּר פֵּרוּשָׁם.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק׳ (מהדורת קה״ת, סְעִיף אֶחָד). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — תְּפִלַּת הַמּוֹעֵד צָרִיךְ לְסַדֵּר

היסוד : הַסְדָּרַת תְּפִלַּת הַמּוֹעֵד

סעיפו היחיד של אדמו״ר הזקן נפרש סביב ארבע נקודות. תחילה צֹרֶךְ הַהַסְדָּרָה — למה תפילת המועד צריכה הסדרה תחילה. אחר כך וְעַכְשָׁו נוֹהֲגִים לְהָקֵל — מנהג ההקלה בזמננו. אחר כך לִמּוּד הַזְּכוּת של המתפלל מִתּוֹךְ הַסֵּפֶר. ולבסוף הַסְּיָג של תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ, בִּפְרָט בְּפִיּוּטִים.

צֹרֶךְ הַהַסְדָּרָה — אֵין אָדָם רָגִיל בָּהֶן

צֹרֶךְ הַהַסְדָּרָה — אֵין אָדָם רָגִיל בָּהֶן

היסוד. תְּפִלָּה שֶׁל מוֹעֲדוֹת וְרָאשֵׁי חֳדָשִׁים אֵינָהּ תְּדִירָה אֶלָּא לִפְרָקִים ; אֵין אָדָם רָגִיל בָּהֶן כָּל כָּךְ, לְפִיכָךְ צָרִיךְ לְהַסְדִּיר אוֹתָן מִתְּחִלָּה (סעיף א).

המסקנה. הַתַּכְלִית — שֶׁתְּהֵא תְּפִלָּתוֹ שְׁגוּרָה בְּפִיו, בְּלִי גִּמְגּוּם שֶׁמְּבַלְבֵּל הַכַּוָּנָה. עיין רמה 1.

וְעַכְשָׁו נוֹהֲגִים לְהָקֵל

וְעַכְשָׁו נוֹהֲגִים לְהָקֵל

היסוד. הָרַב מְצַיֵּן בְּפֵרוּשׁ שֶׁוְּעַכְשָׁו נוֹהֲגִים לְהָקֵל בָּזֶה — בִּזְמַנֵּנוּ מְקִלִּין בַּהַסְדָּרָה הַמֻּקְדֶּמֶת (סעיף א).

המסקנה. הָרַב אֵינוֹ מִסְתַּפֵּק בְּצִיּוּן הַמִּנְהָג : הוּא מֵבִיא אֶת לִמּוּד הַזְּכוּת שֶׁבִּיסוֹדוֹ. עיין רמה 3.

מִתּוֹךְ הַסֵּפֶר — רוֹאֶה מַה שֶּׁמִּתְפַּלֵּל

מִתּוֹךְ הַסֵּפֶר — רוֹאֶה מַה שֶּׁמִּתְפַּלֵּל

היסוד. לֹא הִצְרִיכוּ הַסְדָּרָה אֶלָּא בְּעַל פֶּה ; אֲבָל מִתּוֹךְ הַסֵּפֶר — אֵין צָרִיךְ לְהַסְדִּיר, שֶׁהֲרֵי רוֹאֶה מַה שֶּׁמִּתְפַּלֵּל (סעיף א).

המסקנה. הַסִּדּוּר הַפָּתוּחַ מְסַלֵּק חֲשַׁשׁ גִּמְגּוּם : רְאִיַּת הַטֶּקְסְט בִּמְקוֹם הַסְדָּרָה. עיין רמה 2.

תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ — בְּפִיּוּטִים

תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ — בְּפִיּוּטִים

היסוד. וּמִכָּל מָקוֹם — תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ הַדְּבָרִים וַדַּאי צָרִיךְ לִלְמֹד וּלְהַסְדִּיר תְּחִלָּה, שֶׁאֵין סֵפֶר מוֹעִיל לוֹ בָּזֶה (סעיף א).

המסקנה. בִּפְרָט בְּפִיּוּטִים שֶׁחָמוּר פֵּרוּשָׁם : שָׁם אֵין הַסֵּפֶר מַסְפִּיק, וְצָרִיךְ לִמּוּד מֻקְדָּם שֶׁל הַפֵּרוּשׁ. עיין רמה 1.

הפניה ללימוד. היקף הסימן — צֹרֶךְ הַהַסְדָּרָה, מִנְהַג הַהֶקֵּל וְהַסְּיָג שֶׁל הַפִּיּוּטִים — נוגע להכנת תפילת המועדות וראשי חודשים. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק׳. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מסעיפו היחיד.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
חוֹבַת הַהַסְדָּרָה או״ח ק׳ — צָרִיךְ לְהַסְדִּיר תְּפִלָּתוֹ קֹדֶם שֶׁיִּתְפַּלֵּל, כְּדֵי שֶׁתְּהֵא ״שְׁגוּרָה בְּפִיו״. מ״ב ק׳ — טַעַם הַהַסְדָּרָה, שֶׁלֹּא תִּתְבַּלְבֵּל תְּפִלָּתוֹ בְּמַה שֶּׁאֵינוֹ רָגִיל. ערוה״ש ק׳ — גֶּדֶר חוֹבַת הַסְדָּרַת תְּפִלַּת מוֹעֵד וְרֹאשׁ חֹדֶשׁ. דרך אדמו״ר הזקן — אֵין אָדָם רָגִיל בָּהֶן (לִפְרָקִים) ; לְפִיכָךְ צָרִיךְ לְהַסְדִּיר מִתְּחִלָּה, שֶׁתְּהֵא שְׁגוּרָה בְּפִיו (סעיף א).
בְּעַל פֶּה אוֹ מִתּוֹךְ הַסֵּפֶר רמ״א ק׳ — וְיֵשׁ אוֹמְרִים דַּוְקָא בְּעַל פֶּה, אֲבָל מִתּוֹךְ הַסִּדּוּר מֻתָּר, שֶׁרוֹאֶה מַה שֶּׁמִּתְפַּלֵּל. מ״ב ק׳ — הַיּוֹם שֶׁמִּתְפַּלְּלִין מִתּוֹךְ הַסִּדּוּר, נָהֲגוּ לְהָקֵל. ערוה״ש ק׳ — הַהֶקֵּל בִּזְמַנֵּנוּ עַל פִּי הַסִּדּוּר. דרך אדמו״ר הזקן — וְעַכְשָׁו נוֹהֲגִים לְהָקֵל ; מִתּוֹךְ הַסֵּפֶר אֵין צָרִיךְ לְהַסְדִּיר, שֶׁהֲרֵי רוֹאֶה מַה שֶּׁמִּתְפַּלֵּל (סעיף א).
תְּפִלָּה שֶׁצָּרִיךְ פֵּרוּשׁ וּפִיּוּטִים או״ח ק׳ — עִקַּר הַדִּין: הַסְדָּרַת תְּפִלַּת מוֹעֵד וְרֹאשׁ חֹדֶשׁ (בְּלֹא פֵּרוּט הַפִּיּוּטִים). מ״ב ק׳ — מִצְוָה לְהָבִין פֵּרוּשׁ הַמִּלּוֹת. ערוה״ש ק׳ — הַהַקְפָּדָה עַל הֲבָנַת הַתְּפִלָּה. דרך אדמו״ר הזקן — וּמִכָּל מָקוֹם, תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ וַדַּאי צָרִיךְ לְהַסְדִּיר תְּחִלָּה, בִּפְרָט בְּפִיּוּטִים שֶׁחָמוּר פֵּרוּשָׁם (סעיף א).

הטבלה נערכה מטקסט המחבר ומהגהת הרמ״א (אורח חיים ק׳, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, מגן אברהם, ט״ז, משנה ברורה, ערוך השולחן, כף החיים). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (סְעִיף אֶחָד) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — צֹרֶךְ הַהַסְדָּרָה

קו א — צֹרֶךְ הַהַסְדָּרָה

יסוד הסימן : אֵין אָדָם רָגִיל בָּהֶן (מוֹעֲדוֹת וְרָאשֵׁי חֳדָשִׁים), לְפִיכָךְ צָרִיךְ לְהַסְדִּיר מִתְּחִלָּה, שֶׁתְּהֵא שְׁגוּרָה בְּפִיו (סעיף א).

קריאת הרב : הָרַב נוֹתֵן אֶת הַטַּעַם — חֶסְרוֹן הַהֶרְגֵּל (לִפְרָקִים, לֹא תָּדִיר) הוּא הַמְחַיֵּב אֶת הַהַסְדָּרָה. עיין רמה 1.

מסקנה למעשה : לְהַסְדִּיר מֵרֹאשׁ אֶת תְּפִלַּת הַמּוֹעֵד וְרֹאשׁ חֹדֶשׁ, שֶׁתִּזְרֹם בְּלִי גִּמְגּוּם.

קו ב — הַהֶקֵּל מִתּוֹךְ הַסֵּפֶר

קו ב — הַהֶקֵּל מִתּוֹךְ הַסֵּפֶר

יסוד הסימן : וְעַכְשָׁו נוֹהֲגִים לְהָקֵל ; מִתּוֹךְ הַסֵּפֶר אֵין צָרִיךְ לְהַסְדִּיר, שֶׁהֲרֵי רוֹאֶה מַה שֶּׁמִּתְפַּלֵּל (סעיף א).

קריאת הרב : הָרַב אֵינוֹ דּוֹחֶה אֶת הַמִּנְהָג — הוּא מְבָאֲרוֹ בְּלִמּוּד זְכוּת : הַהַסְדָּרָה נִצְרְכָה רַק בְּעַל פֶּה. עיין רמה 3.

מסקנה למעשה : הַמִּתְפַּלֵּל מִתּוֹךְ הַסִּדּוּר הַפָּתוּחַ פָּטוּר מֵהַסְדָּרָה בְּעַל פֶּה.

קו ג — הֲבָנַת הַפֵּרוּשׁ

קו ג — הֲבָנַת הַפֵּרוּשׁ

יסוד הסימן : וּמִכָּל מָקוֹם תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ הַדְּבָרִים וַדַּאי צָרִיךְ לְהַסְדִּיר תְּחִלָּה, שֶׁאֵין סֵפֶר מוֹעִיל לוֹ בָּזֶה (סעיף א).

קריאת הרב : הָרַב מַצִּיב גְּבוּל לַהֶקֵּל — רְאִיַּת הַטֶּקְסְט אֵינָהּ מַחְלִיפָה הֲבָנָה, בִּפְרָט בְּפִיּוּטִים שֶׁחָמוּר פֵּרוּשָׁם. עיין רמה 2.

מסקנה למעשה : בַּפִּיּוּטִים וּבַקְּטָעִים הַקָּשִׁים — לִלְמֹד אֶת הַפֵּרוּשׁ מֵרֹאשׁ, שֶׁאֵין הַסֵּפֶר לְבַדּוֹ מַסְפִּיק.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק׳ ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק׳ (תְּפִלַּת הַמּוֹעֵד צָרִיךְ לְסַדֵּר)

  • ① הַסְדָּרָה. לִפְנֵי הַמּוֹעֲדוֹת וְרָאשֵׁי חֳדָשִׁים — לְהַסְדִּיר אֶת הַתְּפִלָּה מִתְּחִלָּה, קֹדֶם שֶׁיִּתְפַּלֵּל, כְּדֵי שֶׁתְּהֵא שְׁגוּרָה בְּפִיו. יסוד : שוע״ר ק׳ סעיף א.
  • ② הַטַּעַם. מִפְּנֵי שֶׁאֵין אָדָם רָגִיל בָּהֶן — שֶׁאֵינוֹ מִתְפַּלֵּל אוֹתָן תָּדִיר אֶלָּא לִפְרָקִים. יסוד : שוע״ר ק׳ סעיף א.
  • ③ עַכְשָׁו לְהָקֵל. הָרַב מֵבִיא שֶׁוְעַכְשָׁו נוֹהֲגִים לְהָקֵל בָּזֶה, עִם לִמּוּד זְכוּת. יסוד : שוע״ר ק׳ סעיף א.
  • ④ מִתּוֹךְ הַסֵּפֶר. הַמִּתְפַּלֵּל מִתּוֹךְ הַסֵּפֶר — אֵין צָרִיךְ לְהַסְדִּיר, שֶׁהֲרֵי רוֹאֶה מַה שֶּׁמִּתְפַּלֵּל ; אֲבָל בְּעַל פֶּה — יַסְדִּיר. יסוד : שוע״ר ק׳ סעיף א.
  • ⑤ פֵּרוּשׁ וּפִיּוּטִים. תְּפִלָּה שֶׁצָּרִיךְ לְהָבִין פֵּרוּשׁ — וַדַּאי צָרִיךְ לְהַסְדִּיר תְּחִלָּה, בִּפְרָט בְּפִיּוּטִים שֶׁחָמוּר פֵּרוּשָׁם. יסוד : שוע״ר ק׳ סעיף א.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק׳ (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַהַסְדָּרָה, הַתְּפִלָּה מִתּוֹךְ הַסֵּפֶר, הֲבָנַת הַפֵּרוּשׁ — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
