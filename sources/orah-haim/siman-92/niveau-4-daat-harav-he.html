דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן צ״ב · הַנִּצְרָךְ לִנְקָבָיו וְדִין רְחִיצָה לַתְּפִלָּה

דרכו של אדמו״ר הזקן בהַנִּצְרָךְ לִנְקָבָיו וְדִין רְחִיצָה לַתְּפִלָּה : הַנִּצְרָךְ לִנְקָבָיו — אַל יִתְפַּלֵּל ; הַהֲכָנוֹת וְהַנְּטִילַת יָדַיִם ; הַמָּקוֹם מְטֻנָּף וְהַרְחָקַת ד׳ אַמּוֹת ; וְצְדָקָה קֹדֶם תְּפִלָּה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן צ״ב — הַנִּצְרָךְ לִנְקָבָיו וְדִין רְחִיצָה לַתְּפִלָּה. אצל הרב הסימן נפרש לי׳ סְעִיפִים : הַנִּצְרָךְ לִנְקָבָיו וּנְקִיּוּת הַגּוּף, הַהֲכָנוֹת וְהַנְּטִילַת יָדַיִם, הַמָּקוֹם מְטֻנָּף וְהַרְחָקַת ד׳ אַמּוֹת, וּצְדָקָה קֹדֶם תְּפִלָּה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן צ״ב — י׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim צ״ב). אדמו״ר הזקן דן בו בְּהַנִּצְרָךְ לִנְקָבָיו וְדִין רְחִיצָה לַתְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן צ״ב — הַנִּצְרָךְ לִנְקָבָיו וְדִין רְחִיצָה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן צ״ב — הַנִּצְרָךְ לִנְקָבָיו וְדִין רְחִיצָה לַתְּפִלָּה — וּבוֹ י׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.92), ואחריו תרגומנו. הסימן כולל י׳ סעיפים (סעיף k → ‎.92.k).

סעיף א הַנִּצְרָךְ לִנְקָבָיו — אַל יִתְפַּלֵּל ; הִתְפַּלֵּל — תְּפִלָּתוֹ תּוֹעֵבָה ; שִׁעוּר פַּרְסָה.

הַנִּצְרָךְ לִנְקָבָיו וְדִין רְחִיצָה וּשְׁאָר הֲכָנוֹת וּבוֹ י׳ סְעִיפִים:גּוּפוֹ כֵּיצַד? הָיָה צָרִיךְ לִנְקָבָיו אַל יִתְפַּלֵּל עַד שֶׁיִּפָּנֶה, מִשּׁוּם שֶׁנֶּאֱמַר: ״הִכּוֹן לִקְרַאת אֱלֹהֶיךָ יִשְׂרָאֵל״. וְאִם הִתְפַּלֵּל — תְּפִלָּתוֹ תּוֹעֵבָה וְצָרִיךְ לַחֲזֹר וּלְהִתְפַּלֵּל. בַּמֶּה דְּבָרִים אֲמוּרִים? בִּגְדוֹלִים, אֲבָל בִּקְטַנִּים — אֵין צָרִיךְ לַחֲזֹר וּלְהִתְפַּלֵּל. וַאֲפִלּוּ בִּגְדוֹלִים אֵינוֹ צָרִיךְ לַחֲזֹר אֶלָּא כְּשֶׁאֵינוֹ יָכוֹל לְהַעֲמִיד עַצְמוֹ שִׁעוּר הִלּוּךְ פַּרְסָה, אֲבָל אִם יָכוֹל לְהַעֲמִיד עַצְמוֹ שִׁעוּר הִלּוּךְ פַּרְסָה — יָצָא בְּדִיעֲבַד. אֲבָל לְכַתְּחִלָּה לֹא יִתְפַּלֵּל בְּגוּף מְשֻׁקָּץ, אֲפִלּוּ אִם יַעֲבֹר זְמַן הַתְּפִלָּה עַד שֶׁיִּפָּנֶה. וַאֲפִלּוּ בְּדִבְרֵי תּוֹרָה אָסוּר כָּל זְמַן שֶׁגּוּפוֹ מְשֻׁקָּץ. וְיֵשׁ מִי שֶׁמַּתִּיר אֲפִלּוּ לְכַתְּחִלָּה אִם יָכוֹל לְהַעֲמִיד עַצְמוֹ שִׁעוּר פַּרְסָה. וְיֵשׁ לִסְמֹךְ עַל זֶה אִם הוּא בְּעִנְיָן שֶׁבְּעוֹד שֶׁיִּפָּנֶה יַעֲבֹר זְמַן הַתְּפִלָּה:

הַנִּצְרָךְ לִנְקָבָיו. הַצָּרִיךְ לִנְקָבָיו אַל יִתְפַּלֵּל עַד שֶׁיִּפָּנֶה (״הִכּוֹן לִקְרַאת אֱלֹהֶיךָ יִשְׂרָאֵל״) ; הִתְפַּלֵּל — תְּפִלָּתוֹ תּוֹעֵבָה וְחוֹזֵר. בְּמָה דְּבָרִים אֲמוּרִים? בִּגְדוֹלִים ; וּבִקְטַנִּים אֵין צָרִיךְ לַחֲזֹר. וְאַף בִּגְדוֹלִים אֵינוֹ חוֹזֵר אֶלָּא כְּשֶׁאֵינוֹ יָכוֹל לְהַעֲמִיד עַצְמוֹ שִׁעוּר פַּרְסָה. וּלְכַתְּחִלָּה לֹא יִתְפַּלֵּל בְּגוּף מְשֻׁקָּץ, וְאַף בְּדִבְרֵי תּוֹרָה אָסוּר. וְיֵשׁ מַתִּיר לְכַתְּחִלָּה אִם יָכוֹל לְהַעֲמִיד עַצְמוֹ שִׁעוּר פַּרְסָה וְעוֹבֵר הַזְּמַן.

סעיף ב נִתְעוֹרֵר בְּאֶמְצַע הַתְּפִלָּה — מַעֲמִיד עַצְמוֹ ; מִשּׁוּם ״בַּל תְּשַׁקְּצוּ״.

וְכָל זֶה כְּשֶׁהֻצְרַךְ קֹדֶם הַתְחָלַת הַתְּפִלָּה, אֲבָל אִם בְּאֶמְצַע תְּפִלָּתוֹ נִתְעוֹרֵר לוֹ תַּאֲוָה בֵּין לִקְטַנִּים בֵּין לִגְדוֹלִים, אִם יָכוֹל לְהַעֲמִיד עַצְמוֹ עַד שֶׁיְּסַיֵּם תְּפִלָּתוֹ — יַעֲמִיד עַצְמוֹ וְלֹא יַפְסִיק אֲפִלּוּ לִקְטַנִּים בְּאֶמְצַע תְּפִלַּת י״ח. וְאִם אֵינוֹ יָכוֹל לְהַעֲמִיד עַצְמוֹ — יַפְסִיק לִגְדוֹלִים, אֲבָל לִקְטַנִּים לֹא יַפְסִיק. וַאֲפִלּוּ מַיִם שׁוֹתְתִין עַל בִּרְכָּיו — מַמְתִּין עַד שֶׁיִּכְלוּ הַמַּיִם וְחוֹזֵר לִתְפִלָּתוֹ, וְאֵין צָרִיךְ לְהַרְחִיק, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ע״ח. אֲבָל בְּאֶמְצַע קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ, אִם רָצָה — מַרְחִיק וּמֵטִיל מַיִם, וְאִם רָצָה — קוֹרֵא כְּדַרְכּוֹ וּמַעֲמִיד עַצְמוֹ עַד שֶׁיְּסַיֵּם תְּפִלָּתוֹ. אֲבָל לִגְדוֹלִים צָרִיךְ לְהַעֲמִיד עַצְמוֹ אִם יָכוֹל וְלֹא יַפְסִיק אֲפִלּוּ בִּקְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ (אוֹ אֲפִלּוּ בִּפְסוּקֵי דְּזִמְרָה מִ״בָּרוּךְ שֶׁאָמַר״ וְאֵילָךְ). וְהוּא שֶׁאֵינוֹ מִתְאַוֶּה כָּל כָּךְ בְּעִנְיָן שֶׁיְּהֵא בּוֹ מִשּׁוּם ״בַּל תְּשַׁקְּצוּ״ כְּשֶׁמַּעֲמִיד עַצְמוֹ, אַף עַל פִּי שֶׁאֵינוֹ יָכוֹל לְהַעֲמִיד עַצְמוֹ שִׁעוּר פַּרְסָה, אֲבָל אִם הוּא בְּעִנְיָן שֶׁיֵּשׁ בּוֹ מִשּׁוּם ״בַּל תְּשַׁקְּצוּ״ — מַפְסִיק בֵּין לִקְטַנִּים בֵּין לִגְדוֹלִים בְּאֶמְצַע קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ, אֲבָל לֹא בְּאֶמְצַע תְּפִלַּת י״ח, שֶׁלֹּא הִתִּירוּ לְהַפְסִיק בְּאֶמְצַע תְּפִלַּת י״ח אֶלָּא מִשּׁוּם חֲשַׁשׁ סַכָּנַת נְפָשׁוֹת בִּלְבַד וּכְמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ק״ד, וְקַל וָחֹמֶר מִשּׁוּם אִסּוּר ״בַּל תְּשַׁקְּצוּ״ שֶׁאֵינוֹ אֶלָּא מִדִּבְרֵי סוֹפְרִים, שֶׁהֵם אָמְרוּ וְהֵם אָמְרוּ (וְאִם מֵרֹב דָּחְקוֹ נִתְעַטֵּשׁ אוֹ שֶׁבִּקֵּשׁ לְהִתְעַטֵּשׁ — יִתְבָּאֵר בְּסִימָן (קי״ג) [ק״ג]):

נִתְעוֹרֵר בְּאֶמְצַע. נִתְעוֹרֵר בְּאֶמְצַע הַתְּפִלָּה — אִם יָכוֹל מַעֲמִיד עַצְמוֹ וְאֵינוֹ מַפְסִיק, אַף לִקְטַנִּים בְּאֶמְצַע תְּפִלַּת י״ח ; אֵינוֹ יָכוֹל — מַפְסִיק לִגְדוֹלִים בִּלְבַד. בִּקְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ — אִם רָצָה מֵטִיל מַיִם. וּמִשּׁוּם ״בַּל תְּשַׁקְּצוּ״ מַפְסִיק בַּקְּרִיאַת שְׁמַע, אֲבָל לֹא בִּתְפִלַּת י״ח (סימן ק״ד).

סעיף ג יִבְדֹּק עַצְמוֹ, יָסִיר כִּיחוֹ וְנִיעוֹ, וְיִרְחַץ יָדָיו — בְּלֹא בְּרָכָה.

וּמִכָּל מָקוֹם, לְכַתְּחִלָּה יִבְדֹּק אֶת עַצְמוֹ יָפֶה קֹדֶם הַתְּפִלָּה שֶׁלֹּא יִצְטָרֵךְ לִנְקָבָיו אַחַר כָּךְ וְיִטָּרֵד בִּתְפִלָּתוֹ. וְכֵן יָסִיר כִּיחוֹ וְנִיעוֹ וְכָל דָּבָר הַטּוֹרְדוֹ. וְאַחַר כָּךְ צָרִיךְ לִרְחֹץ יָדָיו בַּמַּיִם אַף עַל פִּי שֶׁנּוֹטֵל יָדָיו שַׁחֲרִית, כֵּיוָן שֶׁנָּגַע בְּטִנֹּפֶת אַחַר נְטִילַת שַׁחֲרִית. וּמִכָּל מָקוֹם, לֹא יְבָרֵךְ עַל רְחִיצָה זוֹ אִם כְּבָר בֵּרַךְ עַל נְטִילַת שַׁחֲרִית, מִטַּעַם שֶׁנִּתְבָּאֵר בְּסִימָן ז׳:

הֲכָנוֹת. לְכַתְּחִלָּה יִבְדֹּק עַצְמוֹ, יָסִיר כִּיחוֹ וְנִיעוֹ, וְיִרְחַץ יָדָיו אַף עַל פִּי שֶׁנָּטַל שַׁחֲרִית ; וְלֹא יְבָרֵךְ עַל רְחִיצָה זוֹ (סימן ז׳).

סעיף ד אֵין לוֹ מַיִם — עַד פַּרְסָה לְפָנָיו, מִיל לַאֲחוֹרָיו ; מְנַקֶּה בְּעָפָר.

וְאִם אֵין לוֹ מַיִם — צָרִיךְ לַחֲזֹר אַחֲרֵיהֶם עַד פַּרְסָה אִם הוּא בַּדֶּרֶךְ וְהַמַּיִם נִמְצָאִים לְפָנָיו בְּרִחוּק פַּרְסָה וְיָכוֹל לְהַגִּיעַ אֲלֵיהֶן קֹדֶם שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה. וְהוּא שֶׁהוֹלֵךְ יְחִידִי, אֲבָל אִם הוֹלֵךְ בְּשַׁיָּרָא — אֵין צָרִיךְ לְהִפָּרֵד מִן הַשַּׁיָּרָא לָלֶכֶת יְחִידִי לִמְקוֹם הַמַּיִם. וְאִם צָרִיךְ לַחֲזֹר לַאֲחוֹרָיו לִמְקוֹם מַיִם עַד מִיל — חוֹזֵר, יוֹתֵר מִמִּיל — אֵינוֹ צָרִיךְ לַחֲזֹר. וְאִם מִתְיָרֵא שֶׁמָּא יַעֲבֹר זְמַן הַתְּפִלָּה עַד שֶׁיַּגִּיעַ לִמְקוֹם הַמַּיִם — לֹא יַחֲזֹר אַחֲרֵיהֶם כְּלָל, רַק יְנַקֶּה יָדָיו בִּצְרוֹר אוֹ בְּעָפָר אוֹ בְּקוֹרָה אוֹ בְּכָל דָּבָר שֶׁמְּנַקֶּה:

אֵין לוֹ מַיִם. אֵין לוֹ מַיִם — חוֹזֵר עַד פַּרְסָה לְפָנָיו, וּבִשְׁיָרָא אֵינוֹ נִפְרָד ; לַאֲחוֹרָיו עַד מִיל. וְאִם עוֹבֵר הַזְּמַן — מְנַקֶּה יָדָיו בִּצְרוֹר, בְּעָפָר אוֹ בְּקוֹרָה.

סעיף ה הֶסַּח הַדַּעַת פּוֹסֵל ; עוֹמֵד בִּתְפִלָּה — עָפָר אוֹ מְחַכֵּךְ בַּכֹּתֶל.

וְכָל זֶה כְּשֶׁיָּדָיו מְלֻכְלָכוֹת אוֹ שֶׁנָּגְעוּ בְּמָקוֹם מְטֻנָּף, אֲבָל אִם אֵינוֹ יוֹדֵעַ לָהֶן שׁוּם לִכְלוּךְ אַף עַל פִּי שֶׁהִסִּיחַ דַּעְתּוֹ מֵהֶם אַחַר נְטִילַת שַׁחֲרִית — אֵין צָרִיךְ לַחֲזֹר וְלִטְּלָן וְלֹא לְנַקּוֹתָן, שֶׁסְּתָם יָדַיִם כְּשֵׁרוֹת לִתְפִלָּה כְּמוֹ שֶׁכְּשֵׁרוֹת לִשְׁאָר בְּרָכוֹת. וְיֵשׁ אוֹמְרִים שֶׁהֶסַּח הַדַּעַת פּוֹסֵל לִתְפִלָּה כְּמוֹ שֶׁפּוֹסֵל לַאֲכִילָה, מִפְּנֵי שֶׁהַיָּדַיִם עַסְקָנִיּוֹת הֵן וְשֶׁמָּא נָגְעוּ בִּמְקוֹם הַטִּנֹּפֶת. וְכֵן עִקָּר. וּמִכָּל מָקוֹם, אִם אֵין מַיִם מְזֻמָּנִים לְפָנָיו — אֵין צָרִיךְ לַחֲזֹר אַחֲרֵיהֶם אִם אֵינוֹ יוֹדֵעַ לָהֶם שׁוּם לִכְלוּךְ, שֶׁלְּעִנְיָן זֶה יֵשׁ לִסְמֹךְ עַל סְבָרָא הָרִאשׁוֹנָה, אֶלָּא יְנַקֶּה יָדָיו בְּכָל דָּבָר שֶׁמְּנַקֶּה, שֶׁנִּקָּיוֹן מוֹעִיל אַף לְיָדַיִם שֶׁל הֶסַּח הַדַּעַת, שֶׁפְּסוּל הֶסַּח הַדַּעַת הוּא מִשּׁוּם שֶׁמָּא מֻדְבָּק בָּהֵן מְעַט זֵעָה אוֹ צוֹאָה, וּכְשֶׁמְּנַקֶּה יִתְקַנְּחוּ. וַאֲפִלּוּ בְּיָדַיִם מְלֻכְלָכוֹת — אֵין צָרִיךְ לַחֲזֹר אַחַר מַיִם אֶלָּא כְּשֶׁאֶפְשָׁר, אֲבָל כְּשֶׁעוֹמֵד בִּתְפִלָּה שֶׁאִי אֶפְשָׁר לָזוּז מִמְּקוֹמוֹ וְנִזְכַּר שֶׁנָּגַע בְּמָקוֹם מְטֻנָּף — דַּי בְּנִקְיוֹן עָפָר אוֹ צְרוֹרוֹת, אוֹ מְחַכֵּךְ יָדָיו בַּכֹּתֶל. אֲבָל לְכַתְּחִלָּה אָסוּר לִגַּע בְּמָקוֹם מְטֻנָּף (אֲפִלּוּ יֵשׁ לוֹ מַיִם מְזֻמָּנִים לְפָנָיו, שֶׁלֹּא לְהַפְסִיק) בַּתְּפִלָּה, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן צ״ז:

הֶסַּח הַדַּעַת. יָדָיו מְלֻכְלָכוֹת אוֹ נָגְעוּ בְּמָקוֹם מְטֻנָּף ; אֲבָל אֵין לָהֶן לִכְלוּךְ — סְתָם יָדַיִם כְּשֵׁרוֹת. וְיֵשׁ אוֹמְרִים הֶסַּח הַדַּעַת פּוֹסֵל וְכֵן עִקָּר, וּמְנַקֶּה בְּכָל דָּבָר. וְעוֹמֵד בִּתְפִלָּה — דַּי בְּעָפָר אוֹ בַּכֹּתֶל ; וּלְכַתְּחִלָּה אָסוּר לִגַּע (סימן צ״ז).

סעיף ו לְתַלְמוּד תּוֹרָה — דַּי בְּנִקָּיוֹן בְּעָלְמָא.

וְכָל זֶה בִּתְפִלָּה, אֲבָל לְתַלְמוּד תּוֹרָה דַּי בְּנִקָּיוֹן בְּעָלְמָא בְּכָל עִנְיָן, אֲפִלּוּ יֵשׁ מַיִם מְזֻמָּנִים לְפָנָיו:

תַּלְמוּד תּוֹרָה. לְתַלְמוּד תּוֹרָה דַּי בְּנִקָּיוֹן בְּעָלְמָא, אֲפִלּוּ יֵשׁ מַיִם מְזֻמָּנִים.

סעיף ז מָקוֹם מְטֻנָּף — הַמְכֻסִּים ; צוֹאַת הָאֹזֶן וְהָאַף — עַל יְדֵי בֶּגֶד.

מָקוֹם הַמְטֻנָּף הֵם מְקוֹמוֹת הַמְכֻסִּים בָּאָדָם, לְפִי שֶׁיֵּשׁ בָּהֶם מִלְמוּלֵי זֵעָה. וְכֵן אִם חִכֵּךְ הָרֹאשׁ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קס״ד. וְכֵן צוֹאַת הָאֹזֶן וְצוֹאַת הָאַף הֵן טִנּוֹפוֹת, וְאָסוּר לִגַּע בָּהֶן בַּתְּפִלָּה כִּי אִם עַל יְדֵי בֶּגֶד:

מָקוֹם מְטֻנָּף. מְקוֹמוֹת הַמְכֻסִּים שֶׁיֵּשׁ בָּהֶם מִלְמוּלֵי זֵעָה, וְכֵן חִכּוּךְ הָרֹאשׁ (סימן קס״ד) ; צוֹאַת הָאֹזֶן וְהָאַף — אָסוּר לִגַּע כִּי אִם עַל יְדֵי בֶּגֶד.

סעיף ח הַמַּשְׁתִּין וְהַמִּתְפַּלֵּל — שִׁעוּר ד׳ אַמּוֹת.

הַמַּשְׁתִּין לֹא יִתְפַּלֵּל עַד שֶׁיִּשְׁהֶה כְּדֵי הִלּוּךְ ד׳ אַמּוֹת, מִשּׁוּם נִיצוֹצוֹת שֶׁבְּאַמָּתוֹ. וְהַמִּתְפַּלֵּל לֹא יַשְׁתִּין עַד שֶׁיִּשְׁהֶה כְּדֵי הִלּוּךְ ד׳ אַמּוֹת אַחַר סִיּוּם תְּפִלָּתוֹ, שֶׁכָּל ד׳ אַמּוֹת תְּפִלָּתוֹ שְׁגוּרָה בְּפִיו וּשְׂפָתָיו רוֹחֲשׁוֹת. אֲבָל לְאַחַר שֶׁשָּׁהָה כְּדֵי הִלּוּךְ ד׳ אַמּוֹת — מֻתָּר לוֹ לְהַשְׁתִּין בַּמָּקוֹם שֶׁהִתְפַּלֵּל. וּמִדַּת חֲסִידוּת שֶׁלֹּא לְהַשְׁתִּין בְּתוֹךְ ד׳ אַמּוֹת שֶׁל תְּפִלָּה:

הַמַּשְׁתִּין. הַמַּשְׁתִּין וְהַמִּתְפַּלֵּל — שִׁעוּר ד׳ אַמּוֹת, מִשּׁוּם נִיצוֹצוֹת וּמִשּׁוּם שֶׁתְּפִלָּתוֹ שְׁגוּרָה בְּפִיו ; וּמִדַּת חֲסִידוּת שֶׁלֹּא לְהַשְׁתִּין בְּתוֹךְ ד׳ אַמּוֹת.

סעיף ט הָרוֹקֵק — ד׳ אַמּוֹת ; נִזְדַּמֵּן לוֹ רֹק — מֻתָּר.

הָרוֹקֵק לֹא יִתְפַּלֵּל עַד שֶׁיִּשְׁהֶה כְּדֵי הִלּוּךְ ד׳ אַמּוֹת. וְהַמִּתְפַּלֵּל לֹא יָרֹק לִרְצוֹנוֹ מֵחֲמַת טִיּוּל עַד שֶׁיִּשְׁהֶה כְּדֵי הִלּוּךְ ד׳ אַמּוֹת. אֲבָל אִם נִזְדַּמֵּן לוֹ רֹק — מֻתָּר אֲפִלּוּ בְּתוֹךְ הַתְּפִלָּה, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן צ״ז:

הָרוֹקֵק. הָרוֹקֵק וְהַמִּתְפַּלֵּל — ד׳ אַמּוֹת, וְדַוְקָא לִרְצוֹנוֹ ; נִזְדַּמֵּן לוֹ רֹק — מֻתָּר אַף בַּתְּפִלָּה (סימן צ״ז).

סעיף י טוֹב לִתֵּן צְדָקָה קֹדֶם הַתְּפִלָּה.

טוֹב לִתֵּן צְדָקָה קֹדֶם הַתְּפִלָּה, שֶׁנֶּאֱמַר: ״אֲנִי בְּצֶדֶק אֶחֱזֶה פָנֶיךָ״:

צְדָקָה. טוֹב לִתֵּן צְדָקָה קֹדֶם הַתְּפִלָּה, שֶׁנֶּאֱמַר ״אֲנִי בְּצֶדֶק אֶחֱזֶה פָנֶיךָ״.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim צ״ב (מהדורת קה״ת, י׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — נְקִיּוּת הַגּוּף לַתְּפִלָּה

היסוד : נְקִיּוּת הַגּוּף קֹדֶם הַתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַנִּצְרָךְ לִנְקָבָיו וּנְקִיּוּת הַגּוּף. אחר כך הַהֲכָנוֹת וְהַנְּטִילַת יָדַיִם. אחר כך הַמָּקוֹם מְטֻנָּף וְהַרְחָקַת ד׳ אַמּוֹת. ולבסוף הַצְּדָקָה קֹדֶם תְּפִלָּה.

הַנִּצְרָךְ לִנְקָבָיו — הִכּוֹן לִקְרַאת אֱלֹהֶיךָ

הַנִּצְרָךְ לִנְקָבָיו — הִכּוֹן לִקְרַאת אֱלֹהֶיךָ

היסוד. הַצָּרִיךְ לִנְקָבָיו אַל יִתְפַּלֵּל (״הִכּוֹן לִקְרַאת אֱלֹהֶיךָ יִשְׂרָאֵל״) ; הִתְפַּלֵּל — תְּפִלָּתוֹ תּוֹעֵבָה וְחוֹזֵר, אֶלָּא אִם יָכוֹל לְהַעֲמִיד עַצְמוֹ שִׁעוּר פַּרְסָה (סעיף א).

המסקנה. לְכַתְּחִלָּה לֹא יִתְפַּלֵּל בְּגוּף מְשֻׁקָּץ, וְאַף תַּאֲוָה בְּאֶמְצַע הַתְּפִלָּה — מַעֲמִיד עַצְמוֹ אִם יָכוֹל (סעיף ב). עיין סימן 91.

נְטִילַת יָדַיִם וְהַהֲכָנוֹת

נְטִילַת יָדַיִם וְהַהֲכָנוֹת

היסוד. יָסִיר כִּיחוֹ וְנִיעוֹ וְיִרְחַץ יָדָיו אַף אַחַר נְטִילַת שַׁחֲרִית בְּלֹא בְּרָכָה (סעיף ג). אֵין מַיִם — עַד פַּרְסָה לְפָנָיו, מִיל לַאֲחוֹרָיו, וְאִם לָאו מְנַקֶּה בְּעָפָר (סעיף ד).

המסקנה. הֶסַּח הַדַּעַת פּוֹסֵל אֶת הַיָּדַיִם — וְכֵן עִקָּר (סעיף ה) ; אֲבָל לְתַלְמוּד תּוֹרָה דַּי בְּנִקָּיוֹן בְּעָלְמָא (סעיף ו).

מָקוֹם מְטֻנָּף וְהַרְחָקַת ד׳ אַמּוֹת

מָקוֹם מְטֻנָּף וְהַרְחָקַת ד׳ אַמּוֹת

היסוד. מָקוֹם מְטֻנָּף — הַמְקוֹמוֹת הַמְכֻסִּים (מִלְמוּלֵי זֵעָה) ; צוֹאַת הָאֹזֶן וְהָאַף — אֵין נוֹגְעִים בָּהֶם בַּתְּפִלָּה כִּי אִם עַל יְדֵי בֶּגֶד (סעיף ז).

המסקנה. הַמַּשְׁתִּין וְהָרוֹקֵק — שִׁעוּר ד׳ אַמּוֹת לִפְנֵי הַתְּפִלָּה וְאַחֲרֶיהָ ; רֹק שֶׁנִּזְדַּמֵּן — מֻתָּר (סעיפים ח-ט). עיין סימן 93.

צְדָקָה קֹדֶם תְּפִלָּה

צְדָקָה קֹדֶם תְּפִלָּה

היסוד. טוֹב לִתֵּן צְדָקָה קֹדֶם הַתְּפִלָּה, שֶׁנֶּאֱמַר : ״אֲנִי בְּצֶדֶק אֶחֱזֶה פָנֶיךָ״ (סעיף י).

המסקנה. הַהֲכָנָה לַתְּפִלָּה אֵינָהּ גּוּפָנִית בִּלְבַד — אֶלָּא גַּם שֶׁל הַלֵּב, בַּצְּדָקָה הַקּוֹדֶמֶת לַפְּגִישָׁה.

הפניה ללימוד. היקף הסימן — הַנְּקִיּוּת, הַנְּטִילַת יָדַיִם וְהַהַרְחָקוֹת — נוגע להכנת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

הַנִּצְרָךְ לִנְקָבָיו וְדִין רְחִיצָה לַתְּפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן צ״ב. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מי׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הַנִּצְרָךְ לִנְקָבָיו או״ח צ״ב — הַצָּרִיךְ לִנְקָבָיו אַל יִתְפַּלֵּל ; הִתְפַּלֵּל — תְּפִלָּתוֹ תּוֹעֵבָה. מ״ב צ״ב — שִׁעוּר פַּרְסָה וְגֶדֶר גּוּף מְשֻׁקָּץ. ערוה״ש צ״ב — דִּין הַנִּצְרָךְ וְהַהֶכְשֵׁר לַתְּפִלָּה. דרך אדמו״ר הזקן — אַל יִתְפַּלֵּל עַד שֶׁיִּפָּנֶה ; לְכַתְּחִלָּה לֹא בְּגוּף מְשֻׁקָּץ, וְאַף בְּדִבְרֵי תּוֹרָה ; וְיֵשׁ מַתִּיר בְּשִׁעוּר פַּרְסָה (סעיפים א-ב).
הֲכָנוֹת וּנְטִילַת יָדַיִם או״ח צ״ב — יָסִיר כִּיחוֹ, יִרְחַץ יָדָיו ; אֵין לוֹ מַיִם מְנַקֶּה. מ״ב צ״ב — פְּרָטֵי נְטִילָה וּמֶרְחַקֵּי הַמַּיִם. ערוה״ש צ״ב — נְטִילָה לַתְּפִלָּה בְּלֹא בְּרָכָה. דרך אדמו״ר הזקן — יִבְדֹּק עַצְמוֹ, יִרְחַץ יָדָיו אַף אַחַר שַׁחֲרִית בְּלֹא בְּרָכָה ; אֵין מַיִם — עַד פַּרְסָה, וּמְנַקֶּה (סעיפים ג-ו).
מָקוֹם מְטֻנָּף וְהַרְחָקוֹת או״ח צ״ב — מָקוֹם מְטֻנָּף ; מַשְׁתִּין וְרוֹקֵק שִׁעוּר ד׳ אַמּוֹת. מ״ב צ״ב — גֶּדֶר הַטִּנּוֹפוֹת וְהַנִּיצוֹצוֹת. ערוה״ש צ״ב — הַרְחָקַת ד׳ אַמּוֹת וּטְעָמֶיהָ. דרך אדמו״ר הזקן — הַמְכֻסִּים וְצוֹאַת הָאֹזֶן וְהָאַף עַל יְדֵי בֶּגֶד ; מַשְׁתִּין וְרוֹקֵק — ד׳ אַמּוֹת (סעיפים ז-ט).
צְדָקָה קֹדֶם תְּפִלָּה או״ח צ״ב — טוֹב לִתֵּן צְדָקָה קֹדֶם תְּפִלָּה. מ״ב צ״ב — טַעַם ״אֲנִי בְּצֶדֶק אֶחֱזֶה פָנֶיךָ״. ערוה״ש צ״ב — עִנְיַן צְדָקָה קֹדֶם תְּפִלָּה. דרך אדמו״ר הזקן — טוֹב לִתֵּן צְדָקָה קֹדֶם הַתְּפִלָּה, שֶׁנֶּאֱמַר ״אֲנִי בְּצֶדֶק אֶחֱזֶה פָנֶיךָ״ (סעיף י).

הטבלה נערכה מטקסט המחבר (אורח חיים צ״ב, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה ד, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ב-כ״ג). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (י׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַנְּקִיּוּת כִּתְנַאי בַּתְּפִלָּה

קו א — הַנְּקִיּוּת כִּתְנַאי בַּתְּפִלָּה

קו א — הַנְּקִיּוּת כִּתְנַאי בַּתְּפִלָּה

יסוד הסימן : הַנִּצְרָךְ לִנְקָבָיו אַל יִתְפַּלֵּל ; גּוּף מְשֻׁקָּץ — לֹא לְכַתְּחִלָּה (סעיפים א-ב).

קריאת הרב : הַנְּקִיּוּת שֶׁל הַגּוּף הִיא תְּנַאי בַּתְּפִלָּה, לֹא נוֹי בְּעָלְמָא.

קו ב — נְטִילַת יָדַיִם וְהֶסַּח הַדַּעַת

קו ב — נְטִילַת יָדַיִם וְהֶסַּח הַדַּעַת

קו ב — נְטִילַת יָדַיִם וְהֶסַּח הַדַּעַת

יסוד הסימן : יִרְחַץ יָדָיו אַף אַחַר שַׁחֲרִית בְּלֹא בְּרָכָה ; הֶסַּח הַדַּעַת פּוֹסֵל (סעיפים ג-ה).

קריאת הרב : וְכֵן עִקָּר שֶׁהֶסַּח הַדַּעַת מְחַיֵּב נִקָּיוֹן חָדָשׁ, שֶׁהַיָּדַיִם עַסְקָנִיּוֹת.

קו ג — הַהַרְחָקוֹת וּנְקִיּוּת הַמָּקוֹם

קו ג — הַהַרְחָקוֹת וּנְקִיּוּת הַמָּקוֹם

קו ג — הַהַרְחָקוֹת וּנְקִיּוּת הַמָּקוֹם

יסוד הסימן : מָקוֹם מְטֻנָּף עַל יְדֵי בֶּגֶד ; מַשְׁתִּין וְרוֹקֵק — ד׳ אַמּוֹת (סעיפים ז-ט).

קריאת הרב : הַהַרְחָקוֹת שׁוֹמְרוֹת עַל כְּבוֹד הַתְּפִלָּה ; וְהַנִּזְדַּמֵּן (רֹק) הוּקַל.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן צ״ב ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן צ״ב (הַנִּצְרָךְ לִנְקָבָיו וְדִין רְחִיצָה לַתְּפִלָּה)

  • ① הַנִּצְרָךְ לִנְקָבָיו. לְהִפָּנוֹת קֹדֶם הַתְּפִלָּה ; גּוּף מְשֻׁקָּץ — אֵין תְּפִלָּה וְאַף לֹא דִּבְרֵי תּוֹרָה. יסוד : שוע״ר צ״ב סעיף א.
  • ② בְּאֶמְצַע הַתְּפִלָּה. נִתְעוֹרְרָה תַּאֲוָה בַּתְּפִלָּה — מַעֲמִיד עַצְמוֹ אִם יָכוֹל ; מִשּׁוּם ״בַּל תְּשַׁקְּצוּ״ מַפְסִיק בַּק״ש, לֹא בָּעֲמִידָה. יסוד : שוע״ר צ״ב סעיף ב.
  • ③ הֲכָנוֹת וּנְטִילַת יָדַיִם. יָסִיר כִּיחוֹ וְנִיעוֹ, יִרְחַץ יָדָיו אַף אַחַר שַׁחֲרִית בְּלֹא בְּרָכָה ; אֵין מַיִם — נִקָּיוֹן בְּעָפָר. יסוד : שוע״ר צ״ב סעיפים ג-ה.
  • ④ מָקוֹם מְטֻנָּף. לֹא לִגַּע בַּמְּקוֹמוֹת הַמְכֻסִּים וְלֹא בְּצוֹאַת הָאֹזֶן וְהָאַף בַּתְּפִלָּה, כִּי אִם עַל יְדֵי בֶּגֶד. יסוד : שוע״ר צ״ב סעיפים ו-ז.
  • ⑤ הַרְחָקוֹת וּצְדָקָה. שִׁעוּר ד׳ אַמּוֹת לַמַּשְׁתִּין וְלָרוֹקֵק ; וְטוֹב לִתֵּן צְדָקָה קֹדֶם תְּפִלָּה (״אֲנִי בְּצֶדֶק אֶחֱזֶה פָנֶיךָ״). יסוד : שוע״ר צ״ב סעיפים ח-י.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן צ״ב (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַנְּקִיּוּת, הַנְּטִילַת יָדַיִם, הַהַרְחָקוֹת — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
