דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״ט · דִּין אֵיךְ יִתְנַהֵג הַיָּחִיד לְכַוֵּן בִּתְפִלָּתוֹ עִם הַצִּבּוּר

דרכו של אדמו״ר הזקן בדִּין אֵיךְ יִתְנַהֵג הַיָּחִיד לְכַוֵּן בִּתְפִלָּתוֹ עִם הַצִּבּוּר : מָתַי יַתְחִיל הַיָּחִיד לְהַשִּׂיג קְדֻשָּׁה וְקַדִּישׁ ; הַהַמְתָּנָה עַד ״שׁוֹמֵעַ תְּפִלָּה״ וְ״מוֹדִים״ לִשְׁחוֹת עִם הַצִּבּוּר ; אֲמִירַת הַקְּדֻשָּׁה מִלָּה בְּמִלָּה עִם הַשְּׁלִיחַ צִבּוּר ; סֵדֶר הָעֲדִיפוּיוֹת וְהַקְּדֻשּׁוֹת שָׁווֹת — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״ט — דִּין אֵיךְ יִתְנַהֵג הַיָּחִיד לְכַוֵּן בִּתְפִלָּתוֹ עִם הַצִּבּוּר. אצל הרב הסימן נפרש לה׳ סְעִיפִים : מָתַי יַתְחִיל הַיָּחִיד (לְהַשִּׂיג קְדֻשָּׁה וְקַדִּישׁ), הַהַמְתָּנָה עַד ״שׁוֹמֵעַ תְּפִלָּה״ וְהַשְּׁחִיָּה עִם הַצִּבּוּר בְּ״מוֹדִים״, אֲמִירַת הַקְּדֻשָּׁה מִלָּה בְּמִלָּה עִם הַשְּׁלִיחַ צִבּוּר, סֵדֶר הָעֲדִיפוּיוֹת (מוּטָב לְבַטֵּל מוֹדִים), וְהַקְּדֻשּׁוֹת שָׁווֹת וְשֶׁאֵינָן שָׁווֹת.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״ט — ה׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״ט). אדמו״ר הזקן דן בו בְּדִין אֵיךְ יִתְנַהֵג הַיָּחִיד לְכַוֵּן בִּתְפִלָּתוֹ עִם הַצִּבּוּר. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״ט — דִּין אֵיךְ יִתְנַהֵג הַיָּחִיד לְכַוֵּן בִּתְפִלָּתוֹ עִם הַצִּבּוּר

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״ט — דִּין אֵיךְ יִתְנַהֵג הַיָּחִיד לְכַוֵּן בִּתְפִלָּתוֹ עִם הַצִּבּוּר — וּבוֹ ה׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.109), ואחריו תרגומנו. הסימן כולל ה׳ סעיפים (סעיף k → ‎.109.k).

סעיף א הַנִּכְנָס וּמָצָא צִבּוּר מִתְפַּלְלִין — אִם יָכוֹל לִגְמֹר קֹדֶם קְדֻשָּׁה יִתְפַּלֵּל.

דִּין אֵיךְ יִתְנַהֵג הַיָּחִיד לְכַוֵּן בִּתְפִלָּתוֹ עִם הַצִּבּוּר וּבוֹ ה׳ סְעִיפִים:הַנִּכְנָס לְבֵית הַכְּנֶסֶת וּמָצָא צִבּוּר מִתְפַּלְלִין י״ח (שֶׁל מִנְחָה) וְאֵין הַשָּׁעָה עוֹבֶרֶת עֲדַיִן, אִם יָכוֹל לְהַתְחִיל י״ח וְלִגְמֹר עַד שֶׁלֹּא יַגִּיעַ הַשְּׁלִיחַ צִבּוּר לִקְדֻשָּׁה — יִתְפַּלֵּל, וְאִם לָאו — אַל יִתְפַּלֵּל עַד שֶׁיַּעֲנֶה אָמֵן אַחַר ״הָאֵל הַקָּדוֹשׁ״, שֶׁאָמֵן זֶה חָשׁוּב כִּקְדֻשָּׁה. וְאִם שָׁמַע כְּבָר קְדֻשָּׁה אוֹ שֶׁיּוֹדֵעַ שֶׁיִּשְׁמַע אַחַר כָּךְ — אֵין צָרִיךְ לְהַמְתִּין. וְאִם מָצָא הַצִּבּוּר מִתְפַּלְּלִין עַרְבִית, אִם יָכוֹל לְהַתְחִיל וְלִגְמֹר עַד שֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לְקַדִּישׁ שֶׁלְּאַחַר י״ח בְּעִנְיָן שֶׁיּוּכַל לַעֲנוֹת אַחֲרָיו ״יְהֵא שְׁמֵהּ רַבָּא״ עַל דֶּרֶךְ שֶׁנִּתְבָּאֵר בְּסִימָן נ״ו אֲפִלּוּ אִם לֹא יוּכַל לַעֲנוֹת ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״ כְּמוֹ שֶׁנִּתְבָּאֵר שָׁם — יִתְפַּלֵּל, וְאִם לָאו — אַל יִתְפַּלֵּל אַף עַל פִּי שֶׁשָּׁמַע כְּבָר קַדִּישׁ בְּעַרְבִית, לְפִי שֶׁהַקַּדִּישׁ אֵין לוֹ קִצְבָּה. וּמִכָּל מָקוֹם, הַקַּדִּישִׁים שֶׁמִּן ״עָלֵינוּ״ וְאֵילָךְ, בֵּין בַּשַּׁחַר בֵּין בָּעֶרֶב, כֵּיוָן שֶׁאֵינָן אֶלָּא מִנְהָג — אֶפְשָׁר שֶׁאֵין צָרִיךְ לְהַמְתִּין עֲלֵיהֶם. אֲבָל אִם פּוֹרְסִים עַל ״שְׁמַע״ וְהוּא לֹא שָׁמַע עֲדַיִן ״בָּרְכוּ״ — צָרִיךְ לְהַמְתִּין בִּשְׁבִיל ״בָּרְכוּ״, וַאֲפִלּוּ אִם אוֹמְרִים אוֹתוֹ בְּלֹא קַדִּישׁ כְּגוֹן בְּעַרְבִית:

הַנִּכְנָס וּמָצָא צִבּוּר מִתְפַּלְלִין. הַנִּכְנָס לְבֵית הַכְּנֶסֶת וּמָצָא צִבּוּר מִתְפַּלְלִין עֲמִידָה (שֶׁל מִנְחָה), וְעֲדַיִן אֵין הַשָּׁעָה עוֹבֶרֶת : אִם יָכוֹל לְהַתְחִיל וְלִגְמֹר קֹדֶם קְדֻשָּׁה — יִתְפַּלֵּל ; וְאִם לָאו — יַמְתִּין עַד שֶׁיַּעֲנֶה אָמֵן אַחַר ״הָאֵל הַקָּדוֹשׁ״ (חָשׁוּב כִּקְדֻשָּׁה). וּבְעַרְבִית — עַד שֶׁיּוּכַל לַעֲנוֹת ״יְהֵא שְׁמֵהּ רַבָּא״ אַחַר הַקַּדִּישׁ שֶׁלְּאַחַר י״ח (עיין סימן נ״ו), שֶׁהַקַּדִּישׁ אֵין לוֹ קִצְבָּה. הַקַּדִּישִׁים מִן ״עָלֵינוּ״ אֵינָם אֶלָּא מִנְהָג ; אֲבָל בִּשְׁבִיל ״בָּרְכוּ״ (פּוֹרְסִין עַל שְׁמַע) — צָרִיךְ לְהַמְתִּין.

סעיף ב עַד ״שׁוֹמֵעַ תְּפִלָּה״ ; וְאִם לָאו — עַד לְאַחַר ״מוֹדִים״, לִשְׁחוֹת עִם הַצִּבּוּר.

וְאִם נִכְנַס לְבֵית הַכְּנֶסֶת אַחַר קְדֻשָּׁה, אוֹ שֶׁנִּכְנַס קֹדֶם לָכֵן וְהִמְתִּין עַד לְאַחַר כָּךְ, אִם יָכוֹל לְהַתְחִיל וְלִגְמֹר עַד שֶׁיַּגִּיעַ הַשְּׁלִיחַ צִבּוּר לְ״שׁוֹמֵעַ תְּפִלָּה״ בִּכְדֵי שֶׁיּוּכַל לַעֲנוֹת אָמֵן אַחַר בִּרְכַּת ״שׁוֹמֵעַ תְּפִלָּה״ שֶׁאָמֵן זֶה גַּם כֵּן חָשׁוּב כִּקְדֻשָּׁה וְקַדִּישׁ — יִתְפַּלֵּל, וְאִם לָאו — אַל יִתְפַּלֵּל עַד לְאַחַר ״מוֹדִים״. וְכֵן בְּשַׁבָּת וְיוֹם טוֹב וּמוּסַף רֹאשׁ חֹדֶשׁ שֶׁאֵין בָּהֶם אָמֵן שֶׁל ״שׁוֹמֵעַ תְּפִלָּה״ — לֹא יִתְפַּלֵּל אֶלָּא אִם כֵּן יָכוֹל לִגְמֹר קֹדֶם שֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לְ״מוֹדִים״, כְּדֵי שֶׁיִּשְׁחֶה עִם הַצִּבּוּר בְּ״מוֹדִים״, שֶׁאִם הֵם יִכְרְעוּ וְהוּא לֹא יִכְרַע עִמָּהֶם — יְהֵא נִרְאֶה כְּכוֹפֵר בְּמִי שֶׁחֲבֵרָיו כּוֹרְעִים וּמִשְׁתַּחֲוִים לוֹ (וְכֵן בְּכָל מָקוֹם שֶׁהַצִּבּוּר כּוֹרְעִים וּמִשְׁתַּחֲוִים כְּגוֹן בְּ״עָלֵינוּ״). וְהוּא הַדִּין אִם יָכוֹל לְהַגִּיעַ לְ״מוֹדִים״, אוֹ לְאַחַת מֵהַבְּרָכוֹת שֶׁשּׁוֹחִים בָּהֶן כְּשֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לְ״מוֹדִים״ — יִתְפַּלֵּל. וְאַף עַל פִּי שֶׁלֹּא יָכוֹל לוֹמַר ״מוֹדִים״ שֶׁהַצִּבּוּר אוֹמְרִים — אֵין בְּכָךְ כְּלוּם, כֵּיוָן שֶׁשּׁוֹחֶה עִמָּהֶם. בַּמֶּה דְּבָרִים אֲמוּרִים? בְּמוּסָף אוֹ בְּמִנְחָה, אוֹ אֲפִלּוּ בִּתְפִלַּת שַׁחֲרִית וּמַמְתִּין קֹדֶם חֲתִימַת ״גָּאַל יִשְׂרָאֵל״. אֲבָל אִם אָמַר עַד ״גָּאַל יִשְׂרָאֵל״ שֶׁאֵינוֹ יָכוֹל עוֹד לְהַמְתִּין וְלַעֲנוֹת מִשּׁוּם סְמִיכַת גְּאֻלָּה לִתְפִלָּה וּמֻכְרָח לְהַתְחִיל וּלְהִתְפַּלֵּל מִיָּד וְנִזְדַּמֵּן לוֹ שֶׁמַּגִּיעַ שְׁלִיחַ צִבּוּר לְ״מוֹדִים״ וְהוּא בְּאַחַת הַבְּרָכוֹת בְּאֶמְצָעָן — יִשְׁחֶה עִמּוֹ, שֶׁמֻּתָּר לִשְׁחוֹת בְּאֶמְצַע כָּל הַבְּרָכוֹת. אֲבָל אִם אֵינוֹ מֻכְרָח לְהַתְחִיל מִיָּד — לֹא יַתְחִיל עַל סְמַךְ שֶׁיִּשְׁחֶה בְּאֶמְצַע אֵיזוֹ בְּרָכָה, כִּי שֶׁמָּא לֹא יוּכַל לְצַמְצֵם שֶׁיִּהְיֶה בְּאֶמְצַע בְּרָכָה כְּשֶׁמַּגִּיעַ שְׁלִיחַ צִבּוּר לְ״מוֹדִים״ אֶלָּא יִהְיֶה בִּתְחִלַּת בְּרָכָה אוֹ בְּסוֹפָהּ וְאֵין שׁוֹחִין בִּתְחִלַּת בְּרָכָה אוֹ בְּסוֹפָהּ אֶלָּא בְּאָבוֹת וְהוֹדָאָה בִּלְבַד כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן (קל״ג) [קי״ג], וְאִם יִתֵּן לִבּוֹ לְצַמְצֵם בָּזֶה — יִטָּרֵד וְלֹא יְכַוֵּן יָפֶה בִּתְפִלָּתוֹ:

עַד ״שׁוֹמֵעַ תְּפִלָּה״ וְ״מוֹדִים״. נִכְנַס אַחַר קְדֻשָּׁה : אִם יָכוֹל לִגְמֹר עַד שֶׁיַּגִּיעַ הַש״ץ לְ״שׁוֹמֵעַ תְּפִלָּה״ לַעֲנוֹת אָמֵן (חָשׁוּב כִּקְדֻשָּׁה וְקַדִּישׁ) — יִתְפַּלֵּל ; וְאִם לָאו — אַל יִתְפַּלֵּל עַד לְאַחַר ״מוֹדִים״, כְּדֵי שֶׁיִּשְׁחֶה עִם הַצִּבּוּר (וְאִם לָאו — כְּכוֹפֵר). בְּשַׁבָּת, יו״ט וּמוּסַף ר״ח (שֶׁאֵין בָּהֶם אָמֵן דְּ״שׁוֹמֵעַ תְּפִלָּה״) — רַק אִם יָכוֹל לְהַגִּיעַ לְ״מוֹדִים״. בְּשַׁחֲרִית מַמְתִּין קֹדֶם חֲתִימַת ״גָּאַל יִשְׂרָאֵל״ ; וּלְאַחַר מִכֵּן (סְמִיכַת גְּאֻלָּה לִתְפִלָּה) מַתְחִיל מִיָּד, וְאִם הוּא בְּאֶמְצַע בְּרָכָה כְּשֶׁהַצִּבּוּר אוֹמְרִים מוֹדִים — יִשְׁחֶה עִמּוֹ (עיין סימן קי״ג).

סעיף ג הַמַּתְחִיל עִם הַשְּׁלִיחַ צִבּוּר — אוֹמֵר עִמּוֹ מִלָּה בְּמִלָּה כָּל הַקְּדֻשָּׁה.

וּכְשֶׁהוּא מֻכְרָח לְהַתְחִיל מִיָּד וּמַתְחִיל עִם הַשְּׁלִיחַ צִבּוּר, כְּשֶׁיַּגִּיעַ עִם שְׁלִיחַ צִבּוּר לְ״נַקְדִּישָׁךְ״ אוֹ לִ״נְקַדֵּשׁ״ כְּפִי נֻסַּח מְדִינוֹת אֵלּוּ — יֹאמַר עִמּוֹ מִלָּה בְּמִלָּה כָּל נֻסַּח הַקְּדֻשָּׁה כְּמוֹ שֶׁהוּא אוֹמֵר, שֶׁאַף שֶׁאֵין הַיָּחִיד אוֹמֵר קְדֻשָּׁה בִּתְפִלָּתוֹ — כְּשֶׁהוּא אוֹמֵר עִם הַשְּׁלִיחַ צִבּוּר אֵינוֹ נִקְרָא יָחִיד. וְכֵן יֹאמַר עִמּוֹ מִלָּה בְּמִלָּה כָּל נֻסַּח בִּרְכַּת ״הָאֵל הַקָּדוֹשׁ״ וּבִרְכַּת ״שׁוֹמֵעַ תְּפִלָּה״, כְּדֵי שֶׁיְּסַיֵּם הַבְּרָכָה עִם הַשְּׁלִיחַ צִבּוּר בְּיַחַד, וְשׁוּב אֵין צָרִיךְ לַעֲנוֹת אָמֵן, כְּמוֹ שֶׁהַשְּׁלִיחַ צִבּוּר אֵין צָרִיךְ שֶׁהַצִּבּוּר עוֹנִים אַחֲרָיו, וַעֲנִיַּת הַצִּבּוּר הִיא גַּם כֵּן עַל בְּרָכוֹת שֶׁל זֶה. וְגַם יְכַוֵּן שֶׁכְּשֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לְ״מוֹדִים״ יַגִּיעַ גַּם הוּא לְ״מוֹדִים״ אוֹ לְ״הַטּוֹב שִׁמְךָ״, כְּדֵי שֶׁיִּשְׁחֶה גַּם הוּא עִם הַצִּבּוּר בְּ״מוֹדִים״. וְאִם יָכוֹל לִגְמֹר כָּל הַתְּפִלָּה עַד שֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לְסִיּוּם ״שׁוֹמֵעַ תְּפִלָּה״ — יָכוֹל לְהִתְפַּלֵּל בִּפְנֵי עַצְמוֹ אַחַר סִיּוּם ״הָאֵל הַקָּדוֹשׁ״. וְכָל זֶה כְּשֶׁמֻּכְרָח לְהִתְפַּלֵּל מִיָּד מִשּׁוּם סְמִיכַת גְּאֻלָּה לִתְפִלָּה, אוֹ מִפְּנֵי שֶׁהַשָּׁעָה עוֹבֶרֶת, אֲבָל אִם אֵינוֹ מֻכְרָח — לֹא יַתְחִיל עִם הַשְּׁלִיחַ צִבּוּר וְלוֹמַר עִמּוֹ נֻסַּח הַקְּדֻשָּׁה, לְפִי שֶׁלְּכַתְּחִלָּה צָרִיךְ לִשְׁתֹּק וְלִשְׁמֹעַ נֻסַּח הַקְּדֻשָּׁה מִפִּי הַשְּׁלִיחַ צִבּוּר וְלַעֲנוֹת אַחֲרָיו ״קָדוֹשׁ״ וּ״בָרוּךְ״ וְ״יִמְלֹךְ״ בִּלְבַד, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ה:

הַמַּתְחִיל עִם הַשְּׁלִיחַ צִבּוּר. הַמֻּכְרָח לְהַתְחִיל מִיָּד עִם הַש״ץ : כְּשֶׁמַּגִּיעַ לְ״נַקְדִּישָׁךְ״/״נְקַדֵּשׁ״ — אוֹמֵר עִמּוֹ מִלָּה בְּמִלָּה כָּל הַקְּדֻשָּׁה (שֶׁכְּשֶׁאוֹמֵר עִם הַש״ץ אֵינוֹ נִקְרָא יָחִיד) ; וְכֵן בִּרְכַּת ״הָאֵל הַקָּדוֹשׁ״ וּ״שׁוֹמֵעַ תְּפִלָּה״, לְסַיֵּם עִמּוֹ בְּלִי לַעֲנוֹת אָמֵן. וְיְכַוֵּן שֶׁ״מוֹדִים״ דְּהַש״ץ יִזְדַּמֵּן עִם ״מוֹדִים״/״הַטּוֹב שִׁמְךָ״ שֶׁלּוֹ. וְכָל זֶה בְּמֻכְרָח (סְמִיכַת גְּאֻלָּה / הַשָּׁעָה עוֹבֶרֶת) ; וְאִם לָאו — לְכַתְּחִלָּה יִשְׁתֹּק וְיִשְׁמַע וְיַעֲנֶה ״קָדוֹשׁ״ ״בָּרוּךְ״ ״יִמְלֹךְ״ בִּלְבַד (עיין סימן קכ״ה).

סעיף ד מוּטָב לְבַטֵּל ״מוֹדִים״ מִלְּבַטֵּל ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״.

מִי שֶׁבָּא לְבֵית הַכְּנֶסֶת תֵּכֶף אַחַר קְדֻשָּׁה, וְאִם יַמְתִּין עַד אַחַר שֶׁיַּגִּיעַ שְׁלִיחַ צִבּוּר לְ״מוֹדִים״ לֹא יוּכַל לַעֲנוֹת ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״ (אַחַר הַקַּדִּישׁ) שֶׁאַחַר י״ח, וְאִם יַמְתִּין עַד אַחַר הַקַּדִּישׁ יַעֲבֹר זְמַן הַתְּפִלָּה אוֹ לֹא יוּכַל לְהִתְפַּלֵּל עַרְבִית עִם הַצִּבּוּר — יַתְחִיל מִיָּד, שֶׁמּוּטָב לְבַטֵּל ״מוֹדִים״ מִלְּבַטֵּל ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״. וְהוּא הַדִּין אִם בָּא סָמוּךְ לִקְדֻשָּׁה, וְאִם יַתְחִיל מִיָּד לֹא יוּכַל לוֹמַר קְדֻשָּׁה, וְאִם יַמְתִּין עַד אַחַר קְדֻשָּׁה לֹא יוּכַל לוֹמַר ״מוֹדִים״, וְאִם יַמְתִּין עַד אַחַר ״מוֹדִים״ יַעֲבֹר זְמַן הַתְּפִלָּה — יַתְחִיל אַחַר קְדֻשָּׁה, שֶׁמּוּטָב לְבַטֵּל ״מוֹדִים״ מִקְּדֻשָּׁה עִם הַצִּבּוּר. וְאִם בָּא סָמוּךְ לַקַּדִּישׁ, וְאִם יִתְפַּלֵּל תֵּכֶף יְבַטֵּל ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״, וְאִם יַמְתִּין עַד אַחַר קַדִּישׁ לֹא יוּכַל לְהִתְפַּלֵּל עַרְבִית עִם הַצִּבּוּר — מוּטָב לְהִתְפַּלֵּל עִם הַצִּבּוּר:

סֵדֶר הָעֲדִיפוּיוֹת — מוּטָב לְבַטֵּל. בָּא תֵּכֶף אַחַר קְדֻשָּׁה : אִם יַמְתִּין לְ״מוֹדִים״ יְבַטֵּל ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״ ; מוּטָב לְבַטֵּל ״מוֹדִים״ מִ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״. בָּא סָמוּךְ לִקְדֻשָּׁה : יַתְחִיל אַחַר קְדֻשָּׁה — מוּטָב לְבַטֵּל ״מוֹדִים״ מִקְּדֻשָּׁה. בָּא סָמוּךְ לַקַּדִּישׁ : מוּטָב לְהִתְפַּלֵּל עִם הַצִּבּוּר.

סעיף ה קְדֻשַּׁת ״וּבָא לְצִיּוֹן״ וְ״יוֹצֵר״ — אֵינוֹ אוֹמֵר עִמָּהֶם קָדוֹשׁ.

יָחִיד הָעוֹמֵד בִּתְפִלַּת י״ח וּכְשֶׁמַּגִּיעַ לִמְקוֹם קְדֻשָּׁה הָיוּ הַצִּבּוּר אוֹמְרִים קְדֻשַּׁת ״וּבָא לְצִיּוֹן״ אוֹ קְדֻשַּׁת ״יוֹצֵר״ — אֵינוֹ אוֹמֵר עִמָּהֶם קָדוֹשׁ, לְפִי שֶׁאֵין הַקְּדֻשּׁוֹת שָׁווֹת. וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁהוּא הַדִּין אִם הָיָה הַצִּבּוּר אוֹמְרִים קְדֻשַּׁת מוּסָף וְהַיָּחִיד מִתְפַּלֵּל שַׁחֲרִית — אֵינוֹ אוֹמֵר עִמָּהֶם ״קָדוֹשׁ״, שֶׁאֵין הַקְּדֻשּׁוֹת שָׁווֹת. וְיֵשׁ אוֹמְרִים שֶׁקְּדֻשַּׁת שַׁחֲרִית וּקְדֻשַּׁת מוּסָף נִקְרָאוֹת קְדֻשּׁוֹת שָׁווֹת, מַה שֶּׁאֵין כֵּן קְדֻשַּׁת ״וּבָא לְצִיּוֹן״ וּקְדֻשַּׁת ״יוֹצֵר״ שֶׁאֵינָן אֶלָּא סִפּוּר דְּבָרִים כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן נ״ט. וְכֵן עִקָּר. (וּמִכָּאן אַתָּה לָמֵד שֶׁהַיָּחִיד כְּשֶׁמַּגִּיעַ בִּתְפִלָּתוֹ לְבִרְכַּת ״אַתָּה קָדוֹשׁ״ — יָכוֹל לַעֲנוֹת קָדוֹשׁ עִם הַצִּבּוּר אַף עַל פִּי שֶׁהוּא לֹא אָמַר וְלֹא שָׁמַע ״נְקַדֵּשׁ״ אוֹ ״נַקְדִּישָׁך״ מִפִּי הַשְּׁלִיחַ צִבּוּר, שֶׁהֲרֵי קְדֻשַּׁת ״יוֹצֵר״ ״וּבָא לְצִיּוֹן״ אֵין בָּהֶם ״נְקַדֵּשׁ״ וְאַף עַל פִּי כֵן לֹא מְנָעוּהוּ אֶלָּא מִפְּנֵי שֶׁאֵין הַקְּדֻשּׁוֹת שָׁווֹת):

קְדֻשּׁוֹת שָׁווֹת וְשֶׁאֵינָן שָׁווֹת. יָחִיד בַּעֲמִידָה שֶׁמַּגִּיעַ לִקְדֻשָּׁה בְּעֵת שֶׁהַצִּבּוּר אוֹמְרִים קְדֻשַּׁת ״וּבָא לְצִיּוֹן״ אוֹ ״יוֹצֵר״ — אֵינוֹ אוֹמֵר עִמָּהֶם קָדוֹשׁ, שֶׁאֵין הַקְּדֻשּׁוֹת שָׁווֹת. וְיֵשׁ אוֹמְרִים כֵּן אַף בְּמוּסָף כְּשֶׁהוּא בְּשַׁחֲרִית ; אֲבָל הָעִקָּר : שַׁחֲרִית וּמוּסָף קְדֻשּׁוֹת שָׁווֹת, מַה שֶּׁאֵין כֵּן ״וּבָא לְצִיּוֹן״/״יוֹצֵר״ שֶׁאֵינָן אֶלָּא סִפּוּר דְּבָרִים (עיין סימן נ״ט). וּמִכָּאן : בִּבְרַכַּת ״אַתָּה קָדוֹשׁ״ שֶׁלּוֹ יָכוֹל לַעֲנוֹת קָדוֹשׁ עִם הַצִּבּוּר.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק״ט (מהדורת קה״ת, ה׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — הַיָּחִיד לְכַוֵּן תְּפִלָּתוֹ עִם הַצִּבּוּר

היסוד : הַיָּחִיד הַמְכַוֵּן תְּפִלָּתוֹ עִם הַצִּבּוּר

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב חמישה צירים. תחילה מָתַי יַתְחִיל הַיָּחִיד לְהַשִּׂיג קְדֻשָּׁה וְקַדִּישׁ. אחר כך הַהַמְתָּנָה עַד ״שׁוֹמֵעַ תְּפִלָּה״ וְהַשְּׁחִיָּה עִם הַצִּבּוּר בְּ״מוֹדִים״. אחר כך אֲמִירַת הַקְּדֻשָּׁה מִלָּה בְּמִלָּה עִם הַשְּׁלִיחַ צִבּוּר. אחר כך סֵדֶר הָעֲדִיפוּיוֹת (מוּטָב לְבַטֵּל). ולבסוף הַקְּדֻשּׁוֹת שָׁווֹת וְשֶׁאֵינָן שָׁווֹת.

מָתַי יַתְחִיל הַיָּחִיד

מָתַי יַתְחִיל הַיָּחִיד

היסוד. אִם יָכוֹל לִגְמֹר קֹדֶם קְדֻשָּׁה — יִתְפַּלֵּל ; וְאִם לָאו — יַמְתִּין לְאָמֵן אַחַר ״הָאֵל הַקָּדוֹשׁ״ (חָשׁוּב כִּקְדֻשָּׁה) ; וּבְעַרְבִית — ״יְהֵא שְׁמֵהּ רַבָּא״ (סעיף א).

המסקנה. הַקַּדִּישׁ אֵין לוֹ קִצְבָּה, וְלָכֵן מַמְתִּין אַף אַחַר שֶׁשָּׁמַע כְּבָר קַדִּישׁ. עיין סימן 56 (קדיש ואמן יהא שמה רבא).

שְׁחִיָּה עִם הַצִּבּוּר בְּ״מוֹדִים״

שְׁחִיָּה עִם הַצִּבּוּר בְּ״מוֹדִים״

היסוד. נִכְנַס אַחַר קְדֻשָּׁה — מְכַוֵּן לְאָמֵן דְּ״שׁוֹמֵעַ תְּפִלָּה״ ; וְאִם לָאו — מַמְתִּין עַד ״מוֹדִים״ לִשְׁחוֹת עִם הַצִּבּוּר, שֶׁאִם לָאו — כְּכוֹפֵר (סעיף ב).

המסקנה. בְּשַׁחֲרִית שׁוֹמֵר סְמִיכַת גְּאֻלָּה לִתְפִלָּה : לְאַחַר ״גָּאַל יִשְׂרָאֵל״ שׁוֹחֶה רַק בְּאֶמְצַע בְּרָכָה. עיין סימן 113 (מקומות הכריעה).

אֲמִירַת הַקְּדֻשָּׁה מִלָּה בְּמִלָּה

אֲמִירַת הַקְּדֻשָּׁה מִלָּה בְּמִלָּה

היסוד. הַמֻּכְרָח הַמַּתְחִיל עִם הַש״ץ אוֹמֵר עִמּוֹ מִלָּה בְּמִלָּה כָּל הַקְּדֻשָּׁה וּבִרְכוֹת ״הָאֵל הַקָּדוֹשׁ״ / ״שׁוֹמֵעַ תְּפִלָּה״, שֶׁאָז אֵינוֹ נִקְרָא יָחִיד (סעיף ג).

המסקנה. בְּלֹא הֶכְרֵחַ — לְכַתְּחִלָּה יִשְׁתֹּק וְיַעֲנֶה קָדוֹשׁ / בָּרוּךְ / יִמְלֹךְ בִּלְבַד. עיין סימן 125 (נוסח הקדושה).

סֵדֶר הָעֲדִיפוּיוֹת — מוּטָב לְבַטֵּל

סֵדֶר הָעֲדִיפוּיוֹת — מוּטָב לְבַטֵּל

היסוד. מוּטָב לְבַטֵּל ״מוֹדִים״ מִ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״ ; וּמוּטָב לְבַטֵּל ״מוֹדִים״ מִקְּדֻשָּׁה עִם הַצִּבּוּר (סעיף ד).

המסקנה. הֶיֵּרַרְכְיָה קְבוּעָה : קְדֻשָּׁה וְאָמֵן יש״ר קוֹדְמִים לַשְּׁחִיָּה שֶׁל מוֹדִים. עיין סימן 55 (דבר שבקדושה).

קְדֻשּׁוֹת שָׁווֹת וְשֶׁאֵינָן שָׁווֹת

קְדֻשּׁוֹת שָׁווֹת וְשֶׁאֵינָן שָׁווֹת

היסוד. בִּקְדֻשַּׁת ״וּבָא לְצִיּוֹן״/״יוֹצֵר״ הַיָּחִיד אֵינוֹ אוֹמֵר קָדוֹשׁ (אֵין הַקְּדֻשּׁוֹת שָׁווֹת) ; אֲבָל שַׁחֲרִית וּמוּסָף שָׁווֹת (סעיף ה).

המסקנה. ״וּבָא לְצִיּוֹן״ וְ״יוֹצֵר״ אֵינָן אֶלָּא סִפּוּר דְּבָרִים. עיין סימן 59 (ברכת יוצר וקדושתה).

הפניה ללימוד. היקף הסימן — הַמָּתַי יַתְחִיל, הַשְּׁחִיָּה עִם הַצִּבּוּר וְסֵדֶר הָעֲדִיפוּיוֹת — נוגע להנהגת המאחר בבית הכנסת. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

אֵיךְ יִתְנַהֵג הַיָּחִיד עִם הַצִּבּוּר — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק״ט. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מה׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הַנִּכְנָס וּמָצָא צִבּוּר — מָתַי יַתְחִיל או״ח ק״ט — אִם יָכוֹל לִגְמֹר קֹדֶם קְדֻשָּׁה אוֹ קַדִּישׁ יִתְפַּלֵּל, וְאִם לָאו אַל יִתְפַּלֵּל. מ״ב ק״ט — פְּרָטֵי הַהַמְתָּנָה עַל קְדֻשָּׁה וְקַדִּישׁ. ערוה״ש ק״ט — גֶּדֶר ״הַשָּׁעָה עוֹבֶרֶת״ וְהַהַמְתָּנָה. דרך אדמו״ר הזקן — עַד ״הָאֵל הַקָּדוֹשׁ״ (אָמֵן חָשׁוּב כִּקְדֻשָּׁה), וּבְעַרְבִית עַד ״יְהֵא שְׁמֵהּ רַבָּא״ שֶׁהַקַּדִּישׁ אֵין לוֹ קִצְבָּה (סעיף א).
עַד ״שׁוֹמֵעַ תְּפִלָּה״ וְהַשְּׁחִיָּה או״ח ק״ט — אַל יִתְפַּלֵּל עַד שֶׁיַּגִּיעַ לְ״מוֹדִים״ אוֹ לְאַחַת מֵהַבְּרָכוֹת שֶׁשּׁוֹחִים בָּהֶן. מ״ב ק״ט — דִּין הַשְּׁחִיָּה עִם הַצִּבּוּר וְ״כְּכוֹפֵר״. ערוה״ש ק״ט — הַהַכְרָעָה בְּשַׁחֲרִית וּסְמִיכַת גְּאֻלָּה לִתְפִלָּה. דרך אדמו״ר הזקן — עַד אָמֵן ״שׁוֹמֵעַ תְּפִלָּה״, וְאִם לָאו עַד ״מוֹדִים״ לִשְׁחוֹת עִם הַצִּבּוּר ; בְּשַׁחֲרִית קֹדֶם ״גָּאַל יִשְׂרָאֵל״ (סעיף ב).
אֲמִירַת הַקְּדֻשָּׁה עִם הַשְּׁלִיחַ צִבּוּר או״ח ק״ט — יֹאמַר עִם הַש״ץ מִלָּה בְּמִלָּה כָּל הַקְּדֻשָּׁה וּבִרְכוֹתֶיהָ. מ״ב ק״ט — כַּוָּנַת סִיּוּם הַבְּרָכָה יַחַד עִם הַש״ץ. ערוה״ש ק״ט — גֶּדֶר ״אֵינוֹ נִקְרָא יָחִיד״. דרך אדמו״ר הזקן — מִלָּה בְּמִלָּה כָּל הַקְּדֻשָּׁה, וְלֹא לְכַתְּחִלָּה אֶלָּא בְּמֻכְרָח ; אַחֶרֶת — קָדוֹשׁ בָּרוּךְ יִמְלֹךְ בִּלְבַד (סעיף ג).
סֵדֶר הָעֲדִיפוּיוֹת — מוּטָב לְבַטֵּל או״ח ק״ט — סֵדֶר הַהַכְרָעָה בֵּין קְדֻשָּׁה, קַדִּישׁ וּ״מוֹדִים״. מ״ב ק״ט — מַה קּוֹדֵם כְּשֶׁאִי אֶפְשָׁר לְקַיֵּם הַכֹּל. ערוה״ש ק״ט — הַהַכְרָעָה בֵּין הַדְּבָרִים שֶׁבִּקְדֻשָּׁה. דרך אדמו״ר הזקן — מוּטָב לְבַטֵּל ״מוֹדִים״ מִ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״ וּמִקְּדֻשָּׁה עִם הַצִּבּוּר (סעיף ד).
קְדֻשּׁוֹת שָׁווֹת וְשֶׁאֵינָן שָׁווֹת או״ח ק״ט — קְדֻשַּׁת דְּסִדְרָא אֵינוֹ אוֹמֵר עִמָּהֶם ; שַׁחֲרִית וּמוּסָף קְדֻשּׁוֹת שָׁווֹת. מ״ב ק״ט — ״שׁוֹמֵעַ כְּעוֹנֶה״ וְגֶדֶר הַקְּדֻשּׁוֹת. ערוה״ש ק״ט — טַעַם ״אֵין הַקְּדֻשּׁוֹת שָׁווֹת״. דרך אדמו״ר הזקן — ״וּבָא לְצִיּוֹן״/״יוֹצֵר״ סִפּוּר דְּבָרִים — אֵינוֹ עוֹנֶה ; שַׁחֲרִית וּמוּסָף שָׁווֹת, וְעוֹנֶה קָדוֹשׁ (סעיף ה).

הטבלה נערכה מטקסט המחבר (אורח חיים ק״ט, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה, רא״ש, תוספות ברכות, תרומת הדשן, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ה׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַזְּמָן וְהַהַמְתָּנָה

קו א — הַזְּמָן וְהַהַמְתָּנָה

יסוד הסימן : אִם יָכוֹל לִגְמֹר קֹדֶם קְדֻשָּׁה יִתְפַּלֵּל ; וְאִם לָאו — יַמְתִּין עַד ״הָאֵל הַקָּדוֹשׁ״ אוֹ ״שׁוֹמֵעַ תְּפִלָּה״ (סעיפים א-ב).

קריאת הרב : הָאָמֵן אַחַר בְּרָכוֹת אֵלּוּ חָשׁוּב כִּקְדֻשָּׁה ; הַקַּדִּישׁ אֵין לוֹ קִצְבָּה. עיין סימן 56.

מסקנה למעשה : לְכַוֵּן אֶת תְּחִלַּת הָעֲמִידָה לְהַשִּׂיג אֶת הַדָּבָר שֶׁבִּקְדֻשָּׁה הַקָּרוֹב.

קו ב — הָאֲמִירָה עִם הַשְּׁלִיחַ צִבּוּר

קו ב — הָאֲמִירָה עִם הַשְּׁלִיחַ צִבּוּר

יסוד הסימן : הַמֻּכְרָח לְהַתְחִיל עִם הַש״ץ אוֹמֵר עִמּוֹ מִלָּה בְּמִלָּה כָּל הַקְּדֻשָּׁה (סעיף ג).

קריאת הרב : כְּשֶׁאוֹמֵר עִם הַש״ץ אֵינוֹ נִקְרָא יָחִיד ; בְּלֹא הֶכְרֵחַ יִשְׁתֹּק וְיַעֲנֶה. עיין סימן 125.

מסקנה למעשה : הַמֻּכְרָח אוֹמֵר הַקְּדֻשָּׁה מִלָּה בְּמִלָּה ; וְהָאַחֵר עוֹנֶה קָדוֹשׁ / בָּרוּךְ / יִמְלֹךְ בִּלְבַד.

קו ג — סֵדֶר הָעֲדִיפוּיוֹת וְהַקְּדֻשּׁוֹת

קו ג — סֵדֶר הָעֲדִיפוּיוֹת וְהַקְּדֻשּׁוֹת

יסוד הסימן : מוּטָב לְבַטֵּל ״מוֹדִים״ ; וְאֵינוֹ אוֹמֵר קָדוֹשׁ עַל קְדֻשָּׁה שֶׁאֵינָהּ שָׁוָה (סעיפים ד-ה).

קריאת הרב : הֶיֵּרַרְכְיָה בְּרוּרָה שֶׁל דְּבָרִים שֶׁבִּקְדֻשָּׁה ; שַׁחֲרִית וּמוּסָף שָׁווֹת. עיין סימן 59.

מסקנה למעשה : לְהַעֲדִיף קְדֻשָּׁה וְאָמֵן יש״ר עַל הַשְּׁחִיָּה, וְלַעֲנוֹת קָדוֹשׁ רַק בִּקְדֻשּׁוֹת שָׁווֹת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״ט ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״ט (דִּין אֵיךְ יִתְנַהֵג הַיָּחִיד לְכַוֵּן בִּתְפִלָּתוֹ עִם הַצִּבּוּר)

  • ① מָתַי יַתְחִיל. מָצָא צִבּוּר בַּעֲמִידָה : אִם יָכוֹל לִגְמֹר קֹדֶם קְדֻשָּׁה — יִתְפַּלֵּל ; וְאִם לָאו — יַמְתִּין לְאָמֵן אַחַר ״הָאֵל הַקָּדוֹשׁ״. יסוד : שוע״ר ק״ט סעיף א.
  • ② בְּעַרְבִית. מַמְתִּין עַד שֶׁיּוּכַל לַעֲנוֹת ״יְהֵא שְׁמֵהּ רַבָּא״ אַחַר הַקַּדִּישׁ שֶׁלְּאַחַר י״ח, שֶׁהַקַּדִּישׁ אֵין לוֹ קִצְבָּה. יסוד : שוע״ר ק״ט סעיף א.
  • ③ שְׁחִיָּה בְּ״מוֹדִים״. וְאִם לָאו — אַל יִתְפַּלֵּל עַד שֶׁיּוּכַל לִשְׁחוֹת עִם הַצִּבּוּר בְּ״מוֹדִים״ (שֶׁאִם לָאו — כְּכוֹפֵר). יסוד : שוע״ר ק״ט סעיף ב.
  • ④ הַמֻּכְרָח. מֻכְרָח לְהַתְחִיל מִיָּד עִם הַש״ץ — אוֹמֵר עִמּוֹ מִלָּה בְּמִלָּה כָּל הַקְּדֻשָּׁה ; וְאַחֶרֶת יִשְׁתֹּק וְיַעֲנֶה קָדוֹשׁ / בָּרוּךְ / יִמְלֹךְ. יסוד : שוע״ר ק״ט סעיף ג.
  • ⑤ סֵדֶר וּקְדֻשּׁוֹת. מוּטָב לְבַטֵּל ״מוֹדִים״ מֵ״אָמֵן יְהֵא שְׁמֵהּ רַבָּא״ ; בְּ״וּבָא לְצִיּוֹן״/״יוֹצֵר״ אֵינוֹ אוֹמֵר קָדוֹשׁ (אֵין הַקְּדֻשּׁוֹת שָׁווֹת). יסוד : שוע״ר ק״ט סעיפים ד-ה.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק״ט (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַמָּתַי יַתְחִיל, הַשְּׁחִיָּה עִם הַצִּבּוּר, סֵדֶר הָעֲדִיפוּיוֹת — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
