דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן פ״ד · דִּין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ

דרכו של אדמו״ר הזקן בדִּין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ : מֶרְחָץ חָדָשׁ — מֻתָּר, שֶׁאֵין קְרִיאַת שֵׁם אוֹסַרְתּוֹ ; וּמֶרְחָץ יָשָׁן נֶחְלָק לִשְׁלֹשָׁה : חִיצוֹן (כֻּלָּם לְבוּשִׁים) — ק״ש וּתְפִלָּה ; אֶמְצָעִי — שְׁאִילַת שָׁלוֹם וְלֹא ק״ש ; פְּנִימִי (כֻּלָּם עֲרֻמִּים) — אֲפִלּוּ שָׁלוֹם אָסוּר, כִּי ״שָׁלוֹם״ שְׁמוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן פ״ד — דִּין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ. אצל הרב הסימן נכתב בסָעִיף אֶחָד הַמְאַגֵּד שְׁלֹשָׁה מִישׁוֹרִים : מֶרְחָץ חָדָשׁ (אֵין קְרִיאַת שֵׁם אוֹסַרְתּוֹ), שְׁלֹשֶׁת הַבָּתִּים שֶׁל הַמֶּרְחָץ הַיָּשָׁן — חִיצוֹן, אֶמְצָעִי, פְּנִימִי — כָּל אֶחָד וְדִינוֹ, וּלְבַסּוֹף שֵׁם שָׁלוֹם, הַפְּסַק דִּין בְּלֹא טַעַם וַעֲנִיַּת אָמֵן.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן פ״ד — סָעִיף אֶחָד מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim פ״ד). אדמו״ר הזקן דן בו בְּדִין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן פ״ד — דִּין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן פ״ד — דִּין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ — וּבוֹ סָעִיף אֶחָד
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.84), ואחריו תרגומנו. הסימן כולל סעיף אחד (סעיף א → ‎.84.1).

סעיף א מֶרְחָץ חָדָשׁ — מֻתָּר ; יָשָׁן : חִיצוֹן — ק״ש, אֶמְצָעִי — שָׁלוֹם, פְּנִימִי — אָסוּר.

אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ וּבוֹ סָעִיף אֶחָד:מֶרְחָץ חָדָשׁ שֶׁלֹּא רָחֲצוּ בּוֹ — מֻתָּר לִקְרוֹת בּוֹ, שֶׁאֵין קְרִיאַת שֵׁם אוֹסַרְתּוֹ כְּמוֹ שֶׁאוֹסֶרֶת בְּבֵית הַכִּסֵּא, לְפִי שֶׁבֵּית הַכִּסֵּא הוּא מָאוּס בְּיוֹתֵר. וּמֶרְחָץ יָשָׁן, בַּבַּיִת הַחִיצוֹן שֶׁכָּל הָעוֹמְדִים שָׁם הֵם לְבוּשִׁים — מֻתָּר לִקְרוֹת שָׁם קְרִיאַת שְׁמַע וּלְהִתְפַּלֵּל. וּבָאֶמְצַע שֶׁקְּצָת מִן הָעוֹמְדִים שָׁם הֵם לְבוּשִׁים וּקְצָתָם עֲרֻמִּים — יֵשׁ שָׁם שְׁאִילַת שָׁלוֹם, אֲבָל לֹא קְרִיאַת שְׁמַע וּתְפִלָּה וְדִבְרֵי תּוֹרָה אֲפִלּוּ בְּשָׁעָה שֶׁאֵין שָׁם אָדָם עָרֹם. אֲבָל מֻתָּר לְהַרְהֵר שָׁם בְּדִבְרֵי תּוֹרָה אֲפִלּוּ כְּשֶׁיֵּשׁ שָׁם עֲרֻמִּים. וּמֻתָּר לִפְסֹק שָׁם דִּין, וּבִלְבַד שֶׁלֹּא יֹאמַר טַעֲמוֹ שֶׁל דָּבָר, שֶׁזֶּה אֵינוֹ אֶלָּא כְּמוֹ הִרְהוּר בְּדִבְרֵי תּוֹרָה שֶׁמְּחַשֵּׁב הַטַּעַם בְּלִבּוֹ. וְהַבַּיִת הַפְּנִימִי שֶׁכָּל הָעוֹמְדִים שָׁם הֵם עֲרֻמִּים — אֲפִלּוּ שְׁאִילַת שָׁלוֹם אָסוּר כִּי ״שָׁלוֹם״ הוּא שְׁמוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁנֶּאֱמַר: ״וַיִּקְרָא לוֹ ה׳ שָׁלוֹם״. וְהוּא הַדִּין שֶׁאָסוּר לִתֵּן שָׁלוֹם לַחֲבֵרוֹ בִּמְבוֹאוֹת הַמְטֻנָּפוֹת. וְאָדָם שֶׁשְּׁמוֹ שָׁלוֹם — יֵשׁ אוֹסְרִין לִקְרוֹת בִּשְׁמוֹ אֶלָּא בִּלְשׁוֹן (לע״ז). וְיֵשׁ מַתִּירִין, כֵּיוָן שֶׁאֵין מִתְכַוֵּן עַל עִנְיַן הַשָּׁלוֹם אֶלָּא לְהַזְכָּרַת שְׁמוֹ שֶׁל אוֹתוֹ אָדָם (וּבְדִבְרֵי סוֹפְרִים הַלֵּךְ אַחַר הַמֵּקֵל. וְכֵן נוֹהֲגִין). וְאָסוּר לַעֲנוֹת שָׁם אָמֵן, וְלֹא לְהַרְהֵר בְּשׁוּם דָּבָר שֶׁבִּקְדוּשָּׁה. וְאִם שָׁאַל לוֹ אָדָם דְּבַר הֲלָכָה — אָסוּר לוֹמַר לוֹ אֵין מְשִׁיבִין בַּמֶּרְחָץ. אֲבָל בַּבַּיִת הָאֶמְצָעִי מֻתָּר לוֹמַר לוֹ כָּךְ:

מֶרְחָץ חָדָשׁ. מֶרְחָץ חָדָשׁ שֶׁלֹּא רָחֲצוּ בּוֹ — מֻתָּר לִקְרוֹת בּוֹ, שֶׁאֵין קְרִיאַת הַשֵּׁם ״מֶרְחָץ״ אוֹסַרְתּוֹ כִּבְבֵית הַכִּסֵּא שֶׁהוּא מָאוּס בְּיוֹתֵר. מֶרְחָץ יָשָׁן : בַּחִיצוֹן שֶׁכֻּלָּם לְבוּשִׁים — מֻתָּר ק״ש וּתְפִלָּה ; בָּאֶמְצָעִי שֶׁמִּקְצָתָם עֲרֻמִּים — יֵשׁ שְׁאִילַת שָׁלוֹם, אֲבָל לֹא ק״ש וּתְפִלָּה וְדִבְרֵי תּוֹרָה, אֲפִלּוּ שֶׁאֵין שָׁם עָרֹם ; אֲבָל מֻתָּר לְהַרְהֵר אֲפִלּוּ כְּשֶׁיֵּשׁ עֲרֻמִּים, וְלִפְסֹק דִּין וּבִלְבַד שֶׁלֹּא יֹאמַר טַעֲמוֹ (כְּהִרְהוּר). בַּפְּנִימִי שֶׁכֻּלָּם עֲרֻמִּים — אֲפִלּוּ שְׁאִילַת שָׁלוֹם אָסוּר, כִּי ״שָׁלוֹם״ שְׁמוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא (״וַיִּקְרָא לוֹ ה׳ שָׁלוֹם״) ; וְכֵן בִּמְבוֹאוֹת הַמְטֻנָּפוֹת. אָדָם שֶׁשְּׁמוֹ שָׁלוֹם — יֵשׁ אוֹסְרִין אֶלָּא בְּלַעַז וְיֵשׁ מַתִּירִין, וּבְדִבְרֵי סוֹפְרִים הַלֵּךְ אַחַר הַמֵּקֵל (וְכֵן נוֹהֲגִין). אָסוּר לַעֲנוֹת שָׁם אָמֵן וּלְהַרְהֵר בְּדָבָר שֶׁבִּקְדֻשָּׁה ; וְאִם שָׁאֲלוֹ דְּבַר הֲלָכָה — אָסוּר אֲפִלּוּ לוֹמַר ״אֵין מְשִׁיבִין בַּמֶּרְחָץ״, אֲבָל בָּאֶמְצָעִי מֻתָּר לוֹמַר לוֹ כָּךְ.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim פ״ד (מהדורת קה״ת, סָעִיף אֶחָד). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ

היסוד : דִּין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה מישורים. תחילה הַמֶּרְחָץ הַחָדָשׁ — אֵין הַשֵּׁם אוֹסֵר. אחר כך הַבַּיִת הַחִיצוֹן — כֻּלָּם לְבוּשִׁים, וְקוֹרְאִים. אחר כך הַבַּיִת הָאֶמְצָעִי — מָקוֹם מְעֹרָב שֶׁבּוֹ הַדִּבּוּר שֶׁבִּקְדֻשָּׁה נִדְחֶה אֲבָל הַמַּחֲשָׁבָה קַיֶּמֶת. ולבסוף הַבַּיִת הַפְּנִימִי — כֻּלָּם עֲרֻמִּים, שֶׁאַף הַשָּׁלוֹם נֶעֱצָר בּוֹ.

מֶרְחָץ חָדָשׁ — אֵין קְרִיאַת שֵׁם אוֹסַרְתּוֹ

מֶרְחָץ חָדָשׁ — אֵין הַשֵּׁם אוֹסֵר

היסוד. מֶרְחָץ חָדָשׁ שֶׁלֹּא רָחֲצוּ בּוֹ — מֻתָּר לִקְרוֹת בּוֹ ; אֵין קְרִיאַת הַשֵּׁם ״מֶרְחָץ״ אוֹסַרְתּוֹ כִּבְבֵית הַכִּסֵּא שֶׁהוּא מָאוּס בְּיוֹתֵר (סעיף א).

המסקנה. הָאוֹסֵר הוּא הַמְּצִיאוּת הַמְאוּסָה, לֹא הַכִּנּוּי. עיין סימן ע״ו · 76 (צואה).

הַבַּיִת הַחִיצוֹן — כֻּלָּם לְבוּשִׁים

הַבַּיִת הַחִיצוֹן — כֻּלָּם לְבוּשִׁים

היסוד. בַּבַּיִת הַחִיצוֹן שֶׁכָּל הָעוֹמְדִים לְבוּשִׁים — מֻתָּר לִקְרוֹת שָׁם ק״ש וּלְהִתְפַּלֵּל (סעיף א).

המסקנה. מֶרְחָץ ״יָשָׁן״ אֵינוֹ אָסוּר בְּמִקְשֶׁה אַחַת ; הַכֹּל לְפִי מַצַּב הַבַּיִת. עיין סימן ע״ט · 79 (ריח רע).

הַבַּיִת הָאֶמְצָעִי — דִּבּוּר מוּל הִרְהוּר

הַבַּיִת הָאֶמְצָעִי — דִּבּוּר מוּל הִרְהוּר

היסוד. יֵשׁ שָׁם שְׁאִילַת שָׁלוֹם, אֲבָל לֹא ק״ש וּתְפִלָּה וְדִבְרֵי תּוֹרָה, אֲפִלּוּ שֶׁאֵין שָׁם עָרֹם ; אֲבָל מֻתָּר לְהַרְהֵר וְלִפְסֹק דִּין בְּלֹא טַעַם (סעיף א).

המסקנה. הַדִּבּוּר שֶׁבִּקְדֻשָּׁה אָסוּר, הַמַּחֲשָׁבָה מֻתֶּרֶת — פְּסַק דִּין בְּלֹא טַעַם כְּהִרְהוּר. עיין סימן פ״א · 81 (מי רגלים).

הַבַּיִת הַפְּנִימִי — שֵׁם שָׁלוֹם

הַבַּיִת הַפְּנִימִי — שֵׁם שָׁלוֹם

היסוד. כֻּלָּם עֲרֻמִּים — אֲפִלּוּ שְׁאִילַת שָׁלוֹם אָסוּר, כִּי ״שָׁלוֹם״ שְׁמוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא ; אָסוּר לַעֲנוֹת אָמֵן וּלְהַרְהֵר בְּדָבָר שֶׁבִּקְדֻשָּׁה (סעיף א).

המסקנה. אָדָם שֶׁשְּׁמוֹ שָׁלוֹם — הַלֵּךְ אַחַר הַמֵּקֵל (וְכֵן נוֹהֲגִין) ; שְׁאֵלַת הֲלָכָה — אַף ״אֵין מְשִׁיבִין בַּמֶּרְחָץ״ אֵין אוֹמְרִים בַּפְּנִימִי, אֲבָל בָּאֶמְצָעִי אוֹמְרִים.

הפניה ללימוד. היקף הסימן — הַמֶּרְחָץ הַחָדָשׁ, שְׁלֹשֶׁת הַבָּתִּים וְשֵׁם שָׁלוֹם — נוגע להנהגה היומיומית במקומות רחצה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן פ״ד. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מסעיפו האחד.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
מֶרְחָץ חָדָשׁ מוּל בֵּית הַכִּסֵּא או״ח פ״ד — מֶרְחָץ חָדָשׁ שֶׁלֹּא רָחֲצוּ בּוֹ מֻתָּר לִקְרוֹת בּוֹ. מ״ב פ״ד — גֶּדֶר הַ״חָדָשׁ״ וּמָתַי נַעֲשֶׂה כְּיָשָׁן. ערוה״ש פ״ד — הַהֶבְדֵּל שֶׁבֵּין מֶרְחָץ לְבֵית הַכִּסֵּא. דרך אדמו״ר הזקן — חָדָשׁ מֻתָּר, שֶׁאֵין קְרִיאַת שֵׁם אוֹסַרְתּוֹ ; בֵּית הַכִּסֵּא מָאוּס בְּיוֹתֵר (סעיף א).
הַבָּתִּים — חִיצוֹן וְאֶמְצָעִי או״ח פ״ד — חִיצוֹן מֻתָּר ; אֶמְצָעִי שְׁאִילַת שָׁלוֹם אֲבָל לֹא ק״ש וּתְפִלָּה. מ״ב פ״ד — פְּרָטֵי שְׁלֹשֶׁת הַבָּתִּים וְדִינָם בַּזְּמַן הַזֶּה. ערוה״ש פ״ד — גֶּדֶר הָעֲרֻמִּים הַקּוֹבֵעַ אֶת הַבַּיִת. דרך אדמו״ר הזקן — חִיצוֹן ק״ש וּתְפִלָּה ; אֶמְצָעִי שָׁלוֹם וְלֹא ק״ש, אֲפִלּוּ אֵין שָׁם עָרֹם (סעיף א).
הַבַּיִת הַפְּנִימִי וּשְׁאִילַת שָׁלוֹם או״ח פ״ד — פְּנִימִי אֲפִלּוּ שְׁאִילַת שָׁלוֹם אָסוּר ; וְאָסוּר לַעֲנוֹת אָמֵן. מ״ב פ״ד — טַעַם ״שָׁלוֹם״ שְׁמוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא. ערוה״ש פ״ד — דִּין אָדָם שֶׁשְּׁמוֹ שָׁלוֹם. דרך אדמו״ר הזקן — אֲפִלּוּ שָׁלוֹם אָסוּר (״וַיִּקְרָא לוֹ ה׳ שָׁלוֹם״) ; שֶׁשְּׁמוֹ שָׁלוֹם — הַלֵּךְ אַחַר הַמֵּקֵל, וְכֵן נוֹהֲגִין (סעיף א).
הִרְהוּר, פְּסַק דִּין וַעֲנִיַּת אָמֵן או״ח פ״ד — הַרְהֵר בְּדִבְרֵי תּוֹרָה מֻתָּר (הַגָּהָה). מ״ב פ״ד — פְּרָטֵי הַהִרְהוּר וּפְסַק הַדִּין בְּלֹא טַעַם. ערוה״ש פ״ד — גֶּדֶר עֲנִיַּת אָמֵן וּתְשׁוּבַת הֲלָכָה. דרך אדמו״ר הזקן — מֻתָּר לְהַרְהֵר וְלִפְסֹק דִּין בְּלֹא טַעַם (כְּהִרְהוּר) ; בַּפְּנִימִי אָסוּר אָמֵן וַאֲפִלּוּ ״אֵין מְשִׁיבִין״ (סעיף א).

הטבלה נערכה מטקסט המחבר (אורח חיים פ״ד, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, ר״ן, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד שבת י׳, עבודה זרה מ״ד, שבועות ל״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (סָעִיף אֶחָד) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַמָּקוֹם לְפִי מַהוּתוֹ

קו א — הַמָּקוֹם לְפִי מַהוּתוֹ

יסוד הסימן : מֶרְחָץ חָדָשׁ מֻתָּר, שֶׁאֵין קְרִיאַת שֵׁם אוֹסַרְתּוֹ (סעיף א).

קריאת הרב : הָאוֹסֵר הוּא הַ״מָּאוּס בְּיוֹתֵר״ שֶׁל בֵּית הַכִּסֵּא, לֹא הַכִּנּוּי ״מֶרְחָץ״. עיין סימן ע״ו · 76.

מסקנה למעשה : לְהַעֲרִיךְ אֶת הַמָּקוֹם לְפִי מַצָּבוֹ הָאֲמִתִּי — מֶרְחָץ חָדָשׁ שֶׁלֹּא רָחֲצוּ בּוֹ אֵין דִּינוֹ כְּיָשָׁן.

קו ב — מִדְרָג לְפִי הָעֶרְוָה

קו ב — מִדְרָג לְפִי הָעֶרְוָה

יסוד הסימן : חִיצוֹן (לְבוּשִׁים) — ק״ש ; אֶמְצָעִי (מְעֹרָב) — שָׁלוֹם ; פְּנִימִי (עֲרֻמִּים) — אָסוּר (סעיף א).

קריאת הרב : שְׁלֹשָׁה בָּתִּים, שְׁלֹשָׁה דִּינִים ; מְצִיאוּת הָעֲרֻמִּים קוֹבַעַת אֶת הַמַּעֲמָד, לֹא עֶצֶם הַהִשְׁתַּיְּכוּת לַמֶּרְחָץ. עיין סימן ע״ט · 79.

מסקנה למעשה : בָּאֶמְצָעִי הַק״ש אֲסוּרָה אֲפִלּוּ בְּשָׁעָה שֶׁאֵין שָׁם עָרֹם.

קו ג — מַחֲשָׁבָה מוּל דִּבּוּר

קו ג — מַחֲשָׁבָה מוּל דִּבּוּר

יסוד הסימן : מֻתָּר לְהַרְהֵר וְלִפְסֹק דִּין בְּלֹא טַעַם ; בַּפְּנִימִי אֲפִלּוּ שָׁלוֹם וְאָמֵן אָסוּר (סעיף א).

קריאת הרב : קַו הַגְּבוּל הוּא הַדִּבּוּר ; פְּסַק דִּין בְּלֹא טַעַם נִשְׁאָר בַּלֵּב, כְּהִרְהוּר. עיין סימן פ״א · 81.

מסקנה למעשה : לְהַרְהֵר בַּתּוֹרָה מֻתָּר בְּכָל מָקוֹם ; לְאָמְרוֹ בַּפֶּה — לְפִי דִּין הַבַּיִת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן פ״ד ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן פ״ד (דִּין אִם מֻתָּר לִקְרוֹת בַּמֶּרְחָץ)

  • ① מֶרְחָץ חָדָשׁ. מֶרְחָץ חָדָשׁ שֶׁלֹּא רָחֲצוּ בּוֹ — מֻתָּר לִקְרוֹת וּלְהִתְפַּלֵּל ; אֵין הַשֵּׁם אוֹסֵר. יסוד : שוע״ר פ״ד סעיף א.
  • ② בֵּית הַחִיצוֹן. כֻּלָּם לְבוּשִׁים — מֻתָּר ק״ש וּתְפִלָּה. יסוד : שוע״ר פ״ד סעיף א.
  • ③ בֵּית הָאֶמְצָעִי. מָקוֹם מְעֹרָב — שְׁאִילַת שָׁלוֹם מֻתֶּרֶת ; ק״ש, תְּפִלָּה וְדִבְרֵי תּוֹרָה אֲסוּרִים, אֲפִלּוּ שֶׁאֵין שָׁם עָרֹם ; לְהַרְהֵר וְלִפְסֹק דִּין בְּלֹא טַעַם מֻתָּר. יסוד : שוע״ר פ״ד סעיף א.
  • ④ בֵּית הַפְּנִימִי. כֻּלָּם עֲרֻמִּים — אֲפִלּוּ שְׁאִילַת שָׁלוֹם אָסוּר (שֵׁם ה׳) ; אָסוּר לַעֲנוֹת אָמֵן וּלְהַרְהֵר בְּדָבָר שֶׁבִּקְדֻשָּׁה. יסוד : שוע״ר פ״ד סעיף א.
  • ⑤ שְׁאֵלַת הֲלָכָה. בַּפְּנִימִי אָסוּר אֲפִלּוּ לוֹמַר ״אֵין מְשִׁיבִין בַּמֶּרְחָץ״ ; בָּאֶמְצָעִי מֻתָּר לוֹמַר כָּךְ. אָדָם שֶׁשְּׁמוֹ שָׁלוֹם — הַלֵּךְ אַחַר הַמֵּקֵל (וְכֵן נוֹהֲגִין). יסוד : שוע״ר פ״ד סעיף א.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן פ״ד (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַמֶּרְחָץ, הַבָּתִּים, שֵׁם שָׁלוֹם — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
