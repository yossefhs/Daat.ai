דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קי״ח · חֲתִימַת בִּרְכַּת הָשִׁיבָה

דרכו של אדמו״ר הזקן בחֲתִימַת בִּרְכַּת הָשִׁיבָה : מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט — מַלְכוּת בַּחֲתִימָה, וּמֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר — הַמֶּלֶךְ הַמִּשְׁפָּט ; בִּרְכַּת הַמִּינִים כּוֹלֶלֶת אֶת הַזֵּדִים, וּבִרְכַּת הַצַּדִּיקִים כּוֹלֶלֶת גֵּרֵי צֶדֶק — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קי״ח — חֲתִימַת בִּרְכַּת הָשִׁיבָה. אצל הרב הסימן נפרש לב׳ סְעִיפִים : חֲתִימַת בִּרְכַּת הָשִׁיבָה — מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט, וְהַמֶּלֶךְ הַמִּשְׁפָּט מֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר — וְאַחַר כָּךְ כְּלִילַת הַזֵּדִים בְּבִרְכַּת הַמִּינִים וְהַגֵּרֵי צֶדֶק בְּבִרְכַּת הַצַּדִּיקִים.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קי״ח — ב׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קי״ח). אדמו״ר הזקן דן בו בַּחֲתִימַת בִּרְכַּת הָשִׁיבָה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קי״ח — חֲתִימַת בִּרְכַּת הָשִׁיבָה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קי״ח — חֲתִימַת בִּרְכַּת הָשִׁיבָה — וּבוֹ ב׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.118), ואחריו תרגומנו. הסימן כולל ב׳ סעיפים (סעיף k → ‎.118.k).

סעיף א חֲתִימַת בִּרְכַּת הָשִׁיבָה — מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט ; מֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר — הַמֶּלֶךְ הַמִּשְׁפָּט.

חֲתִימַת בִּרְכַּת הָשִׁיבָה וּבוֹ ב׳ סְעִיפִים:״הָשִׁיבָה שׁוֹפְטֵינוּ״ חוֹתֵם בָּהּ ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״, מַה שֶׁאֵין כֵּן בִּשְׁאָר בְּרָכוֹת שֶׁאֵין מַלְכוּת בַּחֲתִימָתָן אֶלָּא בְּמִשְׁפָּט בִּלְבַד, מִשּׁוּם שֶׁנֶּאֱמַר: ״מֶלֶךְ בְּמִשְׁפָּט יַעֲמִיד אָרֶץ״. וּמֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר חוֹתֵם ״הַמֶּלֶךְ הַמִּשְׁפָּט״, וּפֵרוּשׁוֹ הוּא מֶלֶךְ הַמִּשְׁפָּט, וְכֵן הוּא לְשׁוֹן הַמִּקְרָא לִפְעָמִים כְּמוֹ ״הָאָרוֹן הַבְּרִית״ וְדוֹמָיו. וְאִם אָמַר ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ — יִתְבָּאֵר בְּסִימָן תקפ״ב:

חֲתִימַת בִּרְכַּת הָשִׁיבָה. בִּרְכַּת ״הָשִׁיבָה שׁוֹפְטֵינוּ״ חוֹתֶמֶת בְּ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ — בְּמַלְכוּת, מַה שֶּׁאֵין כֵּן בִּשְׁאָר בְּרָכוֹת שֶׁאֵין מַלְכוּת בַּחֲתִימָתָן אֶלָּא מִשְׁפָּט בִּלְבַד, מִשּׁוּם ״מֶלֶךְ בְּמִשְׁפָּט יַעֲמִיד אָרֶץ״. מֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר חוֹתְמִים ״הַמֶּלֶךְ הַמִּשְׁפָּט״, פֵּרוּשׁוֹ מֶלֶךְ הַמִּשְׁפָּט (לְשׁוֹן הַמִּקְרָא, כְּמוֹ ״הָאָרוֹן הַבְּרִית״). וְהָאוֹמֵר ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ — יִתְבָּאֵר בְּסִימָן תקפ״ב.

סעיף ב בִּרְכַּת הַמִּינִים כּוֹלֶלֶת אֶת הַזֵּדִים ; בִּרְכַּת הַצַּדִּיקִים כּוֹלֶלֶת גֵּרֵי צֶדֶק.

בְּ״בִרְכַּת הַמִּינִים״ צָרִיךְ לִכְלֹל גַּם הַ״זֵּדִים״, שֶׁנֶּאֱמַר: ״וְשֶׁבֶר פֹּשְׁעִים וְחַטָּאִים יַחְדָּו וְעֹזְבֵי ה׳ יִכְלוּ״, ״עֹזְבֵי ה׳״ אֵלּוּ הַמִּינִים שֶׁאֵינָם מַאֲמִינִים בְּדַת מֹשֶׁה שֶׁהָיְתָה מִן הַשָּׁמַיִם, וּ״פֹשְׁעִים״ אֵלּוּ הַזֵּדִים, כִּי פּוֹשְׁעִים אֵלּוּ הַמּוֹרְדִים, כְּמוֹ שֶׁכָּתוּב: ״מֶלֶךְ מוֹאָב פָּשַׁע בִּי״. וּבְ״בִרְכַּת הַצַּדִּיקִים״ צָרִיךְ לִכְלֹל גַּם גֵּרֵי צֶדֶק, מִשּׁוּם שֶׁנֶּאֱמַר: ״מִפְּנֵי שֵׂיבָה תָּקוּם וְגו׳״, וּכְתִיב אַחֲרָיו: ״וְכִי יָגוּר אִתְּךָ גֵּר וְגו׳״, הֲרֵי גֵּרִים סְמוּכִים לַצַּדִּיקִים, לְכָךְ כּוֹלְלָם בִּבְרָכָה אַחַת:

הַמִּינִים וְהַצַּדִּיקִים. בְּבִרְכַּת הַמִּינִים צָרִיךְ לִכְלֹל גַּם אֶת הַזֵּדִים, שֶׁנֶּאֱמַר ״וְשֶׁבֶר פֹּשְׁעִים וְחַטָּאִים יַחְדָּו וְעֹזְבֵי ה׳ יִכְלוּ״ : ״עֹזְבֵי ה׳״ אֵלּוּ הַמִּינִים שֶׁאֵינָם מַאֲמִינִים בְּדַת מֹשֶׁה מִן הַשָּׁמַיִם, וּ״פֹשְׁעִים״ אֵלּוּ הַזֵּדִים — הַמּוֹרְדִים, כְּמוֹ ״מֶלֶךְ מוֹאָב פָּשַׁע בִּי״. וּבְבִרְכַּת הַצַּדִּיקִים צָרִיךְ לִכְלֹל גַּם גֵּרֵי צֶדֶק, שֶׁנֶּאֱמַר ״מִפְּנֵי שֵׂיבָה תָּקוּם״ וְאַחֲרָיו ״וְכִי יָגוּר אִתְּךָ גֵּר״ — גֵּרִים סְמוּכִים לַצַּדִּיקִים, לְכָךְ כּוֹלְלָם בִּבְרָכָה אַחַת.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קי״ח (מהדורת קה״ת, ב׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — חֲתִימַת בִּרְכַּת הָשִׁיבָה

היסוד : חֲתִימַת בִּרְכַּת הָשִׁיבָה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבע נקודות. תחילה חֲתִימַת בִּרְכַּת הָשִׁיבָה — מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט — בְּמַלְכוּת. אחר כך הַשִּׁנּוּי מֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר — הַמֶּלֶךְ הַמִּשְׁפָּט. אחר כך כְּלִילַת הַזֵּדִים בְּבִרְכַּת הַמִּינִים. ולבסוף כְּלִילַת גֵּרֵי צֶדֶק בְּבִרְכַּת הַצַּדִּיקִים.

מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט — מַלְכוּת בַּחֲתִימָה

מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט — מַלְכוּת בַּחֲתִימָה

היסוד. בִּרְכַּת הָשִׁיבָה חוֹתֶמֶת בְּ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ ; מַה שֶּׁאֵין כֵּן שְׁאָר בְּרָכוֹת שֶׁאֵין מַלְכוּת בַּחֲתִימָתָן אֶלָּא מִשְׁפָּט (סעיף א).

המסקנה. הַטַּעַם : ״מֶלֶךְ בְּמִשְׁפָּט יַעֲמִיד אָרֶץ״ — הַמִּשְׁפָּט שֶׁבִּבְרָכָה זוֹ מְעוֹרֵר אֶת הַמַּלְכוּת. עיין סימן 117.

הַמֶּלֶךְ הַמִּשְׁפָּט — מֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר

הַמֶּלֶךְ הַמִּשְׁפָּט — מֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר

היסוד. מֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר חוֹתְמִים ״הַמֶּלֶךְ הַמִּשְׁפָּט״, פֵּרוּשׁוֹ מֶלֶךְ הַמִּשְׁפָּט — לְשׁוֹן הַמִּקְרָא (כְּמוֹ ״הָאָרוֹן הַבְּרִית״) (סעיף א).

המסקנה. הָאוֹמֵר ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ בְּיָמִים אֵלּוּ — יִתְבָּאֵר בְּסִימָן תקפ״ב. עיין סימן 119.

בִּרְכַּת הַמִּינִים — הַזֵּדִים

בִּרְכַּת הַמִּינִים — הַזֵּדִים

היסוד. בְּבִרְכַּת הַמִּינִים צָרִיךְ לִכְלֹל גַּם אֶת הַזֵּדִים, שֶׁנֶּאֱמַר ״וְשֶׁבֶר פֹּשְׁעִים וְחַטָּאִים יַחְדָּו וְעֹזְבֵי ה׳ יִכְלוּ״ (סעיף ב).

המסקנה. ״עֹזְבֵי ה׳״ אֵלּוּ הַמִּינִים ; ״פֹשְׁעִים״ אֵלּוּ הַזֵּדִים, הַמּוֹרְדִים (״מֶלֶךְ מוֹאָב פָּשַׁע בִּי״). עיין סימן 117.

בִּרְכַּת הַצַּדִּיקִים — גֵּרֵי צֶדֶק

בִּרְכַּת הַצַּדִּיקִים — גֵּרֵי צֶדֶק

היסוד. בְּבִרְכַּת הַצַּדִּיקִים צָרִיךְ לִכְלֹל גַּם גֵּרֵי צֶדֶק, שֶׁנֶּאֱמַר ״מִפְּנֵי שֵׂיבָה תָּקוּם״ וְאַחֲרָיו ״וְכִי יָגוּר אִתְּךָ גֵּר״ (סעיף ב).

המסקנה. הַגֵּרִים סְמוּכִים לַצַּדִּיקִים בַּתּוֹרָה — לְכָךְ כּוֹלְלָם בִּבְרָכָה אַחַת. עיין סימן 119.

הפניה ללימוד. היקף הסימן — הַחֲתִימָה בְּמַלְכוּת, הַשִּׁנּוּי בַּעֲשֶׂרֶת יְמֵי תְשׁוּבָה וּכְלִילַת הַזֵּדִים וְגֵרֵי צֶדֶק — נוגע לנוסח התפילה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

חֲתִימַת בִּרְכַּת הָשִׁיבָה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קי״ח. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מב׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
חֲתִימַת הָשִׁיבָה בְּמַלְכוּת או״ח קי״ח — הָשִׁיבָה חוֹתֵם בָּהּ ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״. מ״ב קי״ח — טַעַם הַמַּלְכוּת בַּחֲתִימָה זוֹ. ערוה״ש קי״ח — גֶּדֶר הַחֲתִימָה בְּמַלְכוּת. דרך אדמו״ר הזקן — ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״, מַלְכוּת בַּחֲתִימָה, מִשּׁוּם ״מֶלֶךְ בְּמִשְׁפָּט יַעֲמִיד אָרֶץ״ (סעיף א).
הַמֶּלֶךְ הַמִּשְׁפָּט בַּעֲשֶׂרֶת יְמֵי תְשׁוּבָה או״ח קי״ח — מֵרֹאשׁ הַשָּׁנָה עַד יוֹם הַכִּפּוּרִים חוֹתֵם ״הַמֶּלֶךְ הַמִּשְׁפָּט״. מ״ב קי״ח — דִּין הַטּוֹעֶה בַּחֲתִימָה בַּעֲשֶׂרֶת יְמֵי תְשׁוּבָה. ערוה״ש קי״ח — פֵּרוּשׁ ״הַמֶּלֶךְ הַמִּשְׁפָּט״. דרך אדמו״ר הזקן — ״הַמֶּלֶךְ הַמִּשְׁפָּט״, פֵּרוּשׁוֹ מֶלֶךְ הַמִּשְׁפָּט (כְּמוֹ ״הָאָרוֹן הַבְּרִית״) ; אָמַר ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ — יִתְבָּאֵר בְּסִימָן תקפ״ב (סעיף א).
בִּרְכַּת הַמִּינִים וְהַזֵּדִים או״ח קי״ח — עוֹסֵק בַּחֲתִימַת הָשִׁיבָה ; דִּין הַזֵּדִים בְּבִרְכַּת הַמִּינִים — עיין בְּנֻסַּח הַבְּרָכָה. מ״ב — כְּלִילַת הַזֵּדִים בְּבִרְכַּת הַמִּינִים. ערוה״ש — הַזֵּדִים וְהַמִּינִים בַּבְּרָכָה. דרך אדמו״ר הזקן — לִכְלֹל אֶת הַ״זֵּדִים״ ; ״עֹזְבֵי ה׳״ אֵלּוּ הַמִּינִים, וּ״פֹשְׁעִים״ אֵלּוּ הַזֵּדִים (״מֶלֶךְ מוֹאָב פָּשַׁע בִּי״) (סעיף ב).
בִּרְכַּת הַצַּדִּיקִים וְגֵרֵי צֶדֶק או״ח קי״ח — דִּין גֵּרֵי צֶדֶק בְּבִרְכַּת הַצַּדִּיקִים — עיין בְּנֻסַּח הַבְּרָכָה. מ״ב — הַזְכָּרַת גֵּרֵי צֶדֶק עִם הַצַּדִּיקִים. ערוה״ש — סְמִיכוּת גֵּרִים לַצַּדִּיקִים. דרך אדמו״ר הזקן — לִכְלֹל גֵּרֵי צֶדֶק ; ״מִפְּנֵי שֵׂיבָה תָּקוּם״ וְאַחֲרָיו ״וְכִי יָגוּר אִתְּךָ גֵּר״ — גֵּרִים סְמוּכִים לַצַּדִּיקִים (סעיף ב).

הטבלה נערכה מטקסט המחבר (אורח חיים קי״ח, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ח-כ״ט). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ב׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — חֲתִימָה בְּמַלְכוּת

קו א — חֲתִימָה בְּמַלְכוּת

יסוד הסימן : בִּרְכַּת הָשִׁיבָה חוֹתֵם ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״, מַה שֶּׁאֵין כֵּן בִּשְׁאָר בְּרָכוֹת (סעיף א).

קריאת הרב : הַמִּשְׁפָּט שֶׁבִּבְרָכָה זוֹ מְעוֹרֵר אֶת הַמַּלְכוּת — ״מֶלֶךְ בְּמִשְׁפָּט יַעֲמִיד אָרֶץ״. עיין סימן 117.

מסקנה למעשה : לַחְתֹּם בִּרְכַּת הָשִׁיבָה בְּ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ כָּל הַשָּׁנָה.

קו ב — הַמֶּלֶךְ הַמִּשְׁפָּט

קו ב — הַמֶּלֶךְ הַמִּשְׁפָּט

יסוד הסימן : מֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר חוֹתֵם ״הַמֶּלֶךְ הַמִּשְׁפָּט״ (סעיף א).

קריאת הרב : פֵּרוּשׁוֹ הוּא מֶלֶךְ הַמִּשְׁפָּט — לְשׁוֹן הַמִּקְרָא (כְּמוֹ ״הָאָרוֹן הַבְּרִית״). עיין סימן 119.

מסקנה למעשה : בַּעֲשֶׂרֶת יְמֵי תְשׁוּבָה לַחְתֹּם ״הַמֶּלֶךְ הַמִּשְׁפָּט״ ; הָאוֹמֵר ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ — יִתְבָּאֵר בְּסִימָן תקפ״ב.

קו ג — כְּלִילַת הַזֵּדִים וְגֵרֵי צֶדֶק

קו ג — כְּלִילַת הַזֵּדִים וְגֵרֵי צֶדֶק

יסוד הסימן : בִּרְכַּת הַמִּינִים כּוֹלֶלֶת הַזֵּדִים, וּבִרְכַּת הַצַּדִּיקִים כּוֹלֶלֶת גֵּרֵי צֶדֶק (סעיף ב).

קריאת הרב : כָּל כְּלִילָה נִסְמֶכֶת עַל סָמוּךְ בַּמִּקְרָא — ״עֹזְבֵי ה׳״/״פֹשְׁעִים״ לַמִּינִים, וּסְמִיכוּת הַגֵּר לַצַּדִּיקִים. עיין סימן 117.

מסקנה למעשה : נֻסַּח שְׁתֵּי הַבְּרָכוֹת כּוֹלֵל זֵדִים וְגֵרֵי צֶדֶק, כָּל אֶחָד בִּבְרָכָתוֹ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קי״ח ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קי״ח (חֲתִימַת בִּרְכַּת הָשִׁיבָה)

  • ① חֲתִימָה. לַחְתֹּם בִּרְכַּת הָשִׁיבָה בְּ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ — בְּמַלְכוּת, מַה שֶּׁאֵין כֵּן בִּשְׁאָר בְּרָכוֹת. יסוד : שוע״ר קי״ח סעיף א.
  • ② הַטַּעַם. הַמַּלְכוּת כָּאן מִשּׁוּם הַפָּסוּק ״מֶלֶךְ בְּמִשְׁפָּט יַעֲמִיד אָרֶץ״. יסוד : שוע״ר קי״ח סעיף א.
  • ③ עֲשֶׂרֶת יְמֵי תְשׁוּבָה. מֵרֹאשׁ הַשָּׁנָה עַד יוֹם כִּפּוּר — לַחְתֹּם ״הַמֶּלֶךְ הַמִּשְׁפָּט״ (= מֶלֶךְ הַמִּשְׁפָּט) ; הָאוֹמֵר ״מֶלֶךְ אוֹהֵב...״ — יִתְבָּאֵר בְּסִימָן תקפ״ב. יסוד : שוע״ר קי״ח סעיף א.
  • ④ בִּרְכַּת הַמִּינִים. לִכְלֹל גַּם אֶת הַזֵּדִים — ״עֹזְבֵי ה׳״ (הַמִּינִים) וּ״פֹשְׁעִים״ (הַזֵּדִים, הַמּוֹרְדִים). יסוד : שוע״ר קי״ח סעיף ב.
  • ⑤ בִּרְכַּת הַצַּדִּיקִים. לִכְלֹל גַּם גֵּרֵי צֶדֶק — סְמוּכִים לַצַּדִּיקִים (״מִפְּנֵי שֵׂיבָה תָּקוּם״ · ״וְכִי יָגוּר אִתְּךָ גֵּר״). יסוד : שוע״ר קי״ח סעיף ב.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קי״ח (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַחֲתִימָה, נֻסַּח הַבְּרָכוֹת, עֲשֶׂרֶת יְמֵי תְשׁוּבָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
