דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן נ״ב · דִּין מִי שֶׁשָּׁהָה לָבוֹא לְבֵית הַכְּנֶסֶת עַד יִשְׁתַּבַּח

דרכו של אדמו״ר הזקן בדִּין מִי שֶׁשָּׁהָה לָבוֹא לְבֵית הַכְּנֶסֶת עַד יִשְׁתַּבַּח: התְּפִלַּת צִבּוּר לְעוֹלָם רְצוּיָה וּמְקֻבֶּלֶת, לְפִיכָךְ הַמְאַחֵר מְדַלֵּג אֶת פְּסוּקֵי דְּזִמְרָה לְפִי הַשְּׁהוּת כְּדֵי לְהִתְפַּלֵּל י״ח עִם הַצִּבּוּר ; וְאִם הִתְחִילוּ יוֹצֵר — אוֹמֵר פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה בְּלֹא בְּרָכָה, וּמְבָרֵךְ כָּל בִּרְכוֹת הַשַּׁחַר — ברגישות חב״ד.

למה רמה 4 מיוחדת לאדמו״ר הזקן? השולחן ערוך של אדמו״ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתבו אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. דף זה מאסף את שיטת הרב על סימן נ״ב — דִּין מִי שֶׁשָּׁהָה לָבוֹא לְבֵית הַכְּנֶסֶת עַד יִשְׁתַּבַּח. הסימן נפרש אצל הרב בב׳ סְעִיפִים: המוּטָב שֶׁל הַתְּפִלָּה בְּצִבּוּר, סֵדֶר הַדִּילּוּג לְפִי הַשְּׁהוּת (ב״ש, תְּהִלָּה לְדָוִד, מִזְמוֹרֵי הַלְלוּיָהּ, יִשְׁתַּבַּח), פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה בְּלֹא בְּרָכָה, וְחִיּוּב כָּל בִּרְכוֹת הַשַּׁחַר ; וְאַחַר כָּךְ דִּין הַבָּא בְּיִשְׁתַּבַּח אוֹ סָמוּךְ לְק״ש, וְהַלְכַתְּחִלָּה לְזָרֵז לָבֹא לְבֵית הַכְּנֶסֶת.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בדף זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן נ״ב — ב׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנסרקה בספריא (Shulchan Arukh HaRav, Orach Chayim נ״ב). אדמו״ר הזקן דן במי שאיחר ומצא ציבור בסוף פְּסוּקֵי דְּזִמְרָה: הַמּוּטָב לְהִתְפַּלֵּל עִם הַצִּבּוּר, סֵדֶר הַדִּילּוּג לְפִי הַשְּׁהוּת, פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה בְּלֹא בְּרָכָה, וּבִרְכוֹת הַשַּׁחַר. הביאורים הם שלנו. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן נ״ב — דִּין מִי שֶׁשָּׁהָה לָבוֹא לְבֵית הַכְּנֶסֶת עַד יִשְׁתַּבַּח

הטקסט המלא של שולחן ערוך אדמו״ר הזקן

שולחן ערוך הרב, סימן נ״ב — דִּין מִי שֶׁשָּׁהָה לָבוֹא לְבֵית הַכְּנֶסֶת עַד יִשְׁתַּבַּח — וּבוֹ ב׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.52), ולאחריו ביאור קצר. הסימן כולל ב׳ סְעִיפִים.

סעיף א מִי שֶׁבָּא וּמָצָא צִבּוּר בְּסוֹף פְּסוּקֵי דְּזִמְרָה — מְדַלֵּג כְּדֵי לְהִתְפַּלֵּל עִם הַצִּבּוּר, וְסֵדֶר הַדִּילּוּג לְפִי הַשְּׁהוּת ; וְאִם הִתְחִילוּ יוֹצֵר — קוֹרֵא פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה בְּלֹא בְּרָכָה.

דִּין מִי שֶׁשָּׁהָה לָבוֹא לְבֵית הַכְּנֶסֶת עַד ״יִשְׁתַּבַּח״ וּבוֹ ב׳ סְעִיפִים:תְּפִלַּת צִּבּוּר הִיא רְצוּיָהּ וּמְקֻבֶּלֶת לְעוֹלָם, שֶׁנֶּאֱמַר: ״כַּבִּיר לֹא יִמְאָס״, וְאוֹמֵר: ״פָּדָה בְשָׁלוֹם נַפְשִׁי מִקְּרָב לִי כִּי בְרַבִּים הָיוּ עִמָּדִי״. לְפִיכָךְ מִי שֶׁבָּא לְבֵית הַכְּנֶסֶת וּמָצָא צִבּוּר בְּסוֹף פְּסוּקֵי דְּזִמְרָה — יֵשׁ לוֹ לְדַלֵּג פְּסוּקֵי דְּזִמְרָה כְּדֵי לְהִתְפַּלֵּל עִם הַצִּבּוּר, שֶׁפְּסוּקֵי דְּזִמְרָה גַּם כֵּן לֹא נִתְקְנוּ אֶלָּא בִּשְׁבִיל הַתְּפִלָּה שֶׁתִּהְיֶה רְצוּיָה וּמְקֻבֶּלֶת עַל יְדֵי שֶׁמְּסַדֵּר שִׁבְחוֹ שֶׁל מָקוֹם תְּחִלָּה וְאַחַר כָּךְ יִתְפַּלֵּל, וְאִם כֵּן מוּטָב לְהִתְפַּלֵּל עִם הַצִּבּוּר שֶׁאָז תְּפִלָּתוֹ מְקֻבֶּלֶת וַדַּאי. וְכֵיצַד מְדַלֵּג? אִם יֵשׁ לוֹ שְׁהוּת לִקְרוֹת מִזְמוֹר ״תְּהִלָּה לְדָוִד״ — דַּי בְּמִזְמוֹר זֶה לְבָרֵךְ לְפָנָיו ״בָּרוּךְ שֶׁאָמַר״ וְאַחֲרָיו ״יִשְׁתַּבַּח״. וְאַף שֶׁתִּקְּנוּ לְבָרֵךְ וְלִקְרוֹת מִ״תְּהִלָּה לְדָוִד״ עַד ״כֹּל הַנְּשָׁמָה״, מִכָּל מָקוֹם כֵּיוָן שֶׁעִקַּר הַתַּקָּנָה לֹא הָיְתָה אֶלָּא בִּשְׁבִיל מִזְמוֹר זֶה כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן נ״א — דַּי לוֹ בְּמִזְמוֹר זֶה, כְּדֵי שֶׁיִּקְרָא קְרִיאַת שְׁמַע וְיִתְפַּלֵּל עִם הַצִבּוּר.וְאִם יֵשׁ לוֹ שְׁהוּת יוֹתֵר — יֹאמַר אַחַר ״תְּהִלָּה לְדָוִד״ ב׳ מִזְמוֹרִים שֶׁל הִלּוּלִים, שֶׁהֵם: ״הַלְלוּ אֶת ה׳ מִן הַשָּׁמַיִם״ וּמִזְמוֹר ״הַלְלוּ אֵל בְּקָדְשׁוֹ״. וְאִם אֵין לוֹ שְׁהוּת לוֹמַר שְׁנֵיהֶם — יֹאמַר ״הַלְלוּ אֵל בְּקָדְשׁוֹ״ לְבַדּוֹ.וְאִם יֵשׁ לוֹ שְׁהוּת יוֹתֵר — יֹאמַר כָּל ה׳ מִזְמוֹרֵי ״הַלְלוּיָהּ״.וְאִם יֵשׁ לוֹ שְׁהוּת יוֹתֵר — יֹאמַר גַּם ״וַיְבָרֶךְ דָּוִיד״ עַד ״לְשֵׁם תִּפְאַרְתֶּךָ״ וּמְדַלֵּג עַד ״יִשְׁתַּבַּח״.וְאִם יֵשׁ לוֹ שְׁהוּת יוֹתֵר — יֹאמַר ״הוֹדוּ לַה׳ קִרְאוּ״ עַד ״וְהוּא רַחוּם״, וּמְדַלֵּג עַד ״וְהוּא רַחוּם״ שֶׁלִּפְנֵי ״אַשְׁרֵי״, כִּי בֵּינְתַיִם אֵינָם רַק פְּסוּקִים מְלֻקָּטִים.וְאִם כְּבָר הִתְחִילוּ הַצִּבּוּר יוֹצֵר וְאֵין לוֹ שְׁהוּת לוֹמַר פְּסוּקֵי דְּזִמְרָה אֲפִלּוּ בְּדִלּוּג — יִקְרָא קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ עִם הַצִּבּוּר וְיִתְפַּלֵּל עִמָּהֶם.וְאַחַר כָּךְ, יֵשׁ אוֹמְרִים שֶׁצָּרִיךְ לִקְרוֹת כָּל הַפְּסוּקֵי דְּזִמְרָה עִם הַבְּרָכוֹת שֶׁלִּפְנֵיהֶם וּלְאַחֲרֵיהֶם. וְיֵשׁ אוֹמְרִים שֶׁאֵין צָרִיךְ כָּל עִקָּר, לְפִי שֶׁלֹּא נִתְקְנוּ אֶלָּא קֹדֶם הַתְּפִלָּה לְפִיכָךְ אֵין לוֹ לְבָרֵךְ כְּלָל לֹא לִפְנֵיהֶם וְלֹא לְאַחֲרֵיהֶם, אֶלָּא אִם רוֹצֶה לִקְרוֹתָם בְּלֹא בְּרָכָה יָפֶה הוּא עוֹשֶׂה, וְהַלְוַאי שֶׁיֹּאמְרֵם כָּל הַיּוֹם. וּלְעִנְיַן הֲלָכָה יֵשׁ לוֹ לִקְרוֹתָם אַחַר הַתְּפִלָּה כְּדֵי לָחֹש לַסְּבָרָא הָרִאשׁוֹנָה, אֲבָל לֹא יְבָרֵךְ לֹא ״בָּרוּךְ שֶׁאָמַר״ וְלֹא ״יִשְׁתַּבַּח״, שֶׁסְּפֵק בְּרָכוֹת לְהָקֵל.וּמִכָּל מָקוֹם יֹאמַר כָּל הַבְּרָכוֹת שֶׁמְּחֻיָּב לְבָרֵךְ בַּבֹּקֶר, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן מ״ו. וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁלֹּא יְבָרֵךְ בִּרְכַּת ״אֱלֹהַי נְשָׁמָה״, מִפְּנֵי שֶׁיָּצָא בְּבִרְכַּת מְחַיֶּה הַמֵּתִים שֶׁבַּתְּפִלָּה. וְטוֹב לָחֹש לִדְבָרָיו לְכַתְּחִלָּה לְבָרֵךְ אוֹתָהּ קֹדֶם הַתְּפִלָּה:

דִּין הַמְאַחֵר — דִּילּוּג כְּדֵי לְהִתְפַּלֵּל עִם הַצִּבּוּר. התְּפִלַּת צִבּוּר לְעוֹלָם רְצוּיָה וּמְקֻבֶּלֶת, שֶׁנֶּאֱמַר ״כַּבִּיר לֹא יִמְאָס״. לְפִיכָךְ הַבָּא וּמָצָא צִבּוּר בְּסוֹף פְּסוּקֵי דְּזִמְרָה — יֵשׁ לוֹ לְדַלֵּג כְּדֵי לְהִתְפַּלֵּל עִם הַצִּבּוּר, שֶׁפְּסוּקֵי דְּזִמְרָה לֹא נִתְקְנוּ אֶלָּא בִּשְׁבִיל הַתְּפִלָּה, וְאִם כֵּן מוּטָב לְהִתְפַּלֵּל עִם הַצִּבּוּר שֶׁתְּפִלָּתוֹ מְקֻבֶּלֶת וַדַּאי. וְכֵיצַד מְדַלֵּג ? יֵשׁ שְׁהוּת לְ״תְּהִלָּה לְדָוִד״ — דַּי בּוֹ, לְבָרֵךְ לְפָנָיו ״בָּרוּךְ שֶׁאָמַר״ וְאַחֲרָיו ״יִשְׁתַּבַּח״ (עִקַּר הַתַּקָּנָה בִּשְׁבִיל מִזְמוֹר זֶה, עַיֵּן סימן נ״א). שְׁהוּת יוֹתֵר — ב׳ מִזְמוֹרֵי הִלּוּלִים ״הַלְלוּ אֶת ה׳ מִן הַשָּׁמַיִם״ וְ״הַלְלוּ אֵל בְּקָדְשׁוֹ״ (וּבְאֵין שְׁהוּת לִשְׁנֵיהֶם — ״הַלְלוּ אֵל בְּקָדְשׁוֹ״ לְבַדּוֹ) ; אַחַר כָּךְ כָּל ה׳ מִזְמוֹרֵי ״הַלְלוּיָהּ״ ; אַחַר כָּךְ ״וַיְבָרֶךְ דָּוִיד״ ; אַחַר כָּךְ ״הוֹדוּ לַה׳ קִרְאוּ״ עַד ״וְהוּא רַחוּם״ (שֶׁבֵּינְתַיִם פְּסוּקִים מְלֻקָּטִים). וְאִם הִתְחִילוּ ״יוֹצֵר״ וְאֵין שְׁהוּת אֲפִלּוּ בְּדִלּוּג — קוֹרֵא ק״ש וּבִרְכוֹתֶיהָ עִם הַצִּבּוּר וּמִתְפַּלֵּל עִמָּהֶם, וְאַחַר כָּךְ קוֹרֵא כָּל פְּסוּקֵי דְּזִמְרָה בְּלֹא בְּרָכָה (שֶׁסְּפֵק בְּרָכוֹת לְהָקֵל). וּמִכָּל מָקוֹם אוֹמֵר כָּל בִּרְכוֹת הַשַּׁחַר (עַיֵּן סימן מ״ו) ; וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁלֹּא לְבָרֵךְ ״אֱלֹהַי נְשָׁמָה״ (שֶׁיָּצָא בִּמְחַיֶּה הַמֵּתִים), וְטוֹב לָחֹש לְבָרְכָהּ קֹדֶם הַתְּפִלָּה.

סעיף ב בָּא כְּשֶׁהִתְחִילוּ ״יִשְׁתַּבַּח״ — יִשְׁמָעֶנּוּ מֵהַשַּׁ״ץ ; סָמוּךְ לְק״ש — יַתְחִיל ״יוֹצֵר״ לְהִתְפַּלֵּל י״ח עִם הַצִּבּוּר ; וּלְכַתְּחִלָּה יְזָרֵז לָבֹא לְבֵית הַכְּנֶסֶת.

אִם בָּא כְּשֶׁהִתְחִילוּ הַקָּהָל ״יִשְׁתַּבַּח״ — טוֹב שֶׁיִּשְׁמַע ״יִשְׁתַּבַּח״ מֵהַשְּׁלִיחַ צִבּוּר. וְאִם בָּא סָמוּךְ לִקְרִיאַת שְׁמַע וְיָכוֹל לְהַתְחִיל ״יוֹצֵר״ וּלְהַגִּיעַ לְהִתְפַּלֵּל י״ח עִם הַצִּבּוּר — יַעֲשֶׂה כֵן, אַף עַל פִּי שֶׁלֹּא יִקְרָא קְרִיאַת שְׁמַע עִם הַצִבּוּר, כִּי עִקַּר הַקְּפֵדָה הִיא שֶׁיִּתְפַּלֵּל י״ח עִם הַצִּבּוּר. וְכָל זֶה בְּדִיעֲבַד שֶׁעָבַר וְשָׁהָה לָבֹא לְבֵית הַכְּנֶסֶת, אֲבָל לְכַתְּחִלָּה חַיָּב כָּל אָדָם לְזָרֵז אֶת עַצְמוֹ לָבֹא לְבֵית הַכְּנֶסֶת, בְּעִנְיָן שֶׁלֹּא יִצְטָרֵךְ לְדַלֵּג כְּלוּם כְּדֵי לְהִתְפַּלֵּל תְּפִלַּת י״ח עִם הַצִּבּוּר:

בָּא בְּ״יִשְׁתַּבַּח״ אוֹ סָמוּךְ לְק״ש. בָּא כְּשֶׁהִתְחִילוּ הַקָּהָל ״יִשְׁתַּבַּח״ — טוֹב שֶׁיִּשְׁמָעֶנּוּ מֵהַשַּׁ״ץ. וְאִם בָּא סָמוּךְ לִקְרִיאַת שְׁמַע וְיָכוֹל לְהַתְחִיל ״יוֹצֵר״ וּלְהַגִּיעַ לְהִתְפַּלֵּל י״ח עִם הַצִּבּוּר — יַעֲשֶׂה כֵן, אַף שֶׁלֹּא יִקְרָא ק״ש עִם הַצִּבּוּר, כִּי עִקַּר הַקְּפֵדָה שֶׁיִּתְפַּלֵּל י״ח עִם הַצִּבּוּר. וְכָל זֶה בְּדִיעֲבַד ; אֲבָל לְכַתְּחִלָּה חַיָּב כָּל אָדָם לְזָרֵז אֶת עַצְמוֹ לָבֹא לְבֵית הַכְּנֶסֶת, בְּעִנְיָן שֶׁלֹּא יִצְטָרֵךְ לְדַלֵּג כְּלוּם כְּדֵי לְהִתְפַּלֵּל י״ח עִם הַצִּבּוּר.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim נ״ב (מהדורת קה״ת, ב׳ סְעִיפִים). ביאור: DAAT. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.

2 · היסוד

היסוד — תְּפִלָּה בְּצִבּוּר וְסֵדֶר הַדִּילּוּג

היסוד: קְדִימַת הַתְּפִלָּה בְּצִבּוּר וְסֵדֶר הַדִּילּוּג

קריאת אדמו״ר הזקן בסימן זה בְּשָׁלֹשׁ תְּנוּעוֹת. תְּחִלָּה המוּטָב: כֵּיוָן שֶׁתְּפִלַּת צִבּוּר לְעוֹלָם רְצוּיָה וּמְקֻבֶּלֶת, מְדַלְּגִים פְּסוּקֵי דְּזִמְרָה כְּדֵי לְהִתְפַּלֵּל עֲמִידָה עִם הַצִּבּוּר (עַיֵּן סימן צ׳, סימן קי״א). אַחַר כָּךְ סֵדֶר הַדִּילּוּג לְפִי הַשְּׁהוּת. וּלְבַסּוֹף, אִם הִתְחִילוּ יוֹצֵר — פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה בְּלֹא בְּרָכָה וְחִיּוּב בִּרְכוֹת הַשַּׁחַר.

תְּפִלַּת צִבּוּר וּפְסוּקֵי דְּזִמְרָה — מוּטָב לְהִתְפַּלֵּל עִם הַצִּבּוּר

קְדִימַת הַתְּפִלָּה עִם הַצִּבּוּר

הַיְסוֹד. התְּפִלַּת צִבּוּר לְעוֹלָם רְצוּיָה, שֶׁנֶּאֱמַר ״כַּבִּיר לֹא יִמְאָס״ ; וּפְסוּקֵי דְּזִמְרָה לֹא נִתְקְנוּ אֶלָּא בִּשְׁבִיל הַתְּפִלָּה.

הַתּוֹצָאָה. לְפִיכָךְ מוּטָב לְדַלֵּג וּלְהִתְפַּלֵּל י״ח עִם הַצִּבּוּר. עַיֵּן סימן צ׳.

סֵדֶר הַדִּילּוּג לְפִי הַשְּׁהוּת

סֵדֶר הַדִּילּוּג לְפִי הַשְּׁהוּת

הַכְּלָל. אוֹמֵר תְּחִלָּה ״בָּרוּךְ שֶׁאָמַר״, ״תְּהִלָּה לְדָוִד״ וְ״יִשְׁתַּבַּח״ ; וּבִשְׁהוּת יוֹתֵר מוֹסִיף מִזְמוֹרֵי ״הַלְלוּיָהּ״, אַחַר כָּךְ ״וַיְבָרֶךְ דָּוִיד״, אַחַר כָּךְ ״הוֹדוּ לַה׳ קִרְאוּ״.

הַהַרְחָבָה. עִקַּר הַתַּקָּנָה ״תְּהִלָּה לְדָוִד״ (עַיֵּן סימן נ״א) ; וּמַה שֶּׁבֵּינְתַיִם אֵינָם רַק פְּסוּקִים מְלֻקָּטִים.

פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה בְּלֹא בְּרָכָה וּבִרְכוֹת הַשַּׁחַר

פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה וּבִרְכוֹת הַשַּׁחַר

הַכְּלָל. אִם הִתְחִילוּ יוֹצֵר, מִתְפַּלֵּל עֲמִידָה עִם הַצִּבּוּר וְאַחַר כָּךְ קוֹרֵא פְּסוּקֵי דְּזִמְרָה בְּלֹא בְּרָכָה (שֶׁסְּפֵק בְּרָכוֹת לְהָקֵל).

הַהַנְהָגָה. אֵין מְדַלְּגִים לְעוֹלָם אֶת בִּרְכוֹת הַשַּׁחַר ; עַיֵּן סימן מ״ו וְסימן מ״ז.

הַפְנָיָה לַלִּמּוּד. הֶקֵּף הַיְסוֹד כָּאן — הַמּוּטָב שֶׁל תְּפִלָּה בְּצִבּוּר, סֵדֶר הַדִּילּוּג וּפְסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה בְּלֹא בְּרָכָה — נוֹגֵעַ בְּהַנְהָגַת הַמְאַחֵר וּבְרַגִּישׁוּת חב״ד לְעֵרֶךְ הַתְּפִלָּה בְּצִבּוּר. הַדַּף מַצִּיג זֹאת בִּרְמַת הָעִקָּרוֹן הַמְאֻמָּת בְּטֶקְסְט הָרַב, בְּלִי לְהוֹסִיף דָּבָר בָּדוּי. לְלִמּוּד מַעֲמִיק וּלְהַנְהָגָה אִישִׁית — לִלְמֹד עִם רַב שֶׁל חב״ד וּבַטְּקָסְטִים עַצְמָם.

דִּין מִי שֶׁשָּׁהָה לָבוֹא לְבֵית הַכְּנֶסֶת עַד יִשְׁתַּבַּח — בָּעִקָּרוֹן · לַלִּמּוּד, רַב שֶׁל חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרכו של אדמו״ר הזקן עם המחבר, הרמ״א (רמ״א) והמשנה ברורה (משנה ברורה) בנקודות המפתח של סימן נ״ב. שורת שולחן ערוך הרב (שולחן ערוך הרב) מובלטת — היא חושפת את רגישות הרב כפי שהיא עולה מב׳ סעיפיו. הרב מפרש את הטעם ואת הפרט במקום שהמחבר מסתפק באמירה.

נושא מחבר רמ״א משנה ברורה דרך אדמו״ר הזקן (שולחן ערוך הרב)
מוּטָב תְּפִלָּה בְּצִבּוּר וְדִילּוּג פְּסוּקֵי דְּזִמְרָה או״ח נ״ב — מָצָא צִבּוּר בְּסוֹף פְּסוּקֵי דְּזִמְרָה, מְדַלֵּג כְּדֵי לְהִתְפַּלֵּל עִם הַצִּבּוּר. או״ח נ״ב (הגה) — הַשְׁלָמַת הַדִּילּוּג בְּ״הוֹדוּ״ וַ״יְבָרֶךְ דָּוִיד״ [הגהות מיימוני פ״ז מהל׳ תפלה]. מ״ב נ״ב / ברכות ל״ב ע״א, כ״א ע״ב — טַעַם הַדִּילּוּג וּתְפִלָּה בְּצִבּוּר. דרך אדמו״ר הזקן — תְּפִלַּת צִבּוּר רְצוּיָה וּמְקֻבֶּלֶת, שֶׁפְּסוּקֵי דְּזִמְרָה בִּשְׁבִיל הַתְּפִלָּה, וְאִם כֵּן מוּטָב עִם הַצִּבּוּר (סעיף א).
סֵדֶר הַדִּילּוּג לְפִי הַשְּׁהוּת או״ח נ״ב — ב״ש, תְּהִלָּה לְדָוִד, הַלְלוּ אֶת ה׳, הַלְלוּ אֵל בְּקָדְשׁוֹ, יִשְׁתַּבַּח. או״ח נ״ב (הגה) — יֵשׁ שְׁהוּת: ״הוֹדוּ לַה׳ קִרְאוּ״ ; אֵין: ב״ש וּתְהִלָּה לְדָוִד וְיִשְׁתַּבַּח [רא״ש והר״י פ׳ אין עומדין]. מ״ב נ״ב — פְּרָטֵי הַדִּילּוּג וְהַשְּׁהוּת. דרך אדמו״ר הזקן — לְפִי הַשְּׁהוּת: תְּהִלָּה לְדָוִד ; יוֹתֵר — מִזְמוֹרֵי הַלְלוּיָהּ, וַיְבָרֶךְ דָּוִיד, הוֹדוּ ; שֶׁבֵּינְתַיִם פְּסוּקִים מְלֻקָּטִים (סעיף א).
פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה וּבִרְכוֹת הַשַּׁחַר או״ח נ״ב — הִתְחִילוּ יוֹצֵר: ק״ש וּבִרְכוֹתֶיהָ עִם הַצִּבּוּר, וְאַחַר כָּךְ פְּסוּקֵי דְּזִמְרָה בְּלֹא בְּרָכָה. או״ח נ״ב (הגה) — וּמִכָּל מָקוֹם יֹאמַר כָּל בִּרְכוֹת הַבֹּקֶר [כל בו וב״י בשם מהרי״א], סי׳ מ״ו וּמ״ז. מ״ב נ״ב — סְפֵק בְּרָכוֹת לְהָקֵל בְּב״ש וְיִשְׁתַּבַּח. דרך אדמו״ר הזקן — אַחַר הַתְּפִלָּה בְּלֹא בְּרָכָה (סְפֵק בְּרָכוֹת לְהָקֵל), וּמִכָּל מָקוֹם כָּל בִּרְכוֹת הַשַּׁחַר (סעיף א).

הטבלה נערכה מטקסט המחבר (אורח חיים נ״ב, סָעִיף אֶחָד, מקור ספריא) ומנושאי הכלים הניתנים לאימות (בית יוסף, טור, רמב״ם, רא״ש פ׳ אין עומדין, הגהות מיימוני פ״ז מהלכות תפלה, כל בו, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ל״ב ע״א, כ״א ע״ב). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ב׳ סְעִיפִים) המובא ב§1. ייחוסי סעיף קטן שאינם ודאיים נותרו « על דרך הסברה ».

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה — מעבר ליישום הפשוט של הכלל. כל אחת בנויה: עיקרון הסימן → קריאת הרב → תוצאה מעשית. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — מוּטָב תְּפִלָּה בְּצִבּוּר, רְצוּיָה וּמְקֻבֶּלֶת

קְדִימַת הַתְּפִלָּה בְּצִבּוּר

עיקרון הסימן: הַמּוּטָב לְהִתְפַּלֵּל עֲמִידָה עִם הַצִּבּוּר (סעיף א).

קריאת הרב: תְּפִלַּת צִבּוּר לְעוֹלָם רְצוּיָה וּמְקֻבֶּלֶת ; וּפְסוּקֵי דְּזִמְרָה רַק בִּשְׁבִיל הַתְּפִלָּה.

תוצאה מעשית: לְדַלֵּג פְּסוּקֵי דְּזִמְרָה כְּדֵי לְהִתְפַּלֵּל י״ח עִם הַצִּבּוּר. עַיֵּן סימן צ׳.

קו ב — סֵדֶר הַדִּילּוּג לְפִי הַשְּׁהוּת

סֵדֶר הַדִּילּוּג לְפִי הַשְּׁהוּת

עיקרון הסימן: סֵדֶר הַדִּילּוּג הַמַּדְרֵגִי (סעיף א).

קריאת הרב: ״בָּרוּךְ שֶׁאָמַר״, ״תְּהִלָּה לְדָוִד״, ״יִשְׁתַּבַּח״ ; אַחַר כָּךְ מִזְמוֹרֵי ״הַלְלוּיָהּ״, ״וַיְבָרֶךְ דָּוִיד״, ״הוֹדוּ״.

תוצאה מעשית: לְהַתְאִים לַשְּׁהוּת ; הָעִיקָּר ״תְּהִלָּה לְדָוִד״. עַיֵּן סימן נ״א.

קו ג — תַּשְׁלוּמֵי פְּסוּקֵי דְּזִמְרָה וּבִרְכוֹת הַשַּׁחַר

פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה וּבִרְכוֹת הַשַּׁחַר

עיקרון הסימן: פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה וּבִרְכוֹת הַשַּׁחַר (סעיף א).

קריאת הרב: אַחַר הַתְּפִלָּה — פְּסוּקֵי דְּזִמְרָה בְּלֹא בְּרָכָה ; אֲבָל כָּל בִּרְכוֹת הַבֹּקֶר.

תוצאה מעשית: לְעוֹלָם לֹא מְדַלְּגִים בִּרְכוֹת הַשַּׁחַר. עַיֵּן סימן מ״ו וְסימן נ״ג.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן נ״ב ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, בהפניה לרב לפרטי המקרים (הַמּוּטָב, סֵדֶר הַדִּילּוּג, פְּסוּקֵי דְּזִמְרָה אַחַר הַתְּפִלָּה, בִּרְכוֹת הַשַּׁחַר).

לחסיד חב״ד — סימן נ״ב (דִּין מִי שֶׁשָּׁהָה לָבוֹא לְבֵית הַכְּנֶסֶת עַד יִשְׁתַּבַּח)

  • ① מוּטָב עִם הַצִּבּוּר. אִחֵר וּמָצָא צִבּוּר בְּסוֹף פְּסוּקֵי דְּזִמְרָה — מְדַלֵּג כְּדֵי לְהִתְפַּלֵּל י״ח עִם הַצִּבּוּר, שֶׁתְּפִלַּת צִבּוּר רְצוּיָה וּמְקֻבֶּלֶת. יסוד: שוע״ר נ״ב סעיף א-ב.
  • ② סֵדֶר הַדִּילּוּג. לְפִי הַשְּׁהוּת: ״בָּרוּךְ שֶׁאָמַר״, ״תְּהִלָּה לְדָוִד״, ״יִשְׁתַּבַּח״ ; וּבְיוֹתֵר — מִזְמוֹרֵי ״הַלְלוּיָהּ״, ״וַיְבָרֶךְ דָּוִיד״, ״הוֹדוּ״. יסוד: שוע״ר נ״ב סעיף א-ב.
  • ③ אַחַר הַתְּפִלָּה בְּלֹא בְּרָכָה. הִתְחִילוּ יוֹצֵר — מִתְפַּלֵּל עִם הַצִּבּוּר וְאַחַר כָּךְ קוֹרֵא פְּסוּקֵי דְּזִמְרָה בְּלֹא בְּרָכָה (סְפֵק בְּרָכוֹת לְהָקֵל). יסוד: שוע״ר נ״ב סעיף א-ב.
  • ④ בִּרְכוֹת הַשַּׁחַר. לְעוֹלָם לֹא מְדַלְּגִים בִּרְכוֹת הַשַּׁחַר ; וּלְכַתְּחִלָּה לְזָרֵז לָבֹא לְבֵית הַכְּנֶסֶת. יסוד: שוע״ר נ״ב סעיף א-ב.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן נ״ב (הטקסט האמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַדִּילּוּג, הַבְּרָכוֹת, הָאִחוּר — יש להתייעץ עם רב, ולחב״ד עם רב של חב״ד.
