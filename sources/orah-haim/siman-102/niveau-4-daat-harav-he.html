דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״ב · שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל

דרכו של אדמו״ר הזקן בדִּין שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל : הַרְחָקַת ד׳ אַמּוֹת — אַדְמַת קֹדֶשׁ ; הֶתֵּר הָעוֹסֵק בַּתּוֹרָה וְהַחַלָּשׁ ; הַיּוֹשֵׁב כְּבָר — בָּא בִּגְבוּלוֹ ; אִסּוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלֵּל וְלִפְסֹעַ ג׳ פְּסִיעוֹת — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״ב — שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל. אצל הרב הסימן נפרש לה׳ סְעִיפִים : הַרְחָקַת ד׳ אַמּוֹת וְאַדְמַת קֹדֶשׁ, הֶתֵּר הָעוֹסֵק בַּתּוֹרָה, הֶתֵּר הַחַלָּשׁ, הַיּוֹשֵׁב כְּבָר (בָּא בִּגְבוּלוֹ), וְאִסּוּר לַעֲבֹר וְלִפְסֹעַ ג׳ פְּסִיעוֹת.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״ב — ה׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״ב). אדמו״ר הזקן דן בו בְּדִין שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״ב — שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״ב — שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל — וּבוֹ ה׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.102), ואחריו תרגומנו. הסימן כולל ה׳ סעיפים (סעיף k → ‎.102.k).

סעיף א אָסוּר לֵישֵׁב בְּתוֹךְ ד׳ אַמּוֹת שֶׁל הַמִּתְפַּלֵּל ; עוֹסֵק בַּתּוֹרָה — מֻתָּר.

שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל וּבוֹ ה׳ סְעִיפִים:אָסוּר לֵישֵׁב בְּתוֹךְ ד׳ אַמּוֹת שֶׁל הַמִּתְפַּלֵּל, בֵּין מִלְּפָנָיו, בֵּין מִלְּאַחֲרָיו, בֵּין מִן הַצְּדָדִים — צָרִיךְ לְהַרְחִיק ד׳ אַמּוֹת, כִּי הַמָּקוֹם אֲשֶׁר הַמִּתְפַּלֵּל עוֹמֵד עָלָיו אַדְמַת קֹדֶשׁ הוּא, הוֹאִיל וּשְׁכִינָה כְּנֶגְדּוֹ כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן צ״ז, וְצָרִיךְ לִנְהֹג כָּבוֹד בְּמָקוֹם זֶה כָּל זְמַן שֶׁהוּא מִתְפַּלֵּל, וּמְקוֹמוֹ שֶׁל אָדָם הוּא ד׳ אַמּוֹת. וְאִם אֵינוֹ רוֹצֶה לִהְיוֹת יוֹשֵׁב וּבָטֵל שָׁם אֶלָּא עָסוּק בַּתּוֹרָה אוֹ בְּשִׁירוֹת וְתִשְׁבָּחוֹת — מֻתָּר לֵישֵׁב, שֶׁכְּשֶׁעוֹסֵק שָׁם בְּדִבְרֵי קְדֻשָּׁה — אֵין גְּנַאי בִּישִׁיבָתוֹ שָׁם. וְהוּא שֶׁמּוֹצִיא דִּבְרֵי תּוֹרָה בְּפִיו, אֲבָל הִרְהוּר בְּדִבְרֵי תּוֹרָה — אֵינוֹ מוֹעִיל לָזֶה, שֶׁהִרְהוּר אֵינוֹ נִרְאֶה וִישִׁיבָתוֹ נִרְאֵית גְּנַאי לְמָקוֹם קָדוֹשׁ:

הַרְחָקַת ד׳ אַמּוֹת. אָסוּר לֵישֵׁב בְּתוֹךְ ד׳ אַמּוֹת שֶׁל הַמִּתְפַּלֵּל (מִלְּפָנָיו, מִלְּאַחֲרָיו וּמִן הַצְּדָדִים), כִּי הַמָּקוֹם אַדְמַת קֹדֶשׁ וּשְׁכִינָה כְּנֶגְדּוֹ (סימן צ״ז), וּמְקוֹמוֹ שֶׁל אָדָם ד׳ אַמּוֹת. אֲבָל עָסוּק בַּתּוֹרָה אוֹ בְּשִׁירוֹת וְתִשְׁבָּחוֹת — מֻתָּר, וְהוּא שֶׁמּוֹצִיא בְּפִיו ; הִרְהוּר אֵינוֹ מוֹעִיל.

סעיף ב הַיּוֹשֵׁב חַלָּשׁ — מֻתָּר לוֹ לֵישֵׁב בְּצַד הַמִּתְפַּלֵּל.

יֵשׁ מִי שֶׁאוֹמֵר שֶׁאִם הַיּוֹשֵׁב בְּצַד הַמִּתְפַּלֵּל הוּא חַלָּשׁ — מֻתָּר לוֹ לֵישֵׁב, שֶׁאֵין יְשִׁיבָתוֹ נִרְאֵית גְּנַאי, לְפִי שֶׁחֻלְשָׁתוֹ מוֹכַחַת עָלָיו שֶׁמִּפְּנֵי כָּךְ יוֹשֵׁב, וְיֵשׁ לִסְמֹךְ עַל דְּבָרָיו אִם אִי אֶפְשָׁר לוֹ בְּקַל לְהַרְחִיק ד׳ אַמּוֹת וְלֵישֵׁב שָׁם:

הֶתֵּר הַחַלָּשׁ. יֵשׁ מִי שֶׁאוֹמֵר : הַיּוֹשֵׁב בְּצַד הַמִּתְפַּלֵּל שֶׁהוּא חַלָּשׁ — מֻתָּר לוֹ לֵישֵׁב, שֶׁחֻלְשָׁתוֹ מוֹכַחַת וְאֵין יְשִׁיבָתוֹ גְּנַאי ; וְיֵשׁ לִסְמֹךְ אִם אִי אֶפְשָׁר בְּקַל לְהַרְחִיק ד׳ אַמּוֹת.

סעיף ג הַיּוֹשֵׁב כְּבָר — אֵין צָרִיךְ לָקוּם ; בְּבֵית הַכְּנֶסֶת — צָרִיךְ.

אִם הַיּוֹשֵׁב יָשַׁב כְּבָר וְעָמַד זֶה בְּצִדּוֹ לְהִתְפַּלֵּל — אֵין צָרִיךְ לָקוּם, שֶׁהֲרֵי זֶה בָּא בִּגְבוּלוֹ. וְהוּא שֶׁמִּתְפַּלֵּל בְּבַיִת שֶׁלֹּא יִחֲדוּהוּ לִתְפִלָּה, אֲבָל בְּבֵית הַכְּנֶסֶת שֶׁיִּחֲדוּהוּ לִתְפִלָּה — צָרִיךְ לָקוּם, מִפְּנֵי שֶׁהוּא מָקוֹם הַמְיֻחָד לַכֹּל, וְלֹא שַׁיָּךְ לוֹמַר זֶה בָּא בִּגְבוּלוֹ. וַאֲפִלּוּ בְּבֵיתוֹ — מִדַּת חֲסִידוּת הוּא לָקוּם, כְּדֵי לְהָרִים מִכְשׁוֹל מֵחֲבֵרוֹ שֶׁעוֹשֶׂה שֶׁלֹּא כַּשּׁוּרָה שֶׁעָמַד לְהִתְפַּלֵּל בְּתוֹךְ ד׳ אַמּוֹת שֶׁל הַיּוֹשֵׁב כְּבָר בְּהֶתֵּר וְאֵינוֹ מְחֻיָּב לָקוּם מִפָּנָיו, וִישִׁיבָתוֹ הִיא גְּנַאי לְמָקוֹם קָדוֹשׁ:

בָּא בִּגְבוּלוֹ. הַיּוֹשֵׁב כְּבָר וְעָמַד זֶה בְּצִדּוֹ — אֵין צָרִיךְ לָקוּם, שֶׁזֶּה בָּא בִּגְבוּלוֹ, וְהוּא בְּבַיִת שֶׁלֹּא יִחֲדוּהוּ ; אֲבָל בְּבֵית הַכְּנֶסֶת הַמְיֻחָד לַכֹּל — צָרִיךְ לָקוּם. וַאֲפִלּוּ בְּבֵיתוֹ — מִדַּת חֲסִידוּת לָקוּם לְהָרִים מִכְשׁוֹל מֵחֲבֵרוֹ.

סעיף ד אָסוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלֵּל ; בְּצִדֵּיהֶם — מֻתָּר.

אָסוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלְלִין בְּתוֹךְ ד׳ אַמּוֹת שֶׁלִּפְנֵיהֶם, שֶׁמָּא תִּתְבַּטֵּל כַּוָּנָתָם. אֲבָל בְּצִדֵּיהֶם מֻתָּר לַעֲבֹר וְלַעֲמֹד שָׁם, וְלֹא יֵלֵךְ לְהָלְאָה שֶׁלֹּא יִהְיֶה כְּנֶגֶד פְּנֵיהֶם, שֶׁכָּל שֶׁיּוּכַל לִרְאוֹתוֹ — אָסוּר, מִפְּנֵי שֶׁמִּתְבַּטֶּלֶת כַּוָּנָתוֹ. וְחַכְמֵי הַזֹּהַר אוֹסְרִים אֲפִלּוּ בְּצִדֵּיהֶם תּוֹךְ ד׳ אַמּוֹת:

אִסּוּר לַעֲבֹר. אָסוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלְלִין בְּתוֹךְ ד׳ אַמּוֹת שֶׁלִּפְנֵיהֶם, שֶׁמָּא תִּתְבַּטֵּל כַּוָּנָתָם ; בְּצִדֵּיהֶם מֻתָּר, וְלֹא יֵלֵךְ לְהָלְאָה כְּנֶגֶד פְּנֵיהֶם. וְחַכְמֵי הַזֹּהַר אוֹסְרִים אֲפִלּוּ בְּצִדֵּיהֶם תּוֹךְ ד׳ אַמּוֹת.

סעיף ה הִשְׁלִים תְּפִלָּתוֹ וְאֶחָד מִתְפַּלֵּל אַחֲרָיו — אָסוּר לִפְסֹעַ ג׳.

אִם הִשְׁלִים תְּפִלָּתוֹ וְהָיָה אָדָם אֶחָד מִתְפַּלֵּל אַחֲרָיו — אָסוּר לִפְסֹעַ ג׳ פְּסִיעוֹת עַד שֶׁיִּגְמֹר מִי שֶׁאַחֲרָיו אֶת תְּפִלָּתוֹ, שֶׁאִם יִפְסַע קֹדֶם לָכֵן — הֲרֵי הוּא עוֹבֵר כְּנֶגֶד הַמִּתְפַּלֵּל. וְצָרִיךְ לְדַקְדֵּק בָּזֶה אֲפִלּוּ אִם הָאַחֲרוֹן הִתְחִיל לְהִתְפַּלֵּל אַחֲרָיו, מֵאַחַר שֶׁכְּבָר הִתְחִיל:

אִסּוּר ג׳ פְּסִיעוֹת. הִשְׁלִים תְּפִלָּתוֹ וְאֶחָד מִתְפַּלֵּל אַחֲרָיו — אָסוּר לִפְסֹעַ ג׳ פְּסִיעוֹת עַד שֶׁיִּגְמֹר, שֶׁאִם יִפְסַע הֲרֵי הוּא עוֹבֵר כְּנֶגֶד הַמִּתְפַּלֵּל ; וְצָרִיךְ לְדַקְדֵּק אֲפִלּוּ הִתְחִיל הָאַחֲרוֹן אַחֲרָיו.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק״ב (מהדורת קה״ת, ה׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל

היסוד : שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב חמישה צירים. תחילה הַרְחָקַת ד׳ אַמּוֹת וְאַדְמַת קֹדֶשׁ. אחר כך הֶתֵּר הָעוֹסֵק בַּתּוֹרָה, וְהֶתֵּר הַחַלָּשׁ. אחר כך הַיּוֹשֵׁב כְּבָר (בָּא בִּגְבוּלוֹ). ולבסוף אִסּוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלֵּל וְאִסּוּר ג׳ פְּסִיעוֹת.

הַרְחָקַת ד׳ אַמּוֹת — אַדְמַת קֹדֶשׁ

הַרְחָקַת ד׳ אַמּוֹת — אַדְמַת קֹדֶשׁ

היסוד. אָסוּר לֵישֵׁב בְּתוֹךְ ד׳ אַמּוֹת שֶׁל הַמִּתְפַּלֵּל (מִלְּפָנָיו, מִלְּאַחֲרָיו, מִן הַצְּדָדִים), שֶׁהַשְּׁכִינָה כְּנֶגְדּוֹ וְהַמָּקוֹם אַדְמַת קֹדֶשׁ (סעיף א).

המסקנה. עֵסֶק בַּתּוֹרָה אוֹ בְּשִׁירוֹת וְתִשְׁבָּחוֹת מַתִּיר — וְהוּא שֶׁמּוֹצִיא בְּפִיו. עיין סימן 97 (קדושת המקום).

הֶתֵּר הַיּוֹשֵׁב הַחַלָּשׁ

הֶתֵּר הַיּוֹשֵׁב הַחַלָּשׁ

היסוד. הַחַלָּשׁ מֻתָּר לוֹ לֵישֵׁב בְּצַד הַמִּתְפַּלֵּל, שֶׁחֻלְשָׁתוֹ מוֹכַחַת (סעיף ב).

המסקנה. סוֹמְכִים כְּשֶׁאִי אֶפְשָׁר בְּקַל לְהַרְחִיק ד׳ אַמּוֹת. עיין סימן 98 (כוונה בתפילה).

הַיּוֹשֵׁב כְּבָר — בָּא בִּגְבוּלוֹ

הַיּוֹשֵׁב כְּבָר — בָּא בִּגְבוּלוֹ

היסוד. הַיּוֹשֵׁב כְּבָר אֵין צָרִיךְ לָקוּם, שֶׁזֶּה בָּא בִּגְבוּלוֹ — בְּבַיִת שֶׁלֹּא יִחֲדוּהוּ (סעיף ג).

המסקנה. בְּבֵית הַכְּנֶסֶת צָרִיךְ לָקוּם, וּבְבֵיתוֹ מִדַּת חֲסִידוּת. עיין סימן 90 (בית הכנסת).

אִסּוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלֵּל

אִסּוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלֵּל

היסוד. אָסוּר לַעֲבֹר בְּתוֹךְ ד׳ אַמּוֹת שֶׁלִּפְנֵי הַמִּתְפַּלֵּל, שֶׁמָּא תִּתְבַּטֵּל כַּוָּנָתָם ; בְּצִדֵּיהֶם מֻתָּר (סעיף ד).

המסקנה. חַכְמֵי הַזֹּהַר אוֹסְרִים אֲפִלּוּ בַּצְּדָדִים תּוֹךְ ד׳ אַמּוֹת. עיין סימן 101 (הפסק בתפילה).

אִסּוּר ג׳ פְּסִיעוֹת

אִסּוּר ג׳ פְּסִיעוֹת

היסוד. הִשְׁלִים תְּפִלָּתוֹ וְאֶחָד מִתְפַּלֵּל אַחֲרָיו — אָסוּר לִפְסֹעַ ג׳ פְּסִיעוֹת עַד שֶׁיִּגְמֹר (סעיף ה).

המסקנה. הַפּוֹסֵעַ קֹדֶם הֲרֵי הוּא כְּעוֹבֵר כְּנֶגֶד הַמִּתְפַּלֵּל — אֲפִלּוּ הִתְחִיל אַחֲרָיו. עיין סימן 100 (סדר התפילה).

הפניה ללימוד. היקף הסימן — הַהַרְחָקָה, הַהֶתֵּרִים וְאִסּוּר הַמַּעֲבָר וְהַפְּסִיעוֹת — נוגע להנהגת בית הכנסת היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק״ב. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מה׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הַרְחָקַת ד׳ אַמּוֹת וְהֶתֵּר הָעוֹסֵק או״ח ק״ב — אָסוּר לֵישֵׁב תּוֹךְ ד׳ אַמּוֹת ; עוֹסֵק בְּתִקּוּנֵי תְּפִלּוֹת אֵין צָרִיךְ לְהַרְחִיק ; יֵשׁ מַתִּיר בְּעוֹסֵק בַּתּוֹרָה. מ״ב ק״ב — פְּרָטֵי הַהַרְחָקָה וּגְדַר הָעוֹסֵק בְּדִבְרֵי תּוֹרָה. ערוה״ש ק״ב — טַעַם הַהַרְחָקָה וְדִין הָעוֹסֵק בַּתּוֹרָה כְּנֶגְדּוֹ. דרך אדמו״ר הזקן — הַטַּעַם : אַדְמַת קֹדֶשׁ וּשְׁכִינָה כְּנֶגְדּוֹ (סימן צ״ז) ; הֶתֵּר עוֹסֵק בַּתּוֹרָה — וְהוּא שֶׁמּוֹצִיא בְּפִיו, הִרְהוּר אֵינוֹ מוֹעִיל (סעיף א).
הֶתֵּר הַיּוֹשֵׁב הַחַלָּשׁ או״ח ק״ב — אִם הַיּוֹשֵׁב בְּצַד הַמִּתְפַּלֵּל חַלָּשׁ — מֻתָּר. מ״ב ק״ב — שִׁעוּר הַחֻלְשָׁה וְהַסְּמִיכָה עַל הַהֶתֵּר. ערוה״ש ק״ב — גֶּדֶר הַחַלָּשׁ בַּיְשִׁיבָה. דרך אדמו״ר הזקן — חֻלְשָׁתוֹ מוֹכַחַת וְאֵין יְשִׁיבָתוֹ גְּנַאי, וְיֵשׁ לִסְמֹךְ אִם אִי אֶפְשָׁר בְּקַל לְהַרְחִיק ד׳ אַמּוֹת (סעיף ב).
הַיּוֹשֵׁב כְּבָר — בָּא בִּגְבוּלוֹ או״ח ק״ב — יָשַׁב כְּבָר וְעָמַד זֶה בְּצִדּוֹ — אֵין צָרִיךְ לָקוּם (מִדַּת חֲסִידוּת לָקוּם). מ״ב ק״ב — פְּרָטֵי בָּא בִּגְבוּלוֹ וְדִין בֵּית הַכְּנֶסֶת. ערוה״ש ק״ב — הַהֶבְדֵּל בֵּין בַּיִת לְבֵית הַכְּנֶסֶת. דרך אדמו״ר הזקן — דַּוְקָא בְּבַיִת ; בְּבֵית הַכְּנֶסֶת (מָקוֹם הַמְיֻחָד לַכֹּל) צָרִיךְ לָקוּם ; וּבְבֵיתוֹ מִדַּת חֲסִידוּת לְהָרִים מִכְשׁוֹל (סעיף ג).
אִסּוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלֵּל או״ח ק״ב — אָסוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלְלִין תּוֹךְ ד׳ אַמּוֹת ; בְּצִדֵּיהֶם מֻתָּר. מ״ב ק״ב — גֶּדֶר הַמַּעֲבָר וְהַהֶתֵּר בַּצְּדָדִים. ערוה״ש ק״ב — דִּין הָעוֹבֵר לִפְנֵי הַמִּתְפַּלֵּל. דרך אדמו״ר הזקן — שֶׁלִּפְנֵיהֶם שֶׁמָּא תִּתְבַּטֵּל כַּוָּנָתָם ; וְחַכְמֵי הַזֹּהַר אוֹסְרִים אֲפִלּוּ בְּצִדֵּיהֶם תּוֹךְ ד׳ אַמּוֹת (סעיף ד).
אִסּוּר ג׳ פְּסִיעוֹת או״ח ק״ב — הִשְׁלִים וְאֶחָד מִתְפַּלֵּל אַחֲרָיו — אָסוּר לִפְסֹעַ ג׳ עַד שֶׁיִּגְמֹר. מ״ב ק״ב — פְּרָטֵי הַפְּסִיעוֹת וְהַמַּמְתִּין לַאֲחוֹרָיו. ערוה״ש ק״ב — דִּין הַפּוֹסֵעַ כְּנֶגֶד הַמִּתְפַּלֵּל אַחֲרָיו. דרך אדמו״ר הזקן — אָסוּר לִפְסֹעַ ג׳ עַד שֶׁיִּגְמֹר הָאַחֲרוֹן ; וְצָרִיךְ לְדַקְדֵּק אֲפִלּוּ הִתְחִיל אַחֲרָיו (סעיף ה).

הטבלה נערכה מטקסט המחבר (אורח חיים ק״ב, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה, רא״ש ותוספות ברכות פרק אין עומדין, מרדכי, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ז, זוהר). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ה׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — אַדְמַת קֹדֶשׁ וְהַרְחָקָה

קו א — אַדְמַת קֹדֶשׁ וְהַרְחָקָה

יסוד הסימן : הַרְחָקַת ד׳ אַמּוֹת מִן הַמִּתְפַּלֵּל (סעיף א).

קריאת הרב : נוֹתֵן אֶת הַטַּעַם — אַדְמַת קֹדֶשׁ וּשְׁכִינָה כְּנֶגְדּוֹ (סימן צ״ז) — וּמַצְרִיךְ לְהוֹצִיא בְּפִיו, שֶׁהַהִרְהוּר אֵינוֹ מוֹעִיל. מסקנה : לְהַרְחִיק ד׳ אַמּוֹת, פְּרָט לְעֵסֶק בְּקוֹל בַּתּוֹרָה.

קו ב — הֶתֵּרֵי הַיְשִׁיבָה

קו ב — הֶתֵּרֵי הַיְשִׁיבָה

יסוד הסימן : הֶתֵּר הָעוֹסֵק בַּתּוֹרָה, הֶתֵּר הַחַלָּשׁ, וְהַיּוֹשֵׁב כְּבָר בָּא בִּגְבוּלוֹ (סעיפים א-ג).

קריאת הרב : מַבְחִין בֵּין בַּיִת לְבֵית הַכְּנֶסֶת — בְּבֵית הַכְּנֶסֶת, מָקוֹם הַמְיֻחָד לַכֹּל, צָרִיךְ לָקוּם. מסקנה : מִדַּת חֲסִידוּת לָקוּם אֲפִלּוּ בְּבֵיתוֹ, לְהָרִים מִכְשׁוֹל. עיין סימן 90.

קו ג — אִסּוּר הַמַּעֲבָר וְהַפְּסִיעוֹת

קו ג — אִסּוּר הַמַּעֲבָר וְהַפְּסִיעוֹת

יסוד הסימן : אָסוּר לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלֵּל וְלִפְסֹעַ ג׳ פְּסִיעוֹת (סעיפים ד-ה).

קריאת הרב : הַטַּעַם שֶׁמָּא תִּתְבַּטֵּל כַּוָּנָתָם, וְחַכְמֵי הַזֹּהַר מַחְמִירִים אֲפִלּוּ בַּצְּדָדִים. מסקנה : לֹא לִפְסֹעַ כָּל זְמַן שֶׁהַמִּתְפַּלֵּל אַחֲרָיו. עיין סימן 100.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״ב ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״ב (שֶׁלֹּא לְהַפְסִיק כְּנֶגֶד הַמִּתְפַּלֵּל)

  • ① הַרְחָקָה. לֹא לֵישֵׁב בְּתוֹךְ ד׳ אַמּוֹת שֶׁל הַמִּתְפַּלֵּל (מִלְּפָנָיו, מִלְּאַחֲרָיו, מִן הַצְּדָדִים) — אַדְמַת קֹדֶשׁ. יסוד : שוע״ר ק״ב סעיף א.
  • ② הֶתֵּר עוֹסֵק. עָסוּק בַּתּוֹרָה אוֹ בְּשִׁירוֹת וְתִשְׁבָּחוֹת בְּקוֹל — מֻתָּר לֵישֵׁב ; הִרְהוּר אֵינוֹ מוֹעִיל. יסוד : שוע״ר ק״ב סעיף א.
  • ③ חַלָּשׁ. חַלָּשׁ מֻתָּר לוֹ לֵישֵׁב בְּצַד הַמִּתְפַּלֵּל, שֶׁחֻלְשָׁתוֹ מוֹכַחַת. יסוד : שוע״ר ק״ב סעיף ב.
  • ④ בָּא בִּגְבוּלוֹ. הַיּוֹשֵׁב אֵין צָרִיךְ לָקוּם בְּבַיִת (בָּא בִּגְבוּלוֹ) ; בְּבֵית הַכְּנֶסֶת — צָרִיךְ לָקוּם ; וּבְבֵיתוֹ מִדַּת חֲסִידוּת. יסוד : שוע״ר ק״ב סעיף ג.
  • ⑤ מַעֲבָר וּפְסִיעוֹת. לֹא לַעֲבֹר כְּנֶגֶד הַמִּתְפַּלֵּל (ד׳ אַמּוֹת) וְלֹא לִפְסֹעַ ג׳ פְּסִיעוֹת כָּל זְמַן שֶׁמִּתְפַּלֵּל אַחֲרָיו. יסוד : שוע״ר ק״ב סעיפים ד-ה.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק״ב (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַהַרְחָקָה, הַבָּא בִּגְבוּלוֹ, אִסּוּר הַמַּעֲבָר וְהַפְּסִיעוֹת — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
