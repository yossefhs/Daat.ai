דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״כ · שֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה

דרכו של אדמו״ר הזקן בשֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה : אוֹמְרִים רְצֵה בְּכָל הַתְּפִלּוֹת — וְלֹא כְּאוֹתָן שֶׁמַּתְחִילִים בְּמִנְחָה וּבְעַרְבִית ״וְאִשֵּׁי יִשְׂרָאֵל״, שֶׁאֵין טַעַם לְמִנְהָגָם ; וּבֵאוּר ״וְאִשֵּׁי יִשְׂרָאֵל״ — נִשְׁמוֹת הַצַּדִּיקִים, עֲבוֹדָה שֶׁבַּלֵּב, אֵשׁ הַהִתְלַהֲבוּת — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״כ — שֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה. אצל הרב הסימן נפרש לב׳ סְעִיפִים : תחילה דִּין אֲמִירַת רְצֵה בְּכָל הַתְּפִלּוֹת — וְלֹא כְּמִנְהַג הַמַּתְחִילִים בְּמִנְחָה וּבְעַרְבִית ״וְאִשֵּׁי יִשְׂרָאֵל וּתְפִלָּתָם״ — ואחר כך בֵּאוּר ״וְאִשֵּׁי יִשְׂרָאֵל״ : נִשְׁמוֹת הַצַּדִּיקִים שֶׁמַּקְרִיב מִיכָאֵל, הַתְּפִלָּה הַנִּקְרָא עֲבוֹדָה שֶׁבַּלֵּב, וְאֵשׁ הַהִתְלַהֲבוּת.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״כ — ב׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״כ). אדמו״ר הזקן דן בו בְּשֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״כ — שֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״כ — שֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה — וּבוֹ ב׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.120), ואחריו תרגומנו. הסימן כולל ב׳ סעיפים (סעיף k → ‎.120.k).

סעיף א אוֹמְרִים ״רְצֵה״ בְּכָל הַתְּפִלּוֹת, וְלֹא כְּאוֹתָן שֶׁמַּתְחִילִים בְּמִנְחָה וּבְעַרְבִית ״וְאִשֵּׁי יִשְׂרָאֵל״.

שֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה וּבוֹ ב׳ סְעִיפִים:אוֹמְרִים ״רְצֵה״ בְּכָל הַתְּפִלּוֹת, וְלֹא כְּאוֹתָן שֶׁנּוֹהֲגִים לְהַתְחִיל בְּמִנְחָה וּבְעַרְבִית ״וְאִשֵּׁי יִשְׂרָאֵל וּתְפִלָּתָם כו׳״, שֶׁאֵין טַעַם לְמִנְהָגָם:

אוֹמְרִים ״רְצֵה״ בְּכָל הַתְּפִלּוֹת. אוֹמְרִים רְצֵה בְּכָל הַתְּפִלּוֹת (שַׁחֲרִית, מִנְחָה, עַרְבִית), וְלֹא כְּאוֹתָן שֶׁנּוֹהֲגִים לְהַתְחִיל בְּמִנְחָה וּבְעַרְבִית ״וְאִשֵּׁי יִשְׂרָאֵל וּתְפִלָּתָם וכו׳״ : שֶׁאֵין טַעַם לְמִנְהָגָם.

סעיף ב ״וְאִשֵּׁי יִשְׂרָאֵל״ — נִשְׁמוֹת הַצַּדִּיקִים ; ״וּתְפִלָּתָם״ — גּוּף הַתְּפִלָּה שֶׁהֵן תֵּבוֹתֶיהָ.

״וְאִשֵּׁי יִשְׂרָאֵל״ הֵן נִשְׁמוֹת הַצַּדִּיקִים שֶׁמַּקְרִיב מִיכָאֵל שַׂר הַגָּדוֹל עַל הַמִּזְבֵּחַ שֶׁל מַעְלָה (וּלְמַטָּה — הֵן הַקָּרְבָּנוֹת שֶׁמַּקְרִיבִים הַצַּדִּיקִים וּמוֹסְרִים נַפְשׁוֹתֵיהֶם לְהַקָּדוֹשׁ בָּרוּךְ הוּא עַל כָּל דִּבּוּר וְדִבּוּר הַיּוֹצֵא מִפִּיהֶם הַמְכֻנֶּה בְּשֵׁם ״מִזְבֵּחַ״) (בְּאַהֲבָה רַבָּה הַבָּאָה אֲלֵיהֶם מִלְמַעְלָה עַל יְדֵי מִיכָאֵל שַׂר הַחֶסֶד), וְזוֹ הִיא תְּפִלָּה הַנִּקְרָא ״עֲבוֹדָה שֶׁבַּלֵּב״. וְהַקָּרְבָּנוֹת אֵלּוּ שֶׁהֵן בְּאֵשׁ הַהִתְלַהֲבוּת — נִקְרָאוֹת ״אִשֵּׁי יִשְׂרָאֵל״, ״וּתְפִלָּתָם״ — הִיא גּוּף הַתְּפִלָּה שֶׁהֵן תֵּבוֹתֶיהָ:

בֵּאוּר ״וְאִשֵּׁי יִשְׂרָאֵל״. הַרַב מְבָאֵר אֶת עִנְיַן הַפָּסוּק. ״וְאִשֵּׁי יִשְׂרָאֵל״ — הֵן נִשְׁמוֹת הַצַּדִּיקִים שֶׁמַּקְרִיב מִיכָאֵל שַׂר הַגָּדוֹל עַל הַמִּזְבֵּחַ שֶׁל מַעְלָה (וּלְמַטָּה — הַקָּרְבָּנוֹת שֶׁהַצַּדִּיקִים מוֹסְרִים נַפְשָׁם עַל כָּל דִּבּוּר הַיּוֹצֵא מִפִּיהֶם, הַמְכֻנֶּה ״מִזְבֵּחַ״ ; בְּאַהֲבָה רַבָּה הַבָּאָה מִלְמַעְלָה עַל יְדֵי מִיכָאֵל שַׂר הַחֶסֶד). וְזוֹ הַתְּפִלָּה הַנִּקְרָא ״עֲבוֹדָה שֶׁבַּלֵּב״. וְהַקָּרְבָּנוֹת שֶׁבְּאֵשׁ הַהִתְלַהֲבוּת נִקְרָאוֹת ״אִשֵּׁי יִשְׂרָאֵל״ ; ״וּתְפִלָּתָם״ — הִיא גּוּף הַתְּפִלָּה שֶׁהֵן תֵּבוֹתֶיהָ.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק״כ (מהדורת קה״ת, ב׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — שֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה

היסוד : שֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שני צירים. תחילה הַדִּין הַמַּעֲשִׂי : אֲמִירַת ״רְצֵה״ בְּכָל הַתְּפִלּוֹת, וְלֹא לְהַחְלִיפָהּ בְּמִנְחָה וּבְעַרְבִית בְּ״וְאִשֵּׁי יִשְׂרָאֵל״. אחר כך הַבֵּאוּר שֶׁל הַפָּסוּק ״וְאִשֵּׁי יִשְׂרָאֵל״ : נִשְׁמוֹת הַצַּדִּיקִים, עֲבוֹדָה שֶׁבַּלֵּב וְאֵשׁ הַהִתְלַהֲבוּת.

אֲמִירַת ״רְצֵה״ בְּכָל הַתְּפִלּוֹת

אֲמִירַת ״רְצֵה״ בְּכָל הַתְּפִלּוֹת

היסוד. אוֹמְרִים ״רְצֵה״ בְּכָל הַתְּפִלּוֹת ; מִנְהַג הַמַּתְחִילִים בְּמִנְחָה וּבְעַרְבִית ״וְאִשֵּׁי יִשְׂרָאֵל וּתְפִלָּתָם״ אֵין לוֹ מָקוֹם — שֶׁאֵין טַעַם לְמִנְהָגָם (סעיף א).

המסקנה. בִּרְכַּת עֲבוֹדָה (רְצֵה) נֶאֱמֶרֶת בְּאֹפֶן אֶחָד בְּשָׁלֹשׁ הַתְּפִלּוֹת. עיין סימן קי״ט · 119 (סֵדֶר וְשִׁנּוּיֵי הַבְּרָכוֹת).

בֵּאוּר ״וְאִשֵּׁי יִשְׂרָאֵל״ — עֲבוֹדָה שֶׁבַּלֵּב

בֵּאוּר ״וְאִשֵּׁי יִשְׂרָאֵל״ — עֲבוֹדָה שֶׁבַּלֵּב

היסוד. ״וְאִשֵּׁי יִשְׂרָאֵל״ — נִשְׁמוֹת הַצַּדִּיקִים שֶׁמַּקְרִיב מִיכָאֵל עַל הַמִּזְבֵּחַ שֶׁל מַעְלָה ; זוֹ הַתְּפִלָּה הַנִּקְרָא ״עֲבוֹדָה שֶׁבַּלֵּב״, בְּאֵשׁ הַהִתְלַהֲבוּת (סעיף ב).

המסקנה. ״וּתְפִלָּתָם״ — גּוּף הַתְּפִלָּה שֶׁהֵן תֵּבוֹתֶיהָ : אֵשׁ הַהִתְלַהֲבוּת וְהַתֵּבוֹת הֵן שְׁנֵי פָּנִים שֶׁל הָעֲבוֹדָה. עיין סימן צ״ח · 98 (כַּוָּנַת הַתְּפִלָּה).

הפניה ללימוד. היקף הסימן — אֲחִידוּת אֲמִירַת רְצֵה וְהַמַּשְׁמָעוּת הַפְּנִימִית שֶׁל ״וְאִשֵּׁי יִשְׂרָאֵל״ כַּעֲבוֹדָה שֶׁבַּלֵּב — נוגע להנהגת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

שֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק״כ. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מב׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
אֲמִירַת רְצֵה בְּכָל תְּפִלָּה או״ח ק״כ — אוֹמְרִים רְצֵה בְּכָל הַתְּפִלּוֹת, וְדַלָא כְּאוֹתָם שֶׁנּוֹהֲגִים שֶׁלֹּא לְאוֹמְרוֹ בְּמִנְחָה. מ״ב ק״כ — טַעַם אֲמִירַת רְצֵה בְּכָל הַתְּפִלּוֹת וְדִין הַנֻּסְחָאוֹת. ערוה״ש ק״כ — בֵּרוּר הַמִּנְהָג בַּאֲמִירַת רְצֵה בְּמִנְחָה וּבְעַרְבִית. דרך אדמו״ר הזקן — אוֹמְרִים ״רְצֵה״ בְּכָל הַתְּפִלּוֹת, וְלֹא כְּאוֹתָן שֶׁמַּתְחִילִים בְּמִנְחָה וּבְעַרְבִית ״וְאִשֵּׁי יִשְׂרָאֵל״, שֶׁאֵין טַעַם לְמִנְהָגָם (סעיף א).
בֵּאוּר ״וְאִשֵּׁי יִשְׂרָאֵל״ או״ח ק״כ — לֹא בֵּאֵר הַמְחַבֵּר עִנְיַן ״וְאִשֵּׁי יִשְׂרָאֵל״, רַק דִּין אֲמִירַת רְצֵה. מ״ב ק״כ — הֶאָרוֹת עַל הַנֻּסָּח וְעַל מַשְׁמָעוּת ״אִשֵּׁי יִשְׂרָאֵל״. ערוה״ש ק״כ — פֵּרוּשׁ ״אִשֵּׁי יִשְׂרָאֵל״ עַל דֶּרֶךְ הַפְּשָׁט. דרך אדמו״ר הזקן — נִשְׁמוֹת הַצַּדִּיקִים שֶׁמַּקְרִיב מִיכָאֵל עַל הַמִּזְבֵּחַ שֶׁל מַעְלָה ; תְּפִלָּה הַנִּקְרָא ״עֲבוֹדָה שֶׁבַּלֵּב״, בְּאֵשׁ הַהִתְלַהֲבוּת ; ״וּתְפִלָּתָם״ — גּוּף הַתְּפִלָּה שֶׁהֵן תֵּבוֹתֶיהָ (סעיף ב).

הטבלה נערכה מטקסט המחבר (אורח חיים ק״כ, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ב׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שתי מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — רְצֵה בְּכָל תְּפִלָּה

קו א — רְצֵה בְּכָל תְּפִלָּה

יסוד הסימן : אוֹמְרִים ״רְצֵה״ בְּכָל הַתְּפִלּוֹת ; שֶׁאֵין טַעַם לְמִנְהָגָם לְהַתְחִיל בְּ״וְאִשֵּׁי יִשְׂרָאֵל״ בְּמִנְחָה וּבְעַרְבִית (סעיף א).

קריאת הרב : אֲחִידוּת הַנֻּסָּח : בִּרְכַּת עֲבוֹדָה שָׁוָה בְּשָׁלֹשׁ הַתְּפִלּוֹת. עיין סימן קי״ט · 119.

מסקנה למעשה : לוֹמַר רְצֵה בְּשַׁחֲרִית, מִנְחָה וְעַרְבִית, בְּלִי לְהַחְלִיפָהּ בְּ״וְאִשֵּׁי יִשְׂרָאֵל״.

קו ב — ״וְאִשֵּׁי יִשְׂרָאֵל״ וַעֲבוֹדָה שֶׁבַּלֵּב

קו ב — ״וְאִשֵּׁי יִשְׂרָאֵל״ וַעֲבוֹדָה שֶׁבַּלֵּב

יסוד הסימן : ״וְאִשֵּׁי יִשְׂרָאֵל״ — נִשְׁמוֹת הַצַּדִּיקִים, קָרְבָּנוֹת בְּאֵשׁ הַהִתְלַהֲבוּת ; תְּפִלָּה הַנִּקְרָא ״עֲבוֹדָה שֶׁבַּלֵּב״ (סעיף ב).

קריאת הרב : הַמַּשְׁמָעוּת הַפְּנִימִית שֶׁל בִּרְכַּת עֲבוֹדָה : הַתְּפִלָּה מַמְשִׁיכָה אֶת הַקָּרְבָּן — הִתְלַהֲבוּת הַלֵּב הִיא אִשָּׁהּ. עיין סימן צ״ח · 98.

מסקנה למעשה : ״וּתְפִלָּתָם״ — גּוּף הַתְּפִלָּה שֶׁהֵן תֵּבוֹתֶיהָ : לְחַבֵּר אֶת הִתְלַהֲבוּת הַלֵּב לַתֵּבוֹת הַנֶּאֱמָרוֹת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״כ ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״כ (שֶׁרָאוּי לוֹמַר רְצֵה בְּכָל תְּפִלָּה)

  • ① אֲמִירַת רְצֵה. לוֹמַר רְצֵה בְּכָל הַתְּפִלּוֹת — שַׁחֲרִית, מִנְחָה, עַרְבִית. יסוד : שוע״ר ק״כ סעיף א.
  • ② לֹא לְשַׁנּוֹת בְּמִנְחָה וְעַרְבִית. לֹא לְהַתְחִיל בְּמִנְחָה אוֹ עַרְבִית בְּ״וְאִשֵּׁי יִשְׂרָאֵל וּתְפִלָּתָם״ בִּמְקוֹמָהּ — שֶׁאֵין טַעַם לְמִנְהָגָם. יסוד : שוע״ר ק״כ סעיף א.
  • ③ בֵּאוּר ״וְאִשֵּׁי יִשְׂרָאֵל״. נִשְׁמוֹת הַצַּדִּיקִים שֶׁמַּקְרִיב מִיכָאֵל עַל הַמִּזְבֵּחַ שֶׁל מַעְלָה ; הַתְּפִלָּה הִיא ״עֲבוֹדָה שֶׁבַּלֵּב״, בְּאֵשׁ הַהִתְלַהֲבוּת. יסוד : שוע״ר ק״כ סעיף ב.
  • ④ ״וּתְפִלָּתָם״. הִיא גּוּף הַתְּפִלָּה שֶׁהֵן תֵּבוֹתֶיהָ — לְחַבֵּר אֶת הִתְלַהֲבוּת הַלֵּב לְתֵבוֹת הַתְּפִלָּה. יסוד : שוע״ר ק״כ סעיף ב.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק״כ (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — אֲמִירַת רְצֵה, הַנֻּסָּח, וּבֵאוּר ״וְאִשֵּׁי יִשְׂרָאֵל״ — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
