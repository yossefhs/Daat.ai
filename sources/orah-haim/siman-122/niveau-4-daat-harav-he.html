דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קכ״ב · דִּינִים הַשַּׁיָּכִים בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״

דרכו של אדמו״ר הזקן בדִּינִים הַשַּׁיָּכִים בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״ : ״יִהְיוּ לְרָצוֹן״ — מִכְּלַל הַתְּפִלָּה, אָסוּר לְהַפְסִיק בֵּינוֹ לְי״ח ; בֵּינוֹ לְ״אֱלֹהַי נְצֹר״ — מֻתָּר לְהַפְסִיק לְדָבָר שֶׁבִּקְדֻשָּׁה ; ד׳ דְּבָרִים וְ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי״ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קכ״ב — דִּינִים הַשַּׁיָּכִים בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״. אצל הרב הסימן נפרש לג׳ סְעִיפִים : מַעֲמַד ״יִהְיוּ לְרָצוֹן״ כְּמִכְּלַל הַתְּפִלָּה וְדִינֵי הַהֶפְסֵק ; ד׳ דְּבָרִים (״עֲשֵׂה לְמַעַן שְׁמֶךָ״…) ; וּמִנְהַג אֲמִירַת ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי״ בְּכָל יוֹם, אַף בְּשַׁבָּת.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קכ״ב — ג׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קכ״ב). אדמו״ר הזקן דן בו בְּדִינִים הַשַּׁיָּכִים בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קכ״ב — דִּינִים הַשַּׁיָּכִים בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קכ״ב — דִּינִים הַשַּׁיָּכִים בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״ — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.122), ואחריו תרגומנו. הסימן כולל ג׳ סעיפים (סעיף k → ‎.122.k).

סעיף א ״יִהְיוּ לְרָצוֹן״ מִכְּלַל הַתְּפִלָּה — אָסוּר לְהַפְסִיק בֵּינוֹ לְי״ח.

דִּינִים הַשַּׁיָּכִים בֵּין י״ח לְיִהְיוּ לְרָצוֹן וּבוֹ ג׳ סְעִיפִים:תַּקָּנַת חֲכָמִים לוֹמַר אַחַר י״ח פָּסוּק ״יִהְיוּ לְרָצוֹן וְגו׳״ כְּמוֹ שֶׁאֲמָרוֹ דָּוִד אַחַר י״ח מִזְמוֹרִים, וְכֵיוָן שֶׁהוּא תִּקּוּן חֲכָמִים וּמִכְּלַל הַתְּפִלָּה — אָסוּר לְהַפְסִיק בֵּינוֹ לְי״ח, אֲפִלּוּ לַעֲנוֹת קַדִּישׁ וּקְדֻשָּׁה, וְאֵין צָרִיךְ לוֹמַר בַּאֲמִירַת ״אֱלֹהַי נְצֹר״ וּשְׁאָר תַּחֲנוּנִים שֶׁאַחַר הַתְּפִלָּה שֶׁאֵינָן אֶלָּא מִנְהָג וְלֹא חוֹבָה כְּלָל, אֶלָּא צָרִיךְ לוֹמַר ״יִהְיוּ לְרָצוֹן״ קֹדֶם ״אֱלֹהַי נְצֹר״. וְאִם בָּא לַחֲזֹר וּלְאָמְרוֹ פַּעַם אַחֶרֶת אַחַר הַתַּחֲנוּנִים — הָרְשׁוּת בְּיָדוֹ. וּבֵין ״יִהְיוּ לְרָצוֹן״ לֶ״אֱלֹהַי נְצֹר״ — מֻתָּר לְהַפְסִיק (וְלַעֲנוֹת כָּל דָּבָר שֶׁבִּקְדֻשָּׁה, אֲבָל לֹא דָּבָר אַחֵר עַד שֶׁיַּעֲקֹר רַגְלָיו, שֶׁהֲרֵי אֲפִלּוּ לָזוּז מִמְּקוֹמוֹ אָסוּר עַד שֶׁיִּפְסַע ג׳ פְּסִיעוֹת לַאֲחוֹרָיו כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ק״ד, וְהֶפְסֵק בְּדִבּוּר חָמוּר מֵהֲלִיכָה מִמְּקוֹמוֹ, שֶׁהֲרֵי לְצֹרֶךְ מֻתָּר לְהַלֵּךְ וְאָסוּר לְדַבֵּר כְּמוֹ שֶׁנִּתְבָּאֵר שָׁם. וַאֲפִלּוּ לָרֹק אָסוּר קֹדֶם שֶׁפָּסַע, לְפִי שֶׁעֲדַיִן עוֹמֵד לִפְנֵי הַמֶּלֶךְ כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן צ״ז, כָּל שֶׁכֵּן לְדַבֵּר לְפָנָיו דְּבָרִים אֲחֵרִים). וְאִם הִתְחִיל לוֹמַר ״אֱלֹהַי נְצֹר״ אוֹ שְׁאָר תַּחֲנוּנִים וְהִתְחִיל שְׁלִיחַ צִבּוּר לְסַדֵּר תְּפִלָּתוֹ וְהִגִּיעַ לִקְדֻשָּׁה — מְקַצֵּר וְעוֹלֶה וּפוֹסֵעַ לְאַחֲרָיו. וַאֲפִלּוּ לֹא קִצֵּר — יָכוֹל לְהַפְסִיק וְלַעֲנוֹת כְּדֶרֶךְ שֶׁמַּפְסִיק לַעֲנוֹת בְּבִרְכוֹת קְרִיאַת שְׁמַע אֲפִלּוּ בָּאֶמְצַע, שֶׁהַתַּחֲנוּנִים אֵינָם חֲמוּרִים כִּתְפִלָּה, אַף עַל פִּי שֶׁעֲדַיִן לֹא עָקַר רַגְלָיו. וְיֵשׁ אוֹמְרִים שֶׁאִם אֵינוֹ אוֹמֵר ״יִהְיוּ לְרָצוֹן״ עַד אַחַר סִיּוּם הַתַּחֲנוּנִים — אֵין בָּזֶה אִסּוּר, שֶׁהֲרֵי בְּבִרְכוֹת י״ח עַצְמָן אִם בָּא לוֹמַר בְּסוֹף כָּל בְּרָכָה מֵעֵין הַבְּרָכָה אוֹמֵר, וְאֵין לְהַחֲמִיר בֵּין י״ח לְיִהְיוּ לְרָצוֹן יוֹתֵר מֵהַבְּרָכוֹת עַצְמָן. וְכֵן נוֹהֲגִין בִּמְדִינוֹת אֵלּוּ. וְכֵיוָן שֶׁמַּפְסִיקִין בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״ בֶּ״אֱלֹהַי נְצֹר״ וּשְׁאָר תַּחֲנוּנִים — מַפְסִיק גַּם כֵּן לַעֲנוֹת כָּל דָּבָר שֶׁבִּקְדֻשָּׁה. וּמִכָּל מָקוֹם, אִם יָכוֹל לְקַצֵּר בְּתַחֲנוּנִים כְּדֵי לוֹמַר גַּם ״יִהְיוּ לְרָצוֹן״ קֹדֶם שֶׁיַּפְסִיק — יוֹתֵר טוֹב לְקַצֵּר. אֲבָל טוֹב לָחֹש לַסְּבָרָא הָרִאשׁוֹנָה שֶׁהוּא עִקָּר, וְלִנְהֹג לְכַתְּחִלָּה לוֹמַר ב׳ פְּעָמִים ״יִהְיוּ לְרָצוֹן״, אֶחָד מִיָּד אַחַר י״ח וְאֶחָד אַחַר הַתַּחֲנוּנִים כַּנָּהוּג:

״יִהְיוּ לְרָצוֹן״. תַּקָּנַת חֲכָמִים לוֹמַר אַחַר י״ח ״יִהְיוּ לְרָצוֹן וְגו׳״, כְּמוֹ דָּוִד אַחַר י״ח מִזְמוֹרִים ; וְכֵיוָן שֶׁהוּא תִּקּוּן חֲכָמִים וּמִכְּלַל הַתְּפִלָּה — אָסוּר לְהַפְסִיק בֵּינוֹ לְי״ח, אֲפִלּוּ לְקַדִּישׁ וּקְדֻשָּׁה ; כָּל שֶׁכֵּן ״אֱלֹהַי נְצֹר״ וּשְׁאָר תַּחֲנוּנִים שֶׁאֵינָן אֶלָּא מִנְהָג — לָכֵן ״יִהְיוּ לְרָצוֹן״ קֹדֶם ״אֱלֹהַי נְצֹר״ (וְאִם בָּא לַחֲזֹר אַחַר הַתַּחֲנוּנִים — הָרְשׁוּת בְּיָדוֹ). בֵּין ״יִהְיוּ לְרָצוֹן״ לֶ״אֱלֹהַי נְצֹר״ — מֻתָּר לְהַפְסִיק לְכָל דָּבָר שֶׁבִּקְדֻשָּׁה, לֹא לְדָבָר אַחֵר עַד שֶׁיַּעֲקֹר רַגְלָיו (ג׳ פְּסִיעוֹת, סימן ק״ד ; לָרֹק אָסוּר — עוֹמֵד לִפְנֵי הַמֶּלֶךְ, סימן צ״ז). הִתְחִיל ״אֱלֹהַי נְצֹר״ וְהִגִּיעַ שַׁ״ץ לִקְדֻשָּׁה — מְקַצֵּר וְעוֹלֶה. וְיֵשׁ אוֹמְרִים : לְאַחֵר ״יִהְיוּ לְרָצוֹן״ עַד אַחַר הַתַּחֲנוּנִים אֵין בּוֹ אִסּוּר, וְכֵן נוֹהֲגִין בִּמְדִינוֹת אֵלּוּ. טוֹב לָחֹשׁ לַסְּבָרָא הָרִאשׁוֹנָה שֶׁהוּא עִקָּר, וְלוֹמַר לְכַתְּחִלָּה ב׳ פְּעָמִים ״יִהְיוּ לְרָצוֹן״.

סעיף ב הָרָגִיל לוֹמַר ד׳ דְּבָרִים אֵלּוּ — זוֹכֶה וּמְקַבֵּל פְּנֵי הַשְּׁכִינָה.

הָרָגִיל לוֹמַר ד׳ דְּבָרִים אֵלּוּ — זוֹכֶה וּמְקַבֵּל פְּנֵי הַשְּׁכִינָה, ״עֲשֵׂה לְמַעַן שְׁמֶךָ״. ״עֲשֵׂה לְמַעַן יְמִינֶךָ״. ״עֲשֵׂה לְמַעַן תּוֹרָתֶךָ״. ״עֲשֵׂה לְמַעַן קְדֻשָּׁתֶךָ״:

ד׳ דְּבָרִים. הָרָגִיל לוֹמַר ד׳ דְּבָרִים אֵלּוּ — זוֹכֶה וּמְקַבֵּל פְּנֵי הַשְּׁכִינָה : ״עֲשֵׂה לְמַעַן שְׁמֶךָ״, ״עֲשֵׂה לְמַעַן יְמִינֶךָ״, ״עֲשֵׂה לְמַעַן תּוֹרָתֶךָ״, ״עֲשֵׂה לְמַעַן קְדֻשָּׁתֶךָ״.

סעיף ג ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי כו׳״ — מֻתָּרִים לְאוֹמְרוֹ בְּשַׁבָּת.

יֵשׁ נוֹהֲגִין לוֹמַר ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי כו׳״ בְּכָל יוֹם, וּמֻתָּרִים לְאוֹמְרוֹ בְּשַׁבָּת:

״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי״. יֵשׁ נוֹהֲגִין לוֹמַר ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי כו׳״ בְּכָל יוֹם, וּמֻתָּרִים לְאוֹמְרוֹ אַף בְּשַׁבָּת.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קכ״ב (מהדורת קה״ת, ג׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

היסוד — דִּינִים הַשַּׁיָּכִים בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״

היסוד : בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים. תחילה מַעֲמַד ״יִהְיוּ לְרָצוֹן״ כְּמִכְּלַל הַתְּפִלָּה וְאִסּוּר הַהֶפְסֵק בֵּינוֹ לְי״ח. אחר כך גֶּדֶר הַהֶפְסֵק בֵּין ״יִהְיוּ לְרָצוֹן״ לַתַּחֲנוּנִים — מֻתָּר לְדָבָר שֶׁבִּקְדֻשָּׁה. ולבסוף ד׳ דְּבָרִים וּמִנְהַג ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי״.

״יִהְיוּ לְרָצוֹן״ — מִכְּלַל הַתְּפִלָּה

״יִהְיוּ לְרָצוֹן״ — מִכְּלַל הַתְּפִלָּה

היסוד. ״יִהְיוּ לְרָצוֹן״ תִּקּוּן חֲכָמִים וּמִכְּלַל הַתְּפִלָּה (כְּדָוִד אַחַר י״ח מִזְמוֹרִים) : אָסוּר לְהַפְסִיק בֵּינוֹ לְי״ח, אֲפִלּוּ לְקַדִּישׁ וּקְדֻשָּׁה ; אוֹמְרוֹ קֹדֶם ״אֱלֹהַי נְצֹר״ (סעיף א).

המסקנה. הַתַּחֲנוּנִים אֵינָן אֶלָּא מִנְהָג — אֵינָן קוֹדְמוֹת לְ״יִהְיוּ לְרָצוֹן״ הַמְתֻקָּן. עיין סימן 121.

הַהֶפְסֵק לַתַּחֲנוּנִים וְלַקְּדֻשָּׁה

גֶּדֶר הַהֶפְסֵק

היסוד. בֵּין ״יִהְיוּ לְרָצוֹן״ לֶ״אֱלֹהַי נְצֹר״ — מֻתָּר לְהַפְסִיק לְדָבָר שֶׁבִּקְדֻשָּׁה, לֹא לְדָבָר אַחֵר עַד שֶׁיַּעֲקֹר רַגְלָיו (ג׳ פְּסִיעוֹת, סימן ק״ד ; לָרֹק אָסוּר — עוֹמֵד לִפְנֵי הַמֶּלֶךְ, סימן צ״ז) (סעיף א).

המסקנה. הִתְחִיל תַּחֲנוּנִים וְהִגִּיעַ שַׁ״ץ לִקְדֻשָּׁה — מְקַצֵּר וְעוֹלֶה, אוֹ מַפְסִיק כִּבְבִרְכוֹת קְרִיאַת שְׁמַע. עיין סימן 104 (ג׳ פְּסִיעוֹת).

ד׳ דְּבָרִים וְ״אֱלֹהַי״

ד׳ דְּבָרִים וְ״אֱלֹהַי״

היסוד. הָרָגִיל לוֹמַר ד׳ דְּבָרִים אֵלּוּ (״עֲשֵׂה לְמַעַן שְׁמֶךָ״…) — זוֹכֶה וּמְקַבֵּל פְּנֵי הַשְּׁכִינָה (סעיף ב).

המסקנה. יֵשׁ נוֹהֲגִין לוֹמַר ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי כו׳״ בְּכָל יוֹם, וּמֻתָּרִים לְאוֹמְרוֹ בְּשַׁבָּת (סעיף ג). עיין סימן 97 (הנהגה לאחר התפלה).

הפניה ללימוד. היקף הסימן — מַעֲמַד ״יִהְיוּ לְרָצוֹן״, גֶּדֶר הַהֶפְסֵק וְד׳ דְּבָרִים — נוגע להנהגת סוף התפלה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּינִים הַשַּׁיָּכִים בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״ — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קכ״ב. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מג׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
״יִהְיוּ לְרָצוֹן״ וְהַהֶפְסֵק לְי״ח או״ח קכ״ב — אֵינוֹ פּוֹסֵק בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״, שֶׁמִּכְּלַל הַתְּפִלָּה הוּא. מ״ב קכ״ב — טַעַם ״יִהְיוּ לְרָצוֹן״ מִכְּלַל הַתְּפִלָּה, וְגֶדֶר הַהֶפְסֵק. ערוה״ש קכ״ב — מְקוֹם ״יִהְיוּ לְרָצוֹן״ וְדִין הַפְסָקָתוֹ. דרך אדמו״ר הזקן — תִּקּוּן חֲכָמִים וּמִכְּלַל הַתְּפִלָּה, אָסוּר לְהַפְסִיק אֲפִלּוּ לְקַדִּישׁ וּקְדֻשָּׁה ; ״יִהְיוּ לְרָצוֹן״ קֹדֶם ״אֱלֹהַי נְצֹר״ (סעיף א).
הַהֶפְסֵק לַקְּדֻשָּׁה וְהַתַּחֲנוּנִים או״ח קכ״ב — בֵּין ״יִהְיוּ לְרָצוֹן״ לִשְׁאָר תַּחֲנוּנִים שַׁפִּיר דָּמֵי ; מְקַצֵּר וְעוֹלֶה. מ״ב קכ״ב — פְּרָטֵי הַהֶפְסֵק לְדָבָר שֶׁבִּקְדֻשָּׁה וְקֹדֶם עֲקִירַת רַגְלָיו. ערוה״ש קכ״ב — גֶּדֶר הַהֶפְסֵק בָּאֱלֹהַי נְצֹר וְלַתַּחֲנוּנִים. דרך אדמו״ר הזקן — מֻתָּר לְהַפְסִיק לְדָבָר שֶׁבִּקְדֻשָּׁה ; לֹא לְדָבָר אַחֵר עַד ג׳ פְּסִיעוֹת (סימן ק״ד, צ״ז) ; הִגִּיעַ שַׁ״ץ לִקְדֻשָּׁה — מְקַצֵּר וְעוֹלֶה (סעיף א).
ד׳ דְּבָרִים וְ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי״ או״ח קכ״ב — הָרָגִיל לוֹמַר ד׳ דְּבָרִים אֵלּוּ זוֹכֶה וּמְקַבֵּל פְּנֵי שְׁכִינָה. מ״ב קכ״ב — עִנְיַן ״עֲשֵׂה לְמַעַן…״ וְאֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי. ערוה״ש קכ״ב — נֹסַח הַתַּחֲנוּנִים שֶׁאַחַר הַתְּפִלָּה. דרך אדמו״ר הזקן — הָרָגִיל בְּד׳ דְּבָרִים — זוֹכֶה וּמְקַבֵּל פְּנֵי הַשְּׁכִינָה ; יֵשׁ נוֹהֲגִין ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי״, וּמֻתָּר בְּשַׁבָּת (סעיפים ב-ג).

הטבלה נערכה מטקסט המחבר (אורח חיים קכ״ב, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמ״א, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ד׳, ט״ז-י״ז). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — ״יִהְיוּ לְרָצוֹן״ מִכְּלַל הַתְּפִלָּה

קו א — ״יִהְיוּ לְרָצוֹן״ מִכְּלַל הַתְּפִלָּה

יסוד הסימן : תִּקּוּן חֲכָמִים וּמִכְּלַל הַתְּפִלָּה ; אָסוּר לְהַפְסִיק בֵּינוֹ לְי״ח, אֲפִלּוּ לְקַדִּישׁ וּקְדֻשָּׁה (סעיף א).

קריאת הרב : ״יִהְיוּ לְרָצוֹן״ קֹדֶם ״אֱלֹהַי נְצֹר״, שֶׁהַתַּחֲנוּנִים אֵינָן אֶלָּא מִנְהָג. עיין סימן 121.

מסקנה למעשה : לֹא לְהַפְסִיק בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״, וְלוֹמְרוֹ קֹדֶם כָּל תַּחֲנוּן.

קו ב — גֶּדֶר הַהֶפְסֵק

קו ב — גֶּדֶר הַהֶפְסֵק

יסוד הסימן : בֵּין ״יִהְיוּ לְרָצוֹן״ לֶ״אֱלֹהַי נְצֹר״ — מֻתָּר לְהַפְסִיק לְדָבָר שֶׁבִּקְדֻשָּׁה, לֹא לְדָבָר אַחֵר עַד עֲקִירַת רַגְלָיו (סעיף א).

קריאת הרב : הֶפְסֵק בְּדִבּוּר חָמוּר מֵהֲלִיכָה ; לָרֹק אָסוּר קֹדֶם שֶׁפָּסַע. עיין סימן 104.

מסקנה למעשה : הִגִּיעַ שַׁ״ץ לִקְדֻשָּׁה — מְקַצֵּר וְעוֹלֶה, וּלְכַתְּחִלָּה לוֹמַר ב׳ פְּעָמִים ״יִהְיוּ לְרָצוֹן״.

קו ג — ד׳ דְּבָרִים וְ״אֱלֹהַי״

קו ג — ד׳ דְּבָרִים וְ״אֱלֹהַי״

יסוד הסימן : הָרָגִיל לוֹמַר ד׳ דְּבָרִים אֵלּוּ — זוֹכֶה וּמְקַבֵּל פְּנֵי הַשְּׁכִינָה (סעיף ב).

קריאת הרב : יֵשׁ נוֹהֲגִין ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי כו׳״ בְּכָל יוֹם, וּמֻתָּר בְּשַׁבָּת. עיין סימן 97.

מסקנה למעשה : רְגִילוּת בְּד׳ הַבַּקָּשׁוֹת וּבַ״אֱלֹהַי״ הַיּוֹמִי, אַף בְּשַׁבָּת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קכ״ב ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קכ״ב (דִּינִים הַשַּׁיָּכִים בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״)

  • ① אֵין מַפְסִיקִין. אֵין לְהַפְסִיק בֵּין י״ח לְ״יִהְיוּ לְרָצוֹן״, אֲפִלּוּ לְקַדִּישׁ וּקְדֻשָּׁה — שֶׁמִּכְּלַל הַתְּפִלָּה הוּא. יסוד : שוע״ר קכ״ב סעיף א.
  • ② קֹדֶם הַתַּחֲנוּנִים. לוֹמַר ״יִהְיוּ לְרָצוֹן״ קֹדֶם ״אֱלֹהַי נְצֹר״ וּשְׁאָר תַּחֲנוּנִים (שֶׁאֵינָן אֶלָּא מִנְהָג). יסוד : שוע״ר קכ״ב סעיף א.
  • ③ הַהֶפְסֵק. בֵּין ״יִהְיוּ לְרָצוֹן״ לֶ״אֱלֹהַי נְצֹר״ — מֻתָּר לְהַפְסִיק לְדָבָר שֶׁבִּקְדֻשָּׁה, לֹא לְדָבָר אַחֵר עַד עֲקִירַת רַגְלָיו ; שַׁ״ץ בַּקְּדֻשָּׁה — מְקַצֵּר וְעוֹלֶה. יסוד : שוע״ר קכ״ב סעיף א.
  • ④ ב׳ פְּעָמִים לְכַתְּחִלָּה. טוֹב (עִקָּר) לוֹמַר ״יִהְיוּ לְרָצוֹן״ ב׳ פְּעָמִים — מִיָּד אַחַר י״ח וְאַחַר הַתַּחֲנוּנִים. יסוד : שוע״ר קכ״ב סעיף א.
  • ⑤ ד׳ דְּבָרִים וְ״אֱלֹהַי״. לְהִתְרַגֵּל בְּד׳ ״עֲשֵׂה לְמַעַן…״ (זוֹכֶה לִפְנֵי הַשְּׁכִינָה) וּבְ״אֱלֹהַי עַד שֶׁלֹּא נוֹצַרְתִּי״ בְּכָל יוֹם, אַף בְּשַׁבָּת. יסוד : שוע״ר קכ״ב סעיפים ב-ג.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קכ״ב (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַהֶפְסֵק, ״יִהְיוּ לְרָצוֹן״, הַתַּחֲנוּנִים — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
