דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן מ״ג · דִּין אֵיךְ לְהִתְנַהֵג בַּתְּפִלִּין בְּהִכָּנְסוֹ לְבֵית הַכִּסֵּא

דרכו של אדמו״ר הזקן בהנהגת התפילין בהכנסו לבית הכסא: לְהַשְׁתִּין בְּבֵית הַכִּסֵּא קָבוּעַ אוֹ עֲרַאי (הַנִּיצוֹצוֹת, מְיֻשָּׁב וּמְעֻמָּד) ; הַחֲלִיצָה בְּרִחוּק ד׳ אַמּוֹת וְהַגְּלִילָה בָּרְצוּעוֹת ; הַשּׁוֹכֵחַ, הָרוֹפֵא וְהַבַּעַל נֶפֶשׁ — ברגישות חב״ד.

למה רמה 4 מיוחדת לאדמו״ר הזקן? השולחן ערוך של אדמו״ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתבו אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. דף זה מאסף את שיטת הרב על סימן מ״ג — דִּין אֵיךְ לְהִתְנַהֵג בַּתְּפִלִּין בְּהִכָּנְסוֹ לְבֵית הַכִּסֵּא. הסימן כולל אצל הרב ט׳ סְעִיפִים, שבהם אדמו״ר הזקן פורש: אִסּוּר הַשְׁתָּנָה בִּתְפִלִּין בְּבֵית הַכִּסֵּא קָבוּעַ (גְּזֵרָה שֶׁמָּא יַעֲשֶׂה בָּהֶן צְרָכָיו) וַחֲשַׁשׁ הַנִּיצוֹצוֹת ; גֶּדֶר קָבוּעַ וַעֲרַאי וּבֵית הַכִּסֵּא שֶׁבַּבַּיִת (מָקוֹם הַמִּשְׁתַּמֵּר) ; הַחֲלִיצָה בְּרִחוּק ד׳ אַמּוֹת וְהַגְּלִילָה בָּרְצוּעוֹת ; דִּין הַלַּיְלָה, הַשּׁוֹכֵחַ, הָרוֹפֵא וְהַבַּעַל נֶפֶשׁ.

הערת מקור. הטקסט העברי המובא בדף זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן מ״ג — ט׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנסרקה בספריא (Shulchan Arukh HaRav, Orach Chayim מ״ג). בתשעת הסעיפים, אדמו״ר הזקן דן באִסּוּר הַשְׁתָּנָה בִּתְפִלִּין בְּבֵית הַכִּסֵּא קָבוּעַ (מִשּׁוּם גְּזֵרָה שֶׁמָּא יַעֲשֶׂה בָּהֶן צְרָכָיו) וּבְשִׁפְשׁוּף הַנִּיצוֹצוֹת (מְיֻשָּׁב וּמְעֻמָּד), בְּגֶדֶר קָבוּעַ וַעֲרַאי וּבֵית הַכִּסֵּא שֶׁבַּבַּיִת (מָקוֹם הַמִּשְׁתַּמֵּר), בַּחֲלִיצָה בְּרִחוּק ד׳ אַמּוֹת וּבַגְּלִילָה בָּרְצוּעוֹת (עִם קְדֻשַּׁת הַשַּׁדַּי), בְּדִין הַלַּיְלָה וּבַשּׁוֹכֵחַ, וּבָרוֹפֵא וּבַעַל נֶפֶשׁ. ההפניה לסֵפֶר תּוֹרָה היא של הרב עצמו (יורה דעה). הביאורים הם שלנו. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן מ״ג — אֵיךְ לְהִתְנַהֵג בַּתְּפִלִּין בְּהִכָּנְסוֹ לְבֵית הַכִּסֵּא

הטקסט המלא של שולחן ערוך אדמו״ר הזקן

שולחן ערוך הרב, סימן מ״ג — אֵיךְ לְהִתְנַהֵג בַּתְּפִלִּין בְּהִכָּנְסוֹ לְבֵית הַכִּסֵּא — וּבוֹ ט׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.43), ולאחריו ביאור קצר. הסימן כולל ט׳ סְעִיפִים.

סעיף א אָסוּר לִכָּנֵס לְבֵית הַכִּסֵּא קָבוּעַ לְהַשְׁתִּין בִּתְפִלִּין שֶׁבְּרֹאשׁוֹ ; בַּעֲרַאי מֻתָּר ; וְאוֹחֲזָן בְּיָדוֹ מְעֻמָּד אָסוּר שֶׁמָּא יְשַׁפְשֵׁף הַנִּיצוֹצוֹת.

דִּין אֵיךְ לְהִתְנַהֵג בַּתְּפִלִּין בְּהִכָּנְסוֹ לְבֵית הַכִּסֵּא וּבוֹ ט׳ סְעִיפִים:אָסוּר לִכָּנֵס לְבֵית הַכִּסֵּא קָבוּעַ לְהַשְׁתִּין בִּתְפִלִּין שֶׁבְּרֹאשׁוֹ אוֹ בִּזְרוֹעוֹ, גְּזֵרָה שֶׁמָּא יַעֲשֶׂה בָּהֶן צְרָכָיו, אֲבָל בְּבֵית הַכִּסֵּא עֲרָאִי דְּהַיְנוּ מָקוֹם שֶׁאֵין רְגִילִין לִפָּנוֹת שָׁם לִגְדוֹלִים — מֻתָּר לְהַשְׁתִּין בָּהֶן, וְאֵין חוֹשְׁשִׁין שֶׁמָּא יִפָּנֶה שָׁם אֲפִלּוּ הוּא מָקוֹם צָנוּעַ. וְיֵשׁ מִי שֶׁאוֹסֵר לְהַשְׁתִּין בִּתְפִלִּין אֲפִלּוּ שֶׁלֹּא בְּבֵית הַכִּסֵּא — גְּזֵרָה שֶׁמָּא יָפִיחַ בָּהֶן, כִּי כֵן דֶּרֶךְ הַמַּשְׁתִּינִים, וּתְפִלִּין צְרִיכִין גּוּף נָקִי. וְיֵשׁ לָחֹש לִדְבָרָיו. בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁהֵן עָלָיו, אֲבָל כְּשֶׁהֵן בְּכִיסָן וְאוֹחֵז הַכִּיס בְּיָדוֹ — מֻתָּר לְהַשְׁתִּין בָּהֶ[ן] לְדִבְרֵי הַכֹּל, שֶׁכְּשֶׁהֵן בְּכִיסָן — אֵין אִסּוּר בַּהֲפָחָה לְדִבְרֵי הַכֹּל, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן מד. וְאִם אֵינָן בְּכִיסָן אֶלָּא אוֹחֲזָן בְּיָדוֹ, אֲפִלּוּ אִם תּוֹפְסָן בְּבִגְדוֹ וּבְיָדוֹ — אָסוּר לְהַשְׁתִּין בָּהֶן מְעֻמָּד אֲפִלּוּ שֶׁלֹּא בְּבֵית הַכִּסֵּא, שֶׁמָּא יְשַׁפְשֵׁף בְּיָד זוֹ אֶת נִיצוֹצוֹת מֵי רַגְלַיִם שֶׁיִּפְּלוּ עַל רַגְלָיו כְּשֶׁמַּשְׁתִּין מְעֻמָּד:

לְהַשְׁתִּין בְּבֵית הַכִּסֵּא קָבוּעַ. אָסוּר לִכָּנֵס לְבֵית הַכִּסֵּא קָבוּעַ לְהַשְׁתִּין בִּתְפִלִּין שֶׁבְּרֹאשׁוֹ (גְּזֵרָה שֶׁמָּא יַעֲשֶׂה בָּהֶן צְרָכָיו) ; וּבַעֲרַאי מֻתָּר. יֵשׁ אוֹסֵר אַף שֶׁלֹּא בְּבֵית הַכִּסֵּא (שֶׁמָּא יָפִיחַ) וְיֵשׁ לָחֹשׁ. בְּכִיסָן וְאוֹחֵז בְּיָדוֹ — מֻתָּר לְדִבְרֵי הַכֹּל ; אֲבָל אוֹחֲזָן בְּיָדוֹ אָסוּר לְהַשְׁתִּין מְעֻמָּד שֶׁמָּא יְשַׁפְשֵׁף הַנִּיצוֹצוֹת.

סעיף ב הֵן בְּחֵיקוֹ וְחָגוּר, אוֹ צְרוּרִין וּקְשׁוּרִין בְּבִגְדוֹ — מֻתָּר לְהַשְׁתִּין בָּהֶם אֲפִלּוּ מְעֻמָּד.

אֲבָל אִם הֵן בְּחֵיקוֹ וְחָגַר חֲגוֹרָה, וְכֵן אִם הֵן צְרוּרִין וּקְשׁוּרִין בְּבִגְדוֹ שֶׁאֵין צָרִיךְ (עַיֵּן בְּמַה שֶּׁנִּתְבָּאֵר לְעֵיל) לְאָחֳזָן בְּיָדוֹ — מֻתָּר לְהַשְׁתִּין בָּהֶם אֲפִלּוּ מְעֻמָּד:

בְּחֵיקוֹ וְחָגוּר. אִם הֵן בְּחֵיקוֹ וְחָגַר, אוֹ צְרוּרִין וּקְשׁוּרִין בְּבִגְדוֹ שֶׁאֵין צָרִיךְ לְאָחֳזָן בְּיָדוֹ — מֻתָּר לְהַשְׁתִּין בָּהֶם אֲפִלּוּ מְעֻמָּד.

סעיף ג מַשְׁתִּין מְיֻשָּׁב אוֹ בַּעֲפַר תִּיחוּחַ מֻתָּר ; אוֹחֲזָן בִּימִינוֹ כְּנֶגֶד לִבּוֹ ; גּוֹלְלָן בָּרְצוּעוֹת ; וְחוֹלֵץ בְּרִחוּק ד׳ אַמּוֹת.

אֲבָל אִם מַשְׁתִּין מְיֻשָּׁב אוֹ בַּעֲפַר תִּחוּחַ שֶׁאֵין נִיצוֹצוֹת נוֹפְלִין עַל רַגְלָיו — מֻתָּר לְהַשְׁתִּין בָּהֶן אֲפִלּוּ כְּשֶׁהֵן מְגֻלִּין בְּיָדוֹ. וּבִלְבַד שֶׁלֹּא יִכָּנֵס בָּהֶן לְבֵית הַכִּסֵּא קָבוּעַ — גְּזֵרָה שֶׁמָּא יַעֲשֶׂה בָּהֶן צְרָכָיו כְּשֶׁהֵן מְגֻלִּין בְּיָדוֹ, כֵּיוָן שֶׁהוּא מָקוֹם שֶׁרְגִילִים לִפָּנוֹת בּוֹ לִגְדוֹלִים. אֲבָל אִם אוֹחֲזָן בְּבִגְדוֹ בְּיָדוֹ — מֻתָּר אֲפִלּוּ לַעֲשׂוֹת בָּהֶן צְרָכָיו בְּבֵית הַכִּסֵּא קָבוּעַ. וְצָרִיךְ שֶׁיֹּאחֲזֵן בִּימִינוֹ כְּנֶגֶד לִבּוֹ, אֲבָל לֹא בִּשְׂמֹאלוֹ, שֶׁמָּא יְקַנַּח בָּהּ, אֲבָל בְּיָמִין אֵין מְקַנְּחִין. וּלְשִׁפְשׁוּף נִיצוֹצוֹת אֵין לָחֹש כְּשֶׁנִּפְנָה לִגְדוֹלִים, שֶׁאַף שֶׁאִי אֶפְשָׁר לִגְדוֹלִים בְּלֹא קְטַנִּים, שֶׁמִּתּוֹךְ שֶׁדּוֹחֵק עַצְמוֹ לִגְדוֹלִים בָּא לִידֵי קְטַנִּים, מִכָּל מָקוֹם מִן הַסְּתָם הוּא עוֹשֶׂה צְרָכָיו מְיֻשָּׁב, וּמְיֻשָּׁב אֵין לָחֹש לְנִיצוֹצוֹת. וְיִזָּהֵר לְגָלְלָן בָּרְצוּעוֹת שֶׁלָּהֶן, שֶׁלֹּא תְּהֵא רְצוּעָה יוֹצְאָה מִתַּחַת יָדוֹ טֶפַח, שֶׁגַּם בָּרְצוּעוֹת יֵשׁ קְדֻשָּׁה, שֶׁהֲרֵי ד׳ וְיוּ"ד שֶׁל שַׁדַּ"י קְבוּעִים בָּהֶן. וּכְשֶׁהוּא יוֹצֵא מַרְחִיק אַרְבַּע אַמּוֹת מִפֶּתַח בֵּית הַכִּסֵּא וּמְנִיחָן עָלָיו. וְכֵן כְּשֶׁנִּכְנָס — צָרִיךְ לְחָלְצָן בְּרִחוּק אַרְבַּע אַמּוֹת מִפֶּתַח בֵּית הַכִּסֵּא, וּמַכְנִיסָן בְּבִגְדוֹ בְּיָדוֹ. וַאֲפִלּוּ הַנִּכְנָס לְבֵית הַכִּסֵּא קָבוּעַ לְהַשְׁתִּין מַיִם — צָרִיךְ לְחָלְצָן בְּרִחוּק אַרְבַּע אַמּוֹת. וְאַף עַל פִּי שֶׁאֵין אִסּוּר מִן הַדִּין לֵילֵךְ בִּתְפִלִּין אֲפִלּוּ תּוֹךְ אַרְבַּע אַמּוֹת שֶׁל צוֹאָה עַצְמָהּ, מִכָּל מָקוֹם כֵּיוָן שֶׁזֶּה צָרִיךְ הוּא לַחֲלֹץ תְּפִלָּיו מִשּׁוּם גְּזֵרַת חֲכָמִים שֶׁמָּא יַעֲשֶׂה בָּהֶן צְרָכָיו — יֵשׁ לוֹ לְחָלְצָן מִיָּד שֶׁמַּגִּיעַ לְאַרְבַּע אַמּוֹת שֶׁל בֵּית הַכִּסֵּא, שֶׁהֵן כְּאַרְבַּע אַמּוֹת שֶׁל צוֹאָה עַצְמָהּ כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן פג. אֲבָל יָכוֹל לְהוֹלִיכָן עִמּוֹ בְּיָדָיו עַד בֵּית הַכִּסֵּא, וְשָׁם יֹאחֲזֵן בְּבִגְדוֹ בְּיַד יָמִין אִם יַשְׁתִּין מְיֻשָּׁב אוֹ בַּעֲפַר תִּחוּחַ. אֲבָל אִם יַשְׁתִּין מְעֻמָּד — לֹא יוֹלִיכֵן עִמּוֹ לְבֵית הַכִּסֵּא, אֶלָּא יִתְּנוֹ לַחֲבֵרוֹ, אֶלָּא אִם כֵּן מוֹלִיכָן בְּכִיסָן. אֲבָל הַנִּכְנָס לְבֵית הַכִּסֵּא קָבוּעַ לַעֲשׂוֹת צְרָכָיו מְיֻשָּׁב — אֵין לוֹ לִתְּנָן לַחֲבֵרוֹ שֶׁכֵּיוָן שֶׁהִתִּירוּ לוֹ חֲכָמִים לְהַכְנִיסָן לְבֵית הַכִּסֵּא בְּבִגְדוֹ בְּיָדוֹ — יֵשׁ לְהַכְנִיסָן עִמּוֹ, כְּדֵי שֶׁיִּשְׁמְרוּהוּ מִן הַמַּזִּיקִים, וְגַם מִצְוָה לְשָׁמְרָן בְּעַצְמוֹ. בַּמֶּה דְּבָרִים אֲמוּרִים שֶׁהַנִּכְנָס לְבֵית הַכִּסֵּא קָבוּעַ צָרִיךְ לְחָלְצָן בְּרִחוּק אַרְבַּע אַמּוֹת? בְּבֵית הַכִּסֵּא שֶׁהַצּוֹאָה שֶׁבּוֹ הִיא מְגֻלָּה בְּלֹא חֲפִירָה, אֲבָל אִם הִיא בַּחֲפִירָה — הֲרֵי הַחֲפִירָה הִיא רְשׁוּת לְעַצְמוֹ וּבֵית הַכִּסֵּא שֶׁלְּמַעְלָה רְשׁוּת לְעַצְמוֹ, וְאֵין לוֹ דִּין בֵּית הַכִּסֵּא כְּלָל כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן פג, וּמֻתָּר לִכָּנֵס בְּתוֹכוֹ בִּתְפִלִּין שֶׁעָלָיו. וּמִכָּל מָקוֹם, קֹדֶם שֶׁיַּשְׁתִּין — צָרִיךְ לְחָלְצָן לְדִבְרֵי הַכֹּל, כְּמוֹ בִּשְׁאָר בֵּית הַכִּסֵּא קָבוּעַ (וְגַם לְכַסּוֹתָן, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן שטו). בַּמֶּה דְּבָרִים אֲמוּרִים? לְהַשְׁתִּין, אֲבָל לַעֲשׂוֹת צְרָכָיו — צָרִיךְ לְחָלְצָן בְּרִחוּק אַרְבַּע אַמּוֹת מִמָּקוֹם שֶׁרוֹצֶה לַעֲשׂוֹת צְרָכָיו, אֲפִלּוּ נִפְנָה בְּמָקוֹם שֶׁאֵינוֹ בֵּית הַכִּסֵּא כְּלָל אֶלָּא בַּחֲצֵרוֹ אוֹ עַל פְּנֵי הַשָּׂדֶה:

מְיֻשָּׁב וְרִחוּק ד׳ אַמּוֹת. מַשְׁתִּין מְיֻשָּׁב אוֹ בַּעֲפַר תִּיחוּחַ — מֻתָּר אַף מְגֻלִּין בְּיָדוֹ, וּבִלְבַד שֶׁלֹּא בְּבֵית הַכִּסֵּא קָבוּעַ. אוֹחֲזָן בִּימִינוֹ כְּנֶגֶד לִבּוֹ (לֹא בִּשְׂמֹאל שֶׁמָּא יְקַנַּח) ; גּוֹלְלָן בָּרְצוּעוֹת שֶׁלֹּא תֵּצֵא רְצוּעָה טֶפַח ; וְחוֹלֵץ בְּרִחוּק ד׳ אַמּוֹת מִפֶּתַח בֵּית הַכִּסֵּא. וְהַחֲפִירָה רְשׁוּת לְעַצְמָהּ.

סעיף ד נִפְנָה בַּלַּיְלָה שֶׁאֵין שְׁהוּת לַהֲנִיחָן — חוֹלְצָן וּמַנִּיחָן בְּכִיס אוֹ בִּכְלִי וְאוֹחֵז הַכְּלִי בְּיָדוֹ.

וְכָל זֶה כְּשֶׁנִּפְנָה בַּיּוֹם, אֲבָל אִם נִפְנָה בַּלַּיְלָה אוֹ סָמוּךְ לַחֲשֵׁכָה שֶׁאֵין שְׁהוּת לַהֲנִיחָן עָלָיו עוֹד אַחַר עֲשִׂיַּת צְרָכָיו — לֹא יַעֲשֶׂה בָּהֶן צְרָכָיו אֲפִלּוּ הֵן גְּלוּלִין בְּבִגְדוֹ, וְלֹא יִכָּנֵס בָּהֶן לְבֵית הַכִּסֵּא קָבוּעַ אֲפִלּוּ לְהַשְׁתִּין מְיֻשָּׁב וְהֵן גְּלוּלִין בְּבִגְדוֹ, אֶלָּא חוֹלְצָן בְּרִחוּק אַרְבַּע אַמּוֹת וּמַנִּיחָן בְּכִיסָן אִם יֵשׁ בּוֹ טֶפַח. אֲבָל אִם אֵין בּוֹ טֶפַח — בָּטֵל הוּא לְגַבֵּי הַתְּפִלִּין כֵּיוָן שֶׁהוּא כִּלְיָן. וְאִם מַנִּיחָן בִּכְלִי שֶׁאֵינוֹ כִּלְיָן אֲפִלּוּ אֵין בּוֹ טֶפַח — אֵינוֹ בָּטֵל לְגַבֵּיהֶן. וְאוֹחֵז הַכִּיס אוֹ הַכְּלִי בְּיָדוֹ וְנִכְנָס אֲפִלּוּ לְבֵית הַכִּסֵּא קָבוּעַ לַעֲשׂוֹת צְרָכָיו אוֹ לְהַשְׁתִּין אֲפִלּוּ מְעֻמָּד:

בַּלַּיְלָה. כָּל זֶה בַּיּוֹם ; אֲבָל נִפְנָה בַּלַּיְלָה אוֹ סָמוּךְ לַחֲשֵׁכָה שֶׁאֵין שְׁהוּת לַהֲנִיחָן — חוֹלְצָן בְּרִחוּק ד׳ אַמּוֹת וּמַנִּיחָן בְּכִיס (אִם יֵשׁ בּוֹ טֶפַח) אוֹ בִּכְלִי, וְאוֹחֵז הַכְּלִי בְּיָדוֹ וְנִכְנָס אֲפִלּוּ לְקָבוּעַ.

סעיף ה בֵּית הַכִּסֵּא שֶׁבַּשָּׂדֶה הִתִּירוּ בְּכִיסָן ; אֲבָל שֶׁבַּבַּיִת לֹא יַכְנִיסֵן כְּלָל, שֶׁיָּכוֹל לְהַנִּיחָן בְּמָקוֹם הַמִּשְׁתַּמֵּר.

וְכָל זֶה שֶׁהִתִּירוּ חֲכָמִים לְהַכְנִיס תְּפִלִּין לְבֵית הַכִּסֵּא קָבוּעַ אֲפִלּוּ כְּשֶׁהֵן בְּכִיסָן, הוּא בְּבֵית הַכִּסֵּא שֶׁבַּשָּׂדֶה, שֶׁאִי אֶפְשָׁר לְהַצְנִיעָן חוּץ לְבֵית הַכִּסֵּא שֶׁמָּא יִטְּלוּ אוֹתָן עוֹבְרֵי דְּרָכִים. אֲבָל בְּבֵית הַכִּסֵּא שֶׁבַּבַּיִת — לֹא יַכְנִיסֵן כְּלָל אֲפִלּוּ בְּכִיסָן כֵּיוָן שֶׁיָּכוֹל לְהַנִּיחָן בְּמָקוֹם הַמִּשְׁתַּמֵּר. וְכֵן לֹא יַעֲשֶׂה בָּהֶן צְרָכָיו בַּחֲצֵרוֹ, אֲפִלּוּ שֶׁלֹּא בְּבֵית הַכִּסֵּא אֲפִלּוּ הֵן בְּכִיסָן. אֲבָל מֻתָּר לְהַשְׁתִּין בָּהֶן שֶׁלֹּא בְּבֵית הַכִּסֵּא קָבוּעַ כְּשֶׁהֵן בְּכִיסָן, אֲפִלּוּ הוּא בְּבֵיתוֹ וַחֲצֵרוֹ:

שָׂדֶה וּבַיִת. מַה שֶּׁהִתִּירוּ לְהַכְנִיס בְּכִיסָן לְבֵית הַכִּסֵּא קָבוּעַ — בְּשֶׁבַּשָּׂדֶה, שֶׁאִי אֶפְשָׁר לְהַצְנִיעָן מִפְּנֵי עוֹבְרֵי דְּרָכִים ; אֲבָל שֶׁבַּבַּיִת לֹא יַכְנִיסֵן כְּלָל אֲפִלּוּ בְּכִיסָן, שֶׁיָּכוֹל לְהַנִּיחָן בְּמָקוֹם הַמִּשְׁתַּמֵּר.

סעיף ו שְׁאָר סְפָרִים וְסֵפֶר תּוֹרָה ; הַנִּכְנָס שֶׁלֹּא לַעֲשׂוֹת צְרָכָיו ; וְדִין הַמֶּרְחָץ.

וַאֲפִלּוּ בְּבֵית הַכִּסֵּא שֶׁבַּשָּׂדֶה, לֹא הִתִּירוּ אֶלָּא בִּתְפִלִּין שֶׁאָדָם לוֹבְשָׁן כָּל הַיּוֹם וְהוֹלֵךְ בָּהֶן בַּשָּׂדֶה, וְאִי אֶפְשָׁר לִטְרֹחַ בְּכָל פַּעַם לַהֲשִׁיבָן לְבֵיתוֹ, וְגַם אִי אֶפְשָׁר לְהַצְנִיעָן בַּשָּׂדֶה מִפְּנֵי עוֹבְרֵי דְּרָכִים. אֲבָל שְׁאָר סְפָרִים — אָסוּר לְהַכְנִיסָן אֲפִלּוּ בְּבֵית הַכִּסֵּא שֶׁבַּשָּׂדֶה בְּבִגְדוֹ בְּיָדוֹ לְהַשְׁתִּין מְיֻשָּׁב, אוֹ לַעֲשׂוֹת צְרָכָיו אֲפִלּוּ שֶׁלֹּא בְּבֵית הַכִּסֵּא קָבוּעַ, כִּי יָכוֹל לְהַנִּיחָם בְּבֵיתוֹ. אֲבָל מֻתָּר לְהַכְנִיסָן בְּכִיסָן לְבֵית הַכִּסֵּא קָבוּעַ שֶׁבַּשָּׂדֶה לַעֲשׂוֹת צְרָכָיו, אוֹ לְהַשְׁתִּין אֲפִלּוּ מְעֻמָּד (אֲבָל בְּבֵית הַכִּסֵּא שֶׁבַּבַּיִת, אוֹ לַעֲשׂוֹת בָּהֶן צְרָכָיו בַּחֲצֵרוֹ אֲפִלּוּ שֶׁלֹּא בְּבֵית הַכִּסֵּא קָבוּעַ — דִּינָם כִּתְפִלִּין שֶׁאָסוּר אֲפִלּוּ בְּכִיסָן). וְסֵפֶר תּוֹרָה — אָסוּר לְהַכְנִיסוֹ אֲפִלּוּ בְּכִיסוֹ לְבֵית הַכִּסֵּא קָבוּעַ אֲפִלּוּ שֶׁבַּשָּׂדֶה כְּמוֹ שֶׁיִּתְבָּאֵר בְּיוֹרֶה דֵּעָה סִימָן רפב. וְכָל זֶה כְּשֶׁרוֹצֶה לַעֲשׂוֹת צְרָכָיו גְּדוֹלִים, אוֹ אֲפִלּוּ קְטַנִּים אֶלָּא שֶׁהוּא בֵּית הַכִּסֵּא קָבוּעַ וְיֵשׁ לָחֹש שֶׁמָּא יִפָּנֶה שָׁם לִגְדוֹלִים. אֲבָל אִם נִכְנָס שֶׁלֹּא לַעֲשׂוֹת צְרָכָיו כְּלָל — יֵשׁ מַתִּירִין לְהַכְנִיסָן בְּיָדוֹ אֲפִלּוּ מְגֻלִּין לְבֵית הַכִּסֵּא קָבוּעַ אֲפִלּוּ שֶׁבַּבַּיִת. וְהוּא הַדִּין לִשְׁאָר סְפָרִים. וְיֵשׁ מִי שֶׁאוֹסֵר בְּנִכְנָס לְבֵית הַכִּסֵּא קָבוּעַ שֶׁלֹּא לַעֲשׂוֹת צְרָכָיו כְּמוֹ שֶׁיִּתְבָּאֵר, כְּמוֹ בְּנִכְנָס לַעֲשׂוֹת צְרָכָיו. וְנָכוֹן לָחֹש לִדְבָרָיו. בַּמֶּה דְּבָרִים אֲמוּרִים? לְהַכְנִיסָן בְּיָדָיו, אֲבָל לִכָּנֵס לְבֵית הַכִּסֵּא בִּתְפִלִּין שֶׁבְּרֹאשׁוֹ אוֹ בִּזְרוֹעוֹ שֶׁלֹּא לַעֲשׂוֹת צְרָכָיו כְּלָל אֲפִלּוּ קְטַנִּים — לְדִבְרֵי הַכֹּל אָסוּר בְּכָל עִנְיָן אֲפִלּוּ הֵן מְכֻסִּין עַל רֹאשׁוֹ וּזְרוֹעוֹ, שֶׁהֲרֵי אֲפִלּוּ בְּבֵית הַמֶּרְחָץ אָסְרוּ לִכָּנֵס בִּתְפִלִּין שֶׁעָלָיו. וְלֹא אָמְרוּ שֶׁאֵין אִסּוּר לִכָּנֵס לְבֵית הַכִּסֵּא קָבוּעַ לְהַשְׁתִּין בִּתְפִלִּין שֶׁעָלָיו, אֶלָּא מִשּׁוּם גְּזֵרָה שֶׁמָּא יִפָּנֶה בָּהֶן, אֶלָּא מִשּׁוּם שֶׁלְּהַשְׁתִּין הוּא חַיָּיו שֶׁל אָדָם וְאִי אֶפְשָׁר בִּלְעָדוֹ, וְלֹא הָיוּ מַטְרִיחִין אוֹתוֹ לַחֲלֹץ תְּפִלָּיו בִּשְׁבִיל כָּךְ, מַה שֶּׁאֵין כֵּן כְּשֶׁנִּכְנָס שֶׁלֹּא לְצֹרֶךְ. וְלִכָּנֵס לְמֶרְחָץ אַף שֶׁהוּא צָרְכּוֹ שֶׁל אָדָם, מִכָּל מָקוֹם כֵּיוָן שֶׁמֻּכְרָח הוּא לְחָלְצָן בַּמֶּרְחָץ כְּשֶׁיַּפְשִׁיט כָּל בְּגָדָיו — הִצְרִיכוּהוּ לְחָלְצָן עַל פֶּתַח הַמֶּרְחָץ שֶׁלֹּא יִכָּנֵס בָּהֶן כְּלָל לְבֵית הַמֶּרְחָץ. וּמִטַּעַם זֶה הִצְרִיכוּהוּ לְחָלְצָן בְּרִחוּק אַרְבַּע אַמּוֹת מִבֵּית הַכִּסֵּא קָבוּעַ כְּשֶׁנִּכְנָס לְהַשְׁתִּין, כְּמוֹ שֶׁנִּתְבָּאֵר לְמַעְלָה:

שְׁאָר סְפָרִים וְהַמֶּרְחָץ. שְׁאָר סְפָרִים — אָסוּר לְהַכְנִיסָן (יָכוֹל לְהַנִּיחָן בְּבֵיתוֹ), וְסֵפֶר תּוֹרָה אָסוּר אֲפִלּוּ בְּכִיסוֹ (יורה דעה). הַנִּכְנָס שֶׁלֹּא לַעֲשׂוֹת צְרָכָיו — יֵשׁ מַתִּירִין וְיֵשׁ אוֹסְרִין, וְנָכוֹן לָחֹשׁ. וְלִכָּנֵס בִּתְפִלִּין שֶׁבְּרֹאשׁוֹ שֶׁלֹּא לְצֹרֶךְ — אָסוּר לְדִבְרֵי הַכֹּל, כְּמוֹ בַּמֶּרְחָץ.

סעיף ז צוֹאָה שֶׁאֵינָהּ בְּבֵית הַכִּסֵּא — מֻתָּר לֵילֵךְ אֶצְלָהּ בִּתְפִלִּין ; וְטוֹב לְכַסּוֹת בְּכוֹבַע.

בַּמֶּה דְּבָרִים אֲמוּרִים? לִכָּנֵס לְבֵית הַכִּסֵּא, אֲבָל צוֹאָה שֶׁאֵינָהּ בְּבֵית הַכִּסֵּא — מֻתָּר לֵילֵךְ אֶצְלָהּ בִּתְפִלִּין שֶׁעַל רֹאשׁוֹ וּזְרוֹעוֹ, אִם דֶּרֶךְ הִלּוּכוֹ הוּא שָׁם, וְאֵין צָרִיךְ לְהַרְחִיק מִמֶּנָּה. וְלָכֵן מֻתָּר לֵילֵךְ בִּתְפִלִּין שֶׁעַל רֹאשׁוֹ בִּמְבוֹאוֹת הַמְטֻנָּפוֹת אַף עַל פִּי שֶׁהַצּוֹאָה נִרְאֶה לָעַיִן, וְאֵין צָרִיךְ לְכַסּוֹת הַתְּפִלִּין שֶׁל רֹאשׁ מִן הַדִּין. וְאַף־עַל־פִּי־כֵן טוֹב לְכַסּוֹת בְּכוֹבַע:

צוֹאָה שֶׁחוּץ לְבֵית הַכִּסֵּא. מֻתָּר לֵילֵךְ אֶצְלָהּ בִּתְפִלִּין שֶׁעַל רֹאשׁוֹ אִם דֶּרֶךְ הִלּוּכוֹ שָׁם, וְאֵין צָרִיךְ לְהַרְחִיק. וְכֵן בִּמְבוֹאוֹת הַמְטֻנָּפוֹת, וְאֵין צָרִיךְ לְכַסּוֹת הַשֶּׁל רֹאשׁ מִן הַדִּין — וְטוֹב לְכַסּוֹת בְּכוֹבַע.

סעיף ח מֻתָּר לָרוֹפֵא לִקַּח עָבִיט שֶׁל מֵי רַגְלַיִם אַף שֶׁתְּפִלִּין בְּרֹאשׁוֹ ; וּבַעַל נֶפֶשׁ יַחֲמִיר.

וְכֵן מֻתָּר לָרוֹפֵא לִקַּח עָבִיט שֶׁל מֵי רַגְלַיִם לִבְדֹּק בּוֹ הַחוֹלֶה אַף עַל פִּי שֶׁתְּפִלִּין בְּרֹאשׁוֹ. וּבַעַל נֶפֶשׁ יַחֲמִיר לְעַצְמוֹ:

הָרוֹפֵא וּבַעַל נֶפֶשׁ. מֻתָּר לָרוֹפֵא לִקַּח עָבִיט שֶׁל מֵי רַגְלַיִם לִבְדֹּק בּוֹ הַחוֹלֶה אַף שֶׁתְּפִלִּין בְּרֹאשׁוֹ ; וּבַעַל נֶפֶשׁ יַחֲמִיר לְעַצְמוֹ.

סעיף ט שָׁכַח וְעָשָׂה צְרָכָיו — מַנִּיחַ יָדוֹ עֲלֵיהֶן עַד גְּמַר עַמּוּד הָרִאשׁוֹן, וְיוֹצֵא וְחוֹלֵץ.

אִם שָׁכַח תְּפִלִּין בְּרֹאשׁוֹ וְעָשָׂה בָּהֶן צְרָכָיו — מַנִּיחַ יָדוֹ עֲלֵיהֶן לְכַסּוֹתָן עַד שֶׁיִּגְמֹר עַמּוּד הָרִאשׁוֹן, שֶׁאִם יַפְסִיק מִיָּד יָבוֹא לִידֵי הִידְרוֹקָן, וְיוֹצֵא וְחוֹלֵץ שֶׁל יָד וְשֶׁל רֹאשׁ וְחוֹזֵר וְנִכְנָס:

הַשּׁוֹכֵחַ. שָׁכַח תְּפִלִּין בְּרֹאשׁוֹ וְעָשָׂה בָּהֶן צְרָכָיו — מַנִּיחַ יָדוֹ עֲלֵיהֶן לְכַסּוֹתָן עַד גְּמַר עַמּוּד הָרִאשׁוֹן (שֶׁאִם יַפְסִיק מִיָּד יָבוֹא לִידֵי הִידְרוֹקָן), וְיוֹצֵא וְחוֹלֵץ שֶׁל יָד וְשֶׁל רֹאשׁ וְחוֹזֵר.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim מ״ג (מהדורת קה״ת, ט׳ סְעִיפִים). ביאור: DAAT. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.

2 · היסוד

היסוד — הַנְהָגַת הַתְּפִלִּין בְּבֵית הַכִּסֵּא

היסוד: ההנהגה בתפילין בבית הכסא

קריאת אדמו״ר הזקן על סימן זה בשלושה מהלכים. תחילה כְּבוֹד הַתְּפִלִּין וְהַנִּיצוֹצוֹת: אֵין נִכְנָסִים לְהַשְׁתִּין בִּתְפִלִּין בְּבֵית הַכִּסֵּא קָבוּעַ (גְּזֵרָה שֶׁמָּא יַעֲשֶׂה בָּהֶן צְרָכָיו), וּמְעֻמָּד חוֹשְׁשִׁים לְשִׁפְשׁוּף הַנִּיצוֹצוֹת. אַחַר כָּךְ גֶּדֶר קָבוּעַ וַעֲרַאי וּבֵית הַכִּסֵּא שֶׁבַּבַּיִת, הַנִּדּוֹן עַל פִּי הַמָּקוֹם הַמִּשְׁתַּמֵּר. וּלְבַסּוֹף הַחֲלִיצָה בְּרִחוּק ד׳ אַמּוֹת וְהַגְּלִילָה בָּרְצוּעוֹת, דִּין הַלַּיְלָה, הַשּׁוֹכֵחַ, הָרוֹפֵא וְהַבַּעַל נֶפֶשׁ.

כְּבוֹד הַתְּפִלִּין וְהַנִּיצוֹצוֹת

כבוד התפילין והניצוצות

היסוד. הַתְּפִלִּין צְרִיכוֹת גּוּף נָקִי ; לִכָּנֵס לְהַשְׁתִּין בְּבֵית הַכִּסֵּא קָבוּעַ בִּתְפִלִּין שֶׁבְּרֹאשׁוֹ אָסוּר (גְּזֵרָה שֶׁמָּא יַעֲשֶׂה בָּהֶן צְרָכָיו). מְעֻמָּד — חוֹשְׁשִׁים לְשִׁפְשׁוּף הַנִּיצוֹצוֹת.

התוצאה. מְיֻשָּׁב אוֹ בַּעֲפַר תִּיחוּחַ — אֵין חֲשַׁשׁ נִיצוֹצוֹת ; וּבַכִּיס הַנֶּאֱחָז בְּיָדוֹ מֻתָּר לְדִבְרֵי הַכֹּל. וְלִכְבוֹד תַּשְׁמִישֵׁי קְדֻשָּׁה עַיֵּן סימן 42 וְסימן 44.

קָבוּעַ וַעֲרַאי וּבֵית הַכִּסֵּא שֶׁבַּבַּיִת

קבוע ועראי ובית הכסא שבבית

העיקרון. בֵּית הַכִּסֵּא קָבוּעַ (שֶׁרְגִילִים לִפָּנוֹת בּוֹ לִגְדוֹלִים) חָמוּר יוֹתֵר ; וּבַעֲרַאי מֻתָּר לְהַשְׁתִּין. הַחֲלוּקָה הַזֹּאת שׁוֹלֶטֶת בְּכָל הַסִּימָן.

ההרחבה. בְּבֵית הַכִּסֵּא שֶׁבַּשָּׂדֶה הִתִּירוּ בְּכִיסָן (מִפְּנֵי עוֹבְרֵי דְּרָכִים) ; אֲבָל בַּבַּיִת לֹא כְּלָל, שֶׁיֵּשׁ מָקוֹם הַמִּשְׁתַּמֵּר. וְסֵפֶר תּוֹרָה חָמוּר יוֹתֵר (יורה דעה).

חֲלִיצָה בְּרִחוּק ד׳ אַמּוֹת וְהַשּׁוֹכֵחַ

חליצה ברחוק ד׳ אמות והשוכח

העיקרון. לַעֲשׂוֹת צְרָכָיו בְּקָבוּעַ — חוֹלֵץ בְּרִחוּק ד׳ אַמּוֹת, גּוֹלְלָן בָּרְצוּעוֹת (קְדֻשַּׁת הַשַּׁדַּי), וְאוֹחֲזָן בִּימִינוֹ כְּנֶגֶד לִבּוֹ.

ההנהגה. בַּלַּיְלָה מַנִּיחָן בִּכְלִי ; הַשּׁוֹכֵחַ מַנִּיחַ יָדוֹ עַד גְּמַר עַמּוּד הָרִאשׁוֹן וְיוֹצֵא ; הָרוֹפֵא מֻתָּר בְּעָבִיט, וְהַבַּעַל נֶפֶשׁ יַחֲמִיר לְעַצְמוֹ.

הפניה ללימוד. משקל היסוד שהונח כאן — הכָּבוֹד לתַשְׁמִישֵׁי קְדֻשָּׁה, חֲשַׁשׁ הַנִּיצוֹצוֹת והגּוּף נָקִי — נוגע גם בהנהגה (השתנה, עשיית צרכים, החֲלִיצָה) וגם ברגישות חב״ד על הקְדֻשָּׁה שסביב המצווה. דף זה מציג זאת ברמת העיקרון המתועד בטקסט של הרב, בלי לצטט פרק או נוסח מדויק, ובלי להוסיף דבר מומצא במשמעות הרוחנית. ללימוד מעמיק של ממד זה — ולהנהגה אישית בקָבוּעַ וַעֲרַאי, בחֲלִיצָה ובשוכח — יש ללמוד עם רב של חב״ד ובתוך המקורות עצמם.

אֵיךְ לְהִתְנַהֵג בַּתְּפִלִּין בְּהִכָּנְסוֹ לְבֵית הַכִּסֵּא — בעיקרון · ללימוד, פנה לרב של חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרכו של אדמו״ר הזקן עם המחבר, הרמ״א (רמ״א) והמשנה ברורה (משנה ברורה) בנקודות המפתח של סימן מ״ג. שורת שולחן ערוך הרב (שולחן ערוך הרב) מובלטת — היא חושפת את רגישות הרב כפי שהיא עולה מטקסטו (§1, ט׳ סעיפים). הרב מפרש את הטעם ואת הפרט במקום שהמחבר מסתפק באמירה.

נושא מחבר רמ״א משנה ברורה דרך אדמו״ר הזקן (שולחן ערוך הרב)
נִיצוֹצוֹת וּכְבוֹד הַתְּפִלִּין או״ח מ״ג:א — אָסוּר לִכָּנֵס לְקָבוּעַ לְהַשְׁתִּין בִּתְפִלִּין ; מְיֻשָּׁב מֻתָּר. או״ח מ״ג:א — דַּוְקָא מְיֻשָּׁב אֵין לָחֹשׁ ; מְעֻמָּד אָסוּר (נִיצוֹצוֹת). מ״ב מ״ג / ברכות כ״ג ע״א — טַעַם הַגְּזֵרָה ; שִׁפְשׁוּף נִיצוֹצוֹת. דרך אדמו״ר הזקן — אָסוּר בְּקָבוּעַ, וּבַעֲרַאי מֻתָּר ; אוֹחֲזָן בְּיָדוֹ מְעֻמָּד אָסוּר שֶׁמָּא יְשַׁפְשֵׁף ; וּבְכִיסָן מֻתָּר (סעיפים א-ב).
קָבוּעַ וַעֲרַאי וּבֵית הַכִּסֵּא שֶׁבַּבַּיִת או״ח מ״ג — בֵּית הַכִּסֵּא שֶׁבַּבַּיִת לֹא יַכְנִיסֵן, שֶׁיֵּשׁ מָקוֹם הַמִּשְׁתַּמֵּר. — מ״ב מ״ג — גֶּדֶר קָבוּעַ וַעֲרַאי ; שָׂדֶה וּבַיִת. דרך אדמו״ר הזקן — בַּשָּׂדֶה הִתִּירוּ בְּכִיסָן ; בַּבַּיִת לֹא כְּלָל ; וְסֵפֶר תּוֹרָה חָמוּר (יורה דעה) (סעיפים ד-ו).
חֲלִיצָה וּגְלִילָה בָּרְצוּעוֹת או״ח מ״ג:ה — חוֹלֵץ בְּרִחוּק ד׳ אַמּוֹת ; אוֹחֵז בִּימִינוֹ. או״ח מ״ג:ה — יְגַלְגְּלֵם בָּרְצוּעוֹת שֶׁלֹּא תֵּצֵא רְצוּעָה טֶפַח. מ״ב מ״ג / רב האי גאון — הַגְּלִילָה ; קְדֻשַּׁת הָרְצוּעוֹת. דרך אדמו״ר הזקן — אוֹחֲזָן בִּימִינוֹ כְּנֶגֶד לִבּוֹ ; גּוֹלְלָן בָּרְצוּעוֹת ; בַּלַּיְלָה בִּכְלִי (סעיפים ג-ד).
הַשּׁוֹכֵחַ, הָרוֹפֵא וּבַעַל נֶפֶשׁ או״ח מ״ג:ח-ט — שָׁכַח — מַנִּיחַ יָדוֹ עַד עַמּוּד הָרִאשׁוֹן ; הָרוֹפֵא וְעָבִיט. — מ״ב מ״ג / ברכות כ״ג ע״א — הִידְרוֹקָן ; בַּעַל נֶפֶשׁ. דרך אדמו״ר הזקן — מַנִּיחַ יָדוֹ עַד גְּמַר עַמּוּד הָרִאשׁוֹן וְיוֹצֵא וְחוֹלֵץ ; רוֹפֵא מֻתָּר וּבַעַל נֶפֶשׁ יַחֲמִיר (סעיפים ח-ט).

הטבלה נערכה מטקסט המחבר (אורח חיים מ״ג, ט׳ סעיפים, מקור ספריא) ומנושאי הכלים הניתנים לאימות (בית יוסף, טור, רמב״ם הלכות תפילין ד, רא״ש, נמוקי יוסף, מגדל עוז, רב האי גאון, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ג ע״א–ע״ב). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ט׳ סְעִיפִים) המובא ב§1. ייחוסי סעיף קטן שאינם ודאיים נותרו « על דרך הסברה ».

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה — מעבר ליישום הפשוט של הכלל. כל אחת בנויה: עיקרון הסימן → קריאת הרב → תוצאה מעשית. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — כְּבוֹד הַתְּפִלִּין וְהַנִּיצוֹצוֹת

כבוד התפילין והניצוצות

עיקרון הסימן: הַשְׁתָּנָה בְּבֵית הַכִּסֵּא קָבוּעַ אוֹ עֲרַאי (סעיפים א-ג).

קריאת הרב: הרב מבחין בין הקָבוּעַ (אָסוּר שֶׁבְּרֹאשׁוֹ, גְּזֵרָה) לָעֲרַאי ; מְעֻמָּד חוֹשְׁשִׁים לַנִּיצוֹצוֹת, מְיֻשָּׁב אֵין ; וּבַכִּיס מֻתָּר לְדִבְרֵי הַכֹּל.

תוצאה מעשית: לֹא לִכָּנֵס לְהַשְׁתִּין בִּתְפִלִּין בְּקָבוּעַ ; מְיֻשָּׁב אֵין חֲשַׁשׁ ; לֹא לְאָחֳזָן מְגֻלִּין בְּיָדוֹ מְעֻמָּד.

קו ב — קָבוּעַ וַעֲרַאי וּבֵית הַכִּסֵּא שֶׁבַּבַּיִת

קבוע ועראי והבית

עיקרון הסימן: הַחֲלִיצָה, הַשָּׂדֶה וְהַבַּיִת (סעיפים ד-ז).

קריאת הרב: הַקְּרִיטֶרְיוֹן הוּא הַמָּקוֹם הַמִּשְׁתַּמֵּר — בַּשָּׂדֶה מַתִּירִים בְּכִיסָן, בַּבַּיִת לֹא כְּלָל ; בַּלַּיְלָה מַנִּיחָן בִּכְלִי ; וְסֵפֶר תּוֹרָה חָמוּר (יורה דעה).

תוצאה מעשית: בַּבַּיִת לֹא לְהַכְנִיס תְּפִלִּין לְבֵית הַכִּסֵּא כְּלָל ; לְהַנִּיחָן בְּמָקוֹם הַמִּשְׁתַּמֵּר.

קו ג — הַשּׁוֹכֵחַ, הָרוֹפֵא וּבַעַל נֶפֶשׁ

השוכח, הרופא ובעל נפש

עיקרון הסימן: הַשּׁוֹכֵחַ, הָרוֹפֵא וְהַבַּעַל נֶפֶשׁ (סעיפים ח-ט).

קריאת הרב: הַשּׁוֹכֵחַ מַנִּיחַ יָדוֹ עַד גְּמַר עַמּוּד הָרִאשׁוֹן (חֲשַׁשׁ הִידְרוֹקָן) וְאָז חוֹלֵץ ; הָרוֹפֵא מֻתָּר בְּעָבִיט, אֲבָל הַבַּעַל נֶפֶשׁ יַחֲמִיר.

תוצאה מעשית: לֹא לְהַפְסִיק פִּתְאוֹם אֶלָּא לְכַסּוֹת בַּיָּד ; לִסְמֹךְ עַל הָרַב לְמַעֲשֶׂה, וּלְהַחֲמִיר לְעַצְמוֹ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן מ״ג ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, בהפניה לרב לפרטי המקרים (השתנה, עשיית צרכים, הלילה, השכחה).

לחסיד חב״ד — סימן מ״ג (אֵיךְ לְהִתְנַהֵג בַּתְּפִלִּין בְּהִכָּנְסוֹ לְבֵית הַכִּסֵּא)

  • ① כבוד וניצוצות. לֹא לִכָּנֵס לְהַשְׁתִּין בִּתְפִלִּין בְּבֵית הַכִּסֵּא קָבוּעַ ; מְיֻשָּׁב אֵין חֲשַׁשׁ נִיצוֹצוֹת, מְעֻמָּד אָסוּר בְּאוֹחֲזָן מְגֻלִּין בְּיָדוֹ. יסוד: שוע״ר מ״ג סעיף א-ג.
  • ② קָבוּעַ / עֲרַאי. לְהַבְחִין בֵּין הַמָּקוֹם הַקָּבוּעַ לָעֲרַאי ; בְּבֵית הַכִּסֵּא שֶׁבַּבַּיִת לֹא לְהַכְנִיס כְּלָל אֲפִלּוּ בְּכִיסָן (מָקוֹם הַמִּשְׁתַּמֵּר). יסוד: שוע״ר מ״ג סעיף ד-ה.
  • ③ חֲלִיצָה וּגְלִילָה. לַעֲשׂוֹת צְרָכָיו — לַחֲלֹץ בְּרִחוּק ד׳ אַמּוֹת, לְגָלְלָן בָּרְצוּעוֹת (שֶׁלֹּא תֵּצֵא רְצוּעָה טֶפַח), לְאָחֳזָן בִּימִינוֹ כְּנֶגֶד לִבּוֹ ; בַּלַּיְלָה בִּכְלִי. יסוד: שוע״ר מ״ג סעיף ג-ד.
  • ④ שְׁאָר סְפָרִים וְהַמֶּרְחָץ. סֵפֶר תּוֹרָה חָמוּר יוֹתֵר (אָסוּר אֲפִלּוּ בְּכִיסוֹ, יורה דעה) ; לֹא לִכָּנֵס בִּתְפִלִּין שֶׁלֹּא לְצֹרֶךְ, כְּמוֹ בַּמֶּרְחָץ. יסוד: שוע״ר מ״ג סעיף ו-ז.
  • ⑤ שָׁכַח, רוֹפֵא וּבַעַל נֶפֶשׁ. הַשּׁוֹכֵחַ מַנִּיחַ יָדוֹ עַל הַתְּפִלִּין עַד גְּמַר עַמּוּד הָרִאשׁוֹן וְיוֹצֵא וְחוֹלֵץ ; הָרוֹפֵא מֻתָּר, וְהַבַּעַל נֶפֶשׁ יַחֲמִיר לְעַצְמוֹ. יסוד: שוע״ר מ״ג סעיף ח-ט.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן מ״ג (הטקסט האמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — קבוע ועראי, החליצה, הלילה, השכחה — יש להתייעץ עם רב, ולחב״ד עם רב של חב״ד.
