דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן פ״א · דִּין צוֹאַת קָטָן

דרכו של אדמו״ר הזקן בדִּין צוֹאַת קָטָן : שִׁעוּר כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס — מַרְחִיקִין כְּמוֹ מִשֶּׁל גָּדוֹל ; הַקָּטָן הַחַלָּשׁ וְהַחֲזָקָה — טוֹב וְיָשָׁר לִהְיוֹת נָקִי אֲפִלּוּ מִצּוֹאַת תִּינוֹק בֶּן יוֹמוֹ ; וְצוֹאָה כְּנֶגְדּוֹ בִּתְפִלָּה — יֵלֵךְ וְיַרְחִיקֶנָּה ד׳ אַמּוֹת, וְחוֹזֵר לְמָקוֹם שֶׁפָּסַק — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן פ״א — דִּין צוֹאַת קָטָן. אצל הרב הסימן נפרש לג׳ סְעִיפִים : הַשִּׁעוּר שֶׁל צוֹאַת קָטָן (כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס), דִּין הַקָּטָן הַחַלָּשׁ עִם הַחֲזָקָה וְהַטּוֹב וְיָשָׁר, וְדִין רָאָה צוֹאָה כְּנֶגְדּוֹ בִּתְפִלָּה וּבִקְרִיאַת שְׁמַע.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן פ״א — ג׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim פ״א). אדמו״ר הזקן דן בו בְּדִין צוֹאַת קָטָן. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן פ״א — דִּין צוֹאַת קָטָן

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן פ״א — דִּין צוֹאַת קָטָן — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.81), ואחריו תרגומנו. הסימן כולל ג׳ סעיפים (סעיף k → ‎.81.k).

סעיף א קָטָן שֶׁיָּכוֹל לֶאֱכֹל כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס — מַרְחִיקִין כְּמוֹ מִשֶּׁל גָּדוֹל.

דִּין צוֹאַת קָטָן וּבוֹ ג׳ סְעִיפִים:קָטָן שֶׁיָּכוֹל לֶאֱכֹל כְּזַיִת דָּגָן שֶׁהֵם ה׳ מִינֵי תְּבוּאָה שֶׁצּוֹאַת מִינִים אֵלּוּ מַסְרַחַת, אֲפִלּוּ אֵינוֹ יָכוֹל לְאָכְלוֹ אֶלָּא עַל יְדֵי בִּשּׁוּל כְּגוֹן דַּייְסָא וְכַיּוֹצֵא בָּהּ, אִם אֵין צָרִיךְ לִשְׁהוֹת מִתְּחִלַּת כְּזַיִת זֶה עַד סוֹפוֹ יוֹתֵר מִמַּה שֶּׁגָּדוֹל בֵּינוֹנִי שׁוֹהֶה בַּאֲכִילַת פְּרָס מִמַּאֲכָל זֶה — צָרִיךְ לְהַרְחִיק מִצּוֹאָתוֹ וּמִמֵּי רַגְלָיו כְּמוֹ מִשֶּׁל גָּדוֹל. אֲבָל אִם צָרִיךְ לִשְׁהוֹת יוֹתֵר מִכַּאן — הֲרֵי זֶה כְּאוֹכֵל עַכְשָׁו חֲצִי זַיִת וּלְמָחָר חֲצִי זַיִת, שֶׁכֵּן הֲלָכָה לְמֹשֶׁה מִסִּינַי שֶׁאֵין אֲכִילָה מִצְטָרֶפֶת לִשְׁהִיָּה אֲרֻכָּה כָּזוֹ. וְשִׁעוּר כַּזַּיִת יִתְבָּאֵר בְּסִימָן תפ״ו, וְשִׁעוּר פְּרָס בְּסִימָן תרי״ב:

שִׁעוּר צוֹאַת קָטָן. קָטָן הַיָּכוֹל לֶאֱכֹל כְּזַיִת דָּגָן (ה׳ מִינֵי תְּבוּאָה שֶׁצּוֹאָתָם מַסְרַחַת), אֲפִלּוּ רַק עַל יְדֵי בִּשּׁוּל (דַּייְסָא) : אִם אֵינוֹ צָרִיךְ לִשְׁהוֹת בִּכְזַיִת זֶה יוֹתֵר מִשִּׁעוּר אֲכִילַת פְּרָס שֶׁל גָּדוֹל בֵּינוֹנִי — צָרִיךְ לְהַרְחִיק מִצּוֹאָתוֹ וּמִמֵּי רַגְלָיו כְּמוֹ מִשֶּׁל גָּדוֹל. אֲבָל אִם צָרִיךְ יוֹתֵר — הֲרֵי זֶה כַּחֲצִי זַיִת עַכְשָׁו וַחֲצִי לְמָחָר, שֶׁהֲלָכָה לְמֹשֶׁה מִסִּינַי אֵין אֲכִילָה מִצְטָרֶפֶת לִשְׁהִיָּה אֲרֻכָּה. שִׁעוּר כְּזַיִת → סִימָן תפ״ו, שִׁעוּר פְּרָס → סִימָן תרי״ב.

סעיף ב הַקָּטָן הַחַלָּשׁ — אֵין צָרִיךְ לְהַרְחִיק ; טוֹב וְיָשָׁר נָקִי אֲפִלּוּ מִצּוֹאַת תִּינוֹק בֶּן יוֹמוֹ.

וְאִם הַקָּטָן הוּא חַלָּשׁ וְאֵינוֹ יָכוֹל לֶאֱכֹל כְּזַּיִת בִּכְדֵי אֲכִילַת פְּרָס, אַף עַל פִּי שֶׁהִגִּיעַ לְשָׁנִים שֶׁאַחֵר כַּיּוֹצֵא בּוֹ יָכוֹל לֶאֱכֹל — אֵין צָרִיךְ לְהַרְחִיק מִצּוֹאָתוֹ וּמִמֵּי רַגְלָיו אֲפִלּוּ הוּא בֵּן ג׳ וד׳ שָׁנִים (וְהוּא שֶׁעֲדַיִן לֹא הָיָה יָכוֹל לֶאֱכֹל מִיָּמָיו כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס). אֲבָל אִם הָיָה יָכוֹל לֶאֱכֹל פַּעַם אַחַת וְאַחַר כָּךְ נֶחֱלַשׁ וְאֵינוֹ יָכוֹל — דִּינוֹ כְּגָדוֹל שֶׁצָּרִיךְ לְהַרְחִיק מִצּוֹאָתוֹ וּמִמֵּי רַגְלָיו (וַאֲפִלּוּ אֵינוֹ יָכוֹל). וַאֲפִלּוּ לֹא אָכַל דָּגָן מִיָּמִים רַבִּים, וַאֲפִלּוּ לֹא אָכַל מֵעוֹלָם, שֶׁאַף שֶׁצּוֹאָה זוֹ אֵינָהּ מִדָּגָן מִכָּל מָקוֹם (כָּל שֶׁהָיָה יָכוֹל לֶאֱכֹל כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס) — בְּנֵי מֵעַיִם מַסְרִיחִים כָּל מִינֵי מַאֲכָל שֶׁיֹּאכַל. וְכָל זֶה מִן הַדִּין אֲבָל טוֹב וְיָשָׁר לִקְרוֹת וּלְהִתְפַּלֵּל בְּמָקוֹם שֶׁהוּא נָקִי לְגַמְרֵי אֲפִלּוּ מִצּוֹאַת תִּינוֹק בֶּן יוֹמוֹ (אִם אֶפְשָׁר בְּקַּל, אַף עַל פִּי שֶׁאֵין אִסּוּר בַּדָּבָר):

הַקָּטָן הַחַלָּשׁ. קָטָן חַלָּשׁ שֶׁאֵינוֹ יָכוֹל לֶאֱכֹל כְּזַיִת בִּכְדֵי אֲכִילַת פְּרָס — אַף שֶׁהִגִּיעַ לְשָׁנִים שֶׁאֲחֵרִים כַּיּוֹצֵא בּוֹ יְכוֹלִים — אֵין צָרִיךְ לְהַרְחִיק, אֲפִלּוּ בֵּן ג׳ וד׳ שָׁנִים (וּבִלְבַד שֶׁלֹּא הָיָה יָכוֹל מֵעוֹלָם). אֲבָל אִם הָיָה יָכוֹל פַּעַם אַחַת וְנֶחֱלַשׁ — דִּינוֹ כְּגָדוֹל (וַאֲפִלּוּ אֵינוֹ יָכוֹל עַכְשָׁו). כִּי כָּל שֶׁהָיָה יָכוֹל — בְּנֵי מֵעַיִם מַסְרִיחִים כָּל מַאֲכָל. וְכָל זֶה מִן הַדִּין ; אֲבָל טוֹב וְיָשָׁר לִקְרוֹת וּלְהִתְפַּלֵּל בְּמָקוֹם נָקִי לְגַמְרֵי אֲפִלּוּ מִצּוֹאַת תִּינוֹק בֶּן יוֹמוֹ (אִם אֶפְשָׁר בְּקַל, אַף שֶׁאֵין אִסּוּר).

סעיף ג רָאָה צוֹאָה כְּנֶגְדּוֹ בִּתְפִלָּה — יֵלֵךְ וְיַרְחִיקֶנָּה ד׳ אַמּוֹת, וְחוֹזֵר לְמָקוֹם שֶׁפָּסַק.

אִם הָיָה עוֹמֵד בְּתְפִלָּה וְרָאָה צוֹאָה כְּנֶגְדּוֹ בָּרִחוּק ד׳ אַמּוֹת (וְאֵינוֹ יָכוֹל לַחֲזֹר פָּנָיו מִמֶּנָּה מִפְּנֵי שֶׁצָּרִיךְ לְהַחֲזִיר פָּנָיו כְּנֶגֶד בֵּית קָדְשֵׁי קָדָשִׁים) — יֵלֵךְ לְפָנָיו כְּדֵי שֶׁיִּזְרְקֶנָּה מֵאַחֲרָיו ד׳ אַמּוֹת. וְאִם אִי אֶפְשָׁר כְּגוֹן שֶׁיֵּשׁ לְפָנָיו נָהָר אוֹ דָּבָר אַחֵר הַמְעַכֵּב — יֵלֵךְ בִּכְדֵי שֶׁיַּנִּיחֶנָּה לִצְדָדִים ד׳ אַמּוֹת. וְכֵן בִּקְרִיאַת שְׁמַע, אִם אִי אֶפְשָׁר לוֹ. וְאֵינוֹ צָרִיךְ לַחֲזֹר אֶלָּא לְמָקוֹם שֶׁפָּסַק וְלֹא לְרֹאשׁ הַתְּפִלָּה אוֹ הַקְּרִיאַת שְׁמַע, אַף עַל פִּי שֶׁהִתְפַּלֵּל וְקָרָא כְּנֶגֶד הַצּוֹאָה, לְפִי שֶׁאֵין זֶה ״זֶבַח רְשָׁעִים תּוֹעֵבָה״, כֵּיוָן שֶׁלֹּא הָיְתָה ד׳ אַמּוֹת סְבִיבוֹ, שֶׁבָּהֶן לְבַדָּם הוּא שֶׁמְּחֻיָּב לִבְדֹּק שֶׁלֹּא תְּהֵא בָּהֶם צוֹאָה אִם הוּא מָקוֹם שֶׁרָאוּי לְהִסְתַּפֵּק בְּצוֹאָה, וְאִם לֹא בָּדַק בָּהֶן פּוֹשֵׁעַ הוּא וּתְפִלָּתוֹ תּוֹעֵבָה, אֲבָל לֹא הִטְרִיחוּ לִבְדֹּק לְפָנָיו כִּמְלֹא עֵינָיו אַף עַל פִּי שֶׁהוּא בְּמָקוֹם הָרָאוּי לְהִסְתַּפֵּק בּוֹ בְּצוֹאָה:

צוֹאָה כְּנֶגְדּוֹ בִּתְפִלָּה. עוֹמֵד בִּתְפִלָּה וְרָאָה צוֹאָה כְּנֶגְדּוֹ בְּרִחוּק ד׳ אַמּוֹת (וְאֵינוֹ יָכוֹל לְהַחֲזִיר פָּנָיו מִמֶּנָּה, שֶׁצָּרִיךְ לְהַחֲזִיר כְּנֶגֶד בֵּית קָדְשֵׁי קָדָשִׁים) — יֵלֵךְ לְפָנָיו שֶׁיִּזְרְקֶנָּה מֵאַחֲרָיו ד׳ אַמּוֹת ; וְאִם אִי אֶפְשָׁר (נָהָר אוֹ מְעַכֵּב) — לִצְדָדִים ד׳ אַמּוֹת, וְכֵן בִּקְרִיאַת שְׁמַע. וְאֵינוֹ חוֹזֵר אֶלָּא לְמָקוֹם שֶׁפָּסַק, לֹא לָרֹאשׁ, שֶׁאֵין זֶה ״זֶבַח רְשָׁעִים תּוֹעֵבָה״ — לֹא הָיְתָה בְּד׳ אַמּוֹת סְבִיבוֹ שֶׁחַיָּב לִבְדֹּק ; וְלֹא הִטְרִיחוּהוּ לִבְדֹּק לְפָנָיו כִּמְלֹא עֵינָיו.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim פ״א (מהדורת קה״ת, ג׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין צוֹאַת קָטָן

היסוד : דִּין צוֹאַת קָטָן

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים. תחילה הַשִּׁעוּר שֶׁל צוֹאַת קָטָן — סַף כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס. אחר כך דִּין הַקָּטָן הַחַלָּשׁ, הַחֲזָקָה שֶׁל קָטָן שֶׁהָיָה יָכוֹל, וְהַטּוֹב וְיָשָׁר. ולבסוף דִּין רָאָה צוֹאָה כְּנֶגְדּוֹ בִּתְפִלָּה וְהַחוֹזֵר לְמָקוֹם שֶׁפָּסַק.

שִׁעוּר צוֹאַת קָטָן — כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס

שִׁעוּר צוֹאַת קָטָן

היסוד. קָטָן הַיָּכוֹל לֶאֱכֹל כְּזַיִת דָּגָן (ה׳ מִינֵי תְּבוּאָה שֶׁצּוֹאָתָם מַסְרַחַת) בִּכְדֵי אֲכִילַת פְּרָס — מַרְחִיקִין מִצּוֹאָתוֹ וּמִמֵּי רַגְלָיו כְּמוֹ מִשֶּׁל גָּדוֹל (סעיף א).

המסקנה. יוֹתֵר מִשִּׁעוּר זֶה שֶׁל זְמַן — אֵין אֲכִילָה מִצְטָרֶפֶת (הֲלָכָה לְמֹשֶׁה מִסִּינַי). עיין סימן 79 (צוֹאַת בְּהֵמָה).

הַקָּטָן הַחַלָּשׁ וְהַחֲזָקָה

הַקָּטָן הַחַלָּשׁ וְהַחֲזָקָה

היסוד. קָטָן חַלָּשׁ שֶׁאֵינוֹ יָכוֹל כְּזַיִת בִּכְדֵי אֲכִילַת פְּרָס — אֵין צָרִיךְ לְהַרְחִיק אֲפִלּוּ בֵּן ג׳ וד׳ שָׁנִים ; אֲבָל הָיָה יָכוֹל פַּעַם — דִּינוֹ כְּגָדוֹל (סעיף ב).

המסקנה. כָּל שֶׁהָיָה יָכוֹל — בְּנֵי מֵעַיִם מַסְרִיחִים כָּל מַאֲכָל ; וְטוֹב וְיָשָׁר לִהְיוֹת נָקִי לְגַמְרֵי אֲפִלּוּ מִצּוֹאַת תִּינוֹק בֶּן יוֹמוֹ. עיין סימן 80 (הַרְחָקָה מִן הַצּוֹאָה).

צוֹאָה כְּנֶגְדּוֹ בִּתְפִלָּה וּבִקְרִיאַת שְׁמַע

צוֹאָה כְּנֶגְדּוֹ בִּתְפִלָּה

היסוד. רָאָה צוֹאָה כְּנֶגְדּוֹ בְּרִחוּק ד׳ אַמּוֹת — יֵלֵךְ לְפָנָיו שֶׁיִּזְרְקֶנָּה מֵאַחֲרָיו ד׳ אַמּוֹת ; וְאִם אִי אֶפְשָׁר — לִצְדָדִים ד׳ אַמּוֹת (סעיף ג).

המסקנה. חוֹזֵר לְמָקוֹם שֶׁפָּסַק, לֹא לָרֹאשׁ, שֶׁאֵין זֶה ״זֶבַח רְשָׁעִים תּוֹעֵבָה״ — לֹא הָיְתָה בְּד׳ אַמּוֹת סְבִיבוֹ. עיין סימן 82 (צוֹאָה בַּבַּיִת).

הפניה ללימוד. היקף הסימן — שִׁעוּר הַהַרְחָקָה, הַחֲזָקָה שֶׁל הַקָּטָן וְדִין רָאָה צוֹאָה כְּנֶגְדּוֹ — נוגע להנהגת הַמָּקוֹם הַנָּקִי לִקְרִיאַת שְׁמַע וְלִתְפִלָּה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן פ״א. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מג׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
שִׁעוּר צוֹאַת קָטָן או״ח פ״א — קָטָן שֶׁהִגִּיעַ לְכְלָל שֶׁאַחֵר כַּיּוֹצֵא בּוֹ יָכוֹל לֶאֱכֹל כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס — מַרְחִיקִין. מ״ב פ״א — פְּרָטֵי הַשִּׁעוּר וְטַעַם ה׳ מִינֵי תְּבוּאָה הַמַּסְרִיחִים. ערוה״ש פ״א — גֶּדֶר בִּכְדֵי אֲכִילַת פְּרָס בַּקָּטָן. דרך אדמו״ר הזקן — כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס, אֲפִלּוּ עַל יְדֵי בִּשּׁוּל (דַּייְסָא) — מַרְחִיקִין כְּמוֹ מִשֶּׁל גָּדוֹל ; הֲלָכָה לְמֹשֶׁה מִסִּינַי אֵין אֲכִילָה מִצְטָרֶפֶת (סעיף א).
הַקָּטָן הַחַלָּשׁ וְהַחֲזָקָה או״ח פ״א — סְתַם הַדִּין בַּקָּטָן שֶׁהִגִּיעַ לַכְּלָל. מ״ב פ״א — דִּין הַחַלָּשׁ וְהַהֶרְגֵּל בַּאֲכִילָה. ערוה״ש פ״א — סְבָרַת בְּנֵי מֵעַיִם מַסְרִיחִים. דרך אדמו״ר הזקן — חַלָּשׁ שֶׁלֹּא אָכַל מֵעוֹלָם — אֵין מַרְחִיקִין ; אָכַל פַּעַם וְנֶחֱלַשׁ — דִּינוֹ כְּגָדוֹל ; טוֹב וְיָשָׁר נָקִי אֲפִלּוּ מִצּוֹאַת תִּינוֹק בֶּן יוֹמוֹ (סעיף ב).
צוֹאָה כְּנֶגְדּוֹ בִּתְפִלָּה או״ח פ״א — הָיָה קוֹרֵא וְרָאָה צוֹאָה כְּנֶגְדּוֹ — יֵלֵךְ שֶׁיִּזְרְקֶנָּה מֵאַחֲרָיו ד׳ אַמּוֹת ; חוֹזֵר לְמָקוֹם שֶׁפָּסַק. מ״ב פ״א — דִּין הַצְּדָדִים וּבְדִיקַת ד׳ אַמּוֹת סְבִיבוֹ. ערוה״ש פ״א — גֶּדֶר ״זֶבַח רְשָׁעִים תּוֹעֵבָה״ בַּצּוֹאָה. דרך אדמו״ר הזקן — יֵלֵךְ שֶׁיִּזְרְקֶנָּה מֵאַחֲרָיו אוֹ לִצְדָדִים ד׳ אַמּוֹת, וְכֵן בִּקְרִיאַת שְׁמַע ; חוֹזֵר לְמָקוֹם שֶׁפָּסַק, שֶׁלֹּא הָיְתָה בְּד׳ אַמּוֹת סְבִיבוֹ (סעיף ג).

הטבלה נערכה מטקסט המחבר (אורח חיים פ״א, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — שִׁעוּר הַהַרְחָקָה

קו א — שִׁעוּר הַהַרְחָקָה

יסוד הסימן : קָטָן שֶׁיָּכוֹל לֶאֱכֹל כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס — מַרְחִיקִין כְּמוֹ מִשֶּׁל גָּדוֹל (סעיף א).

קריאת הרב : הַמֶּדֶד הוּא יְכֹלֶת מַמָּשִׁית לֶאֱכֹל כְּזַיִת דָּגָן בִּזְמַן אֲכִילַת פְּרָס, אֲפִלּוּ עַל יְדֵי בִּשּׁוּל. עיין סימן 79.

מסקנה למעשה : כָּל עוֹד קָטָן יָכוֹל לֶאֱכֹל כָּךְ — מַרְחִיקִין מִצּוֹאָתוֹ וּמִמֵּי רַגְלָיו כְּמוֹ מִגָּדוֹל מַמָּשׁ.

קו ב — חֲזָקָה וְטוֹב וְיָשָׁר

קו ב — חֲזָקָה וְטוֹב וְיָשָׁר

יסוד הסימן : חַלָּשׁ שֶׁלֹּא הָיָה יָכוֹל מֵעוֹלָם — אֵין מַרְחִיקִין ; אָכַל פַּעַם וְנֶחֱלַשׁ — דִּינוֹ כְּגָדוֹל (סעיף ב).

קריאת הרב : מִשֶּׁהִגִּיעָה הַיְּכֹלֶת — בְּנֵי מֵעַיִם מַסְרִיחִים כָּל מַאֲכָל, וְהַמֶּדֶד אֵינוֹ תָּלוּי בְּמַה שֶּׁאוֹכֵל הַיּוֹם. עיין סימן 80.

מסקנה למעשה : מִן הַדִּין מַבְחִינִים לְפִי הַיְּכֹלֶת ; אֲבָל טוֹב וְיָשָׁר, כְּשֶׁאֶפְשָׁר בְּקַל, לִהְיוֹת נָקִי לְגַמְרֵי אֲפִלּוּ מִצּוֹאַת תִּינוֹק בֶּן יוֹמוֹ.

קו ג — הַרְחָקַת הַצּוֹאָה בַּתְּפִלָּה

קו ג — הַרְחָקַת הַצּוֹאָה בַּתְּפִלָּה

יסוד הסימן : רָאָה צוֹאָה כְּנֶגְדּוֹ בְּד׳ אַמּוֹת — יֵלֵךְ וְיַרְחִיקֶנָּה מֵאַחֲרָיו אוֹ לִצְדָדִים ד׳ אַמּוֹת (סעיף ג).

קריאת הרב : חוֹבַת הַבְּדִיקָה אֵינָהּ אֶלָּא בְּד׳ אַמּוֹת סְבִיבוֹ ; מֵעֵבֶר לָהֶן — לֹא הִטְרִיחוּ לִבְדֹּק כִּמְלֹא עֵינָיו. עיין סימן 82.

מסקנה למעשה : חוֹזֵר רַק לְמָקוֹם שֶׁפָּסַק — תְּפִלָּתוֹ שֶׁכְּבָר אָמַר אֵינָהּ תּוֹעֵבָה, שֶׁלֹּא הָיְתָה הַצּוֹאָה בְּד׳ אַמּוֹתָיו.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן פ״א ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן פ״א (דִּין צוֹאַת קָטָן)

  • ① שִׁעוּר. קָטָן הַיָּכוֹל לֶאֱכֹל כְּזַיִת דָּגָן בִּכְדֵי אֲכִילַת פְּרָס (אֲפִלּוּ דַּייְסָא) — מַרְחִיקִין מִצּוֹאָתוֹ וּמִמֵּי רַגְלָיו כְּמוֹ מִגָּדוֹל. יסוד : שוע״ר פ״א סעיף א.
  • ② אֵין מִצְטָרֵף. צָרִיךְ יוֹתֵר מֵאֲכִילַת פְּרָס — אֵין אֲכִילָה מִצְטָרֶפֶת (הֲלָכָה לְמֹשֶׁה מִסִּינַי) ; שִׁעוּר כְּזַיִת → סימן תפ״ו, שִׁעוּר פְּרָס → סימן תרי״ב. יסוד : שוע״ר פ״א סעיף א.
  • ③ חַלָּשׁ. קָטָן חַלָּשׁ שֶׁלֹּא יָכוֹל מֵעוֹלָם — אֵין צָרִיךְ לְהַרְחִיק, אֲפִלּוּ בֵּן ג׳ וד׳ שָׁנִים ; אֲבָל הָיָה יָכוֹל וְנֶחֱלַשׁ — דִּינוֹ כְּגָדוֹל. יסוד : שוע״ר פ״א סעיף ב.
  • ④ טוֹב וְיָשָׁר. מִן הַדִּין מַבְחִינִים ; אֲבָל טוֹב וְיָשָׁר לִקְרוֹת וּלְהִתְפַּלֵּל בְּמָקוֹם נָקִי לְגַמְרֵי, אֲפִלּוּ מִצּוֹאַת תִּינוֹק בֶּן יוֹמוֹ, אִם אֶפְשָׁר בְּקַל. יסוד : שוע״ר פ״א סעיף ב.
  • ⑤ צוֹאָה כְּנֶגְדּוֹ. רָאָה צוֹאָה כְּנֶגְדּוֹ בִּתְפִלָּה — יֵלֵךְ וְיַרְחִיקֶנָּה ד׳ אַמּוֹת (מֵאַחֲרָיו אוֹ לִצְדָדִים), וְכֵן בִּקְרִיאַת שְׁמַע, וְחוֹזֵר לְמָקוֹם שֶׁפָּסַק. יסוד : שוע״ר פ״א סעיף ג.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן פ״א (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַשִּׁעוּר, הַחֲזָקָה, הַרְחָקַת הַצּוֹאָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
