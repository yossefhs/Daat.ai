דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ע״ט · דִּין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה

דרכו של אדמו״ר הזקן בדִּין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה : הַרְחָקַת ד׳ אַמּוֹת מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ, וְמַחֲנֵהוּ קָדוֹשׁ כִּמְלֹא עֵינָיו ; הַשִּׁנּוּי רְשׁוּת (גָּבוֹהַּ אוֹ נָמוּךְ י׳ טְפָחִים) ; צוֹאַת בְּהֵמָה חַיָּה וָעוֹף וְרֵיחַ רַע שֶׁאֵין לוֹ עִקָּר — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ע״ט — דִּין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה. אצל הרב הסימן נפרש לי״א סְעִיפִים : הַהַרְחָקָה וְהַמַּחֲנֶה קָדוֹשׁ, הַשִּׁנּוּי רְשׁוּת וְהַהֶפְסֵק, צוֹאַת בְּהֵמָה חַיָּה וָעוֹף, וְהָאַשְׁפָּה עִם רֵיחַ רַע שֶׁאֵין לוֹ עִקָּר.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ע״ט — י״א סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ע״ט). אדמו״ר הזקן דן בו בְּדִין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ע״ט — דִּין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ע״ט — דִּין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה — וּבוֹ י״א סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.79), ואחריו תרגומנו. הסימן כולל י״א סעיפים (סעיף k → ‎.79.k).

סעיף א צוֹאַת אָדָם מֵאַחֲרָיו — ד׳ אַמּוֹת ; מַחֲנֵהוּ קָדוֹשׁ כִּמְלֹא עֵינָיו.

מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה וּבוֹ י״א סְעִיפִים:הָיְתָה צוֹאַת אָדָם מֵאַחֲרָיו אֲפִלּוּ אֵין לָהּ רֵיחַ — צָרִיךְ לְהַרְחִיק מִמֶּנָּה ד׳ אַמּוֹת, שֶׁנֶּאֱמַר: ״וְהָיָה מַחֲנֶיךָ קָדוֹשׁ״, וּמַחֲנֵהוּ שֶׁל אָדָם דְּהַיְנוּ חֲנָיָתוֹ — הַיְינוּ ד׳ אַמּוֹת לְכָל רוּחַ (שֶׁזֶּהוּ מְקוֹמוֹ שֶׁל אָדָם כְּשֶׁיִּשְׁכֹּב וְיִפְשֹׁט יָדָיו וְרַגְלָיו כְּמוֹ שֶׁיִּתְבָּאֵר סִימָן שמ״ט), חוּץ מִכְּנֶגֶד פָּנָיו שֶׁשָּׁם עִקָּר חֲנָיָתוֹ. וְצָרִיךְ לִהְיוֹת מַחֲנֵהוּ קָדוֹשׁ כִּמְלֹא עֵינָיו. וַאֲפִלּוּ בַּלַּיְלָה שֶׁאֵינוֹ רוֹאֶה אוֹתָהּ, אוֹ שֶׁהוּא סוּמָא — צָרִיךְ לְהַרְחִיק עַד מָקוֹם שֶׁהַפִּקֵּחַ לֹא יוּכַל לִרְאוֹתָהּ בַּיּוֹם. וּצְדָדִים שֶׁלְּפָנָיו — הֲרֵי הֵם כִּלְפָנָיו, וְשֶׁלְּאַחֲרָיו — כִּלְאַחֲרָיו (וּצְדָדִין מַמָּשׁ — יִתְבָּאֵר בְּסִימָן פ״א):

צוֹאַת אָדָם מֵאַחֲרָיו. צוֹאַת אָדָם מֵאַחֲרָיו — אַף בְּלֹא רֵיחַ — מַרְחִיק ד׳ אַמּוֹת (״וְהָיָה מַחֲנֶיךָ קָדוֹשׁ״ ; מַחֲנֵהוּ ד׳ אַמּוֹת לְכָל רוּחַ), חוּץ מִכְּנֶגֶד פָּנָיו — מַחֲנֵהוּ קָדוֹשׁ כִּמְלֹא עֵינָיו. בַּלַּיְלָה אוֹ סוּמָא — עַד מָקוֹם שֶׁאֵין הַפִּקֵּחַ רוֹאֶה בַּיּוֹם. צְדָדִין כִּלְפָנָיו וּכְלְאַחֲרָיו.

סעיף ב יֵשׁ לָהּ רֵיחַ — ד׳ אַמּוֹת מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ.

וְאִם יֵשׁ לָהּ רֵיחַ — צָרִיךְ שֶׁלֹּא יִהְיֶה רֵיחָהּ בְּתוֹךְ ד׳ אַמּוֹת שֶׁלְּאַחֲרָיו וּצְדָדִין, שֶׁאִם לֹא כֵן — אֵין מַחֲנֵהוּ קָדוֹשׁ. לְפִיכָךְ צָרִיךְ לְהַרְחִיק ד׳ אַמּוֹת מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ. וַאֲפִלּוּ אִם יֵשׁ לוֹ חֳלִי שֶׁאֵינוֹ מֵרִיחַ — צָרִיךְ לְהַרְחִיק ד׳ אַמּוֹת מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ לְמִי שֶׁמֵּרִיחַ (אֲבָל מִלְּפָנָיו אֵין צָרִיךְ לְהַרְחִיק מְלֹא עֵינָיו מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ, אֶלָּא כֵּיוָן שֶׁאֵינוֹ יָכוֹל לִרְאוֹתָהּ — הֲרֵי מַחֲנֵהוּ קָדוֹשׁ):

יֵשׁ לָהּ רֵיחַ. יֵשׁ לָהּ רֵיחַ — מַרְחִיק ד׳ אַמּוֹת מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ (שֶׁאֵין מַחֲנֵהוּ קָדוֹשׁ). וְאַף חוֹלֶה שֶׁאֵינוֹ מֵרִיחַ — לְפִי הַמֵּרִיחַ. וּמִלְּפָנָיו — כֵּיוָן שֶׁאֵינוֹ רוֹאֶה, מַחֲנֵהוּ קָדוֹשׁ.

סעיף ג שְׁלִיחַ צִבּוּר וְצוֹאָה בְּבֵית הַכְּנֶסֶת — שׁוֹתֵק עַד שֶׁיּוֹצִיאוּהָ.

שְׁלִיחַ צִבּוּר הַמִּתְפַּלֵל וְצוֹאָה בְּבֵית הַכְּנֶסֶת אוֹ בְּאֵיזֶה בַּיִת שֶׁמִּתְפַּלֵל שָׁם, אֲפִלּוּ הוּא לְאַחֲרָיו בְּכָל הַבַּיִת — צָרִיךְ הוּא לִשְׁתֹּק עַד שֶׁיּוֹצִיאוּהוּ, מֵאַחַר שֶׁמּוֹצִיא רַבִּים יְדֵי חוֹבָתָן וְאִי אֶפְשָׁר שֶׁלֹּא יִהְיֶה אֶחָד מִן הַקָּהָל בְּתוֹךְ ד׳ אַמּוֹת מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ, וְאִי אֶפְשָׁר לוֹ לְכַוֵּן וּלִשְׁמֹעַ מֵהַשְּׁלִיחַ צִבּוּר, וְאֵין הַשָּׁלִיחַ צִבּוּר מוֹצִיאוֹ יְדֵי חוֹבָתוֹ אֶלָּא אִם כֵּן שׁוֹמֵעַ מִמֶּנּוּ. וְאִם הוּא בִּפְסוּקֵי דְּזִמְרָה שֶׁאֵינוֹ מוֹצִיא אֲחֵרִים יְדֵי חוֹבָתָן — מֻתָּר לַשְּׁלִיחַ צִבּוּר לוֹמַר כְּדַרְכּוֹ אִם הוּא חוּץ לד׳ אַמּוֹת. וּבְמֵי רַגְלַיִם שֶׁאֵין לָהֶם רֵיחַ — יְכוֹלִים כָּל הַקָּהָל לְהַרְחִיק ד׳ אַמּוֹת, וְיָכוֹל הַשְּׁלִיחַ צִבּוּר לוֹמַר כְּדַרְכּוֹ בְּכָל עִנְיָן:

שְׁלִיחַ צִבּוּר. ש״ץ שֶׁמִּתְפַּלֵּל וְצוֹאָה בְּבֵית הַכְּנֶסֶת — אַף לְאַחֲרָיו — שׁוֹתֵק עַד שֶׁיּוֹצִיאוּהָ (מוֹצִיא רַבִּים). בִּפְסוּקֵי דְּזִמְרָה — מֻתָּר חוּץ לד׳ אַמּוֹת. מֵי רַגְלַיִם בְּלֹא רֵיחַ — הַקָּהָל מַרְחִיק ד׳ אַמּוֹת.

סעיף ד גָּבוֹהַּ אוֹ נָמוּךְ י׳ טְפָחִים — רְשׁוּת לְעַצְמוֹ.

הָיְתָה בְּמָקוֹם גָּבוֹהַּ י׳ טְפָחִים אוֹ נָמוּךְ י׳ טְפָחִים — הֲרֵי זֶה רְשׁוּת לְעַצְמוֹ וְנִקְרָא מַחֲנֵהוּ קָדוֹשׁ, אַף עַל פִּי שֶׁיּוֹשֵׁב בְּצִדָּהּ וְרוֹאֶה אוֹתָהּ וְקוֹרֵא. וְהוּא שֶׁאֵין רֵיחָהּ מַגִּיעַ אֵלָיו. וְיֵשׁ אוֹמְרִים שֶׁצָּרִיךְ שֶׁיִּהְיֶה רֹחַב מָקוֹם הַהוּא ד׳ טְפָחִים עַל ד׳ טְפָחִים וְאָז הוּא מָקוֹם חָשׁוּב לַחֲלֹק רְשׁוּת לְעַצְמוֹ כְּמוֹ בָּרְשֻׁיּוֹת שֶׁל שַׁבָּת. וְיֵשׁ לָחוּשׁ לְדִבְרֵיהֶם לְהַחֲמִיר בְּשֶׁל תּוֹרָה:

גָּבוֹהַּ אוֹ נָמוּךְ י׳ טְפָחִים. גָּבוֹהַּ אוֹ נָמוּךְ י׳ טְפָחִים — רְשׁוּת לְעַצְמוֹ, מַחֲנֵהוּ קָדוֹשׁ, אַף שֶׁרוֹאֶה — וּבִלְבַד שֶׁאֵין הָרֵיחַ מַגִּיעַ. וְיֵשׁ אוֹמְרִים ד׳ טְפָחִים עַל ד׳ ; וְיֵשׁ לָחוּשׁ לְהַחֲמִיר בְּשֶׁל תּוֹרָה.

סעיף ה בְּחֶדֶר אַחֵר וְהַפֶּתַח פָּתוּחַ — מֻתָּר אִם אֵין רֵיחַ.

וְכֵן אִם הָיְתָה בְּחֶדֶר אֶחָד וְהוּא קוֹרֵא בְּחֶדֶר אַחֵר, אֲפִלּוּ הַפֶּתַח פָּתוּחַ בֵּינֵיהֶם וְהוּא יוֹשֵׁב אֵצֶל הַפֶּתַח בְּסָמוּךְ לָהּ וְרוֹאֶה אוֹתָהּ — מֻתָּר אִם אֵין רֵיחָהּ מַגִּיעַ אֵלָיו, לְפִי שֶׁהַפֶּתַח הִיא כְּסָתוּם, כֵּיוָן שֶׁיֵּשׁ לָהּ גִּפּוּפִים מִכָּאן וּמִכָּאן הַמַּבְדִּילִים בֵּין חֶדֶר לְחֶדֶר. וְיֵשׁ מִי שֶׁאוֹסֵר מִי שֶׁרוֹאֶה אוֹתָהּ אַף עַל פִּי שֶׁהִיא בִּרְשׁוּת אַחֶרֶת. וְהָעִקָּר כִּסְבָרָא הָרִאשׁוֹנָה, שֶׁכֵּיוָן שֶׁהִיא בִּרְשׁוּת אַחֶרֶת אַף עַל פִּי שֶׁרוֹאֶה אוֹתָהּ — הֲרֵי זֶה כְּצוֹאָה בַּעֲשָׁשִׁית שֶׁאֵין אִסּוּר כְּלָל בִּרְאִיָּתָהּ. וְאַף עַל פִּי כֵן טוֹב לָחוּשׁ לִדְבָרָיו. וְאִם מַעֲצִים עֵינָיו אוֹ בַּלַּיְלָה — מֻתָּר בָּזֶה לְדִבְרֵי הַכֹּל, כֵּיוָן שֶׁהִיא בִּרְשׁוּת אַחֶרֶת וְגַם אֵינוֹ רוֹאֶה אוֹתָהּ:

בְּחֶדֶר אַחֵר. בְּחֶדֶר אַחֵר וְהַפֶּתַח פָּתוּחַ וְרוֹאֶה — מֻתָּר אִם אֵין רֵיחַ (פֶּתַח כְּסָתוּם, גִּפּוּפִים). יֵשׁ מִי שֶׁאוֹסֵר בָּרוֹאֶה ; וְהָעִקָּר כָּרִאשׁוֹנָה (כְּצוֹאָה בַּעֲשָׁשִׁית), וְטוֹב לָחוּשׁ. מַעֲצִים עֵינָיו אוֹ לַיְלָה — מֻתָּר לְדִבְרֵי הַכֹּל.

סעיף ו הֶעֱבִירוּ צוֹאָה וְנִשְׁאַר רֵיחַ — יֵשׁ לְהַחֲמִיר.

אִם הֶעֱבִירוּ צוֹאָה בַּבַּיִת וְנִשְׁאַר עֲדַיִן רֵיחַ רַע בַּבַּיִת — יֵשׁ לְהַחֲמִיר שֶׁלֹּא לִקְרוֹת בּוֹ עַד שֶׁיִּכְלֶה הָרֵיחַ, אַף עַל פִּי שֶׁהַצּוֹאָה כְּבָר חָלְפָה וְעָבְרָה. וּמִכָּל מָקוֹם מִי שֶׁאֵינוֹ מֵרִיחַ — מֻתָּר לִקְרוֹת בָּזֶה, כֵּיוָן שֶׁאֵין צוֹאָה בַּבַּיִת:

הֶעֱבִירוּ צוֹאָה. הֶעֱבִירוּ צוֹאָה וְנִשְׁאַר רֵיחַ — יֵשׁ לְהַחֲמִיר עַד שֶׁיִּכְלֶה. וּמִי שֶׁאֵינוֹ מֵרִיחַ — מֻתָּר, שֶׁאֵין צוֹאָה בַּבַּיִת.

סעיף ז חָצֵר קְטַנָּה שֶׁנִּפְרְצָה לַגְּדוֹלָה — דִּין הָרְשֻׁיּוֹת.

חָצֵר קְטַנָּה שֶׁנִּפְרְצָה בְּמִלּוּאָהּ לַגְּדוֹלָּה וּבַגְּדוֹלָה יֵשׁ גִּפּוּפִים הָעוֹדְפִים עַל הַפִּרְצָה מִכָּאן וּמִכָּאן — הַקְּטַנָּה אֵינָהּ חֲשׁוּבָה כְּבַיִת בִּפְנֵי עַצְמָהּ אֶלָּא הוּא כְּקֶרֶן זָוִית שֶׁל הַגְּדוֹלָה הוֹאִיל וְנִפְרְצָה לָהּ בְּמִלּוּאָהּ, וְהַגְּדוֹלָה הִיא כְּבַיִת בִּפְנֵי עַצְמָהּ הוֹאִיל וְנִשְׁתַּיְּרוּ בָּהּ גִּפּוּפִים מִכָּאן וּמִכָּאן. לְפִיכָךְ אִם צוֹאָה בַּגְּדוֹלָה — אָסוּר לִקְרוֹת בַּקְּטַנָּה (עַד שֶׁיַּרְחִיק כְּשִׁעוּר), אֲבָל אִם צוֹאָה בַּקְּטַנָּה — מֻתָּר לִקְרוֹת בַּגְּדוֹלָה בְּלֹא הַרְחָקָה, כֵּיוָן שֶׁהוּא כְּבַיִת בִּפְנֵי עַצְמוֹ, רַק שֶׁלֹּא יִהְיֶה רֵיחָהּ מַגִּיעַ אֵלָיו, וְגַם שֶׁלֹּא יִרְאֶנָּה לְדִבְרֵי הָאוֹסְרִים לְעֵיל סָעִיף ה׳ (בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁכָּתְלֵי אֹרֶךְ הַקְּטַנָּה נִכְנָסִים לְתוֹךְ הַגְּדוֹלָה וּמֻפְלָגִים שָׁם ג׳ טְפָחִים מִכָּתְלֵי אֹרֶךְ הַגְּדוֹלָה, שֶׁאִם לֹא כֵן אַף הַקְּטַנָּה הִיא כְּבַיִת בִּפְנֵי עַצְמָהּ עַל יְדֵי גִּפּוּפֵי הַגְּדוֹלָה, מִשּׁוּם נִרְאֶה מִבַּחוּץ וְשָׁוֶה מִבִּפְנִים, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ש״ס):

חָצֵר קְטַנָּה. קְטַנָּה שֶׁנִּפְרְצָה בְּמִלּוּאָהּ לַגְּדוֹלָה (וּבַגְּדוֹלָה גִּפּוּפִים) — הַקְּטַנָּה אֵינָהּ בַּיִת, הַגְּדוֹלָה בַּיִת. צוֹאָה בַּגְּדוֹלָה — אָסוּר בַּקְּטַנָּה ; בַּקְּטַנָּה — מֻתָּר בַּגְּדוֹלָה בְּלֹא הַרְחָקָה, וּבִלְבַד שֶׁאֵין רֵיחַ וְשֶׁלֹּא יִרְאֶנָּה (לְהָאוֹסְרִים סָעִיף ה׳).

סעיף ח צוֹאַת כֶּלֶב, חֲמוֹר, חָתוּל וּנְמִיָּה — כְּצוֹאַת אָדָם.

צוֹאַת כֶּלֶב אוֹ חֲזִיר, אִם נָתַן בָּהֶם עוֹרוֹת — מַרְחִיק מֵהֶם כְּמוֹ מִצּוֹאַת אָדָם. וְאִם לָאו — דִּינָן כְּמוֹ צוֹאַת שְׁאָר בְּהֵמָה חַיָּה וָעוֹף, שֶׁאֵין צָרִיךְ לְהִתְרַחֵק מֵהֶם כְּלָל אִם אֵין בָּהֶם רֵיחַ. וְאִם יֵשׁ בָּהֶם רֵיחַ — אֵין צָרִיךְ לְהַרְחִיק אֶלָּא עַד שֶׁיִּכְלֶה הָרֵיחַ וְלֹא יוֹתֵר. חוּץ מִצּוֹאָה רַכָּה שֶׁל חֲמוֹר הַבָּא מִן הַדֶּרֶךְ וּמִצּוֹאַת חָתוּל וּנְמִיָּה שֶׁהֵן כְּצוֹאַת אָדָם, לְפִי שֶׁסְּתָמָן רֵיחָן רַע. וְלָכֵן אֲפִלּוּ יִזְדַּמֵּן פַּעַם אַחַת שֶׁאֵין לָהֶם רֵיחַ — אָסְרוּ כְּצוֹאַת אָדָם. וְאִם יֵשׁ לָהֶם רֵיחַ — צָרִיךְ לְהַרְחִיק ד׳ אַמּוֹת מִמָּקוֹם שֶׁיִּכְלֶה הָרֵיחַ. וְכֵן הַדִּין בִּנְבֵלָה מַסְרַחַת. וּמִכָּל מָקוֹם הַהוֹלֵךְ בַּדֶּרֶךְ אַף עַל פִּי שֶׁרוֹאֶה צוֹאַת בְּהֵמָה כְּנֶגְדּוֹ, אִם אֵין הָרֵיחַ בָּא לוֹ — אֵינוֹ חוֹשֵׁשׁ לַמִּעוּט לִתְלוֹתָהּ בְּשֶׁל חֲמוֹר. אֲבָל בְּסָמוּךְ לָעִיר — יֵשׁ אוֹמְרִים שֶׁיֵּשׁ לָחוּשׁ, לְפִי שֶׁרֹב בְּהֵמוֹת הַמְּצוּיִים שָׁם הֵם חֲמוֹרִים, בַּמְּקוֹמוֹת שֶׁהַחֲמוֹרִים מְצוּיִים:

צוֹאַת כֶּלֶב וַחֲזִיר. כֶּלֶב וַחֲזִיר שֶׁנָּתַן עוֹרוֹת — כְּאָדָם ; וְאִם לָאו — כִּשְׁאָר בְּהֵמָה. חוּץ מֵרַכָּה שֶׁל חֲמוֹר הַבָּא מִן הַדֶּרֶךְ, חָתוּל וּנְמִיָּה וּנְבֵלָה מַסְרַחַת — כְּאָדָם. בַּדֶּרֶךְ, אֵין הָרֵיחַ בָּא — אֵין חוֹשְׁשִׁין לַמִּעוּט ; בְּסָמוּךְ לָעִיר — יֵשׁ לָחוּשׁ (רֹב חֲמוֹרִים).

סעיף ט מֵי רַגְלֵי חֲמוֹר וְצוֹאַת תַּרְנְגוֹלִים אֲדֻמִּים.

יֵשׁ אוֹסְרִין לִקְרוֹת כְּנֶגֶד מֵי רַגְלֵי חֲמוֹר הַבָּא מִן הַדֶּרֶךְ וּכְנֶגֶד צוֹאַת תַּרְנְגוֹלִים אֲדֻמִּים (שֶׁקּוֹרִין ענגלי״שן הא״ן בִּלְשׁוֹן אַשְׁכְּנַז אוֹ אינדי״ק), שֶׁתַּרְנְגוֹלִין אֵלּוּ צוֹאָתָן מַסְרַחַת מְאֹד (וְלָכֵן הִיא כְּצוֹאַת אָדָם, שֶׁאַף עַל פִּי שֶׁיִּזְדַּמֵּן שֶׁאֵין לָהּ רֵיחַ רַע — אָסוּר. וְאִם יֵשׁ לָהּ רֵיחַ — צָרִיךְ לְהַרְחִיק ד׳ אַמּוֹת מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ, כְּמוֹ בְּצוֹאַת אָדָם וְלֹא כְּצוֹאַת שְׁאָר עוֹפוֹת שֶׁיֵּשׁ לָהֶן רֵיחַ). אֲבָל צוֹאַת שְׁאָר תַּרְנְגוֹלִים הַהוֹלְכִים בַּבַּיִת — דִּינוֹ כְּדִין צוֹאַת שְׁאָר עוֹפוֹת. אֲבָל הַלּוּל שֶׁלָּהֶם יֵשׁ בּוֹ סִרְחוֹן לָרֹב, וְלָכֵן דִּינוֹ כְּצוֹאַת אָדָם:

מֵי רַגְלֵי חֲמוֹר וְתַרְנְגוֹלִים אֲדֻמִּים. יֵשׁ אוֹסְרִין מֵי רַגְלֵי חֲמוֹר הַבָּא מִן הַדֶּרֶךְ וְצוֹאַת תַּרְנְגוֹלִים אֲדֻמִּים (מַסְרַחַת מְאֹד) — כְּאָדָם. שְׁאָר תַּרְנְגוֹלִים בַּבַּיִת — כִּשְׁאָר עוֹפוֹת ; אֲבָל הַלּוּל — כְּאָדָם.

סעיף י אַשְׁפָּה שֶׁרֵיחָהּ רַע וְרֵיחַ שֶׁנִּשְׁאַר.

אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע כְּנֶגֶד אַשְׁפָּה שֶׁרֵיחָהּ רַע (אֲפִלּוּ אֵין הָרֵיחַ מִצּוֹאָה מְגֻלָּה שֶׁעָלֶיהָ אֶלָּא מֵהַטְּמוּנָה וּמְכֻסָּה), אוֹ מִצּוֹאָה שֶׁחָלְפָה וְעָבְרָה וְנִשְׁאַר רֵיחָהּ שָׁם (וְהוּא שֶׁהָרֵיחַ מַגִּיעַ אֵלָיו):

אַשְׁפָּה שֶׁרֵיחָהּ רַע. אָסוּר לִקְרוֹת ק״ש כְּנֶגֶד אַשְׁפָּה שֶׁרֵיחָהּ רַע (אַף מִן הַטְּמוּנָה) אוֹ מִצּוֹאָה שֶׁעָבְרָה וְנִשְׁאַר רֵיחָהּ — וְהוּא שֶׁהָרֵיחַ מַגִּיעַ.

סעיף י״א רֵיחַ שֶׁאֵין לוֹ עִקָּר — יָצָא מִמֶּנּוּ אוֹ מֵחֲבֵרוֹ.

רֵיחַ רַע שֶׁאֵין לוֹ עִקָּר, כְּגוֹן שֶׁיָּצָא מִמֶּנּוּ רוּחַ מִלְּמַטָּה — גָּזְרוּ עָלָיו חֲכָמִים מִשּׁוּם רֵיחַ רַע שֶׁיֵּשׁ לוֹ עִקָּר, וְאָסוּר בְּדִבְרֵי תּוֹרָה עַד שֶׁיִּכְלֶה הָרֵיחַ, וְאֵין צָרִיךְ לְהַרְחִיק כְּלוּם. וְאִם יָצָא מֵחֲבֵרוֹ — מֻתָּר בְּדִבְרֵי תּוֹרָה, מִפְּנֵי שֶׁהוּא דָּבָר שֶׁאִי אֶפְשָׁר לִזָּהֵר מִמֶּנּוּ, שֶׁהַתַּלְמִידִים שֶׁבְּבֵית הַמִּדְרָשׁ קְצָתָם גּוֹרְסִים בַּלַּיְלָה וּקְצָתָם יְשֵׁנִים וּמְפִיחִים בְּתוֹךְ הַשֵּׁנָה, לְפִיכָךְ לֹא גָּזְרוּ עַל שֶׁל חֲבֵרוֹ לְדִבְרֵי תּוֹרָה. וְלָכֵן אֲפִלּוּ אִם לוֹמֵד בְּיָחִיד וְאֶפְשָׁר לוֹ לָצֵאת לְמָקוֹם אַחֵר — אֵינוֹ צָרִיךְ לָצֵאת, כֵּיוָן שֶׁלֹּא גָּזְרוּ עַל זֶה כְּלָל לְדִבְרֵי תּוֹרָה. אֲבָל לִקְרוֹת קְרִיאַת שְׁמַע (אוֹ לְהִתְפַּלֵּל) גָּזְרוּ אֲפִלּוּ עַל שֶׁל חֲבֵרוֹ עַד שֶׁיִּכְלֶה הָרֵיחַ, כֵּיוָן שֶׁלְּעוֹלָם אֶפְשָׁר לוֹ לֵילֵךְ לְמָקוֹם אַחֵר לִקְרוֹת אוֹ לְהִתְפַּלֵּל שָׁם:

רֵיחַ שֶׁאֵין לוֹ עִקָּר. רֵיחַ שֶׁאֵין לוֹ עִקָּר (הִפְלִיחַ) — גָּזְרוּ כְּשֶׁיֵּשׁ לוֹ עִקָּר, אָסוּר בְּדִבְרֵי תּוֹרָה עַד שֶׁיִּכְלֶה, בְּלֹא הַרְחָקָה. מֵחֲבֵרוֹ — מֻתָּר בְּדִבְרֵי תּוֹרָה (אִי אֶפְשָׁר לִזָּהֵר) ; אֲבָל ק״ש וּתְפִלָּה — אָסוּר אַף מֵחֲבֵרוֹ.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ע״ט (מהדורת קה״ת, י״א סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה

היסוד : דִּין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַהַרְחָקָה וְהַמַּחֲנֶה קָדוֹשׁ. אחר כך הַשִּׁנּוּי רְשׁוּת וְהַהֶפְסֵק. אחר כך צוֹאַת בְּהֵמָה חַיָּה וָעוֹף. ולבסוף הָאַשְׁפָּה וְהָרֵיחַ רַע שֶׁאֵין לוֹ עִקָּר.

הַרְחָקָה וּמַחֲנֶה קָדוֹשׁ

הַרְחָקָה וּמַחֲנֶה קָדוֹשׁ

היסוד. מֵאַחֲרָיו אַף בְּלֹא רֵיחַ — ד׳ אַמּוֹת (״וְהָיָה מַחֲנֶיךָ קָדוֹשׁ״) ; עִם רֵיחַ — מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ ; מִלְּפָנָיו — כִּמְלֹא עֵינָיו (סעיפים א-ב).

המסקנה. ש״ץ שֶׁצּוֹאָה בְּבֵית הַכְּנֶסֶת — שׁוֹתֵק עַד שֶׁיּוֹצִיאוּהָ (סעיף ג). עיין סימן ע״ח · 78 (הרהור והפסק בק״ש).

שִׁנּוּי רְשׁוּת וְהֶפְסֵק

שִׁנּוּי רְשׁוּת וְהֶפְסֵק

היסוד. גָּבוֹהַּ אוֹ נָמוּךְ י׳ טְפָחִים — רְשׁוּת לְעַצְמוֹ ; חֶדֶר אַחֵר וּפֶתַח פָּתוּחַ — מֻתָּר אִם אֵין רֵיחַ (סעיפים ד-ה).

המסקנה. הֶעֱבִירוּ צוֹאָה וְנִשְׁאַר רֵיחַ — יֵשׁ לְהַחֲמִיר ; דִּין חָצֵר קְטַנָּה וּגְדוֹלָה (סעיפים ו-ז). עיין סימן פ״א · 81 (צואה על בשרו).

צוֹאַת בְּהֵמָה חַיָּה וָעוֹף

צוֹאַת בְּהֵמָה חַיָּה וָעוֹף

היסוד. כֶּלֶב וַחֲזִיר עִם עוֹרוֹת — כְּאָדָם ; חֲמוֹר מִן הַדֶּרֶךְ, חָתוּל וּנְמִיָּה — כְּאָדָם (סעיף ח).

המסקנה. מֵי רַגְלֵי חֲמוֹר וְתַרְנְגוֹלִים אֲדֻמִּים — כְּאָדָם ; הַלּוּל — כְּאָדָם (סעיף ט). עיין סימן פ״ז · 87 (צואה בבית).

אַשְׁפָּה וְרֵיחַ רַע שֶׁאֵין לוֹ עִקָּר

אַשְׁפָּה וְרֵיחַ רַע שֶׁאֵין לוֹ עִקָּר

היסוד. אַשְׁפָּה שֶׁרֵיחָהּ רַע וְצוֹאָה שֶׁעָבְרָה וְנִשְׁאַר רֵיחָהּ — אָסוּר בְּק״ש (סעיף י).

המסקנה. רֵיחַ שֶׁאֵין לוֹ עִקָּר — אָסוּר בְּדִבְרֵי תּוֹרָה ; מֵחֲבֵרוֹ מֻתָּר בְּדִבְרֵי תּוֹרָה אֲבָל לֹא בְּק״ש (סעיף יא). עיין סימן ע״ו · 76 (מקום הראוי לק״ש).

הפניה ללימוד. היקף הסימן — הַהַרְחָקָה, הַשִּׁנּוּי רְשׁוּת וְגִדְרֵי הָרֵיחַ — נוגע להנהגת ק״ש ותפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ע״ט. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מי״א סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הַרְחָקָה וּמַחֲנֶה קָדוֹשׁ או״ח ע״ט — צוֹאָה מֵאַחֲרָיו ד׳ אַמּוֹת מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ ; מִלְּפָנָיו מְלֹא עֵינָיו. מ״ב ע״ט — שִׁעוּר הַהַרְחָקָה וְגֶדֶר מַחֲנֶה קָדוֹשׁ. ערוה״ש ע״ט — פְּרָטֵי הָרֵיחַ וְהָרְאִיָּה בַּצּוֹאָה. דרך אדמו״ר הזקן — מֵאַחֲרָיו ד׳ אַמּוֹת אַף בְּלֹא רֵיחַ ; מִלְּפָנָיו כִּמְלֹא עֵינָיו ; ש״ץ שׁוֹתֵק עַד שֶׁיּוֹצִיאוּהָ (סעיפים א-ג).
שִׁנּוּי רְשׁוּת וְהֶפְסֵק או״ח ע״ט — גָּבוֹהַּ אוֹ נָמוּךְ י׳ טְפָחִים אוֹ בַּיִת אַחֵר — קוֹרֵא אִם אֵין רֵיחַ. מ״ב ע״ט — הֶפְסֵק וְשִׁנּוּי רְשׁוּת בַּצּוֹאָה וּבָרֵיחַ. ערוה״ש ע״ט — גֶּדֶר הָרְשֻׁיּוֹת וְהַגִּפּוּפִים. דרך אדמו״ר הזקן — י׳ טְפָחִים אוֹ חֶדֶר אַחֵר — רְשׁוּת לְעַצְמוֹ ; פֶּתַח כְּסָתוּם, וְטוֹב לָחוּשׁ ; חָצֵר קְטַנָּה וּגְדוֹלָה (סעיפים ד-ז).
צוֹאַת בְּהֵמָה חַיָּה וָעוֹף או״ח ע״ט — כֶּלֶב וַחֲזִיר, חֲמוֹר, חָתוּל וּנְמִיָּה כְּצוֹאַת אָדָם. מ״ב ע״ט — גִּדְרֵי צוֹאַת בְּהֵמָה חַיָּה וָעוֹף. ערוה״ש ע״ט — דִּין הַהוֹלֵךְ בַּדֶּרֶךְ וְהַסָּמוּךְ לָעִיר. דרך אדמו״ר הזקן — עוֹרוֹת כְּאָדָם ; חֲמוֹר, חָתוּל, נְמִיָּה וְתַרְנְגוֹלִים אֲדֻמִּים — כְּאָדָם ; הַלּוּל כְּאָדָם (סעיפים ח-ט).
אַשְׁפָּה וְרֵיחַ שֶׁאֵין לוֹ עִקָּר או״ח ע״ט — אָסוּר כְּנֶגֶד אַשְׁפָּה שֶׁרֵיחָהּ רַע ; רוּחַ מִלְּמַטָּה עַד שֶׁיִּכְלֶה הָרֵיחַ. מ״ב ע״ט — רֵיחַ שֶׁיֵּשׁ וְשֶׁאֵין לוֹ עִקָּר. ערוה״ש ע״ט — דִּין הַהֲפָחָה בַּתּוֹרָה וּבַתְּפִלָּה. דרך אדמו״ר הזקן — אַשְׁפָּה וְצוֹאָה שֶׁעָבְרָה — אָסוּר ; רֵיחַ שֶׁאֵין לוֹ עִקָּר אָסוּר בְּק״ש אַף מֵחֲבֵרוֹ (סעיפים י-יא).

הטבלה נערכה מטקסט המחבר (אורח חיים ע״ט, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, רשב״א, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ד-כ״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (י״א סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַרְחָקָה וּמַחֲנֶה קָדוֹשׁ

קו א — הַרְחָקָה וּמַחֲנֶה קָדוֹשׁ

יסוד הסימן : צוֹאָה מֵאַחֲרָיו ד׳ אַמּוֹת ; עִם רֵיחַ מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ (סעיפים א-ג).

קריאת הרב : הַמַּחֲנֵהוּ קָדוֹשׁ מוֹדֵד אֶת הַהַרְחָקָה — מִלְּפָנָיו כִּמְלֹא עֵינָיו, מֵאַחֲרָיו וּצְדָדִין ד׳ אַמּוֹת. עיין סימן ע״ח · 78.

מסקנה למעשה : ש״ץ שׁוֹתֵק עַד שֶׁיּוֹצִיאוּהָ.

קו ב — שִׁנּוּי רְשׁוּת וְהֶפְסֵק

קו ב — שִׁנּוּי רְשׁוּת וְהֶפְסֵק

יסוד הסימן : גָּבוֹהַּ אוֹ נָמוּךְ י׳ טְפָחִים אוֹ חֶדֶר אַחֵר — רְשׁוּת לְעַצְמוֹ (סעיפים ד-ז).

קריאת הרב : שִׁנּוּי רְשׁוּת מוֹעִיל לַצּוֹאָה עַצְמָהּ כְּשֶׁאֵין הָרֵיחַ מַגִּיעַ ; הָעִקָּר כָּרִאשׁוֹנָה (כְּצוֹאָה בַּעֲשָׁשִׁית) וְטוֹב לָחוּשׁ. עיין סימן פ״א · 81.

מסקנה למעשה : פֶּתַח פָּתוּחַ עִם גִּפּוּפִים — כְּסָתוּם, וְטוֹב שֶׁלֹּא יִרְאֶנָּה.

קו ג — גִּדְרֵי הָרֵיחַ

קו ג — גִּדְרֵי הָרֵיחַ

יסוד הסימן : צוֹאַת בְּהֵמָה, אַשְׁפָּה, רֵיחַ שֶׁאֵין לוֹ עִקָּר (סעיפים ח-יא).

קריאת הרב : הָרֵיחַ הוּא הַקּוֹבֵעַ — סְתָם חֲמוֹר, חָתוּל, נְמִיָּה וְתַרְנְגוֹלִים אֲדֻמִּים מַסְרִיחִים וְדִינָם כְּאָדָם. עיין סימן ע״ו · 76.

מסקנה למעשה : רֵיחַ שֶׁאֵין לוֹ עִקָּר — אָסוּר בְּק״ש (וּבִתְפִלָּה) אַף מֵחֲבֵרוֹ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ע״ט ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ע״ט (דִּין מִי שֶׁנִּזְדַּמֵּן לוֹ צוֹאָה בִּשְׁעַת קְרִיאָה)

  • ① צוֹאַת אָדָם. צוֹאָה מֵאַחֲרָיו — מַרְחִיק ד׳ אַמּוֹת (אַף בְּלֹא רֵיחַ) ; מִלְּפָנָיו — כִּמְלֹא עֵינָיו. יסוד : שוע״ר ע״ט סעיף א.
  • ② עִם רֵיחַ. עִם רֵיחַ — ד׳ אַמּוֹת מִמָּקוֹם שֶׁכָּלֶה הָרֵיחַ, אַף לְמִי שֶׁאֵינוֹ מֵרִיחַ (לְפִי הַמֵּרִיחַ). יסוד : שוע״ר ע״ט סעיף ב.
  • ③ שְׁלִיחַ צִבּוּר. ש״ץ שֶׁצּוֹאָה בְּבֵית הַכְּנֶסֶת — שׁוֹתֵק עַד שֶׁיּוֹצִיאוּהָ ; בִּפְסוּקֵי דְּזִמְרָה חוּץ לד׳ אַמּוֹת מֻתָּר. יסוד : שוע״ר ע״ט סעיף ג.
  • ④ שִׁנּוּי רְשׁוּת. גָּבוֹהַּ/נָמוּךְ י׳ טְפָחִים אוֹ חֶדֶר נִפְרָד — רְשׁוּת לְעַצְמוֹ אִם אֵין רֵיחַ ; וְטוֹב שֶׁלֹּא יִרְאֶנָּה. יסוד : שוע״ר ע״ט סעיפים ד-ה, ז.
  • ⑤ רֵיחַ שֶׁנִּשְׁאַר. נִשְׁאַר רֵיחַ אַחַר שֶׁהֶעֱבִירוּ צוֹאָה — יֵשׁ לְהַחֲמִיר ; אַשְׁפָּה שֶׁרֵיחָהּ רַע — אָסוּר. יסוד : שוע״ר ע״ט סעיפים ו, י.
  • ⑥ בְּהֵמָה וְעוֹף. חֲמוֹר מִן הַדֶּרֶךְ, חָתוּל, נְמִיָּה וְתַרְנְגוֹלִים אֲדֻמִּים — כְּצוֹאַת אָדָם. יסוד : שוע״ר ע״ט סעיפים ח-ט.
  • ⑦ רֵיחַ שֶׁאֵין לוֹ עִקָּר. הֵפִיחַ — אָסוּר בְּדִבְרֵי תּוֹרָה עַד שֶׁיִּכְלֶה ; שֶׁל חֲבֵרוֹ מֻתָּר בְּדִבְרֵי תּוֹרָה אֲבָל לֹא בְּק״ש. יסוד : שוע״ר ע״ט סעיף יא.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ע״ט (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַהַרְחָקָה, הַשִּׁנּוּי רְשׁוּת, הָרֵיחַ הָרַע — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
