דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן מ״ז · הִלְכוֹת בִּרְכַּת הַתּוֹרָה

דרכו של אדמו״ר הזקן בבִּרְכַּת הַתּוֹרָה: לִזָּהֵר בָּהּ מְאֹד וְשַׁלְשֶׁלֶת הַלּוֹמֵד לִשְׁמָהּ ; עַל מַה מְבָרְכִין — מִקְרָא, מִשְׁנָה, תַּלְמוּד, מִדְרָשׁ, הַכּוֹתֵב וְהַמְהַרְהֵר ; הַנֻּסָּח (עַל דִּבְרֵי תוֹרָה, וְהַעֲרֶב נָא, אֲשֶׁר בָּחַר בָּנוּ) ; אַהֲבַת עוֹלָם הַפּוֹטֶרֶת וְהַהֶפְסֵק ; לַיְלָה וָיוֹם ; וְנָשִׁים מְבָרְכוֹת — ברגישות חב״ד.

לָמָּה רמה 4 מֻקְדֶּשֶׁת לְאַדְמוֹ״ר הַזָּקֵן? שֻׁלְחָן עָרוּךְ הָרַב אֵינוֹ פֵּרוּשׁ עַל הַמְחַבֵּר — הוּא שֻׁלְחָן עָרוּךְ עַצְמָאִי וּשְׁלֵם, מֵאֵת אַדְמוֹ״ר הַזָּקֵן (רַבִּי שְׁנֵאוּר זַלְמַן מִלִּיאַדִי, מְיַסֵּד חַב״ד, תַּלְמִיד הַמַּגִּיד מִמֶּזְרִיטְשׁ). יִחוּדוֹ: הוּא מְשַׁלֵּב הֲלָכָה + טַעֲמֵי הַמִּצְווֹת + פְּנִימִיּוּת בְּחִבּוּר אֶחָד, וּפוֹסֵק בְּחֲרִיפוּת תַּלְמוּדִית מְיֻחֶדֶת.

לְחַב״ד, אַדְמוֹ״ר הַזָּקֵן הוּא הַפּוֹסֵק הָאַחֲרוֹן. עַמּוּד זֶה מְכַנֵּס אֶת דֶּרֶךְ הָרַב עַל סִימָן מ״ז — הִלְכוֹת בִּרְכַּת הַתּוֹרָה. אֵצֶל הָרַב הַסִּימָן בֶּן עֲשָׂרָה סְעִיפִים (י׳ סְעִיפִים): חֲשִׁיבוּת בִּרְכַּת הַתּוֹרָה וְשַׁלְשֶׁלֶת הַלּוֹמֵד לִשְׁמָהּ ; עַל מַה מְבָרְכִין (מִקְרָא, מִשְׁנָה, תַּלְמוּד, מִדְרָשׁ), הַכּוֹתֵב וְהַמְהַרְהֵר ; הַנֻּסָּח (עַל דִּבְרֵי תוֹרָה, וְהַעֲרֶב נָא, אֲשֶׁר בָּחַר בָּנוּ) וְהַוָּא״ו שֶׁל הַעֲרֶב ; אַהֲבַת עוֹלָם הַפּוֹטֶרֶת וְהַהֶפְסֵק (שֵׁנָה, לַיְלָה וָיוֹם) ; וְנָשִׁים מְבָרְכוֹת.

← לַהַקְדָּמָה הַכְּלָלִית עַל שִׁיטַת אַדְמוֹ״ר הַזָּקֵן

הֶעָרַת מָקוֹר. הַטֶּקְסְט הָעִבְרִי הַמּוּבָא בְּעַמּוּד זֶה הוּא הַטֶּקְסְט הַמָּלֵא וְהָאֲמִתִּי שֶׁל שֻׁלְחָן עָרוּךְ הָרַב, אורח חיים סימן מ״ז — עֲשָׂרָה סְעִיפִים (י׳ סְעִיפִים) מִיָּדוֹ שֶׁל הָרַב — לְפִי מַהֲדוּרַת קֶהוֹת כְּפִי שֶׁדֻּיְּגָה בְּסֵפַרְיָא (Shulchan Arukh HaRav, Orach Chayim מ״ז). הָרַב דָּן בַּחֲשִׁיבוּת בִּרְכַּת הַתּוֹרָה, בְּעַל מַה מְבָרְכִין, בַּנֻּסָּח, בְּאַהֲבַת עוֹלָם הַפּוֹטֶרֶת, בַּהֶפְסֵק וּבַלַּיְלָה, וּבְנָשִׁים. הַתַּרְגּוּמִים שֶׁלָּנוּ. אֵין מוּבָא דָּבָר מִחוּץ לַטֶּקְסְט הַמְאֻמָּת הַזֶּה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן מ״ז — הִלְכוֹת בִּרְכַּת הַתּוֹרָה

הַטֶּקְסְט הַמָּלֵא שֶׁל שֻׁלְחָן עָרוּךְ הָרַב

שולחן ערוך הרב, סימן מ״ז — הִלְכוֹת בִּרְכַּת הַתּוֹרָה — וּבוֹ י׳ סְעִיפִים
לְשׁוֹן אַדְמוֹ״ר הַזָּקֵן עַצְמוֹ (מְקוֹר סֵפַרְיָא Shulchan_Arukh_HaRav,_Orach_Chayim.47), וְאַחֲרָיו בֵּאוּר קָצָר. הַסִּימָן בֶּן עֲשָׂרָה סְעִיפִים (י׳ סְעִיפִים).

סעיף א חֲשִׁיבוּת בִּרְכַּת הַתּוֹרָה — לִזָּהֵר בָּהּ מְאֹד, שֶׁלֹּא חָרְבָה הָאָרֶץ.

דִּינֵי בִּרְכַּת הַתּוֹרָה וּבוֹ י׳ סְעִיפִים:בִּרְכַּת הַתּוֹרָה צָרִיךְ לִזָּהֵר בָּהּ מְאֹד, שֶׁלֹּא חָרְבָה הָאָרֶץ אֶלָּא מִפְּנֵי שֶׁלֹּא בֵּרְכוּ בַּתּוֹרָה תְּחִלָּה, לְפִי שֶׁלֹּא הָיְתָה חֲשׁוּבָה בְּעֵינֵיהֶם כָּל כָּךְ לְבָרֵךְ עָלֶיהָ, לְכָךְ הָיוּ מְזַלְזְלִים בְּבִרְכָתָהּ, לָכֵן לֹא הֵגֵנָּה עֲלֵיהֶם תּוֹרָתָם אַף עַל פִּי שֶׁעָסְקוּ בָּהּ הַרְבֵּה. וּלְכָךְ יִרְאֶה כָּל אָדָם שֶׁתִּהְיֶה כְּלִי חֶמְדָּתוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁהָיָה מִשְׁתַּעֲשֵׁעַ בָּהּ בְּכָל יוֹם, חֲשׁוּבָה בְּעֵינָיו לְבָרֵךְ עָלֶיהָ בְּשִׂמְחָה יוֹתֵר מֵעַל כָּל הֲנָאוֹת שֶׁבָּעוֹלָם, שֶׁזֶּה מוֹרֶה שֶׁעוֹסֵק בָּהּ לִשְׁמָהּ. וְיִזְכֶּה לְשַׁלְשֶׁלֶת הַנִּמְשָׁךְ לָעוֹסֵק בָּהּ לִשְׁמָהּ, שֶׁהוּא ״לֹא יָמוּשׁוּ מִפִּיךָ וּמִפִּי זַרְעֲךָ וְגו׳״, שֶׁתֵּעָשֶׂה שְׁאֵלָתוֹ וּבַקָּשָׁתוֹ שֶׁמְּבַקֵּשׁ בְּבִרְכָתָהּ ״וְנִהְיֶה אֲנַחְנוּ וְצֶאֱצָאֵינוּ״. וְיֵשׁ מְדַקְדְּקִים לְהוֹסִיף ״וְצֶאֱצָאֵי צֶאֱצָאֵינוּ״, כִּי כָּל שֶׁהוּא וּבְנוֹ וּבֶן בְּנוֹ תַּלְמִיד חָכָם — שׁוּב אֵין תּוֹרָה פּוֹסֶקֶת מִזַּרְעוֹ לְעוֹלָם. וְאֵין צָרִיךְ לְכָךְ, כִּי בִּכְלַל ״צֶאֱצָאֵינוּ״ הֵם גַּם כֵּן בְּנֵי בָּנִים:

חֲשִׁיבוּת בִּרְכַּת הַתּוֹרָה. צָרִיךְ לִזָּהֵר בָּהּ מְאֹד ; שֶׁלֹּא חָרְבָה הָאָרֶץ אֶלָּא מִפְּנֵי שֶׁלֹּא בֵּרְכוּ בַּתּוֹרָה תְּחִלָּה. יַחֲשִׁיב אָדָם אֶת הַתּוֹרָה — כְּלִי חֶמְדָּתוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא — וִיבָרֵךְ עָלֶיהָ בְּשִׂמְחָה, וְיִזְכֶּה לְשַׁלְשֶׁלֶת ״לֹא יָמוּשׁוּ מִפִּיךָ וּמִפִּי זַרְעֲךָ״. וְיֵשׁ מְדַקְדְּקִים לְהוֹסִיף ״וְצֶאֱצָאֵי צֶאֱצָאֵינוּ״, וְאֵין צָרִיךְ, שֶׁבִּכְלַל ״צֶאֱצָאֵינוּ״ הֵם.

סעיף ב מְבָרֵךְ לְמִקְרָא, מִשְׁנָה, תַּלְמוּד וּמִדְרָשׁ ; הַמְהַרְהֵר אֵין צָרִיךְ.

צָרִיךְ לְבָרֵךְ בֵּין לְמִקְרָא לְבַד, בֵּין לְמִשְׁנָה לְבַדָּהּ, בֵּין לְתַלְמוּד לְבַדּוֹ, בֵּין לְמִדְרָשׁ לְבַדּוֹ, שֶׁהַכֹּל תּוֹרָה הִיא וְנִתְּנָה לְמֹשֶׁה מִסִּינַי. וְהַמְהַרְהֵר בְּדִבְרֵי תּוֹרָה — אֵין צָרִיךְ לְבָרֵךְ. וְהוּא הַדִּין שֶׁיָכוֹל לִפְסֹק דִּין בְּלֹא נְתִינַת טַעַם לִדְבָרָיו, שֶׁזֶּהוּ גַּם כֵּן אֵינוֹ אֶלָּא הִרְהוּר, וְהִרְהוּר לָאו כְּדִבּוּר הוּא, כְּמוֹ שֶׁ[יִּ]תְבָּאֵר בְּסִימָן ס״ב:

עַל מַה מְבָרְכִין. מְבָרֵךְ בֵּין לְמִקְרָא לְבַד, בֵּין לְמִשְׁנָה, בֵּין לְתַלְמוּד, בֵּין לְמִדְרָשׁ — שֶׁהַכֹּל תּוֹרָה וְנִתְּנָה מִסִּינַי. וְהַמְהַרְהֵר אֵין צָרִיךְ לְבָרֵךְ ; וְהוּא הַדִּין הַפּוֹסֵק דִּין בְּלֹא נְתִינַת טַעַם, שֶׁגַּם זֶה הִרְהוּר, וְהִרְהוּר לָאו כְּדִבּוּר, כִּמְבֹאָר בְּסִימָן ס״ב.

סעיף ג הַכּוֹתֵב בְּדִבְרֵי תוֹרָה — מַחֲלֹקֶת ; לְעִנְיַן מַעֲשֶׂה יוֹצִיא בְּפִיו.

יֵשׁ מִי שֶׁאוֹמֵר שֶׁהַכּוֹתֵב בְּדִבְרֵי תּוֹרָה אַף עַל פִּי שֶׁאֵינוֹ קוֹרֵא — צָרִיךְ לְבָרֵךְ. וְהוּא שֶׁכּוֹתֵב סְפָרִים לְעַצְמוֹ דֶּרֶךְ לִמּוּדוֹ וּמֵבִין מַה שֶּׁהוּא כּוֹתֵב, אֲבָל סוֹפֵר הַמַּעְתִּיק וְאֵינוֹ מְבַקֵּשׁ לְהָבִין — אֵין צָרִיךְ לְבָרֵךְ. וְכֵן אִם כּוֹתֵב אֵיזֶה פָּסוּק בְּאִגֶּרֶת הָרְשׁוּת לִדְבַר צַחוּת — אֵינוֹ צָרִיךְ לְבָרֵךְ, כֵּיוָן שֶׁאֵינוֹ מִתְכַּוֵּן לְלִמּוּד. וּלְעִנְיַן מַעֲשֶׂה אֵין לִסְמֹךְ עַל דְּבָרָיו לְבָרֵךְ עַל כְּתִיבָה לְבַדָּהּ בְּשׁוּם עִנְיָן, שֶׁהֲרֵי אֵינוֹ רַק מְהַרְהֵר בְּדִבְרֵי תּוֹרָה, אֶלָּא רָאוּי לְכָל כּוֹתֵב בְּדִבְרֵי תּוֹרָה שֶׁיּוֹצִיא מִפִּיו קְצָת תֵּבוֹת לְהִנָּצֵל מִבְּרָכָה לְבַטָּלָה, אִם אֵינוֹ אוֹמֵר פְּסוּקֵי בִּרְכַּת כֹּהֲנִים אוֹ שְׁאָר דָּבָר מִן הַתּוֹרָה אַחַר הַבְּרָכָה כְּמוֹ שֶׁנּוֹהֲגִין:

הַכּוֹתֵב בְּדִבְרֵי תוֹרָה. יֵשׁ מִי שֶׁאוֹמֵר שֶׁהַכּוֹתֵב אַף שֶׁאֵינוֹ קוֹרֵא — מְבָרֵךְ, וּבִלְבַד שֶׁכּוֹתֵב לְעַצְמוֹ וּמֵבִין. אֲבָל סוֹפֵר מַעְתִּיק, אוֹ כּוֹתֵב פָּסוּק בְּאִגֶּרֶת לְצַחוּת — אֵינוֹ מְבָרֵךְ. וּלְעִנְיַן מַעֲשֶׂה אֵין לִסְמֹךְ לְבָרֵךְ עַל כְּתִיבָה לְבַדָּהּ, שֶׁאֵינוֹ אֶלָּא מְהַרְהֵר, אֶלָּא יוֹצִיא מִפִּיו קְצָת תֵּבוֹת לְהִנָּצֵל מִבְּרָכָה לְבַטָּלָה.

סעיף ד נֻסַּח הַבְּרָכָה — עַל דִּבְרֵי תוֹרָה, וְהַעֲרֶב נָא, אֲשֶׁר בָּחַר בָּנוּ ; גֵּר.

שֶׁרָגִיל לִלְמֹד תָּמִיד, וְאַף כְּשֶׁיּוֹצֵא לְהִתְעַסֵּק בְּפַרְנָסָתוֹ מְמַהֵר לַעֲשׂוֹת צְרָכָיו בְּכָל מַה דְּאֶפְשָׁר כְּדֵי לַחֲזֹר וְלִלְמֹד, שֶׁעֵסֶק זֶה אֵינוֹ חָשׁוּב הֶפְסֵק לְעִנְיַן בְּרָכָה, כֵּיוָן שֶׁבְּשָׁעָה שֶׁהוּא מִתְעַסֵּק בּוֹ דַּעְתּוֹ עַל לִמּוּדוֹ. וְהוּא הַדִּין לַמֶּרְחָץ וּלְבֵית הַכִּסֵּא שֶׁאֵינָן חֲשׁוּבִין הֶפְסֵק, לְפִי שֶׁאַף בְּעוֹדוֹ בַּמֶּרְחָץ וּבֵית הַכִּסֵּא — אֵינוֹ מַסִּיחַ דַּעְתּוֹ מִלִּלְמֹד אַחַר כָּךְ. וְכֵן אִם הִשְׁכִּיב עַצְמוֹ לִישֹׁן וְאֵינוֹ יָכוֹל לִישֹׁן — אֵין זֶה הֶפְסֵק, לְפִי שֶׁבְּעוֹדוֹ נֵעוֹר דַּעְתּוֹ עַל לִמּוּדוֹ. וְכֵן אִם יָשַׁן עַל אַצִּילֵי יָדָיו שְׁנַת עֲרַאי — אֵינָהּ חֲשׁוּבָה הֶפְסֵק.אֲבָל יָשַׁן שְׁנַת קֶבַע עַל מִטָּתוֹ — חֲשׁוּבָה הֶפְסֵק. וְכֵן מִי שֶׁאֵין דַּרְכּוֹ לִלְמֹד תָּמִיד וּבֵרַךְ וְלָמַד שִׁעוּרוֹ וְיָצָא לַעֲסָקָיו וְהִסִּיחַ דַּעְתּוֹ מִלִּלְמֹד עוֹד בְּאוֹתוֹ הַיּוֹם וְאַחַר כָּךְ נִמְלַךְ לַחֲזֹר וְלִלְמֹד — צָרִיךְ לַחֲזֹר וּלְבָרֵךְ. וְיֵשׁ אוֹמְרִים שֶׁבִּרְכַּת הַתּוֹרָה הִיא כְּבִרְכוֹת הַשַּׁחַר שֶׁלֹּא תִּקְּנוּ אוֹתָן אֶלָּא פַּעַם אַחַת בַּיּוֹם, וּלְפִיכָךְ אֲפִלּוּ יָשַׁן שְׁנַת קֶבַע עַל מִטָּתוֹ אוֹ שֶׁיָּצָא לַעֲסָקָיו וְהִסִּיחַ דַּעְתּוֹ — אֵין צָרִיךְ לַחֲזֹר וּלְבָרֵךְ. וְכֵן נָהֲגוּ.וּכְשֶׁלּוֹמֵד בַּלַּיְלָה כָּל זְמַן שֶׁלֹּא יָשַׁן — אֵין צָרִיךְ לְבָרֵךְ לְדִבְרֵי הַכֹּל, שֶׁהַלַּיְלָה הוֹלֵךְ אַחַר הַיּוֹם. וּכְשֶׁיָּשַׁן שְׁנַת קֶבַע וְהִשְׁכִּים קֹדֶם עֲלוֹת הַשַּׁחַר — צָרִיךְ לְבָרֵךְ לְדִבְרֵי הַכֹּל, שֶׁאַף לִסְבָרָא אַחֲרוֹנָה שֶׁהִיא כְּבִרְכוֹת הַשַּׁחַר — הֲרֵי גַּם בִּרְכוֹת הַשַּׁחַר זְמַנָּן הוּא גַּם כֵּן קֹדֶם אוֹר הַיּוֹם, מִיָּד שֶׁנֵּעוֹר מִשְּׁנָתוֹ.וְאִם הָיָה נֵעוֹר כָּל הַלַּיְלָה — לַסְּבָרָא הָרִאשׁוֹנָה אֵין צָרִיךְ לְבָרֵךְ כְּשֶׁיֵּאוֹר הַיּוֹם כָּל זְמַן שֶׁלֹּא הִסִּיחַ דַּעְתּוֹ מִלִּלְמֹד, וְלַסְּבָרָא הָאַחֲרוֹנָה — צָרִיךְ לְבָרֵךְ כְּשֶׁיֵּאוֹר הַיּוֹם, כְּמוֹ שְׁאָר בִּרְכוֹת הַשַּׁחַר שֶׁנּוֹהֲגִין לְבָרֵךְ אַף עַל פִּי שֶׁלֹּא נִתְחַיֵּב בָּהֶן. וְהָרוֹצֶה לָצֵאת יְדֵי סָפֵק — יִשְׁמַע בִּרְכַּת הַתּוֹרָה מֵאַחֵר וְיִתְכַוֵּן לָצֵאת יְדֵי חוֹבָתוֹ אִם אֶפְשָׁר, וְאִם לָאו — יְבָרֵךְ בְּעַצְמוֹ:

הֶפְסֵק, שֵׁנָה, וְלַיְלָה. יֵשׁ אוֹמְרִים שֶׁאֵין צָרִיךְ לִלְמֹד מִיָּד אַחַר הַבְּרָכָה (תַּלְמוּד תּוֹרָה חִיּוּבוֹ תָּמִיד), וְיֵשׁ חוֹלְקִים, וְנָכוֹן שֶׁלֹּא לְהַפְסִיק. וְכֵן נָהֲגוּ לוֹמַר בִּרְכַּת כֹּהֲנִים סָמוּךְ לָהּ. הָרָגִיל לִלְמֹד תָּמִיד — עֲסָקָיו, מֶרְחָץ, בֵּית הַכִּסֵּא וּשְׁנַת עֲרַאי אֵינָם הֶפְסֵק, אֲבָל שְׁנַת קֶבַע חֲשׁוּבָה הֶפְסֵק. וְיֵשׁ אוֹמְרִים כְּבִרְכוֹת הַשַּׁחַר — פַּעַם אַחַת בַּיּוֹם, וְכֵן נָהֲגוּ. וּבַלַּיְלָה כָּל שֶׁלֹּא יָשַׁן — אֵין צָרִיךְ, שֶׁהַלַּיְלָה הוֹלֵךְ אַחַר הַיּוֹם. וְהָרוֹצֶה לָצֵאת יְדֵי סָפֵק יִשְׁמַע מֵאַחֵר.

סעיף ח הַמַּשְׁכִּים קֹדֶם אוֹר הַיּוֹם מְבָרֵךְ, וְאֵין צָרִיךְ לַחֲזֹר בְּבֵית הַכְּנֶסֶת.

הַמַּשְׁכִּים קֹדֶם אוֹר הַיּוֹם לִלְמֹד — מְבָרֵךְ בִּרְכַּת הַתּוֹרָה. וְאֵין צָרִיךְ לַחֲזֹר וּלְבָרֵךְ בְּבֵית הַכְּנֶסֶת אֲפִלּוּ אִם הוּא שְׁלִיחַ צִבּוּר וְיוֹרֵד לִפְנֵי הַתֵּבָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן מ״ו וְסִימָן ו׳:

הַמַּשְׁכִּים. הַמַּשְׁכִּים קֹדֶם אוֹר הַיּוֹם לִלְמֹד — מְבָרֵךְ בִּרְכַּת הַתּוֹרָה, וְאֵין צָרִיךְ לַחֲזֹר וּלְבָרֵךְ בְּבֵית הַכְּנֶסֶת אֲפִלּוּ הוּא שְׁלִיחַ צִבּוּר, כִּמְבֹאָר בְּסִימָן מ״ו וְסִימָן ו׳.

סעיף ט כָּל סֵדֶר הַבְּרָכוֹת חוּץ מֵהַנּוֹתֵן לַשֶּׂכְוִי וּפָרָשַׁת הַתָּמִיד ; נְטִילַת יָדַיִם.

הַמַּשְׁכִּים קֹדֶם אוֹר הַיּוֹם — יָכוֹל לְבָרֵךְ כָּל סֵדֶר הַבְּרָכוֹת, חוּץ מִבִּרְכַּת ״הַנּוֹתֵן לַשֶּׂכְוִי בִינָה״ וּ״פָרָשַׁת הַתָּמִיד״, עַיֵּן סִימָן א׳ שֶׁיַּמְתִּין מִלְּאָמְרָהּ עַד שֶׁיֵּאוֹר הַיּוֹם. וְאִם שָׁמַע קוֹל תַּרְנְגוֹל — יָכוֹל לְבָרֵךְ ״הַנּוֹתֵן לַשֶּׂכְוִי בִינָה״ מִיָּד אַחַר חֲצוֹת לַיְלָה. וּלְכַתְּחִלָּה יִטֹּל יָדָיו קֹדֶם שֶׁיְּבָרֵךְ לִלְמֹד. וְאִם אֵין לוֹ מַיִם — יָכוֹל לְבָרֵךְ וְלִלְמֹד בְּלֹא נְטִילָה, רַק שֶׁיְּנַקֶּה יָדָיו בִּצְרוֹר אוֹ בְּעָפָר אוֹ בְּקוֹרָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ד׳:

סֵדֶר הַבְּרָכוֹת קֹדֶם אוֹר ; נְטִילַת יָדַיִם. יָכוֹל לְבָרֵךְ כָּל סֵדֶר הַבְּרָכוֹת חוּץ מִ״הַנּוֹתֵן לַשֶּׂכְוִי״ וּ״פָרָשַׁת הַתָּמִיד״ — עַיֵּן סִימָן א׳ שֶׁיַּמְתִּין. שָׁמַע קוֹל תַּרְנְגוֹל — מְבָרֵךְ ״הַנּוֹתֵן לַשֶּׂכְוִי״ אַחַר חֲצוֹת. וּלְכַתְּחִלָּה יִטֹּל יָדָיו, וּבְאֵין מַיִם — יְנַקֶּה בִּצְרוֹר אוֹ עָפָר אוֹ קוֹרָה.

סעיף י נָשִׁים מְבָרְכוֹת בִּרְכוֹת הַתּוֹרָה ; חַיָּבוֹת בִּתְפִלָּה כַּאֲנָשִׁים.

נָשִׁים מְבָרְכוֹת בִּרְכוֹת הַתּוֹרָה, שֶׁהֲרֵי חַיָּבוֹת לִלְמֹד מִצְווֹת שֶׁלָּהֶן, לֵידַע הֵיאַךְ לַעֲשׂוֹתָן וְהֵיאַךְ לִזָּהֵר מִכָּל לֹא תַעֲשֶׂה שֶׁהֵן מֻזְהָרוֹת בָּהֶן כַּאֲנָשִׁים, וְהֵן חַיָּבוֹת בִּתְפִלָּה כְּמוֹ אֲנָשִׁים כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ק״ו (וְעוֹד שֶׁבִּרְכַּת הַתּוֹרָה מְבָרְכִין קֹדֶם ״פָּרָשַׁת הַתָּמִיד״ וּבִקְרִיאַת ״פָּרָשַׁת הַתָּמִיד״ הֵן שָׁווֹת לָאֲנָשִׁים, שֶׁהֲרֵי תְּפִלָּה בִּמְקוֹם תָּמִיד תִּקְּנוּהָ):

נָשִׁים. נָשִׁים מְבָרְכוֹת בִּרְכוֹת הַתּוֹרָה, שֶׁחַיָּבוֹת לִלְמֹד מִצְווֹתֵיהֶן לֵידַע הֵיאַךְ לַעֲשׂוֹתָן וּלְהִזָּהֵר מִלֹּא תַעֲשֶׂה, וְחַיָּבוֹת בִּתְפִלָּה כַּאֲנָשִׁים (סִימָן ק״ו) ; וְעוֹד שֶׁבִּרְכַּת הַתּוֹרָה קֹדֶם פָּרָשַׁת הַתָּמִיד, וּבָהּ שָׁווֹת לָאֲנָשִׁים.

מָקוֹר: Sefaria — Shulchan Arukh HaRav, Orach Chayim מ״ז (מַהֲדוּרַת קֶהוֹת, עֲשָׂרָה סְעִיפִים / י׳ סְעִיפִים). בֵּאוּר: DAAT. אֵין מוּבָא דָּבָר מִחוּץ לַטֶּקְסְט הַזֶּה.

הַיְסוֹד: עַל מַה מְבָרְכִין וְאַהֲבַת עוֹלָם הַפּוֹטֶרֶת

קְרִיאַת אַדְמוֹ״ר הַזָּקֵן בַּסִּימָן בִּשְׁלֹשָׁה מַהֲלָכִים. תְּחִלָּה הַחֲשִׁיבוּת: לִזָּהֵר בָּהּ מְאֹד, שֶׁלֹּא חָרְבָה הָאָרֶץ אֶלָּא עַל שֶׁלֹּא בֵּרְכוּ תְּחִלָּה — הַקֶּשֶׁר בֵּין חֲשִׁיבוּת הַבְּרָכָה לְלִמּוּד לִשְׁמָהּ. אַחַר כָּךְ עַל מַה מְבָרְכִין: מִקְרָא, מִשְׁנָה, תַּלְמוּד, מִדְרָשׁ הַכֹּל תּוֹרָה ; הַכּוֹתֵב הַמֵּבִין מְבָרֵךְ, וְהַמְהַרְהֵר לֹא, שֶׁהִרְהוּר לָאו כְּדִבּוּר. לְבַסּוֹף אַהֲבַת עוֹלָם פּוֹטֶרֶת וְהַהֶפְסֵק: טִיב הַבְּרָכָה, הַצֹּרֶךְ לַעֲסֹק מִיָּד, וְהַסָּפֵק הַמְחַיֵּב לְבָרֵךְ קֹדֶם הַתְּפִלָּה.

חֲשִׁיבוּת בִּרְכַּת הַתּוֹרָה וְלִשְׁמָהּ

חֲשִׁיבוּת הַבְּרָכָה וְלִמּוּד לִשְׁמָהּ

הַיְסוֹד. צָרִיךְ לִזָּהֵר בָּהּ מְאֹד — חָרְבָה הָאָרֶץ עַל שֶׁלֹּא בֵּרְכוּ תְּחִלָּה וְלֹא הֶחְשִׁיבוּהָ. בְּרָכָה בְּשִׂמְחָה מוֹרָה שֶׁעוֹסֵק לִשְׁמָהּ.

הַמַּסְקָנָה. הַלּוֹמֵד לִשְׁמָהּ זוֹכֶה לַשַּׁלְשֶׁלֶת — לֹא יָמוּשׁוּ מִפִּיךָ וּמִפִּי זַרְעֲךָ. עַל חֲשִׁיבוּת לִמּוּד הַשַּׁחַר, עַיֵּן סימן 1.

עַל מַה מְבָרְכִין — כּוֹתֵב וּמְהַרְהֵר

עַל מַה מְבָרְכִין: הַכּוֹתֵב וְהַמְהַרְהֵר

הַכְּלָל. מְבָרְכִין לְמִקְרָא, מִשְׁנָה, תַּלְמוּד, מִדְרָשׁ — הַכֹּל תּוֹרָה מִסִּינַי. הַנֻּסָּח: עַל דִּבְרֵי תוֹרָה, וְהַעֲרֶב נָא, אֲשֶׁר בָּחַר בָּנוּ.

הַהַרְחָבָה. הַכּוֹתֵב הַמֵּבִין מְבָרֵךְ ; הַמְהַרְהֵר לֹא — הִרְהוּר לָאו כְּדִבּוּר. עַל מְקוֹם בִּרְכַּת הַתּוֹרָה בַּסֵּדֶר, עַיֵּן סימן 46.

אַהֲבַת עוֹלָם פּוֹטֶרֶת וְהַהֶפְסֵק

אַהֲבַת עוֹלָם הַפּוֹטֶרֶת וְהַהֶפְסֵק

הַכְּלָל. אַהֲבַת עוֹלָם פּוֹטֶרֶת אִם לָמַד מִיָּד בְּלֹא הֶפְסֵק ; בִּרְכַּת הַתּוֹרָה אֵינָהּ כְּבִרְכוֹת הַמִּצְווֹת (חִיּוּב לִמּוּד תָּמִיד).

הַהַנְהָגָה. מִכָּאן הַמַּחֲלֹקֶת בַּשֵּׁנָה (שְׁנַת קֶבַע) וּבַלַּיְלָה (הַלַּיְלָה הוֹלֵךְ אַחַר הַיּוֹם), וְהָעֵצָה לְבָרֵךְ קֹדֶם הַתְּפִלָּה. עַל בִּרְכַּת הַתּוֹרָה בִּקְרִיאַת הַצִּבּוּר, עַיֵּן סימן 139.

הַפְנָיָה לַלִּמּוּד. הֶקֵּף הַיְסוֹד שֶׁהֻנַּח כָּאן — חֲשִׁיבוּת בִּרְכַּת הַתּוֹרָה, עַל מַה מְבָרְכִין וְאַהֲבַת עוֹלָם הַפּוֹטֶרֶת — נוֹגֵעַ גַּם לַהַנְהָגָה הַיּוֹמְיוֹמִית וְגַם לִרְגִישׁוּת חַב״ד עַל קְדֻשַּׁת הַלִּמּוּד. עַמּוּד זֶה מַצִּיג זֹאת בְּמַדְרֵגַת הַכְּלָל הַמְאֻמָּת בַּלָּשׁוֹן שֶׁל הָרַב, בְּלֹא לְהוֹסִיף דָּבָר. לְלִמּוּד מַעֲמִיק וּלְהַנְהָגָה אִישִׁית — לִלְמֹד עִם רַב חַב״ד וּבַמְּקוֹרוֹת עַצְמָם.

הִלְכוֹת בִּרְכַּת הַתּוֹרָה — בַּכְּלָל · לְלִמּוּד, עַיֵּן רַב חַב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כֹּחַ הַפְּסָק שֶׁל אַדְמוֹ״ר הַזָּקֵן

הַשְׁוָאַת דֶּרֶךְ אַדְמוֹ״ר הַזָּקֵן עִם הַמְחַבֵּר, הָרְמָ״א וְהַמִּשְׁנָה בְּרוּרָה בְּנִקּוּדוֹת הַמַּפְתֵּחַ שֶׁל סִימָן מ״ז. שׁוּרַת שֻׁלְחָן עָרוּךְ הָרַב מֻדְגֶּשֶׁת — הִיא מַצִּיגָה אֶת רְגִישׁוּתוֹ שֶׁל הָרַב כְּפִי שֶׁעוֹלֶה מֵעֲשֶׂרֶת סְעִיפָיו. הָרַב מְפָרֵשׁ אֶת הַטַּעַם וְאֶת הַפְּרָט בַּמָּקוֹם שֶׁהַמְחַבֵּר מִסְתַּפֵּק בַּהֲלָכָה.

נוֹשֵׂא Mehaber Rama Mishna Beroura דֶּרֶךְ אַדְמוֹ״ר הַזָּקֵן (שולחן ערוך הרב)
חֲשִׁיבוּת בִּרְכַּת הַתּוֹרָה או״ח מ״ז — צָרִיךְ לִזָּהֵר בְּבִרְכַּת הַתּוֹרָה מְאֹד. — מ״ב מ״ז / נדרים פ״א ע״א — עַל מָה אָבְדָה הָאָרֶץ. דֶּרֶךְ אַדְמוֹ״ר הַזָּקֵן — לִזָּהֵר מְאֹד, שַׁלְשֶׁלֶת הַלּוֹמֵד לִשְׁמָהּ וְ״צֶאֱצָאֵי צֶאֱצָאֵינוּ״ (סעיף א).
עַל מַה מְבָרְכִין ; כּוֹתֵב וּמְהַרְהֵר או״ח מ״ז — מִקְרָא, מִשְׁנָה, תַּלְמוּד ; הַמְהַרְהֵר אֵין צָרִיךְ. רמ״א מ״ז — אַף לְמִדְרָשׁ מְבָרֵךְ. מ״ב מ״ז — כּוֹתֵב לְעַצְמוֹ וּמֵבִין. דֶּרֶךְ אַדְמוֹ״ר הַזָּקֵן — כּוֹתֵב הַמֵּבִין מְבָרֵךְ, סוֹפֵר מַעְתִּיק לֹא ; לְמַעֲשֶׂה יוֹצִיא בְּפִיו (סעיף ב–ג).
נֻסָּח ; וְהַעֲרֶב בְּוָא״ו או״ח מ״ז — עַל דִּבְרֵי תוֹרָה, וְהַעֲרֶב נָא, אֲשֶׁר בָּחַר בָּנוּ. רמ״א מ״ז — יֵשׁ אוֹמְרִים בְּלֹא וָא״ו. מ״ב מ״ז — יוֹתֵר טוֹב בְּוָא״ו ; סְמוּכָה לַחֲבֶרְתָּהּ. דֶּרֶךְ אַדְמוֹ״ר הַזָּקֵן — ג׳ טְעָמִים ; יוֹתֵר טוֹב ״וְהַעֲרֶב״ בְּוָא״ו, אַף עַל פִּי הַסּוֹד (סעיף ד–ה).
אַהֲבַת עוֹלָם ; הֶפְסֵק וְשֵׁנָה או״ח מ״ז — אַהֲבַת עוֹלָם פּוֹטֶרֶת אִם לָמַד מִיָּד. — מ״ב מ״ז / ביאור הלכה — סְפֵק בְּרָכוֹת וּשְׁנַת קֶבַע. דֶּרֶךְ אַדְמוֹ״ר הַזָּקֵן — יֵשׁ לָחֹשׁ לִשְׁנֵיהֶם, וְכֵן נָהֲגוּ שֶׁאֵין צָרִיךְ אַחַר שְׁנַת קֶבַע (סעיף ו–ז).
נָשִׁים בְּבִרְכַּת הַתּוֹרָה או״ח מ״ז — נָשִׁים מְבָרְכוֹת בִּרְכַּת הַתּוֹרָה. — מ״ב מ״ז — חַיָּבוֹת בְּפָרָשִׁיּוֹת שֶׁלָּהֶן. דֶּרֶךְ אַדְמוֹ״ר הַזָּקֵן — חַיָּבוֹת לִלְמֹד מִצְווֹתֵיהֶן וּבִתְפִלָּה כַּאֲנָשִׁים (סעיף י).

הַטַּבְלָה עַל פִּי לְשׁוֹן הַמְחַבֵּר (אורח חיים מ״ז, מְקוֹר סֵפַרְיָא) וְנוֹשְׂאֵי כֵּלִים מְאֻמָּתִים (בית יוסף, טור, רמב״ם הלכות תפילה ז, רא״ש, ר״ן פ״ק דשבת, רשב״א בשם המאור, אבודרהם, אגור, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות י״א ע״ב, כ״א ע״א, נדרים פ״א ע״א). טוּר אַדְמוֹ״ר הַזָּקֵן עַל פִּי לְשׁוֹן שֻׁלְחָן עָרוּךְ הָרַב (י׳ סְעִיפִים) שֶׁבְּ§1. יִחוּסִים מְסֻפָּקִים — עַל דֶּרֶךְ הַסְּבָרָה.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קַוֵּי הַיְסוֹד שֶׁל שִׁיטַת הָרַב בַּסִּימָן

שָׁלֹשׁ מְגַמּוֹת הַמְאַפְיְנוֹת אֶת רְגִישׁוּת אַדְמוֹ״ר הַזָּקֵן בַּסִּימָן — מֵעֵבֶר לְעֶצֶם הַהֲלָכָה. כָּל אַחַת בַּמִּבְנֶה: כְּלַל הַסִּימָן ← קְרִיאַת הָרַב ← מַסְקָנָה לְמַעֲשֶׂה. כָּל קַו נִשְׁעָן עַל לְשׁוֹן הָרַב (§1).

קו א — לִזָּהֵר בָּהּ מְאֹד

לִזָּהֵר בָּהּ מְאֹד

כְּלַל הַסִּימָן: חֲשִׁיבוּת בִּרְכַּת הַתּוֹרָה וְעַל מַה מְבָרְכִין (סעיפים א–ג).

קְרִיאַת הָרַב: חָרְבָה הָאָרֶץ עַל שֶׁלֹּא בֵּרְכוּ ; הַכּוֹתֵב הַמֵּבִין מְבָרֵךְ, הַמְהַרְהֵר לֹא.

לְמַעֲשֶׂה: לְבָרֵךְ בְּשִׂמְחָה עַל מִקְרָא/מִשְׁנָה/תַּלְמוּד/מִדְרָשׁ ; לְמַעֲשֶׂה יוֹצִיא בְּפִיו.

קו ב — נֻסָּח וּסְפֵק בְּרָכָה

הַנֻּסָּח וּסְפֵק הַבְּרָכָה

כְּלַל הַסִּימָן: הַנֻּסָּח (עַל דִּבְרֵי תוֹרָה, וְהַעֲרֶב נָא, אֲשֶׁר בָּחַר בָּנוּ) וְהַוָּא״ו (סעיפים ד–ה).

קְרִיאַת הָרַב: יוֹתֵר טוֹב ״וְהַעֲרֶב״ בְּוָא״ו — שֶׁלֹּא לְהַפְסִיק בְּאֶמְצַע בְּרָכָה אַחַת.

לְמַעֲשֶׂה: לֵילֵךְ אַחַר הַנֻּסָּח הַמְקֻבָּל ; לוֹמַר וְהַעֲרֶב בְּוָא״ו.

קו ג — אַהֲבַת עוֹלָם וְהַלַּיְלָה

אַהֲבַת עוֹלָם, שֵׁנָה וְלַיְלָה

כְּלַל הַסִּימָן: אַהֲבַת עוֹלָם הַפּוֹטֶרֶת, הַהֶפְסֵק, וְלַיְלָה וָיוֹם (סעיפים ו–י).

קְרִיאַת הָרַב: בִּרְכַּת הַתּוֹרָה אֵינָהּ כְּבִרְכוֹת הַמִּצְווֹת ; וְכֵן נָהֲגוּ אַחַר שְׁנַת קֶבַע.

לְמַעֲשֶׂה: לְבָרֵךְ קֹדֶם הַתְּפִלָּה ; מַשְׁכִּים קֹדֶם אוֹר — מְבָרֵךְ ; לֵילֵךְ אַחַר רַבּוֹ בַּסָּפֵק. עַיֵּן סימן 48.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הַהַנְהָגָה הַחַב״דִית לְמַעֲשֶׂה

נְקֻדּוֹת הַנְהָגָה הַנּוֹבְעוֹת יְשִׁירוֹת מִסִּימָן מ״ז בִּרְגִישׁוּת אַדְמוֹ״ר הַזָּקֵן וְחַב״ד — בְּמַדְרֵגַת הַכְּלָל, בְּהַפְנָיָה לָרַב לִפְרָטֵי הַמִּקְרִים (הַחֲשִׁיבוּת, עַל מַה מְבָרְכִין, הַנֻּסָּח, אַהֲבַת עוֹלָם, הַהֶפְסֵק וְהַנָּשִׁים).

לֶחָסִיד חַב״ד — סִימָן מ״ז (הִלְכוֹת בִּרְכַּת הַתּוֹרָה)

  • ① לִזָּהֵר בָּהּ מְאֹד. לְהַחֲשִׁיב אֶת בִּרְכַּת הַתּוֹרָה וּלְבָרְכָהּ בְּשִׂמְחָה ; לִזָּהֵר בָּהּ מְאֹד. יְסוֹד: שו״ע הרב מ״ז סעיף א.
  • ② עַל מַה מְבָרְכִין. לְבָרֵךְ עַל מִקְרָא, מִשְׁנָה, תַּלְמוּד, מִדְרָשׁ ; הַכּוֹתֵב הַמֵּבִין מְבָרֵךְ, הַמְהַרְהֵר לֹא — וּלְמַעֲשֶׂה יוֹצִיא בְּפִיו. יְסוֹד: שו״ע הרב מ״ז סעיף ב–ג.
  • ③ הַנֻּסָּח. לוֹמַר עַל דִּבְרֵי תוֹרָה, וְהַעֲרֶב נָא, אֲשֶׁר בָּחַר בָּנוּ ; ״וְהַעֲרֶב״ בְּוָא״ו. יְסוֹד: שו״ע הרב מ״ז סעיף ד–ה.
  • ④ אַהֲבַת עוֹלָם וְהֶפְסֵק. לְבָרֵךְ בִּרְכַּת הַתּוֹרָה קֹדֶם הַתְּפִלָּה, וְלֹא לִסְמֹךְ עַל אַהֲבַת עוֹלָם ; אַחַר שְׁנַת קֶבַע — וְכֵן נָהֲגוּ שֶׁלֹּא לַחֲזֹר. יְסוֹד: שו״ע הרב מ״ז סעיף ו–ז.
  • ⑤ הַנָּשִׁים. נָשִׁים מְבָרְכוֹת בִּרְכַּת הַתּוֹרָה, שֶׁחַיָּבוֹת לִלְמֹד מִצְווֹתֵיהֶן וּבִתְפִלָּה. יְסוֹד: שו״ע הרב מ״ז סעיף י.

⚠ סָעִיף זֶה מַצִּיג הַנְהָגָה חַב״דִית עַל פִּי שֻׁלְחָן עָרוּךְ הָרַב, סימן מ״ז (הַטֶּקְסְט הָאֲמִתִּי שֶׁבְּ§1). אֵינוֹ מַחֲלִיף פְּסַק רַב לְמִקְרֶה פְּרָטִי. לְכָל שְׁאֵלָה מַעֲשִׂית — בִּרְכַּת הַתּוֹרָה, הַנֻּסָּח, הַהֶפְסֵק, הַנָּשִׁים — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד, עִם רַב חַב״ד.
