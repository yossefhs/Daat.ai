דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן פ״ט · זְמַן תְּפִלַּת הַשַּׁחַר

דרכו של אדמו״ר הזקן בזְמַן תְּפִלַּת הַשַּׁחַר : נֵץ הַחַמָּה — לְכַתְּחִלָּה, וְעַמּוּד הַשַּׁחַר — בְּדִיעֲבַד ; עַד שְׁלִישׁ הַיּוֹם, וְאַחַר כָּךְ עַד חֲצוֹת וְתַשְׁלוּמִין ; אִסּוּר שָׁלוֹם, עֲסָקִים, אֲכִילָה וְלִמּוּד קֹדֶם תְּפִלָּה, וְהַשְּׁעַת הַדְּחָק לַדֶּרֶךְ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן פ״ט — זְמַן תְּפִלַּת הַשַּׁחַר. אצל הרב הסימן נפרש לט׳ סְעִיפִים : הַזְּמַן (נֵץ הַחַמָּה, עַמּוּד הַשַּׁחַר, שְׁלִישׁ הַיּוֹם), הָעַד חֲצוֹת וְהַתַּשְׁלוּמִין, אִסּוּר שָׁלוֹם וַעֲסָקִים, אֲכִילָה וְהַפְסָקָה, תַּלְמוּד תּוֹרָה, מֶרְחָץ, וְהַשְּׁעַת הַדְּחָק לַדֶּרֶךְ.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן פ״ט — ט׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim פ״ט). אדמו״ר הזקן דן בו בְּזְמַן תְּפִלַּת הַשַּׁחַר. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן פ״ט — זְמַן תְּפִלַּת הַשַּׁחַר

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן פ״ט — זְמַן תְּפִלַּת הַשַּׁחַר — וּבוֹ ט׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.89), ואחריו תרגומנו. הסימן כולל ט׳ סעיפים (סעיף k → ‎.89.k).

סעיף א אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה תִּקְּנוּ ג׳ תְּפִלּוֹת ; זְמַן הַשַּׁחַר מִנֵּץ עַד שְׁלִישׁ הַיּוֹם.

הִלְכוֹת תְּפִלָּה וּזְמַנָּהּ וּבוֹ ט׳ סְעִיפִים:אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה תִּקְּנוּ לְהִתְפַּלֵּל בְּכָל יוֹם ג׳ תְּפִלּוֹת שֶׁל י״ח בְּרָכוֹת שֶׁהִתְפַּלְלוּ הָאָבוֹת, וְקָבְעוּ שְׁתַּיִם מֵהֶן חוֹבָה, שֶׁהֵן שַׁחֲרִית וּמִנְחָה, כְּנֶגֶד ב׳ תְּמִידִין שֶׁהֵן חוֹבָה בְּכָל יוֹם, וְשֶׁל עַרְבִית תִּקְּנוּ רְשׁוּת, כְּנֶגֶד אֵיבָרִים וּפְדָרִים שֶׁאִם לֹא נִתְעַכְּלוּ מִבְּעוֹד יוֹם הֵם קְרֵבִים וְהוֹלְכִים כָּל הַלַּיְלָה, אֲבָל עַכְשָׁו כְּבָר קִבְּלוּהוּ כָּל יִשְׂרָאֵל עֲלֵיהֶם לְחוֹבָה, כְּמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמָהּ. וּכְשֶׁרַבּוּ הָאֶפִּיקוֹרְסִין בְּיִשְׂרָאֵל, וְהָיוּ מְצֵרִין לָהֶם מְאֹד, וּמְסִיתִין אֶת הָעָם לִכְפֹּר בַּתּוֹרָה שֶׁבְּעַל פֶּה — עָמַד רַבָּן גַּמְלִיאֵל וּבֵית דִּינוֹ וְהוֹסִיפוּ ״בִּרְכַּת הַמִּינִים״ בְּכָל תְּפִלַּת י״ח. וּזְמַן הַתְּפִלּוֹת הוּא כִּזְמַן הַתְּמִידִין, שֶׁתָּמִיד שֶׁל שַׁחַר זְמַנּוֹ מַתְחִיל מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר, שֶׁהוּא אוֹר הַנּוֹצֵץ מֵהַחַמָּה בַּמִּזְרָח, מַהֲלַךְ ד׳ מִילִין קֹדֶם נֵץ הַחַמָּה. אֶלָּא שֶׁהָיוּ מַמְתִּינִים עַד שֶׁהֵאִיר כָּל פְּנֵי הַמִּזְרָח עַד שֶׁבְּחֶבְרוֹן, לְפִי שֶׁפַּעַם אַחַת בָּאוּ לִידֵי טָעוּת עַל יְדֵי שֶׁעָלָה מְאוֹר הַלְּבָנָה וְדִמּוּ שֶׁהוּא אוֹר הַנּוֹצֵץ בַּמִּזְרָח בַּשַּׁחַר. וְנִמְשַׁךְ זְמַנּוֹ עַד סוֹף שָׁעָה ד׳ מֵעֲלוֹת הַשַּׁחַר, שֶׁהוּא שְׁלִישׁ הַיּוֹם, שֶׁמִּכָּאן וְאֵילָךְ אֵינוֹ נִקְרָא בֹּקֶר. לְפִיכָךְ, גַּם תְּפִלַּת הַשַּׁחַר זְמַנָּהּ מַתְחִיל מֵעֲלוֹת הַשַּׁחַר. אֶלָּא שֶׁלְּכַתְּחִלָּה מִצְוָה לְהַתְחִיל עִם נֵץ הַחַמָּה, דְּהַיְנוּ אַחַר יְצִיאָתָהּ וְלֹא קֹדֶם לָכֵן, מִשּׁוּם שֶׁנֶּאֱמַר: ״יִירָאוּךָ עִם שָׁמֶשׁ וגו׳״. וְרָאוּי לִזָּהֵר בָּזֶה בִּימֵי הַסְּלִיחוֹת שֶׁרְגִילִין לְהַקְדִּים קֹדֶם נֵץ הַחַמָּה, וְאֵינוֹ נָכוֹן. אֲבָל בְּדִיעֲבַד אִם עָבַר וְהִתְפַּלֵּל מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר — יָצָא. וְאַף לְכַתְּחִלָּה יָכוֹל לַעֲשׂוֹת כֵּן בִּשְׁעַת הַדְּחָק, כְּמוֹ שֶׁיִּתְבָּאֵר. וְנִמְשָׁךְ זְמַנָּהּ עַד סוֹף שְׁלִישׁ הַיּוֹם, שֶׁצָּרִיךְ לִגְמֹר הַתְּפִלָּה קֹדֶם שֶׁיַּעֲבֹר שְׁלִישׁ הַיּוֹם, בֵּין שֶׁהַיּוֹם אָרֹךְ בֵּין שֶׁהַיּוֹם קָצָר. כְּגוֹן, אִם הַיּוֹם אָרֹךְ י״ח שָׁעוֹת מֵעֲלוֹת הַשַּׁחַר עַד צֵאת הַכּוֹכָבִים — שְׁלִישׁ הַיּוֹם הוּא ו׳ שָׁעוֹת מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר. וְאִם הוּא ט׳ שָׁעוֹת — שְׁלִישִׁיתוֹ הוּא ג׳ שָׁעוֹת. וְלָכֵן צָרִיךְ לִזָּהֵר בַּחֹרֶף לְהַשְׁכִּים לְהִתְפַּלֵּל בְּעִנְיָן שֶׁיִּגְמְרוּ תְּפִלַּת י״ח קֹדֶם שְׁלִישׁ הַיּוֹם, אַף עַל פִּי שֶׁכְּבָר קָרְאוּ קְרִיאַת שְׁמַע בְּעוֹנָתָהּ קֹדֶם הַתְּפִלָּה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן מ״ו. וְאִם טָעָה אוֹ עָבַר וְלֹא הִתְפַּלֵּל עַד אַחַר שְׁלִישׁ הַיּוֹם עַד חֲצוֹת — יָצָא יְדֵי חוֹבַת תְּפִלָּה, אֲבָל לֹא יָצָא יְדֵי חוֹבַת תְּפִלָּה בִּזְמַנָּהּ. וּמִכָּל מָקוֹם, אֵין צָרִיךְ לְהִתְפַּלֵּל בְּמִנְחָה שְׁתַּיִם, לְפִי שֶׁגַּם בְּתַשְׁלוּמִין שֶׁבְּמִנְחָה אֵין לוֹ שְׂכַר תְּפִלָּה בִּזְמַנָּהּ:

זְמַן תְּפִלַּת הַשַּׁחַר. אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה תִּקְּנוּ ג׳ תְּפִלּוֹת ; שַׁחֲרִית וּמִנְחָה חוֹבָה כְּנֶגֶד הַתְּמִידִין, וְעַרְבִית רְשׁוּת. זְמַן שַׁחֲרִית מֵעֲלוֹת הַשַּׁחַר ; לְכַתְּחִלָּה מִצְוָה עִם נֵץ הַחַמָּה (״יִירָאוּךָ עִם שָׁמֶשׁ״), וּבְדִיעֲבַד מֵעַמּוּד הַשַּׁחַר ; עַד שְׁלִישׁ הַיּוֹם, וְאַחַר כָּךְ עַד חֲצוֹת יָצָא אֲבָל לֹא בִּזְמַנָּהּ.

סעיף ב אֲפִלּוּ לְכַתְּחִלָּה עַד חֲצוֹת ; אַחַר חֲצוֹת — מִנְחָה שְׁתַּיִם בְּטָעוּת אוֹ אֹנֶס.

וַאֲפִלּוּ לְכַתְּחִלָּה יָכוֹל לְהִתְפַּלֵּל שַׁחֲרִית עַד חֲצוֹת אִם עָבַר וְלֹא הִתְפַּלֵּל קֹדֶם לָכֵן, אֲפִלּוּ עָבַר בְּמֵזִיד שֶׁאֵין תַּשְׁלוּמִין בְּמִנְחָה — יֵשׁ לוֹ תַּשְׁלוּמִין עַד חֲצוֹת, כֵּיוָן שֶׁלֹּא הִגִּיעַ עֲדַיִן זְמַן תְּפִלָּה אַחֶרֶת. אֲבָל אַחַר חֲצוֹת מִיָּד, כֵּיוָן שֶׁהִגִּיעַ כְּבָר זְמַן תְּפִלַּת הַמִּנְחָה, אַף עַל פִּי שֶׁאֵינוֹ יָכוֹל לְהִתְפַּלֵּל מִנְחָה עַד אַחַר חֲצִי שָׁעָה אַחַר חֲצוֹת — זֶהוּ מִשּׁוּם שֶׁרֹב בְּנֵי אָדָם (אֵינָם בְּקִיאִים) לְכַוֵּן הַשָּׁעוֹת כָּל כָּךְ, אֲבָל בֶּאֱמֶת זְמַנָּהּ מַתְחִיל מִיָּד אַחַר חֲצוֹת, לְפִיכָךְ אֵין לוֹ תַּשְׁלוּמִין עוֹד לִתְפִלַּת הַשַּׁחַר. אֶלָּא אִם כֵּן טָעָה אוֹ נֶאֱנַס וְלֹא הִתְפַּלֵּל אוֹתָהּ בִּזְמַנָּהּ. וְגַם בְּטָעוּת אוֹ אֹנֶס אֵין לוֹ תַּשְׁלוּמִין אֶלָּא לְהִתְפַּלֵּל מִנְחָה שְׁתַּיִם, כְּלוֹמַר שֶׁמִּתְּחִלָּה יִתְפַּלֵּל מִנְחָה שֶׁהוּא חוֹבַת שָׁעָה, וְאָז כֵּיוָן שֶׁהוּא עָסוּק בַּתְּפִלָּה — חוֹזֵר וּמַשְׁלִים מַה שֶּׁטָּעָה בִּתְפִלּוֹתָיו, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ק״ח. לְפִיכָךְ, צָרִיךְ לְהַמְתִּין חֲצִי שָׁעָה אַחַר חֲצוֹת כְּדֵי לְהִתְפַּלֵּל מִנְחָה תְּחִלָּה, וְאַחַר כָּךְ יַחֲזֹר לְהִתְפַּלֵּל לְשֵׁם תַּשְׁלוּמִין שֶׁל שַׁחֲרִית. וְאִם הָפַךְ — לֹא יָצָא, כְּמוֹ שֶׁיִּתְבָּאֵר שָׁם:

עַד חֲצוֹת וְתַשְׁלוּמִין. אֲפִלּוּ לְכַתְּחִלָּה עַד חֲצוֹת אִם עָבַר, אֲפִלּוּ בְּמֵזִיד, כֵּיוָן שֶׁלֹּא הִגִּיעַ זְמַן תְּפִלָּה אַחֶרֶת. אַחַר חֲצוֹת מִיָּד זְמַן מִנְחָה, וְאֵין תַּשְׁלוּמִין אֶלָּא בְּטָעוּת אוֹ אֹנֶס — מִנְחָה שְׁתַּיִם : תְּחִלָּה מִנְחָה, וְאַחַר כָּךְ תַּשְׁלוּמִין (סימן ק״ח) ; וְאִם הָפַךְ — לֹא יָצָא.

סעיף ג אָסוּר לְהַקְדִּים לְפֶתַח חֲבֵרוֹ לִתֵּן לוֹ שָׁלוֹם קֹדֶם תְּפִלָּה.

כֵּיוָן שֶׁהִגִּיעַ זְמַן תְּפִלָּה, דְּהַיְנוּ מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר, אָסוּר לְאָדָם לְהַקְדִּים לְפֶתַח חֲבֵרוֹ לִתֵּן לוֹ שָׁלוֹם, וְרֶמֶז לַדָּבָר שֶׁנֶּאֱמַר: ״חִדְלוּ לָכֶם מִן הָאָדָם אֲשֶׁר נְשָׁמָה בְּאַפּוֹ כִּי בַמֶּה נֶחְשָׁב הוּא״, כְּלוֹמַר בַּמֶּה חֲשַׁבְתּוֹ לָזֶה שֶׁהִקְדַּמְתָּ כְּבוֹדוֹ לִכְבוֹדִי, וְעִקַּר כְּבוֹדוֹ שֶׁל מָקוֹם וַעֲבוֹדָתוֹ הִיא תְּפִלַּת י״ח. וּלְפִיכָךְ, אֲפִלּוּ הִשְׁכִּים וּבֵרַךְ כָּל בִּרְכוֹת הַשַּׁחַר וְאָמַר כַּמָּה שִׁירוֹת וְתִשְׁבָּחוֹת, וַאֲפִלּוּ קִבֵּל עָלָיו עֹל מַלְכוּת שָׁמַיִם בִּקְרִיאַת שְׁמַע בְּעוֹנָתָהּ קֹדֶם הַתְּפִלָּה — אָסוּר לְהַשְׁכִּים לְפֶתַח חֲבֵרוֹ לִתֵּן לוֹ שָׁלוֹם, אוֹ אֲפִלּוּ לוֹמַר לוֹ ״צַפְרָא דְּמָרֵיהּ טַב״ אִם אֵינוֹ הוֹלֵךְ אֶלָּא לְהַקְבִּיל פָּנָיו. אֲבָל אִם הֻצְרַךְ לָלֶכֶת לִרְאוֹת אֵיזֶהוּ עֵסֶק וְאַגַּב זֶה הוֹלֵךְ לְפֶתַח חֲבֵרוֹ לוֹמַר לוֹ ״צַפְרָא דְּמָרֵיהּ טַב״ — מֻתָּר. אֲבָל לֹא יִתֵּן לוֹ שָׁלוֹם, כִּי שְׁמוֹ שֶׁל הַקָּבָּ״ה שָׁלוֹם, וְאֵין זֶה כְּבוֹד הַמָּקוֹם לְחַלֵּק כָּבוֹד בִּשְׁמוֹ לְבָשָׂר וָדָם קֹדֶם שֶׁחָלַק לוֹ כָּבוֹד בַּעֲבוֹדַת הַתְּפִלָּה. וְכֵן אָסוּר לִכְרֹעַ לוֹ עַל בִּרְכָּיו אֲפִלּוּ בְּלֹא נְתִינַת שָׁלוֹם (כִּי זֶהוּ כָּבוֹד גָּדוֹל כְּמוֹ כָּבוֹד שֶׁבַּתְּפִלָּה שֶׁכּוֹרְעִים בָּהּ לְהַקָּדוֹשׁ בָּרוּךְ הוּא, וְאָסוּר לְהַקְדִּים כֵּן לְבָשָׂר וָדָם). וְהוּא שֶׁמַּשְׁכִּים לְפִתְחוֹ, אֲבָל אִם מְצָאוֹ בַּשּׁוּק אוֹ בְּבֵית הַכְּנֶסֶת — יָכוֹל לִכְרֹעַ לוֹ וְלִתֵּן לוֹ שָׁלוֹם מִן הַדִּין. אֲבָל נָהֲגוּ שֶׁלֹּא לִתֵּן שָׁלוֹם קֹדֶם הַתְּפִלָּה, אֲפִלּוּ מְצָאוֹ בַּשּׁוּק, אֲפִלּוּ בְּלֹא כְּרִיעָה, וַאֲפִלּוּ הוּא רָגִיל לִתֵּן לוֹ שָׁלוֹם בְּכָל יוֹם, אֶלָּא יֹאמַר ״צַפְרָא דְּמָרֵיהּ טַב״, שֶׁעַל יְדֵי שֶׁמְּשַׁנֶּה בַּלָּשׁוֹן — יִתֵּן לֵב שֶׁהוּא אָסוּר לְהִתְעַכֵּב בִּדְבָרִים אֲחֵרִים כְּלָל עַד שֶׁיִּתְפַּלֵּל, כְּמוֹ שֶׁיִּתְבָּאֵר. וְיֵשׁ אוֹסְרִין לִכְרֹעַ עַל בִּרְכָּיו אֲפִלּוּ מְצָאוֹ בַּשּׁוּק וַאֲפִלּוּ בְּלֹא נְתִינַת שָׁלוֹם כְּלָל. וְיֵשׁ לָחֹש לְדִבְרֵיהֶם. וּמִכָּל מָקוֹם יֵשׁ לְהָקֵל בָּזֶה אִם הִתְחִיל כְּבָר לְבָרֵךְ בִּרְכוֹת הַשַּׁחַר, הוֹאִיל וְאִסּוּר זֶה לֹא נִזְכַּר בַּתַּלְמוּד אֶלָּא הוּא מִסְּבָרָא. אֲבָל אִם הוּא מַשְׁכִּים לְפִתְחוֹ — אָסוּר לִכְרֹעַ לוֹ (אֲפִלּוּ בְּלֹא נְתִינַת שָׁלוֹם) עַד שֶׁיִּתְפַּלֵּל י״ח:

הַקְדָּמַת שָׁלוֹם. מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר אָסוּר לְהַקְדִּים לְפֶתַח חֲבֵרוֹ לִתֵּן לוֹ שָׁלוֹם (״חִדְלוּ לָכֶם מִן הָאָדָם״), אֲפִלּוּ אַחַר בִּרְכוֹת הַשַּׁחַר וְק״ש. יֹאמַר רַק ״צַפְרָא דְּמָרֵיהּ טַב״ אַגַּב עֵסֶק, כִּי שְׁמוֹ שֶׁל הַקָּבָּ״ה שָׁלוֹם. אָסוּר גַּם לִכְרֹעַ. יֵשׁ לְהָקֵל אַחַר בִּרְכוֹת הַשַּׁחַר, שֶׁהוּא מִסְּבָרָא.

סעיף ד אָסוּר לְהִתְעַסֵּק בִּצְרָכָיו וְלָצֵאת לַדֶּרֶךְ עַד שֶׁיִּתְפַּלֵּל.

אָסוּר לְאָדָם לְהִתְעַסֵּק בִּצְרָכָיו קֹדֶם תְּפִלַּת י״ח בַּשַּׁחַר, כְּדֵי שֶׁלֹּא יִפְנֶה לִבּוֹ לְשׁוּם דָּבָר עַד שֶׁיִּתְפַּלֵּל. וְיֵשׁ נוֹהֲגִין לְהָקֵל בָּזֶה לְאַחַר שֶׁאָמְרוּ בִּרְכוֹת הַשַּׁחַר קֹדֶם בָּרוּךְ שֶׁאָמַר. וְאֵינוֹ נָכוֹן. וְכֵן אָסוּר לָצֵאת לַדֶּרֶךְ בַּבֹּקֶר עַד שֶׁיִּתְפַּלֵּל י״ח, אֶלָּא אִם כֵּן יוֹצֵא בְּשַׁיָּרָא וְאֵינָהּ מַמְתֶּנֶת לוֹ:

עֲסָקָיו וְהַדֶּרֶךְ. אָסוּר לְהִתְעַסֵּק בִּצְרָכָיו קֹדֶם תְּפִלַּת י״ח, כְּדֵי שֶׁלֹּא יִפְנֶה לִבּוֹ ; וְהַנּוֹהֲגִין לְהָקֵל אַחַר בִּרְכוֹת הַשַּׁחַר — אֵינוֹ נָכוֹן. וְכֵן אָסוּר לָצֵאת לַדֶּרֶךְ עַד שֶׁיִּתְפַּלֵּל, אֶלָּא בְּשַׁיָּרָא שֶׁאֵינָהּ מַמְתֶּנֶת.

סעיף ה אָסוּר לֶאֱכֹל וְלִשְׁתּוֹת ; מַיִם וּרְפוּאָה מֻתָּרִין.

אָסְרוּ חֲכָמִים לֶאֱכֹל אוֹ לִשְׁתּוֹת מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר עַד שֶׁיִּתְפַּלֵּל י״ח. וְסָמְכוּ אִסּוּר זֶה עַל מַה שֶּׁכָּתוּב: ״לֹא תֹאכְלוּ עַל הַדָּם״, כְּלוֹמַר, לֹא תֹּאכְלוּ קֹדֶם שֶׁתִּתְפַּלְלוּ עַל דַּמְכֶם. וְכָל הָאוֹכֵל וְשׁוֹתֶה וְאַחַר כָּךְ מִתְפַּלֵּל — עָלָיו הַכָּתוּב אוֹמֵר: ״וְאֹתִי הִשְׁלַכְתָּ אַחֲרֵי גַוֶּךָ״, אַל תִּקְרָא ״גַּוֶּךָ״, שֶׁמַּשְׁמָעוֹ כָּל צָרְכֵי גּוּפְךָ, אֶלָּא ״גֵּאֶיךָ״ — לְשׁוֹן גַּאֲוָה, אָמַר הַקָּבָּ״ה: לְאַחַר שֶׁאָכַל וְשָׁתָה זֶה וְנִתְגָּאָה — קִבֵּל עָלָיו עֹל מַלְכוּת שָׁמַיִם. לְפִיכָךְ, מֻתָּר לִשְׁתּוֹת מַיִם קֹדֶם הַתְּפִלָּה אַף עַל פִּי שֶׁהֵם צֹרֶךְ הַגּוּף, הוֹאִיל וְלֹא שַׁיָּךְ בָּהֶם גַּאֲוָה. וַאֲפִלּוּ בְּשַׁבָּת וְיוֹם טוֹב שֶׁצָּרִיךְ לְקַדֵּשׁ עַל הַכּוֹס לִפְנֵי הַסְּעוּדָה — אֵין הַקִּדּוּשׁ אוֹסְרוֹ מִלִּשְׁתּוֹת קֹדֶם תְּפִלַּת הַשַּׁחַר, כֵּיוָן שֶׁעֲדַיִן לֹא הִגִּיעַ זְמַנּוֹ, שֶׁאֵין קִדּוּשׁ אֶלָּא בִּמְקוֹם סְעוּדָה וְסָמוּךְ לָהּ, וַהֲרֵי אִי אֶפְשָׁר לוֹ לֶאֱכֹל כְּלוּם קֹדֶם תְּפִלַּת הַשַּׁחַר. וַאֲפִלּוּ כָּל מִינֵי אֳכָלִים וְכָל מִינֵי מַשְׁקִין שֶׁשַּׁיָּךְ בָּהֶם גַּאֲוָה — מֻתָּר לְאָכְלָם וְלִשְׁתּוֹתָם לְצֹרֶךְ רְפוּאָה, שֶׁאֵין בָּרְפוּאָה מִשּׁוּם גַּאֲוָה. אֲבָל לִרְעָבוֹ וְלִצְמָאוֹ — אָסוּר. אֶלָּא אִם כֵּן אֵינוֹ יָכוֹל לְכַוֵּן דַּעְתּוֹ בִּתְפִלָּתוֹ עַד שֶׁיֹּאכַל אוֹ שֶׁיִּשְׁתֶּה. וַאֲפִלּוּ עַכְשָׁו שֶׁאֵין מְכַוְּנִים כָּל כָּךְ בַּתְּפִלָּה, מִכָּל מָקוֹם אִם רוֹצֶה לֶאֱכֹל וְלִשְׁתּוֹת כְּדֵי לְכַוֵּן — הָרְשׁוּת בְּיָדוֹ:

אֲכִילָה וּשְׁתִיָּה. אָסְרוּ חֲכָמִים לֶאֱכֹל וְלִשְׁתּוֹת עַד שֶׁיִּתְפַּלֵּל (״לֹא תֹאכְלוּ עַל הַדָּם״). מַיִם מֻתָּר שֶׁאֵין בּוֹ גַּאֲוָה, אֲפִלּוּ בְּשַׁבָּת קֹדֶם הַקִּדּוּשׁ ; וּרְפוּאָה מֻתֶּרֶת ; לִרְעָבוֹ וְלִצְמָאוֹ אָסוּר, אֶלָּא אִם אֵינוֹ יָכוֹל לְכַוֵּן דַּעְתּוֹ.

סעיף ו הִתְחִיל לֶאֱכֹל קֹדֶם עֲלוֹת הַשַּׁחַר — צָרִיךְ לְהַפְסִיק.

אִם הִתְחִיל לֶאֱכֹל קֹדֶם עֲלוֹת הַשַּׁחַר — צָרִיךְ לְהַפְסִיק כְּשֶׁעוֹלֶה עַמּוּד הַשַּׁחַר. וְאַף עַל פִּי שֶׁבִּתְפִלַּת הַמִּנְחָה אֵין צָרִיךְ לְהַפְסִיק, מִכָּל מָקוֹם כָּאן הוֹאִיל וְאִסּוּר הָאֲכִילָה קֹדֶם הַתְּפִלָּה הִסְמִיכוּהוּ חֲכָמִים עַל לֹא תַּעֲשֶׂה שֶׁבַּתּוֹרָה שֶׁהוּא ״לֹא תֹאכְלוּ עַל הַדָּם״ — צָרִיךְ לְהַפְסִיק אֲפִלּוּ הִתְחִיל בְּהֶתֵּר. וְיֵשׁ אוֹמְרִים שֶׁאֵין צָרִיךְ לְהַפְסִיק לִתְפִלָּה אֲפִלּוּ הִתְחִיל בְּאִסּוּר, אֶלָּא לִקְרִיאַת שְׁמַע בִּלְבַד שֶׁהִיא מִן הַתּוֹרָה צָרִיךְ לְהַפְסִיק אִם הִתְחִיל בְּאִסּוּר, וְקוֹרֵא קְרִיאַת שְׁמַע בְּלֹא בִּרְכוֹתֶיהָ, שֶׁבִּרְכוֹתֶיהָ הֵן מִדִּבְרֵי סוֹפְרִים. וְהָעִקָּר כַּסְּבָרָא הָרִאשׁוֹנָה. וְאִם בֵּרַךְ בִּרְכַּת הַמָּזוֹן וְעוֹסֵק בִּשְׁתִיָּה — צָרִיךְ לְהַפְסִיק לְדִבְרֵי הַכֹּל:

הִתְחִיל לֶאֱכֹל. הִתְחִיל לֶאֱכֹל קֹדֶם עֲלוֹת הַשַּׁחַר — צָרִיךְ לְהַפְסִיק בָּעֲלוֹת עַמּוּד הַשַּׁחַר, שֶׁנִּסְמַךְ אִסּוּרוֹ עַל ״לֹא תֹאכְלוּ עַל הַדָּם״, אֲפִלּוּ הִתְחִיל בְּהֶתֵּר ; וְיֵשׁ אוֹמְרִים שֶׁמַּפְסִיק רַק לְק״ש, וְהָעִקָּר כַּסְּבָרָא הָרִאשׁוֹנָה. בֵּרַךְ בִּרְכַּת הַמָּזוֹן וְעוֹסֵק בִּשְׁתִיָּה — מַפְסִיק לְדִבְרֵי הַכֹּל.

סעיף ז אֲפִלּוּ לִלְמֹד אָסוּר ; מְלַמֵּד לַאֲחֵרִים — מֻתָּר.

אֲפִלּוּ לִלְמֹד אָסוּר מִשֶּׁהִגִּיעַ זְמַן הַתְּפִלָּה אִם הוּא רָגִיל לְהִתְפַּלֵּל בְּבֵית מִדְרָשׁוֹ בְּיָחִיד וְאֵינוֹ רָגִיל לֵילֵךְ לְבֵית הַכְּנֶסֶת, יֵשׁ לָחֹש שֶׁמָּא יִמָּשֵׁךְ בְּלִמּוּדוֹ וְיַעֲבֹר זְמַן קְרִיאַת שְׁמַע וּתְפִלָּה. אֲבָל מִי שֶׁרָגִיל לֵילֵךְ לְבֵית הַכְּנֶסֶת — מֻתָּר לִלְמֹד בְּבֵיתוֹ עַד שְׁעַת תְּפִלַּת הַצִּבּוּר. וְאִם הוּא מְלַמֵּד לַאֲחֵרִים, אֲפִלּוּ אֵינוֹ רָגִיל לֵילֵךְ לְבֵית הַכְּנֶסֶת — מֻתָּר, כֵּיוָן שֶׁהַשָּׁעָה עוֹבֶרֶת, שֶׁזְּכוּת הָרַבִּים דָּבָר גָּדוֹל הוּא, וְאִם לֹא יִלְמְדוּ עַכְשָׁו יִתְבַּטְּלוּ וְלֹא יוּכְלוּ לִלְמֹד אַחַר כָּךְ. וּלְפִיכָךְ, אֲפִלּוּ אִם נִמְשַׁךְ בְּלִמּוּדוֹ עַד שֶׁזְּמַן קְרִיאַת שְׁמַע וּתְפִלָּה עוֹבֵר — לֹא יַפְסִיק מִלִּמּוּדוֹ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ק״ו:

תַּלְמוּד תּוֹרָה. אֲפִלּוּ לִלְמֹד אָסוּר לְמִי שֶׁרָגִיל לְהִתְפַּלֵּל בְּיָחִיד בְּבֵית מִדְרָשׁוֹ (שֶׁמָּא יִמָּשֵׁךְ) ; הָרָגִיל לְבֵית הַכְּנֶסֶת — מֻתָּר עַד תְּפִלַּת הַצִּבּוּר ; וּמְלַמֵּד לַאֲחֵרִים — מֻתָּר, שֶׁזְּכוּת הָרַבִּים דָּבָר גָּדוֹל, וְאֵינוֹ מַפְסִיק (סימן ק״ו).

סעיף ח מֻתָּר לְהִסְתַּפֵּר וְלִכָּנֵס לַמֶּרְחָץ סָמוּךְ לְשַׁחֲרִית.

מֻתָּר לִכָּנֵס לַמֶּרְחָץ וּלְהִסְתַּפֵּר סָמוּךְ לְשַׁחֲרִית, שֶׁלֹּא גָּזְרוּ אֶלָּא סָמוּךְ לְמִנְחָה שֶׁהוּא דָּבָר הַמָּצוּי שֶׁרֹב הָעָם נִכְנָסִין שָׁם בַּיּוֹם, אֲבָל בַּשַּׁחַר דָּבָר שֶׁאֵינוֹ מָצוּי הוּא וְלֹא גָּזְרוּ בּוֹ. אֲבָל לַהֲפֹךְ בְּעוֹרוֹת וְכָל מְלָאכָה כַּיּוֹצֵא בָּהּ שֶׁאָסְרוּ סָמוּךְ לְמִנְחָה — אָסוּר גַּם סָמוּךְ לְשַׁחֲרִית, שֶׁהוּא דָּבָר הַמָּצוּי שֶׁאָדָם מַשְׁכִּים לִמְלַאכְתּוֹ קֹדֶם עֲלוֹת הַשַּׁחַר:

מֶרְחָץ וְתִסְפֹּרֶת. מֻתָּר לִכָּנֵס לַמֶּרְחָץ וּלְהִסְתַּפֵּר סָמוּךְ לְשַׁחֲרִית, שֶׁלֹּא גָּזְרוּ אֶלָּא סָמוּךְ לְמִנְחָה שֶׁהוּא דָּבָר הַמָּצוּי בַּיּוֹם ; אֲבָל לַהֲפֹךְ בְּעוֹרוֹת וְכָל מְלָאכָה כַּיּוֹצֵא בָּהּ — אָסוּר גַּם סָמוּךְ לְשַׁחֲרִית, שֶׁאָדָם מַשְׁכִּים לִמְלַאכְתּוֹ.

סעיף ט שְׁעַת הַדְּחָק לַדֶּרֶךְ — מִתְפַּלֵּל מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר.

בִּשְׁעַת הַדְּחָק, כְּגוֹן שֶׁצָּרִיךְ לְהַשְׁכִּים לָצֵאת לַדֶּרֶךְ, בֵּין בְּרַגְלָיו בֵּין בְּקָרוֹן אוֹ בִּסְפִינָה, וְאֵין הַשַּׁיָּרָא מַמְתֶּנֶת לוֹ עַד אַחַר נֵץ הַחַמָּה לְהִתְפַּלֵּל תְּפִלַּת י״ח כְּמִצְוָתָהּ בְּבֵיתוֹ מְעֻמָּד — יָכוֹל לְהִתְפַּלֵּל מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר קֹדֶם צֵאתוֹ. וְאִם הֵאִיר כְּבָר הַיּוֹם עַד שֶׁיַּכִּיר אֶת חֲבֵרוֹ בְּרִחוּק ד׳ אַמּוֹת שֶׁהוּא זְמַן קְרִיאַת שְׁמַע — יִקְרָא גַּם קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ בְּבֵיתוֹ, וְיִסְמֹךְ גְּאֻלָּה לִתְפִלָּה. וְאִם לֹא הֵאִיר הַיּוֹם כָּל כָּךְ — יִתְפַּלֵּל י״ח לְבַדָּהּ בְּבֵיתוֹ מְעֻמָּד, וְיַמְתִּין מִלִּקְרוֹת קְרִיאַת שְׁמַע עַד שֶׁיַּגִּיעַ זְמַנָּהּ אִם אֶפְשָׁר לִקְרוֹתָהּ בַּדֶּרֶךְ כְּמִצְוָתָהּ, דְּהַיְנוּ שֶׁיַּעֲמֹד וִיכַוֵּן בְּפָסוּק רִאשׁוֹן כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן נ״ח. אֲבָל תְּפִלָּה שֶׁהִיא צְרִיכָה עֲמִידָה בְּכֻלָּהּ — מִן הַסְּתָם לֹא יִזְדַּמֵּן לוֹ בַּדֶּרֶךְ לְהִתְפַּלֵּל מְעֻמָּד (שֶׁבְּנֵי הַשַּׁיָּרָא לֹא יַמְתִּינוּ לוֹ, אוֹ שֶׁעַל יְדֵי כָּךְ יִטָּרֵד וְלֹא יוּכַל לְכַוֵּן כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן צ״ד). לְפִיכָךְ צָרִיךְ לְהִתְפַּלֵּל מְעֻמָּד קֹדֶם צֵאתוֹ, אַף עַל פִּי שֶׁאֵינוֹ סוֹמֵךְ גְּאֻלָּה לִתְפִלָּה, וְגַם עֲדַיִן לֹא הִגִּיעַ עִקַּר זְמַנָּהּ שֶׁהוּא נֵץ הַחַמָּה, לְפִי שֶׁתְּפִלָּה מְעֻמָּד שְׁקוּלָה כְּנֶגֶד כֻּלָּם וּגְדוֹלָה מֵהֶם. וְעַכְשָׁו אֵין רֹב הָעוֹלָם נִזְהָרִין בָּזֶה, לְפִי שֶׁתְּפִלָּה מְעֻמָּד אֵינָהּ גְּדוֹלָה מִסְּמִיכַת גְּאֻלָּה לִתְפִלָּה אֶלָּא מִשּׁוּם שֶׁעוֹמֵד יָכוֹל לְכַוֵּן הֵיטֵב, וְעַכְשָׁו אֵינָן מְכַוְּנִים כָּל כָּךְ בַּתְּפִלָּה. וּמִכָּל מָקוֹם, אִם הוּא בַּדֶּרֶךְ וְיָרֵא שֶׁמָּא יַעֲבֹר זְמַן קְרִיאַת שְׁמַע וְיָדוּעַ לוֹ שֶׁיַּגִּיעַ לַמָּלוֹן קֹדֶם שֶׁיַּעֲבֹר זְמַן תְּפִלָּה — יָכוֹל לִקְרוֹת קְרִיאַת שְׁמַע בִּלְבַדָּהּ, וּלְהַמְתִּין עִם בִּרְכוֹתֶיהָ וְעִם הַתְּפִלָּה עַד שֶׁיַּגִּיעַ לַמָּלוֹן, כְּדֵי לְהִתְפַּלֵּל מְעֻמָּד וְלִסְמֹךְ גְּאֻלָּה לִתְפִלָּה (אֲבָל מִן הַדִּין אֵין צָרִיךְ לְהַמְתִּין כְּלָל, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן צ״ד):

שְׁעַת הַדְּחָק. הַצָּרִיךְ לְהַשְׁכִּים לַדֶּרֶךְ וְאֵין הַשַּׁיָּרָא מַמְתֶּנֶת — מִתְפַּלֵּל מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר קֹדֶם צֵאתוֹ. הֵאִיר הַיּוֹם — קוֹרֵא גַּם ק״ש וּבִרְכוֹתֶיהָ וְסוֹמֵךְ גְּאֻלָּה לִתְפִלָּה ; וְאִם לָאו — י״ח לְבַדָּהּ מְעֻמָּד, שֶׁשְּׁקוּלָה כְּנֶגֶד כֻּלָּם. וְעַכְשָׁו אֵין רֹב הָעוֹלָם נִזְהָרִין (סימן צ״ד).

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim פ״ט (מהדורת קה״ת, ט׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — זְמַן תְּפִלַּת הַשַּׁחַר

היסוד : זְמַן תְּפִלַּת הַשַּׁחַר

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה זְמַן הַתְּפִלָּה — נֵץ הַחַמָּה, עַמּוּד הַשַּׁחַר וּשְׁלִישׁ הַיּוֹם. אחר כך הָעַד חֲצוֹת וְהַתַּשְׁלוּמִין. אחר כך אִסּוּר הַהַקְדָּמָה קֹדֶם תְּפִלָּה — שָׁלוֹם, עֲסָקִים, אֲכִילָה וְלִמּוּד. ולבסוף הַשְּׁעַת הַדְּחָק וְהַיּוֹצֵא לַדֶּרֶךְ.

זְמַן הַתְּפִלָּה — נֵץ הַחַמָּה וְעַמּוּד הַשַּׁחַר

זְמַן הַתְּפִלָּה — נֵץ הַחַמָּה וְעַמּוּד הַשַּׁחַר

היסוד. זְמַן שַׁחֲרִית מֵעֲלוֹת הַשַּׁחַר ; לְכַתְּחִלָּה מִצְוָה עִם נֵץ הַחַמָּה (״יִירָאוּךָ עִם שָׁמֶשׁ״), וּבְדִיעֲבַד מֵעַמּוּד הַשַּׁחַר ; עַד שְׁלִישׁ הַיּוֹם (סעיף א).

המסקנה. צָרִיךְ לִזָּהֵר בַּחֹרֶף לְסַיֵּם י״ח קֹדֶם שְׁלִישׁ הַיּוֹם. עיין סימן נ״ח · 58 (זמן קריאת שמע).

עַד חֲצוֹת וְהַתַּשְׁלוּמִין

עַד חֲצוֹת וְהַתַּשְׁלוּמִין

היסוד. אֲפִלּוּ לְכַתְּחִלָּה עַד חֲצוֹת ; אַחַר חֲצוֹת מִיָּד זְמַן מִנְחָה (סעיף ב).

המסקנה. אַחַר חֲצוֹת אֵין תַּשְׁלוּמִין אֶלָּא בְּטָעוּת אוֹ אֹנֶס בְּמִנְחָה שְׁתַּיִם. עיין סימן ק״ח · 108 (תשלומין).

אֵין מַקְדִּימִין דָּבָר לַתְּפִלָּה

אֵין מַקְדִּימִין דָּבָר לַתְּפִלָּה

היסוד. אָסוּר שָׁלוֹם, עֲסָקִים, אֲכִילָה וְלִמּוּד קֹדֶם תְּפִלָּה (סעיפים ג-ז).

המסקנה. עִקַּר כְּבוֹד הַמָּקוֹם הִיא הַתְּפִלָּה ; מַיִם וּרְפוּאָה מֻתָּרִין, מְלַמֵּד לַאֲחֵרִים מֻתָּר. עיין סימן צ׳ · 90 (בית הכנסת).

שְׁעַת הַדְּחָק וְהַיּוֹצֵא לַדֶּרֶךְ

שְׁעַת הַדְּחָק וְהַיּוֹצֵא לַדֶּרֶךְ

היסוד. שְׁעַת הַדְּחָק לַדֶּרֶךְ — מֵעַמּוּד הַשַּׁחַר קֹדֶם צֵאתוֹ (סעיף ט).

המסקנה. תְּפִלָּה מְעֻמָּד שְׁקוּלָה כְּנֶגֶד כֻּלָּם. עיין סימן צ״ד · 94 (תפילה בדרך).

הפניה ללימוד. היקף הסימן — הַזְּמַן, הָעַד חֲצוֹת, הָאִסּוּר קֹדֶם תְּפִלָּה וְהַשְּׁעַת הַדְּחָק — נוגע להנהגת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן פ״ט. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מט׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
זְמַן תְּפִלַּת הַשַּׁחַר או״ח פ״ט — מִנֵּץ הַחַמָּה לְכַתְּחִלָּה, מֵעַמּוּד הַשַּׁחַר בְּדִיעֲבַד ; עַד סוֹף ד׳ שָׁעוֹת (שְׁלִישׁ הַיּוֹם). מ״ב פ״ט — פְּרָטֵי הַזְּמַן וְאַזְהָרָה עַל הַסְּלִיחוֹת קֹדֶם נֵץ. ערוה״ש פ״ט — חֶשְׁבּוֹן הַשָּׁעוֹת וּזְמַנֵּי הַתְּפִלָּה. דרך אדמו״ר הזקן — לְכַתְּחִלָּה עִם נֵץ הַחַמָּה (״יִירָאוּךָ עִם שָׁמֶשׁ״) ; בְּדִיעֲבַד מֵעֲלוֹת הַשַּׁחַר ; עַד שְׁלִישׁ הַיּוֹם (סעיף א).
שַׁחֲרִית עַד חֲצוֹת וְתַשְׁלוּמִין או״ח פ״ט — אַחַר ד׳ שָׁעוֹת עַד חֲצוֹת שְׂכַר תְּפִלָּה ; אַחַר חֲצוֹת אָסוּר. מ״ב פ״ט — דִּין הַמֵּזִיד וּפְרָטֵי הַתַּשְׁלוּמִין. ערוה״ש פ״ט — גֶּדֶר הַתַּשְׁלוּמִין בְּמִנְחָה. דרך אדמו״ר הזקן — אֲפִלּוּ לְכַתְּחִלָּה עַד חֲצוֹת ; אַחַר חֲצוֹת מִיָּד — מִנְחָה שְׁתַּיִם רַק בְּטָעוּת אוֹ אֹנֶס (סימן ק״ח) (סעיף ב).
הַקְדָּמַת שָׁלוֹם וַעֲסָקִים קֹדֶם תְּפִלָּה או״ח פ״ט — אָסוּר לְהַקְדִּים לְפֶתַח חֲבֵרוֹ ; צַפְרָא דְּמָרֵיהּ טַב מֻתָּר ; אָסוּר בַּעֲסָקָיו וּבַדֶּרֶךְ. מ״ב פ״ט — פְּרָטֵי הַשָּׁלוֹם וְהַכְּרִיעָה. ערוה״ש פ״ט — גֶּדֶר הָאִסּוּר וּמִנְהַג הָעוֹלָם. דרך אדמו״ר הזקן — אָסוּר שָׁלוֹם וּכְרִיעָה (״שְׁמוֹ שֶׁל הַקָּבָּ״ה שָׁלוֹם״) ; יֵשׁ לְהָקֵל אַחַר בִּרְכוֹת הַשַּׁחַר, שֶׁהוּא מִסְּבָרָא (סעיפים ג-ד).
אֲכִילָה, לִמּוּד וְשַׁיָּרָא (שְׁעַת הַדְּחָק) או״ח פ״ט — אָסוּר לֶאֱכֹל ; מַיִם וּרְפוּאָה מֻתָּר ; לִלְמֹד אָסוּר לְמִי שֶׁאֵין רָגִיל ; שְׁעַת הַדְּחָק מֵעַמּוּד הַשַּׁחַר. מ״ב פ״ט — פְּרָטֵי הָאֲכִילָה וְהַהַפְסָקָה. ערוה״ש פ״ט — דִּין הַיּוֹצֵא לַדֶּרֶךְ בַּשַּׁיָּרָא. דרך אדמו״ר הזקן — הִתְחִיל לֶאֱכֹל — יַפְסִיק ; מְלַמֵּד לַאֲחֵרִים מֻתָּר (סימן ק״ו) ; שְׁעַת הַדְּחָק — מְעֻמָּד קֹדֶם צֵאתוֹ, שְׁקוּלָה כְּנֶגֶד כֻּלָּם (סעיפים ה-ט).

הטבלה נערכה מטקסט המחבר (אורח חיים פ״ט, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה ג, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ו-ל׳). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ט׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — זְמַן תְּפִלַּת הַשַּׁחַר

קו א — זְמַן תְּפִלַּת הַשַּׁחַר

יסוד הסימן : מִנֵּץ הַחַמָּה לְכַתְּחִלָּה, מֵעַמּוּד הַשַּׁחַר בְּדִיעֲבַד ; עַד שְׁלִישׁ הַיּוֹם (סעיף א).

קריאת הרב : הַמִּצְוָה עִם נֵץ הַחַמָּה, וּמַזְהִיר עַל הַהַקְדָּמָה בִּימֵי הַסְּלִיחוֹת. עיין סימן נ״ח · 58.

מסקנה למעשה : לְהִזָּהֵר בַּחֹרֶף לְסַיֵּם י״ח קֹדֶם שְׁלִישׁ הַיּוֹם.

קו ב — אֵין מַקְדִּימִין דָּבָר לַתְּפִלָּה

קו ב — אֵין מַקְדִּימִין דָּבָר לַתְּפִלָּה

יסוד הסימן : אָסוּר שָׁלוֹם, עֲסָקָיו, אֲכִילָה וְלִמּוּד קֹדֶם תְּפִלָּה (סעיפים ג-ז).

קריאת הרב : עִקַּר כְּבוֹד הַמָּקוֹם הִיא הַתְּפִלָּה. עיין סימן צ׳ · 90.

מסקנה למעשה : מַיִם וּרְפוּאָה מֻתָּרִין ; מְלַמֵּד לַאֲחֵרִים מֻתָּר (זְכוּת הָרַבִּים).

קו ג — שְׁעַת הַדְּחָק וְהַכַּוָּנָה

קו ג — שְׁעַת הַדְּחָק וְהַכַּוָּנָה

יסוד הסימן : שְׁעַת הַדְּחָק לַדֶּרֶךְ — מֵעַמּוּד הַשַּׁחַר (סעיף ט).

קריאת הרב : תְּפִלָּה מְעֻמָּד שְׁקוּלָה כְּנֶגֶד כֻּלָּם — עָדִיף לְהִתְפַּלֵּל מְעֻמָּד בְּבֵיתוֹ. עיין סימן צ״ד · 94.

מסקנה למעשה : עַכְשָׁו אֵין רֹב הָעוֹלָם נִזְהָרִין, שֶׁהָעִקָּר הַכַּוָּנָה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן פ״ט ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן פ״ט (זְמַן תְּפִלַּת הַשַּׁחַר)

  • ① זְמַן. לְהִתְפַּלֵּל שַׁחֲרִית לְכַתְּחִלָּה עִם נֵץ הַחַמָּה ; בְּדִיעֲבַד מֵעֲלוֹת הַשַּׁחַר ; לְסַיֵּם קֹדֶם שְׁלִישׁ הַיּוֹם. יסוד : שוע״ר פ״ט סעיף א.
  • ② עַד חֲצוֹת. עָבַר הַזְּמַן — עַד חֲצוֹת ; אַחַר כָּךְ מִנְחָה שְׁתַּיִם רַק בְּטָעוּת אוֹ אֹנֶס. יסוד : שוע״ר פ״ט סעיף ב.
  • ③ שָׁלוֹם. לֹא לִתֵּן שָׁלוֹם וְלֹא לִכְרֹעַ קֹדֶם תְּפִלָּה ; רַק ״צַפְרָא דְּמָרֵיהּ טַב״. יסוד : שוע״ר פ״ט סעיף ג.
  • ④ עֲסָקָיו וְדֶרֶךְ. אֵין עֲסָקִים וְאֵין יְצִיאָה לַדֶּרֶךְ קֹדֶם י״ח, אֶלָּא בְּשַׁיָּרָא. יסוד : שוע״ר פ״ט סעיף ד.
  • ⑤ אֲכִילָה. אֵין אוֹכְלִין וְשׁוֹתִין ; מַיִם וּרְפוּאָה מֻתָּרִין ; הִתְחִיל — יַפְסִיק. יסוד : שוע״ר פ״ט סעיפים ה-ו.
  • ⑥ לִמּוּד. אֲפִלּוּ לִלְמֹד אָסוּר לַיָּחִיד ; מְלַמֵּד לַאֲחֵרִים מֻתָּר. יסוד : שוע״ר פ״ט סעיף ז.
  • ⑦ מֶרְחָץ / שְׁעַת הַדְּחָק. מֻתָּר מֶרְחָץ וְתִסְפֹּרֶת סָמוּךְ לְשַׁחֲרִית ; בִּשְׁעַת הַדְּחָק לַדֶּרֶךְ — מְעֻמָּד מֵעַמּוּד הַשַּׁחַר. יסוד : שוע״ר פ״ט סעיפים ח-ט.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן פ״ט (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַזְּמַן, הָעַד חֲצוֹת, הָאִסּוּר קֹדֶם תְּפִלָּה, הַשְּׁעַת הַדְּחָק — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
