דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קי״ג · דִּינֵי הַכְּרִיעוֹת בְּי״ח בְּרָכוֹת

דרכו של אדמו״ר הזקן בדִּינֵי הַכְּרִיעוֹת בְּי״ח בְּרָכוֹת : מְקוֹם הַכְּרִיעוֹת — בְּ״אָבוֹת״ וּבְ״הוֹדָאָה״ תְּחִלָּה וָסוֹף ; שִׁעוּר הַכְּרִיעָה — עַד שֶׁיִּתְפַּקְּקוּ חֻלְיוֹת הַשִּׁדְרָה וְלֹא כִּיוּהֲרָא ; כּוֹרֵעַ בְּ״בָרוּךְ״ וְזוֹקֵף בַּשֵּׁם ; וּמַרְאִית הָעַיִן וּמַטְבֵּעַ הַתְּאָרִים — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קי״ג — דִּינֵי הַכְּרִיעוֹת בְּי״ח בְּרָכוֹת. אצל הרב הסימן נפרש לט׳ סְעִיפִים : הַמְּקוֹם הַכְּרִיעוֹת (בְּ״אָבוֹת״ וּבְ״הוֹדָאָה״ תְּחִלָּה וָסוֹף), הַכְּרִיעָה בְּרֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר, הַכּוֹרֵעַ בְּמָקוֹם שֶׁלֹּא תִּקְּנוּ, הַשִּׁעוּר וְהָאֹפֶן, הַכּוֹרֵעַ בְּ״בָרוּךְ״ וְזוֹקֵף בַּשֵּׁם, הַמַּרְאִית הָעַיִן, וְהַמַּטְבֵּעַ שֶׁטָּבְעוּ חֲכָמִים עִם הַמַּשְׁתִּיקִין.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קי״ג — ט׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קי״ג). אדמו״ר הזקן דן בו בְּדִינֵי הַכְּרִיעוֹת בְּי״ח בְּרָכוֹת. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קי״ג — דִּינֵי הַכְּרִיעוֹת בְּי״ח בְּרָכוֹת

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קי״ג — דִּינֵי הַכְּרִיעוֹת בְּי״ח בְּרָכוֹת — וּבוֹ ט׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.113), ואחריו תרגומנו. הסימן כולל ט׳ סעיפים (סעיף k → ‎.113.k).

סעיף א אֵלּוּ בְּרָכוֹת שֶׁשּׁוֹחִין בָּהֶן — ״אָבוֹת״ וְ״הוֹדָאָה״ תְּחִלָּה וָסוֹף.

דִּינֵי הַכְּרִיעוֹת בְּי״ח בְּרָכוֹת וּבוֹ ט׳ סְעִיפִים:אֵלּוּ בְּרָכוֹת שֶׁשּׁוֹחִין בָּהֶן: בְּ״אָבוֹת״ תְּחִלָּה וָסוֹף, וּבְ״הוֹדָאָה״ תְּחִלָּה וָסוֹף. וְאִם בָּא לִשְׁחוֹת בְּסוֹף כָּל בְּרָכָה אוֹ בִּתְחִלָּתָהּ — מְלַמְּדִין אוֹתוֹ שֶׁלֹּא יִשְׁחֶה, שֶׁלֹּא יָבוֹא לַעֲקֹר תַּקָּנַת חֲכָמִים, שֶׁלֹּא יֹאמְרוּ כָּל אֶחָד מַחֲמִיר כְּמוֹ שֶׁהוּא רוֹצֶה וְאֵין כָּאן תַּקָּנַת חֲכָמִים כְּלָל לִשְׁחוֹת, וְיָבֹאוּ לְהָקֵל עַל יְדֵי כָּךְ בְּ״אָבוֹת״ וּבְ״הוֹדָאָה״. אֲבָל בְּאֶמְצָעִיתָן מֻתָּר לִשְׁחוֹת, שֶׁאֵין לָחֹש שֶׁיָּבוֹאוּ לַעֲקֹר תַּקָּנַת חֲכָמִים עַל יְדֵי זֶה, כֵּיוָן שֶׁחֲכָמִים לֹא תִּקְּנוּ כְּלָל לִשְׁחוֹת בְּאֶמְצַע שׁוּם בְּרָכָה:

אֵלּוּ בְּרָכוֹת שֶׁשּׁוֹחִין בָּהֶן. שׁוֹחִין בְּ״אָבוֹת״ (תְּחִלָּה וָסוֹף) וּבְ״הוֹדָאָה״ (תְּחִלָּה וָסוֹף). הַבָּא לִשְׁחוֹת בְּסוֹף כָּל בְּרָכָה אוֹ בִּתְחִלָּתָהּ — מְלַמְּדִין אוֹתוֹ שֶׁלֹּא יִשְׁחֶה, שֶׁלֹּא לַעֲקֹר תַּקָּנַת חֲכָמִים (שֶׁלֹּא יֹאמְרוּ כָּל אֶחָד מַחֲמִיר כִּרְצוֹנוֹ וְיָבֹאוּ לְהָקֵל אַף בְּ״אָבוֹת״ וּבְ״הוֹדָאָה״). אֲבָל בְּאֶמְצָעִיתָן — מֻתָּר לִשְׁחוֹת, שֶׁלֹּא תִּקְּנוּ כְּלָל לִשְׁחוֹת בְּאֶמְצַע שׁוּם בְּרָכָה.

סעיף ב הַכּוֹרֵעַ בְּ״זָכְרֵנוּ״ וּבְ״מִי כָמוֹךָ״ — צָרִיךְ לִזְקֹף בְּסוֹף הַבְּרָכָה.

הַנּוֹהֲגִים לִשְׁחוֹת בְּרֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר כְּשֶׁאוֹמְרִים ״זָכְרֵנוּ״ וְ״מִי כָמוֹךָ״ — צְרִיכִים לִזְקֹף כְּשֶׁמַּגִּיעִים לְסוֹף הַבְּרָכָה. וְאַף עַל פִּי שֶׁבְּאָבוֹת כּוֹרֵעַ בְּסוֹף הַבְּרָכָה, מִכָּל מָקוֹם צָרִיךְ לִזְקֹף מְעַט בְּסוֹף ״זָכְרֵנוּ״ קֹדֶם שֶׁיֹּאמַר אֵיזוֹ תֵּבוֹת הַסְּמוּכוֹת לְ״בָרוּךְ אַתָּה״, כְּדֵי שֶׁיּוּכַל אַחַר כָּךְ לַחֲזֹר וְלִשְׁחוֹת כְּשֶׁיֹּאמַר ״בָּרוּךְ אַתָּה״, שֶׁלֹּא דַּי בַּשְּׁחִיָּה הָרִאשׁוֹנָה מִפְּנֵי שֶׁצָּרִיךְ לִכְרֹעַ בְּ״בָרוּךְ״ כְּמוֹ שֶׁיִּתְבָּאֵר, וְעוֹד כְּדֵי שֶׁיִּהְיֶה נִרְאֶה שֶׁחוֹזֵר וְכוֹרֵעַ מִשּׁוּם חִיּוּב, שֶׁשְּׁחִיָּה רִאשׁוֹנָה הָיְתָה רְשׁוּת. וּבְסוֹף ״מִי כָמוֹךָ״ אֵין צָרִיךְ לִזְקֹף, עַד שֶׁיַּגִּיעַ לְ״בָרוּךְ אַתָּה״ שֶׁאָז זוֹקֵף מְעַט, כְּדֵי שֶׁלֹּא לְהוֹסִיף עַל תַּקָּנַת חֲכָמִים שֶׁלֹּא תִּקְּנוּ לִשְׁחוֹת אֶלָּא בְּ״בָרוּךְ אַתָּה״ שֶׁבְּ״אָבוֹת״ וְ״הוֹדָאָה״. וְכֵן הַנּוֹהֲגִים לְהִתְפַּלֵּל כָּל הַתְּפִלָּה בְּרֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר בִּכְרִיעָה — צְרִיכִים לִזְקֹף בְּסוֹף כָּל בְּרָכָה וּבִתְחִלַּת כָּל בְּרָכָה, שֶׁלֹּא לְהוֹסִיף עַל תַּקָּנַת חֲכָמִים שֶׁלֹּא תִּקְּנוּ אֶלָּא בִּתְחִלַּת אָבוֹת וְהוֹדָאָה וּבְסוֹפָן:

זָכְרֵנוּ וּמִי כָמוֹךָ. הַנּוֹהֲגִים לִשְׁחוֹת בְּרֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר בְּ״זָכְרֵנוּ״ וּבְ״מִי כָמוֹךָ״ — צְרִיכִים לִזְקֹף בְּסוֹף הַבְּרָכָה. אַף שֶׁבְּ״אָבוֹת״ כּוֹרֵעַ בַּסּוֹף, צָרִיךְ לִזְקֹף מְעַט בְּסוֹף ״זָכְרֵנוּ״ קֹדֶם ״בָּרוּךְ אַתָּה״, כְּדֵי לַחֲזֹר וְלִשְׁחוֹת בְּ״בָרוּךְ״ מִשּׁוּם חִיּוּב (שֶׁהָרִאשׁוֹנָה רְשׁוּת). וּבְ״מִי כָמוֹךָ״ זוֹקֵף רַק בְּ״בָרוּךְ אַתָּה״. וְהַמִּתְפַּלֵּל כָּל הַתְּפִלָּה בִּכְרִיעָה — זוֹקֵף בְּסוֹף וּבִתְחִלַּת כָּל בְּרָכָה, שֶׁלֹּא לְהוֹסִיף עַל הַתַּקָּנָה.

סעיף ג הַכּוֹרֵעַ בַּ״הוֹדָאוֹת״ וּבִ״וְכָל קוֹמָה״ — מְגֻנֶּה ; ״וַאֲנַחְנוּ כּוֹרְעִים״ — יָפֶה.

הַכּוֹרֵעַ בְּ״הוֹדָאָה״ שֶׁבְּבִרְכַּת הַמָּזוֹן וּבְ״הוֹדָאָה״ שֶׁל הַלֵּל שֶׁהִיא ״הוֹדוּ לַה׳ כִּי טוֹב וְגו׳״ — הֲרֵי זֶה מְגֻנֶּה. וְהוּא הַדִּין בִּשְׁאָר הוֹדָאוֹת, כְּגוֹן ״לְךָ לְבַדְּךָ אֲנַחְנוּ מוֹדִים״, ״וְעַתָּה אֱלֹקֵינוּ מוֹדִים אֲנַחְנוּ לָךְ״, שֶׁאֵין לִכְרֹעַ בְּ״בָרוּךְ אַתָּה״ וְלֹא בְּ״הוֹדָאָה״ אֶלָּא בִּמְקוֹמוֹת שֶׁתִּקְּנוּ חֲכָמִים, בְּ״אָבוֹת״ תְּחִלָּה וָסוֹף וּבְ״הוֹדָאָה״ שֶׁבַּתְּפִלָּה תְּחִלָּה וָסוֹף, מִפְּנֵי שֶׁלְּשׁוֹן ״הוֹדָאָה״ אֵינוֹ מִלְּשׁוֹן ״הִשְׁתַּחֲוָאָה״ כְּלָל, אֶלָּא מִלְּשׁוֹן ״הַחְזָקַת טוֹבָה״ וְ״הִלּוּל״, וְהַכּוֹרֵעַ מֵחֲמַת לְשׁוֹן הוֹדָאָה שֶׁמַּזְכִּיר — הֲרֵי זֶה מְגֻנֶּה וְסָכָל הוּא וּכְרִיעָתוֹ סִכְלוּת. אֲבָל מִי שֶׁכּוֹרֵעַ בְּאֵיזֶה מְקוֹמוֹת שֶׁל שְׁבָחִים אוֹ תַּחֲנוּנִים, אוֹ מֵחֲמַת אֵיזֶה טַעַם שֶׁיֵּשׁ לוֹ שֶׁנִּרְאֶה לוֹ שֶׁרָאוּי לִשְׁחוֹת בְּאוֹתוֹ מָקוֹם, אוֹ מֵחֲמַת הַמִּנְהָג כְּמוֹ הַכְּרִיעוֹת שֶׁבַּקַּדִּישׁ, אוֹ מֵחֲמַת שֶׁנִּתְלַהֵב לִבּוֹ בַּתְּפִלָּה — הֲרֵי זֶה מְשֻׁבָּח, וְאֵינוֹ כְּמוֹסִיף עַל תַּקָּנַת חֲכָמִים שֶׁלֹּא תִּקְּנוּ אֶלָּא בְּ״הוֹדָאָה״ שֶׁבַּתְּפִלָּה בִּלְבַדָּהּ, וַהֲרֵי זֶה דּוֹמֶה כְּמִי שֶׁכּוֹרֵעַ בְּאֶמְצַע בְּרָכוֹת שֶׁל תְּפִלָּה שֶׁאֵין בָּזֶה כְּמוֹסִיף עַל תַּקָּנַת חֲכָמִים, כֵּיוָן שֶׁלֹּא תִּקְּנוּ כְּלָל לִשְׁחוֹת בְּאֶמְצַע שׁוּם בְּרָכָה כְּמוֹ שֶׁנִּתְבָּאֵר. אֲבָל הַכּוֹרֵעַ בַּ״הוֹדָאוֹת״ הַנִּזְכָּרִים לְעֵיל מֵחֲמַת לְשׁוֹן הוֹדָאָה שֶׁמַּזְכִּיר — הֲרֵי זֶה סָכָל. וְכֵן הַכּוֹרֵעַ בִּ״וְכָל קוֹמָה לְפָנֶיךָ תִּשְׁתַּחֲוֶה״ — הֲרֵי זֶה גַּם כֵּן מְגֻנֶּה, מִפְּנֵי שֶׁכּוֹרֵעַ מֵחֲמַת לְשׁוֹן הִשְׁתַּחֲוָאָה שֶׁמַּזְכִּיר וּבֶאֱמֶת אֵין לִכְרֹעַ מִפְּנֵי זֶה. אֲבָל בְּ״וַאֲנַחְנוּ כּוֹרְעִים״ כֵּיוָן שֶׁאֵינוֹ אוֹמֵר עַל עַצְמוֹ אֶלָּא עַל כָּל הָעוֹלָם כֻּלּוֹ שֶׁמִּשְׁתַּחֲוִים לְפָנָיו יִתְבָּרַךְ — יָפֶה הֵם עוֹשִׂים. וּכְמוֹ כֵן מַה שֶּׁנּוֹהֲגִים לִכְרֹעַ בְּמוּסָף יוֹם הַכִּפּוּרִים בִּ״וְהַכֹּהֲנִים וְהָעָם וכו׳ הָיוּ כּוֹרְעִים״, גַּם כֵּן אֵינוֹ אוֹמֵר עַל עַצְמוֹ אֶלָּא לְזֵכֶר מַה שֶּׁעָשׂוּ הַכֹּהֲנִים וְהָעָם בִּזְמַן בֵּית הַמִּקְדָּשׁ וְאָנוּ מְדַמִּים עַצְמֵנוּ בִּמְקוֹם הַכֹּהֲנִים — גַּם כֵּן יָפֶה עוֹשִׂים. וְכֵן מַה שֶּׁאוֹמְרִים בְּסֵדֶר עֲבוֹדָה בְּמוּסָף יוֹם כִּפּוּר ״חָטְאוּ עָווּ פָּשְׁעוּ בְּנֵי יִשְׂרָאֵל״ — אָנוּ מְדַמִּין עַצְמֵנוּ כְּאִלּוּ אֲנַחְנוּ אוֹתָן בֵּית יִשְׂרָאֵל, וְיָפֶה עוֹשִׂים אוֹתָן הַמַּכִּים עַל לִבָּם כְּדֶרֶךְ שֶׁעוֹשִׂים בְּאָמְרָם ״אָשַׁמְנוּ״:

הַכּוֹרֵעַ בַּהוֹדָאוֹת. הַכּוֹרֵעַ בְּ״הוֹדָאָה״ שֶׁבְּבִרְכַּת הַמָּזוֹן וּבַהַלֵּל, וְכֵן שְׁאָר הוֹדָאוֹת — מְגֻנֶּה, שֶׁלְּשׁוֹן ״הוֹדָאָה״ אֵינוֹ הִשְׁתַּחֲוָאָה אֶלָּא הַחְזָקַת טוֹבָה ; וְהַכּוֹרֵעַ מֵחֲמַת הַלָּשׁוֹן — סָכָל. אֲבָל הַכּוֹרֵעַ מֵחֲמַת שְׁבָחִים/תַּחֲנוּנִים, מִנְהָג (כְּרִיעוֹת שֶׁבַּקַּדִּישׁ) אוֹ הִתְלַהֲבוּת — מְשֻׁבָּח. ״וְכָל קוֹמָה״ — מְגֻנֶּה ; ״וַאֲנַחְנוּ כּוֹרְעִים״ (עַל כָּל הָעוֹלָם) — יָפֶה, וְכֵן ״וְהַכֹּהֲנִים וְהָעָם״ בְּמוּסָף יוֹם הַכִּפּוּרִים.

סעיף ד יִכְרַע עַד שֶׁיִּתְפַּקְּקוּ חֻלְיוֹת הַשִּׁדְרָה, וְלֹא כִּיוּהֲרָא.

הַמִּתְפַּלֵּל צָרִיךְ שֶׁיִּכְרַע עַד שֶׁיִּתְפַּקְּקוּ כָּל חֻלְיוֹת שֶׁבַּשִּׁדְרָה. וְלֹא יִכְרַע מֵאֶמְצַע מָתְנָיו וְרֹאשׁוֹ יִשָּׁאֵר זָקוּף, אֶלָּא גַּם רֹאשׁוֹ יָכֹף כְּאַגְמוֹן. וְלֹא יִשְׁחֶה כָּל כָּךְ עַד שֶׁיִּהְיֶה פִּיו כְּנֶגֶד חֲגוֹר שֶׁל מִכְנָסַיִם, מִפְּנֵי שֶׁנִּרְאֶה כְּיוּהֲרָא שֶׁשּׁוֹחֶה יוֹתֵר מִדַּאי. וְזָקֵן וְחוֹלֶה שֶׁאֵינוֹ יָכוֹל לִשְׁחוֹת עַד שֶׁיִּתְפַּקְּקוּ, כֵּיוָן שֶׁהִרְכִּין רֹאשׁוֹ — דַּיּוֹ, מֵאַחַר שֶׁנִּכָּר שֶׁחָפֵץ לִכְרֹעַ אֶלָּא שֶׁמִּצְטַעֵר:

שִׁעוּר הַכְּרִיעָה. הַמִּתְפַּלֵּל יִכְרַע עַד שֶׁיִּתְפַּקְּקוּ כָּל חֻלְיוֹת הַשִּׁדְרָה, וְגַם רֹאשׁוֹ יָכֹף כְּאַגְמוֹן (לֹא רַק מֵאֶמְצַע מָתְנָיו). וְלֹא יִשְׁחֶה כָּל כָּךְ עַד שֶׁפִּיו כְּנֶגֶד הַחֲגוֹר, שֶׁנִּרְאֶה כְּיוּהֲרָא. זָקֵן וְחוֹלֶה שֶׁאֵינוֹ יָכוֹל — דַּיּוֹ בְּהַרְכָּנַת רֹאשׁוֹ, שֶׁנִּכָּר שֶׁחָפֵץ לִכְרֹעַ אֶלָּא שֶׁמִּצְטַעֵר.

סעיף ה כּוֹרֵעַ בְּ״בָרוּךְ״ וְזוֹקֵף בַּשֵּׁם — ״ה׳ זֹקֵף כְּפוּפִים״.

הַמִּתְפַּלֵּל כּוֹרֵעַ בְּ״בָרוּךְ״ וְזוֹקֵף בַּשֵּׁם, עַל שֵׁם ״ה׳ זֹקֵף כְּפוּפִים״. וּמִכָּל מָקוֹם, בַּכְּרִיעָה שֶׁבִּתְחִלַּת בְּרָכָה אִם רָצָה שֶׁלֹּא לִזְקֹף עַד סָמוּךְ לַחֲתִימָה — הָרְשׁוּת בְּיָדוֹ, אֶלָּא שֶׁאִם רָצָה לִזְקֹף מִיָּד — צָרִיךְ לִזְקֹף בַּשֵּׁם. וְכֵן בַּכְּרִיעָה שֶׁבְּסוֹף בְּרָכָה שֶׁהוּא חַיָּב לִזְקֹף מִיָּד כְּדֵי שֶׁלֹּא לִהְיוֹת שָׁחוּחַ בִּתְחִלַּת בְּרָכָה שֶׁלְּאַחֲרֶיהָ — אֲזַי צָרִיךְ הוּא לִזְקֹף בַּשֵּׁם, שֶׁמִּשּׁוּם ״ה׳ זֹקֵף כְּפוּפִים״ לֹא הִצְרִיכוּ לִזְקֹף מִכְּרִיעָתוֹ כְּלָל, אֶלָּא אִם מִבִּלְעֲדֵי כֵן הוּא זוֹקֵף, אֲזַי מִשּׁוּם ״ה׳ זֹקֵף כְּפוּפִים״ יִזְקֹף בַּשֵּׁם. וְלָכֵן הַנּוֹהֲגִין לְהִתְפַּלֵּל בְּרֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר בִּכְרִיעָה — אֵין צְרִיכִים לִזְקֹף אֶלָּא בְּסוֹף בְּרָכָה וּתְחִלָּתָהּ, כְּמוֹ שֶׁנִּתְבָּאֵר:

בָּרוּךְ וְזוֹקֵף בַּשֵּׁם. כּוֹרֵעַ בְּ״בָרוּךְ״ וְזוֹקֵף בַּשֵּׁם, עַל שֵׁם ״ה׳ זֹקֵף כְּפוּפִים״. בַּכְּרִיעָה שֶׁבִּתְחִלַּת בְּרָכָה — אִם רָצָה לְהַמְתִּין עַד סָמוּךְ לַחֲתִימָה הָרְשׁוּת בְּיָדוֹ, וְאִם זוֹקֵף מִיָּד — בַּשֵּׁם. וְכֵן בַּסּוֹף שֶׁחַיָּב לִזְקֹף מִיָּד — בַּשֵּׁם. וְהַמִּתְפַּלֵּל בְּרֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר בִּכְרִיעָה — זוֹקֵף רַק בְּסוֹף בְּרָכָה וּתְחִלָּתָהּ.

סעיף ו כּוֹרֵעַ בִּמְהִירוּת וְזוֹקֵף בְּנַחַת ; עַל פִּי הַסּוֹד בְּ״בָרוּךְ״ בְּבִרְכָּיו.

כְּשֶׁהוּא כּוֹרֵעַ — כּוֹרֵעַ בִּמְהִירוּת בְּפַעַם אַחַת, וּכְשֶׁהוּא זוֹקֵף — זוֹקֵף בְּנַחַת רֹאשׁוֹ תְּחִלָּה וְאַחַר כָּךְ גּוּפוֹ, שֶׁלֹּא תְּהֵא הַכְּרִיעָה נִרְאֵית עָלָיו כְּמַשּׁאוֹי אִם יִזְקֹף בִּמְהִירוּת. וְעַל פִּי הַסּוֹד כְּשֶׁאוֹמֵר ״בָּרוּךְ״ — יִכְרַע בְּבִרְכָּיו, וּבְ״אַתָּה״ — יִשְׁתַּחֲוֶה עַד שֶׁיִּתְפַּקְּקוּ כָּל הַחֻלְיוֹת:

מְהִירוּת וְנַחַת. כְּשֶׁכּוֹרֵעַ — בִּמְהִירוּת בְּפַעַם אַחַת ; כְּשֶׁזּוֹקֵף — בְּנַחַת, רֹאשׁוֹ תְּחִלָּה וְאַחַר כָּךְ גּוּפוֹ, שֶׁלֹּא תֵּרָאֶה הַכְּרִיעָה כְּמַשּׁאוֹי. וְעַל פִּי הַסּוֹד : בְּ״בָרוּךְ״ כּוֹרֵעַ בְּבִרְכָּיו, וּבְ״אַתָּה״ מִשְׁתַּחֲוֶה עַד שֶׁיִּתְפַּקְּקוּ כָּל הַחֻלְיוֹת.

סעיף ז נָכְרִי וּשְׁתִי וָעֵרֶב בְּיָדוֹ — לֹא יִשְׁחֶה מִפְּנֵי מַרְאִית הָעַיִן.

הַמִּתְפַּלֵּל וּבָא כְּנֶגְדּוֹ נָכְרִי וּשְׁתִי וָעֵרֶב בְּיָדוֹ וְהִגִּיעַ לְמָקוֹם שֶׁשּׁוֹחִין בּוֹ — לֹא יִשְׁחֶה אַף עַל פִּי שֶׁלִּבּוֹ לַשָּׁמַיִם, מִפְּנֵי מַרְאִית הָעַיִן שֶׁנִּרְאֶה כְּמִשְׁתַּחֲוֶה לַעֲבוֹדָה זָרָה (וְכָל מָקוֹם שֶׁאָסְרוּ מִפְּנֵי מַרְאִית הָעַיִן אֲפִלּוּ בְּחַדְרֵי חֲדָרִים אָסוּר). וְאִם הַנָּכְרִי הוּא שַׂר אוֹ הֶגְמוֹן שֶׁאַף מִבִּלְעֲדֵי הַשְּׁתִי וָעֵרֶב שֶׁבְּיָדוֹ מִשְׁתַּחֲוִים לוֹ — אֶפְשָׁר לְהָקֵל אֲפִלּוּ בִּמְקוֹם רוֹאִים, כְּמוֹ שֶׁיִּתְבָּאֵר בְּיוֹרֶה דֵּעָה סִימָן ק״נ:

מַרְאִית הָעַיִן. הַמִּתְפַּלֵּל וְנָכְרִי עִם שְׁתִי וָעֵרֶב בְּיָדוֹ בָּא כְּנֶגְדּוֹ, וְהִגִּיעַ לִמְקוֹם כְּרִיעָה — לֹא יִשְׁחֶה, אַף שֶׁלִּבּוֹ לַשָּׁמַיִם, מִפְּנֵי מַרְאִית הָעַיִן (שֶׁנִּרְאֶה כְּמִשְׁתַּחֲוֶה לַעֲבוֹדָה זָרָה ; וְכָל שֶׁאָסְרוּ מִפְּנֵי מַרְאִית הָעַיִן — אַף בְּחַדְרֵי חֲדָרִים אָסוּר). וְאִם הַנָּכְרִי שַׂר אוֹ הֶגְמוֹן שֶׁמִּשְׁתַּחֲוִים לוֹ אַף בְּלֹא הַשְּׁתִי וָעֵרֶב — אֶפְשָׁר לְהָקֵל אַף בִּמְקוֹם רוֹאִים (יוֹרֶה דֵּעָה סִימָן ק״נ).

סעיף ח אֵין לְהוֹסִיף עַל תְּאָרָיו — ״הָאֵל הַגָּדוֹל הַגִּבּוֹר וְהַנּוֹרָא״.

אֵין לְהוֹסִיף עַל תְּאָרָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא יוֹתֵר מֵ״הָאֵל הַגָּדוֹל הַגִּבּוֹר וְהַנּוֹרָא״, לְפִי שֶׁכְּשֶׁאוֹמֵר יוֹתֵר מֵהֶם וּמְסַיֵּם — נִרְאֶה כְּאִלּוּ סִיֵּם כָּל שְׁבָחָיו, וּמִי יוּכַל לְהַשְׁמִיעַ כָּל תְּהִלָּתוֹ, וְגַם אֵלּוּ הַשְּׁלֹשָׁה כִּנּוּיִים אִם לֹא שֶׁכְּתָבָן מֹשֶׁה בַּתּוֹרָה וּבָאוּ אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה וְתִקְּנוּם בַּתְּפִלָּה — לֹא הָיִינוּ אוֹמְרִים אוֹתָם, לְפִיכָךְ אֵין לְשַׁנּוֹת מִמַּטְבֵּעַ שֶׁטָּבְעוּ חֲכָמִים. בַּמֶּה דְּבָרִים אֲמוּרִים? בִּתְפִלַּת י״ח, אֲבָל בְּתַחֲנוּנִים אוֹ בַּקָּשׁוֹת וּשְׁבָחִים שֶׁאָדָם אוֹמֵר מֵעַצְמוֹ — יָכוֹל לְהַרְבּוֹת בִּתְאָרִים כְּמוֹ שֶׁיִּרְצֶה, מִפְּנֵי שֶׁכְּשֶׁהוּא מִתְחַנֵּן בֵּינוֹ לְבֵין עַצְמוֹ וְאוֹמֵר תְּאָרִים נִרְאֶה שֶׁאֵינוֹ אוֹמְרָם אֶלָּא לְפִי שֶׁאֵלּוּ הַתְּאָרִים צְרִיכִים לְבַקָּשָׁתוֹ, מַה שֶּׁאֵין כֵּן בִּתְפִלַּת י״ח שֶׁבְּזֹאת הַבְּרָכָה הָרִאשׁוֹנָה לֹא בָּאוּ בְּכָאן הַתְּאָרִים אֶלָּא כְּדֵי לְהַזְכִּיר שִׁבְחוֹ שֶׁל מָקוֹם בִּלְבַד, שֶׁהֲרֵי עֲדַיִן אֵין אָנוּ שׁוֹאֲלִין שׁוּם צֹרֶךְ מֵאִתּוֹ. וְיֵשׁ אוֹסְרִין בְּכָל עִנְיָן, לְפִי שֶׁכָּל הַשְּׁלֵמוּת שֶׁהֵם אֶצְלֵנוּ אֵין אֶצְלוֹ יִתְבָּרַךְ מִמִּינָם דָּבָר אֶלָּא כֻּלָּם חֶסְרוֹנוֹת בְּחֵיקוֹ יִתְבָּרַךְ, וְהוּא מָשָׁל לְמֶלֶךְ שֶׁהָיָה לוֹ אֶלֶף אֲלָפִים דִּינְרֵי זָהָב וּמְקַלְּסִין אוֹתוֹ בְּשֶׁל כֶּסֶף שֶׁגְּנַאי הוּא לוֹ (שֶׁכָּל הַשְּׁבָחִים וְהַכִּנּוּיִים לֹא יֻחֲסוּ לוֹ יִתְבָּרַךְ עַל צַד הַחִיּוּב, כִּי שְׁמוֹ מְרוֹמָם עַל כָּל בְּרָכָה וּתְהִלָּה, אֶלָּא עַל דֶּרֶךְ שְׁלִילָה לִשְׁלֹל הֲפָכָם מִשְּׁמוֹ יִתְבָּרַךְ, כַּנּוֹדָע לְהַיּוֹדְעִים עִנְיַן אַחְדוּתוֹ יִתְבָּרַךְ הַפָּשׁוּט בְּתַכְלִית הַפְּשִׁיטוּת). וְהָעִקָּר כַּסְּבָרָא הָרִאשׁוֹנָה. וְלָכֵן נָהֲגוּ לוֹמַר תְּחִנּוֹת עֲנֵנ[וּ] הַמְסֻדָּרִים בְּא״ב וְכָל אוֹת תֹּאַר וְשֶׁבַח בִּפְנֵי עַצְמוֹ, וְכֵן הַרְבֵּה תְּפִלּוֹת דּוֹמוֹת לָזוֹ. וְאַף עַל פִּי כֵן נָכוֹן לְמִי שֶׁיִּרְצֶה לְהַאֲרִיךְ בְּשִׁבְחֵי הַמָּקוֹם שֶׁיֹּאמַר אוֹתָן בִּקְרִיאַת פְּסוּקִים, לָצֵאת לְדִבְרֵי הַכֹּל:

מַטְבֵּעַ הַתְּאָרִים. אֵין לְהוֹסִיף עַל תְּאָרָיו יוֹתֵר מֵ״הָאֵל הַגָּדוֹל הַגִּבּוֹר וְהַנּוֹרָא״, שֶׁהַמּוֹסִיף וּמְסַיֵּם נִרְאֶה כִּמְסַיֵּם כָּל שְׁבָחָיו, וּמִי יַשְׁמִיעַ כָּל תְּהִלָּתוֹ. אַף שְׁלֹשָׁה אֵלּוּ — לוּלֵי שֶׁכְּתָבָן מֹשֶׁה וְתִקְּנוּם אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה לֹא הָיִינוּ אוֹמְרִים ; אֵין לְשַׁנּוֹת מִמַּטְבֵּעַ שֶׁטָּבְעוּ חֲכָמִים. בַּמֶּה דְּבָרִים אֲמוּרִים? בִּתְפִלַּת י״ח ; אֲבָל בְּתַחֲנוּנִים יָכוֹל לְהַרְבּוֹת בִּתְאָרִים. וְיֵשׁ אוֹסְרִין, וְהָעִקָּר כַּסְּבָרָא הָרִאשׁוֹנָה. וְנָכוֹן לְהַאֲרִיךְ בַּשְּׁבָחִים בִּקְרִיאַת פְּסוּקִים, לָצֵאת לְדִבְרֵי הַכֹּל.

סעיף ט ״מִי שֶׁרִחֵם עַל קַן צִפּוֹר״ — מַשְׁתִּיקִין אוֹתוֹ.

הָאוֹמֵר בְּתַחֲנוּנָיו ״מִי שֶׁרִחֵם עַל קַן צִפּוֹר יְרַחֵם עָלֵינוּ״ — מַשְׁתִּיקִין אוֹתוֹ, שֶׁאֵין מִצְווֹת הַמָּקוֹם רַחֲמָנוּת אֶלָּא גְּזֵרוֹת עַל עַמּוֹ, לְהוֹדִיעַ שֶׁהֵם עֲבָדָיו וְשׁוֹמְרֵי מִצְו‍ֹתָיו וּגְזֵרַת חֻקּוֹתָיו אַף בִּדְבָרִים שֶׁיֵּשׁ לַשָּׂטָן וּלְאֻמּוֹת הָעוֹלָם לְהָשִׁיב עֲלֵיהֶם וְלוֹמַר מַה צֹּרֶךְ בְּמִצְוָה זוֹ. וְכֵן הָאוֹמֵר ״עַל הַטּוֹב שֶׁאַתָּה עוֹשֶׂה לָנוּ יִזָּכֵר שְׁמֶךָ״ — מַשְׁתִּיקִין אוֹתוֹ, שֶׁחַיָּב אָדָם לְבָרֵךְ עַל הָרָעָה כְּשֵׁם שֶׁמְּבָרֵךְ עַל הַטּוֹבָה כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ר״ל:

מַשְׁתִּיקִין אוֹתוֹ. הָאוֹמֵר בְּתַחֲנוּנָיו ״מִי שֶׁרִחֵם עַל קַן צִפּוֹר יְרַחֵם עָלֵינוּ״ — מַשְׁתִּיקִין אוֹתוֹ, שֶׁאֵין מִצְווֹת הַמָּקוֹם רַחֲמָנוּת אֶלָּא גְּזֵרוֹת עַל עַמּוֹ. וְכֵן הָאוֹמֵר ״עַל הַטּוֹב שֶׁאַתָּה עוֹשֶׂה לָנוּ יִזָּכֵר שְׁמֶךָ״ — מַשְׁתִּיקִין אוֹתוֹ, שֶׁחַיָּב אָדָם לְבָרֵךְ עַל הָרָעָה כְּשֵׁם שֶׁמְּבָרֵךְ עַל הַטּוֹבָה (סִימָן ר״ל).

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קי״ג (מהדורת קה״ת, ט׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

היסוד — דִּינֵי הַכְּרִיעוֹת בְּי״ח בְּרָכוֹת

היסוד : דִּינֵי הַכְּרִיעוֹת בָּעֲמִידָה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַמְּקוֹם הַכְּרִיעוֹת — הֵיכָן שׁוֹחִין וְהֵיכָן אֵין שׁוֹחִין. אחר כך הַכּוֹרֵעַ בְּמָקוֹם שֶׁלֹּא תִּקְּנוּ — מְגֻנֶּה אוֹ מְשֻׁבָּח. אחר כך הַשִּׁעוּר וְהָאֹפֶן — עַד שֶׁיִּתְפַּקְּקוּ, ״בָּרוּךְ״ וְהַשֵּׁם. ולבסוף הַמַּרְאִית הָעַיִן וְהַמַּטְבֵּעַ — שְׁתִי וָעֵרֶב, תְּאָרִים, וְהַמַּשְׁתִּיקִין.

מְקוֹם הַכְּרִיעוֹת — אָבוֹת וְהוֹדָאָה

מְקוֹם הַכְּרִיעוֹת — אָבוֹת וְהוֹדָאָה

היסוד. שׁוֹחִין בְּ״אָבוֹת״ וּבְ״הוֹדָאָה״, תְּחִלָּה וָסוֹף. הַבָּא לְהוֹסִיף בָּרֹאשׁ/בַּסּוֹף שֶׁל בְּרָכָה אַחֶרֶת — מְלַמְּדִין אוֹתוֹ שֶׁלֹּא יִשְׁחֶה, שֶׁלֹּא לַעֲקֹר תַּקָּנַת חֲכָמִים ; אֲבָל בְּאֶמְצַע — מֻתָּר (סעיפים א-ב).

המסקנה. מְקוֹם הַכְּרִיעָה נִקְבַּע בַּתַּקָּנָה, לֹא בְּמַשְׁמַע הַתֵּבוֹת. עיין סימן 112.

הַכּוֹרֵעַ בְּמָקוֹם שֶׁלֹּא תִּקְּנוּ

הַכּוֹרֵעַ בְּמָקוֹם שֶׁלֹּא תִּקְּנוּ

היסוד. הַכּוֹרֵעַ בְּ״הוֹדָאָה״ אוֹ בְּ״וְכָל קוֹמָה״ מֵחֲמַת הַלָּשׁוֹן — מְגֻנֶּה וְסָכָל ; אֲבָל מֵחֲמַת שְׁבָחִים, תַּחֲנוּנִים, מִנְהָג אוֹ הִתְלַהֲבוּת — מְשֻׁבָּח, וְ״וַאֲנַחְנוּ כּוֹרְעִים״ (עַל כָּל הָעוֹלָם) — יָפֶה (סעיף ג).

המסקנה. הַמֶּדֶד אֵינוֹ הַתֵּבָה אֶלָּא הַ״מוֹסִיף עַל תַּקָּנַת חֲכָמִים״. עיין סימן 111.

שִׁעוּר הַכְּרִיעָה וְאָפְנָהּ

שִׁעוּר הַכְּרִיעָה וְאָפְנָהּ

היסוד. עַד שֶׁיִּתְפַּקְּקוּ חֻלְיוֹת הַשִּׁדְרָה, כְּאַגְמוֹן, וְלֹא כְּיוּהֲרָא ; כּוֹרֵעַ בִּמְהִירוּת וְזוֹקֵף בְּנַחַת ; כּוֹרֵעַ בְּ״בָרוּךְ״ וְזוֹקֵף בַּשֵּׁם (סעיפים ד-ו).

המסקנה. צוּרַת הַגּוּף מְשָׁרֶתֶת אֶת הַכַּוָּנָה, בְּלִי גֹּזֶם וּבְלִי מַשּׁאוֹי. עיין סימן 114.

מַרְאִית הָעַיִן וּמַטְבֵּעַ הַתְּפִלָּה

מַרְאִית הָעַיִן וּמַטְבֵּעַ הַתְּפִלָּה

היסוד. נָכְרִי וּשְׁתִי וָעֵרֶב בְּיָדוֹ — לֹא יִשְׁחֶה (מַרְאִית הָעַיִן) ; אֵין לְהוֹסִיף עַל תְּאָרָיו בִּתְפִלַּת י״ח ; מַשְׁתִּיקִין אֶת הָאוֹמֵר ״קַן צִפּוֹר״ (סעיפים ז-ט).

המסקנה. שׁוֹמְרִים עַל הַמַּטְבֵּעַ שֶׁטָּבְעוּ חֲכָמִים וּמַרְחִיקִים מַרְאִית פְּסוּלָה. עיין סימן 110.

הפניה ללימוד. היקף הסימן — מְקוֹם הַכְּרִיעוֹת, שִׁעוּר הַכְּרִיעָה וּמַטְבֵּעַ הַתְּפִלָּה — נוגע להנהגת העֲמִידָה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קי״ג. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מט׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
מְקוֹם הַכְּרִיעוֹת או״ח קי״ג — שׁוֹחִין בְּ״אָבוֹת״ וּבְ״הוֹדָאָה״ תְּחִלָּה וָסוֹף ; בְּאֶמְצַע — יָכוֹל לִשְׁחוֹת. מ״ב קי״ג — גֶּדֶר הַ״מְלַמְּדִין אוֹתוֹ שֶׁלֹּא יִשְׁחֶה״ וַעֲקִירַת הַתַּקָּנָה. ערוה״ש קי״ג — מְקוֹמוֹת הַכְּרִיעָה בָּעֲמִידָה. דרך אדמו״ר הזקן — בְּ״אָבוֹת״ וּבְ״הוֹדָאָה״ תְּחִלָּה וָסוֹף ; בְּאֶמְצַע מֻתָּר, שֶׁלֹּא לַעֲקֹר תַּקָּנַת חֲכָמִים (סעיף א).
הַכּוֹרֵעַ שֶׁלֹּא בִּמְקוֹמוֹ או״ח קי״ג — הַכּוֹרֵעַ בְּ״וְכָל קוֹמָה״ אוֹ בַּ״הוֹדָאוֹת״ הֲרֵי זֶה מְגֻנֶּה. מ״ב קי״ג — לְשׁוֹן הוֹדָאָה אֵינוֹ הִשְׁתַּחֲוָאָה ; פְּרָטֵי הַמְגֻנֶּה. ערוה״ש קי״ג — גֶּדֶר ״וַאֲנַחְנוּ כּוֹרְעִים״ וְהַכְּרִיעוֹת שֶׁבַּמִּנְהָג. דרך אדמו״ר הזקן — מֵחֲמַת לְשׁוֹן הוֹדָאָה — סָכָל ; אֲבָל מִנְהָג/הִתְלַהֲבוּת — מְשֻׁבָּח, וְ״וַאֲנַחְנוּ כּוֹרְעִים״ יָפֶה (סעיף ג).
שִׁעוּר הַכְּרִיעָה וְ״בָרוּךְ״ או״ח קי״ג — עַד שֶׁיִּתְפַּקְּקוּ חֻלְיוֹת ; כּוֹרֵעַ בְּ״בָרוּךְ״ זוֹקֵף בַּשֵּׁם. מ״ב קי״ג — דִּין הַזָּקֵן וְהַחוֹלֶה וְ״כְּאַגְמוֹן״. ערוה״ש קי״ג — גֶּדֶר הַיּוּהֲרָא בַּשְּׁחִיָּה יְתֵרָה. דרך אדמו״ר הזקן — עַד שֶׁיִּתְפַּקְּקוּ ; לֹא כְּיוּהֲרָא ; כּוֹרֵעַ בִּמְהִירוּת זוֹקֵף בְּנַחַת ; עַל פִּי הַסּוֹד בְּ״בָרוּךְ״ בְּבִרְכָּיו (סעיפים ד-ו).
מַטְבֵּעַ הַתְּאָרִים וְהַמַּשְׁתִּיקִין או״ח קי״ג — אֵין לְהוֹסִיף עַל ״הָאֵל הַגָּדוֹל הַגִּבּוֹר וְהַנּוֹרָא״ ; ״קַן צִפּוֹר״ מַשְׁתִּיקִין. מ״ב קי״ג — גֶּדֶר הַשִּׁנּוּי מִמַּטְבֵּעַ בַּתַּחֲנוּנִים וּבַתְּפִלָּה. ערוה״ש קי״ג — טַעַם הַ״מַּשְׁתִּיקִין״ וְגֶדֶר הַשְּׁבָחִים. דרך אדמו״ר הזקן — בִּתְפִלַּת י״ח אֵין לְשַׁנּוֹת ; בַּתַּחֲנוּנִים יָכוֹל לְהַרְבּוֹת, וְנָכוֹן בִּקְרִיאַת פְּסוּקִים ; ״קַן צִפּוֹר״ מַשְׁתִּיקִין (סעיפים ח-ט).

הטבלה נערכה מטקסט המחבר (אורח חיים קי״ג, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפלה ה, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ל״ד, מגילה כ״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ט׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — לֹא לַעֲקֹר תַּקָּנָה

קו א — לֹא לַעֲקֹר תַּקָּנָה

יסוד הסימן : שׁוֹחִין בְּ״אָבוֹת״ וּבְ״הוֹדָאָה״ ; הַמּוֹסִיף בָּרֹאשׁ/בַּסּוֹף — מְלַמְּדִין אוֹתוֹ שֶׁלֹּא יִשְׁחֶה (סעיף א).

קריאת הרב : הַמֶּדֶד שֶׁלֹּא יָבוֹא לַעֲקֹר תַּקָּנַת חֲכָמִים ; בְּאֶמְצַע — מֻתָּר, שֶׁאֵין שָׁם תַּקָּנָה. עיין סימן 112.

מסקנה למעשה : לִשְׁחוֹת בְּדִיּוּק בְּאַרְבָּעָה מְקוֹמוֹת שֶׁתִּקְּנוּ, לֹא פָּחוֹת וְלֹא יוֹתֵר.

קו ב — אֹפֶן וְשִׁעוּר

קו ב — אֹפֶן וְשִׁעוּר

יסוד הסימן : עַד שֶׁיִּתְפַּקְּקוּ חֻלְיוֹת ; לֹא כְּיוּהֲרָא ; כּוֹרֵעַ בִּמְהִירוּת וְזוֹקֵף בְּנַחַת (סעיפים ד-ו).

קריאת הרב : כּוֹרֵעַ בְּ״בָרוּךְ״ וְזוֹקֵף בַּשֵּׁם, וְעַל פִּי הַסּוֹד בְּ״בָרוּךְ״ בְּבִרְכָּיו. עיין סימן 114.

מסקנה למעשה : לְהַרְכִּין רֹאשׁ וְגוּף בְּלִי יוּהֲרָא ; לָקוּם בְּנַחַת, הָרֹאשׁ תְּחִלָּה.

קו ג — מַטְבֵּעַ שֶׁטָּבְעוּ חֲכָמִים

קו ג — מַטְבֵּעַ שֶׁטָּבְעוּ חֲכָמִים

יסוד הסימן : אֵין לְהוֹסִיף עַל תְּאָרָיו בִּתְפִלַּת י״ח ; מַשְׁתִּיקִין אֶת הָאוֹמֵר ״קַן צִפּוֹר״ (סעיפים ח-ט).

קריאת הרב : בַּתַּחֲנוּנִים יָכוֹל לְהַרְבּוֹת, וְנָכוֹן בִּקְרִיאַת פְּסוּקִים לָצֵאת לְדִבְרֵי הַכֹּל. עיין סימן 110.

מסקנה למעשה : לִשְׁמֹר עַל הַמַּטְבֵּעַ הַקָּבוּעַ בָּעֲמִידָה ; הוֹסָפַת שְׁבָחִים — בִּתְפִלּוֹת אִישִׁיּוֹת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קי״ג ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קי״ג (דִּינֵי הַכְּרִיעוֹת בְּי״ח בְּרָכוֹת)

  • ① מְקוֹם הַכְּרִיעוֹת. שׁוֹחִין בְּ״אָבוֹת״ וּבְ״הוֹדָאָה״, תְּחִלָּה וָסוֹף ; אֵין לְהוֹסִיף בָּרֹאשׁ/בַּסּוֹף שֶׁל בְּרָכָה אַחֶרֶת (מְלַמְּדִין אוֹתוֹ שֶׁלֹּא יִשְׁחֶה). יסוד : שוע״ר קי״ג סעיף א.
  • ② רֹאשׁ הַשָּׁנָה וְיוֹם כִּפּוּר. הַכּוֹרֵעַ בְּ״זָכְרֵנוּ״/״מִי כָמוֹךָ״ — זוֹקֵף בְּסוֹף הַבְּרָכָה, כְּדֵי לַחֲזֹר וְלִשְׁחוֹת בְּ״בָרוּךְ אַתָּה״. יסוד : שוע״ר קי״ג סעיף ב.
  • ③ שֶׁלֹּא בִּמְקוֹמוֹ. כּוֹרֵעַ מֵחֲמַת לְשׁוֹן ״הוֹדָאָה״ אוֹ ״וְכָל קוֹמָה״ — מְגֻנֶּה ; מֵחֲמַת מִנְהָג/הִתְלַהֲבוּת אוֹ ״וַאֲנַחְנוּ כּוֹרְעִים״ — יָפֶה. יסוד : שוע״ר קי״ג סעיף ג.
  • ④ שִׁעוּר וְאֹפֶן. יִכְרַע עַד שֶׁיִּתְפַּקְּקוּ חֻלְיוֹת, כְּאַגְמוֹן, לֹא כְּיוּהֲרָא ; כּוֹרֵעַ בִּמְהִירוּת, זוֹקֵף בְּנַחַת ; כּוֹרֵעַ בְּ״בָרוּךְ״ זוֹקֵף בַּשֵּׁם. יסוד : שוע״ר קי״ג סעיפים ד-ו.
  • ⑤ מַרְאִית הָעַיִן וּמַטְבֵּעַ. נָכְרִי וּשְׁתִי וָעֵרֶב בְּיָדוֹ — לֹא יִשְׁחֶה ; אֵין לְהוֹסִיף עַל הַתְּאָרִים בִּתְפִלַּת י״ח ; ״קַן צִפּוֹר״ — מַשְׁתִּיקִין. יסוד : שוע״ר קי״ג סעיפים ז-ט.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קי״ג (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — מְקוֹם הַכְּרִיעוֹת, שִׁעוּר הַכְּרִיעָה, מַטְבֵּעַ הַתְּפִלָּה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
