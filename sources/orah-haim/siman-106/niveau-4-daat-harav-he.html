דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״ו · מִי הֵם הַפְּטוּרִים מִתְּפִלָּה

דרכו של אדמו״ר הזקן במִי הֵם הַפְּטוּרִים מִתְּפִלָּה : כָּל הַפָּטוּר מִק״ש פָּטוּר מִתְּפִלָּה, חוּץ מֵהַמְלַוִּין אֶת הַמֵּת ; נָשִׁים וַעֲבָדִים — חַיָּבִים בִּתְפִלָּה ; חִנּוּךְ הַקְּטַנִּים ; וּמִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ — הַהַפְסָקָה לק״ש וְלִתְפִלָּה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״ו — מִי הֵם הַפְּטוּרִים מִתְּפִלָּה. אצל הרב הסימן נפרש לד׳ סְעִיפִים : הַשְׁוָאַת חִיּוּב ק״ש וּתְפִלָּה עִם פְּטוֹר מְלַוֵּי הַמֵּת, חִיּוּב נָשִׁים וַעֲבָדִים וִיסוֹדוֹ, חִנּוּךְ הַקְּטַנִּים, וּמִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ (חַיֵּי עוֹלָם וְחַיֵּי שָׁעָה).

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״ו — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״ו). אדמו״ר הזקן דן בו בְּמִי הֵם הַפְּטוּרִים מִתְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״ו — מִי הֵם הַפְּטוּרִים מִתְּפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״ו — מִי הֵם הַפְּטוּרִים מִתְּפִלָּה — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.106), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.106.k).

סעיף א כָּל הַפְּטוּרִים מִק״ש — פְּטוּרִים מִתְּפִלָּה ; חוּץ מִמְּלַוֵּי הַמֵּת.

מִי הֵם הַפְּטוּרִים מִתְּפִלָּה וּבוֹ ד׳ סְעִיפִים:כָּל הַפְּטוּרִים מִקְּרִיאַת שְׁמַע — פְּטוּרִים מִתְּפִלָּה. וְכָל שֶׁחַיָּב בִּקְרִיאַת שְׁמַע — חַיָּב בִּתְפִלָּה, חוּץ מֵהַמְלַוִּין אֶת הַמֵּת שֶׁאֵין לַמִּטָּה צֹרֶךְ בָּהֶן, שֶׁאַף עַל פִּי שֶׁחַיָּבִים בִּקְרִיאַת שְׁמַע — פְּטוּרִים מִתְּפִלָּה, הוֹאִיל וְהִיא מִדִּבְרֵי סוֹפְרִים, וְעוֹד שֶׁהִיא בַּעֲמִידָה וְאֵין יְכוֹלִין לִשְׁהוֹת כָּל כָּךְ, אֲבָל קְרִיאַת שְׁמַע עִקַּר כַּוָּנָתָהּ וַעֲמִידָתָהּ אֵינָהּ אֶלָּא בְּפָסוּק רִאשׁוֹן וּבְקַל יוּכַל לַעֲמֹד וּלְכַוֵּן:

הַפְּטוּרִים וְהַחַיָּבִים. כָּל הַפָּטוּר מִק״ש פָּטוּר מִתְּפִלָּה, וְכָל הַחַיָּב בִּק״ש חַיָּב בִּתְפִלָּה — חוּץ מִמְּלַוֵּי הַמֵּת שֶׁאֵין לַמִּטָּה צֹרֶךְ בָּהֶן : אַף שֶׁחַיָּבִים בִּק״ש — פְּטוּרִים מִתְּפִלָּה, הוֹאִיל וְהִיא מִדִּבְרֵי סוֹפְרִים וּבַעֲמִידָה וְאֵין יְכוֹלִין לִשְׁהוֹת, אֲבָל ק״ש עִקַּר כַּוָּנָתָהּ פָּסוּק רִאשׁוֹן, וּבְקַל יוּכַל לַעֲמֹד וּלְכַוֵּן.

סעיף ב נָשִׁים וַעֲבָדִים — חַיָּבִים בִּתְפִלָּה ; וְכֵן עִקָּר שַׁחֲרִית וּמִנְחָה.

נָשִׁים וַעֲבָדִים אַף עַל פִּי שֶׁפְּטוּרִים מִקְּרִיאַת שְׁמַע — חַיָּבִים בִּתְפִלָּה, מִפְּנֵי שֶׁהוּא מִצְוַת עֲשֵׂה שֶׁלֹּא הַזְּמַן גְּרָמָא, שֶׁזְּמַנֵּי הַתְּפִלָּה הֵם מִדִּבְרֵי סוֹפְרִים, וְעִקַּר מִצְוַת הַתְּפִלָּה הִיא מִן הַתּוֹרָה, שֶׁנֶּאֱמַר: ״וּלְעָבְדוֹ בְּכָל לְבַבְכֶם״, אֵיזוֹ עֲבוֹדָה שֶׁבַּלֵּב — זוֹ תְּפִלָּה, אֶלָּא שֶׁאֵין לָהּ נֻסָּח יָדוּעַ מִן הַתּוֹרָה וְיָכוֹל לְהִתְפַּלֵּל בְּכָל נֻסָּח שֶׁיִּרְצֶה וּבְכָל עֵת שֶׁיִּרְצֶה, וּמִשֶּׁהִתְפַּלֵּל פַּעַם אַחַת בַּיּוֹם אוֹ בַּלַּיְלָה — יָצָא יְדֵי חוֹבָתוֹ מִן הַתּוֹרָה. וְעַל פִּי סְבָרָא זוֹ נָהֲגוּ רֹב הַנָּשִׁים שֶׁאֵינָן מִתְפַּלְלִין י״ח בִּתְמִידוּת שַׁחַר וָעֶרֶב, לְפִי שֶׁאוֹמְרוֹת מִיָּד בַּבֹּקֶר סָמוּךְ לַנְּטִילָה אֵיזֶה בַּקָּשָׁה וּמִן הַתּוֹרָה יוֹצְאוֹת בָּזֶה, וְאֶפְשָׁר שֶׁאַף חֲכָמִים לֹא חִיְּבוּם יוֹתֵר. אֲבָל יֵשׁ אוֹמְרִים שֶׁעִקַּר מִצְוַת תְּפִלָּה הִיא מִדִּבְרֵי סוֹפְרִים, שֶׁהֵם אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה שֶׁתִּקְּנוּ י״ח בְּרָכוֹת עַל הַסֵּדֶר, לְהִתְפַּלֵּל אוֹתָן שַׁחֲרִית וּמִנְחָה חוֹבָה וְעַרְבִית רְשׁוּת, וְאַף עַל פִּי שֶׁהוּא מִצְוַת עֲשֵׂה מִדִּבְרֵי סוֹפְרִים שֶׁהַזְּמַן גְּרָמָא, וְהַנָּשִׁים פְּטוּרוֹת מִכָּל מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָא אֲפִלּוּ הִיא מִדִּבְרֵי סוֹפְרִים כְּגוֹן הַלֵּל וְקִדּוּשׁ הַלְּבָנָה — אַף עַל פִּי כֵן חִיְּבוּ אוֹתָן בִּתְפִלַּת שַׁחֲרִית וּמִנְחָה, הוֹאִיל וּתְפִלָּה הִיא בַּקָּשַׁת רַחֲמִים. וְכֵן עִקָּר. אֲבָל תְּפִלַּת עַרְבִית שֶׁהוּא רְשׁוּת, אַף עַל פִּי שֶׁעַכְשָׁו כְּבָר קִבְּלוּהוּ עֲלֵיהֶם כָּל יִשְׂרָאֵל לְחוֹבָה, מִכָּל מָקוֹם הַנָּשִׁים לֹא קִבְּלוּ עֲלֵיהֶם וְרֻבָּן אֵינָן מִתְפַּלְּלוֹת עַרְבִית בְּמ״ש:

נָשִׁים וַעֲבָדִים. אַף שֶׁפְּטוּרִים מִק״ש — חַיָּבִים בִּתְפִלָּה, שֶׁהִיא מ״ע שֶׁלֹּא הַזְּמַן גְּרָמָא : עִיקָּרָהּ מִן הַתּוֹרָה (״וּלְעָבְדוֹ בְּכָל לְבַבְכֶם״ — אֵיזוֹ עֲבוֹדָה שֶׁבַּלֵּב ? זוֹ תְּפִלָּה), בְּלֹא נֻסָּח קָבוּעַ, וּפַעַם אַחַת בַּיּוֹם דַּי מִן הַתּוֹרָה. וְעַל פִּי זֶה נָהֲגוּ רֹב נָשִׁים שֶׁאֵינָן מִתְפַּלְלוֹת י״ח בִּתְמִידוּת. אֲבָל יֵשׁ אוֹמְרִים שֶׁעִיקָּרָהּ מִדִּבְרֵי סוֹפְרִים (אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה שֶׁתִּקְּנוּ י״ח בְּרָכוֹת — שַׁחֲרִית וּמִנְחָה חוֹבָה, עַרְבִית רְשׁוּת) : אַף שֶׁהַזְּמַן גְּרָמָא, חִיְּבוּ אוֹתָן שַׁחֲרִית וּמִנְחָה, הוֹאִיל וּתְפִלָּה הִיא בַּקָּשַׁת רַחֲמִים — וְכֵן עִקָּר. אֲבָל עַרְבִית (רְשׁוּת), הַנָּשִׁים לֹא קִבְּלוּ.

סעיף ג קְטַנִּים שֶׁהִגִּיעוּ לְחִנּוּךְ — לְחַנְּכָם לְהִתְפַּלֵּל עֶרֶב וָבֹקֶר.

קְטַנִּים שֶׁהִגִּיעוּ לְחִנּוּךְ — חַיָּבִים לְחַנְּכָם לְהִתְפַּלֵּל י״ח עֶרֶב וָבֹקֶר. וּמִכָּל מָקוֹם, רַשַּׁאי לִתֵּן לָהֶם לֶאֱכֹל קֹדֶם תְּפִלַּת שַׁחֲרִית וְאָסוּר לְעַנּוֹתָם, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן רס״ט וְשמ״ג לְעִנְיַן לְהַאֲכִילָם קֹדֶם קִדּוּשׁ:

חִנּוּךְ הַקְּטַנִּים. קְטַנִּים שֶׁהִגִּיעוּ לְחִנּוּךְ — חַיָּבִים לְחַנְּכָם לְהִתְפַּלֵּל י״ח עֶרֶב וָבֹקֶר. וּמִכָּל מָקוֹם, רַשַּׁאי לְהַאֲכִילָם קֹדֶם תְּפִלַּת שַׁחֲרִית וְאָסוּר לְעַנּוֹתָם, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן רס״ט וְשמ״ג לְעִנְיַן לְהַאֲכִילָם קֹדֶם קִדּוּשׁ.

סעיף ד מִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ — מַפְסִיק לק״ש וְלֹא לִתְפִלָּה ; אֲבָל אָנוּ מַפְסִיקִין.

מִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ כְּגוֹן רַבִּי שִׁמְעוֹן בֶּן יוֹחַאי וַחֲבֵרָיו — מַפְסִיק לִקְרִיאַת שְׁמַע וּמְבָרֵךְ לְפָנֶיהָ וּלְאַחֲרֶיהָ, אֲבָל אֵינוֹ מִתְפַּלֵּל כָּל עִקָּר, כֵּיוָן שֶׁמִּבַּלְעֲדֵי כֵן אֵינוֹ מְבַטֵּל מִלִּמּוּדוֹ אֲפִלּוּ רֶגַע — גַּם בִּשְׁבִיל הַתְּפִלָּה לֹא יְבַטֵּל. וְאַף שֶׁצָּרִיךְ לְבַטֵּל וּלְהַפְסִיק לִמּוּדוֹ לַעֲשִׂיַּת כָּל הַמִּצְווֹת אֲפִלּוּ שֶׁל דִּבְרֵי סוֹפְרִים, מִכָּל מָקוֹם תְּפִלָּה הוֹאִיל וְאֵינָהּ אֶלָּא בַּקָּשַׁת רַחֲמִים עַל עִנְיְנֵי עוֹלָם הַזֶּה — אֵין לוֹ לְהָנִיחַ חַיֵּי הָעוֹלָם וְלַעֲסֹק בְּחַיֵּי שָׁעָה. אֲבָל אָנוּ מַפְסִיקִין בֵּין לִקְרִיאַת שְׁמַע בֵּין לִתְפִלָּה, כֵּיוָן שֶׁמִּבַּלְעֲדֵי כֵן אָנוּ מְבַטְּלִין הַרְבֵּה. וּמִכָּל מָקוֹם, אִם מְלַמֵּד לַאֲחֵרִים וְאִם לֹא יִלְמְדוּ עַכְשָׁו יִתְבַּטְּלוּ וְלֹא יוּכְלוּ לְהִתְקַבֵּץ לִלְמֹד אַחַר הַתְּפִלָּה — לֹא יִפְסֹק מִלִּמּוּדוֹ עִמָּהֶם אֲפִלּוּ אִם יַעֲבֹר זְמַן הַתְּפִלָּה. וּמִכָּל מָקוֹם, פּוֹסֵק וְקוֹרֵא פָּסוּק רִאשׁוֹן שֶׁל קְרִיאַת שְׁמַע, כְּדֵי לְקַבֵּל מַלְכוּת שָׁמַיִם בִּזְמַן קְרִיאַת שְׁמַע. וְיַחֲזֹר אַחַר הֲלָכָה שֶׁיֵּשׁ בָּהּ יְצִיאַת מִצְרַיִם, כְּדֵי לְהַזְכִּיר יְצִיאַת מִצְרַיִם בִּזְמַן קְרִיאַת שְׁמַע. וְאַחַר לִמּוּדוֹ טוֹב שֶׁיִּגְמֹר כָּל קְרִיאַת שְׁמַע. וְאַף עַל פִּי שֶׁעָבַר זְמַנָּהּ — לֹא הִפְסִיד, כְּאָדָם שֶׁקּוֹרֵא בַּתּוֹרָה. וְאִם אֵין הַשָּׁעָה עוֹבֶרֶת וְיֵשׁ לוֹ שְׁהוּת עֲדַיִן לִקְרוֹת קְרִיאַת שְׁמַע וּלְהִתְפַּלֵּל — אֵינוֹ פּוֹסֵק כְּלָל, אֲפִלּוּ לוֹמֵד יְחִידִי, וַאֲפִלּוּ הִתְחִיל לִלְמֹד בְּאִסּוּר לְאַחַר שֶׁכְּבָר הִגִּיעַ זְמַן קְרִיאַת שְׁמַע וּתְפִלָּה, שֶׁאָסוּר לוֹ לְהַתְחִיל לִלְמֹד עַד שֶׁיִּתְפַּלֵּל תְּחִלָּה אִם אֵינוֹ רָגִיל לֵילֵךְ לְבֵית הַכְּנֶסֶת כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן פ״ט, אַף עַל פִּי כֵן אִם הִתְחִיל אֵין צָרִיךְ לְהַפְסִיק, וְרַשַּׁאי לִלְמֹד כָּל זְמַן שֶׁיֵּשׁ שְׁהוּת עֲדַיִן לִקְרוֹת קְרִיאַת שְׁמַע וּלְהִתְפַּלֵּל אַחַר כָּךְ:

מִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ. כְּגוֹן רשב״י וַחֲבֵרָיו — מַפְסִיק לק״ש וּמְבָרֵךְ לְפָנֶיהָ וּלְאַחֲרֶיהָ, אֲבָל אֵינוֹ מִתְפַּלֵּל כָּל עִקָּר : כֵּיוָן שֶׁאֵינוֹ מְבַטֵּל מִלִּמּוּדוֹ אֲפִלּוּ רֶגַע — גַּם לַתְּפִלָּה לֹא יְבַטֵּל. שֶׁאַף שֶׁמַּפְסִיקִין לְכָל מִצְוָה, תְּפִלָּה שֶׁאֵינָהּ אֶלָּא בַּקָּשַׁת רַחֲמִים עַל עִנְיְנֵי עוֹלָם הַזֶּה — אֵין לוֹ לְהָנִיחַ חַיֵּי עוֹלָם וְלַעֲסֹק בְּחַיֵּי שָׁעָה. אֲבָל אָנוּ מַפְסִיקִין בֵּין לק״ש בֵּין לִתְפִלָּה, שֶׁמִּבַּלְעֲדֵי כֵן אָנוּ מְבַטְּלִין הַרְבֵּה. וְאִם מְלַמֵּד לַאֲחֵרִים וְיִתְבַּטְּלוּ — לֹא יִפְסֹק אֲפִלּוּ יַעֲבֹר הַזְּמַן ; מִכָּל מָקוֹם פּוֹסֵק וְקוֹרֵא פָּסוּק רִאשׁוֹן שֶׁל ק״ש (קַבָּלַת מַלְכוּת שָׁמַיִם) וַהֲלָכָה שֶׁיֵּשׁ בָּהּ יְצִיאַת מִצְרַיִם. וְאִם יֵשׁ שְׁהוּת לק״ש וּתְפִלָּה — אֵינוֹ פּוֹסֵק כְּלָל (עיין סימן פ״ט).

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק״ו (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — מִי הֵם הַפְּטוּרִים מִתְּפִלָּה

היסוד : מִי הֵם הַפְּטוּרִים מִתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַשְׁוָאַת חִיּוּב ק״ש וּתְפִלָּה עִם פְּטוֹר מְלַוֵּי הַמֵּת. אחר כך חִיּוּב נָשִׁים וַעֲבָדִים וִיסוֹדוֹ. אחר כך חִנּוּךְ הַקְּטַנִּים. ולבסוף מִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ וְחַיֵּי עוֹלָם וְחַיֵּי שָׁעָה.

הַפְּטוּרִים וְהַחַיָּבִים — חוּץ מִמְּלַוֵּי הַמֵּת

הַפְּטוּרִים וְהַחַיָּבִים — חוּץ מִמְּלַוֵּי הַמֵּת

היסוד. כָּל הַפָּטוּר מִק״ש פָּטוּר מִתְּפִלָּה ; וְכָל שֶׁחַיָּב בִּק״ש חַיָּב בִּתְפִלָּה — חוּץ מִמְּלַוֵּי הַמֵּת שֶׁאֵין לַמִּטָּה צֹרֶךְ בָּהֶן (סעיף א).

המסקנה. טַעַם הָרַב : הַתְּפִלָּה מִדִּבְרֵי סוֹפְרִים וּבַעֲמִידָה, וְק״ש עִקָּרוֹ פָּסוּק רִאשׁוֹן — בְּקַל בַּעֲמִידָה. עיין סימן 105.

נָשִׁים וַעֲבָדִים — חַיָּבִים בִּתְפִלָּה

נָשִׁים וַעֲבָדִים — חַיָּבִים בִּתְפִלָּה

היסוד. נָשִׁים וַעֲבָדִים — חַיָּבִים בִּתְפִלָּה, מ״ע שֶׁלֹּא הַזְּמַן גְּרָמָא ; מַחֲלֹקֶת אִם עִיקָּרָהּ מִן הַתּוֹרָה אוֹ מִדִּבְרֵי סוֹפְרִים (סעיף ב).

המסקנה. וְכֵן עִקָּר : חַיָּבוֹת שַׁחֲרִית וּמִנְחָה, שֶׁהַתְּפִלָּה בַּקָּשַׁת רַחֲמִים ; עַרְבִית (רְשׁוּת) — לֹא קִבְּלוּ. עיין סימן 107.

חִנּוּךְ הַקְּטַנִּים

חִנּוּךְ הַקְּטַנִּים

היסוד. קְטַנִּים שֶׁהִגִּיעוּ לְחִנּוּךְ — לְחַנְּכָם לְהִתְפַּלֵּל י״ח עֶרֶב וָבֹקֶר (סעיף ג).

המסקנה. רַשַּׁאי לְהַאֲכִילָם קֹדֶם שַׁחֲרִית וְאָסוּר לְעַנּוֹתָם (עיין סימן רס״ט וְשמ״ג). עיין סימן 107.

מִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ — חַיֵּי עוֹלָם

מִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ — חַיֵּי עוֹלָם

היסוד. כְּגוֹן רשב״י וַחֲבֵרָיו — מַפְסִיק לק״ש וְלֹא לִתְפִלָּה ; אֲבָל אָנוּ מַפְסִיקִין לִשְׁתֵּיהֶן (סעיף ד).

המסקנה. אֵין לוֹ לְהָנִיחַ חַיֵּי עוֹלָם וְלַעֲסֹק בְּחַיֵּי שָׁעָה ; מִכָּל מָקוֹם קוֹרֵא פָּסוּק רִאשׁוֹן שֶׁל ק״ש. עיין סימן 89.

הפניה ללימוד. היקף הסימן — הַשְׁוָאַת ק״ש וּתְפִלָּה, חִיּוּב הַנָּשִׁים, הַחִנּוּךְ וְתוֹרָתוֹ אֻמָּנוּתוֹ — נוגע להנהגת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

מִי הֵם הַפְּטוּרִים מִתְּפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק״ו. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הַפְּטוּרִים וְהַחַיָּבִים בִּתְפִלָּה או״ח ק״ו — כָּל הַפָּטוּר מִק״ש פָּטוּר מִתְּפִלָּה ; מְלַוֵּי הַמֵּת פְּטוּרִים. מ״ב ק״ו — טַעַם פְּטוֹר מְלַוֵּי הַמֵּת וְגֶדֶר הַחִיּוּב. ערוה״ש ק״ו — הַשְׁוָאַת חִיּוּב ק״ש וּתְפִלָּה. דרך אדמו״ר הזקן — מְלַוֵּי הַמֵּת פְּטוּרִים, הוֹאִיל וְהִיא מִדִּבְרֵי סוֹפְרִים וּבַעֲמִידָה, וְק״ש עִקָּרָהּ פָּסוּק רִאשׁוֹן (סעיף א).
נָשִׁים וַעֲבָדִים בִּתְפִלָּה או״ח ק״ו — נָשִׁים וַעֲבָדִים חַיָּבִים בִּתְפִלָּה, מ״ע שֶׁלֹּא הַזְּמַן גְּרָמָא. מ״ב ק״ו — פְּרָטֵי חִיּוּב הַנָּשִׁים שַׁחֲרִית וּמִנְחָה. ערוה״ש ק״ו — גֶּדֶר חִיּוּב הַנָּשִׁים בַּתְּפִלָּה. דרך אדמו״ר הזקן — מַחֲלֹקֶת אִם עִיקָּרָהּ מִן הַתּוֹרָה (״וּלְעָבְדוֹ בְּכָל לְבַבְכֶם״) ; וְכֵן עִקָּר — חַיָּבוֹת שַׁחֲרִית וּמִנְחָה, וְעַרְבִית רְשׁוּת (סעיף ב).
חִנּוּךְ הַקְּטַנִּים או״ח ק״ו — קְטַנִּים שֶׁהִגִּיעוּ לְחִנּוּךְ חַיָּבִים לְחַנְּכָם. מ״ב ק״ו — גִּיל הַחִנּוּךְ לַתְּפִלָּה. ערוה״ש ק״ו — חִיּוּב הָאָב בְּחִנּוּךְ בְּנוֹ. דרך אדמו״ר הזקן — לְחַנְּכָם לְהִתְפַּלֵּל י״ח עֶרֶב וָבֹקֶר ; רַשַּׁאי לְהַאֲכִילָם קֹדֶם וְאָסוּר לְעַנּוֹתָם (סעיף ג).
מִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ או״ח ק״ו — כְּגוֹן רשב״י — מַפְסִיק לק״ש וְלֹא לִתְפִלָּה ; אֲבָל אָנוּ מַפְסִיקִין לִשְׁתֵּיהֶן. מ״ב ק״ו — גֶּדֶר תּוֹרָתוֹ אֻמָּנוּתוֹ בַּזְּמַן הַזֶּה. ערוה״ש ק״ו — הַהַפְסָקָה לַתְּפִלָּה בַּזְּמַן הַזֶּה. דרך אדמו״ר הזקן — אֵינוֹ מִתְפַּלֵּל כָּל עִיקָּר (חַיֵּי עוֹלָם וְחַיֵּי שָׁעָה) ; אֲבָל אָנוּ מַפְסִיקִין, וּפוֹסֵק לְפָסוּק רִאשׁוֹן שֶׁל ק״ש (סעיף ד).

הטבלה נערכה מטקסט המחבר (אורח חיים ק״ו, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה א ו-ו, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ׳, שבת י״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — תְּפִלָּה כְּבַקָּשַׁת רַחֲמִים

קו א — תְּפִלָּה כְּבַקָּשַׁת רַחֲמִים

יסוד הסימן : אוֹתוֹ טַעַם עוֹבֵר בַּסִּימָן — הַתְּפִלָּה בַּקָּשַׁת רַחֲמִים (סעיפים א-ב-ד).

קריאת הרב : טַעַם זֶה פּוֹטֵר אֶת מְלַוֵּי הַמֵּת, מְחַיֵּב אֶת הַנָּשִׁים שַׁחֲרִית וּמִנְחָה, וּמוֹצִיא אֶת תּוֹרָתוֹ אֻמָּנוּתוֹ. עיין סימן 105.

מסקנה למעשה : הַתְּפִלָּה מוּבֶנֶת כְּבַקָּשָׁה, לֹא כַּאֲמִירָה גְּרֵידָא.

קו ב — עִיקַּר מִצְוַת תְּפִלָּה

קו ב — עִיקַּר מִצְוַת תְּפִלָּה

יסוד הסימן : מַחֲלֹקֶת אִם עִיקָּרָהּ מִן הַתּוֹרָה (״וּלְעָבְדוֹ בְּכָל לְבַבְכֶם״) אוֹ מִדִּבְרֵי סוֹפְרִים (סעיף ב).

קריאת הרב : הָרַב מֵבִיא שְׁתֵּי הַדֵּעוֹת וּפוֹסֵק — וְכֵן עִקָּר — שֶׁהַנָּשִׁים חַיָּבוֹת שַׁחֲרִית וּמִנְחָה. עיין סימן 107.

מסקנה למעשה : הַנָּשִׁים מִתְפַּלְלוֹת שַׁחֲרִית וּמִנְחָה ; עַרְבִית (רְשׁוּת) כְּפִי הַמִּנְהָג.

קו ג — חַיֵּי עוֹלָם וְחַיֵּי שָׁעָה

קו ג — חַיֵּי עוֹלָם וְחַיֵּי שָׁעָה

יסוד הסימן : מִי שֶׁתּוֹרָתוֹ אֻמָּנוּתוֹ אֵינוֹ מִתְפַּלֵּל כָּל עִיקָּר ; אֲבָל אָנוּ מַפְסִיקִין (סעיף ד).

קריאת הרב : אֵין לוֹ לְהָנִיחַ חַיֵּי עוֹלָם (הַלִּמּוּד) וְלַעֲסֹק בְּחַיֵּי שָׁעָה ; אֲבָל אָנוּ, שֶׁמְּבַטְּלִין הַרְבֵּה, מַפְסִיקִין. עיין סימן 89.

מסקנה למעשה : לְמַעֲשֶׂה מַפְסִיקִין לק״ש וְלִתְפִלָּה ; תּוֹרָתוֹ אֻמָּנוּתוֹ אֲמִתִּי נָדִיר בַּזְּמַן הַזֶּה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״ו ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״ו (מִי הֵם הַפְּטוּרִים מִתְּפִלָּה)

  • ① הַפְּטוּרִים וְהַחַיָּבִים. כָּל הַפָּטוּר מִק״ש פָּטוּר מִתְּפִלָּה וְלַהֵפֶךְ — חוּץ מִמְּלַוֵּי הַמֵּת, פְּטוּרִים מִתְּפִלָּה (מִדִּבְרֵי סוֹפְרִים, בַּעֲמִידָה). יסוד : שוע״ר ק״ו סעיף א.
  • ② נָשִׁים. הַנָּשִׁים חַיָּבוֹת שַׁחֲרִית וּמִנְחָה (בַּקָּשַׁת רַחֲמִים, וְכֵן עִקָּר) ; עַרְבִית (רְשׁוּת) — רֻבָּן לֹא קִבְּלוּ. יסוד : שוע״ר ק״ו סעיף ב.
  • ③ חִנּוּךְ. לְחַנֵּךְ אֶת הַקְּטַנִּים שֶׁהִגִּיעוּ לְחִנּוּךְ לְהִתְפַּלֵּל י״ח עֶרֶב וָבֹקֶר. יסוד : שוע״ר ק״ו סעיף ג.
  • ④ לְעַנּוֹתָם. רַשַּׁאי לְהַאֲכִיל אֶת הַקְּטַנִּים קֹדֶם שַׁחֲרִית וְאָסוּר לְעַנּוֹתָם (עיין סימן רס״ט וְשמ״ג). יסוד : שוע״ר ק״ו סעיף ג.
  • ⑤ תּוֹרָתוֹ אֻמָּנוּתוֹ. לְמַעֲשֶׂה אָנוּ מַפְסִיקִין לק״ש וְלִתְפִלָּה (שֶׁלֹּא כְּרשב״י) ; תּוֹרָתוֹ אֻמָּנוּתוֹ קוֹרֵא לְפָחוֹת פָּסוּק רִאשׁוֹן שֶׁל ק״ש. יסוד : שוע״ר ק״ו סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק״ו (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַפְּטוּרִים, חִיּוּב הַנָּשִׁים, הַחִנּוּךְ, הַהַפְסָקָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
