דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן נ״א · דִּינֵי הַתְּפִלָּה מִבָּרוּךְ שֶׁאָמַר עַד יִשְׁתַּבַּח

דרכו של אדמו״ר הזקן בפְּסוּקֵי דְּזִמְרָה — מִבָּרוּךְ שֶׁאָמַר עַד יִשְׁתַּבַּח: שתי הבְּרָכוֹת המקיפות אותם ; אִסּוּר הַהֶפְסֵק (בְּדִבּוּר, שְׁאִילַת שָׁלוֹם) ; הכַּוָּנָה בְּפוֹתֵחַ אֶת יָדֶךָ ; לַעֲמֹד, וּמִזְמוֹר לְתוֹדָה בִּנְגִינָה — ברגישות חב״ד.

למה רמה 4 מיוחדת לאדמו״ר הזקן? השולחן ערוך של אדמו״ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתבו אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. דף זה מאסף את שיטת הרב על סימן נ״א — דִּינֵי הַתְּפִלָּה מִבָּרוּךְ שֶׁאָמַר עַד יִשְׁתַּבַּח. הסימן כולל אצל הרב י״ג סְעִיפִים: תַּקָּנַת פְּסוּקֵי דְּזִמְרָה וּשְׁתֵּי בִּרְכוֹתֵיהֶם (בָּרוּךְ שֶׁאָמַר, יִשְׁתַּבַּח), אִסּוּר הַהֶפְסֵק וּשְׁאִילַת שָׁלוֹם, הַכַּוָּנָה בְּפוֹתֵחַ אֶת יָדֶךָ, דִּקְדּוּק שִׁירַת הַיָּם, וּמִזְמוֹר לְתוֹדָה בִּנְגִינָה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בדף זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן נ״א — י״ג סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנסרקה בספריא (Shulchan Arukh HaRav, Orach Chayim נ״א). אדמו״ר הזקן דן בדִּינֵי הַתְּפִלָּה מִבָּרוּךְ שֶׁאָמַר עַד יִשְׁתַּבַּח — שתי הברכות, ההֶפְסֵק, שְׁאִילַת שָׁלוֹם, הכַּוָּנָה וּמִזְמוֹר לְתוֹדָה. הביאורים הם שלנו. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן נ״א — דִּינֵי הַתְּפִלָּה מִבָּרוּךְ שֶׁאָמַר עַד יִשְׁתַּבַּח

הטקסט המלא של שולחן ערוך אדמו״ר הזקן

שולחן ערוך הרב, סימן נ״א — דִּינֵי הַתְּפִלָּה מִבָּרוּךְ שֶׁאָמַר עַד יִשְׁתַּבַּח — וּבוֹ י״ג סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.51), ולאחריו ביאור קצר. הסימן כולל י״ג סְעִיפִים.

סעיף א תִּקְּנוּ לוֹמַר פְּסוּקֵי דְּזִמְרָה בְּכָל יוֹם, בְּרָכָה לִפְנֵיהֶם — בָּרוּךְ שֶׁאָמַר, וּלְאַחֲרֵיהֶם — יִשְׁתַּבַּח.

דִּינֵי תְּפִלָּה מִ״בָּרוּךְ שֶׁאָמַר״ עַד ״יִשְׁתַּבַּח״ וּבוֹ י״ג סְעִיפִים:תִּקְּנוּ לוֹמַר פְּסוּקֵי דְּזִמְרָה בְּכָל יוֹם, דְּהַיְנוּ מִ״תְּהִלָּה לְדָוִד״ עַד ״כֹּל הַנְּשָׁמָה תְּהַלֵּל וְגו״, כְּדֵי לְסַדֵּר שִׁבְחוֹ שֶׁל מָקוֹם בָּרוּךְ הוּא תְּחִלָּה וְאַחַר כָּךְ יִתְפַּלֵּל, כִּי כֵן עָשָׂה מֹשֶׁה רַבֵּנוּ עָלָיו הַשָּׁלוֹם בִּתְפִלָּתוֹ, שֶׁאָמַר תְּחִלָּה ״אַתָּה הַחִלּוֹתָ לְהַרְאוֹת אֶת עַבְדְּךָ אֶת גָּדְלְךָ וְגו׳״ וְאַחַר כָּךְ ״אֶעְבְּרָה נָא וְגו׳״, וּכְדֵי לַעֲמֹד בִּתְפִלָּה מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה. וְלָמָּה בָּחֲרוּ בְּמִזְמוֹרִים אֵלּוּ? לְפִי שֶׁאָמְרוּ: כָּל הָאוֹמֵר תְּהִלָּה לְדָוִד ג׳ פְּעָמִים בְּכָל יוֹם — מֻבְטָח לוֹ שֶׁהוּא בֶּן עוֹלָם הַבָּא, וְאַגַּב אוֹתוֹ מִזְמוֹר תִּקְּנוּ לוֹמַר עַד סִיּוּם תְּהִלִּים שֶׁיֵּשׁ שָׁם י׳ פְּעָמִים ״הַלְלוּיָהּ״ כְּנֶגֶד עֲשָׂרָה הִלּוּלִים שֶׁבְּסֵפֶר תְּהִלִּים: נִצּוּחַ, נִגּוּן, שִׁיר וְכו׳. וְתִקְּנוּ בְּרָכָה לִפְנֵיהֶם וּלְאַחֲרֵיהֶם, לִפְנֵיהֶם — ״בָּרוּךְ שֶׁאָמַר״, וְאַחֲרֵיהֶם — ״יִשְׁתַּבַּח״. וְיֵשׁ מְקוֹמוֹת שֶׁנּוֹהֲגִין לְהוֹסִיף ״הוֹדוּ לַה׳ קִרְאוּ בִשְׁמוֹ״ כַּאֲשֶׁר הוּא כָּתוּב בְּסֵפֶר דִּבְרֵי הַיָּמִים, לְפִי שֶׁדָּוִד קְבָעוֹ לְאָמְרָהּ בְּכָל יוֹם לִפְנֵי הָאָרוֹן.וּ״מִזְמוֹר לְתוֹדָה״, לְפִי שֶׁכָּל הַשִּׁירוֹת עֲתִידִין לְהִבָּטֵל חוּץ מִ״מִּזְמוֹר לְתוֹדָה״, וְלָכֵן יֵשׁ לִמְשֹׁךְ אוֹתוֹ בִּנְגִינָה. וְנוֹהֲגִין בִּמְדִינוֹת אֵלּוּ שֶׁלֹּא לְאָמְרָהּ בְּשַׁבָּת וְיוֹם טוֹב, מִפְּנֵי שֶׁאֵין תּוֹדָה קְרֵבָה בָּהֶן. וְכֵן בִּימֵי הַפֶּסַח אֵין תּוֹדָה קְרֵבָה מִשּׁוּם חָמֵץ שֶׁבָּהּ. וְכֵן בְּעֶרֶב פֶּסַח, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תכ״ט. וְכֵן בְּעֶרֶב יוֹם כִּפּוּר, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תר״ד. אֲבָל בְּט׳ בְּאָב אוֹמְרִים אוֹתוֹ, שֶׁהֲרֵי בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ הָיָה קַיָּם הָיוּ מַקְרִיבִין תּוֹדָה גַּם בְּט׳ בְּאָב.וְיֵשׁ שֶׁמּוֹסִיפִין עוֹד פְּסוּקִים אֲחֵרִים, וְכָל מָקוֹם וּמָקוֹם לְפִי מִנְהָגוֹ. וְיֵשׁ נוֹהֲגִין לוֹמַר כָּל הַפְּסוּקִים וְהַמִּזְמוֹרִים שֶׁמּוֹסִיפִים קֹדֶם ״בָּרוּךְ שֶׁאָמַר״. וּבִמְדִינוֹת אֵלּוּ נוֹהֲגִין לְאָמְרָם אַחַר ״בָּרוּךְ שֶׁאָמַר״, כְּדֵי שֶׁיִּהְיֶה לְכֻלָּם בְּרָכָה לִפְנֵיהֶם וּלְאַחֲרֵיהֶם.וְנוֹהֲגִין לְהוֹסִיף קֹדֶם בִּרְכַּת ״יִשְׁתַּבַּח״: ״וַיְבָרֶךְ דָּוִיד״ עַד ״מְהַלְלִים לְשֵׁם תִּפְאַרְתֶּךָ״ כַּאֲשֶׁר הוּא בְּסֵפֶר דִּבְרֵי הַיָּמִים, ״וִיבָרְכוּ שֵׁם כְּבוֹדֶךָ״ עַד שִׁירַת הַיָּם כַּאֲשֶׁר הוּא בְּסֵפֶר נְחֶמְיָה, וְשִׁירַת הַיָּם, לְפִי שֶׁכָּל אוֹתָם ט״ו לְשׁוֹנוֹת שֶׁל שֶׁבַח הַסְּדוּרוֹת דְּבִרְכוֹת יִשְׁתַּבַּח דְּרָשׁוּם חֲכָמִים מִתּוֹךְ שִׁירַת הַיָּם וּמִתּוֹךְ אוֹתָן פְּסוּקִים שֶׁל ״וַיְבָרֶךְ דָּוִיד״:

תַּקָּנַת פְּסוּקֵי דְּזִמְרָה. תִּקְּנוּ לוֹמַר בְּכָל יוֹם מִתְּהִלָּה לְדָוִד עַד כֹּל הַנְּשָׁמָה, לְסַדֵּר שִׁבְחוֹ שֶׁל מָקוֹם קֹדֶם שֶׁיִּתְפַּלֵּל כְּמֹשֶׁה רַבֵּנוּ ; בְּרָכָה לִפְנֵיהֶם — בָּרוּךְ שֶׁאָמַר, וּלְאַחֲרֵיהֶם — יִשְׁתַּבַּח. מִזְמוֹר לְתוֹדָה נִמְשָׁךְ בִּנְגִינָה, וְאֵין אוֹמְרִים בְּשַׁבָּת וְיוֹם טוֹב, בְּפֶסַח וְעֶרֶב פֶּסַח (חָמֵץ) וְעֶרֶב יוֹם כִּפּוּר, אֲבָל בְּט׳ בְּאָב אוֹמְרִים. וּמוֹסִיפִין וַיְבָרֶךְ דָּוִיד וְשִׁירַת הַיָּם קֹדֶם יִשְׁתַּבַּח.

סעיף ב יֵשׁ לוֹמַר בָּרוּךְ שֶׁאָמַר בְּנִגּוּן וּמְעֻמָּד, פ״ז תֵּבוֹת, וְיֶאֱחֹז בְּב׳ צִיצִית.

יֵשׁ לוֹמַר ״בָּרוּךְ שֶׁאָמַר״ בְּנִגּוּן וּבִנְעִימָה, כִּי הוּא שִׁיר נָאֶה וְנֶחְמָד. וְתִקְּנוּהוּ אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה עַל יְדֵי אִגֶּרֶת שֶׁנָּפְלָה מִן הַשָּׁמַיִם וּמְצָאוּהוּ כָּתוּב בּוֹ, וְיֵשׁ בָּהּ פ״ז תֵּבוֹת, וְסִימָנוֹ ״רֹאשׁוֹ כֶּתֶם פָּז״. וְקִבְּלוּ לְאָמְרוֹ מְעֻמָּד אֲפִלּוּ בְּיָחִיד. וְעַל פִּי הַסּוֹד יֵשׁ לוֹמַר: ״בְּפֶה עַמּוֹ״ וְלֹא ״בְּפִי עַמּוֹ״, ״בַּתִּשְׁבָּחוֹת״ בְּחִירִ״ק הַתָּי״ו. וְיֶאֱחֹז בְּב׳ צִיצִית שֶׁבַּטַּלִּית שֶׁהֵן כְּנֶגֶד פָּנָיו בִּשְׁעַת אֲמִירַת ״בָּרוּךְ שֶׁאָמַר״:

בָּרוּךְ שֶׁאָמַר. לְאָמְרוֹ בְּנִגּוּן וּבִנְעִימָה, שֶׁהוּא שִׁיר נָאֶה ; תִּקְּנוּהוּ אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה, וְיֵשׁ בּוֹ פ״ז תֵּבוֹת (רֹאשׁוֹ כֶּתֶם פָּז). אוֹמְרוֹ מְעֻמָּד אֲפִלּוּ בְּיָחִיד, וְאוֹחֵז בְּב׳ צִיצִית שֶׁכְּנֶגֶד פָּנָיו.

סעיף ג סִיֵּם בָּרוּךְ שֶׁאָמַר קֹדֶם הַשְּׁלִיחַ צִבּוּר — עוֹנֶה אָמֵן, וְאֵין זֶה הֶפְסֵק.

אִם סִיֵּם ״בָּרוּךְ שֶׁאָמַר״ קֹדֶם שֶׁסִּיֵּם הַשְּׁלִיחַ צִבּוּר — עוֹנֶה אַחֲרָיו אָמֵן. וְאֵין אָמֵן זֶה חָשׁוּב הֶפְסֵק בֵּין פְּסוּקֵי דְּזִמְרָה לִבְרָכָה שֶׁלִּפְנֵיהֶם, לְפִי שֶׁגַּם אָמֵן שֶׁבַח הוּא וּכְעֵין זִמְרָה (אֲבָל אִם עֲדַיִן לֹא סִיֵּם הַבְּרָכָה — לֹא יַעֲנֶה, שֶׁאֵין מַפְסִיקִין בְּאֶמְצַע שׁוּם בְּרָכָה אֲפִלּוּ בַּעֲנִיַּת אָמֵן, אֶלָּא בְּאָמֵן שֶׁל ״הָאֵל הַקָּדוֹשׁ״ וְשֶׁל ״שׁוֹמֵעַ תְּפִלָּה״ בִּלְבַד, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ס״ו). וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁבְּרָכָה זוֹ שֶׁל ״בָּרוּךְ שֶׁאָמַר״ הוֹאִיל וְאֵינָהּ מֻזְכֶּרֶת בַּתַּלְמוּד — מַפְסִיקִין בָּהּ לְכָל אָמֵן. וְאִם סִיֵּם עִם הַשְּׁלִיחַ צִבּוּר בְּבַת אַחַת — יֵשׁ אוֹמְרִים שֶׁלֹּא יַעֲנֶה אָמֵן, מִפְּנֵי שֶׁהָעוֹנֶה אָמֵן אַחַר בִּרְכוֹתָיו — הֲרֵי זֶה מְגֻנֶּה. וְאַף שֶׁיֵּשׁ מָקוֹם לַחֲלֹק וְלוֹמַר שֶׁאֵין זֶה עוֹנֶה אָמֵן אַחַר בִּרְכוֹתָיו, שֶׁעַל כָּל פָּנִים הוּא עוֹנֶה אַחַר בִּרְכַּת הַשְּׁלִיחַ צִבּוּר — אַף עַל פִּי כֵן יֵשׁ לָחֹש לְדִבְרֵיהֶם.וּמִכָּל מָקוֹם, אִם סִיֵּם עִם הַשְּׁלִיחַ צִבּוּר בְּבַת אַחַת בִּרְכַּת ״יִשְׁתַּבַּח״ וְכַיּוֹצֵא בָּהּ מֵהַבְּרָכוֹת שֶׁיֵּשׁ מְקוֹמוֹת נוֹהֲגִין לַעֲנוֹת בָּהּ אָמֵן אֲפִלּוּ אַחַר בִּרְכוֹת עַצְמוֹ כְּמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמָן, אַף שֶׁבִּמְדִינוֹת אֵלּוּ אֵין נוֹהֲגִין כֵן לַעֲנוֹת אַחַר בִּרְכַּת עַצְמוֹ, מִכָּל מָקוֹם כְּשֶׁמְּסַיֵּם עִם הַשְּׁלִיחַ צִבּוּר בְּבַת אַחַת — יָכוֹל לַעֲנוֹת. וְהוּא הַדִּין אִם הַשְּׁלִיחַ צִבּוּר סִיֵּם בְּרָכָה אַחַת וְהוּא סִיֵּם בְּרָכָה אַחֶרֶת — יָכוֹל לַעֲנוֹת אַחַר בִּרְכַּת הַשְּׁלִיחַ צִבּוּר, אֲפִלּוּ סִיֵּם הוּא בְּרָכָה שֶׁאֵין עוֹנִין עָלָיו אַחַר בִּרְכַּת עַצְמוֹ לְדִבְרֵי הַכֹּל:

עֲנִיַּת אָמֵן. סִיֵּם בָּרוּךְ שֶׁאָמַר קֹדֶם הַשְּׁלִיחַ צִבּוּר — עוֹנֶה אַחֲרָיו אָמֵן, וְאֵינוֹ הֶפְסֵק שֶׁגַּם אָמֵן שֶׁבַח הוּא. אֲבָל בְּאֶמְצַע הַבְּרָכָה אֵין מַפְסִיקִין. סִיֵּם עִם הַחַזָּן בְּבַת אַחַת — יֵשׁ אוֹמְרִים שֶׁלֹּא יַעֲנֶה (מְגֻנֶּה אַחַר בִּרְכוֹתָיו), וְיֵשׁ לָחֹש.

סעיף ד צָרִיךְ לִזָּהֵר מִלְּהַפְסִיק בְּדִבּוּר מִבָּרוּךְ שֶׁאָמַר עַד סוֹף י״ח.

צָרִיךְ לִזָּהֵר מִלְּהַפְסִיק בְּדִבּוּר מִשֶּׁיַּתְחִיל ״בָּרוּךְ שֶׁאָמַר״ עַד סוֹף י״ח, לְפִי שֶׁ״בָּרוּךְ שֶׁאָמַר״ הִיא בְּרָכָה שֶׁלִּפְנֵי פְּסוּקֵי דְּזִמְרָה וְ״יִשְׁתַּבַּח״ בְּרָכָה לְאַחֲרֵיהֶם וְאָסוּר לְהַפְסִיק בֵּינֵיהֶם אֲפִלּוּ לְצֹרֶךְ מִצְוָה, כְּמוֹ שֶׁאָסוּר לְהַפְסִיק בִּקְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ. וּבֵין ״יִשְׁתַּבַּח״ לְ״יוֹצֵר אוֹר״ — גַּם כֵּן אֵין לְהַפְסִיק, וְהַמַּפְסִיק — עֲבֵרָה הִיא בְּיָדוֹ, וְחוֹזֵר עָלָיו מֵעוֹרְכֵי הַמִּלְחָמָה, לְפִי שֶׁפְּסוּקֵי דְּזִמְרָה נִתְקְנוּ לְהַסְדִּיר שִׁבְחוֹ שֶׁל מָקוֹם בָּרוּךְ הוּא קֹדֶם הַתְּפִלָּה, וְכֵיוָן שֶׁכֵּן הוּא — אֵינוֹ בְּדִין לְדַבֵּר בֵּין סִדּוּר הַשֶּׁבַח וְהַתְּפִלָּה, אֶלָּא אִם כֵּן לְצֹרֶךְ מִצְוָה כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן נ״ד.וּבֵין בִּרְכוֹת קְרִיאַת שְׁמַע לִתְפִלָּה — אָסוּר לְהַפְסִיק אֲפִלּוּ לְצֹרֶךְ מִצְוָה, מִפְּנֵי שֶׁצָּרִיךְ לִסְמֹךְ גְּאֻלָּה לִתְפִלָּה:

הֶפְסֵק בְּדִבּוּר. צָרִיךְ לִזָּהֵר מִלְּהַפְסִיק מִשֶּׁיַּתְחִיל בָּרוּךְ שֶׁאָמַר עַד סוֹף י״ח, שֶׁהִיא בְּרָכָה לִפְנֵי פְּסוּקֵי דְּזִמְרָה וְיִשְׁתַּבַּח לְאַחֲרֵיהֶם, כִּקְרִיאַת שְׁמַע ; וְהַמַּפְסִיק עֲבֵרָה בְּיָדוֹ. וּבֵין גְּאֻלָּה לִתְפִלָּה אָסוּר אֲפִלּוּ לְצֹרֶךְ מִצְוָה.

סעיף ה שְׁאִילַת שָׁלוֹם — בֵּין הַפְּרָקִים וּבָאֶמְצַע ; וְלַעֲנִיַּת אָמֵן מֻתָּר.

וּשְׁאִילַת שָׁלוֹם מִ״בָּרוּךְ שֶׁאָמַר״ עַד ״יִשְׁתַּבַּח״ — דִּינָהּ כְּמוֹ בִּקְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ, שֶׁבֵּין הַפְּרָקִים — שׁוֹאֵל מִפְּנֵי הַכָּבוֹד וּמֵשִׁיב שָׁלוֹם לְכָל אָדָם, וּבָאֶמְצַע — שׁוֹאֵל מִפְּנֵי הַיִּרְאָה וּמֵשִׁיב מִפְּנֵי הַכָּבוֹד, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ס״ו. וְאֵלּוּ הֵן בֵּין הַפְּרָקִים: בֵּין ״בָּרוּךְ שֶׁאָמַר״ לְהַתְחָלַת הַמִּזְמוֹרִים, וּבֵין כָּל מִזְמוֹר לַחֲבֵרוֹ, וּבֵין סוֹף הַמִּזְמוֹרִים לְ״יִשְׁתַּבַּח״. וּפְסוּקִים הַמְלֻקָּטִים שֶׁאֵינָם מִזְמוֹר שָׁלֵם — יֵשׁ מִי שֶׁאוֹמֵר שֶׁכָּל שְׁנַיִם אוֹ שְׁלֹשָׁה פְּסוּקִים הַנֶּאֱמָרִים כָּאן כְּסֵדֶר שֶׁהֵם כְּתוּבִים בִּמְקוֹמָם — דִּינָם כְּמִזְמוֹר שָׁלֵם. וּכְשֶׁמַּפְסִיק בֵּין אֶחָד לַחֲבֵרוֹ — דִּינוֹ כְּמַפְסִיק בְּאֶמְצַע מִזְמוֹר. וּבֵין סוֹפָם לִתְחִלַּת פָּסוּק אַחֵר — דִּינוֹ כְּבֵין הַפְּרָקִים, כְּגוֹן בִּמְדִינוֹת אֵלּוּ מֵ״הוֹדוּ״ עַד ״אָמֵן וְהַלֵּל לַה׳״, הוּא פֶּרֶק אֶחָד אַף עַל פִּי שֶׁאֵינָם מִזְמוֹר שָׁלֵם בְּדִבְרֵי הַיָּמִים. וְכֵן ב׳ פְּסוּקִים ״תְּנוּ עֹז וְגו׳ נוֹרָא אֱלֹהִים וְגו׳״ עַד ״בָּרוּךְ אֱלֹהִים״ — הֵם גַּם כֵּן פֶּרֶק אֶחָד, הוֹאִיל וְהֵם סְמוּכִים זֶה לָזֶה בִּתְהִלִּים. וְכֵן ב׳ פְּסוּקִים ״אֵל נְקָמוֹת וְגו׳ הִנָּשֵׂא וְגו׳״ עַד ״עַל גֵּאִים״. וְכֵן ג׳ פְּסוּקִים: ״נַפְשֵׁנוּ חִכְּתָה וְגו׳ כִּי בוֹ וְגו׳ יְהִי חַסְדְּךָ וְגו׳״. וְכֵן ג׳ פְּסוּקִים: ״יְהִי שֵׁם ה׳ וְגו׳ מִמִּזְרַח וְגו׳ רָם וְגו׳״. וְכֵן ב׳ פְּסוּקִים: ״בָּרוּךְ ה׳ אֱלֹהִים וְגו׳ וּבָרוּךְ שֵׁם וְגו׳״. וְכֵן מִ״וַּיְבָרֶךְ דָּוִיד״ עַד ״לְשֵׁם תִּפְאַרְתֶּךָ״, וּמֵ״אַתָּה הוּא״ עַד ״בְּמַיִם עַזִּים״, וּמִ״וַּיּוֹשַׁע״ עַד סוֹף וּמֻתָּר לַעֲנוֹת אָמֵן עַל כָּל הַבְּרָכוֹת מִ״בָּרוּךְ שֶׁאָמַר״ עַד ״יִשְׁתַּבַּח״ אֲפִלּוּ בְּאֶמְצַע הַמִּזְמוֹר, וְאֵין צָרִיךְ לוֹמַר בֵּין הַמִּזְמוֹרִים לַבְּרָכָה שֶׁלִּפְנֵיהֶם אוֹ שֶׁלְּאַחֲרֵיהֶם (לְפִי שֶׁאָמֵן הוּא גַּם כֵּן שֶׁבַח וּכְעֵין זִמְרָה — אֵינוֹ חָשׁוּב הֶפְסֵק בִּפְסוּקֵי דְּזִמְרָה).וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁאֲפִלּוּ בְּאֶמְצַע הַבְּרָכָה שֶׁלִּפְנֵיהֶם מֻתָּר לַעֲנוֹת אָמֵן, אַף עַל פִּי שֶׁהִיא פּוֹתַחַת בְּבָרוּךְ וְחוֹתֶמֶת בְּבָרוּךְ — אֵין לָחֹש בְּמַה שֶּׁמַּפְסִיק בֵּין פְּתִיחָה לַחֲתִימָה בַּעֲנִיַּת אָמֵן, הוֹאִיל וּבְרָכָה זוֹ אֵינָהּ מֻזְכֶּרֶת בַּתַּלְמוּד.וְאִם הוּא צָרִיךְ לְהַפְסִיק בְּאֶמְצַע פְּסוּקֵי דְּזִמְרָה — טוֹב שֶׁיֹּאמַר ״בָּרוּךְ ה׳ לְעוֹלָם״ עַד ״וַיְבָרֶךְ דָּוִיד״ קֹדֶם שֶׁיְּדַבֵּר. וּכְשֶׁיַּחֲזֹר לַמָּקוֹם שֶׁפָּסַק — יֹאמַר גַּם כֵּן אֵלּוּ הַפְּסוּקִים, לְפִי שֶׁהֵם כְּמוֹ בְּרָכָה:

שְׁאִילַת שָׁלוֹם. כִּקְרִיאַת שְׁמַע: בֵּין הַפְּרָקִים שׁוֹאֵל מִפְּנֵי הַכָּבוֹד וּמֵשִׁיב לְכָל אָדָם, וּבָאֶמְצַע מִפְּנֵי הַיִּרְאָה. אֲבָל לַעֲנִיַּת אָמֵן — מֻתָּר אֲפִלּוּ בְּאֶמְצַע. וְאִם צָרִיךְ לְהַפְסִיק — טוֹב שֶׁיֹּאמַר בָּרוּךְ ה׳ לְעוֹלָם קֹדֶם.

סעיף ו אָסוּר לוֹמַר מִי יְמַלֵּל אַחַר בָּרוּךְ שֶׁאָמַר קֹדֶם הוֹדוּ.

אָסוּר לוֹמַר פָּסוּק ״מִי יְמַלֵּל גְּבוּרוֹת ה׳״ אַחַר ״בָּרוּךְ שֶׁאָמַר״ קֹדֶם ״הוֹדוּ״, מִפְּנֵי שֶׁנִּרְאֶה כְּאוֹמֵר: מִי שֶׁרוֹצֶה לְמַלֵּל גְּבוּרוֹת ה׳ וּלְהַשְׁמִיעַ כָּל תְּהִלָּתוֹ — יֹאמַר ״הוֹדוּ לַה׳ וְגו׳״, וַהֲרֵי זֶה חֵרוּף:

מִי יְמַלֵּל. אָסוּר לוֹמַר פָּסוּק מִי יְמַלֵּל גְּבוּרוֹת ה׳ אַחַר בָּרוּךְ שֶׁאָמַר קֹדֶם הוֹדוּ, מִפְּנֵי שֶׁנִּרְאֶה כְּחֵרוּף.

סעיף ז צָרִיךְ לְהַפְסִיק מְעַט בֵּין אֱלִילִים וּבֵין וַה׳ שָׁמַיִם עָשָׂה.

צָרִיךְ לְהַפְסִיק מְעַט בֵּין ״אֱלִילִים״ וּבֵין ״וַה׳ שָׁמַיִם עָשָׂה״, שֶׁלֹּא יְהֵא נִרְאֶה כְּחוֹזֵר לְמַעְלָה. וְגַם בֵּין ״כִּי״ לְ״כָּל״ יֵשׁ פְּסִיק, וְלָכֵן הַכָּ״ף דְּגוּשָׁה. וּבֵין ״הָעַמִּים״ לֶ״אֱלִילִים״ וּבֵין ״שָׁמַיִם״ לְ״עָשָׂה״ — צָרִיךְ גַּם כֵּן לְהַפְסִיק קְצָת שֶׁלֹּא תִּבָּלַע הַמֵּ״ם. וְכֵן בְּכָל מֵ״ם שֶׁבְּסוֹף תֵּבָה וְתֵבָה שֶׁלְּאַחֲרֶיהָ מַתְחֶלֶת בְּאָלֶ״ף אוֹ בְּעַיִ״ן — צָרִיךְ לִזָּהֵר כֵּן, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ס״א:

הַפְסָקָה בֵּין אֱלִילִים. צָרִיךְ לְהַפְסִיק מְעַט בֵּין אֱלִילִים וּבֵין וַה׳ שָׁמַיִם עָשָׂה, וְכֵן בְּכָל מֵ״ם סוֹפִית שֶׁהַתֵּבָה שֶׁאַחֲרֶיהָ מַתְחֶלֶת בְּאָלֶ״ף אוֹ עַיִ״ן, שֶׁלֹּא תִּבָּלַע — כְּמוֹ שֶׁיִּתְבָּאֵר סִימָן ס״א.

סעיף ח צָרִיךְ לְכַוֵּן בְּפוֹתֵחַ אֶת יָדֶךָ, וְאִם לֹא כִּוֵּן — חוֹזֵר.

צָרִיךְ לְכַוֵּן בְּ״פוֹתֵחַ אֶת יָדֶךָ״ פֵּרוּשׁ הַמִּלּוֹת הַמּוֹצִיא בִּשְׂפָתָיו, וְאִם לֹא כִּוֵּן — צָרִיךְ לַחֲזֹר וְלוֹמַר בְּכַוָּנָה, לְפִי שֶׁעִקַּר מַה שֶּׁקָּבְעוּ לוֹמַר ״תְּהִלָּה לְדָוִד״ בְּכָל יוֹם הוּא מִפְּנֵי שֶׁבַח שֶׁבְּפָסוּק זֶה, שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא הוּא מַשְׁגִּיחַ עַל בִּרְיוֹתָיו לְפַרְנֵס. וְאַף עַכְשָׁו שֶׁאֵין אָנוּ חוֹזְרִין בִּשְׁבִיל חֶסְרוֹן כַּוָּנָה לְפִי שֶׁקָּרוֹב שֶׁאַף כְּשֶׁיַּחֲזֹר לֹא יְכַוֵּן כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ק״א, מִכָּל מָקוֹם כֵּיוָן שֶׁאֵינוֹ אֶלָּא פָּסוּק אֶחָד — חוֹזֵר, שֶׁבְּקַל יוּכַל לְכַוֵּן כְּשֶׁיִּרְצֶה:

פּוֹתֵחַ אֶת יָדֶךָ. צָרִיךְ לְכַוֵּן פֵּרוּשׁ הַמִּלּוֹת שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא מַשְׁגִּיחַ לְפַרְנֵס, וְאִם לֹא כִּוֵּן — חוֹזֵר, שֶׁהוּא פָּסוּק אֶחָד וּבְקַל יְכַוֵּן.

סעיף ט נוֹהֲגִין לוֹמַר וַאֲנַחְנוּ נְבָרֵךְ יָהּ אַחַר תְּהִלָּה לְדָוִד.

נוֹהֲגִין לוֹמַר פָּסוּק ״וַאֲנַחְנוּ נְבָרֵךְ יָהּ״ אַחַר ״תְּהִלָּה לְדָוִד״ לְסַיֵּם בְּ״הַלְלוּיָהּ״, כְּדֵי לְשַׁלְשֵׁל ״הַלְלוּיָהּ״ אַחַר ״הַלְלוּיָהּ״, שֶׁכָּל הַמִּזְמוֹרוֹת שֶׁלְּאַחֲרֶיהָ פּוֹתְחוֹת בְּ״הַלְלוּיָהּ״ וּמִסְתַּיְּמוֹת בְּ״הַלְלוּיָהּ״. וְעוֹד, לְפִי שֶׁכָּל הָאוֹמֵר ״תְּהִלָּה לְדָוִד״ מֻבְטָח לוֹ שֶׁהוּא בֶּן עוֹלָם הַבָּא, לָכֵן אוֹמְרִים אַחֲרָיו: וַאֲנַחְנוּ נִזְכֶּה לְבָרֵךְ יָהּ גַּם לָעוֹלָם הַבָּא. וּמִטַּעַם זֶה אוֹמְרִים אוֹתוֹ גַּם בְּמִנְחָה, אַף עַל פִּי שֶׁאֵין אַחֲרָיו מִזְמוֹר הַלְלוּיָהּ:

וַאֲנַחְנוּ נְבָרֵךְ יָהּ. נוֹהֲגִין לוֹמְרוֹ אַחַר תְּהִלָּה לְדָוִד לְסַיֵּם בְּהַלְלוּיָהּ, וּלְרַמֵּז שֶׁנִּזְכֶּה לְבָרֵךְ יָהּ גַּם לָעוֹלָם הַבָּא ; וּמִטַּעַם זֶה אוֹמְרוֹ גַּם בְּמִנְחָה.

סעיף י נוֹהֲגִין לִכְפֹּל כֹּל הַנְּשָׁמָה וְה׳ יִמְלֹךְ, סוֹף פְּסוּקֵי דְּזִמְרָה וְהַשִּׁירָה.

נוֹהֲגִין לִכְפֹּל ״כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ״, לְפִי שֶׁהוּא סוֹף פְּסוּקֵי דְּזִמְרָה. וְכֵן פָּסוּק ״ה׳ יִמְלֹךְ לְעֹלָם וָעֶד״, שֶׁהוּא סוֹף הַשִּׁירָה. וְיֵשׁ מוֹסִיפִין פָּסוּק ״כִּי בָא סוּס פַּרְעֹה וְגו׳״, שֶׁגַּם הוּא מֵהַשִּׁירָה, עַיֵּן בְּיוֹרֶה דֵּעָה סִימָן ער״ה:

כְּפִילַת כֹּל הַנְּשָׁמָה. נוֹהֲגִין לִכְפֹּל כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ (סוֹף פְּסוּקֵי דְּזִמְרָה) וְה׳ יִמְלֹךְ (סוֹף הַשִּׁירָה) ; וְיֵשׁ מוֹסִיפִין כִּי בָא סוּס פַּרְעֹה — עַיֵּן בְּיוֹרֶה דֵּעָה.

סעיף יא אֵין לְהִשְׁתַּחֲווֹת בִּוְעַתָּה וּבְכָל קוֹמָה ; נָהֲגוּ לַעֲמֹד לְוַיְבָרֶךְ דָּוִיד.

כְּשֶׁמַּגִּיעַ לִ״וְעַתָּה אֱלֹהֵינוּ מוֹדִים אֲנַחְנוּ לָךְ״ אוֹ לִ״וְכָל קוֹמָה לְפָנֶיךָ תִּשְׁתַּחֲוֶה״ — אֵין לִשְׁחוֹת וּלְהִשְׁתַּחֲווֹת שָׁם, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קי״ג. וְנָהֲגוּ לַעֲמֹד לְ״וַיְבָרֶךְ דָּוִיד״ עַד ״תִּפְאַרְתֶּךָ״. וְיֵשׁ מוֹסִיפִין לַעֲמֹד עַד שֶׁאוֹמְרִים ״אַתָּה הוּא ה׳ הָאֱלֹהִים״ — רָאשֵׁי תֵּבוֹת ״אֶהְיֶה״, וְאַחַר כָּךְ יוֹשְׁבִין. וּכְשֶׁאוֹמְרִים ״וְאַתָּה מוֹשֵׁל בַּכֹּל״ — נוֹתְנִים צְדָקָה מְעֻמָּד:

וְעַתָּה וּבְכָל קוֹמָה. אֵין לִשְׁחוֹת בִּוְעַתָּה אֱלֹהֵינוּ וּבִוְכָל קוֹמָה — עַיֵּן סִימָן קי״ג. וְנָהֲגוּ לַעֲמֹד לְוַיְבָרֶךְ דָּוִיד, וְיֵשׁ עַד אַתָּה הוּא ה׳ הָאֱלֹהִים ; וּבִוְאַתָּה מוֹשֵׁל בַּכֹּל נוֹתְנִים צְדָקָה מְעֻמָּד.

סעיף יב דִּקְדּוּקֵי דָּגֵשׁ וְרָפֶה בְּשִׁירַת הַיָּם כְּפִי הַמְּסוֹרוֹת.

״מִי כָּמֹכָה נֶאְדָּר וְגו׳״ — הַכָּ״ף דְּגוּשָׁה, אֲבָל ״מִי כָמֹכָה בָּאֵלִים״ וְכֵן ״כָּל עַצְמוֹתַי תֹּאמַרְנָה ה׳ מִי כָמוֹךָ״ — הַכָ״ף רְפוּיָה, ״יִדְּמוּ כָּאָבֶן״ — הַכָּ״ף דְּגוּשָׁה, ״עַם זוּ גָּאַלְתָּ״ — הַג׳ דְּגוּשָׁה, שֶׁכָּךְ הוּא בַּמְּסוֹרוֹת. וְכֵן בִּשְׁאָר דָּגוּשׁ וְרָפֶה — צָרִיךְ לִזָּהֵר וְלִקְרוֹת כְּמוֹ שֶׁנִּמְסַר בַּמְּסוֹרוֹת. וְלֹא יְרַפֶּה הֶחָזָק וְלֹא יְחַזֵּק הָרָפֶה, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ס״א:

דָּגֵשׁ וְרָפֶה. דִּקְדּוּקֵי שִׁירַת הַיָּם כְּפִי הַמְּסוֹרוֹת — מִי כָּמֹכָה נֶאְדָּר (כ׳ דְּגוּשָׁה) לְעֻמַּת מִי כָמֹכָה בָּאֵלִים (רְפוּיָה) וְכַיּוֹצֵא ; לֹא יְרַפֶּה הֶחָזָק וְלֹא יְחַזֵּק הָרָפֶה, כְּמוֹ שֶׁיִּתְבָּאֵר סִימָן ס״א.

סעיף יג אֵין אוֹמְרִים הַזְּמִירוֹת בִּמְרוּצָה כִּי אִם בְּנַחַת מִלָּה בְּמִלָּה.

אֵין אוֹמְרִים הַזְּמִירוֹת בִּמְרוּצָה, כִּי אִם בְּנַחַת מִלָּה בְּמִלָּה כְּדֵי לְכַוֵּן בָּהֶם יָפֶה. וְאוֹתָן שֶׁאוֹמְרִים אוֹתָם בִּמְרוּצָה (כְּשֶׁיֵּשׁ שָׁם מִנְיָן) — לֹא יָפֶה הֵם עוֹשִׂים שֶׁמְּקַצְּרִים בְּשִׁבְחוֹ שֶׁל מָקוֹם בִּשְׁבִיל שְׁאִילַת צָרְכָּם, הֲיֵשׁ מוֹשֵׁל שֶׁיִּתְרַצֶּה בְּכָךְ? וְהוּא הַדִּין לְאוֹתָם שֶׁמְּכַוְּנִים בַּתְּפִלָּה בִּבְרָכוֹת שֶׁהֵן תְּחִנָּה וּבַקָּשַׁת צָרְכֵי הַגּוּף וְלֹא בַּבְּרָכוֹת שֶׁהֵן שֶׁבַח וְהוֹדָאָה לַמָּקוֹם בָּרוּךְ הוּא:

בְּנַחַת. אֵין אוֹמְרִים הַזְּמִירוֹת בִּמְרוּצָה כִּי אִם בְּנַחַת מִלָּה בְּמִלָּה לְכַוֵּן יָפֶה ; הַמְקַצְּרִים בְּשִׁבְחוֹ שֶׁל מָקוֹם בִּשְׁבִיל שְׁאִילַת צָרְכָּם — לֹא יָפֶה הֵם עוֹשִׂים.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim נ״א (מהדורת קה״ת, י״ג סְעִיפִים). ביאור: DAAT. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.

2 · היסוד

היסוד — פְּסוּקֵי דְּזִמְרָה וּבִרְכוֹתֵיהֶם

היסוד: שתי הברכות ואיסור ההפסק

קריאת אדמו״ר הזקן על סימן זה בשלושה צירים. תחילה שתי הברכות המקיפות את פְּסוּקֵי דְּזִמְרָה — בָּרוּךְ שֶׁאָמַר לִפְנֵיהֶם, יִשְׁתַּבַּח לְאַחֲרֵיהֶם. אַחַר כָּךְ אִסּוּר הַהֶפְסֵק (בְּדִבּוּר, שְׁאִילַת שָׁלוֹם) בֵּין שְׁתֵּי הַבְּרָכוֹת, כִּקְרִיאַת שְׁמַע. וּלְבַסּוֹף הַכַּוָּנָה, הַנַּחַת וְדִין מִזְמוֹר לְתוֹדָה.

בָּרוּךְ שֶׁאָמַר וְיִשְׁתַּבַּח — בְּרָכָה לִפְנֵיהֶם וּלְאַחֲרֵיהֶם

שתי הברכות של פְּסוּקֵי דְּזִמְרָה

העיקרון. תִּקְּנוּ בְּרָכָה לִפְנֵי פְּסוּקֵי דְּזִמְרָה (בָּרוּךְ שֶׁאָמַר) וְאַחֲרֵיהֶם (יִשְׁתַּבַּח), לְהַקִּיף אֶת הַשֶּׁבַח הַיּוֹמִי (סעיף א). בָּרוּךְ שֶׁאָמַר בְּנִגּוּן וּמְעֻמָּד, וְיֵשׁ בּוֹ פ״ז תֵּבוֹת (סעיף ב).

התוצאה. עוֹנֶה אָמֵן אַחַר בָּרוּךְ שֶׁאָמַר שֶׁל הַחַזָּן וְאֵינוֹ הֶפְסֵק, שֶׁאָמֵן שֶׁבַח הוּא (סעיף ג). עַל הַבָּא בְּאִחוּר וְעַל בִּרְכוֹת הַתְּפִלָּה עַיֵּן סימן 52.

אִסּוּר הַהֶפְסֵק בִּפְסוּקֵי דְּזִמְרָה

אִסּוּר הַהֶפְסֵק בֵּין שְׁתֵּי הַבְּרָכוֹת

העיקרון. צָרִיךְ לִזָּהֵר מִלְּהַפְסִיק בְּדִבּוּר מִבָּרוּךְ שֶׁאָמַר עַד סוֹף הָעֲמִידָה, אֲפִלּוּ לְצֹרֶךְ מִצְוָה — כִּקְרִיאַת שְׁמַע (סעיף ד). וְהַמַּפְסִיק — עֲבֵרָה בְּיָדוֹ.

ההרחבה. בִּשְׁאִילַת שָׁלוֹם מַבְחִינִים בֵּין הַפְּרָקִים וּבָאֶמְצַע ; וּלְעִנְיַן עֲנִיַּת אָמֵן פְּסוּקֵי דְּזִמְרָה קַלִּים מִקְּרִיאַת שְׁמַע (סעיף ה). עַיֵּן סימן 66 (ק״ש) וְסימן 53 (הֶפְסֵק לְצֹרֶךְ מִצְוָה).

כַּוָּנָה, נַחַת וּמִזְמוֹר לְתוֹדָה

כַּוָּנָה, נַחַת וּמִזְמוֹר לְתוֹדָה

העיקרון. צָרִיךְ כַּוָּנָה בְּפוֹתֵחַ אֶת יָדֶךָ, וְאִם לֹא כִּוֵּן חוֹזֵר (סעיף ח) ; וְאוֹמְרִים הַזְּמִירוֹת בְּנַחַת מִלָּה בְּמִלָּה, לֹא בִּמְרוּצָה (סעיף יג).

התוצאה. מִזְמוֹר לְתוֹדָה בִּנְגִינָה, עִם יְמֵי הַיּוֹצְאִים מִן הַכְּלָל (סעיף א) ; וְעוֹמְדִים לְוַיְבָרֶךְ דָּוִיד וְאֵין מִשְׁתַּחֲוִים בִּוְעַתָּה / וְכָל קוֹמָה (סעיף יא). עַיֵּן סימן 50.

הפניה ללימוד. משקל הסימן הזה — שתי הבְּרָכוֹת, אִסּוּר הַהֶפְסֵק וְדִקְדּוּק פְּסוּקֵי דְּזִמְרָה — נוגע גם בהנהגה היומיומית של התפילה וגם ברגישות חב״ד על דִּקְדּוּק המילים. דף זה מציג זאת ברמת העיקרון המתועד בטקסט של הרב, בלי להוסיף דבר מומצא. ללימוד מעמיק ולהנהגה אישית — יש ללמוד עם רב של חב״ד ובתוך המקורות עצמם.

דִּינֵי תְּפִלָּה מִבָּרוּךְ שֶׁאָמַר עַד יִשְׁתַּבַּח — בעיקרון · ללימוד, פנה לרב של חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרכו של אדמו״ר הזקן עם המחבר, הרמ״א (רמ״א) והמשנה ברורה (משנה ברורה) בנקודות המפתח של סימן נ״א. שורת שולחן ערוך הרב (שולחן ערוך הרב) מובלטת — היא חושפת את רגישות הרב כפי שהיא עולה מי״ג סעיפיו. הרב מפרש את הטעם ואת הפרט במקום שהמחבר מסתפק באמירה.

נושא מחבר רמ״א משנה ברורה דרך אדמו״ר הזקן (שולחן ערוך הרב)
בָּרוּךְ שֶׁאָמַר וְיִשְׁתַּבַּח או״ח נ״א — בָּרוּךְ שֶׁאָמַר לִפְנֵי פְּסוּקֵי דְּזִמְרָה וְיִשְׁתַּבַּח לְאַחֲרֵיהֶם. רמ״א — נָהֲגוּ לַעֲמֹד לְבָרוּךְ שֶׁאָמַר, וַיְבָרֶךְ דָּוִיד וְיִשְׁתַּבַּח. מ״ב נ״א / ברכות ל״ב ע״א — יְסַדֵּר שִׁבְחוֹ וְאַחַר כָּךְ יִתְפַּלֵּל. דרך אדמו״ר הזקן — בְּרָכָה לִפְנֵיהֶם וּלְאַחֲרֵיהֶם, בָּרוּךְ שֶׁאָמַר מְעֻמָּד וּבְנִגּוּן, פ״ז תֵּבוֹת (סעיפים א-ב).
הֶפְסֵק בְּדִבּוּר וּשְׁאִילַת שָׁלוֹם או״ח נ״א — לֹא לְהַפְסִיק מִבָּרוּךְ שֶׁאָמַר עַד סוֹף הַתְּפִלָּה. — מ״ב נ״א — פְּרָטֵי בֵּין הַפְּרָקִים וְהֶפְסֵק לְצֹרֶךְ מִצְוָה. דרך אדמו״ר הזקן — אָסוּר אֲפִלּוּ לְצֹרֶךְ מִצְוָה כִּקְרִיאַת שְׁמַע ; שְׁאִילַת שָׁלוֹם כְּבֵין הַפְּרָקִים ; אָמֵן מֻתָּר (סעיפים ד-ה).
מִזְמוֹר לְתוֹדָה וּכְפִילַת פְּסוּקִים או״ח נ״א — מִזְמוֹר לְתוֹדָה בִּנְגִינָה. רמ״א — אֵין אוֹמְרוֹ בְּשַׁבָּת וְיוֹם טוֹב, פֶּסַח וְעֶרֶב פֶּסַח. מ״ב נ״א — טַעַם חָמֵץ וְעֶרֶב יוֹם כִּפּוּר. דרך אדמו״ר הזקן — מִזְמוֹר לְתוֹדָה בִּנְגִינָה, אֲבָל בְּט׳ בְּאָב אוֹמְרִים ; וְכוֹפֵל כֹּל הַנְּשָׁמָה וְה׳ יִמְלֹךְ (סעיפים א,י).

הטבלה נערכה מטקסט המחבר (אורח חיים נ״א, מקור ספריא) ומנושאי הכלים הניתנים לאימות (בית יוסף, טור, רמב״ם הלכות תפילה ז, רא״ש, רבנו ירוחם, אבודרהם, כל בו, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ל״ב ע״א, י״ד ע״א, שבת קי״ח ע״ב). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (י״ג סְעִיפִים) המובא ב§1. ייחוסי סעיף קטן שאינם ודאיים נותרו « על דרך הסברה ».

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה — מעבר ליישום הפשוט. כל אחת בנויה: עיקרון הסימן → קריאת הרב → תוצאה מעשית. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — בָּרוּךְ שֶׁאָמַר וְיִשְׁתַּבַּח כְּבִרְכוֹת פְּסוּקֵי דְּזִמְרָה

שתי הברכות וגדר השבח

עיקרון הסימן: הַבְּרָכוֹת לִפְנֵיהֶם וּלְאַחֲרֵיהֶם שֶׁל פְּסוּקֵי דְּזִמְרָה (סעיפים א-ג).

קריאת הרב: בָּרוּךְ שֶׁאָמַר בְּנִגּוּן וּמְעֻמָּד, פ״ז תֵּבוֹת ; יִשְׁתַּבַּח כּוֹלֶלֶת ט״ו לְשׁוֹנוֹת שֶׁל שֶׁבַח (סעיפים א-ב).

תוצאה מעשית: לְהַקִּיף אֶת פְּסוּקֵי דְּזִמְרָה בִּשְׁתֵּי הַבְּרָכוֹת ; וְלַעֲמֹד לְבָרוּךְ שֶׁאָמַר.

קו ב — אִסּוּר הַהֶפְסֵק וּשְׁאִילַת שָׁלוֹם

אִסּוּר הַהֶפְסֵק וּשְׁאִילַת שָׁלוֹם

עיקרון הסימן: לֹא לְהַפְסִיק מִבָּרוּךְ שֶׁאָמַר עַד סוֹף י״ח (סעיפים ד-ה).

קריאת הרב: הָאִסּוּר כִּקְרִיאַת שְׁמַע ; שְׁאִילַת שָׁלוֹם בֵּין הַפְּרָקִים וּבָאֶמְצַע, וַעֲנִיַּת אָמֵן קַלָּה יוֹתֵר. עַיֵּן סימן 66.

תוצאה מעשית: לֹא לְדַבֵּר בֵּין שְׁתֵּי הַבְּרָכוֹת ; וּמֻתָּר לַעֲנוֹת אָמֵן אֲפִלּוּ בְּאֶמְצַע.

קו ג — כַּוָּנָה, דִּקְדּוּק וּמִזְמוֹר לְתוֹדָה

כַּוָּנָה, דִּקְדּוּק וּמִזְמוֹר לְתוֹדָה

עיקרון הסימן: פּוֹתֵחַ אֶת יָדֶךָ, הַדִּקְדּוּק וּמִזְמוֹר לְתוֹדָה (סעיפים ז-יג).

קריאת הרב: הַכַּוָּנָה מְעַכֶּבֶת בְּפוֹתֵחַ אֶת יָדֶךָ (חוֹזֵר) ; דִּקְדּוּק בֵּין אֱלִילִים ; בְּנַחַת מִלָּה בְּמִלָּה. עַיֵּן סימן 113.

תוצאה מעשית: לְכַוֵּן בְּפוֹתֵחַ אֶת יָדֶךָ ; לוֹמַר בְּנַחַת, וְאֵין לְהִשְׁתַּחֲווֹת בִּוְעַתָּה / וְכָל קוֹמָה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן נ״א ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, בהפניה לרב לפרטי המקרים (שתי הברכות, ההֶפְסֵק, שְׁאִילַת שָׁלוֹם, הכַּוָּנָה וּמִזְמוֹר לְתוֹדָה).

לחסיד חב״ד — סימן נ״א (דִּינֵי הַתְּפִלָּה מִבָּרוּךְ שֶׁאָמַר עַד יִשְׁתַּבַּח)

  • ① שתי הברכות. לוֹמַר בָּרוּךְ שֶׁאָמַר לִפְנֵי פְּסוּקֵי דְּזִמְרָה (מְעֻמָּד, בְּנִגּוּן) וּיִשְׁתַּבַּח לְאַחֲרֵיהֶם. יסוד: שוע״ר נ״א סעיפים א-ב.
  • ② אין להפסיק. אֵין הֶפְסֵק בְּדִבּוּר מִבָּרוּךְ שֶׁאָמַר עַד סוֹף הָעֲמִידָה, אֲפִלּוּ לְצֹרֶךְ מִצְוָה. יסוד: שוע״ר נ״א סעיף ד.
  • ③ שְׁאִילַת שָׁלוֹם / אָמֵן. שׁוֹאֵל וּמֵשִׁיב לְפִי בֵּין הַפְּרָקִים / בָּאֶמְצַע ; וְאָמֵן מֻתָּר אֲפִלּוּ בְּאֶמְצַע. יסוד: שוע״ר נ״א סעיף ה.
  • ④ כַּוָּנָה בְּפוֹתֵחַ אֶת יָדֶךָ. לְכַוֵּן בְּפוֹתֵחַ אֶת יָדֶךָ ; וְאִם לֹא — חוֹזֵר. יסוד: שוע״ר נ״א סעיף ח.
  • ⑤ מִזְמוֹר לְתוֹדָה. בִּנְגִינָה ; לֹא בְּשַׁבָּת/יו״ט, פֶּסַח, עֶרֶב פֶּסַח, עֶרֶב כִּפּוּר, אֲבָל כֵּן בְּט׳ בְּאָב ; וְלוֹמַר בְּנַחַת. יסוד: שוע״ר נ״א סעיפים א, יג.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן נ״א (הטקסט האמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — ההֶפְסֵק, הכַּוָּנָה, מִזְמוֹר לְתוֹדָה — יש להתייעץ עם רב, ולחב״ד עם רב של חב״ד.
