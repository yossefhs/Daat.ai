דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ע״ג · דִּין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו

דרכו של אדמו״ר הזקן בדִּין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו : נְגִיעַת בָּשָׂר — טַלִּית מַפְסֶקֶת מִמָּתְנַיִם וּלְמַטָּה ; אִשְׁתּוֹ כְּגוּפוֹ — קוֹרֵא בַּחֲזָרַת פָּנִים, וְיֵשׁ מִי שֶׁאוֹסֵר ; הַיָּשֵׁן עִם בָּנָיו — קְטַנִּים בְּלֹא הַפְסָקָה, גְּדוֹלִים צָרִיךְ הַפְסָקָה, וְגִיל הַקַּטְנוּת — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ע״ג — דִּין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו, כיצד יתנהג בשעת קריאת שמע. אצל הרב הסימן נפרש לג׳ סְעִיפִים : נְגִיעַת בָּשָׂר וְהַפְסָקַת טַלִּית מִמָּתְנַיִם וּלְמַטָּה ; אִשְׁתּוֹ כְּגוּפוֹ וַחֲזָרַת פָּנִים ; הַיָּשֵׁן עִם בָּנָיו עִם חֲלֻקַּת קָטָן / גָּדוֹל וְגִיל הַקַּטְנוּת.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ע״ג — ג׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ע״ג). אדמו״ר הזקן דן בו בְּדִין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ע״ג — דִּין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ע״ג — דִּין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.73), ואחריו תרגומנו. הסימן כולל ג׳ סעיפים (סעיף k → ‎.73.k).

סעיף א שְׁנַיִם הַיְשֵׁנִים בְּטַלִּית אֶחָד וּבְשָׂרָם נוֹגֵעַ — טַלִּית מַפְסֶקֶת מִמָּתְנַיִם וּלְמַטָּה.

הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו אֵיךְ יִתְנַהֵג וּבוֹ ג׳ סְעִיפִים:שְׁנַיִם שֶׁהֵם יְשֵׁנִים עֲרֻמִּים וּמִתְכַּסִּים בְּטַלִּית אֶחָד וּבְשַׂר שְׁנֵיהֶם נוֹגְעִים זֶה בָּזֶה — לֹא יִקְרָא קְרִיאַת שְׁמַע, אֶלָּא אִם כֵּן הָיְתָה טַלִּית מַפְסֶקֶת בֵּינֵיהֶם מִמָּתְנֵיהֶם וּלְמַטָּה, מִפְּנֵי שֶׁנְּגִיעַת בְּשָׂרָם מִמָּתְנֵיהֶם וּלְמַטָּה מְבִיאָה אוֹתוֹ לִידֵי הִרְהוּר בִּשְׁעַת קְרִיאָתוֹ, וְאָסוּר מִשּׁוּם ״וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר״. וְאַף עַל פִּי שֶׁאֵין הַכָּתוּב מְדַבֵּר אֶלָּא בְּעֶרְוָתוֹ מַמָּשׁ כְּשֶׁהִיא מְגֻלָּה וְנִרְאֵית כְּמוֹ שֶׁיִּתְבָּאֵר שָׁם, מִכָּל מָקוֹם מִדִּבְרֵי סוֹפְרִים בְּשָׁעָה שֶׁ״ה׳ אֱלֹהֶיךָ מִתְהַלֵּךְ בְּקֶרֶב מַחֲנֶךָ״, דְּהַיְנוּ כְּשֶׁעוֹסֵק בִּקְרִיאַת שְׁמַע אוֹ בִּתְפִלָּה אוֹ בְּתַלְמוּד תּוֹרָה כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ע״ד — כָּל דָּבָר שֶׁמֵּבִיא לִידֵי הִרְהוּר נִקְרָא עֶרְוָה, כְּמוֹ שֶׁאָמְרוּ: טֶפַח בְּאִשָּׁה עֶרְוָה, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ע״ה. וְאַף עַל פִּי (ב) שֶׁאֲפִלּוּ לָעֶרְוָה מַמָּשׁ מוֹעִיל כִּסּוּי — כָּאן אֵין כִּסּוּי מוֹעִיל, כֵּיוָן שֶׁמִּכָּל מָקוֹם הוּא בָּא לִידֵי הִרְהוּר עַל יְדֵי שֶׁמַּרְגִּישׁ נְגִיעַת בְּשַׂר חֲבֵרוֹ בִּבְשָׂרוֹ מִמָּתְנַיִם וּלְמַטָּה. וּמִטַּעַם זֶה אָסְרוּ שְׁמִיעַת קוֹל זֶמֶר אִשָּׁה שֶׁהִיא עֶרְוָה, אַף עַל פִּי שֶׁהַקּוֹל אֵינוֹ נִרְאֶה אֶלָּא נִשְׁמָע לוֹ מְבִיאוֹ לִידֵי הִרְהוּר, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ע״ה. (בַּמֶּה דְּבָרִים אֲמוּרִים? (ג) כְּשֶׁקּוֹרֵא קְרִיאַת שְׁמַע, אוֹ מִתְפַּלֵּל, אוֹ מְדַבֵּר בְּדִבְרֵי תּוֹרָה בְּפִיו, אֲבָל לְהַרְהֵר בְּדִבְרֵי תּוֹרָה מֻתָּר אֲפִלּוּ כְּשֶׁעֶרְוָתוֹ מַמָּשׁ מְגֻלָּה, מִטַּעַם שֶׁיִּתְבָּאֵר בְּסִימָן ע״ה. כָּל שֶׁכֵּן שֶׁשּׁוֹמֵעַ קוֹל אִשָּׁה, אוֹ שֶׁשּׁוֹכֵב עִם אִשְׁתּוֹ בְּקֵרוּב בָּשָׂר אֲפִלּוּ שֶׁלֹּא בַּחֲזָרַת פָּנִים, אַף עַל פִּי שֶׁאָסוּר לִקְרוֹת בְּפִיו כְּמוֹ שֶׁיִּתְבָּאֵר. וּמִטַּעַם זֶה מֻתָּר לְהַרְהֵר בִּדְבָרִים שֶׁבְּקְדֻשָּׁהּ בִּשְׁעַת תַּשְׁמִישׁ, שֶׁהִרְהוּר אֵינוֹ כְּדִבּוּר לְעִנְיָן זֶה, אַף עַל פִּי שֶׁהוּא כְּדִבּוּר לְעִנְיָן לְהַרְהֵר בִּמְבוֹאוֹת הַמְטֻנָּפוֹת):

נְגִיעַת בָּשָׂר. שְׁנַיִם הַיְשֵׁנִים תַּחַת כִּסּוּי אֶחָד וּבְשָׂרָם נוֹגֵעַ זֶה בָּזֶה — לֹא יִקְרָא ק״ש, אֶלָּא אִם כֵּן טַלִּית מַפְסֶקֶת מִמָּתְנֵיהֶם וּלְמַטָּה, מִפְּנֵי שֶׁהַנְּגִיעָה מְבִיאָה לִידֵי הִרְהוּר, וְאָסוּר מִשּׁוּם ״וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר״. וְאַף שֶׁהַכָּתוּב בְּעֶרְוָה מַמָּשׁ, מִדִּבְרֵי סוֹפְרִים בִּשְׁעַת ק״ש, תְּפִלָּה אוֹ תַּלְמוּד תּוֹרָה — כָּל דָּבָר הַמֵּבִיא לִידֵי הִרְהוּר נִקְרָא עֶרְוָה (טֶפַח בְּאִשָּׁה עֶרְוָה, סִימָן ע״ה). כָּאן אֵין כִּסּוּי מוֹעִיל, אֶלָּא הַפְסָקָה. בַּמֶּה דְּבָרִים אֲמוּרִים ? כְּשֶׁקּוֹרֵא, מִתְפַּלֵּל אוֹ מְדַבֵּר תּוֹרָה בְּפִיו ; אֲבָל לְהַרְהֵר מֻתָּר.

סעיף ב אִשְׁתּוֹ כְּגוּפוֹ — אֵין צָרִיךְ הַפְסָקָה ; קוֹרֵא בַּחֲזָרַת פָּנִים ; וְיֵשׁ מִי שֶׁאוֹסֵר.

הָיְתָה אִשְׁתּוֹ יְשֵׁנָה עִמּוֹ — אֵין צָרִיךְ לְהַפְסָקַת טַלִּית בֵּינוֹ לְבֵינָהּ, מִפְּנֵי שֶׁאִשְׁתּוֹ כְּגוּפוֹ, שֶׁהוֹאִיל שֶׁרָגִיל בָּהּ לֹא יָבוֹא לִידֵי הִרְהוּר. אֲבָל מִכָּל מָקוֹם אֵינוֹ קוֹרֵא אֶלָּא בַּחֲזָרַת פָּנִים מִמֶּנָּה, דְּהַיְנוּ שֶׁיַּחֲזִירוּ שְׁנֵיהֶם אֲחוֹרֵיהֶם זֶה לָזֶה, כְּדֵי שֶׁלֹּא יִהְיֶה בְּשָׂרָהּ נוֹגֵעַ בְּעֶרְוָתוֹ אוֹ בְּשָׂרוֹ בְּעֶרְוָתָהּ וְיָבוֹא לִידֵי הִרְהוּר. וְיֵשׁ מִי שֶׁאוֹסֵר אַף בְּאִשְׁתּוֹ שֶׁלֹּא בְּהַפְסָקַת טַלִּית. וְנָכוֹן לָחוּשׁ לִדְבָרָיו:

אִשְׁתּוֹ כְּגוּפוֹ. הָיְתָה אִשְׁתּוֹ יְשֵׁנָה עִמּוֹ — אֵין צָרִיךְ הַפְסָקַת טַלִּית, מִפְּנֵי שֶׁאִשְׁתּוֹ כְּגוּפוֹ, שֶׁרָגִיל בָּהּ וְלֹא יָבוֹא לִידֵי הִרְהוּר. אֲבָל אֵינוֹ קוֹרֵא אֶלָּא בַּחֲזָרַת פָּנִים, שֶׁיַּחֲזִירוּ שְׁנֵיהֶם אֲחוֹרֵיהֶם זֶה לָזֶה. וְיֵשׁ מִי שֶׁאוֹסֵר אַף בְּאִשְׁתּוֹ שֶׁלֹּא בְּהַפְסָקַת טַלִּית, וְנָכוֹן לָחוּשׁ לִדְבָרָיו.

סעיף ג בָּנָיו קְטַנִּים — בְּלֹא הַפְסָקָה ; גְּדוֹלִים — צָרִיךְ ; וְגִיל הַקַּטְנוּת.

הָיָה יָשֵׁן עִם בָּנָיו בְּעוֹדָם קְטַנִּים — מֻתָּר לִקְרוֹת בַּחֲזָרַת פָּנִים בְּלֹא הַפְסָקַת טַלִּית, שֶׁאֵין לָחוּשׁ שֶׁמָּא יָבוֹא לִידֵי הִרְהוּר מֵחֲמַת נְגִיעַת בְּשַׂר הַקְּטַנִּים בִּבְשָׂרוֹ, כָּל שֶׁאֵין בְּשָׂרוֹ נוֹגֵעַ בְּעֶרְוָתָן וְלֹא בְּשָׂרָם בְּעֶרְוָתוֹ. וְאִם הֵם גְּדוֹלִים — צָרִיךְ לְהַפְסָקַת טַלִּית. עַד כַּמָּה הֵם חֲשׁוּבִים קְטַנִּים? הַתִּינוֹק עַד שֶׁיִּהְיֶה לוֹ י״ב שָׁנָה, וְהַתִּינֹקֶת עַד שֶׁתִּהְיֶה לָהּ י״א שָׁנָה. וַאֲפִלּוּ הֵבִיאוּ ב׳ שְׂעָרוֹת בְּתוֹךְ זְמַן הַזֶּה — אֵינָן כְּלוּם. וּבִשְׁנַת י״ג לַתִּינוֹק וּשְׁנַת י״ב לַתִּינֹקֶת, אִם הֵבִיאוּ ב׳ שְׂעָרוֹת — אָסוּר בְּלֹא הַפְסָקַת טַלִּית, וְאִם לָאו — מֻתָּר. וּמִשְּׁנַת י״ג וְאֵילָךְ לַתִּינוֹק וי״ב לַתִּינֹקֶת אֲפִלּוּ לֹא הֵבִיאוּ ב׳ שְׂעָרוֹת — אָסוּר בְּלֹא הַפְסָקַת טַלִּית:

הַיָּשֵׁן עִם בָּנָיו. הָיָה יָשֵׁן עִם בָּנָיו בְּעוֹדָם קְטַנִּים — מֻתָּר לִקְרוֹת בַּחֲזָרַת פָּנִים בְּלֹא הַפְסָקַת טַלִּית, כָּל שֶׁאֵין בְּשָׂרוֹ נוֹגֵעַ בְּעֶרְוָתָן וְלֹא בְּשָׂרָם בְּעֶרְוָתוֹ. וְאִם הֵם גְּדוֹלִים — צָרִיךְ לְהַפְסָקַת טַלִּית. עַד כַּמָּה קְטַנִּים ? הַתִּינוֹק עַד י״ב שָׁנָה, הַתִּינֹקֶת עַד י״א ; וַאֲפִלּוּ הֵבִיאוּ ב׳ שְׂעָרוֹת קֹדֶם — אֵינָן כְּלוּם. בִּשְׁנַת י״ג / י״ב לְפִי הַשְּׂעָרוֹת ; וּמִכָּאן וְאֵילָךְ — אָסוּר בְּלֹא הַפְסָקָה.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ע״ג (מהדורת קה״ת, ג׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו

היסוד : דִּין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים, בִּזְהִירוּת וּבִצְנִיעוּת. תחילה נְגִיעַת בָּשָׂר וְהַפְסָקַת טַלִּית מִמָּתְנַיִם וּלְמַטָּה. אחר כך אִשְׁתּוֹ כְּגוּפוֹ וַחֲזָרַת פָּנִים. ולבסוף הַיָּשֵׁן עִם בָּנָיו עִם חֲלֻקַּת קָטָן / גָּדוֹל וְגִיל הַקַּטְנוּת.

נְגִיעַת בָּשָׂר — הַפְסָקַת טַלִּית

נְגִיעַת בָּשָׂר — הַפְסָקַת טַלִּית

היסוד. שְׁנַיִם תַּחַת כִּסּוּי אֶחָד וּבְשָׂרָם נוֹגֵעַ — לֹא יִקְרָא ק״ש, אֶלָּא אִם כֵּן טַלִּית מַפְסֶקֶת מִמָּתְנַיִם וּלְמַטָּה, מִפְּנֵי הַהִרְהוּר (סעיף א).

המסקנה. כָּל הַמֵּבִיא לִידֵי הִרְהוּר נִקְרָא עֶרְוָה בִּשְׁעַת ק״ש ; כָּאן אֵין כִּסּוּי מוֹעִיל אֶלָּא הַפְסָקָה. עיין סימן 75 (טפח באשה, קול באשה).

אִשְׁתּוֹ כְּגוּפוֹ — חֲזָרַת פָּנִים

אִשְׁתּוֹ כְּגוּפוֹ — חֲזָרַת פָּנִים

היסוד. אִשְׁתּוֹ — אֵין צָרִיךְ הַפְסָקַת טַלִּית, שֶׁאִשְׁתּוֹ כְּגוּפוֹ ; אֲבָל אֵינוֹ קוֹרֵא אֶלָּא בַּחֲזָרַת פָּנִים, אֲחוֹרֵיהֶם זֶה לָזֶה (סעיף ב).

המסקנה. וְיֵשׁ מִי שֶׁאוֹסֵר אַף בְּאִשְׁתּוֹ בְּלֹא הַפְסָקָה, וְנָכוֹן לָחוּשׁ לִדְבָרָיו. עיין סימן 74 (ק״ש כנגד ערוה).

הַיָּשֵׁן עִם בָּנָיו — גִּיל הַקַּטְנוּת

הַיָּשֵׁן עִם בָּנָיו — גִּיל הַקַּטְנוּת

היסוד. בָּנָיו קְטַנִּים — בַּחֲזָרַת פָּנִים בְּלֹא הַפְסָקָה ; גְּדוֹלִים — צָרִיךְ הַפְסָקַת טַלִּית (סעיף ג).

המסקנה. הַגְּבוּל : תִּינוֹק י״ב, תִּינֹקֶת י״א ; אחר כך שְׁנַת י״ג / י״ב לְפִי ב׳ שְׂעָרוֹת ; מִכָּאן וְאֵילָךְ — אָסוּר בְּלֹא הַפְסָקָה. עיין סימן 72.

הפניה ללימוד. היקף הסימן — הַנְּגִיעָה, חֲזָרַת פָּנִים וְגִיל הַקַּטְנוּת — נוגע לשאלות של צניעות והנהגה בשעת ק״ש. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בצניעות ובלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ע״ג. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מג׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
נְגִיעַת בָּשָׂר וְהַהַפְסָקָה או״ח ע״ג — בְּשַׂר שְׁנֵיהֶם נוֹגֵעַ לֹא יִקְרָא, אֶלָּא אִם כֵּן טַלִּית מַפְסֶקֶת מִמָּתְנֵיהֶם וּלְמַטָּה. מ״ב ע״ג — טַעַם הַנְּגִיעָה הַמְּבִיאָה לִידֵי הִרְהוּר וְגֶדֶר הַהַפְסָקָה. ערוה״ש ע״ג — גֶּדֶר קֵרוּב בָּשָׂר וְהַהֶכְשֵׁר לְלִקְרוֹת. דרך אדמו״ר הזקן — נוֹגֵעַ מִמָּתְנַיִם וּלְמַטָּה מֵבִיא לִידֵי הִרְהוּר ; אֵין כִּסּוּי מוֹעִיל, רַק הַפְסָקָה (סעיף א).
אִשְׁתּוֹ כְּגוּפוֹ וַחֲזָרַת פָּנִים או״ח ע״ג — אִשְׁתּוֹ כְּגוּפוֹ, קוֹרֵא בַּחֲזָרַת פָּנִים בְּלֹא הַפְסָקָה ; וְיֵשׁ מִי שֶׁאוֹסֵר וְנָכוֹן לָחוּשׁ. מ״ב ע״ג — פְּרָטֵי חֲזָרַת פָּנִים וְהַחֲשָׁשׁ לַדֵּעָה הָאוֹסֶרֶת. ערוה״ש ע״ג — גֶּדֶר ״כְּגוּפוֹ״ וְהַרְגִּילוּת שֶׁאֵינוֹ בָּא לִידֵי הִרְהוּר. דרך אדמו״ר הזקן — אֵין צָרִיךְ הַפְסָקָה, כְּגוּפוֹ ; אַךְ בַּחֲזָרַת פָּנִים, וְיֵשׁ אוֹסֵר וְנָכוֹן לָחוּשׁ (סעיף ב).
הַיָּשֵׁן עִם בָּנָיו וְגִיל הַקַּטְנוּת או״ח ע״ג — קְטַנִּים בְּלֹא הַפְסָקָה, גְּדוֹלִים צָרִיךְ ; תִּינוֹק עַד י״ב, תִּינֹקֶת עַד י״א. מ״ב ע״ג — פְּרָטֵי הַשָּׁנִים וְדִין הַסִּימָנִים (ב׳ שְׂעָרוֹת). ערוה״ש ע״ג — גֶּדֶר קָטָן וְגָדוֹל וְהַהַכְרָעָה בַּשָּׁנִים. דרך אדמו״ר הזקן — קְטַנִּים בְּלֹא הַפְסָקָה ; י״ב לַתִּינוֹק, י״א לַתִּינֹקֶת ; שְׁנַת י״ג/י״ב לְפִי הַשְּׂעָרוֹת ; וְאֵילָךְ — אָסוּר (סעיף ג).

הטבלה נערכה מטקסט המחבר (אורח חיים ע״ג, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ד). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — נְגִיעַת בָּשָׂר

קו א — נְגִיעַת בָּשָׂר

יסוד הסימן : בָּשָׂר בְּבָשָׂר תַּחַת כִּסּוּי אֶחָד — לֹא יִקְרָא, אֶלָּא טַלִּית מַפְסֶקֶת מִמָּתְנַיִם וּלְמַטָּה (סעיף א).

קריאת הרב : הַטַּעַם הוּא הַהִרְהוּר ; כִּסּוּי אֵינוֹ מוֹעִיל כְּשֶׁהַנְּגִיעָה נִשְׁאֶרֶת, רַק הַפְסָקָה. עיין סימן 75.

מסקנה למעשה : לְהַפְסִיק בְּטַלִּית מִמָּתְנַיִם וּלְמַטָּה קֹדֶם הַקְּרִיאָה.

קו ב — אִשְׁתּוֹ כְּגוּפוֹ

קו ב — אִשְׁתּוֹ כְּגוּפוֹ

יסוד הסימן : אֵין צָרִיךְ הַפְסָקָה, כְּגוּפוֹ ; מִכָּל מָקוֹם בַּחֲזָרַת פָּנִים (סעיף ב).

קריאת הרב : וְיֵשׁ מִי שֶׁאוֹסֵר, וְנָכוֹן לָחוּשׁ לִדְבָרָיו — הַזְּהִירוּת קוֹדֶמֶת. עיין סימן 74.

מסקנה למעשה : לְהַחֲזִיר פָּנִים (אֲחוֹרֵיהֶם זֶה לָזֶה) ; וְהַנָּכוֹן אַף לְהַפְסִיק.

קו ג — גִּיל הַקַּטְנוּת

קו ג — גִּיל הַקַּטְנוּת

יסוד הסימן : קְטַנִּים בְּלֹא הַפְסָקָה ; גְּדוֹלִים צָרִיךְ ; תִּינוֹק עַד י״ב, תִּינֹקֶת עַד י״א (סעיף ג).

קריאת הרב : הַגִּיל הוּא הַקָּנֶה מִדָּה הָרִאשׁוֹן ; ב׳ שְׂעָרוֹת אֵינָן נֶחְשָׁבוֹת אֶלָּא מִשְּׁנַת י״ג / י״ב. עיין סימן 72.

מסקנה למעשה : עִם בֵּן שֶׁנַּעֲשָׂה גָּדוֹל, לְהַפְסִיק קֹדֶם הַקְּרִיאָה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ע״ג ברגישות אדמו״ר הזקן וחב״ד — מוצגות בצניעות, ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ע״ג (דִּין הַיָּשֵׁן עִם אִשְׁתּוֹ וּבָנָיו)

  • ① נְגִיעַת בָּשָׂר. שְׁנַיִם שֶׁבְּשָׂרָם נוֹגֵעַ תַּחַת כִּסּוּי אֶחָד — לֹא יִקְרָא ק״ש, אֶלָּא אִם כֵּן טַלִּית מַפְסֶקֶת מִמָּתְנַיִם וּלְמַטָּה ; אֵין כִּסּוּי בְּעָלְמָא מוֹעִיל. יסוד : שוע״ר ע״ג סעיף א.
  • ② אִשְׁתּוֹ כְּגוּפוֹ. עִם אִשְׁתּוֹ — אֵין צָרִיךְ הַפְסָקָה (כְּגוּפוֹ), אֲבָל קוֹרֵא בַּחֲזָרַת פָּנִים, אֲחוֹרֵיהֶם זֶה לָזֶה. יסוד : שוע״ר ע״ג סעיף ב.
  • ③ וְיֵשׁ אוֹסֵר. יֵשׁ מִי שֶׁאוֹסֵר אַף בְּאִשְׁתּוֹ בְּלֹא הַפְסָקָה — וְנָכוֹן לָחוּשׁ לִדְבָרָיו : הַהַנְהָגָה הַנְּכוֹנָה חוֹשֶׁשֶׁת לְדַעְתּוֹ. יסוד : שוע״ר ע״ג סעיף ב.
  • ④ בָּנָיו קְטַנִּים. עִם בָּנָיו הַקְּטַנִּים — מֻתָּר בַּחֲזָרַת פָּנִים בְּלֹא הַפְסָקָה, כָּל שֶׁאֵין נְגִיעָה בַּעֶרְוָה. יסוד : שוע״ר ע״ג סעיף ג.
  • ⑤ גִּיל הַקַּטְנוּת. תִּינוֹק עַד י״ב, תִּינֹקֶת עַד י״א ; אחר כך שְׁנַת י״ג / י״ב לְפִי ב׳ שְׂעָרוֹת ; מִכָּאן וְאֵילָךְ גָּדוֹל — אָסוּר בְּלֹא הַפְסָקַת טַלִּית. יסוד : שוע״ר ע״ג סעיף ג.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ע״ג (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַנְּגִיעָה, חֲזָרַת פָּנִים, גִּיל הַקַּטְנוּת — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
