דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן מ׳ · דִּין אֵיךְ לִנְהֹג בִּקְדֻשַּׁת הַתְּפִלִּין

דרכו של אדמו״ר הזקן בדין קדושת התפילין וההנהגה בה: אִסּוּר תְּלִיָּתָן (דֶּרֶךְ בִּזָּיוֹן), הַכִּיס וְכְּלִי בְּתוֹךְ כְּלִי ; מְקוֹם הַתְּפִלִּין סְמוּךְ לַמִּטָּה (מַרְגְּלוֹת, מְרַאֲשׁוֹת, הַצַּד) ; הַיָּדַיִם עַסְקָנִיּוֹת אַחַר הַקֶּרִי ; וְהַחֲלִיצָה לִסְעוּדַת קֶבַע — ברגישות חב״ד.

למה רמה 4 מיוחדת לאדמו״ר הזקן? השולחן ערוך של אדמו״ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתבו אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. דף זה מאסף את שיטת הרב על סימן מ׳ — דִּין אֵיךְ לִנְהֹג בִּקְדֻשַּׁת הַתְּפִלִּין. הסימן כולל אצל הרב י״א סְעִיפִים, שבהם אדמו״ר הזקן פורש: אִסּוּר תְּלִיַּת הַתְּפִלִּין (דֶּרֶךְ בִּזָּיוֹן) וְעִנְיַן הַכִּיס ; גֶּדֶר הַכְּלִי בְּתוֹךְ כְּלִי וְהַמְיֻחָד (מְיֻחָד) לְתַשְׁמִישׁ הַמִּטָּה וְלַמְּזוּזָה ; מְקוֹם הַתְּפִלִּין סְמוּךְ לַמִּטָּה (מַרְגְּלוֹת, מְרַאֲשׁוֹת, הַצַּד) ; הַיָּדַיִם עַסְקָנִיּוֹת וְהַגּוּף נָקִי אַחַר הַקֶּרִי ; וְהַחֲלִיצָה לִסְעוּדַת קֶבַע.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בדף זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן מ׳ — י״א סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנסרקה בספריא (Shulchan Arukh HaRav, Orach Chayim מ׳). באחד עשר הסעיפים, אדמו״ר הזקן דן באִסּוּר תְּלִיַּת הַתְּפִלִּין (מִשּׁוּם דֶּרֶךְ בִּזָּיוֹן) וְעִנְיַן הַכִּיס, בכְּלִי בְּתוֹךְ כְּלִי וּבַמְיֻחָד (לְתַשְׁמִישׁ הַמִּטָּה וְלַמְּזוּזָה, עם ההפניה להלכות מזוזה, יורה דעה), במקום התפילין סמוך למיטה (מַרְגְּלוֹת, מְרַאֲשׁוֹת, כְּנֶגֶד צִדּוֹ), בהַיָּדַיִם עַסְקָנִיּוֹת וְהַקֶּרִי (אוֹחֵז בָּרְצוּעוֹת וְלֹא בַּבָּתִּים), וּבחֲלִיצָה לִסְעוּדַת קֶבַע. הביאורים הם שלנו. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן מ׳ — אֵיךְ לִנְהֹג בִּקְדֻשַּׁת הַתְּפִלִּין

הטקסט המלא של שולחן ערוך אדמו״ר הזקן

שולחן ערוך הרב, סימן מ׳ — אֵיךְ לִנְהֹג בִּקְדֻשַּׁת הַתְּפִלִּין — וּבוֹ י״א סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.40), ולאחריו ביאור קצר. הסימן כולל י״א סְעִיפִים.

סעיף א אָסוּר לִתְלוֹת תְּפִלִּין עַל הַיָּתֵד מִשּׁוּם דֶּרֶךְ בִּזָּיוֹן ; וּבְאַקְרַאי הָאוֹחֵז בְּיָדוֹ אֵין לְהַקְפִּיד ; וּתְפִלָּה שֶׁל רֹאשׁ בִּשְׁעַת הֲנָחָה מֻתָּר.

דִּין אֵיךְ לִנְהֹג בִּקְדֻשַּׁת הַתְּפִלִּין וּבוֹ י״א סְעִיפִים:אָסוּר לִתְלוֹת תְּפִלִּין עַל הַיָּתֵד, בֵּין בָּרְצוּעוֹת וְהַבָּתִּים תְּלוּיִים לְמַטָּה, בֵּין בַּבָּתִּים וְהָרְצוּעוֹת תְּלוּיוֹת לְמַטָּה, מִפְּנֵי שֶׁהוּא דֶּרֶךְ בִּזָּיוֹן. וּמִכָּל מָקוֹם, בְּאַקְרַאי בְּעָלְמָא שֶׁאוֹחֵז הַתְּפִלִּין בְּיָדוֹ וְהָרְצוּעוֹת תְּלוּיוֹת לְמַטָּה — אֵין לְהַקְפִּיד. אֲבָל אִם הַתְּפִלִּין תְּלוּיִים — אָסוּר אֲפִלּוּ בְּאַקְרַאי בְּעָלְמָא. וְאֵין בִּכְלַל זֶה מַה שֶּׁהַתְּפִלִּין שֶׁל רֹאשׁ תְּלוּיָה בָּאֲוִיר וּמַחֲזִיקִין בַּיָּד בָּרְצוּעוֹת וּמְשִׂימִין עַל הָרֹאשׁ, כִּי זֶהוּ צֹרֶךְ הֲנָחָתָן:

אִסּוּר תְּלִיַּת הַתְּפִלִּין. אָסוּר לִתְלוֹת תְּפִלִּין עַל הַיָּתֵד — בֵּין שֶׁהָרְצוּעוֹת תְּלוּיוֹת בֵּין שֶׁהַבָּתִּים תְּלוּיִים — מִשּׁוּם דֶּרֶךְ בִּזָּיוֹן. אַךְ בְּאַקְרַאי בְּעָלְמָא הָאוֹחֵז בְּיָדוֹ וְהָרְצוּעוֹת תְּלוּיוֹת — אֵין לְהַקְפִּיד ; וְאִם הַתְּפִלִּין עַצְמָן תְּלוּיִים אָסוּר אַף בְּאַקְרַאי. וְשֶׁל רֹאשׁ הַתְּלוּיָה בָּאֲוִיר בִּשְׁעַת הֲנָחָה — צֹרֶךְ הֲנָחָתָן, וְאֵין בָּזֶה בִּזָּיוֹן.

סעיף ב כְּשֶׁהֵן בְּכִיסָן מֻתָּר לִתְלוֹתָן ; אֲבָל סֵפֶר תּוֹרָה אָסוּר אֲפִלּוּ בְּכִיסָהּ ; וְסִדּוּרִים בְּלוּחוֹת אֲסוּרִים.

כְּשֶׁהֵן בְּכִיסָן — מֻתָּר לִתְלוֹתָן. אֲבָל סֵפֶר תּוֹרָה — אָסוּר לִתְלוֹתָהּ אֲפִלּוּ בְּכִיסָהּ. וְסִדּוּרִים הַקְּבוּעִים בְּלוּחוֹת שֶׁבָּהֶן שַׁלְשֶׁלֶת שֶׁל כֶּסֶף — אָסוּר לִתְלוֹתָן בָּהּ, שֶׁהַלּוּחוֹת אֵינָן דּוֹמוֹת לַכִּיס, לְפִי שֶׁהֵן מְחֻבָּרוֹת לַסִּדּוּר, וַהֲרֵי הֵן כַּסִּדּוּר עַצְמוֹ:

בְּכִיסָן מֻתָּר. כְּשֶׁהֵן בְּכִיסָן מֻתָּר לִתְלוֹתָן ; אֲבָל סֵפֶר תּוֹרָה אָסוּר אֲפִלּוּ בְּכִיסָהּ. וְסִדּוּרִים הַקְּבוּעִים בְּלוּחוֹת — הַלּוּחוֹת מְחֻבָּרוֹת לַסִּדּוּר וַהֲרֵי הֵן כַּסִּדּוּר עַצְמוֹ, וְאָסוּר לִתְלוֹתָן בָּהֶן.

סעיף ג בַּיִת שֶׁיֵּשׁ בּוֹ תְּפִלִּין אָסוּר לְשַׁמֵּשׁ עַד שֶׁיּוֹצִיאֵן אוֹ מְחִצָּה אוֹ כְּלִי בְּתוֹךְ כְּלִי ; וְהַמְיֻחָד לָהֶם אֵינוֹ מוֹעִיל.

בַּיִת שֶׁיֵּשׁ בּוֹ תְּפִלִּין — אָסוּר לְשַׁמֵּשׁ בּוֹ מִטָּתוֹ, עַד שֶׁיּוֹצִיאֵן, אוֹ שֶׁיַּעֲשֶׂה לִפְנֵיהֶם מְחִצָּה גָּבוֹהַּ עֲשָׂרָה טְפָחִים אַף עַל פִּי שֶׁהֵן מְגֻלִּין, אוֹ שֶׁיְּכַסֵּן שְׁנֵי כִּסּוּיִים כְּלִי בְּתוֹךְ כְּלִי. וְהוּא שֶׁאֵין הַשֵּׁנִי מְיֻחָד לָהֶם, אוֹ שֶׁהַחִיצוֹן מְיֻחָד לָהֶם וְהַפְּנִימִי אֵין מְיֻחָד לָהֶם, וְאֵין צָרִיךְ לוֹמַר אִם שְׁנֵיהֶם אֵינָן מְיֻחָדִים לָהֶם, אֲבָל אִם שְׁנֵיהֶם מְיֻחָדִים וַאֲפִלּוּ מֵאָה שֶׁהֵם מְיֻחָדִים לָהֶם — חֲשׁוּבִים כִּכְלִי אֶחָד, אַף עַל פִּי שֶׁאֵין מַנִּיחִין לְעוֹלָם הַתְּפִלִּין כְּמוֹ שֶׁהֵן בְּתוֹךְ הַחִיצוֹן, אֶלָּא מַנִּיחִין אוֹתָן בְּתוֹךְ הַפְּנִימִי וְהַפְּנִימִי בְּתוֹךְ הַחִיצוֹן. לְפִיכָךְ אוֹתָן הַמַּנִּיחִין כִּיס הַתְּפִלִּין וְהַטַּלִּית לְתוֹךְ כִּיס גָּדוֹל — הֲרֵי גַּם כִּיס הַגָּדוֹל נִקְרָא כֶּלְיָן הַמְיֻחָד לָהֶן, וְצָרִיךְ עוֹד כְּלִי עַל גַּבָּן. אַךְ אִם כִּיס הַתְּפִלִּין מְחֻבָּר לְמַעְלָה עַל כִּיס הַגָּדוֹל — הֲרֵי צַד הַשֵּׁנִי שֶׁל כִּיס הַגָּדוֹל אֵינוֹ נִקְרָא כֶּלְיָן, וּמוֹעִיל לָהֶן אִם הֲפָכוֹ:

כְּלִי בְּתוֹךְ כְּלִי וְתַשְׁמִישׁ הַמִּטָּה. בַּיִת שֶׁיֵּשׁ בּוֹ תְּפִלִּין אָסוּר לְשַׁמֵּשׁ עַד שֶׁיּוֹצִיאֵן, אוֹ מְחִצָּה עֲשָׂרָה טְפָחִים, אוֹ כְּלִי בְּתוֹךְ כְּלִי — וּבִלְבַד שֶׁאֵין הַשֵּׁנִי מְיֻחָד לָהֶם. וְהַמְיֻחָד לָהֶם, אֲפִלּוּ מֵאָה, כִּכְלִי אֶחָד. לְפִיכָךְ כִּיס הַתְּפִלִּין בְּתוֹךְ כִּיס גָּדוֹל — גַּם הַגָּדוֹל מְיֻחָד וְצָרִיךְ עוֹד כְּלִי, אֶלָּא אִם כֵּן מְחֻבָּר לְמַעְלָה שֶׁאָז צַד הַשֵּׁנִי מוֹעִיל בַּהֲפִיכָתוֹ.

סעיף ד פָּרַשׂ טַלִּית עַל גַּבֵּי כִּיס שֶׁל תְּפִלִּין הַמֻּנָּח — מֻתָּר אַף שֶׁאֵינָהּ כְּרוּכָה תַּחְתָּיו.

אִם פָּרַשׂ טַלִּית עַל גַּבֵּי כִּיס שֶׁל תְּפִלִּין שֶׁמֻּנָּח, אַף עַל פִּי שֶׁאֵינָהּ כְּרוּכָה תַּחְתָּיו מִלְּמַטָּה — מֻתָּר:

פְּרִיסַת טַלִּית. פָּרַשׂ טַלִּית עַל גַּבֵּי כִּיס הַתְּפִלִּין הַמֻּנָּח — מֻתָּר, אַף שֶׁאֵינָהּ כְּרוּכָה תַּחְתָּיו מִלְּמַטָּה.

סעיף ה מְזוּזָה קְבוּעָה בַּחֶדֶר — יֵשׁ מַצְרִיךְ כְּלִי בְּתוֹךְ כְּלִי, וְיֵשׁ דַּי בִּסְתִימַת הַנֶּקֶב שֶׁהַכֹּתֶל חוֹלֵק רְשׁוּת לְעַצְמוֹ.

אִם הַמְּזוּזָה קְבוּעָה בִּפְנִים בְּחֶדֶר — יֵשׁ מִי שֶׁמַּצְרִיךְ כְּלִי בְּתוֹךְ כְּלִי, דְּהַיְנוּ לְכַסּוֹתָהּ בִּשְׁנֵי כִּסּוּיִין, וְאִם יֵשׁ זְכוּכִית עַל הַשֵּׁם דַּי בִּפְרִיסַת סוּדָר אֶחָד עַל הַזְּכוּכִית. וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁמְּזוּזָה כֵּיוָן שֶׁקְּבוּעָה כֻּלָּהּ בַּכֹּתֶל וְאֵין בָּהּ אֶלָּא נֶקֶב בִּלְבַד — דַּי שֶׁיִּסְתֹּם אֶת הַנֶּקֶב בַּחֲתִיכַת שַׁעֲוָה אוֹ דָּבָר אַחֵר שֶׁמְּבַטְּלוֹ שָׁם, וְאֵין צָרִיךְ כְּלִי בְּתוֹךְ כְּלִי, לְפִי שֶׁהַכֹּתֶל חוֹלֵק רְשׁוּת לְעַצְמוֹ לְכָל מַה שֶּׁבְּתוֹכוֹ כְּשֶׁאֵין נֶקֶב פָּתוּחַ מִשָּׁם לַבַּיִת. וְאַף אִם הַמְּזוּזָה קְבוּעָה בִּשְׁפוֹפֶרֶת — מִכָּל מָקוֹם כֵּיוָן שֶׁהִיא מְחֻבֶּרֶת לַכֹּתֶל הֲרֵי הִיא כְּכֹתֶל. אֲבָל סְתִימַת זְכוּכִית — אֵינָהּ מוֹעֶלֶת, שֶׁכֵּיוָן שֶׁהַמְּזוּזָה נִרְאֵית דֶּרֶךְ הַזְּכוּכִית — אָסוּר, כְּמוֹ שֶׁאָסְרוּ עֶרְוָה בַּעֲשָׁשִׁית מִשּׁוּם "לֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר", וְתַשְׁמִישׁ הַמִּטָּה הִיא גַּם כֵּן גְּנַאי וְדוֹמֶה לְעֶרְוָה, וְצָרִיךְ סְתִימָה גְּמוּרָה שֶׁאֵין הַמְּזוּזָה נִרְאֵית מִתּוֹכָהּ. וְיֵשׁ לִסְמֹךְ עַל דְּבָרָיו לְהָקֵל בְּדִבְרֵי סוֹפְרִים, בְּשֶׁגַּם שֶׁיֵּשׁ לוֹמַר שֶׁאַף מִי שֶׁהִצְרִיךְ כְּלִי בְּתוֹךְ כְּלִי — לֹא הִצְרִיךְ אֶלָּא בִּכְלִי לְבַד, מִפְּנֵי שֶׁאֵינוֹ מְבַטְּלוֹ שָׁם, וְאֵינוֹ מוֹעִיל לִסְתֹּם פְּתִיחַת נֶקֶב הַכֹּתֶל לַבַּיִת שֶׁלֹּא יְהֵא כָּל מַה שֶּׁבְּתוֹכוֹ נִדּוֹן כְּאִלּוּ הוּא בְּתוֹךְ הַבַּיִת:

הַמְּזוּזָה בַּחֶדֶר. יֵשׁ מַצְרִיךְ כְּלִי בְּתוֹךְ כְּלִי לַמְּזוּזָה, וְאִם יֵשׁ זְכוּכִית דַּי בְּסוּדָר אֶחָד ; וְיֵשׁ אוֹמֵר שֶׁכֵּיוָן שֶׁקְּבוּעָה בַּכֹּתֶל דַּי בִּסְתִימַת הַנֶּקֶב, שֶׁהַכֹּתֶל חוֹלֵק רְשׁוּת לְעַצְמוֹ. אֲבָל סְתִימַת זְכוּכִית אֵינָהּ מוֹעֶלֶת, שֶׁהַנִּרְאֶה דֶּרֶךְ הַזְּכוּכִית אָסוּר (כְּעֶרְוָה בַּעֲשָׁשִׁית) ; וְיֵשׁ לִסְמֹךְ לְהָקֵל בְּדִבְרֵי סוֹפְרִים.

סעיף ו אֲפִלּוּ כְּלִי בְּתוֹךְ כְּלִי אָסוּר תַּחַת מַרְגְּלוֹתָיו וּכְנֶגֶד רֹאשׁוֹ ; שֶׁלֹּא כְּנֶגֶד רֹאשׁוֹ בְּאֵין אִשְׁתּוֹ עִמּוֹ — מֻתָּר.

אֲפִלּוּ הִנִּיחָן כְּלִי בְּתוֹךְ כְּלִי — אָסוּר לְהַנִּיחָן תַּחַת מַרְגְּלוֹתָיו, אֲפִלּוּ אֵין אִשְׁתּוֹ עִמּוֹ, מִפְּנֵי שֶׁנּוֹהֵג בָּהֶן מִנְהַג בִּזָּיוֹן. וְכֵן לְהַנִּיחָן בַּמִּטָּה כְּנֶגֶד צִדּוֹ — דִּינָם כְּתַחַת מַרְגְּלוֹתָיו. וַאֲפִלּוּ תַּחַת מְרַאֲשׁוֹתָיו — אָסוּר כְּנֶגֶד רֹאשׁוֹ אֲפִלּוּ כְּלִי בְּתוֹךְ כְּלִי אֲפִלּוּ אֵין אִשְׁתּוֹ עִמּוֹ. אֲבָל שֶׁלֹּא כְּנֶגֶד רֹאשׁוֹ, אִם אֵין אִשְׁתּוֹ עִמּוֹ — מֻתָּר אֲפִלּוּ שֶׁלֹּא בִּכְלִי כְּלָל, רַק שֶׁיַּנִּיחֵן בְּעִנְיָן שֶׁלֹּא יִתְגַּלְגְּלוּ מִשָּׁם לִצְדָדִין, כְּגוֹן שֶׁהִנִּיחָן בֵּין כַּר הַתַּחְתּוֹן לַכַּר שֶׁתַּחַת רֹאשׁוֹ, שֶׁלֹּא כְּנֶגֶד רֹאשׁוֹ. וְאִם אִשְׁתּוֹ עִמּוֹ — צָרִיךְ כְּלִי בְּתוֹךְ כְּלִי, דְּהַיְנוּ שֶׁיַּנִּיחֵם בְּתוֹךְ כְּלִי אֶחָד וְהַכַּר שֶׁתַּחַת רֹאשׁוֹ הוּא כְּלִי שֵׁנִי. וְלֹא הִתִּירוּ חֲכָמִים לְהַנִּיחַ תְּפִלִּין תַּחַת מְרַאֲשׁוֹתָיו (כְּשֶׁאִשְׁתּוֹ עִמּוֹ) אֲפִלּוּ כְּלִי בְּתוֹךְ כְּלִי, אֶלָּא כְּדֵי שֶׁיִּשְׁתַּמְּרוּ מִן הַגַּנָּבִים וְהָעַכְבָּרִים, וּלְפִיכָךְ אִם יֵשׁ לוֹ מָקוֹם אַחֵר שֶׁמִּשְׁתַּמְּרִין שָׁם לֹא יַנִּיחֵן כְּלָל בַּמִּטָּה:

מַרְגְּלוֹת, מְרַאֲשׁוֹת וְצַד. אֲפִלּוּ כְּלִי בְּתוֹךְ כְּלִי אָסוּר תַּחַת מַרְגְּלוֹתָיו (מִנְהַג בִּזָּיוֹן) וּכְנֶגֶד צִדּוֹ, וְאַף כְּנֶגֶד רֹאשׁוֹ בִּמְרַאֲשׁוֹתָיו אָסוּר. שֶׁלֹּא כְּנֶגֶד רֹאשׁוֹ, בְּאֵין אִשְׁתּוֹ עִמּוֹ — מֻתָּר אַף בְּלֹא כְּלִי ; וּבְאִשְׁתּוֹ עִמּוֹ צָרִיךְ כְּלִי בְּתוֹךְ כְּלִי, וְלֹא הִתִּירוּ אֶלָּא לְשָׁמְרָן, וְאִם יֵשׁ מָקוֹם אַחֵר לֹא יַנִּיחֵן בַּמִּטָּה.

סעיף ז מָקוֹם גָּבוֹהַּ אוֹ נָמוּךְ שְׁלֹשָׁה טְפָחִים מִמְּרַאֲשׁוֹתָיו — מֻתָּר לְהַנִּיחָן אַף שֶׁאִשְׁתּוֹ עִמּוֹ, וּבִלְבַד כְּלִי בְּתוֹךְ כְּלִי.

הָיָה יוֹצֵא מֵהַמִּטָּה מָקוֹם שֶׁהוּא גָּבוֹהַּ שְׁלֹשָׁה טְפָחִים לְמַעְלָה מְרַאֲשׁוֹתָיו אוֹ נָמוּךְ שְׁלֹשָׁה טְפָחִים לְמַטָּה מְרַאֲשׁוֹתָיו — מֻתָּר לְהַנִּיחָן עָלָיו אַף עַל פִּי שֶׁאִשְׁתּוֹ עִמּוֹ בַּמִּטָּה וּבִלְבַד שֶׁיַּנִּיחֶהָ כְּלִי בְּתוֹךְ כְּלִי:

מָקוֹם רָחוֹק מֵהַמִּטָּה. מָקוֹם גָּבוֹהַּ אוֹ נָמוּךְ שְׁלֹשָׁה טְפָחִים מִמְּרַאֲשׁוֹתָיו — מֻתָּר לְהַנִּיחָן אַף שֶׁאִשְׁתּוֹ עִמּוֹ בַּמִּטָּה, וּבִלְבַד כְּלִי בְּתוֹךְ כְּלִי.

סעיף ח אִשְׁתּוֹ עִמּוֹ וְאֵינוֹ רוֹצֶה לְשַׁמֵּשׁ — כְּאֵין אִשְׁתּוֹ עִמּוֹ ; וְיֵשׁ אוֹסֵר שֶׁמָּא יִשְׁכַּח, וְנָכוֹן לָחֹשׁ.

אִשְׁתּוֹ עִמּוֹ וְאֵינוֹ רוֹצֶה לְשַׁמֵּשׁ — נִקְרָא אֵין אִשְׁתּוֹ עִמּוֹ. וְיֵשׁ מִי שֶׁאוֹסֵר לִהְיוֹת אִשְׁתּוֹ עִמּוֹ בַּמִּטָּה עַד שֶׁיַּנִּיחַ הַתְּפִלִּין וְהַסְּפָרִים שֶׁבַּחֶדֶר כְּלִי בְּתוֹךְ כְּלִי, שֶׁמָּא יִשְׁכַּח עֲלֵיהֶן וִישַׁמֵּשׁ. (וְנָכוֹן לָחֹש לִדְבָרָיו אַף עַל פִּי שֶׁאֵין אִסּוּר בַּדָּבָר מִן הַדִּין, שֶׁהֲרֵי מִן הַדִּין מֻתָּר לִישֹׁן עִם אִשְׁתּוֹ בְּתִשְׁעָה בְּאָב וְאֵין חוֹשְׁשִׁין שֶׁמָּא יִשְׁכַּח, אֶלָּא שֶׁנָּכוֹן הַדָּבָר לְהַחֲמִיר כְּמוֹ שֶׁיִּתְבָּאֵר סִימָן תקנד):

אֵין אִשְׁתּוֹ עִמּוֹ. אִשְׁתּוֹ עִמּוֹ וְאֵינוֹ רוֹצֶה לְשַׁמֵּשׁ — כְּאֵין אִשְׁתּוֹ עִמּוֹ. וְיֵשׁ אוֹסֵר עַד שֶׁיַּנִּיחַ הַתְּפִלִּין וְהַסְּפָרִים כְּלִי בְּתוֹךְ כְּלִי שֶׁמָּא יִשְׁכַּח וִישַׁמֵּשׁ, וְנָכוֹן לָחֹשׁ לִדְבָרָיו לְהַחֲמִיר (כְּמוֹ שֶׁיִּתְבָּאֵר סִימָן תקנד).

סעיף ט שָׁכַח וְשִׁמֵּשׁ בִּתְפִלִּין — לֹא יֶאֱחֹז בַּבָּתִּים וְלֹא בָּרְצוּעוֹת עַד שֶׁיִּטֹּל יָדָיו, שֶׁהַיָּדַיִם עַסְקָנִיּוֹת.

שָׁכַח וְשִׁמֵּשׁ מִטָּתוֹ בִּתְפִלִּין — לֹא יֶאֱחֹז לֹא בַּבָּתִּים וְלֹא בָּרְצוּעוֹת עַד שֶׁיִּטֹּל יָדָיו, מִפְּנֵי שֶׁהַיָּדַיִם עַסְקָנִיּוֹת הֵן וְשֶׁמָּא נָגְעוּ בִּמְקוֹם הַטִּנֹּפֶת, וּמִיָּד שֶׁנּוֹטֵל יָדָיו — צָרִיךְ לְחָלְצָן עַד שֶׁיְּקַנַּח הַקֶּרִי אִם יֶשְׁנוֹ עָלָיו, וְאָז יָכוֹל לַחֲזֹר וּלְלָבְשָׁן, שֶׁבַּעַל קֶרִי מֻתָּר בִּתְפִלִּין לְאַחַר שֶׁקִּנַּח הַקֶּרִי, אֲבָל לֹא בְּעוֹד שֶׁהַקֶּרִי עָלָיו:

שָׁכַח וְשִׁמֵּשׁ. לֹא יֶאֱחֹז בַּבָּתִּים וְלֹא בָּרְצוּעוֹת עַד שֶׁיִּטֹּל יָדָיו, שֶׁהַיָּדַיִם עַסְקָנִיּוֹת ; וּמִיָּד שֶׁנָּטַל צָרִיךְ לְחָלְצָן עַד שֶׁיְּקַנַּח הַקֶּרִי, שֶׁבַּעַל קֶרִי מֻתָּר בִּתְפִלִּין לְאַחַר שֶׁקִּנַּח.

סעיף י שָׁכַח וְיָשַׁן וְרָאָה קֶרִי — יֶאֱחֹז בָּרְצוּעוֹת וְלֹא בַּבָּתִּים ; אֲבָל בְּשִׁמֵּשׁ אֵין מַתִּירִין אֲפִלּוּ בָּרְצוּעָה.

שָׁכַח וְיָשַׁן בָּהֶן וְרָאָה קֶרִי וְצָרִיךְ לְסַלְּקָן בְּעוֹד שֶׁהַקֶּרִי עָלָיו — לֹא יֶאֱחֹז בַּבָּתִּים, אֶלָּא יֶאֱחֹז בָּרְצוּעוֹת וּמְסִירָן עַד שֶׁיְּקַנַּח הַקֶּרִי מֵעָלָיו וְיִטֹּל יָדָיו. אֲבָל לֹא יִטֹּל יָדָיו מִיָּד שֶׁנֵּעוֹר מִשְּׁנָתוֹ, מִפְּנֵי שֶׁאֵין לָחֹש בָּהֶם כָּל־כָּךְ שֶׁמָּא נָגְעוּ בִּמְקוֹם הַטִּנֹּפֶת כְּשֶׁיָּשַׁן, שֶׁתְּהֵא נְטִילָתָן כְּדַאי לְהַשְׁהוֹת עָלָיו הַתְּפִלִּין בְּעוֹד שֶׁהַקֶּרִי עָלָיו עַד שֶׁיִּטֹּל יָדָיו. וְאַף־עַל־פִּי־כֵן אֵין מַתִּירִין לוֹ לֶאֱחֹז בַּבָּתִּים, כֵּיוָן שֶׁאֶפְשָׁר עַל יְדֵי הָרְצוּעוֹת. אֲבָל בְּשִׁמֵּשׁ מִטָּתוֹ — אֵין מַתִּירִים אֲפִלּוּ עַל יְדֵי הָרְצוּעָה, הוֹאִיל וְהַיָּדַיִם עַסְקָנִיּוֹת הֵן:

שָׁכַח וְיָשַׁן וְרָאָה קֶרִי. יֶאֱחֹז בָּרְצוּעוֹת וְלֹא בַּבָּתִּים, וְאֵין נוֹטֵל יָדָיו מִיָּד, שֶׁאֵין לָחֹש כָּל־כָּךְ בַּשֵּׁנָה. אֲבָל בְּשִׁמֵּשׁ אֵין מַתִּירִין אֲפִלּוּ עַל יְדֵי הָרְצוּעָה, שֶׁהַיָּדַיִם עַסְקָנִיּוֹת.

סעיף יא הַנִּכְנָס לִסְעוּדַת קֶבַע — חוֹלְצָן וּמַנִּיחָן עַל הַשֻּׁלְחָן ; אֲבָל אֲכִילַת עֲרַאי אֵין צָרִיךְ לְחָלְצָן.

הַנִּכְנָס לִסְעוּדַת קֶבַע — חוֹלְצָן וּמַנִּיחָן עַל הַשֻּׁלְחָן, שֶׁיְּהֵא מְזֻמָּנִים לוֹ בִּשְׁעַת גְּמַר הַסְּעוּדָה, שֶׁיַּחֲזֹר וִינִיחֵן עָלָיו קֹדֶם בִּרְכַּת הַמָּזוֹן. בַּמֶּה דְּבָרִים אֲמוּרִים? בִּסְעוּדַת קֶבַע שֶׁיֵּשׁ לָחֹש שֶׁמָּא יִשְׁתַּכֵּר וְיִתְגַּנֶּה בִּתְפִלָּיו, אֲבָל לַאֲכִילַת עֲרַאי — אֵין צָרִיךְ לְחָלְצָן כְּלָל:

חֲלִיצָה לִסְעוּדַת קֶבַע. הַנִּכְנָס לִסְעוּדַת קֶבַע חוֹלְצָן וּמַנִּיחָן עַל הַשֻּׁלְחָן, וּמַחֲזִירָן קֹדֶם בִּרְכַּת הַמָּזוֹן, מִשּׁוּם שֶׁמָּא יִשְׁתַּכֵּר וְיִתְגַּנֶּה ; אֲבָל בַּאֲכִילַת עֲרַאי אֵין צָרִיךְ לְחָלְצָן.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim מ׳ (מהדורת קה״ת, י״א סְעִיפִים). ביאור: DAAT. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.

2 · היסוד

היסוד — קְדֻשַּׁת הַתְּפִלִּין וְהַהַנְהָגָה בָּהּ

היסוד: קדושת התפילין וההנהגה בה

קריאת אדמו״ר הזקן על סימן זה בשלושה מהלכים. תחילה קְדֻשַּׁת הַתְּפִלִּין וּכְבוֹדָן: אֵין תּוֹלִין אוֹתָן (דֶּרֶךְ בִּזָּיוֹן), מַנִּיחִין בַּכִּיס, וְסֵפֶר תּוֹרָה חָמוּר יוֹתֵר. אַחַר כָּךְ גֶּדֶר הַכְּלִי בְּתוֹךְ כְּלִי וְהַמְיֻחָד, הַשּׁוֹלֵט בְּתַשְׁמִישׁ הַמִּטָּה בַּחֶדֶר שֶׁיֵּשׁ בּוֹ תְּפִלִּין אוֹ מְזוּזָה, וּמְקוֹם הַתְּפִלִּין סְמוּךְ לַמִּטָּה (מַרְגְּלוֹת, מְרַאֲשׁוֹת, הַצַּד). וּלְבַסּוֹף טָהֳרַת הַגּוּף וְהַיָּדַיִם: הַיָּדַיִם עַסְקָנִיּוֹת אַחַר הַקֶּרִי (אוֹחֵז בָּרְצוּעוֹת וְלֹא בַּבָּתִּים), וְהַחֲלִיצָה לִסְעוּדַת קֶבַע מֵחֲשַׁשׁ שִׁכְרוּת.

כְּבוֹד הַתְּפִלִּין וְאִסּוּר תְּלִיָּתָן

כבוד התפילין ואיסור תלייתן

היסוד. הַתְּפִלִּין תַּשְׁמִישֵׁי קְדֻשָּׁה ; אֵין תּוֹלִין אוֹתָן עַל הַיָּתֵד — בֵּין הָרְצוּעוֹת בֵּין הַבָּתִּים — שֶׁהוּא דֶּרֶךְ בִּזָּיוֹן. אֲבָל בְּכִיסָן מֻתָּר לִתְלוֹתָן.

התוצאה. סֵפֶר תּוֹרָה חָמוּר יוֹתֵר (אָסוּר אֲפִלּוּ בְּכִיסָהּ) ; וְשֶׁל רֹאשׁ הַנֶּאֱחָז בָּרְצוּעוֹת לְשֵׁם הֲנָחָה אֵינוֹ בִּכְלַל הָאִסּוּר, שֶׁזֶּהוּ צֹרֶךְ הֲנָחָה. וּלְתַשְׁמִישֵׁי קְדֻשָּׁה עַיֵּן סימן מ״ב · 42 וְסימן מ״ד · 44.

כְּלִי בְּתוֹךְ כְּלִי וְהַמְיֻחָד לָהֶם

כלי בתוך כלי והמיוחד

העיקרון. לְשַׁמֵּשׁ בַּחֶדֶר שֶׁיֵּשׁ בּוֹ תְּפִלִּין — צָרִיךְ לְהוֹצִיאָן, אוֹ מְחִצָּה עֲשָׂרָה טְפָחִים, אוֹ כְּלִי בְּתוֹךְ כְּלִי — וּבִלְבַד שֶׁאֵין הַשֵּׁנִי מְיֻחָד לָהֶם ; שֶׁהַמְיֻחָד (אֲפִלּוּ מֵאָה) כִּכְלִי אֶחָד.

ההרחבה. אוֹתוֹ גֶּדֶר בַּמְּזוּזָה הַקְּבוּעָה בַּחֶדֶר (הַהַפְנָיָה לְהִלְכוֹת מְזוּזָה, יורה דעה): לְדֵעָה אַחַת כְּלִי בְּתוֹךְ כְּלִי, וּלְדֵעָה אַחֶרֶת דַּי בִּסְתִימַת הַנֶּקֶב, שֶׁהַכֹּתֶל חוֹלֵק רְשׁוּת לְעַצְמוֹ.

הַיָּדַיִם עַסְקָנִיּוֹת וּסְעוּדַת קֶבַע

הידים עסקניות וסעודת קבע

העיקרון. אַחַר קֶרִי (בְּשִׁמֵּשׁ אוֹ בַּשֵּׁנָה) אֵין אוֹחֵז בַּבָּתִּים: הַיָּדַיִם עַסְקָנִיּוֹת וְשֶׁמָּא נָגְעוּ בִּמְקוֹם הַטִּנֹּפֶת. בְּשִׁמֵּשׁ מַמְתִּין לִנְטִילַת יָדַיִם ; בַּשֵּׁנָה אוֹחֵז בָּרְצוּעוֹת וּמְסִירָן עַד שֶׁיְּקַנַּח הַקֶּרִי.

ההנהגה. לִסְעוּדַת קֶבַע חוֹלֵץ וּמַנִּיחָן עַל הַשֻּׁלְחָן עַד בִּרְכַּת הַמָּזוֹן, מֵחֲשַׁשׁ שִׁכְרוּת שֶׁיִּתְגַּנֶּה בָּהֶן ; וּבַאֲכִילַת עֲרַאי אֵין צָרִיךְ.

הפניה ללימוד. משקל היסוד שהונח כאן — הכָּבוֹד לתַשְׁמִישֵׁי קְדֻשָּׁה, הכְּלִי בְּתוֹךְ כְּלִי והגּוּף נָקִי — נוגע גם בהנהגה (סדר הנחת התפילין, חדר המיטה, ניקיון הידיים) וגם ברגישות חב״ד על הקְדֻשָּׁה שסביב המצווה. דף זה מציג זאת ברמת העיקרון המתועד בטקסט של הרב, בלי לצטט פרק או נוסח מדויק, ובלי להוסיף דבר מומצא במשמעות הרוחנית. ללימוד מעמיק של ממד זה — ולהנהגה אישית בכלי בתוך כלי, במזוזה ובקרי — יש ללמוד עם רב של חב״ד ובתוך המקורות עצמם.

אֵיךְ לִנְהֹג בִּקְדֻשַּׁת הַתְּפִלִּין — בעיקרון · ללימוד, פנה לרב של חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרכו של אדמו״ר הזקן עם המחבר, הרמ״א (רמ״א) והמשנה ברורה (משנה ברורה) בנקודות המפתח של סימן מ׳. שורת שולחן ערוך הרב (שולחן ערוך הרב) מובלטת — היא חושפת את רגישות הרב כפי שהיא עולה מטקסטו (§1, י״א סעיפים). הרב מפרש את הטעם ואת הפרט במקום שהמחבר מסתפק באמירה.

נושא מחבר רמ״א משנה ברורה דרך אדמו״ר הזקן (שולחן ערוך הרב)
תליית תפילין וכבודן או״ח מ׳:א-ב — אָסוּר לִתְלוֹת תְּפִלִּין (דֶּרֶךְ בִּזָּיוֹן) ; בְּכִיסָן מֻתָּר ; סֵפֶר תּוֹרָה אָסוּר אֲפִלּוּ בְּכִיסָהּ. — מ״ב מ׳ / ברכות כ״ד ע״א — טַעַם הַבִּזָּיוֹן ; דִּין הַכִּיס וְהַשֶּׁל רֹאשׁ בִּשְׁעַת הֲנָחָה. דרך אדמו״ר הזקן — אָסוּר לִתְלוֹת בֵּין בָּרְצוּעוֹת בֵּין בַּבָּתִּים ; בְּאַקְרַאי הָאוֹחֵז בְּיָדוֹ אֵין לְהַקְפִּיד ; וְשֶׁל רֹאשׁ בִּשְׁעַת הֲנָחָה — צֹרֶךְ הֲנָחָה (סעיפים א-ב).
כלי בתוך כלי ותשמיש המטה או״ח מ׳:ב-ג — עַד שֶׁיּוֹצִיאֵן אוֹ מְחִצָּה אוֹ כְּלִי בְּתוֹךְ כְּלִי ; הַמְיֻחָד לָהֶם כִּכְלִי אֶחָד. — מ״ב מ׳ / ברכות כ״ג ע״ב — גֶּדֶר הַמְיֻחָד ; דִּין הַמְּזוּזָה בַּחֶדֶר (יו״ד סי׳ ר״מ). דרך אדמו״ר הזקן — שְׁנֵי כִּסּוּיִים שֶׁאֵין הַשֵּׁנִי מְיֻחָד ; כִּיס גָּדוֹל מְיֻחָד וְצָרִיךְ עוֹד כְּלִי ; וְהַמְּזוּזָה — יֵשׁ מַצְרִיךְ וְיֵשׁ דַּי בִּסְתִימַת הַנֶּקֶב (סעיפים ג-ה, ח).
מרגלות, מראשות והידים עסקניות או״ח מ׳:ג-ז — אָסוּר תַּחַת מַרְגְּלוֹתָיו וּכְנֶגֶד רֹאשׁוֹ ; שָׁכַח וְשִׁמֵּשׁ — לֹא יֶאֱחֹז בַּבָּתִּים עַד נְטִילַת יָדַיִם. — מ״ב מ׳ / ברכות כ״ג ע״ב-כ״ד ע״א — כְּנֶגֶד רֹאשׁוֹ ; קֶרִי וַאֲחִיזַת הָרְצוּעוֹת. דרך אדמו״ר הזקן — שֶׁלֹּא כְּנֶגֶד רֹאשׁוֹ בְּאֵין אִשְׁתּוֹ עִמּוֹ מֻתָּר אַף בְּלֹא כְּלִי ; בְּקֶרִי יֶאֱחֹז בָּרְצוּעוֹת, וּבְשִׁמֵּשׁ אֲפִלּוּ בָּרְצוּעָה אָסוּר (סעיפים ו-י).
חליצה לסעודת קבע או״ח מ׳:ח — הַנִּכְנָס לִסְעוּדַת קֶבַע חוֹלְצָן וּמַנִּיחָן עַל הַשֻּׁלְחָן. — מ״ב מ׳ / מנחות ל״ו ע״א — חֲשַׁשׁ שִׁכְרוּת ; קֶבַע וַעֲרַאי. דרך אדמו״ר הזקן — חוֹלְצָן עַד גְּמַר הַסְּעוּדָה וְחוֹזֵר קֹדֶם בִּרְכַּת הַמָּזוֹן ; מִשּׁוּם שֶׁמָּא יִשְׁתַּכֵּר וְיִתְגַּנֶּה ; וַעֲרַאי אֵין צָרִיךְ (סעיף יא).

הטבלה נערכה מטקסט המחבר (אורח חיים מ׳, ח׳ סעיפים, מקור ספריא) ומנושאי הכלים הניתנים לאימות (בית יוסף, טור, רמב״ם הלכות תפילין ד, רא״ש, אורחות חיים, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ג ע״ב–כ״ד ע״א, מנחות ל״ו ע״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (י״א סְעִיפִים) המובא ב§1. ייחוסי סעיף קטן שאינם ודאיים נותרו « על דרך הסברה ».

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה — מעבר ליישום הפשוט של הכלל. כל אחת בנויה: עיקרון הסימן → קריאת הרב → תוצאה מעשית. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — כְּבוֹד הַתְּפִלִּין וְאִסּוּר בִּזָּיוֹן

כבוד התפילין ואיסור הביזיון

עיקרון הסימן: אִסּוּר תְּלִיָּה וְעִנְיַן הַכִּיס (סעיפים א-ב).

קריאת הרב: הרב מבחין בין הדֶּרֶךְ בִּזָּיוֹן (אָסוּר) לָאַקְרַאי (מֻתָּר) ; סֵפֶר תּוֹרָה חָמוּר יוֹתֵר, אֲבָל הֲנָחַת הַשֶּׁל רֹאשׁ אֵינָהּ בִּזָּיוֹן.

תוצאה מעשית: לְהַנִּיחַ אֶת הַתְּפִלִּין בַּכִּיס ; לֹא לִתְלוֹתָן וְלֹא לִנְהֹג בָּהֶן קַלּוּת רֹאשׁ.

קו ב — כְּלִי בְּתוֹךְ כְּלִי וְהַמְיֻחָד

כלי בתוך כלי והמיוחד

עיקרון הסימן: תַּשְׁמִישׁ הַמִּטָּה וּמְקוֹם הַתְּפִלִּין סְמוּךְ לַמִּטָּה (סעיפים ג-ח).

קריאת הרב: הַקְּרִיטֶרְיוֹן הוּא הַמְיֻחָד — כְּלִי הַמְיֻחָד לַתְּפִלִּין בָּטֵל לָהֶם וְאֵינוֹ אֶלָּא כְּאֶחָד ; וְהַמְּזוּזָה וְהַכֹּתֶל בְּאוֹתוֹ גֶּדֶר (הַהַפְנָיָה לְהִלְכוֹת מְזוּזָה, יורה דעה).

תוצאה מעשית: לִישֹׁן אוֹ לְשַׁמֵּשׁ — כְּלִי בְּתוֹךְ כְּלִי בְּכֶלִי חִיצוֹן שֶׁאֵינוֹ מְיֻחָד, אוֹ מָקוֹם אַחֵר מִשְׁתַּמֵּר חוּץ לַמִּטָּה.

קו ג — גּוּף נָקִי, הַיָּדַיִם עַסְקָנִיּוֹת וּסְעוּדַת קֶבַע

גוף נקי, הידים עסקניות וסעודת קבע

עיקרון הסימן: הַקֶּרִי, נְטִילַת יָדַיִם וְהַחֲלִיצָה (סעיפים ט-יא).

קריאת הרב: אַחַר שִׁמֵּשׁ אֵין נוֹגְעִים לֹא בַּבָּתִּים וְלֹא בָּרְצוּעוֹת עַד נְטִילַת יָדַיִם ; אַחַר שֵׁנָה אוֹחֵז בָּרְצוּעוֹת ; לִסְעוּדַת קֶבַע חוֹלֵץ מֵחֲשַׁשׁ שִׁכְרוּת.

תוצאה מעשית: לִטֹּל יָדַיִם קֹדֶם שֶׁיֹּאחַז בַּבָּתִּים בַּבֹּקֶר ; לַחֲלֹץ תְּפִלִּין לִסְעוּדַת קֶבַע וּלְהַחֲזִירָן קֹדֶם בִּרְכַּת הַמָּזוֹן ; וְלִסְמֹךְ עַל הָרַב לְמַעֲשֶׂה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן מ׳ ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, בהפניה לרב לפרטי המקרים (סדר הנחת התפילין, חדר המיטה, הקרי, סעודת הקבע).

לחסיד חב״ד — סימן מ׳ (אֵיךְ לִנְהֹג בִּקְדֻשַּׁת הַתְּפִלִּין)

  • ① כבוד התפילין. לֹא לִתְלוֹת תְּפִלִּין (דֶּרֶךְ בִּזָּיוֹן) ; לְהַנִּיחָן בַּכִּיס. סֵפֶר תּוֹרָה חָמוּר יוֹתֵר (אָסוּר אֲפִלּוּ בְּכִיסָהּ). יסוד: שוע״ר מ׳ סעיף א-ב.
  • ② חדר המיטה. לִישֹׁן אוֹ לְשַׁמֵּשׁ בַּמָּקוֹם שֶׁיֵּשׁ בּוֹ תְּפִלִּין: לְהוֹצִיאָן, מְחִצָּה עֲשָׂרָה טְפָחִים, אוֹ כְּלִי בְּתוֹךְ כְּלִי (הַחִיצוֹן אֵינוֹ מְיֻחָד). וְכֵן בַּמְּזוּזָה (יורה דעה). יסוד: שוע״ר מ׳ סעיף ג-ה, ח.
  • ③ סמוך למיטה. אָסוּר תַּחַת מַרְגְּלוֹתָיו, כְּנֶגֶד צִדּוֹ, וּכְנֶגֶד רֹאשׁוֹ בִּמְרַאֲשׁוֹתָיו ; שֶׁלֹּא כְּנֶגֶד רֹאשׁוֹ בְּאֵין אִשְׁתּוֹ עִמּוֹ מֻתָּר אַף בְּלֹא כְּלִי ; מָקוֹם ג׳ טְפָחִים מֵהַמִּטָּה, בְּכֶלִי בְּתוֹךְ כְּלִי. יסוד: שוע״ר מ׳ סעיף ו-ז.
  • ④ הידיים בבוקר. אַחַר שֵׁנָה אוֹ קֶרִי, לֶאֱחֹז בָּרְצוּעוֹת וְלֹא בַּבָּתִּים ; לִטֹּל יָדַיִם (הַיָּדַיִם עַסְקָנִיּוֹת) ; וּבְשִׁמֵּשׁ אֲפִלּוּ בָּרְצוּעָה אָסוּר עַד נְטִילַת יָדַיִם. יסוד: שוע״ר מ׳ סעיף ט-י.
  • ⑤ סעודת קבע. לַחֲלֹץ תְּפִלִּין לִסְעוּדַת קֶבַע (חֲשַׁשׁ שִׁכְרוּת) וּלְהַנִּיחָן עַל הַשֻּׁלְחָן עַד בִּרְכַּת הַמָּזוֹן ; וּבַאֲכִילַת עֲרַאי אֵין צָרִיךְ. יסוד: שוע״ר מ׳ סעיף יא.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן מ׳ (הטקסט האמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — כלי בתוך כלי, חדר המיטה, הקרי, סעודת הקבע — יש להתייעץ עם רב, ולחב״ד עם רב של חב״ד.
