דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן מ״ה · דִּין תְפִלִּין בְבֵית הַקְּבָרות וּבְבֵית הַמֶּרְחָץ

דרכו של אדמו״ר הזקן בדין תפילין בבית הקברות ובבית המרחץ: שֶׁלֹּא לִכָּנֵס בְד׳ אַמּוֹת שֶּל מֵת בִתְפִלִּין מְגֻלּות (לֹעֵג לָרָשׁ) ; מַה שֶּׁמַּתִּיר הַכִּסּוּי — לְרַבּוֹת הָרְצוּעוֹת ; וּמִדְרַג הַבֵית הַמֶּרְחָץ (חִיצוֹן / אֶמְצָעִי / פְנִימִי) מוּל הַבֵית הַטְבִילָה — ברגישות חב״ד.

למה רמה 4 מיוחדת לאדמו״ר הזקן? השולחן ערוך של אדמו״ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתבו אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. דף זה מאסף את שיטת הרב על סימן מ״ה — דִּין תְפִלִּין בְבֵית הַקְּבָרות וּבְבֵית הַמֶּרְחָץ. הסימן כולל אצל הרב שלושה סעיפים (ג׳ סְעִיפִים), שבהם פורש אדמו״ר הזקן: (1) אִסּוּר לִכָּנֵס בְד׳ אַמּוֹת שֶּל מֵת אוֹ קֶבֶר בִתְפִלִּין מְגֻלּות — מִשּׁוּם לֹעֵג לָרָשׁ — וְחֻמְרַת בֵית הַקְּבָרות הַמֻּקָּף מְחִצָּה ; (2) שֶׁכָּל זֶה כְשֶׁהֵן מְגֻלּות — אֲבָל מְכֻסּות מֻתָּר, וּבִלְבַד שֶׁגַּם הָרְצוּעוֹת מְכֻסּות ; (3) הַהַנְהָגָה בְבֵית הַמֶּרְחָץ לְפִי הַחֶדֶר (חִיצוֹן, אֶמְצָעִי, פְנִימִי) וְהַהֶבְדֵּל מִבֵית הַטְבִילָה בְצוֹנֵן.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בדף זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן מ״ה — שלושה סעיפים (ג׳ סְעִיפִים) מידי הרב — לפי מהדורת קה״ת כפי שנסרקה בספריא. בשלושת הסעיפים דן אדמו״ר הזקן: באיסור לִכָּנֵס בְד׳ אַמּוֹת שֶּל מֵת אוֹ קֶבֶר וּתְפִלִּין מְגֻלּות בְרֹאשׁוֹ (מִשּׁוּם לֹעֵג לָרָשׁ) וּבְבֵית הַקְּבָרות הַמֻּקָּף מְחִצָּה ; בְמַה שֶּׁמַּתִּיר הַכִּסּוּי — לְרַבּוֹת הָרְצוּעוֹת ; וּבַהַנְהָגָה בְבֵית הַמֶּרְחָץ (חִיצוֹן / אֶמְצָעִי / פְנִימִי) וּבְבֵית הַטְבִילָה. הביאורים הם שלנו. קישור ספריא של כל סעיף מופיע למטה, סעיף אחר סעיף. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן מ״ה — דִּין תְפִלִּין בְבֵית הַקְּבָרות וּבְבֵית הַמֶּרְחָץ

הטקסט המלא של שולחן ערוך אדמו״ר הזקן

שולחן ערוך הרב, סימן מ״ה — דִּין תְפִלִּין בְבֵית הַקְּבָרות וּבְבֵית הַמֶּרְחָץ — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא, מהדורת קה״ת), ולאחריו ביאור קצר. הסימן כולל שלושה סעיפים (ג׳ סְעִיפִים).

סעיף א אָסוּר לִכָּנֵס בְד׳ אַמּוֹת שֶּל מֵת אוֹ קֶבֶר וּתְפִלִּין בְרֹאשׁוֹ (לֹעֵג לָרָשׁ); בֵית הַקְּבָרות מֻקָּף מְחִצָּה — לְהַחְמִיר.

דִּין תְּפִלִּין בְּבֵית הַקְּבָרוֹת וּבְבֵית הַמֶּרְחָץ וּבוֹ ג׳ סְעִיפִים:אָסוּר לִכָּנֵס בְּתוֹךְ אַרְבַּע אַמּוֹת שֶׁל מֵת אוֹ שֶׁל קֶבֶר וּתְפִלִּין בְּרֹאשׁוֹ, מִשּׁוּם "לֹעֵג לָרָשׁ". וּבֵית הַקְּבָרוֹת שֶׁהוּא מֻקָּף מְחִצָּה — יֵשׁ לְהַחֲמִיר שֶׁלֹּא לִכָּנֵס בְּתוֹכוֹ כְּלָל בִּתְפִלִּין שֶׁבְּרֹאשׁוֹ אֲפִלּוּ רָחוֹק הַרְבֵּה מִן הַקְּבָרוֹת, שֶׁמָּא יִתְקָרֵב בְּתוֹךְ אַרְבַּע אַמּוֹת שֶׁל אֵיזֶה קֶבֶר בְּלִי דַּעַת. אֲבָל חוּץ לַמְּחִצָּה — מֻתָּר אֲפִלּוּ בְּסָמוּךְ לָהּ אַף עַל פִּי שֶׁהוּא תּוֹךְ אַרְבַּע אַמּוֹת לְהַקְּבָרוֹת, מִפְּנֵי שֶׁהַמְחִצָּה מַפְסֶקֶת וְאֵינוֹ נִרְאֶה כְּ"לֹעֵג לָרָשׁ":

בְבֵית הַקְּבָרות וּבְד׳ אַמּוֹת שֶּל מֵת. אָסוּר לִכָּנֵס בְתוֹךְ אַרְבַּע אַמּוֹת שֶּל מֵת אוֹ שֶּל קֶבֶר וּתְפִלִּין בְרֹאשׁוֹ, מִשּׁוּם לֹעֵג לָרָשׁ (שֶׁהַמֵּת אֵינוֹ יָכוֹל עוֹד לְקַיֵּם מִצְווֹת). וּבֵית הַקְּבָרות הַמֻּקָּף מְחִצָּה — יֵשׁ לְהַחְמִיר שֶׁלֹּא לִכָּנֵס בְתוֹכוֹ כְלָל בִתְפִלִּין שֶּבְרֹאשׁוֹ, אֲפִלּוּ רָחוֹק הַרְבֵּה מִן הַקְּבָרות, שֶׁמָּא יִתְקָרֵב בְלִי דַעַת בְתוֹךְ ד׳ אַמּוֹת שֶּל קֶבֶר. אֲבָל חוּץ לַמְחִצָּה — מֻתָּר אֲפִלּוּ בְסָמוּךְ לָהּ, מִפְּנֵי שֶׁהַמְחִצָּה מַפְסֶקֶת וְאֵינוֹ נִרְאֶה כְלֹעֵג לָרָשׁ.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim מ״ה (סעיף א)

סעיף ב בַמֶּה דְבָרִים אֲמוּרִים? כְשֶׁהֵן מְגֻלִּין; אֲבָל מְכֻסִּין — מֻתָּר, וְצָרִיךְ שֶׁגַּם הָרְצוּעוֹת מְכֻסּוֹת.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁהֵן מְגֻלִּין, אֲבָל אִם הֵן מְכֻסִּין — אֵין כָּאן "לֹעֵג לָרָשׁ", וּמֻתָּר בְּכָל עִנְיָן. וְצָרִיךְ שֶׁגַּם הָרְצוּעוֹת תִּהְיֶינָה מְכֻסּוֹת. לְפִיכָךְ אַף עַל פִּי שֶׁמֻּתָּר לִכָּנֵס בִּתְפִלִּין שֶׁל יָד לְבַדָּהּ הוֹאִיל וְהִיא מְכֻסָּה — צָרִיךְ לִזָּהֵר בָּרְצוּעָה שֶׁעַל אֶצְבָּעוֹ שֶׁתִּהְיֶה גַּם כֵּן מְכֻסָּה:

מְגֻלִּין אוֹ מְכֻסִּין. בַמֶּה דְבָרִים אֲמוּרִים? כְשֶׁהֵן מְגֻלִּין ; אֲבָל אִם הֵן מְכֻסִּין — אֵין כָּאן לֹעֵג לָרָשׁ, וּמֻתָּר בְכָל עִנְיָן. וְצָרִיךְ שֶׁגַּם הָרְצוּעוֹת תִּהְיֶינָה מְכֻסּוֹת. לְפִיכָךְ, אַף עַל פִּי שֶׁמֻּתָּר לִכָּנֵס בִתְפִלִּין שֶּל יָד לְבַדָּהּ הוֹאִיל וְהִיא מְכֻסָּה — צָרִיךְ לִזָּהֵר שֶׁגַּם הָרְצוּעָה שֶּעַל אֶצְבָּעוֹ תִּהְיֶה מְכֻסָּה.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim מ״ה (סעיף ב)

סעיף ג בֵית הַמֶּרְחָץ — חִיצוֹן מֻתָּר, אֶמְצָעִי וּפְנִימִי אָסוּר; בֵית הַטְבִילָה בְצוֹנֵן — מֻתָּר כְשֶׁאֵין שָׁם עָרֹם.

בְּבֵית הַמֶּרְחָץ, בַּבַּיִת הַחִיצוֹן שֶׁכָּל הָעוֹמְדִים בּוֹ הֵן לְבוּשִׁים — מֻתָּר לְהָנִיחַ שָׁם תְּפִלִּין עַל רֹאשׁוֹ וּזְרוֹעוֹ לְכַתְּחִלָּה. וּבַבַּיִת הָאֶמְצָעִי שֶׁמִּקְצָת בְּנֵי אָדָם עוֹמְדִים שָׁם לְבוּשִׁים וּמִקְצָתָן עֲרֻמִּים — אָסוּר לְהָנִיחַ שָׁם תְּפִלִּין לְכַתְּחִלָּה אֲפִלּוּ בִּשְׁעַת שֶׁאֵין שָׁם אָדָם עָרֹם. אֲבָל אִם הָיוּ בְּרֹאשׁוֹ — אֵין צָרִיךְ לְחָלְצָן כְּשֶׁנִּכְנָס לְשָׁם. וּבַבַּיִת הַפְּנִימִי שֶׁכָּל הָעוֹמְדִים בּוֹ הֵם עֲרֻמִּים, אֲפִלּוּ הָיוּ בְּרֹאשׁוֹ — צָרִיךְ לְחָלְצָן כְּשֶׁנִּכְנָס לְשָׁם, אֲפִלּוּ בְּשָׁעָה שֶׁאֵין שָׁם אָדָם. בַּמֶּה דְּבָרִים אֲמוּרִים? בַּמֶּרְחָץ שֶׁזֻּהֲמָתוֹ רַבָּה מֵהֶבֶל הַחַמִּין שֶׁמִּשְׁתַּמְּשִׁין בּוֹ, אֲבָל בְּבֵית הַטְּבִילָה אַף עַל פִּי שֶׁעוֹמְדִים בּוֹ עֲרֻמִּים, כֵּיוָן שֶׁתַּשְׁמִישׁוֹ בְּצוֹנֵן — אֵין בּוֹ זֻהֲמָא וּמֻתָּר לִכָּנֵס בּוֹ בִּתְפִלִּין, כְּשֶׁאֵין שָׁם אָדָם עָרֹם. לָכֵן מְבָרְכִין בִּרְכַּת הַטְּבִילָה בַּמִּקְוֶה אֲפִלּוּ קֹדֶם שֶׁיָּרַד לַמַּיִם, אַף עַל פִּי שֶׁלְּמַעְלָה מֵהַמַּיִם עוֹמְדִים שָׁם עֲרֻמִּים. אֲבָל אִם יֵשׁ שָׁם עָרֹם — אָסוּר לִכָּנֵס לְתוֹכוֹ בִּתְפִלִּין אוֹ בִּשְׁאָר כִּתְבֵי הַקֹּדֶשׁ, שֶׁאָסוּר לַעֲמֹד עָרֹם בִּפְנֵי כִּתְבֵי הַקֹּדֶשׁ (וְלָכֵן צָרִיךְ לִזָּהֵר לְכַסּוֹת הַקְּטַנִּים שֶׁלֹּא יִהְיוּ עֲרֻמִּים בִּפְנֵי הַסְּפָרִים, עַיֵּן סוֹף סִימָן ערה).(סְלִיק הִלְכוֹת תְּפִלִּין)

בְבֵית הַמֶּרְחָץ וּבְבֵית הַטְבִילָה. בַבַּיִת הַחִיצוֹן שֶׁכָּל הָעוֹמְדִים בּוֹ לְבוּשִׁים — מֻתָּר לְהָנִיחַ תְּפִלִּין עַל רֹאשׁוֹ וּזְרוֹעוֹ לְכַתְּחִלָּה. בַבַּיִת הָאֶמְצָעִי שֶׁמִּקְצָתָם לְבוּשִׁים וּמִקְצָתָם עֲרֻמִּים — אָסוּר לְהָנִיחַ לְכַתְּחִלָּה, אֲפִלּוּ בְשָׁעָה שֶׁאֵין שָׁם עָרֹם ; אֲבָל אִם הָיוּ בְרֹאשׁוֹ אֵין צָרִיךְ לְחָלְצָן. בַבַּיִת הַפְּנִימִי שֶׁכֻּלָּם עֲרֻמִּים — אֲפִלּוּ הָיוּ בְרֹאשׁוֹ צָרִיךְ לְחָלְצָן, אֲפִלּוּ בְשָׁעָה שֶׁאֵין שָׁם אָדָם. בַמֶּה דְבָרִים אֲמוּרִים? בַמֶּרְחָץ שֶׁזֻּהֲמָתוֹ רַבָּה ; אֲבָל בֵית הַטְבִילָה שֶׁתַּשְׁמִישׁוֹ בְצוֹנֵן — אֵין בּוֹ זֻהֲמָא וּמֻתָּר לִכָּנֵס בּוֹ בִתְפִלִּין כְשֶׁאֵין שָׁם עָרֹם ; לָכֵן מְבָרְכִין בִּרְכַּת הַטְבִילָה בַמִּקְוֶה אֲפִלּוּ קֹדֶם שֶׁיָּרַד לַמַּיִם. אֲבָל אִם יֵשׁ שָׁם עָרֹם — אָסוּר לִכָּנֵס בִתְפִלִּין אוֹ בִשְׁאָר כִתְבֵי הַקֹּדֶשׁ, שֶׁאָסוּר לַעֲמֹד עָרֹם בִפְנֵיהֶם (וְלָכֵן צָרִיךְ לְכַסּוֹת אֶת הַקְּטַנִּים).

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim מ״ה (סעיף ג)

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim מ״ה (מהדורת קה״ת, ג׳ סעיפים). קישור כל סעיף מופיע למעלה. ביאור: DAAT. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.

2 · היסוד

היסוד — לֹעֵג לָרָשׁ, כִּסּוּי וּמְקומות שֶּל גְנַאי

היסוד: לֹעֵג לָרָשׁ, כִּסּוּי וּמְקומות שֶּל גְנַאי

קריאת אדמו״ר הזקן על סימן זה בשלושה מהלכים. תחילה לֹעֵג לָרָשׁ (ברכות י״ח ע״א) : הַמֵּת, שֶׁמִּצְווֹתָיו בָּטְלוּ (מִצְווֹת בְטֵלות לַמֵתִים), לֹא יְהֵא נִלְעָג בְמִצְוָה הַנַּעֲשֵׂית גְלוּיָה לְיָדוֹ — וּמִכָּאן אִסּוּר ד׳ אַמּוֹת וְחֻמְרַת בֵית הַקְּבָרות הַמֻּקָּף מְחִצָּה. אַחַר כָּךְ כִּסּוּי : מַה שֶּׁ«מְכֻסִּין» מַתִּיר — שֶׁאֵין לֹעֵג לָרָשׁ אֶלָּא בִּמְגֻלֶּה — לְרַבּוֹת הָרְצוּעוֹת. וּלְבַסּוֹף מְקומות שֶּל גְנַאי וְהָערווה : מִדְרַג בֵית הַמֶּרְחָץ (חִיצוֹן / אֶמְצָעִי / פְנִימִי) וְהֶבְדֵּלוֹ מִבֵית הַטְבִילָה בְצוֹנֵן.

לֹעֵג לָרָשׁ וְד׳ אַמּוֹת שֶּל מֵת

לֹעֵג לָרָשׁ וד׳ אמות

היסוד. מִצְווֹת בְטֵלות לַמֵתִים ; לְקַיֵּם מִצְוָה גְלוּיָה — לָשֵׂאת תְּפִלִּין מְגֻלּות — בְתוֹךְ ד׳ אַמּוֹת שֶּל מֵת אוֹ קֶבֶר הוּא לֹעֵג לָרָשׁ כְלַפֵּי מִי שֶׁאֵינוֹ יָכוֹל עוֹד לְקַיְּמָהּ (ברכות י״ח ע״א).

התוצאה. בֵית הַקְּבָרות הַמֻּקָּף מְחִצָּה — יֵשׁ לְהַחְמִיר שֶׁלֹּא לִכָּנֵס בְתוֹכוֹ כְלָל בִתְפִלִּין שֶּבְרֹאשׁוֹ, שֶׁמָּא יִכָּנֵס בְלִי דַעַת בְד׳ אַמּוֹת ; אֲבָל חוּץ לַמְחִצָּה הִיא מַפְסֶקֶת. לְעִנְיַן הִלְכוֹת אֲבֵלוּת וּכְבוֹד הַמֵּת עַיֵּן יורה דעה (הלכות אבלות) וְסימן 43.

כִּסּוּי הַתְּפִלִּין וְהָרְצוּעוֹת

מה שהכיסוי מתיר

העיקרון. אֵין לֹעֵג לָרָשׁ אֶלָּא כְשֶׁהַתְּפִלִּין מְגֻלּות ; אִם הֵן מְכֻסּות, בָּטֵל הַגְנַאי וּמֻתָּר בְכָל עִנְיָן.

ההרחבה. הַכִּסּוּי צָרִיךְ לִהְיוֹת שָׁלֵם : אַף הָרְצוּעוֹת צְרִיכוֹת כִּסּוּי, לְרַבּוֹת רְצוּעַת הַשֶּׁל יָד שֶּעַל הָאֶצְבַּע. לְעִנְיַן כְבוֹד תַּשְׁמִישֵׁי קְדֻשָּׁה עַיֵּן סימן 44.

מְקומות שֶּל גְנַאי — בֵית הַמֶּרְחָץ

מקומות של גנאי — בית המרחץ

העיקרון. בֵית הַמֶּרְחָץ מִתְדַּרֵג לְפִי מַצַּב הָעוֹמְדִים : חִיצוֹן (כֻּלָּם לְבוּשִׁים) — מֻתָּר לְכַתְּחִלָּה ; אֶמְצָעִי (לְבוּשִׁים וַעֲרֻמִּים) — אָסוּר לְכַתְּחִלָּה, וְאֵין חוֹלְצִין מַה שֶּׁכְּבָר מֻנָּח ; פְּנִימִי (כֻּלָּם עֲרֻמִּים) — חוֹלְצִין אַף מַה שֶּׁמֻּנָּח.

ההבחנה. כָּל זֶה בַמֶּרְחָץ שֶׁזֻּהֲמָתוֹ רַבָּה (מֵהֶבֶל הַחַמִּין) ; אֲבָל בֵית הַטְבִילָה בְצוֹנֵן אֵין בּוֹ זֻהֲמָא — מֻתָּר לִכָּנֵס בּוֹ בִתְפִלִּין כְשֶׁאֵין שָׁם עָרֹם, וּמִכָּאן בִּרְכַּת הַטְבִילָה בַמִּקְוֶה. וְנִשְׁאָר אִסּוּר עֲמִידָה עָרֹם בִפְנֵי כִתְבֵי הַקֹּדֶשׁ (ערווה מול קְדֻשָּׁה).

הפניה ללימוד. משקל היסוד שהונח כאן — לֹעֵג לָרָשׁ וד׳ אמות, מה שמתיר הכִּסּוּי, מדרג בֵית הַמֶּרְחָץ והערווה מול הקְדֻשָּׁה — נוגע גם בהנהגה (בבית הקברות, במרחץ, במקווה) וגם ברגישות חב״ד על כבוד תַּשְׁמִישֵׁי קְדֻשָּׁה וכבוד המת. דף זה מציג זאת ברמת העיקרון המתועד בטקסט של הרב, בלי להוסיף דבר מומצא. ללימוד מעמיק ולהנהגה אישית — בית הקברות, הכיסוי, בית המרחץ — יש ללמוד עם רב של חב״ד ובתוך המקורות עצמם.

דִּין תְפִלִּין בְבֵית הַקְּבָרות וּבְבֵית הַמֶּרְחָץ — בעיקרון · ללימוד, פנה לרב של חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרכו של אדמו״ר הזקן עם המחבר, הרמ״א (רמ״א) והמשנה ברורה (משנה ברורה) בנקודות המפתח של סימן מ״ה. שורת שולחן ערוך הרב (שולחן ערוך הרב) מובלטת — היא חושפת את רגישות הרב כפי שהיא עולה מטקסטו (§1, ג׳ סעיפים). הרב מפרש את הטעם ואת הפרט במקום שהמחבר מסתפק באמירה.

נושא מחבר רמ״א משנה ברורה דרך אדמו״ר הזקן (שולחן ערוך הרב)
בֵית הַקְּבָרות, ד׳ אַמּוֹת וְלֹעֵג לָרָשׁ או״ח מ״ה ס״א — אָסוּר לִכָּנֵס בְד׳ אַמּוֹת שֶּל מֵת וּתְפִלִּין בְרֹאשׁוֹ מִשּׁוּם לֹעֵג לָרָשׁ. — מ״ב מ״ה / ברכות י״ח ע״א — טַעַם לֹעֵג לָרָשׁ, מִצְווֹת בְטֵלות לַמֵתִים. דרך אדמו״ר הזקן — אָסוּר לִכָּנֵס בְד׳ אַמּוֹת שֶּל מֵת אוֹ קֶבֶר; בֵית הַקְּבָרות מֻקָּף מְחִצָּה — לְהַחְמִיר, וְחוּץ לַמְחִצָּה מֻתָּר (סעיף א).
מְגֻלִּין מוּל מְכֻסִּין וְהָרְצוּעוֹת או״ח מ״ה ס״א — וְאִם הֵם מְכֻסִּים מֻתָּר. — מ״ב מ״ה — גֶדֶר הַכִּסּוּי; אַף הָרְצוּעוֹת. דרך אדמו״ר הזקן — בִּמְכֻסִּין אֵין לֹעֵג לָרָשׁ, וּמֻתָּר בְכָל עִנְיָן; וְצָרִיךְ שֶׁגַּם הָרְצוּעוֹת מְכֻסּוֹת (סעיף ב).
בֵית הַמֶּרְחָץ וּבֵית הַטְבִילָה או״ח מ״ה ס״ב — חִיצוֹן מֻתָּר, אֶמְצָעִי אָסוּר לְכַתְּחִלָּה, פְנִימִי חוֹלֵץ. — מ״ב מ״ה — דַרְגוֹת הַמֶּרְחָץ; זֻהֲמָא מוּל צוֹנֵן. דרך אדמו״ר הזקן — הוֹסִיף בֵית הַטְבִילָה בְצוֹנֵן שֶּאֵין בּוֹ זֻהֲמָא וּמֻתָּר כְשֶׁאֵין שָׁם עָרֹם; וְאִסּוּר עֲמִידָה עָרֹם בִפְנֵי כִתְבֵי הַקֹּדֶשׁ (סעיף ג).

הטבלה נערכה מטקסט המחבר (אורח חיים מ״ה, ב׳ סעיפים, מקור ספריא) ומנושאי הכלים הניתנים לאימות (בית יוסף, טור, רמב״ם הלכות תפילין ד / הלכות ספר תורה, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות י״ח ע״א, שבת י׳ ע״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סעיפים) המובא ב§1. ייחוסי סעיף קטן שאינם ודאיים נותרו « על דרך הסברה ».

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה — מעבר ליישום הפשוט של הכלל. כל אחת בנויה: עיקרון הסימן → קריאת הרב → תוצאה מעשית. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — לֹעֵג לָרָשׁ וְד׳ אַמּוֹת

לֹעֵג לָרָשׁ וד׳ אמות

עיקרון הסימן: לֹא לִכָּנֵס בְד׳ אַמּוֹת שֶּל מֵת בִתְפִלִּין מְגֻלּות (סעיף א).

קריאת הרב: הַמֵּת אֵינוֹ יָכוֹל עוֹד לְקַיֵּם מִצְווֹת (מִצְווֹת בְטֵלות לַמֵתִים) ; לְשֵׂאתָן מְגֻלּות לְיָדוֹ הוּא לֹעֵג לָרָשׁ ; בֵית קְבָרות מֻקָּף מְחִצָּה — לְהַחְמִיר שֶׁלֹּא לִכָּנֵס כְלָל.

תוצאה מעשית: לְכַסּוֹת אוֹ לַחֲלֹץ אֶת הַתְּפִלִּין קֹדֶם הַכְּנִיסָה לְבֵית הַקְּבָרות אוֹ לְד׳ אַמּוֹת.

קו ב — כִּסּוּי הַתְּפִלִּין

מה שהכיסוי מתיר

עיקרון הסימן: מְגֻלִּין / מְכֻסִּין (סעיף ב).

קריאת הרב: אִם הֵן מְכֻסִּין אֵין לֹעֵג לָרָשׁ — מֻתָּר בְכָל עִנְיָן ; הַכִּסּוּי כּוֹלֵל אֶת הָרְצוּעוֹת.

תוצאה מעשית: בְבֵית הַקְּבָרות, תְּפִלִּין מְכֻסּות הֵיטֵב — בַּיִת וּרְצוּעוֹת — מַתִּירוֹת.

קו ג — בֵית הַמֶּרְחָץ וּבֵית הַטְבִילָה

בית המרחץ ובית הטבילה

עיקרון הסימן: מִדְרַג חַדְרֵי בֵית הַמֶּרְחָץ (סעיף ג).

קריאת הרב: חִיצוֹן מֻתָּר, אֶמְצָעִי אָסוּר לְכַתְּחִלָּה, פְנִימִי חוֹלֵץ ; אֲבָל בֵית הַטְבִילָה בְצוֹנֵן אֵין בּוֹ זֻהֲמָא — מֻתָּר כְשֶׁאֵין עָרֹם, וּמִכָּאן בִּרְכַּת הַטְבִילָה.

תוצאה מעשית: לַעֲמֹד עַל מַצַּב הַחֶדֶר וְהָעוֹמְדִים ; לֹא לַעֲמֹד עָרֹם בִפְנֵי כִתְבֵי הַקֹּדֶשׁ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן מ״ה ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, בהפניה לרב לפרטי המקרים (בית הקברות, הכיסוי, בית המרחץ).

לחסיד חב״ד — סימן מ״ה (דִּין תְפִלִּין בְבֵית הַקְּבָרות וּבְבֵית הַמֶּרְחָץ)

  • ① בבית הקברות. לֹא לִכָּנֵס בְד׳ אַמּוֹת שֶּל מֵת אוֹ קֶבֶר בִתְפִלִּין מְגֻלּות שֶּבְרֹאשׁוֹ — מִשּׁוּם לֹעֵג לָרָשׁ ; בֵית קְבָרות מֻקָּף מְחִצָּה, לְהַחְמִיר שֶׁלֹּא לִכָּנֵס כְלָל. יסוד: שוע״ר מ״ה סעיף א.
  • ② מכוסות. אִם הַתְּפִלִּין — בַּיִת וּרְצוּעוֹת — מְכֻסּות, בָּטֵל הָאִסּוּר וּמֻתָּר בְכָל עִנְיָן. יסוד: שוע״ר מ״ה סעיף ב.
  • ③ בבית המרחץ. לְפִי הַחֶדֶר : חִיצוֹן (כֻּלָּם לְבוּשִׁים) מֻתָּר ; אֶמְצָעִי אָסוּר לְכַתְּחִלָּה, בְלִי לַחֲלֹץ מַה שֶּׁמֻּנָּח ; פְּנִימִי (כֻּלָּם עֲרֻמִּים) חוֹלֵץ. יסוד: שוע״ר מ״ה סעיף ג.
  • ④ במקווה. בֵית הַטְבִילָה בְצוֹנֵן אֵין בּוֹ זֻהֲמָא — מֻתָּר לִכָּנֵס בּוֹ בִתְפִלִּין כְשֶׁאֵין שָׁם עָרֹם (וּמִכָּאן בִּרְכַּת הַטְבִילָה) ; אֲבָל לֹא לַעֲמֹד עָרֹם בִפְנֵי כִתְבֵי הַקֹּדֶשׁ. יסוד: שוע״ר מ״ה סעיף ג.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן מ״ה (הטקסט האמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — בית הקברות, הכיסוי, בית המרחץ — יש להתייעץ עם רב, ולחב״ד עם רב של חב״ד.
