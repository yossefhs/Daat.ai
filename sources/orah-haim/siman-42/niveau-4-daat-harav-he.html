דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן מ״ב · דִּין אִם מֻתָּר לְשַׁנּוֹת תְּפִלִּין שֶׁל יָד לְשֶׁל רֹאשׁ

דרכו של אדמו״ר הזקן בשינוי התפילין ובדין ההזמנה: אֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה (רֻבּוֹ שֶׁל שד״י בְּשֶׁל רֹאשׁ) ; מִשֶּׁל יָד לְשֶׁל רֹאשׁ מֻתָּר ; עִנְיַן הַהַזְמָנָה וְהַתְּנַאי ; וְהַזְמָנָה מִילְתָא (סוּדָר, כִּיס, קְלָף מְעֻבָּד לִשְׁמוֹ) — ברגישות חב״ד.

למה רמה 4 מיוחדת לאדמו״ר הזקן? השולחן ערוך של אדמו״ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתבו אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. דף זה מאסף את שיטת הרב על סימן מ״ב — דִּין אִם מֻתָּר לְשַׁנּוֹת תְּפִלִּין שֶׁל יָד לְשֶׁל רֹאשׁ. הסימן כולל אצל הרב ו׳ סְעִיפִים, שבהם אדמו״ר הזקן פורש: אִסּוּר שִׁנּוּי הַשֶּׁל רֹאשׁ לְשֶׁל יָד (אֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה) וְהֶתֵּר הַהֵפֶךְ ; עִנְיַן הַהַזְמָנָה וְהַתְּנַאי ; הָרְצוּעָה שֶׁנִּפְסְקָה וְהַגְּנִיזָה ; הַסּוּדָר / כִּיס הַמֻּזְמָן (הַזְמָנָה מִילְתָא), הַקָּטָן, וְהַהֶבְדֵּל בֵּין תַּשְׁמִישֵׁי קְדֻשָּׁה לְגוּף הַקְּדֻשָּׁה (קְלָף מְעֻבָּד לִשְׁמוֹ).

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בדף זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן מ״ב — ו׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנסרקה בספריא (Shulchan Arukh HaRav, Orach Chayim מ״ב). בששת הסעיפים, אדמו״ר הזקן דן באִסּוּר שִׁנּוּי הַשֶּׁל רֹאשׁ לְשֶׁל יָד (אֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה, רֻבּוֹ שֶׁל שד״י) וּבְהֶתֵּר הַהֵפֶךְ, בַּתְּנַאי הַמְבַטֵּל, בָּרְצוּעָה שֶׁנִּפְסְקָה וּצְרִיכָה גְּנִיזָה, בַּסּוּדָר וּבַכִּיס הַמֻּזְמָנִים (הַזְמָנָה מִילְתָא), בַּקָּטָן שֶׁיֵּשׁ לוֹ מַעֲשֶׂה וְאֵין לוֹ מַחֲשָׁבָה, וּבַהֶבְדֵּל בֵּין תַּשְׁמִישֵׁי קְדֻשָּׁה לְגוּף הַקְּדֻשָּׁה (קְלָף מְעֻבָּד לִשְׁמוֹ, עם ההפניה לסימן לב). הביאורים הם שלנו. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן מ״ב — אִם מֻתָּר לְשַׁנּוֹת תְּפִלִּין שֶׁל יָד לְשֶׁל רֹאשׁ

הטקסט המלא של שולחן ערוך אדמו״ר הזקן

שולחן ערוך הרב, סימן מ״ב — אִם מֻתָּר לְשַׁנּוֹת תְּפִלִּין שֶׁל יָד לְשֶׁל רֹאשׁ — וּבוֹ ו׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.42), ולאחריו ביאור קצר. הסימן כולל ו׳ סְעִיפִים.

סעיף א אָסוּר לְשַׁנּוֹת שֶׁל רֹאשׁ לְשֶׁל יָד (אֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה, רֻבּוֹ שֶׁל שד״י) ; מִשֶּׁל יָד לְשֶׁל רֹאשׁ מֻתָּר, וּבַחֲדָשִׁים אַף לְהֵפֶךְ.

אִם מֻתָּר לְשַׁנּוֹת תְּפִלִּין שֶׁל יָד לְשֶׁל רֹאשׁ וּבוֹ ו׳ סְעִיפִים:אָסוּר לְשַׁנּוֹת תְּפִלִּין שֶׁל רֹאשׁ לַעֲשׂוֹתָהּ שֶׁל יָד, אֲפִלּוּ אֵין לוֹ תְּפִלִּין שֶׁל יָד וְיֵשׁ לוֹ שְׁנַיִם שֶׁל רֹאשׁ. וַאֲפִלּוּ לִקַּח רְצוּעָה מִשֶּׁל רֹאשׁ לְשֶׁל יָד — אָסוּר. מִפְּנֵי שֶׁאֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לִקְדֻשָּׁה קַלָּה, וְשֶׁל רֹאשׁ קְדֻשָּׁתוֹ חֲמוּרָה, שֶׁרֻבּוֹ שֶׁל שֵׁם שד״י הוּא בְּשֶׁל רֹאשׁ — הַשִּׁי״ן בַּבָּתִּים וְהַד׳ בַּקֶּשֶׁר, אֲבָל בְּשֶׁל יָד אֵין בּוֹ — כִּי אִם הַיּוּ"ד. אֲבָל מִשֶּׁל יָד לְשֶׁל רֹאשׁ — מֻתָּר לְשַׁנּוֹת, שֶׁיַּעֲשֶׂה לוֹ אַרְבָּעָה בָּתִּים בְּתוֹכוֹ. וְאִם הָיוּ חֲדָשִׁים שֶׁלֹּא הֱנִיחָן עֲדַיִן עָלָיו — מֻתָּר לְשַׁנּוֹת אֲפִלּוּ מִשֶּׁל רֹאשׁ לְשֶׁל יָד, שֶׁטּוֹלֶה עֲלֵיהֶם מִכְסֵה עוֹר אֶחָד וְנִרְאֵית כְּבַיִת אֶחָד בַּחוּץ, וְאַף שֶׁהֵם אַרְבָּעָה מִבִּפְנִים — אֵין בְּכָךְ כְּלוּם בְּדִיעֲבַד כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן לב, וְאֵין בָּזֶה הוֹרָדָה מִקְּדֻשָּׁה חֲמוּרָה לִקְדֻשָּׁה קַלָּה הֵימֶנָּה אַף עַל פִּי שֶׁזִּמְּנָם וֶהֱכִינָם לְצֹרֶךְ הָרֹאשׁ, לְפִי שֶׁהַהַזְמָנָה אֵינָהּ כְּלוּם:

אֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה. אָסוּר לְשַׁנּוֹת שֶׁל רֹאשׁ לְשֶׁל יָד, וַאֲפִלּוּ לִקַּח רְצוּעָה — שֶׁאֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה, וְרֻבּוֹ שֶׁל שֵׁם שד״י בְּשֶׁל רֹאשׁ (הַשִּׁי״ן בַּבָּתִּים וְהַדָּלֶת בַּקֶּשֶׁר), וּבְשֶׁל יָד רַק הַיּוּ״ד. אֲבָל מִשֶּׁל יָד לְשֶׁל רֹאשׁ מֻתָּר. וּבַחֲדָשִׁים שֶׁלֹּא הֱנִיחָן — אַף מִשֶּׁל רֹאשׁ לְשֶׁל יָד מֻתָּר בְּמִכְסֵה עוֹר אֶחָד (סימן לב), שֶׁהַהַזְמָנָה אֵינָהּ כְּלוּם.

סעיף ב הִתְנָה עֲלֵיהֶם מִתְּחִלָּה — יָכוֹל לְשַׁנּוֹתָן אֲפִלּוּ לְבָשָׁן אָדָם עַל רֹאשׁוֹ אַחַר כָּךְ.

אִם הִתְנָה עֲלֵיהֶם מִתְּחִלָּה שֶׁיּוּכַל לְשַׁנּוֹתָהּ לְשֶׁל יָד — יָכוֹל לְשַׁנּוֹתָהּ אֲפִלּוּ לְבָשָׁן אָדָם עַל רֹאשׁוֹ אַחַר־כָּךְ:

הִתְנָה מִתְּחִלָּה. אִם הִתְנָה עֲלֵיהֶם מִתְּחִלָּה שֶׁיּוּכַל לְשַׁנּוֹתָן לְשֶׁל יָד — יָכוֹל לְשַׁנּוֹתָן אֲפִלּוּ לְבָשָׁן אָדָם עַל רֹאשׁוֹ אַחַר כָּךְ.

סעיף ג נִפְסְקָה רְצוּעָה שֶׁל יָד סָמוּךְ לַקֶּשֶׁר — לֹא יַעֲשֶׂנָּה כְּרִיכוֹת, שֶׁמּוֹרִידָהּ מִקְּדֻשָּׁתָהּ, אֶלָּא צְרִיכָה גְּנִיזָה.

אִם נִפְסְקָה הָרְצוּעָה שֶׁל יָד סָמוּךְ לַקֶּשֶׁר וְהוֹפֵךְ רֹאשׁ הַשֵּׁנִי לְמַעְלָה וְעוֹשֶׂה בּוֹ הַקֶּשֶׁר וְרֹאשׁ הָרִאשׁוֹן יִהְיֶה לְמַטָּה — לֹא יִקְשֹׁר שָׁם לְמַטָּה חֲתִיכַת הָרְצוּעָה שֶׁהָיָה הַקֶּשֶׁר בָּהּ שֶׁנִּפְסְקָה שָׁם, מִפְּנֵי שֶׁמּוֹרִידָהּ מִקְּדֻשָּׁתָהּ, כֵּיוָן שֶׁהָיָה בָּהּ הַקֶּשֶׁר וְהַיּוּ"ד וְעַתָּה יַעֲשֶׂה בָּהּ כְּרִיכוֹת הָאֶצְבָּעוֹת, אֶלָּא צְרִיכָה גְּנִיזָה. וְאִם אֵין הָרְצוּעָה אֲרֻכָּה כָּל־כָּךְ — אֲזַי לֹא יַעֲשֶׂה כָּל־כָּךְ כְּרִיכוֹת סְבִיב הַיָּד:

נִפְסְקָה הָרְצוּעָה שֶׁל יָד. נִפְסְקָה סָמוּךְ לַקֶּשֶׁר וְהוֹפֵךְ הָרֹאשׁ הַשֵּׁנִי לְמַעְלָה — לֹא יִקְשֹׁר לְמַטָּה אֶת חֲתִיכַת הָרְצוּעָה שֶׁהָיָה בָּהּ הַקֶּשֶׁר וְהַיּוּ״ד, מִפְּנֵי שֶׁמּוֹרִידָהּ מִקְּדֻשָּׁתָהּ לַעֲשׂוֹתָהּ כְּרִיכוֹת, אֶלָּא צְרִיכָה גְּנִיזָה ; וְאִם אֵינָהּ אֲרֻכָּה כָּל כָּךְ — לֹא יַרְבֶּה בִּכְרִיכוֹת סְבִיב הַיָּד.

סעיף ד סוּדָר שֶׁהִזְמִינוֹ וְצָרַר בּוֹ תְּפִלִּין — נֶאֱסַר לְמָעוֹת, אֶלָּא אִם כֵּן הִתְנָה ; וְאָדָם אֵינוֹ אוֹסֵר דָּבָר שֶׁאֵינוֹ שֶׁלּוֹ.

סוּדָר שֶׁהִזְמִינוֹ לִצְרֹר בּוֹ תְּפִלִּין לְעוֹלָם וְלֹא לְפִי שָׁעָה בִּלְבַד וְאַחַר־כָּךְ צָרַר בּוֹ תְּפִלִּין פַּעַם אַחַת, אֲפִלּוּ צְרָרָן בּוֹ לְפִי שָׁעָה בִּסְתָם וְלֹא עַל דַּעַת שֶׁיְּהֵא מְיֻחָד לְכָךְ לְעוֹלָם — נֶאֱסַר לְהִשְׁתַּמֵּשׁ בּוֹ דְּבַר חֹל כְּגוֹן לִצְרֹר בּוֹ מָעוֹת, אֶלָּא אִם כֵּן הִתְנָה עָלָיו בַּתְּחִלָּה שֶׁיּוּכַל לִצְרֹר בּוֹ מָעוֹת כְּשֶׁיִּרְצֶה, אוֹ שֶׁבְּשָׁעָה שֶׁצָּרַר בּוֹ הַתְּפִלִּין הִתְנָה בְּדַעְתּוֹ שֶׁצּוֹרְרָן בּוֹ עַל דַּעַת לְפַנּוֹתָן לְאַחַר זְמַן וְיַחֲזֹר לִהְיוֹת חֹל כְּמוֹ שֶׁהָיָה קֹדֶם שֶׁצָּרַר בּוֹ. וְאִם עָשָׂה כִּיס לְשֵׁם תְּפִלִּין אֲפִלּוּ צָרַר בּוֹ עַל דַּעַת לְפַנּוֹתָן — אָסוּר. וְהוּא הַדִּין אִם הָיָה כִּיס וְהוֹסִיף בּוֹ אֵיזֶה דָּבָר לְשֵׁם תְּפִלִּין וְאַחַר־כָּךְ צְרָרָן בּוֹ אֲפִלּוּ עַל דַּעַת לְפַנּוֹתָן — אָסוּר עַד שֶׁיִּטֹּל מַה שֶּׁחִדֵּשׁ. אֲבָל אִם לֹא צְרָרָן בּוֹ מֵעוֹלָם — אֵינוֹ נֶאֱסָר בְּהַזְמָנָה, אֲפִלּוּ עָשָׂה כִּיס מֵחָדָשׁ לְשֵׁם תְּפִלִּין. וְכֵן אִם צְרָרָן בּוֹ אֲפִלּוּ פְּעָמִים רַבּוֹת לְפִי שָׁעָה וְלֹא הִזְמִינוֹ מֵעוֹלָם לְשֵׁם תְּפִלִּין קֹדֶם שֶׁצְּרָרָן בּוֹ, וְגַם כְּשֶׁצְּרָרָן בּוֹ לֹא הָיָה אָז בְּדַעְתּוֹ שֶׁיִּצְרֹר בּוֹ תְּפִלִּין לְעוֹלָם — מֻתָּר לְהִשְׁתַּמֵּשׁ חֹל אֲפִלּוּ בְּשָׁעָה שֶׁמִּשְׁתַּמֵּשׁ בּוֹ קֹדֶשׁ, כְּגוֹן לִצְרֹר בּוֹ תְּפִלִּין וּמָעוֹת יַחַד. אֲבָל אִם צְרָרָן בּוֹ אֲפִלּוּ פַּעַם אַחַת עַל דַּעַת לְצָרְרָן בּוֹ לְעוֹלָם — הֲרֵי זֶה כְּאִלּוּ הִזְמִינוֹ וּצְרָרָן בּוֹ וְנֶאֱסַר. וְכָל זֶה בְּסוּדָר אוֹ כִּיס שֶׁלּוֹ, אֲבָל אָדָם אֵינוֹ אוֹסֵר דָּבָר שֶׁאֵינוֹ שֶׁלּוֹ, אֶלָּא אִם כֵּן גָּנַב בֶּגֶד וַעֲשָׂאוֹ כִּיס, שֶׁאָז קְנָאוֹ בְּשִׁנּוּי מַעֲשֶׂה וַאֲסָרוֹ:

סוּדָר שֶׁהִזְמִינוֹ וְצָרַר בּוֹ. סוּדָר שֶׁהִזְמִינוֹ לִצְרֹר בּוֹ תְּפִלִּין לְעוֹלָם וְאַחַר כָּךְ צָרַר בּוֹ פַּעַם אַחַת — נֶאֱסַר לְהִשְׁתַּמֵּשׁ בּוֹ חֹל (לִצְרֹר מָעוֹת), אֶלָּא אִם כֵּן הִתְנָה אוֹ כִּוֵּן לְפַנּוֹתָן וְשֶׁיַּחֲזֹר לְחֹל. וְכִיס לְשֵׁם תְּפִלִּין — אָסוּר אַף עַל דַּעַת לְפַנּוֹתָן. אֲבָל אִם לֹא צָרַר מֵעוֹלָם — אֵינוֹ נֶאֱסָר בְּהַזְמָנָה ; וְאִם צָרַר בְּלֹא הַזְמָנָה — מֻתָּר אַף בְּיַחַד. וְאָדָם אֵינוֹ אוֹסֵר דָּבָר שֶׁאֵינוֹ שֶׁלּוֹ (אֶלָּא אִם גָּנַב וְקָנָהוּ בְּשִׁנּוּי מַעֲשֶׂה).

סעיף ה קָטָן שֶׁצָּרַר בְּסוּדָר מֻזְמָן — נֶאֱסַר ; אֲבָל הַנֶּאֱסָר בְּהַזְמָנָה לְבַדָּהּ אֵינוֹ נֶאֱסָר בְּהַזְמָנַת קָטָן.

קָטָן שֶׁצָּרַר תְּפִלִּין בְּסוּדָר שֶׁהִזְמִינוֹ לְכָךְ — נֶאֱסַר, אֲבָל דָּבָר הַנֶּאֱסָר בְּהַזְמָנָה לְבַדָּהּ כְּמוֹ שֶׁיִּתְבָּאֵר — אֵינוֹ נֶאֱסָר בְּהַזְמָנַת קָטָן, מִפְּנֵי שֶׁחֵרֵשׁ שׁוֹטֶה וְקָטָן יֵשׁ לָהֶם מַעֲשֶׂה וְאֵין לָהֶם מַחֲשָׁבָה, אֲפִלּוּ גִּלּוּ מַחֲשַׁבְתָּם בְּדִבּוּרָם:

קָטָן שֶׁצָּרַר. קָטָן שֶׁצָּרַר תְּפִלִּין בְּסוּדָר שֶׁהִזְמִינוֹ — נֶאֱסַר ; אֲבָל הַנֶּאֱסָר בְּהַזְמָנָה לְבַדָּהּ אֵינוֹ נֶאֱסָר בְּהַזְמָנַת קָטָן, שֶׁחֵרֵשׁ שׁוֹטֶה וְקָטָן יֵשׁ לָהֶם מַעֲשֶׂה וְאֵין לָהֶם מַחֲשָׁבָה, אַף שֶׁגִּלּוּ מַחֲשַׁבְתָּם בְּדִבּוּרָם.

סעיף ו הַזְמָנָה מוֹעֶלֶת בְּתַשְׁמִישֵׁי קְדֻשָּׁה ; גּוּף הַקְּדֻשָּׁה (קְלָף מְעֻבָּד לִשְׁמוֹ) אָסוּר לְחֹל אֶלָּא בִּתְנַאי ; וּלְהַחֲמִיר כַּסְּבָרָא הָרִאשׁוֹנָה.

יֵשׁ אוֹמְרִים שֶׁלֹּא אָמְרוּ שֶׁהַזְמָנָה בִּלְבַדָּהּ אֵינָהּ מוֹעֶלֶת לֶאֱסֹר לְהִשְׁתַּמֵּשׁ בּוֹ חֹל אֶלָּא כְּשֶׁהִזְמִינוֹ לִהְיוֹת תַּשְׁמִישֵׁי קְדֻשָּׁה, כְּגוֹן כִּיס שֶׁל תְּפִלִּין. וְכֵן רְצוּעוֹת שֶׁל תְּפִלִּין — תַּשְׁמִישֵׁי קְדֻשָּׁה הֵן, וְלֹא גּוּף הַקְּדֻשָּׁה. אֲבָל אִם הִזְמִינוֹ לִהְיוֹת גּוּף הַקְּדֻשָּׁה, כְּגוֹן קְלָף הַמְעֻבָּד לְשֵׁם סֵפֶר תּוֹרָה אוֹ תְּפִלִּין וּמְזוּזוֹת — אָסוּר לִכְתֹּב עָלָיו דִּבְרֵי חֹל, אֶלָּא אִם כֵּן הִתְנָה עָלָיו. אֲבָל מֻתָּר לְהוֹרִידוֹ לִקְדֻשָּׁה קַלָּה, שֶׁהֲרֵי תְּפִלִּין שֶׁל רֹאשׁ מֻתָּר לַעֲשׂוֹתָהּ שֶׁל יָד אִם הֵם חֲדָשִׁים, אַף עַל פִּי שֶׁהֵם גּוּף הַקְּדֻשָּׁה. וְכֵן קְלָף הַמְעֻבָּד לְשֵׁם סֵפֶר תּוֹרָה — מֻתָּר לִכְתֹּב עָלָיו תְּפִלִּין וּמְזוּזוֹת כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן לב, אַף עַל פִּי שֶׁקְּדֻשַּׁת סֵפֶר תּוֹרָה גְּדוֹלָה מִקְּדֻשָּׁתָן. וְגֵט לֹא נִקְרָא דָּבָר שֶׁבִּקְדֻשָּׁה כְּלָל, וְאָסוּר לְכָתְבוֹ עַל קְלָף הַמְעֻבָּד לְשֵׁם סֵפֶר תּוֹרָה אֶלָּא אִם כֵּן הִתְנָה. וְיֵשׁ אוֹמְרִים שֶׁאַף בְּגוּף הַקְּדֻשָּׁה — הַזְמָנָה אֵינָהּ כְּלוּם, עַד שֶׁיַּעֲשֶׂה בּוֹ מַעֲשֶׂה שֶׁל גּוּף הַקְּדֻשָּׁה, כְּגוֹן שֶׁיַּתְחִיל לִכְתֹּב עַל הַקְּלָף סֵפֶר תּוֹרָה אוֹ תְּפִלִּין אֵיזֶה דָּבָר בְּחוּטֵי צִיצִית הַתְּלוּיִים בַּבֶּגֶד כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן כא (וְאֵין צָרִיךְ לוֹמַר בְּתַשְׁמִישֵׁי קְדֻשָּׁה, כְּגוֹן לִכְרֹךְ אֵיזֶה דָּבָר בָּרְצוּעָה הַקָּבוּעַ בִּתְפִלִּין שֶׁעָלָיו, שֶׁאֲפִלּוּ תְּנַאי אֵינוֹ מוֹעִיל בָּזֶה מִשּׁוּם בִּזּוּי מִצְוָה). וּלְעִנְיַן הֲלָכָה — יֵשׁ לְהַחֲמִיר כַּסְּבָרָא הָרִאשׁוֹנָה, אַף עַל פִּי שֶׁהָעִקָּר כַּסְּבָרָא הַשְּׁנִיָּה. וְאַף־עַל־פִּי־כֵן נָהֲגוּ עַכְשָׁו לִכְתֹּב גִּטִּין וּקְמֵיעוֹת עַל קְלָף הַמְעֻבָּד לְשֵׁם סֵפֶר תּוֹרָה, לְפִי שֶׁכֵּיוָן שֶׁנָּהֲגוּ כֵּן — הֲרֵי זֶה כְּאִלּוּ הִתְנָה עָלָיו מִתְּחִלָּה, שֶׁהֲרֵי כְּשֶׁהַסּוֹפְרִים מְתַקְּנִים אוֹתָן — מְתַקְּנִים אוֹתָן עַל דַּעַת לְהִשְׁתַּמֵּשׁ בּוֹ הֵן לִדְבַר קְדֻשָּׁה הֵן לִדְבַר הָרְשׁוּת, כֵּיוָן שֶׁמִּנְהָגָם בְּכָךְ. (וְכֵן מַה שֶּׁנָּהֲגוּ הַרְבֵּה בְּנֵי אָדָם שֶׁמִּשְׁתַּמְּשִׁין בַּשַּׂק שֶׁל תְּפִלִּין כְּשֶׁהֵם בַּדֶּרֶךְ הוּא מִטַּעַם זֶה, שֶׁכֵּיוָן שֶׁנָּהֲגוּ כֵּן — הֲרֵי זֶה כְּאִלּוּ הִתְנוּ עָלָיו כְּשֶׁעֲשָׂאוּהוּ אוֹ כְּשֶׁקְּנָאוּהוּ שֶׁיִּשְׁתַּמְּשׁוּ בּוֹ בַּדֶּרֶךְ כְּמִנְהָגָם):

תַּשְׁמִישֵׁי קְדֻשָּׁה וְגוּף הַקְּדֻשָּׁה. יֵשׁ אוֹמְרִים שֶׁהַזְמָנָה לְבַדָּהּ אֵינָהּ אוֹסֶרֶת אֶלָּא בְּתַשְׁמִישֵׁי קְדֻשָּׁה (כְּגוֹן כִּיס, וְכֵן הָרְצוּעוֹת). אֲבָל גּוּף הַקְּדֻשָּׁה — קְלָף הַמְעֻבָּד לִשְׁמוֹ — אָסוּר לִכְתֹּב עָלָיו חֹל אֶלָּא אִם הִתְנָה, אַךְ מֻתָּר לְהוֹרִידוֹ לִקְדֻשָּׁה קַלָּה (כְּשֶׁל רֹאשׁ חָדָשׁ לְשֶׁל יָד, וְסִימָן לב). וְגֵט אֵינוֹ דָּבָר שֶׁבִּקְדֻשָּׁה. וְיֵשׁ אוֹמְרִים שֶׁאַף בְּגוּף הַקְּדֻשָּׁה הַזְמָנָה אֵינָהּ כְּלוּם עַד שֶׁיַּעֲשֶׂה מַעֲשֶׂה, וְאָז אֵין תְּנַאי מוֹעִיל ; אֲבָל בְּתַשְׁמִישֵׁי קְדֻשָּׁה תְּנַאי מוֹעִיל, וּבְתַשְׁמִישֵׁי מִצְוָה (ציצית, שופר, לולב, נר חנוכה) אֵין צָרִיךְ תְּנַאי. וּלְעִנְיַן הֲלָכָה יֵשׁ לְהַחֲמִיר כַּסְּבָרָא הָרִאשׁוֹנָה, אַף שֶׁהָעִקָּר כַּשְּׁנִיָּה ; וְנָהֲגוּ לִכְתֹּב גִּטִּין וּקְמֵיעוֹת עַל קְלָף מְעֻבָּד לְשֵׁם ס״ת, שֶׁהַמִּנְהָג כִּתְנַאי, וְכֵן הַשַּׂק שֶׁל תְּפִלִּין בַּדֶּרֶךְ.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim מ״ב (מהדורת קה״ת, ו׳ סְעִיפִים). ביאור: DAAT. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.

2 · היסוד

היסוד — שִׁנּוּי הַתְּפִלִּין וְדִין הַהַזְמָנָה

היסוד: שינוי התפילין ודין ההזמנה

קריאת אדמו״ר הזקן על סימן זה בשלושה מהלכים. תחילה מַדְרֵגוֹת הַקְּדֻשָּׁה: אֵין מוֹרִידִין אֶת הַשֶּׁל רֹאשׁ (שֶׁבּוֹ רֻבּוֹ שֶׁל שד״י) לְשֶׁל יָד — אֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה — וְהַהֵפֶךְ מֻתָּר, וְחֲדָשִׁים מִשְׁתַּנִּים אַף לְמַטָּה. אַחַר כָּךְ עִנְיַן הַהַזְמָנָה וְהַתְּנַאי: הַהַזְמָנָה לְבַדָּהּ ״אֵינָהּ כְּלוּם״ בְּגוּף הַתְּפִלִּין, אֲבָל סוּדָר מֻזְמָן וּמְשֻׁמָּשׁ נֶאֱסָר, וְהַתְּנַאי מְבַטֵּל. וּלְבַסּוֹף הַזְמָנָה מִילְתָא: הַהֶבְדֵּל בֵּין תַּשְׁמִישֵׁי קְדֻשָּׁה לְגוּף הַקְּדֻשָּׁה, הַקְּלָף הַמְעֻבָּד לִשְׁמוֹ, הַקָּטָן וְהַפְּסַק הָאַחֲרוֹן.

אֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה

כבוד התפילין ואין מורידין

היסוד. לַתְּפִלִּין מַדְרֵגוֹת קְדֻשָּׁה: הַשֶּׁל רֹאשׁ עוֹלֶה עַל הַשֶּׁל יָד, שֶׁרֻבּוֹ שֶׁל שד״י בּוֹ (מַעֲלִין בַּקֹּדֶשׁ וְאֵין מוֹרִידִין) ; לְפִיכָךְ אֵין מְשַׁנִּין שֶׁל רֹאשׁ לְשֶׁל יָד, וַאֲפִלּוּ רְצוּעָה.

התוצאה. מִשֶּׁל יָד לְשֶׁל רֹאשׁ מֻתָּר ; וּבַחֲדָשִׁים שֶׁלֹּא הֱנִיחָם אַף מִשֶּׁל רֹאשׁ לְשֶׁל יָד בְּמִכְסֵה עוֹר, שֶׁ״הַהַזְמָנָה אֵינָהּ כְּלוּם״. וְעַל הַקְּלָף הַמְעֻבָּד לִשְׁמוֹ עַיֵּן סימן 32 ; וְעַל הָרְצוּעוֹת סימן 44.

הַהַזְמָנָה וְהַתְּנַאי

ההזמנה והתנאי

העיקרון. הַהַזְמָנָה לְבַדָּהּ ״אֵינָהּ כְּלוּם״ בְּגוּף הַתְּפִלִּין (וּמִשּׁוּם כָּךְ מֻתָּר לְשַׁנּוֹת חֲדָשִׁים) ; אֲבָל סוּדָר שֶׁהִזְמִינוֹ לְעוֹלָם וְצָרַר בּוֹ פַּעַם אַחַת — נֶאֱסַר לְחֹל (מָעוֹת).

ההרחבה. הַתְּנַאי מִתְּחִלָּה מְבַטֵּל אֶת הָאִסּוּר: אַף לְאַחַר שֶׁלְּבָשָׁן (סעיף ב), וְאַף בַּסּוּדָר (סעיף ד). וְכֵן אָדָם אֵינוֹ אוֹסֵר דָּבָר שֶׁאֵינוֹ שֶׁלּוֹ.

תַּשְׁמִישֵׁי קְדֻשָּׁה וְגוּף הַקְּדֻשָּׁה

תשמישי קדושה וגוף הקדושה

העיקרון. הַזְמָנָה מִילְתָא מַבְדִּיל בֵּין תַּשְׁמִישׁ קְדֻשָּׁה (כִּיס, רְצוּעוֹת), שֶׁהַהַזְמָנָה + הַשִּׁמּוּשׁ אוֹסְרִים בּוֹ חֹל, לְבֵין גּוּף הַקְּדֻשָּׁה (קְלָף מְעֻבָּד לִשְׁמוֹ), שֶׁאָסוּר לִכְתֹּב עָלָיו חֹל אֶלָּא בִּתְנַאי.

ההנהגה. הַקָּטָן יֵשׁ לוֹ מַעֲשֶׂה וְאֵין לוֹ מַחֲשָׁבָה. וּלְהֲלָכָה יֵשׁ לְהַחֲמִיר כַּסְּבָרָא הָרִאשׁוֹנָה, אַף שֶׁהָעִקָּר כַּשְּׁנִיָּה.

הפניה ללימוד. משקל היסוד שהונח כאן — אֵין מוֹרִידִין, הַהַזְמָנָה וְהַתְּנַאי, וְהַזְמָנָה מִילְתָא — נוגע גם במעשה (לשנות תפילין או לא, הסודר / כיס, הקלף המעובד לשמו) וגם ברגישות חב״ד על קדושת חפצי המצווה. דף זה מציג זאת ברמת העיקרון המתועד בטקסט של הרב, בלי להוסיף דבר מומצא. ללימוד מעמיק ולהנהגה אישית — שינוי תפילין, הסודר המוזמן, הקלף המעובד לשמו — יש ללמוד עם רב של חב״ד ובתוך המקורות עצמם.

אִם מֻתָּר לְשַׁנּוֹת תְּפִלִּין שֶׁל יָד לְשֶׁל רֹאשׁ — בעיקרון · ללימוד, פנה לרב של חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרכו של אדמו״ר הזקן עם המחבר, הרמ״א (רמ״א) והמשנה ברורה (משנה ברורה) בנקודות המפתח של סימן מ״ב. שורת שולחן ערוך הרב (שולחן ערוך הרב) מובלטת — היא חושפת את רגישות הרב כפי שהיא עולה מטקסטו (§1, ו׳ סעיפים). הרב מפרש את הטעם ואת הפרט במקום שהמחבר מסתפק באמירה.

נושא מחבר רמ״א משנה ברורה דרך אדמו״ר הזקן (שולחן ערוך הרב)
שינוי תפילין (של ראש לשל יד) או״ח מ״ב:א — מִשֶּׁל רֹאשׁ לְשֶׁל יָד אָסוּר (אֵין מוֹרִידִין) ; מִשֶּׁל יָד לְשֶׁל רֹאשׁ מֻתָּר ; חֲדָשִׁים מֻתָּר לְשַׁנּוֹת. — מ״ב מ״ב / מנחות ל״ד ע״ב, ברכות כ״ח ע״א — רֻבּוֹ שֶׁל שד״י ; מַעֲלִין בַּקֹּדֶשׁ. דרך אדמו״ר הזקן — אֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה (הַשִּׁי״ן וְהַדָּלֶת בְּשֶׁל רֹאשׁ) ; וּבַחֲדָשִׁים אַף לְשֶׁל יָד בְּמִכְסֵה עוֹר, שֶׁהַהַזְמָנָה אֵינָהּ כְּלוּם (סעיף א).
התנאי וההזמנה או״ח מ״ב:ב — הִתְנָה מִתְּחִלָּה — יָכוֹל לְשַׁנּוֹתָן אַף לְאַחַר שֶׁלְּבָשָׁן. — מ״ב מ״ב — גֶּדֶר הַתְּנַאי בַּתְּפִלִּין וּבַתַּשְׁמִישִׁים. דרך אדמו״ר הזקן — הַתְּנַאי מִתְּחִלָּה מְבַטֵּל — אַף לְבָשָׁן אָדָם עַל רֹאשׁוֹ אַחַר כָּךְ (סעיף ב).
הזמנה מילתא — סודר, כיס, קטן או״ח מ״ב:ג — סוּדָר שֶׁאִזְמְנֵיהּ וְצָר בֵּיהּ — נֶאֱסַר לִצְרֹר בּוֹ זוּזֵי. רמ״א מ״ב:ג — אִם הִתְנָה עָלָיו — הַכֹּל שָׁרֵי. מ״ב מ״ב / סנהדרין מ״ז ע״ב–מ״ח ע״ב — הַזְמָנָה מִילְתָא ; אַזְמְנֵיהּ וְצָר ; קָטָן. דרך אדמו״ר הזקן — מֻזְמָן וּמְשֻׁמָּשׁ נֶאֱסָר לְחֹל אֶלָּא בִּתְנַאי ; אָדָם אֵינוֹ אוֹסֵר שֶׁאֵינוֹ שֶׁלּוֹ ; וְקָטָן יֵשׁ לוֹ מַעֲשֶׂה וְאֵין לוֹ מַחֲשָׁבָה (סעיפים ג-ה).
תשמישי קדושה מול גוף הקדושה או״ח מ״ב:ג — קְלָף מְעֻבָּד לְשֵׁם תְּפִלִּין — אָסוּר לִכְתֹּב עָלָיו חֹל. — מ״ב מ״ב / ביאור הלכה — תַּשְׁמִישֵׁי קְדֻשָּׁה, גּוּף הַקְּדֻשָּׁה וְהַתְּנַאי. דרך אדמו״ר הזקן — הַזְמָנָה מוֹעֶלֶת בְּתַשְׁמִישֵׁי קְדֻשָּׁה ; בְּגוּף הַקְּדֻשָּׁה אֵין תְּנַאי מוֹעִיל מִשֶּׁנַּעֲשָׂה מַעֲשֶׂה ; וּלְהֲלָכָה יֵשׁ לְהַחֲמִיר כַּסְּבָרָא הָרִאשׁוֹנָה (סעיף ו).

הטבלה נערכה מטקסט המחבר (אורח חיים מ״ב, ג׳ סעיפים, מקור ספריא) ומנושאי הכלים הניתנים לאימות (בית יוסף, טור, רמב״ם הלכות תפילין ג, רא״ש, ר״ן ס״פ נגמר הדין, רבנו ירוחם, מהרי״א, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד מנחות ל״ד ע״ב–ל״ה ע״א / צ״ט ע״א, סנהדרין מ״ז ע״ב–מ״ח ע״ב, ברכות כ״ח ע״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ו׳ סְעִיפִים) המובא ב§1. ייחוסים שאינם ודאיים נותרו « על דרך הסברה ».

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה — מעבר ליישום הפשוט של הכלל. כל אחת בנויה: עיקרון הסימן → קריאת הרב → תוצאה מעשית. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — אֵין מוֹרִידִין וְשִׁנּוּי הַתְּפִלִּין

אין מורידין ושינוי התפילין

עיקרון הסימן: אִסּוּר שִׁנּוּי הַשֶּׁל רֹאשׁ לְשֶׁל יָד וְהֶתֵּר הַהֵפֶךְ (סעיף א).

קריאת הרב: הַמַּדְרֵגָה תְּלוּיָה בְּרֻבּוֹ שֶׁל שד״י שֶׁבְּשֶׁל רֹאשׁ ; אֵין מוֹרִידִין מִקְּדֻשָּׁה חֲמוּרָה לְקַלָּה. וְהַחֲדָשִׁים יוֹצְאִים מִן הַכְּלָל (מִכְסֵה עוֹר), שֶׁהַהַזְמָנָה אֵינָהּ כְּלוּם.

תוצאה מעשית: לֹא לַעֲשׂוֹת שֶׁל רֹאשׁ שֶׁל יָד וְלֹא לִקַּח מִמֶּנּוּ רְצוּעָה ; וְהַהֵפֶךְ מֻתָּר.

קו ב — הַהַזְמָנָה וְהַתְּנַאי

ההזמנה והתנאי

עיקרון הסימן: הַתְּנַאי וְהַסּוּדָר הַמֻּזְמָן (סעיפים ב, ד).

קריאת הרב: הַהַזְמָנָה לְבַדָּהּ אֵינָהּ אוֹסֶרֶת אֶלָּא סוּדָר מֻזְמָן וּמְשֻׁמָּשׁ ; הַתְּנַאי מִתְּחִלָּה מְבַטֵּל, אַף לְאַחַר לְבִישָׁה.

תוצאה מעשית: בְּסוּדָר / כִּיס תְּפִלִּין — לְהַתְנוֹת אִם רוֹצֶה לְהַשְׁאִיר בּוֹ שִׁמּוּשׁ חֹל ; וְאִם לָאו, הַהַזְמָנָה וְהַשִּׁמּוּשׁ אוֹסְרִים.

קו ג — תַּשְׁמִישֵׁי קְדֻשָּׁה מוּל גּוּף הַקְּדֻשָּׁה

תשמישי קדושה מול גוף הקדושה והקטן

עיקרון הסימן: הָרְצוּעָה שֶׁנִּפְסְקָה, הַקָּטָן, וְהַקְּלָף הַמְעֻבָּד לִשְׁמוֹ (סעיפים ג, ה, ו).

קריאת הרב: הַקָּטָן יֵשׁ לוֹ מַעֲשֶׂה וְאֵין לוֹ מַחֲשָׁבָה ; רְצוּעַת הַקֶּשֶׁר אֵינָהּ יוֹרֶדֶת מִקְּדֻשָּׁתָהּ (גְּנִיזָה) ; וּבְגוּף הַקְּדֻשָּׁה אֵין תְּנַאי מוֹעִיל מִשֶּׁנַּעֲשָׂה מַעֲשֶׂה.

תוצאה מעשית: לִגְנֹז רְצוּעָה שֶׁנִּפְסְקָה ; לֹא לִכְתֹּב חֹל עַל קְלָף מְעֻבָּד לִשְׁמוֹ ; וְלִסְמֹךְ עַל הָרַב לְמַעֲשֶׂה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן מ״ב ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, בהפניה לרב לפרטי המקרים (לשנות תפילין, הסודר המוזמן, הקלף המעובד לשמו).

לחסיד חב״ד — סימן מ״ב (אִם מֻתָּר לְשַׁנּוֹת תְּפִלִּין שֶׁל יָד לְשֶׁל רֹאשׁ)

  • ① שינוי התפילין. לֹא לַעֲשׂוֹת שֶׁל רֹאשׁ שֶׁל שֶׁל יָד (אֵין מוֹרִידִין) — וְלֹא לִקַּח מִמֶּנּוּ רְצוּעָה ; וְהַהֵפֶךְ מֻתָּר, וְחֲדָשִׁים מִשְׁתַּנִּים אַף לְמַטָּה (מִכְסֵה עוֹר). יסוד: שוע״ר מ״ב סעיף א.
  • ② התנאי. תְּנַאי מִתְּחִלָּה מַתִּיר לְשַׁנּוֹת אֶת הַתְּפִלִּין, אַף לְאַחַר שֶׁלְּבָשָׁן. יסוד: שוע״ר מ״ב סעיף ב.
  • ③ רצועה שנפסקה. רְצוּעָה שֶׁל יָד שֶׁנִּפְסְקָה סָמוּךְ לַקֶּשֶׁר אֵינָהּ יוֹרֶדֶת לִכְרִיכוֹת — צְרִיכָה גְּנִיזָה. יסוד: שוע״ר מ״ב סעיף ג.
  • ④ הסודר / כיס של תפילין. מֻזְמָן לְעוֹלָם וּמְשֻׁמָּשׁ פַּעַם אַחַת — נֶאֱסָר לְחֹל בְּלֹא תְּנַאי ; אָדָם אֵינוֹ אוֹסֵר דָּבָר שֶׁאֵינוֹ שֶׁלּוֹ ; וְקָטָן אֵינוֹ אוֹסֵר בְּהַזְמָנָתוֹ לְבַדָּהּ. יסוד: שוע״ר מ״ב סעיף ד-ה.
  • ⑤ קלף מעובד לשמו / גוף הקדושה. לֹא לִכְתֹּב חֹל עַל קְלָף מְעֻבָּד לִשְׁמוֹ בְּלֹא תְּנַאי ; וּלְהֲלָכָה יֵשׁ לְהַחֲמִיר כַּסְּבָרָא הָרִאשׁוֹנָה. יסוד: שוע״ר מ״ב סעיף ו.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן מ״ב (הטקסט האמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — לשנות תפילין, הסודר המוזמן, הקלף המעובד לשמו — יש להתייעץ עם רב, ולחב״ד עם רב של חב״ד.
