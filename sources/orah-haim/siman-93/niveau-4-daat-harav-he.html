דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן צ״ג · קֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת

דרכו של אדמו״ר הזקן בקֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת : לִשְׁהוֹת שָׁעָה קֹדֶם וְאַחַר הַתְּפִלָּה כְּדֵי לְכַוֵּן ; לַעֲמֹד מִתּוֹךְ כֹּבֶד רֹאשׁ וְלֹא מִתּוֹךְ שְׂחוֹק וְקַלּוּת רֹאשׁ, אֶלָּא מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה ; מִתּוֹךְ הֲלָכָה פְּסוּקָה וְלֹא מִתּוֹךְ דִּין וּפִלְפּוּל ; וְהָעוֹסֵק בְּצָרְכֵי צִבּוּר כְּעוֹסֵק בַּתּוֹרָה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן צ״ג — קֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת (הַהֲכָנָה הַפְּנִימִית קֹדֶם הַתְּפִלָּה). אצל הרב הסימן נפרש לד׳ סְעִיפִים : הַשְּׁהִיָּה שָׁעָה קֹדֶם וְאַחַר הַתְּפִלָּה לְשֵׁם כַּוָּנָה, הַכֹּבֶד רֹאשׁ מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה, הַהֲלָכָה פְּסוּקָה (וְלֹא מִתּוֹךְ דִּין וּפִלְפּוּל), וְהָעוֹסֵק בְּצָרְכֵי צִבּוּר כְּעוֹסֵק בַּתּוֹרָה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן צ״ג — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim צ״ג). אדמו״ר הזקן דן בו בְּקֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן צ״ג — קֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן צ״ג — קֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.93), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.93.k).

סעיף א לִשְׁהוֹת שָׁעָה קֹדֶם וְאַחַר הַתְּפִלָּה — כְּדֵי לְכַוֵּן ; חֲסִידִים הָרִאשׁוֹנִים שָׁעָה מַמָּשׁ.

קֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת וּבוֹ ד׳ סְעִיפִים:הַמִּתְפַּלֵּל צָרִיךְ שֶׁיִּשְׁהֶה בַּמָּקוֹם שֶׁמִּתְפַּלֵּל שָׁעָה אַחַת קֹדֶם תְּפִלָּתוֹ, כְּדֵי שֶׁיְּכַוֵּן לִבּוֹ לַמָּקוֹם, שֶׁנֶּאֱמַר: ״אַשְׁרֵי יוֹשְׁבֵי בֵיתֶךָ״ וְאַחַר כָּךְ ״יְהַלְלוּךָ סֶּלָה״, וְשָׁעָה אַחַת אַחַר תְּפִלָּתוֹ, שֶׁלֹּא תְּהֵא נִרְאֵית עָלָיו כְּמַשְּׂאוֹי שֶׁמְּמַהֵר לָצֵאת מִמֶּנָּה, שֶׁנֶּאֱמַר: ״אַךְ צַדִּיקִים יוֹדוּ לִשְׁמֶךָ״ וְאַחַר כָּךְ ״יֵשְׁבוּ יְשָׁרִים אֶת פָּנֶיךָ״. וַחֲסִידִים הָרִאשׁוֹנִים הָיוּ שׁוֹהִין שָׁעָה מַמָּשׁ מִשָּׁעוֹת בֵּינוֹנִית, שֶׁהֵן כ״ד בְּמֵעֵת לְעֵת, וְלִשְׁאָר עַם דַּי בְּשָׁעָה מוּעֶטֶת כְּדֵי הִלּוּךְ ב׳ פְּתָחִים שֶׁהֵם ח׳ טְפָחִים, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן צ׳. יֵשׁ נוֹהֲגִין לְהִתְפַּלֵּל מִתּוֹךְ הַסֵּפֶר, כְּדֵי לְכַוֵּן מְאֹד. וְהַכֹּל לְפִי מַה שֶּׁהוּא מַרְגִּישׁ בְּנַפְשׁוֹ:

לִשְׁהוֹת קֹדֶם וְאַחַר. הַמִּתְפַּלֵּל יִשְׁהֶה שָׁעָה אַחַת קֹדֶם תְּפִלָּתוֹ כְּדֵי שֶׁיְּכַוֵּן לִבּוֹ לַמָּקוֹם (״אַשְׁרֵי יוֹשְׁבֵי בֵיתֶךָ״ וְ״יְהַלְלוּךָ סֶּלָה״), וְשָׁעָה אַחַת אַחַר תְּפִלָּתוֹ שֶׁלֹּא תְּהֵא נִרְאֵית כְּמַשְּׂאוֹי. חֲסִידִים הָרִאשׁוֹנִים שָׁעָה מַמָּשׁ (כ״ד בְּמֵעֵת לְעֵת) ; לִשְׁאָר עַם שָׁעָה מוּעֶטֶת (כְּדֵי הִלּוּךְ ב׳ פְּתָחִים, סימן צ׳), וְהַכֹּל לְפִי מַה שֶּׁמַּרְגִּישׁ בְּנַפְשׁוֹ.

סעיף ב כֹּבֶד רֹאשׁ וְהַכְנָעָה ; לֹא מִתּוֹךְ שְׂחוֹק אֶלָּא מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה.

אֵין עוֹמְדִין לְהִתְפַּלֵּל אֶלָּא מִתּוֹךְ כֹּבֶד רֹאשׁ, שֶׁהִיא הַכְנָעָה מֵאֵימַת הַקָּבָּ״ה, שֶׁנֶּאֱמַר: ״עִבְדוּ אֶת ה׳ בְּיִרְאָה״, דְּהַיְנוּ הַתְּפִלָּה, שֶׁהִיא לָנוּ בִּמְקוֹם עֲבוֹדָה — עֲשׂוּ אוֹתָהּ בְּיִרְאָה. וְלֹא יַעֲמֹד לֹא מִתּוֹךְ שְׂחוֹק וְקַלּוּת רֹאשׁ וּדְבָרִים בְּטֵלִים, וְלֹא מִתּוֹךְ כַּעַס, אֶלָּא מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה, כְּגוֹן דִּבְרֵי תַּנְחוּמִים שֶׁל תּוֹרָה, כְּגוֹן סָמוּךְ לִגְאֻלַּת מִצְרַיִם, אוֹ סָמוּךְ לִ״תְהִלָּה לְדָוִד״ שֶׁכָּתוּב בּוֹ: ״רְצוֹן יְרֵאָיו יַעֲשֶׂה … שׁוֹמֵר ה׳ אֶת כָּל אֹהֲבָיו״:

כֹּבֶד רֹאשׁ. אֵין עוֹמְדִין אֶלָּא מִתּוֹךְ כֹּבֶד רֹאשׁ — הַכְנָעָה מֵאֵימַת הַקָּבָּ״ה (״עִבְדוּ אֶת ה׳ בְּיִרְאָה״). לֹא מִתּוֹךְ שְׂחוֹק וְקַלּוּת רֹאשׁ וְלֹא מִתּוֹךְ כַּעַס, אֶלָּא מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה (דִּבְרֵי תַּנְחוּמִים : סָמוּךְ לִגְאֻלַּת מִצְרַיִם אוֹ לִ״תְהִלָּה לְדָוִד״).

סעיף ג לֹא מִתּוֹךְ דִּין וּפִלְפּוּל אֶלָּא מִתּוֹךְ הֲלָכָה פְּסוּקָה — שִׂמְחָה שֶׁל מִצְוָה.

אֵין עוֹמְדִין לְהִתְפַּלֵּל מִתּוֹךְ דִּין וְלֹא מִתּוֹךְ הֲלָכָה, שֶׁלֹּא יְהֵא לִבּוֹ טָרוּד בָּהּ בִּתְפִלָּתוֹ וְלֹא יוּכַל לְכַוֵּן יָפֶה, אֶלָּא מִתּוֹךְ הֲלָכָה פְּסוּקָה, שֶׁאֵין לְפַלְפֵּל בָּהּ כְּלוּם וְלֹא יִטָּרֵד בָּהּ בַּתְּפִלָּה, וְהִיא נִקְרֵאת גַּם כֵּן שִׂמְחָה שֶׁל מִצְוָה, שֶׁנֶּאֱמַר: ״פִּקּוּדֵי ה׳ יְשָׁרִים מְשַׂמְּחֵי לֵב״. וּמִכָּל מָקוֹם, אִם עָסַק בְּפִלְפּוּל הֲלָכָה וְהִתְחִילוּ צִבּוּר לְהִתְפַּלֵּל — יִתְפַּלֵּל עִמָּהֶם, וּבִפְרָט עַכְשָׁו שֶׁאֵין מְכַוְּנִים כָּל כָּךְ בַּתְּפִלָּה. וְלֹא הִזְהִירוּ אֶלָּא לְכַתְּחִלָּה, שֶׁלֹּא יַעֲסֹק בְּפִלְפּוּל קֹדֶם הַתְּפִלָּה:

הֲלָכָה פְּסוּקָה. לֹא מִתּוֹךְ דִּין וְלֹא מִתּוֹךְ הֲלָכָה, שֶׁלֹּא יְהֵא לִבּוֹ טָרוּד, אֶלָּא מִתּוֹךְ הֲלָכָה פְּסוּקָה שֶׁאֵין לְפַלְפֵּל בָּהּ (״פִּקּוּדֵי ה׳ יְשָׁרִים מְשַׂמְּחֵי לֵב״). וְאִם עָסַק בְּפִלְפּוּל וְהִתְחִילוּ צִבּוּר — יִתְפַּלֵּל עִמָּהֶם ; לֹא הִזְהִירוּ אֶלָּא לְכַתְּחִלָּה.

סעיף ד הָעוֹסֵק בְּצָרְכֵי צִבּוּר כְּעוֹסֵק בַּתּוֹרָה ; אֵינוֹ פּוֹסֵק, וְאֵין תַּשְׁלוּמִין.

הָעוֹסֵק בְּצָרְכֵי צִבּוּר — הֲרֵי זֶה כְּעוֹסֵק בַּתּוֹרָה. יֵשׁ מְפָרְשִׁים לְעִנְיַן לַעֲמֹד מִתּוֹכוֹ לְהִתְפַּלֵּל, שֶׁגַּם זוֹ שִׂמְחָה הִיא לוֹ שֶׁעוֹסֵק בְּצָרְכֵי צִבּוּר, וּבִלְבַד שֶׁיְּהֵא דָּבָר שֶׁאֵין בּוֹ טִרְדָּה, דּוֹמֶה לַהֲלָכָה פְּסוּקָה. וְיֵשׁ מְפָרְשִׁים לְעִנְיָן שֶׁאֵין צָרִיךְ לִפְסֹק וּלְהִתְפַּלֵּל, כְּמוֹ שֶׁהָעוֹסֵק בַּתּוֹרָה וְתוֹרָתוֹ אֻמָּנוּתוֹ — אֵין צָרִיךְ לִפְסֹק לִתְפִלָּה. וּכְבָר נִתְבָּאֵר בְּסִימָן ע׳ שֶׁהָעוֹסֵק בְּצָרְכֵי צִבּוּר אֲפִלּוּ לִקְרִיאַת שְׁמַע אֵינוֹ פּוֹסֵק, וְכָל שֶׁכֵּן לִתְפִלָּה. וְאֵין צָרִיךְ לְהִתְפַּלֵּל מִנְחָה שְׁתַּיִם בִּשְׁבִיל תַּשְׁלוּמֵי שַׁחֲרִית, שֶׁכֵּיוָן שֶׁבִּשְׁעַת חוֹבָתוֹ הָיָה פָּטוּר מִמֶּנָּה מִן הַדִּין — אֵין צָרִיךְ לְתַשְׁלוּמִין כְּלָל:

הָעוֹסֵק בְּצָרְכֵי צִבּוּר. הֲרֵי זֶה כְּעוֹסֵק בַּתּוֹרָה. יֵשׁ מְפָרְשִׁים לַעֲמֹד מִתּוֹכוֹ לְהִתְפַּלֵּל (שֶׁגַּם זוֹ שִׂמְחָה), וְיֵשׁ מְפָרְשִׁים שֶׁאֵין צָרִיךְ לִפְסֹק — כְּתוֹרָתוֹ אֻמָּנוּתוֹ. וּכְבָר נִתְבָּאֵר בְּסִימָן ע׳ שֶׁאֲפִלּוּ לִקְרִיאַת שְׁמַע אֵינוֹ פּוֹסֵק, וְאֵין צָרִיךְ תַּשְׁלוּמִין כְּלָל.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim צ״ג (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — קֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת

היסוד : קֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַשְּׁהִיָּה שָׁעָה קֹדֶם וְאַחַר הַתְּפִלָּה. אחר כך הַכֹּבֶד רֹאשׁ וְהַשִּׂמְחָה שֶׁל מִצְוָה. אחר כך הַהֲלָכָה פְּסוּקָה (וְלֹא מִתּוֹךְ דִּין וּפִלְפּוּל). ולבסוף הָעוֹסֵק בְּצָרְכֵי צִבּוּר.

לִשְׁהוֹת שָׁעָה — קֹדֶם וְאַחַר הַתְּפִלָּה

לִשְׁהוֹת שָׁעָה — קֹדֶם וְאַחַר הַתְּפִלָּה

היסוד. לִשְׁהוֹת שָׁעָה אַחַת קֹדֶם הַתְּפִלָּה כְּדֵי שֶׁיְּכַוֵּן לִבּוֹ לַמָּקוֹם, וְשָׁעָה אַחַת אַחַר, שֶׁלֹּא תְּהֵא נִרְאֵית עָלָיו כְּמַשְּׂאוֹי (סעיף א).

המסקנה. חֲסִידִים הָרִאשׁוֹנִים שָׁעָה מַמָּשׁ ; לִשְׁאָר עַם שָׁעָה מוּעֶטֶת, וְהַכֹּל לְפִי מַה שֶּׁמַּרְגִּישׁ בְּנַפְשׁוֹ. עיין סימן 90 (מקום קבוע לתפילה).

כֹּבֶד רֹאשׁ — מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה

כֹּבֶד רֹאשׁ — מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה

היסוד. אֵין עוֹמְדִין לְהִתְפַּלֵּל אֶלָּא מִתּוֹךְ כֹּבֶד רֹאשׁ — הַכְנָעָה מֵאֵימַת הַקָּבָּ״ה (״עִבְדוּ אֶת ה׳ בְּיִרְאָה״) (סעיף ב).

המסקנה. לֹא מִתּוֹךְ שְׂחוֹק וְקַלּוּת רֹאשׁ וְלֹא מִתּוֹךְ כַּעַס, אֶלָּא מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה (דִּבְרֵי תַּנְחוּמִים שֶׁל תּוֹרָה). עיין סימן 98 (כוונת התפילה).

מִתּוֹךְ הֲלָכָה פְּסוּקָה

מִתּוֹךְ הֲלָכָה פְּסוּקָה

היסוד. לֹא מִתּוֹךְ דִּין וְלֹא מִתּוֹךְ הֲלָכָה (שֶׁאֵינָהּ פְּסוּקָה), שֶׁלֹּא יְהֵא לִבּוֹ טָרוּד, אֶלָּא מִתּוֹךְ הֲלָכָה פְּסוּקָה שֶׁאֵין לְפַלְפֵּל בָּהּ (״פִּקּוּדֵי ה׳ יְשָׁרִים מְשַׂמְּחֵי לֵב״) (סעיף ג).

המסקנה. אִם עָסַק בְּפִלְפּוּל וְהִתְחִילוּ צִבּוּר — יִתְפַּלֵּל עִמָּהֶם ; לֹא הִזְהִירוּ אֶלָּא לְכַתְּחִלָּה. עיין סימן 92 (הכנה לתפילה).

הָעוֹסֵק בְּצָרְכֵי צִבּוּר

הָעוֹסֵק בְּצָרְכֵי צִבּוּר

היסוד. הָעוֹסֵק בְּצָרְכֵי צִבּוּר הֲרֵי זֶה כְּעוֹסֵק בַּתּוֹרָה : יֵשׁ מְפָרְשִׁים לַעֲמֹד מִתּוֹכוֹ לְהִתְפַּלֵּל, וְיֵשׁ מְפָרְשִׁים שֶׁאֵין צָרִיךְ לִפְסֹק (סעיף ד).

המסקנה. וּכְבָר נִתְבָּאֵר בְּסִימָן ע׳ שֶׁאֲפִלּוּ לִקְרִיאַת שְׁמַע אֵינוֹ פּוֹסֵק, וְאֵין תַּשְׁלוּמִין. עיין סימן 70 (עוסק בצרכי ציבור).

הפניה ללימוד. היקף הסימן — הַשְּׁהִיָּה, הַכֹּבֶד רֹאשׁ מִתּוֹךְ שִׂמְחָה וְהַהֲלָכָה פְּסוּקָה — נוגע להכנה היומיומית של הַתְּפִלָּה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

קֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן צ״ג. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
לִשְׁהוֹת קֹדֶם וְאַחַר הַתְּפִלָּה או״ח צ״ג — שָׁעָה קֹדֶם לְכַוֵּן, וְשָׁעָה אַחַר שֶׁלֹּא כְמַשְּׂאוֹי. מ״ב צ״ג — שִׁעוּר הַשְּׁהִיָּה וַחֲסִידִים הָרִאשׁוֹנִים. ערוה״ש צ״ג — גֶּדֶר הַשְּׁהִיָּה וְהַכַּוָּנָה קֹדֶם וְאַחַר. דרך אדמו״ר הזקן — שָׁעָה אַחַת קֹדֶם וְאַחַר ; חֲסִידִים הָרִאשׁוֹנִים שָׁעָה מַמָּשׁ, לִשְׁאָר עַם שָׁעָה מוּעֶטֶת, וְהַכֹּל לְפִי מַרְגִּישׁ בְּנַפְשׁוֹ (סעיף א).
כֹּבֶד רֹאשׁ וְשִׂמְחָה שֶׁל מִצְוָה או״ח צ״ג — בְּאֵימָה וְהַכְנָעָה, לֹא מִתּוֹךְ שְׂחוֹק וְקַלּוּת רֹאשׁ אֶלָּא מִתּוֹךְ שִׂמְחָה. מ״ב צ״ג — גֶּדֶר כֹּבֶד רֹאשׁ וְהַשִּׂמְחָה שֶׁל מִצְוָה. ערוה״ש צ״ג — הַכְנָעָה מֵאֵימַת הַשֵּׁם בַּתְּפִלָּה. דרך אדמו״ר הזקן — כֹּבֶד רֹאשׁ שֶׁהִיא הַכְנָעָה מֵאֵימַת הַקָּבָּ״ה (״עִבְדוּ אֶת ה׳ בְּיִרְאָה״) ; אֶלָּא מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה (סעיף ב).
מִתּוֹךְ הֲלָכָה פְּסוּקָה או״ח צ״ג — לֹא מִתּוֹךְ דִּין וְלֹא מִתּוֹךְ הֲלָכָה, אֶלָּא מִתּוֹךְ הֲלָכָה פְּסוּקָה. מ״ב צ״ג — טַעַם ״פִּקּוּדֵי ה׳ יְשָׁרִים מְשַׂמְּחֵי לֵב״. ערוה״ש צ״ג — גֶּדֶר הַטִּרְדָּה וְהַהֲלָכָה הַפְּסוּקָה. דרך אדמו״ר הזקן — מִתּוֹךְ הֲלָכָה פְּסוּקָה שֶׁאֵין לְפַלְפֵּל בָּהּ ; וְאִם הִתְחִילוּ צִבּוּר — יִתְפַּלֵּל עִמָּהֶם, וְלֹא הִזְהִירוּ אֶלָּא לְכַתְּחִלָּה (סעיף ג).
הָעוֹסֵק בְּצָרְכֵי צִבּוּר או״ח צ״ג — הָעוֹסֵק בְּצָרְכֵי צִבּוּר כְּעוֹסֵק בַּתּוֹרָה ; וְיֵשׁ מְפָרְשִׁים שֶׁאֵינוֹ פּוֹסֵק. מ״ב צ״ג — פְּרָטֵי הָעוֹסֵק בְּצָרְכֵי צִבּוּר וְהַתַּשְׁלוּמִין. ערוה״ש צ״ג — גֶּדֶר תּוֹרָתוֹ אֻמָּנוּתוֹ בְּצָרְכֵי צִבּוּר. דרך אדמו״ר הזקן — כְּעוֹסֵק בַּתּוֹרָה ; אֲפִלּוּ לִקְרִיאַת שְׁמַע אֵינוֹ פּוֹסֵק (סימן ע׳), וְאֵין צָרִיךְ תַּשְׁלוּמִין כְּלָל (סעיף ד).

הטבלה נערכה מטקסט המחבר (אורח חיים צ״ג, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה ד-ה, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ל׳-ל״א). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַשְּׁהִיָּה וְהַכַּוָּנָה

קו א — הַשְּׁהִיָּה וְהַכַּוָּנָה

יסוד הסימן : לִשְׁהוֹת שָׁעָה קֹדֶם וְאַחַר הַתְּפִלָּה — כְּדֵי שֶׁיְּכַוֵּן לִבּוֹ לַמָּקוֹם (סעיף א).

קריאת הרב : חֲסִידִים הָרִאשׁוֹנִים שָׁעָה מַמָּשׁ ; וְהַכֹּל לְפִי מַה שֶּׁהוּא מַרְגִּישׁ בְּנַפְשׁוֹ. עיין סימן 90.

מסקנה למעשה : לְהָכִין הַלֵּב קֹדֶם הַתְּפִלָּה וְלֹא לָצֵאת מִמֶּנָּה בִּמְהִירוּת ; הַשִּׁעוּר לְפִי מַה שֶּׁמַּרְגִּישׁ בְּנַפְשׁוֹ.

קו ב — כֹּבֶד רֹאשׁ מִתּוֹךְ שִׂמְחָה

קו ב — כֹּבֶד רֹאשׁ מִתּוֹךְ שִׂמְחָה

יסוד הסימן : אֵין עוֹמְדִין אֶלָּא מִתּוֹךְ כֹּבֶד רֹאשׁ ; לֹא מִתּוֹךְ שְׂחוֹק וְלֹא מִתּוֹךְ כַּעַס, אֶלָּא מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה, וּמִתּוֹךְ הֲלָכָה פְּסוּקָה (סעיפים ב-ג).

קריאת הרב : הַכֹּבֶד רֹאשׁ אֵינוֹ עַצְבוּת אֶלָּא הַכְנָעָה מֵאֵימַת הַקָּבָּ״ה, מְאֻחֶדֶת עִם הַשִּׂמְחָה שֶׁל מִצְוָה. עיין סימן 98.

מסקנה למעשה : לָגֶשֶׁת לַתְּפִלָּה בְּרֹצֶן רְצִינִי וְשָׂמֵחַ ; לֹא לְהִתְפַּלֵּל מִתּוֹךְ שְׂחוֹק, קַלּוּת רֹאשׁ אוֹ כַּעַס.

קו ג — הָעוֹסֵק בְּצָרְכֵי צִבּוּר

קו ג — הָעוֹסֵק בְּצָרְכֵי צִבּוּר

יסוד הסימן : הָעוֹסֵק בְּצָרְכֵי צִבּוּר כְּעוֹסֵק בַּתּוֹרָה ; יֵשׁ מְפָרְשִׁים שֶׁאֵין צָרִיךְ לִפְסֹק (סעיף ד).

קריאת הרב : אֲפִלּוּ לִקְרִיאַת שְׁמַע אֵינוֹ פּוֹסֵק (סימן ע׳), וְאֵין תַּשְׁלוּמִין. עיין סימן 70.

מסקנה למעשה : עֲבוֹדַת הַצִּבּוּר שְׁקוּלָה כְּתַלְמוּד תּוֹרָה ; מִי שֶׁהָיָה פָּטוּר מִן הַדִּין — אֵין לוֹ תַּשְׁלוּמִין.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן צ״ג ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן צ״ג (קֹדֶם שֶׁיִּתְפַּלֵּל צָרִיךְ לִשְׁהוֹת)

  • ① לִשְׁהוֹת. לִשְׁהוֹת שָׁעָה קֹדֶם הַתְּפִלָּה כְּדֵי לְכַוֵּן, וְשָׁעָה אַחַר שֶׁלֹּא כְמַשְּׂאוֹי ; הַכֹּל לְפִי מַה שֶּׁמַּרְגִּישׁ בְּנַפְשׁוֹ. יסוד : שוע״ר צ״ג סעיף א.
  • ② כֹּבֶד רֹאשׁ. לְהִתְפַּלֵּל מִתּוֹךְ כֹּבֶד רֹאשׁ — לֹא מִתּוֹךְ שְׂחוֹק וְקַלּוּת רֹאשׁ וְלֹא מִתּוֹךְ כַּעַס, אֶלָּא מִתּוֹךְ שִׂמְחָה שֶׁל מִצְוָה. יסוד : שוע״ר צ״ג סעיף ב.
  • ③ הֲלָכָה פְּסוּקָה. לֹא מִתּוֹךְ דִּין וּפִלְפּוּל אֶלָּא מִתּוֹךְ הֲלָכָה פְּסוּקָה ; וְאִם הִתְחִילוּ צִבּוּר בְּעֵת לִמּוּדוֹ — יִתְפַּלֵּל עִמָּהֶם. יסוד : שוע״ר צ״ג סעיף ג.
  • ④ צָרְכֵי צִבּוּר. הָעוֹסֵק בְּצָרְכֵי צִבּוּר כְּעוֹסֵק בַּתּוֹרָה ; אֲפִלּוּ לִקְרִיאַת שְׁמַע אֵינוֹ פּוֹסֵק, וְאֵין תַּשְׁלוּמִין. יסוד : שוע״ר צ״ג סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן צ״ג (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַשְּׁהִיָּה, הַכֹּבֶד רֹאשׁ, הַהֲלָכָה פְּסוּקָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
