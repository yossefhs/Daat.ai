דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן נ״ט · דִּין בְּרָכָה רִאשׁוֹנָה שֶׁבְּיוֹצֵר

שיטת אדמו״ר הזקן בְּדִין בְּרָכָה רִאשׁוֹנָה שֶׁבְּיוֹצֵר : נֻסַּח « יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ » וּמִדַּת לַיְלָה בַּיּוֹם (נֶגֶד הַמִּינִים), קְדֻשָּׁה שֶׁבְּיוֹצֵר, הַשְּׁלִיחַ צִבּוּר מוֹצִיא, לוֹמַר בְּנַחַת עִם הַשְּׁלִיחַ צִבּוּר, וְאוֹר חָדָשׁ — ברגישות חב״ד.

למה רמה 4 לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מכנס את שיטת הרב על סימן נ״ט — דִּין בְּרָכָה רִאשׁוֹנָה שֶׁבְּיוֹצֵר. אצל הרב הסימן נפרש בְּו׳ סְעִיפִים : נֻסַּח יוֹצֵר אוֹר וְדִינֵי הַטָּעוּת, קְדֻשָּׁה שֶׁבְּיוֹצֵר, מַעֲרִיצִים בְּמ״ם, הַשְּׁלִיחַ צִבּוּר מוֹצִיא, טָעָה וְיָרַד אַחֵר וְאוֹר חָדָשׁ.

← לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט השלם והאמיתי של שולחן ערוך הרב, אורח חיים סימן נ״ט — ו׳ סְעִיפִים מידו של הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim נ״ט). אדמו״ר הזקן דן שם בְּדִין בְּרָכָה רִאשׁוֹנָה שֶׁבְּיוֹצֵר. הביאורים הם משלנו. שום קטע אינו מובא מחוץ לטקסט המאומת הזה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן נ״ט — דִּין בְּרָכָה רִאשׁוֹנָה שֶׁבְּיוֹצֵר

הטקסט השלם של שולחן ערוך הרב

שולחן ערוך הרב, סימן נ״ט — דִּין בְּרָכָה רִאשׁוֹנָה שֶׁבְּיוֹצֵר — וּבוֹ ו׳ סְעִיפִים
לשון אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.59), ולאחריה ביאורנו. הסימן כולל שישה סעיפים (ו׳ סְעִיפִים ; סעיף k → ‎.59.k).

סעיף א נֻסַּח בְּרָכָה רִאשׁוֹנָה — יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ, וּמִדַּת לַיְלָה בַּיּוֹם ; טָעָה מַעֲרִיב עֲרָבִים.

דִּין בְּרָכָה רִאשׁוֹנָה בְּיוֹצֵר וּבוֹ ו׳ סְעִיפִים:שֶׁבַע בְּרָכוֹת תִקְּנוּ חֲכָמִים לִקְרִיאַת שְׁמַע: בַּשַּׁחַר — שְׁתַּיִם לְפָנֶיהָ וְאַחַת לְאַחֲרֶיהָ, וּבָעֶרֶב — שְׁתַּיִם לְפָנֶיהָ וּשְׁתַּיִם לְאַחֲרֶיהָ, וְסָמְכוּ אוֹתָן עַל פָּסוּק ״שֶׁבַע בַּיּוֹם הִלַּלְתִּיךָ וְגו׳״. בְּרָכָה רִאשׁוֹנָה פּוֹתַחַת בְּ״בָרוּךְ״: ״בָּרוּךְ אַתָּה ה׳ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ״. וְתִקְּנוּ לְהַזְכִּיר מִדַּת לַיְלָה בַּיּוֹם, לְהוֹצִיא מִלֵּב הַמִּינִים שֶׁאוֹמְרִים שֶׁמִּי שֶׁבָּרָא אוֹר לֹא בָּרָא חֹשֶׁךְ. וְחוֹתֶמֶת בְּ״בָרוּךְ״: ״בָּרוּךְ אַתָּה ה׳ יוֹצֵר הַמְּאוֹרוֹת״. וְאִם טָעָה וְאָמַר ״בָּרוּךְ אַתָּה ה׳ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם אֲשֶׁר בִּדְבָרוֹ מַעֲרִיב עֲרָבִים״ וְנִזְכַּר מִיָּד וְאָמַר ״יוֹצֵר אוֹר״ עַד סִיּוּם הַבְּרָכָה וְחָתַם בְּ״יוֹצֵר הַמְּאוֹרוֹת״ — יָצָא. אֲבָל אִם אָמַר ״אֲשֶׁר בִּדְבָרוֹ מַעֲרִיב עֲרָבִים״ וְלֹא אָמַר (מִיָּד) ״יוֹצֵר אוֹר״ אַף עַל פִּי שֶׁחָתַם בְּ״יוֹצֵר הַמְּאוֹרוֹת״ — לֹא יָצָא, שֶׁהֲרֵי כָּל פְּתִיחַת הַבְּרָכָה לֹא הָיְתָה כְּתִקּוּנָהּ (שֶׁלֹּא בֵּרֵךְ כְּלָל עַל יְצִירַת הָאוֹר, מַה שֶּׁאֵין כֵּן כְּשֶׁנִּזְכַּר מִיָּד וְאָמַר ״יוֹצֵר אוֹר״ — תִּקֵּן טָעוּתוֹ וּבֵרַךְ עַל יְצִירַת הָאוֹר). וְאִם אָמַר ״בָּרוּךְ אַתָּה ה׳ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ אֲשֶׁר בִּדְבָרוֹ מַעֲרִיב עֲרָבִים בְּחָכְמָה פּוֹתֵחַ״ כו׳ אֲפִלּוּ עַד סִיּוּם הַבְּרָכָה וְחָתַם בְּ״יוֹצֵר הַמְּאוֹרוֹת״ — יָצָא, שֶׁכֵּיוָן שֶׁפְּתִיחַת הַבְּרָכָה וַחֲתִימָתָהּ כְּתִקּוּנָהּ אַף עַל פִּי שֶׁבָּאֶמְצַע שִׁנָּה מִמַּטְבֵּעַ שֶׁטָּבְעוּ חֲכָמִים — יָצָא בְּדִיעֲבַד. אֲבָל אִם חָתַם ״הַמַּעֲרִיב עֲרָבִים״ אַף עַל פִּי שֶׁפָּתַח בְּ״יוֹצֵר אוֹר״ בִּלְבַד וְאָמַר כָּל הַבְּרָכָה כְּתִקּוּנָהּ עַד הַחֲתִימָה — לֹא יָצָא, שֶׁכָּל הַבְּרָכָה צְרִיכָה לִהְיוֹת פְּתִיחָתָהּ וַחֲתִימָתָהּ כְּתִקּוּנָהּ אֲפִלּוּ בְּדִיעֲבַד. וְכָל מָקוֹם שֶׁלֹּא יָצָא אִם לֹא נִזְכַּר שֶׁטָּעָה עַד שֶׁגָּמַר ״אַהֲבָה רַבָּה״ אוֹ ״אַהֲבַת עוֹלָם״ — לֹא יַפְסִיק לַחֲזֹר לְ״יוֹצֵר אוֹר״ בֵּינָהּ לִקְרִיאַת שְׁמַע, אֶלָּא יֹאמַר ״יוֹצֵר אוֹר״ אַחַר תְּפִלַּת י״ח. וְאַף עַל פִּי שֶׁמַּקְדִּים קְרִיאַת שְׁמַע לְבִרְכַּת ״יוֹצֵר״ שֶׁתִקְּנוּ לְפָנֶיהָ — אֵין זֶה מְעַכֵּב בְּדִיעֲבַד, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ס׳:

נֻסַּח וְטָעוּת. שֶׁבַע בְּרָכוֹת לַק״ש (« שֶׁבַע בַּיּוֹם הִלַּלְתִּיךָ ») ; בַּשַּׁחַר שְׁתַּיִם לְפָנֶיהָ וְאַחַת לְאַחֲרֶיהָ. הָרִאשׁוֹנָה פּוֹתַחַת בְּ« יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ », מַזְכִּיר מִדַּת לַיְלָה בַּיּוֹם לְהוֹצִיא מִלֵּב הַמִּינִים, וְחוֹתֶמֶת « יוֹצֵר הַמְּאוֹרוֹת ». טָעָה וְאָמַר « מַעֲרִיב עֲרָבִים » וְנִזְכַּר מִיָּד וְאָמַר « יוֹצֵר אוֹר » — יָצָא ; וְאִם לָאו — לֹא יָצָא. הַכֹּל תָּלוּי בַּפְּתִיחָה וּבַחֲתִימָה כְּתִקּוּנָן, אֲפִלּוּ בְּדִיעֲבַד. נִזְכַּר רַק אַחַר אַהֲבָה רַבָּה — יֹאמַר « יוֹצֵר אוֹר » אַחַר תְּפִלַּת י״ח (עיין סימן 60).

סעיף ב קְדֻשָּׁה שֶׁבְּיוֹצֵר — הַיָּחִיד אוֹמְרָהּ כְּסִפּוּר דְּבָרִים, וְהַמִּנְהָג.

יֵשׁ אוֹמְרִים שֶׁהַקְּדֻשָּׁה שֶׁבְּ״יוֹצֵר״ יָכוֹל הַיָּחִיד לְאָמְרָהּ, שֶׁאֵינָהּ דּוֹמָה לַקְּדֻשָּׁה שֶׁבַּתְּפִלָּה שֶׁאוֹמְרִים ״נַקְדִּישָׁךְ״ אוֹ ״נְקַדֵּשׁ״ שֶׁאֲנַחְנוּ מַקְדִּישִׁים אוֹתוֹ וְאֵין רַשַּׁאי לְהַקְדִּישׁ בְּפָחוֹת מֵעֲשָׂרָה, אֲבָל קְדֻשָּׁה שֶׁל ״יוֹצֵר״ שֶׁאֵין אָנוּ מַקְדִּישִׁין אוֹתוֹ אֶלָּא מַזְכִּירִין הַקְּדֻשָּׁה שֶׁאוֹמְרִים לוֹ מְשָׁרְתָיו — מֻתָּר בְּיָחִיד, שֶׁאֵינָהּ אֶלָּא כְּמוֹ סִפּוּר דְּבָרִים. וְכֵן קְדֻשַּׁת ״וּבָא לְצִיּוֹן״ כֵּיוָן שֶׁאֵינָהּ אֶלָּא קְרִיאַת פְּסוּקִים בְּעָלְמָא — אֵין בְּכָךְ כְּלוּם. וְיֵשׁ אוֹמְרִים שֶׁאֵין הַיָּחִיד אוֹמֵר לֹא קְדֻשַּׁת ״יוֹצֵר״ וְלֹא קְדֻשַּׁת ״וּבָא לְצִיּוֹן״, אֶלָּא כְּשֶׁמַּגִּיעַ לַקְּדֻשָּׁה שֶׁבְּ״יוֹצֵר״ — יֹאמַר ״וְאוֹמְרִים בְּיִרְאָה קָדוֹשׁ וְהָאוֹפַנִּים כו׳ וְאוֹמְרִים בָּרוּךְ״. וְטוֹב לָחוּשׁ לְדִבְרֵיהֶם וְלִזָּהֵר שֶׁיָּחִיד יֹאמְרֶנָה בְּנִגּוּן וּטְעָמִים כְּקוֹרֵא בַּתּוֹרָה. אֲבָל הַמִּנְהָג כַּסְּבָרָא הָרִאשׁוֹנָה. וּכְשֶׁעוֹנִין קְדֻשָּׁה זוֹ בְּצִבּוּר — יֵשׁ לַעֲנוֹת בְּקוֹל רָם, וְיֵשׁ לוֹמַר אוֹתָהּ מְיֻשָּׁב. וְלֹא שֶׁיֵּשֵׁב אִם הָיָה עוֹמֵד, אֶלָּא שֶׁאִם הָיָה יוֹשֵׁב — אֵין לוֹ לַעֲמֹד לִקְדֻשָּׁה זוֹ כְּמוֹ לִקְדֻשָּׁה שֶׁבַּתְּפִלָּה:

קְדֻשָּׁה שֶׁבְּיוֹצֵר. יֵשׁ אוֹמְרִים שֶׁהַיָּחִיד אוֹמְרָהּ, שֶׁאֵינָהּ כִּקְדֻשַּׁת הַתְּפִלָּה (« נַקְדִּישָׁךְ ») הָאֲסוּרָה בְּפָחוֹת מֵעֲשָׂרָה, אֶלָּא כְּסִפּוּר דְּבָרִים (מַזְכִּיר הַקְּדֻשָּׁה שֶׁאוֹמְרִים מְשָׁרְתָיו) ; וְכֵן « וּבָא לְצִיּוֹן ». וְיֵשׁ אוֹמְרִים שֶׁלֹּא יֹאמְרֶנָּה, אֶלָּא « וְאוֹמְרִים בְּיִרְאָה קָדוֹשׁ ». וְטוֹב לְאָמְרָהּ בְּנִגּוּן וּטְעָמִים כְּקוֹרֵא בַּתּוֹרָה ; אֲבָל הַמִּנְהָג כַּסְּבָרָא הָרִאשׁוֹנָה. בְּצִבּוּר : בְּקוֹל רָם וּמְיֻשָּׁב (עיין סימן 66).

סעיף ג מַעֲרִיצִים וּמַקְדִּישִׁים וּמַמְלִיכִים — הַכֹּל בְּמ״ם וְלֹא בְּנו״ן.

יֵשׁ לוֹמַר ״מַעֲרִיצִים וּמַקְדִּישִׁים וּמַמְלִיכִים״— הַכֹּל בְּמ״ם וְלֹא בְּנו״ן, כִּי כָּל נֻסַּח הַבְּרָכָה הוּא לְשׁוֹן מִקְרָא וּלְשׁוֹן מִקְרָא הוּא בְּמ״ם וְלֹא בְּנו״ן (וְיֵשׁ נוֹהֲגִין מִטַּעַם הַיָּדוּעַ לָהֶם לוֹמַר בְּנו״ן):

בְּמ״ם וְלֹא בְּנו״ן. יֵשׁ לוֹמַר « מַעֲרִיצִים וּמַקְדִּישִׁים וּמַמְלִיכִים » — הַכֹּל בְּמ״ם, כִּי כָּל נֻסַּח הַבְּרָכָה לְשׁוֹן מִקְרָא, וּלְשׁוֹן מִקְרָא בְּמ״ם וְלֹא בְּנו״ן. וְיֵשׁ נוֹהֲגִין, מִטַּעַם הַיָּדוּעַ לָהֶם, לוֹמַר בְּנו״ן.

סעיף ד הַשְּׁלִיחַ צִבּוּר מוֹצִיא ; לוֹמַר בְּנַחַת עִם הַשְּׁלִיחַ צִבּוּר ; אֵין לַעֲנוֹת אָמֵן אַחַר הַבּוֹחֵר.

בְּרָכוֹת אֵלּוּ שֶׁל קְרִיאַת שְׁמַע — הַשְּׁלִיחַ צִבּוּר מוֹצִיא אֶת הָרַבִּים יְדֵי חוֹבָתָן אֲפִלּוּ מִי שֶׁהוּא בָּקִי. וְאֵינָן דּוֹמוֹת לִתְפִלָּה שֶׁאֵין הַשְּׁלִיחַ צִבּוּר מוֹצִיא אֶלָּא מִי שֶׁאֵינוֹ בָּקִי, לְפִי שֶׁהַתְּפִלָּה בַּקָּשַׁת רַחֲמִים הִיא וְצָרִיךְ כָּל אֶחָד לְבַקֵּשׁ רַחֲמִים עַל עַצְמוֹ אִם יוֹדֵעַ, מַה שֶּׁאֵין כֵּן בִּבְרָכוֹת אֵלּוּ וְכַיּוֹצֵא בָּהֶן. וּמִכָּל מָקוֹם, אֵינוֹ מוֹצִיא אֶלָּא שְׁלִיחַ צִבּוּר אֶת הַצִּבּוּר, דְּהַיְנוּ כְּשֶׁיֵּשׁ שָׁם ט׳ הַשּׁוֹמְעִים וְעוֹנִין אָמֵן. אֲבָל כְּשֶׁאֵין ט׳ שׁוֹמְעִין — יָחִיד הוּא, וְאֵין יָחִיד מוֹצִיא אֶת הַיָּחִיד אֲפִלּוּ מִי שֶׁאֵינוֹ בָּקִי אֶלָּא בְּבִרְכַּת הַנֶּהֱנִין וּבִרְכַּת הַמִּצְווֹת וְכַיּוֹצֵא בָּהֶן מִשְּׁאָר כָּל הַבְּרָכוֹת. חוּץ מִבִּרְכַּת קְרִיאַת שְׁמַע וְכָל בִּרְכוֹת הַשַּׁחַר שֶׁהֵן בִּרְכוֹת שֶׁבַח וְהוֹדָאָה, וְנִתַּקְנוּ לְאָמְרָם בְּפִיו, אוֹ שֶׁיִּשְׁמַע אוֹתָן בְּרָב עָם שֶׁהִיא הַדְרַת מֶלֶךְ כְּשֶׁט׳ שׁוֹמְעִין מֵאֶחָד וְעוֹנִין אַחֲרָיו אָמֵן. וּבִרְכַּת הַמָּזוֹן אַף עַל פִּי שֶׁהוּא גַּם כֵּן בִּרְכַּת הוֹדָאָה עַל הֶעָבָר, מִכָּל מָקוֹם כֵּיוָן שֶׁהוּא בִּרְכַּת הַנֶּהֱנִין — הֵקֵלּוּ בָּהֶן. וּמִכָּל מָקוֹם הוֹאִיל וּבִרְכוֹת קְרִיאַת שְׁמַע הֵן אֲרֻכּוֹת — אֵין לִסְמֹךְ עַל שְׁמִיעָתָן מֵהַשְּׁלִיחַ צִבּוּר אֲפִלּוּ בְּצִבּוּר, כִּי שֶׁמָּא יִפְנֶה לְבָבוֹ בְּאֶמְצַע הַבְּרָכָה לִדְבָרִים אֲחֵרִים וְלֹא יִשְׁמַע מֵהַשְּׁלִיחַ צִבּוּר. לָכֵן יֵשׁ לוֹמַר כָּל הַבְּרָכָה בְּנַחַת עִם הַשְּׁלִיחַ צִבּוּר, וְאָז אַף עַל פִּי שֶיִפְנֶה לְבָבוֹ (וְיִקְרָא מִקְצָתָהּ בְּלֹא כַּוָּנָה) — יָצָא (שֶׁהַכַּוָּנָה אֵינָהּ מְעַכֶּבֶת בְּדִיעֲבַד אֶלָּא בְּפָסוּק רִאשׁוֹן שֶׁל קְרִיאַת שְׁמַע וּבִבְרָכָה רִאשׁוֹנָה שֶׁל תְּפִלַּת י״ח). וְיִזָּהֵר לְמַהֵר לְסַיֵּם הַבְּרָכָה קֹדֶם שֶׁיְּסַיֵּם הַשְּׁלִיחַ צִבּוּר כְּדֵי שֶׁיַּעֲנֶה אָמֵן אַחֲרָיו, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן נ״א. וְיֵשׁ אוֹמְרִים שֶׁאַחַר סִיּוּם בִּרְכַּת ״הַבּוֹחֵר בְּעַמּוֹ יִשְׂרָאֵל בְּאַהֲבָה״ — לֹא יַעֲנֶה אָמֵן, שֶׁלֹּא לְהַפְסִיק בֵּין קְרִיאַת שְׁמַע לַבְּרָכָה שֶׁלְּפָנֶיהָ, כְּמוֹ שֶׁאָסוּר לְהַפְסִיק בֵּין כָּל דְּבַר מִצְוָה אוֹ דְּבַר הֲנָאָה שֶׁמְּבָרְכִין עָלָיו לְהַבְּרָכָה שֶׁלְּפָנָיו. וְיֵשׁ אוֹמְרִים שֶׁעוֹנִין אָמֵן גַּם אַחַר בְּרָכָה זוֹ. וְאֵין חֲשָׁשׁ בְּמַה שֶּׁמַּפְסִיק בֵּינָהּ לִקְרִיאַת שְׁמַע, לְפִי שֶׁבִּרְכוֹת קְרִיאַת שְׁמַע אֵינָן דּוֹמוֹת לִשְׁאָר בִּרְכוֹת הַמִצְווֹת, שֶׁהֲרֵי אֵין מְבָרְכִין ״אֲשֶׁר קִדְּשָׁנוּ בְּמִצְווֹתָיו וְצִוָּנוּ לִקְרוֹת קְרִיאַת שְׁמַע״, אֶלָּא בְּרָכוֹת בִּפְנֵי עַצְמָן הֵן, וְתִקְּנוּ לְבָרֵךְ אוֹתָן קֹדֶם קְרִיאַת שְׁמַע. וְכֵן הַמִּנְהָג פָּשׁוּט בִּמְדִינוֹת אֵלּוּ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ס״א:

הַשְּׁלִיחַ צִבּוּר מוֹצִיא / בְּנַחַת. בְּבִרְכוֹת ק״ש הַשְּׁלִיחַ צִבּוּר מוֹצִיא אֲפִלּוּ הַבָּקִי (שֶׁלֹּא כִּתְפִלָּה שֶׁהִיא בַּקָּשַׁת רַחֲמִים). וּמִכָּל מָקוֹם רַק בְּצִבּוּר (ט׳ שׁוֹמְעִין וְעוֹנִין אָמֵן) : אֵין יָחִיד מוֹצִיא יָחִיד בְּבִרְכוֹת שֶׁבַח אֵלּוּ. וּלְפִי שֶׁהֵן אֲרֻכּוֹת אֵין לִסְמֹךְ עַל הַשְּׁמִיעָה, אֶלָּא יֹאמַר הַכֹּל בְּנַחַת עִם הַשְּׁלִיחַ צִבּוּר, וְיַקְדִּים לְסַיֵּם כְּדֵי לַעֲנוֹת אָמֵן (עיין סימן 47). יֵשׁ אוֹמְרִים שֶׁאַחַר « הַבּוֹחֵר בְּעַמּוֹ יִשְׂרָאֵל בְּאַהֲבָה » לֹא יַעֲנֶה אָמֵן (הֶפְסֵק) ; וְיֵשׁ אוֹמְרִים שֶׁעוֹנִין, וְכֵן הַמִּנְהָג — עיין סימן 61.

סעיף ה טָעָה הַשְּׁלִיחַ צִבּוּר וְיָרַד אַחֵר — מִמָּקוֹם שֶׁפָּסַק אוֹ מֵרֹאשׁ הַבְּרָכָה.

אִם טָעָה הַשְּׁלִיחַ צִבּוּר בְּבִרְכַּת ״יוֹצֵר״ בְּעִנְיָן שֶׁצָּרִיךְ לַעֲמֹד אַחֵר תַּחְתָּיו, אִם טָעָה קֹדֶם קְדֻשָּׁה — הֲרֵי זֶה הָעוֹמֵד תַּחְתָּיו צָרִיךְ לַחֲזֹר לְרֹאשׁ הַבְּרָכָה, כְּמוֹ בְּטוֹעֶה בְּבִרְכַּת הַתְּפִלָּה וּכְמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ו. וְאִם טָעָה לְאַחַר שֶׁעָנוּ הַצִּבּוּר ״קָדוֹשׁ כו׳״ — לֹא יַתְחִיל מֵרֹאשׁ הַבְּרָכָה, מִפְּנֵי שֶׁעֲנִיַּת קָדוֹשׁ חֲשׁוּבָה כְּסִיּוּם בְּרָכָה לְעִנְיָן זֶה, וּמִ״וְהָאוֹפַנִּים״ הוּא כְּמוֹ רֹאשׁ בְּרָכָה אַחֶרֶת, וְכֵן עֲנִיַּת ״בָּרוּךְ כְּבוֹד כו׳״ הוּא כְּמוֹ סִיּוּם בְּרָכָה לְעִנְיָן זֶה, וְ״לָאֵל בָּרוּךְ״ הוּא כְּמוֹ רֹאשׁ בְּרָכָה אַחֶרֶת. לְפִיכָךְ, אִם טָעָה בְּ״אוֹפַנִּים״ — יַחֲזֹר וְיַתְחִיל זֶה שֶׁתַּחְתָּיו מִ״וְהָאוֹפַנִּים״. וְאִם טָעָה מִ״לָאֵל בָּרוּךְ נְעִימוֹת״ וְאֵילָךָ — יַחֲזֹר וְיַתְחִיל ״לָאֵל בָּרוּךְ כו׳״:

טָעָה הַשְּׁלִיחַ צִבּוּר וְיָרַד אַחֵר. טָעָה קֹדֶם קְדֻשָּׁה — הָעוֹמֵד תַּחְתָּיו חוֹזֵר לְרֹאשׁ הַבְּרָכָה (כְּטוֹעֶה בַּתְּפִלָּה, סי׳ קכ״ו ; עיין סימן 61). טָעָה אַחַר שֶׁעָנוּ « קָדוֹשׁ » — עֲנִיַּת קָדוֹשׁ חֲשׁוּבָה כְּסִיּוּם בְּרָכָה, וּ« וְהָאוֹפַנִּים » כְּרֹאשׁ בְּרָכָה אַחֶרֶת : טָעָה בְּ« אוֹפַנִּים » → מַתְחִיל « וְהָאוֹפַנִּים » ; מִ« לָאֵל בָּרוּךְ » וְאֵילָךְ → מַתְחִיל « לָאֵל בָּרוּךְ ».

סעיף ו אוֹר חָדָשׁ — יֵשׁ אוֹמְרִים שֶׁאֵין לוֹמַר, וְיֵשׁ מְיַשְּׁבִים הַמִּנְהָג.

יֵשׁ אוֹמְרִים שֶׁאֵין לוֹמַר ״אוֹר חָדָשׁ כו׳״, לְפִי שֶׁצָּרִיךְ לוֹמַר מֵעֵין פְּתִיחָה וַחֲתִימָה סָמוּךְ לַחֲתִימָה, וְזֶה אֵינוֹ מֵעֵין פְּתִיחָה וַחֲתִימָה, לְפִי שֶׁחֲכָמִים סִדְּרוּ הַבְּרָכָה עַל אוֹר הַמִּתְחַדֵּשׁ בְּכָל יוֹם וְלֹא עַל אוֹר הֶעָתִיד לְהִתְחַדֵּשׁ. וְיֵשׁ מְיַשְּׁבִים הַמִּנְהָג לוֹמַר שֶׁהִיא מֵעֵין פְּתִיחָה וַחֲתִימָה, לְפִי שֶׁ״יוֹצֵר אוֹר״ זֶהוּ הָאוֹר שֶׁבָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא בְּו׳ יְמֵי בְּרֵאשִׁית וּגְנָזוֹ לַצַּדִּיקִים לֶעָתִיד לָבוֹא, וְעַל אוֹר זֶה נֶאֱמַר: ״וְהָלְכוּ גוֹיִם לְאוֹרֵךְ״, וְזֶהוּ שֶׁאוֹמֵר ״אוֹר חָדָשׁ״ שֶׁחִדֵּשׁ בְּו׳ יְמֵי בְּרֵאשִׁית — עָתִיד הַקָּדוֹשׁ בָּרוּךְ הוּא לְחַדֵּשׁ לָנוּ:

אוֹר חָדָשׁ. יֵשׁ אוֹמְרִים שֶׁאֵין לוֹמַר « אוֹר חָדָשׁ », שֶׁצָּרִיךְ מֵעֵין פְּתִיחָה וַחֲתִימָה סָמוּךְ לַחֲתִימָה, וְזֶה עַל אוֹר הֶעָתִיד וְלֹא עַל אוֹר הַמִּתְחַדֵּשׁ בְּכָל יוֹם. וְיֵשׁ מְיַשְּׁבִים הַמִּנְהָג : « יוֹצֵר אוֹר » הוּא הָאוֹר שֶׁבָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא בְּו׳ יְמֵי בְּרֵאשִׁית וּגְנָזוֹ לַצַּדִּיקִים לֶעָתִיד לָבוֹא (« וְהָלְכוּ גוֹיִם לְאוֹרֵךְ ») — וְזֶהוּ « אוֹר חָדָשׁ » שֶׁעָתִיד לְחַדֵּשׁ.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim נ״ט (מהדורת קה״ת, ו׳ סְעִיפִים). ביאור : DAAT. שום קטע אינו מובא מחוץ לטקסט המאומת הזה.

2 · היסוד

הַיְסוֹד — בְּרָכָה רִאשׁוֹנָה שֶׁבְּיוֹצֵר

היסוד : בְּרָכָה רִאשׁוֹנָה שֶׁבְּיוֹצֵר

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים : נֻסַּח יוֹצֵר אוֹר וּמִדַּת לַיְלָה בַּיּוֹם ; קְדֻשָּׁה שֶׁבְּיוֹצֵר ; הַשְּׁלִיחַ צִבּוּר מוֹצִיא וְלוֹמַר בְּנַחַת ; וְלבסוף טָעָה וְיָרַד אַחֵר וְאוֹר חָדָשׁ.

נֻסַּח יוֹצֵר אוֹר וּמִדַּת לַיְלָה בַּיּוֹם

נֻסַּח יוֹצֵר אוֹר וּמִדַּת לַיְלָה

העיקרון. « יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ » ; מַזְכִּיר מִדַּת לַיְלָה בַּיּוֹם לְהוֹצִיא מִלֵּב הַמִּינִים ; חֲתִימָה « יוֹצֵר הַמְּאוֹרוֹת » (סעיף א).

המסקנה. הַכֹּל תָּלוּי בַּפְּתִיחָה וּבַחֲתִימָה כְּתִקּוּנָן, אֲפִלּוּ בְּדִיעֲבַד. עיין סימן 60.

קְדֻשָּׁה שֶׁבְּיוֹצֵר

קְדֻשָּׁה שֶׁבְּיוֹצֵר

העיקרון. יֵשׁ אוֹמְרִים שֶׁהַיָּחִיד אוֹמְרָהּ (כְּסִפּוּר דְּבָרִים) ; וְיֵשׁ אוֹמְרִים « וְאוֹמְרִים בְּיִרְאָה » (סעיף ב).

המסקנה. הַמִּנְהָג : הַיָּחִיד אוֹמְרָהּ ; בְּצִבּוּר בְּקוֹל רָם וּמְיֻשָּׁב. עיין סימן 66.

הַשְּׁלִיחַ צִבּוּר מוֹצִיא וְעִם הַשְּׁלִיחַ צִבּוּר בְּנַחַת

הַשְּׁלִיחַ צִבּוּר מוֹצִיא וְעִם הַשְּׁלִיחַ צִבּוּר בְּנַחַת

העיקרון. הַשְּׁלִיחַ צִבּוּר מוֹצִיא אֲפִלּוּ בָּקִי, וְרַק בְּצִבּוּר (ט׳) ; לוֹמַר בְּנַחַת עִם הַשְּׁלִיחַ צִבּוּר (סעיפים ג-ד).

המסקנה. אֵין לַעֲנוֹת אָמֵן אַחַר « הַבּוֹחֵר » כַּסְּבָרָא הָרִאשׁוֹנָה ; מַעֲרִיצִים בְּמ״ם. עיין סימן 61.

טָעָה וְיָרַד אַחֵר וְאוֹר חָדָשׁ

טָעָה וְיָרַד אַחֵר וְאוֹר חָדָשׁ

העיקרון. קֹדֶם קְדֻשָּׁה — מֵרֹאשׁ הַבְּרָכָה ; אַחֲרֶיהָ — מִמָּקוֹם שֶׁפָּסַק (סעיף ה) ; אוֹר חָדָשׁ (סעיף ו).

המסקנה. « וְהָאוֹפַנִּים » / « לָאֵל בָּרוּךְ » כְּרֹאשׁ בְּרָכָה ; « אוֹר חָדָשׁ » — יֵשׁ מְיַשְּׁבִים. עיין סימן 47.

הפניה ללימוד. היקף הסימן — נֻסַּח הַבְּרָכָה הָרִאשׁוֹנָה, קְדֻשָּׁה שֶׁבְּיוֹצֵר, הַשְּׁלִיחַ צִבּוּר מוֹצִיא וְאוֹר חָדָשׁ — נוגע להנהגה היומיומית של ברכות ק״ש. עמוד זה מציג אותו ברמת העיקרון המאומת בלשון הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית — ללמוד עם רב חב״די ובתוך המקורות עצמם.

3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של פסק אדמו״ר הזקן

השוואת שיטת אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) והערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן נ״ט. שורת שולחן ערוך הרב מודגשת — היא מציגה את רגישותו הייחודית של הרב כפי שהיא עולה מששת סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן שיטת אדמו״ר הזקן (שולחן ערוך הרב)
נֻסַּח בְּרָכָה רִאשׁוֹנָה וּמִדַּת לַיְלָה בַּיּוֹם או״ח נ״ט — « יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ », מַזְכִּיר מִדַּת לַיְלָה בַּיּוֹם, חוֹתֵם « יוֹצֵר הַמְּאוֹרוֹת ». מ״ב נ״ט — דִּינֵי טָעָה בֵּין יוֹצֵר לְמַעֲרִיב, פְּתִיחָה וַחֲתִימָה. ערוה״ש נ״ט — בֵּאוּר הַנֻּסַּח וְטַעַם מִדַּת לַיְלָה נֶגֶד הַמִּינִים. שיטת אדמו״ר הזקן — מֵבִיא כָּל דִּינֵי הַטָּעוּת ; הַכֹּל תָּלוּי בַּפְּתִיחָה וּבַחֲתִימָה כְּתִקּוּנָן אֲפִלּוּ בְּדִיעֲבַד (סעיף א).
קְדֻשָּׁה שֶׁבְּיוֹצֵר לַיָּחִיד או״ח נ״ט — הַיָּחִיד אוֹמֵר קְדֻשַּׁת יוֹצֵר, שֶׁהִיא כְּסִפּוּר דְּבָרִים. מ״ב נ״ט — יֹאמְרֶנָּה בְּנִגּוּן וּטְעָמִים ; פְּרָטֵי הַמִּנְהָג. ערוה״ש נ״ט — עִנְיַן הַקְּדֻשָּׁה שֶׁבְּיוֹצֵר וְהַהֶכְרֵעַ. שיטת אדמו״ר הזקן — יֵשׁ אוֹמְרִים וְיֵשׁ אוֹמְרִים ; וְהַמִּנְהָג כַּסְּבָרָא הָרִאשׁוֹנָה, בְּצִבּוּר בְּקוֹל רָם וּמְיֻשָּׁב (סעיף ב).
הַשְּׁלִיחַ צִבּוּר מוֹצִיא וַעֲנִיַּת אָמֵן או״ח נ״ט — הַשְּׁלִיחַ צִבּוּר מוֹצִיא ; לוֹמַר עִם הַשְּׁלִיחַ צִבּוּר בְּנַחַת. מ״ב נ״ט — מַחֲלֹקֶת עֲנִיַּת אָמֵן אַחַר הַבּוֹחֵר. ערוה״ש נ״ט — דִּין הַהֶפְסֵק בֵּין גְּאֻלָּה לִק״ש. שיטת אדמו״ר הזקן — מוֹצִיא אֲפִלּוּ בָּקִי, וְרַק בְּצִבּוּר בְּעֲשָׂרָה ; לוֹמַר הַכֹּל בְּנַחַת עִם הַשְּׁלִיחַ צִבּוּר (סעיף ד).
טָעָה וְיָרַד אַחֵר וְאוֹר חָדָשׁ או״ח נ״ט — טָעָה קֹדֶם קְדֻשָּׁה חוֹזֵר לָרֹאשׁ ; אַחֲרֶיהָ מִמָּקוֹם שֶׁפָּסַק. מ״ב נ״ט — פְּרָטֵי « וְהָאוֹפַנִּים » וְ« לָאֵל בָּרוּךְ ». ערוה״ש נ״ט — דִּין הָאוֹמֵר « אוֹר חָדָשׁ ». שיטת אדמו״ר הזקן — מֵרֹאשׁ אוֹ מִמָּקוֹם שֶׁפָּסַק ; וּבְ« אוֹר חָדָשׁ » — יֵשׁ אוֹמְרִים וְיֵשׁ מְיַשְּׁבִים (סעיפים ה-ו).

הטבלה נערכה מלשון המחבר (אורח חיים נ״ט, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע א, רא״ש, הר״י, רוקח, הגהות מיימוניות, אבודרהם, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות י״א-י״ב, כ״א). טור אדמו״ר הזקן נשען על הטקסט האמיתי של שולחן ערוך הרב (ו׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלושה כיוונים המאפיינים את רגישות אדמו״ר הזקן בסימן זה. כל אחד לפי המבנה : עיקרון הסימן ← קריאת הרב ← מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — נֻסַּח הַבְּרָכָה וְהַטָּעוּת

נֻסַּח הַבְּרָכָה וְהַטָּעוּת

עיקרון הסימן : « יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ », מִדַּת לַיְלָה בַּיּוֹם, חֲתִימָה « יוֹצֵר הַמְּאוֹרוֹת » (סעיף א).

קריאת הרב : דִּינֵי הַטָּעוּת (יוֹצֵר / מַעֲרִיב) תְּלוּיִים בַּפְּתִיחָה וּבַחֲתִימָה כְּתִקּוּנָן. עיין סימן 60.

מסקנה למעשה : להקפיד על פתיחת הברכה וחתימתה.

קו ב — קְדֻשָּׁה שֶׁבְּיוֹצֵר וְהַשְּׁלִיחַ צִבּוּר מוֹצִיא

קְדֻשָּׁה שֶׁבְּיוֹצֵר וְהַשְּׁלִיחַ צִבּוּר מוֹצִיא

עיקרון הסימן : קְדֻשָּׁה שֶׁבְּיוֹצֵר (כְּסִפּוּר דְּבָרִים) ; הַשְּׁלִיחַ צִבּוּר מוֹצִיא בְּעֲשָׂרָה (סעיפים ב-ד).

קריאת הרב : הַיָּחִיד אוֹמְרָהּ (הַמִּנְהָג) ; לוֹמַר בְּנַחַת עִם הַשְּׁלִיחַ צִבּוּר. עיין סימן 47.

מסקנה למעשה : לומר בעצמו את הברכה, בנחת, עם השליח ציבור.

קו ג — טָעָה וְיָרַד אַחֵר וְאוֹר חָדָשׁ

טָעָה וְיָרַד אַחֵר וְאוֹר חָדָשׁ

עיקרון הסימן : קֹדֶם קְדֻשָּׁה מֵרֹאשׁ ; אַחֲרֶיהָ מִמָּקוֹם שֶׁפָּסַק ; אוֹר חָדָשׁ (סעיפים ה-ו).

קריאת הרב : עֲנִיַּת קָדוֹשׁ כְּסִיּוּם בְּרָכָה ; « אוֹר חָדָשׁ » — יֵשׁ מְיַשְּׁבִים. עיין סימן 61.

מסקנה למעשה : לנהוג כמנהג ב« אוֹר חָדָשׁ » ובחזרת השליח ציבור.

5 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה החב״דית למעשה

נקודות הנהגה הנובעות ישירות מסימן נ״ט ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן נ״ט (דִּין בְּרָכָה רִאשׁוֹנָה שֶׁבְּיוֹצֵר)

  • ① נֻסַּח. לומר « יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ », להזכיר מִדַּת לַיְלָה בַּיּוֹם, לחתום « יוֹצֵר הַמְּאוֹרוֹת ». יסוד : שוע״ר נ״ט סעיף א.
  • ② קְדֻשָּׁה שֶׁבְּיוֹצֵר. היחיד אומרה (הַמִּנְהָג) ; בְּצִבּוּר בְּקוֹל רָם וּמְיֻשָּׁב. יסוד : שוע״ר נ״ט סעיף ב.
  • ③ עִם הַשְּׁלִיחַ צִבּוּר. לומר את הברכות בְּנַחַת עִם הַשְּׁלִיחַ צִבּוּר, ולסיים לפניו כדי לענות אמן ; מַעֲרִיצִים בְּמ״ם. יסוד : שוע״ר נ״ט סעיפים ג-ד.
  • ④ הַשְּׁלִיחַ צִבּוּר מוֹצִיא. מוֹצִיא אפילו בקי, אך רק בְּצִבּוּר (ט׳ שׁוֹמְעִין) ; אין לענות אמן אחר « הַבּוֹחֵר » כסברא הראשונה. יסוד : שוע״ר נ״ט סעיף ד.
  • ⑤ טָעָה וְיָרַד אַחֵר. קֹדֶם קְדֻשָּׁה — מֵרֹאשׁ הַבְּרָכָה ; אַחֲרֶיהָ — מִמָּקוֹם שֶׁפָּסַק (« וְהָאוֹפַנִּים » / « לָאֵל בָּרוּךְ »). יסוד : שוע״ר נ״ט סעיף ה.
  • ⑥ אוֹר חָדָשׁ. ב« אוֹר חָדָשׁ » — יֵשׁ אוֹמְרִים וְיֵשׁ מְיַשְּׁבִים ; לנהוג כמנהג. יסוד : שוע״ר נ״ט סעיף ו.

⚠ חלק זה מציג את ההנהגה החב״דית המבוססת על שולחן ערוך הרב, סימן נ״ט (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב במקרה פרטי. לכל שאלה מעשית — הַנֻּסַּח, הַקְּדֻשָּׁה שֶׁבְּיוֹצֵר, הַשְּׁלִיחַ צִבּוּר, אוֹר חָדָשׁ — יש לשאול את רבך, ולחב״ד רב חב״די.
