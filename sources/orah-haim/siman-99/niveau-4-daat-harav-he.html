דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן צ״ט · דִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה

דרכו של אדמו״ר הזקן בדִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה : שָׁתָה רְבִיעִית — אַל יִתְפַּלֵּל ; יָכוֹל לְדַבֵּר לִפְנֵי הַמֶּלֶךְ — שָׁתוּי, וּתְפִלָּתוֹ תְּפִלָּה ; אֵינוֹ יָכוֹל — שִׁכּוֹר, וּתְפִלָּתוֹ תּוֹעֵבָה ; שֵׁנָה וְדֶרֶךְ מְפִיגִין ; וְהַיֵּינוֹת שֶׁלָּנוּ אֵינָם חֲזָקִים — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן צ״ט — דִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה. אצל הרב הסימן נפרש לד׳ סְעִיפִים : הַהֶבְדֵּל שֶׁבֵּין שָׁתוּי לְשִׁכּוֹר לְפִי שְׁעוּר רְבִיעִית וְ״לְדַבֵּר לִפְנֵי הַמֶּלֶךְ״, הַיַּיִן שֶׁבְּתוֹךְ הַסְּעוּדָה, מְפִיגֵי הַיַּיִן (שֵׁנָה וְדֶרֶךְ), וּמַעֲמַד הַיֵּינוֹת שֶׁלָּנוּ עִם הַסְּמִיכָה עַל הַהַרְגָּשָׁה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן צ״ט — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim צ״ט). אדמו״ר הזקן דן בו בְּדִין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן צ״ט — דִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן צ״ט — דִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.99), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.99.k).

סעיף א שָׁתָה רְבִיעִית — אַל יִתְפַּלֵּל ; שָׁתוּי לְעֻמַּת שִׁכּוֹר וּתְפִלָּתוֹ תּוֹעֵבָה.

דִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה וּבוֹ ד׳ סְעִיפִים:שָׁתָה רְבִיעִית יַיִן בְּבַת אַחַת — אַל יִתְפַּלֵּל עַד שֶׁיָּסִיר יֵינוֹ מֵעָלָיו אַף עַל פִּי שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה. וְאִם שָׁתָה יוֹתֵר, אִם הוּא יָכוֹל לְדַבֵּר לִפְנֵי הַמֶּלֶךְ — אֵינוֹ נִקְרָא שִׁכּוֹר אֶלָּא שָׁתוּי וּתְפִלָּתוֹ תְּפִלָּה אִם עָבַר וְהִתְפַּלֵּל, וְאֵין צָרִיךְ לַחֲזֹר וּלְהִתְפַּלֵּל כְּשֶׁיָּפוּג יֵינוֹ אַף עַל פִּי שֶׁלֹּא עָבַר עֲדַיִן זְמַן תְּפִלָּה. וְאִם אֵינוֹ יָכוֹל לְדַבֵּר לִפְנֵי הַמֶּלֶךְ — נִקְרָא שִׁכּוֹר. וְאִם עָבַר וְהִתְפַּלֵּל — תְּפִלָּתוֹ תּוֹעֵבָה וּכְאִלּוּ עוֹבֵד עֲבוֹדָה זָרָה, שֶׁנֶּאֱמַר: ״וַיַּחְשְׁבֶהָ עֵלִי לְשִׁכֹּרָה וגו׳ וַתֹּאמֶר וגו׳ אַל תִּתֵּן אֶת אֲמָתְךָ לִפְנֵי בַּת בְּלִיָּעַל וגו׳״, וְאֵין בְּלִיַּעַל אֶלָּא עֲבוֹדָה זָרָה, שֶׁנֶּאֱמַר: ״יָצְאוּ אֲנָשִׁים בְּנֵי בְלִיַּעַל מִקִּרְבֶּךָ וַיַּדִּיחוּ וגו׳״. וְצָרִיךְ הוּא לַחֲזֹר וּלְהִתְפַּלֵּל כְּשֶׁיָּסִיר יֵינוֹ מֵעָלָיו. וַאֲפִלּוּ עָבַר זְמַן הַתְּפִלָּה — מַשְׁלִים אוֹתָהּ בִּזְמַן תְּפִלָּה שֶׁלְּאַחֲרֶיהָ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ק״ח. וְדִין קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ לְעִנְיַן שִׁכּוֹר הוּא כְּדִין תְּפִלָּה, וּלְעִנְיַן שָׁתוּי יִתְבָּאֵר בְּסִימָן קפ״ה. אֲבָל שְׁאָר בְּרָכוֹת יָכוֹל לְבָרֵךְ אַף עַל פִּי שֶׁהוּא שִׁכּוֹר:

שָׁתָה רְבִיעִית. שָׁתָה רְבִיעִית יַיִן בְּבַת אַחַת — אַל יִתְפַּלֵּל עַד שֶׁיָּסִיר יֵינוֹ, אַף שֶׁיַּעֲבֹר הַזְּמַן. שָׁתָה יוֹתֵר : יָכוֹל לְדַבֵּר לִפְנֵי הַמֶּלֶךְ — אֵינוֹ שִׁכּוֹר אֶלָּא שָׁתוּי, וּתְפִלָּתוֹ תְּפִלָּה בְּדִיעֲבַד ; אֵינוֹ יָכוֹל — נִקְרָא שִׁכּוֹר וּתְפִלָּתוֹ תּוֹעֵבָה כְּאִלּוּ עוֹבֵד עֲבוֹדָה זָרָה, וְחוֹזֵר וּמִתְפַּלֵּל כְּשֶׁיָּסִיר יֵינוֹ (וְאַף שֶׁעָבַר הַזְּמַן — מַשְׁלִים, סימן ק״ח). דִּין ק״ש וּבִרְכוֹתֶיהָ לְעִנְיַן שִׁכּוֹר כְּדִין תְּפִלָּה (וּלְעִנְיַן שָׁתוּי — סימן קפ״ה) ; אֲבָל שְׁאָר בְּרָכוֹת מְבָרֵךְ אַף שִׁכּוֹר.

סעיף ב יַיִן שֶׁבְּתוֹךְ הַסְּעוּדָה אֵינוֹ מְשַׁכֵּר.

יַיִן שֶׁבְּתוֹךְ הַסְּעוּדָה אֵינוֹ מְשַׁכֵּר. וְיֵשׁ אוֹמְרִים גַּם יַיִן שֶׁלִּפְנֵי הַמָּזוֹן, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תע״ג:

יַיִן שֶׁבְּתוֹךְ הַסְּעוּדָה. יַיִן הַנִּשְׁתֶּה בְּתוֹךְ הַסְּעוּדָה אֵינוֹ מְשַׁכֵּר. לְפִי הַיֵּשׁ אוֹמְרִים כֵּן גַּם יַיִן שֶׁלִּפְנֵי הַמָּזוֹן, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תע״ג.

סעיף ג שֵׁנָה וְדֶרֶךְ מִיל מְפִיגִין אֶת הַיַּיִן ; הַכֹּל לְפִי רֹב הַיַּיִן.

שֵׁנָה כָּל שֶׁהוּא מְפִיגָה אֶת הַיַּיִן, אִם לֹא שָׁתָה אֶלָּא רְבִיעִית. אֲבָל אִם שָׁתָה יוֹתֵר מֵרְבִיעִית — כָּל שֶׁכֵּן שֶׁשֵּׁנָה מוּעֶטֶת מְשַׁכַּרְתּוֹ. אֲבָל שֵׁנָה הַרְבֵּה — וַדַּאי מְפִיגָתוֹ. וְכֵן דֶּרֶךְ מִיל מְפִיגָה אֶת הַיַּיִן, בֵּין מְהַלֵּךְ בְּרַגְלָיו בֵּין רוֹכֵב עַל גַּבֵּי בְּהֵמָה אִם לֹא שָׁתָה אֶלָּא רְבִיעִית. אֲבָל אִם שָׁתָה יוֹתֵר מֵרְבִיעִית — כָּל שֶׁכֵּן שֶׁדֶּרֶךְ טוֹרַדְתּוֹ, אִם מְהַלֵּךְ בְּרַגְלָיו. אֲבָל אִם הָיָה רָכוּב שֶׁאֵין לוֹ טֹרַח כָּל כָּךְ — אֲפִלּוּ בְּיוֹתֵר מֵרְבִיעִית דֶּרֶךְ מְפִיגָתוֹ. וְהוּא שֶׁרָכוּב ג׳ מִילִין. וּמִכָּל מָקוֹם הַכֹּל לְפִי רֹב הַיַּיִן, שֶׁאִם הִרְבָּה לִשְׁתּוֹת מְאֹד — אֵין דֶּרֶךְ ג׳ מִילִין מְפִיגָתוֹ:

שֵׁנָה וְדֶרֶךְ. שֵׁנָה כָּל שֶׁהוּא מְפִיגָה אֶת הַיַּיִן — רַק בְּשָׁתָה רְבִיעִית ; בְּיוֹתֵר — שֵׁנָה מוּעֶטֶת מְשַׁכַּרְתּוֹ, וְרַק שֵׁנָה הַרְבֵּה מְפִיגָתוֹ. וְכֵן דֶּרֶךְ מִיל מְפִיגָה, בְּרַגְלָיו אוֹ רָכוּב, בְּשָׁתָה רְבִיעִית ; בְּיוֹתֵר וּמְהַלֵּךְ בְּרַגְלָיו — דֶּרֶךְ טוֹרַדְתּוֹ ; אֲבָל רָכוּב בְּלֹא טֹרַח — אֲפִלּוּ בְּיוֹתֵר מֵרְבִיעִית דֶּרֶךְ מְפִיגָתוֹ, וְהוּא בְּרָכוּב ג׳ מִילִין. וּמִכָּל מָקוֹם הַכֹּל לְפִי רֹב הַיַּיִן : הִרְבָּה מְאֹד — אֵין ג׳ מִילִין מְפִיגָתוֹ.

סעיף ד הַיֵּינוֹת שֶׁלָּנוּ אֵינָם חֲזָקִים ; אֲבָל שִׁכּוֹר לְגַמְרֵי — צָרִיךְ לִזָּהֵר.

כָּל אֶחָד שֶׁהוּא שָׁתוּי — דַּי לוֹ לְהַמְתִּין לְפִי שֶׁמַּרְגִּישׁ בְּנַפְשׁוֹ שֶׁיָּפִיג יֵינוֹ. וְלָכֵן אֵין נִזְהָרִין עַכְשָׁו בַּיֵּינוֹת שֶׁלָּנוּ שֶׁמִּתְפַּלְלִים אַף עַל פִּי שֶׁשָּׁתוּ רְבִיעִית יַיִן וְיוֹתֵר, כֵּיוָן שֶׁאָנוּ מַרְגִּישִׁים וְיוֹדְעִים בַּיֵּינוֹת שֶׁלָּנוּ שֶׁאֵינָם חֲזָקִים. וְכָל שֶׁכֵּן אִם מִתְפַּלְלִים מִתּוֹךְ הַסִּדּוּר שֶׁבְּיָדָם, שֶׁאֵין חוֹשְׁשִׁין לְשִׁכְרוּת מְעַט. וְאַף בְּעַל פֶּה אֵין לָחֹש כָּל כָּךְ עַכְשָׁו מִלְּהִתְפַּלֵּל כְּשֶׁהוּא שָׁתוּי אִם יַעֲבֹר זְמַן הַתְּפִלָּה עַד שֶׁיָּפִיג יֵינוֹ, כֵּיוָן שֶׁעַכְשָׁו אֵין מְכַוְּנִים כָּל כָּךְ בַּתְּפִלָּה. אֲבָל שִׁכּוֹר לְגַמְרֵי שֶׁאֵינוֹ יָכוֹל לְדַבֵּר לִפְנֵי הַמֶּלֶךְ — צָרִיךְ לִזָּהֵר אַף עַכְשָׁו כֵּיוָן שֶׁתְּפִלָּתוֹ תּוֹעֵבָה וּכְאִלּוּ עוֹבֵד עֲבוֹדָה זָרָה. וְצָרִיךְ לַחֲזֹר וּלְהִתְפַּלֵּל:

הַיֵּינוֹת שֶׁלָּנוּ. כָּל שָׁתוּי — דַּי לוֹ לְהַמְתִּין לְפִי שֶׁמַּרְגִּישׁ בְּנַפְשׁוֹ שֶׁיָּפִיג יֵינוֹ. וְלָכֵן אֵין נִזְהָרִין עַכְשָׁו בַּיֵּינוֹת שֶׁלָּנוּ — מִתְפַּלְלִים אַף שֶׁשָּׁתוּ רְבִיעִית וְיוֹתֵר, שֶׁאֵינָם חֲזָקִים ; וְכָל שֶׁכֵּן מִתּוֹךְ הַסִּדּוּר, שֶׁאֵין חוֹשְׁשִׁין לְשִׁכְרוּת מְעַט ; וְאַף בְּעַל פֶּה אֵין לָחֹשׁ כָּל כָּךְ, שֶׁעַכְשָׁו אֵין מְכַוְּנִים כָּל כָּךְ. אֲבָל שִׁכּוֹר לְגַמְרֵי שֶׁאֵינוֹ יָכוֹל לְדַבֵּר לִפְנֵי הַמֶּלֶךְ — צָרִיךְ לִזָּהֵר אַף עַכְשָׁו, שֶׁתְּפִלָּתוֹ תּוֹעֵבָה, וְחוֹזֵר וּמִתְפַּלֵּל.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim צ״ט (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה

היסוד : דִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַהֶבְדֵּל שֶׁבֵּין שָׁתוּי לְשִׁכּוֹר לְפִי שְׁעוּר רְבִיעִית וְ״לְדַבֵּר לִפְנֵי הַמֶּלֶךְ״. אחר כך הַיַּיִן שֶׁבְּתוֹךְ הַסְּעוּדָה. אחר כך מְפִיגֵי הַיַּיִן — שֵׁנָה וְדֶרֶךְ. ולבסוף מַעֲמַד הַיֵּינוֹת שֶׁלָּנוּ וְהַסְּמִיכָה עַל הַהַרְגָּשָׁה.

שָׁתוּי לְעֻמַּת שִׁכּוֹר — רְבִיעִית וְלִדְבַּר לִפְנֵי הַמֶּלֶךְ

שָׁתוּי לְעֻמַּת שִׁכּוֹר — רְבִיעִית וּלְדַבֵּר לִפְנֵי הַמֶּלֶךְ

היסוד. שָׁתָה רְבִיעִית בְּבַת אַחַת — אַל יִתְפַּלֵּל עַד שֶׁיָּסִיר יֵינוֹ, אַף שֶׁיַּעֲבֹר הַזְּמַן. בְּיוֹתֵר : יָכוֹל לְדַבֵּר לִפְנֵי הַמֶּלֶךְ → שָׁתוּי, וּתְפִלָּתוֹ בְּדִיעֲבַד ; אֵינוֹ יָכוֹל → שִׁכּוֹר, וּתְפִלָּתוֹ תּוֹעֵבָה (סעיף א).

המסקנה. הַשִּׁכּוֹר חוֹזֵר וּמִתְפַּלֵּל, וּמַשְׁלִים בַּתְּפִלָּה שֶׁלְּאַחֲרֶיהָ (סימן ק״ח) ; ק״ש וּבִרְכוֹתֶיהָ כְּדִין תְּפִלָּה. עיין סימן 98 (כוונת התפלה).

יַיִן שֶׁבְּתוֹךְ הַסְּעוּדָה

יַיִן שֶׁבְּתוֹךְ הַסְּעוּדָה

היסוד. יַיִן שֶׁבְּתוֹךְ הַסְּעוּדָה אֵינוֹ מְשַׁכֵּר ; הַיֵּשׁ אוֹמְרִים מַרְחִיב גַּם לְיַיִן שֶׁלִּפְנֵי הַמָּזוֹן (סעיף ב).

המסקנה. הַפַּת הַמְלַוָּה אֶת הַיַּיִן מְמַעֶטֶת כֹּחוֹ — וְהַהַפְנָיָה לְסִימָן תע״ג קוֹשֶׁרֶת זֹאת לְלֵיל הַסֵּדֶר. עיין סימן 100 (סדר התפלה).

מְפִיגֵי הַיַּיִן — שֵׁנָה וְדֶרֶךְ

מְפִיגֵי הַיַּיִן — שֵׁנָה וְדֶרֶךְ

היסוד. שֵׁנָה כָּל שֶׁהוּא וְדֶרֶךְ מִיל מְפִיגִין — רַק בְּשָׁתָה רְבִיעִית. בְּיוֹתֵר, שֵׁנָה מוּעֶטֶת אוֹ הֲלִיכָה מְשַׁכְּרוֹת ; רַק שֵׁנָה הַרְבֵּה אוֹ ג׳ מִילִין רָכוּב מְפִיגִין (סעיף ג).

המסקנה. הַכֹּל לְפִי רֹב הַיַּיִן : הִרְבָּה מְאֹד — אֵין ג׳ מִילִין מְפִיגָתוֹ. עיין סימן 98 (זמן התפלה).

הַיֵּינוֹת שֶׁלָּנוּ — הַהַרְגָּשָׁה וְהַסִּדּוּר

הַיֵּינוֹת שֶׁלָּנוּ — הַהַרְגָּשָׁה וְהַסִּדּוּר

היסוד. דַּי לְהַמְתִּין לְפִי שֶׁמַּרְגִּישׁ בְּנַפְשׁוֹ ; יֵינוֹת שֶׁלָּנוּ אֵינָם חֲזָקִים, מִתְפַּלְלִים אַף בִּרְבִיעִית וְיוֹתֵר, וְכָל שֶׁכֵּן מִתּוֹךְ הַסִּדּוּר (סעיף ד).

המסקנה. אֲבָל שִׁכּוֹר לְגַמְרֵי — צָרִיךְ לִזָּהֵר אַף עַכְשָׁו, שֶׁתְּפִלָּתוֹ תּוֹעֵבָה. עיין סימן 100 (עמידה בתפלה).

הפניה ללימוד. היקף הסימן — הַהֶבְדֵּל שָׁתוּי / שִׁכּוֹר, מְפִיגֵי הַיַּיִן וּמַעֲמַד הַיֵּינוֹת שֶׁלָּנוּ — נוגע להנהגת התפלה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן צ״ט. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
שָׁתוּי וְשִׁכּוֹר וּשְׁעוּר רְבִיעִית או״ח צ״ט — שָׁתָה רְבִיעִית אַל יִתְפַּלֵּל ; יָכוֹל לְדַבֵּר לִפְנֵי הַמֶּלֶךְ תְּפִלָּתוֹ תְּפִלָּה, וְאִם לָאו תּוֹעֵבָה וְחוֹזֵר. מ״ב צ״ט — גֶּדֶר הָרְבִיעִית וְשְׁעוּר ״לְדַבֵּר לִפְנֵי הַמֶּלֶךְ״. ערוה״ש צ״ט — הַהֶפְרֵשׁ שֶׁבֵּין שָׁתוּי לְשִׁכּוֹר לְעִנְיַן תְּפִלָּה. דרך אדמו״ר הזקן — רְבִיעִית בְּבַת אַחַת אַל יִתְפַּלֵּל ; יָכוֹל לְדַבֵּר — שָׁתוּי וּתְפִלָּתוֹ תְּפִלָּה ; אֵינוֹ יָכוֹל — שִׁכּוֹר וּתְפִלָּתוֹ תּוֹעֵבָה, וְחוֹזֵר (תַּשְׁלוּמִין סימן ק״ח) (סעיף א).
יַיִן שֶׁבְּתוֹךְ הַסְּעוּדָה או״ח צ״ט — לֹא נִתְפָּרֵשׁ בַּמְחַבֵּר, וְהוּא מִדִּינֵי שְׁעוּר הַשִּׁכְרוּת. מ״ב צ״ט — יַיִן שֶׁעִם הַפַּת אֵינוֹ מְשַׁכֵּר כָּל כָּךְ. ערוה״ש צ״ט — הַפַּת מְמַעֶטֶת כֹּחַ הַיַּיִן. דרך אדמו״ר הזקן — יַיִן שֶׁבְּתוֹךְ הַסְּעוּדָה אֵינוֹ מְשַׁכֵּר ; וְיֵשׁ אוֹמְרִים גַּם שֶׁלִּפְנֵי הַמָּזוֹן (סימן תע״ג) (סעיף ב).
מְפִיגֵי הַיַּיִן — שֵׁנָה וְדֶרֶךְ או״ח צ״ט — דֶּרֶךְ מִיל וְשֵׁנָה כָּל שֶׁהוּא מְפִיגִין ; וְהַיְנוּ בִּרְבִיעִית. מ״ב צ״ט — פְּרָטֵי הַמְּהַלֵּךְ וְהָרוֹכֵב וּשְׁעוּר הַדֶּרֶךְ. ערוה״ש צ״ט — גֶּדֶר הַפָּגַת הַיַּיִן בַּשֵּׁנָה וּבַדֶּרֶךְ. דרך אדמו״ר הזקן — בִּרְבִיעִית שֵׁנָה וְדֶרֶךְ מִיל מְפִיגִין ; בְּיוֹתֵר — שֵׁנָה מוּעֶטֶת וְדֶרֶךְ בָּרֶגֶל מְשַׁכְּרִין, וְרָכוּב ג׳ מִילִין מֵפִיג ; הַכֹּל לְפִי רֹב הַיַּיִן (סעיף ג).
הַיֵּינוֹת שֶׁלָּנוּ וְשִׁכּוֹר לְגַמְרֵי או״ח צ״ט — כָּל שָׁתוּי לְפִי מַה שֶּׁמַּרְגִּישׁ ; יֵינוֹת שֶׁלָּנוּ אֵין נִזְהָרִין וְכָל שֶׁכֵּן מִתּוֹךְ סִדּוּר. מ״ב צ״ט — הַקֻּלָּא בַּיֵּינוֹת שֶׁלָּנוּ וּבַמִּתְפַּלֵּל מִתּוֹךְ הַסִּדּוּר. ערוה״ש צ״ט — הַהַנְהָגָה בַּזְּמַן הַזֶּה בַּיֵּינוֹת שֶׁאֵינָם חֲזָקִים. דרך אדמו״ר הזקן — דַּי לְפִי שֶׁמַּרְגִּישׁ ; יֵינוֹת שֶׁלָּנוּ אֵינָם חֲזָקִים וְאַף בְּעַל פֶּה אֵין לָחֹשׁ כָּל כָּךְ ; אֲבָל שִׁכּוֹר לְגַמְרֵי צָרִיךְ לִזָּהֵר אַף עַכְשָׁו וְחוֹזֵר (סעיף ד).

הטבלה נערכה מטקסט המחבר (אורח חיים צ״ט, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפלה ד, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ל״א, עירובין ס״ד). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — שָׁתוּי וְשִׁכּוֹר

קו א — שָׁתוּי וְשִׁכּוֹר

יסוד הסימן : רְבִיעִית — אַל יִתְפַּלֵּל ; הַהַכְרָעָה בְּ״לְדַבֵּר לִפְנֵי הַמֶּלֶךְ״ (סעיף א).

קריאת הרב : הַשָּׁתוּי תְּפִלָּתוֹ בְּדִיעֲבַד ; הַשִּׁכּוֹר תְּפִלָּתוֹ תּוֹעֵבָה וְחוֹזֵר (תַּשְׁלוּמִין סימן ק״ח). עיין סימן 98.

מסקנה למעשה : לִבְחֹן אֶת יְכָלְתּוֹ לְדַבֵּר לִפְנֵי מֶלֶךְ קֹדֶם שֶׁמִּתְפַּלֵּל אַחַר שְׁתִיָּה.

קו ב — מְפִיגֵי הַיַּיִן

קו ב — מְפִיגֵי הַיַּיִן

יסוד הסימן : שֵׁנָה וְדֶרֶךְ מְפִיגִין בִּרְבִיעִית ; בְּיוֹתֵר — מְשַׁכְּרִין, וְהַכֹּל לְפִי רֹב הַיַּיִן (סעיפים ב-ג).

קריאת הרב : הַסְּעוּדָה כְּבָר מְמַעֶטֶת אֶת הַיַּיִן ; שֵׁנָה מוּעֶטֶת אוֹ הֲלִיכָה עֲלוּלוֹת לְהוֹסִיף שִׁכְרוּת. עיין סימן 100.

מסקנה למעשה : לֹא לִסְמֹךְ עַל מְנוּחָה קְצָרָה אוֹ דֶּרֶךְ קְצָרָה לְהָפִיג שִׁכְרוּת חֲזָקָה.

קו ג — הַיֵּינוֹת שֶׁלָּנוּ

קו ג — הַיֵּינוֹת שֶׁלָּנוּ וְהַהַרְגָּשָׁה

יסוד הסימן : דַּי לְפִי שֶׁמַּרְגִּישׁ בְּנַפְשׁוֹ ; יֵינוֹת שֶׁלָּנוּ אֵינָם חֲזָקִים (סעיף ד).

קריאת הרב : קֻלָּא אֲמִתִּית בַּזְּמַן הַזֶּה — וְכָל שֶׁכֵּן מִתּוֹךְ הַסִּדּוּר — אַךְ אֵינָהּ נוֹגַעַת בְּשִׁכּוֹר לְגַמְרֵי. עיין סימן 98.

מסקנה למעשה : לִסְמֹךְ עַל הַהַרְגָּשָׁה בְּשִׁכְרוּת מְעַט, אֲבָל לְהִמָּנַע לְגַמְרֵי בְּשִׁכְרוּת גְּמוּרָה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן צ״ט ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן צ״ט (דִּין שָׁתוּי וְשִׁכּוֹר לִתְפִלָּה)

  • ① רְבִיעִית. שָׁתָה רְבִיעִית יַיִן בְּבַת אַחַת — אַל יִתְפַּלֵּל עַד שֶׁיָּסִיר יֵינוֹ, אַף שֶׁיַּעֲבֹר הַזְּמַן. יסוד : שוע״ר צ״ט סעיף א.
  • ② שָׁתוּי אוֹ שִׁכּוֹר. בְּיוֹתֵר מֵרְבִיעִית : יָכוֹל לְדַבֵּר לִפְנֵי הַמֶּלֶךְ → שָׁתוּי (תְּפִלָּתוֹ בְּדִיעֲבַד) ; אֵינוֹ יָכוֹל → שִׁכּוֹר (תְּפִלָּתוֹ תּוֹעֵבָה, וְחוֹזֵר, תַּשְׁלוּמִין סימן ק״ח). יסוד : שוע״ר צ״ט סעיף א.
  • ③ בְּתוֹךְ הַסְּעוּדָה. יַיִן שֶׁבְּתוֹךְ הַסְּעוּדָה אֵינוֹ מְשַׁכֵּר (וְיֵשׁ אוֹמְרִים גַּם שֶׁלִּפְנֵי הַמָּזוֹן, סימן תע״ג). יסוד : שוע״ר צ״ט סעיף ב.
  • ④ מְפִיגֵי הַיַּיִן. שֵׁנָה וְדֶרֶךְ מִיל מְפִיגִין — רַק בִּרְבִיעִית ; בְּיוֹתֵר עֲלוּלִים לְהוֹסִיף, וְהַכֹּל לְפִי רֹב הַיַּיִן. יסוד : שוע״ר צ״ט סעיף ג.
  • ⑤ הַיֵּינוֹת שֶׁלָּנוּ. יֵינוֹת שֶׁלָּנוּ אֵינָם חֲזָקִים — מִתְפַּלְלִים אַף בִּרְבִיעִית וְיוֹתֵר, וְכָל שֶׁכֵּן מִתּוֹךְ הַסִּדּוּר ; אֲבָל שִׁכּוֹר לְגַמְרֵי — צָרִיךְ לִזָּהֵר אַף עַכְשָׁו. יסוד : שוע״ר צ״ט סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן צ״ט (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — שְׁעוּר רְבִיעִית, הַהֶבְדֵּל שָׁתוּי / שִׁכּוֹר, מְפִיגֵי הַיַּיִן — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
