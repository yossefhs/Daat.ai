דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ס״ג · לֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע וְשֶׁלֹּא יִישַׁן

דרכו של אדמו״ר הזקן בלֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע : לִקְרוֹת ק״ש בְּכָל תְּנוּחָה — כְּדַרְכּוֹ אֲבָל לֹא פְּרַקְדָּן ; הַמַּחְמִיר לַעֲמֹד נִקְרָא עֲבַרְיָן ; הָעֲמִידָה וְהַכַּוָּנָה בְּפָסוּק רִאשׁוֹן ; הַיָּשֵׁן שֶׁמְּעִירִים ; שֶׁלֹּא יִרְמֹז וְשֶׁלֹּא לְהַפְסִיק אֶת הַקֶּבַע — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ס״ג — לֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע וְשֶׁלֹּא יִישַׁן. אצל הרב הסימן נפרש לי׳ סְעִיפִים : תְּנוּחוֹת הַקְּרִיאָה וּפְרַקְדָּן, הַמַּחְמִיר לַעֲמֹד הַנִּקְרָא עֲבַרְיָן, הָעֲמִידָה בַּדֶּרֶךְ, עִיקַּר הַכַּוָּנָה בְּפָסוּק רִאשׁוֹן, הַיָּשֵׁן וְהַמִּתְנַמְנֵם, וְדִין לֹא יִרְמֹז וּבִיטּוּל מְלָאכָה — עֲשֵׂה אוֹתָם קֶבַע.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ס״ג — י׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ס״ג). אדמו״ר הזקן דן בו בְּלֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע וְשֶׁלֹּא יִישַׁן. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ס״ג — לֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע וְשֶׁלֹּא יִישַׁן

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ס״ג — לֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע וְשֶׁלֹּא יִישַׁן — וּבוֹ י׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.63), ואחריו תרגומנו. הסימן כולל י׳ סעיפים (סעיף k → ‎.63.k).

סעיף א כָּל אָדָם קוֹרֵא כְּדַרְכּוֹ — עוֹמֵד, יוֹשֵׁב, שׁוֹכֵב, רוֹכֵב ; אֲבָל לֹא פְּרַקְדָּן.

לֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע וְשֶׁלֹּא יִישַׁן וּבוֹ י׳ סְעִיפִים:כָּל אָדָם קוֹרִין קְרִיאַת שְׁמַע כְּדַרְכָּן, בֵּין עוֹמְדִין בֵּין יוֹשְׁבִין, בֵּין שׁוֹכְבִין בֵּין מְהַלְּכִין, בֵּין רוֹכְבִין עַל גַּבֵּי בְּהֵמָה, שֶׁנֶּאֱמַר: ״וּבְלֶכְתְּךָ בַדֶּרֶךְ״, אִם כֵּן לָמָּה נֶאֱמַר ״בְּשָׁכְבְּךָ וּבְקוּמֶךָ״? בִּזְמַן שֶׁבְּנֵי אָדָם שׁוֹכְבִין וּבִזְמַן שֶׁבְּנֵי אָדָם עוֹמְדִין. וּמִכָּל מָקוֹם פְּרַקְדָּן, דְּהַיְנוּ שֶׁהוּא מֻטָּל וּפָנָיו טוּחוֹת בַּקַּרְקַע אוֹ מֻשְׁלָךְ עַל גַּבָּיו וּפָנָיו לְמַעְלָה, אֲפִלּוּ אִם הוּא בְּעִנְיָן שֶׁאֵין אִסּוּר בִּשְׁכִיבָה זוֹ מִשּׁוּם חֲשָׁשׁ שֶׁמָּא יָבוֹא לִידֵי קִשּׁוּי כְּגוֹן שֶׁמֻּטֶּה עַל צִדּוֹ מְעַט — אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע עַד שֶׁיִּתְהַפֵּךְ עַל צִדּוֹ לְגַמְרֵי, מִפְּנֵי שֶׁמְּקַבֵּל מַלְכוּת שָׁמַיִם דֶּרֶךְ שְׂרָרָה וְגַאֲוָה, אֶלָּא אִם כֵּן הָיָה בַּעַל בָּשָׂר הַרְבֵּה וְאֵינוֹ יָכוֹל לְהִתְהַפֵּךְ עַל צִדּוֹ אוֹ שֶׁהוּא חוֹלֶה. וְיֵשׁ אוֹמְרִים שֶׁאֲפִלּוּ לְשׁוֹכֵב עַל צִדּוֹ לֹא הִתִּירוּ לִקְרוֹת כָּךְ אֶלָּא כְּשֶׁשּׁוֹכֵב עָרֹם תַּחַת סְדִינוֹ, שֶׁלֹּא הִטְרִיחוּ עָלָיו לַעֲמֹד וְלִלְבֹּשׁ בְּגָדָיו. וְהָעִקָּר כַּסְּבָרָא הָרִאשׁוֹנָה:

כָּל אָדָם כְּדַרְכּוֹ. קוֹרְאִים ק״ש בְּכָל תְּנוּחָה — עוֹמֵד, יוֹשֵׁב, שׁוֹכֵב, מְהַלֵּךְ אוֹ רוֹכֵב (״וּבְלֶכְתְּךָ בַדֶּרֶךְ״) ; ״בְּשָׁכְבְּךָ וּבְקוּמֶךָ״ — זְמַן וְלֹא תְּנוּחָה. אֲבָל לֹא פְּרַקְדָּן ; וַאֲפִלּוּ מֻטֶּה מְעַט עַל צִדּוֹ אָסוּר עַד שֶׁיִּתְהַפֵּךְ לְגַמְרֵי, מִפְּנֵי שֶׁמְּקַבֵּל מַלְכוּת שָׁמַיִם דֶּרֶךְ שְׂרָרָה וְגַאֲוָה, אֶלָּא אִם כֵּן בַּעַל בָּשָׂר אוֹ חוֹלֶה. וְיֵשׁ אוֹמְרִים עַל צִדּוֹ רַק כְּשֶׁעָרֹם תַּחַת סְדִינוֹ. וְהָעִיקָּר כַּסְּבָרָא הָרִאשׁוֹנָה.

סעיף ב הַמַּחְמִיר לַעֲמֹד כְּשֶׁהוּא יוֹשֵׁב — נִקְרָא עֲבַרְיָן, וְחַיָּב מִיתָה.

מִי שֶׁרוֹצֶה לְהַחֲמִיר לַעֲמֹד כְּשֶׁהוּא יוֹשֵׁב וְלִקְרוֹתָהּ מְעֻמָּד — נִקְרָא עֲבַרְיָן, שֶׁעוֹבֵר עַל דִּבְרֵי חֲכָמִים. וְאַף שֶׁהַחֲכָמִים הִתִּירוּ גַּם מְעֻמָּד, מִכָּל מָקוֹם כֵּיוָן שֶׁלְּפִי דִּבְרֵיהֶם אֵין מַעֲלָה כְּלָל בִּמְעֻמָּד מִבְּיוֹשֵׁב, שֶׁזֶּה שֶׁנֶּאֱמַר ״וּבְקוּמֶךָ״ אֵין הַכָּתוּב מְדַבֵּר כְּלָל בַּעֲמִידָה אֶלָּא בִּזְמַן קִימָה, וְהוּא מַחְמִיר לַעֲמֹד — הֲרֵי נִרְאֶה שֶׁאֵינוֹ חוֹשֵׁשׁ לְדִבְרֵיהֶם וּלְפֵרוּשָׁם בַּכָּתוּב, וְחַיָּב מִיתָה. וְאֵין דּוֹמֶה כְּלָל לִשְׁאָר חֻמְרוֹת שֶׁאָנוּ מַחֲמִירִין מֵאֵיזֶה טַעַם בְּמַה שֶׁהֵקֵלּוּ חֲכָמִים, שֶׁאַף חֲכָמִים הָיוּ מוֹדִים שֶׁיֵּשׁ מַעֲלָה בְּחֻמְרָא מֵאוֹתוֹ טַעַם, אֶלָּא שֶׁלֹּא רָצוּ לְהַחֲמִיר:

הַמַּחְמִיר לַעֲמֹד. יוֹשֵׁב הָרוֹצֶה חוּמְרָא לַעֲמֹד וְלִקְרוֹת מְעֻמָּד — נִקְרָא עֲבַרְיָן, שֶׁעוֹבֵר עַל דִּבְרֵי חֲכָמִים, וְחַיָּב מִיתָה. שֶׁ״וּבְקוּמֶךָ״ אֵינוֹ בַּעֲמִידָה אֶלָּא בִּזְמַן קִימָה ; בְּהַחְמִירוֹ נִרְאֶה שֶׁאֵינוֹ חוֹשֵׁשׁ לְפֵרוּשָׁם. וְאֵין דּוֹמֶה לִשְׁאָר חֻמְרוֹת שֶׁחֲכָמִים מוֹדִים בְּמַעֲלָתָן.

סעיף ג הָיָה מְהַלֵּךְ וְעוֹמֵד לְכַוֵּן — רַשַּׁאי וְתָבוֹא עָלָיו בְּרָכָה.

בַּמֶּה דְּבָרִים אֲמוּרִים? בְּיוֹשֵׁב שֶׁעוֹמֵד, אֲבָל אִם הָיָה מְהַלֵּךְ וְרוֹצֶה לַעֲמֹד וְלִקְרוֹת כָּל הַג׳ פָּרָשִׁיּוֹת שֶׁל קְרִיאַת שְׁמַע כְּדֵי שֶׁיּוּכַל לְכַוֵּן הֵיטֵב — רַשַׁאי וְתָבוֹא עָלָיו בְּרָכָה, כֵּיוָן שֶׁאֵינוֹ מִתְכַוֵּן בָּעֲמִידָה אֶלָּא בִּשְׁבִיל הַכַּוָּנָה, מַה שֶּׁאֵין כֵּן בַּיוֹשֵׁב שֶׁיּוּכַל לְכַוֵּן כְּמוֹ מְעֻמָּד. לָכֵן מַה שֶׁנוֹהֲגִין לוֹמַר בְּיוֹם הַכִּפּוּרִים ״בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד״ מְעֻמָּד — מֻתָּר לַעֲמֹד כְּשֶׁמַּגִּיעַ לִקְרִיאַת שְׁמַע, אַף עַל פִּי שֶׁהָיָה יוֹשֵׁב וְיָכוֹל לְכַוֵּן, מִכָּל מָקוֹם אֵינוֹ עוֹשֶׂה מִשּׁוּם חוֹבַת קְרִיאַת שְׁמַע אֶלָּא מִשּׁוּם חוֹבַת הַיּוֹם:

מְהַלֵּךְ שֶׁעוֹמֵד לְכַוֵּן. אֵין הָעֲבַרְיָן אֶלָּא בְּיוֹשֵׁב שֶׁעוֹמֵד ; אֲבָל הַמְהַלֵּךְ הָרוֹצֶה לַעֲמֹד וְלִקְרוֹת ג׳ פָּרָשִׁיּוֹת כְּדֵי לְכַוֵּן — רַשַּׁאי וְתָבוֹא עָלָיו בְּרָכָה, שֶׁאֵינוֹ עוֹמֵד אֶלָּא בִּשְׁבִיל הַכַּוָּנָה. וְכֵן בְּיוֹם הַכִּפּוּרִים לַעֲמֹד לְ״בָרוּךְ שֵׁם״ — מֻתָּר, מִשּׁוּם חוֹבַת הַיּוֹם וְלֹא חוֹבַת ק״ש.

סעיף ד מְהַלֵּךְ — עוֹמֵד בְּפָסוּק רִאשׁוֹן ; רוֹכֵב מַעֲמִיד בְּהֶמְתּוֹ, וּבַעֲגָלָה אֵין צָרִיךְ.

וְאִם רָצָה לִקְרוֹת כָּל הַג׳ פָּרָשִׁיּוֹת כְּשֶׁהוּא מְהַלֵּךְ — הָרְשׁוּת בְּיָדוֹ, מִלְּבַד פָּסוּק רִאשׁוֹן שֶׁצְּרִיכִין לַעֲמֹד מֵהִלּוּכוֹ (אוֹ לֵישֵׁב) וְלִקְרוֹתוֹ כְּדֵי לְכַוֵּן הֵיטֵב. וְהוּא הַדִּין אִם הָיָה רוֹכֵב עַל גַּבֵּי בְּהֵמָה — צָרִיךְ לְהַעֲמִידָהּ, שֶׁאֵינוֹ יָכוֹל לְכַוֵּן כָּל כָּךְ שֶׁהוּא טָרוּד בַּהֲלִיכָתָהּ. אֲבָל אִם יוֹשֵׁב בָּעֲגָלָה — אֵין צָרִיךְ לְהַעֲמִידָהּ:

מְהַלֵּךְ וְרוֹכֵב. רָצָה לִקְרוֹת ג׳ פָּרָשִׁיּוֹת מְהַלֵּךְ — הָרְשׁוּת בְּיָדוֹ, מִלְּבַד פָּסוּק רִאשׁוֹן שֶׁצָּרִיךְ לַעֲמֹד (אוֹ לֵישֵׁב) וּלְכַוֵּן. וְכֵן הָרוֹכֵב עַל בְּהֵמָה — מַעֲמִידָהּ (טָרוּד בַּהֲלִיכָתָהּ) ; אֲבָל יוֹשֵׁב בָּעֲגָלָה — אֵין צָרִיךְ לְהַעֲמִידָהּ.

סעיף ה עִיקַּר הַכַּוָּנָה בְּפָסוּק רִאשׁוֹן ; לֹא כִּוֵּן — חוֹזֵר וְקוֹרֵא.

אַף עַל פִּי שֶׁלְּכַתְּחִלָּה צָרִיךְ לְכַוֵּן בְּכָל הַג׳ פָּרָשִׁיּוֹת — עִקָּר הַכַּוָּנָה הִיא בְּפָסוּק רִאשׁוֹן, שֶׁהוּא קַבָּלַת מַלְכוּת שָׁמַיִם. וְהוּא הַדִּין לְ״בָרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד״, שֶׁהוּא גַּם כֵּן קַבָּלַת מַלְכוּת שָׁמַיִם. אֲבָל מִ״וְאָהַבְתָּ״ וְאֵילָךְ הוּא לְשׁוֹן צַוָּאָה. לְפִיכָךְ אִם קָרָא קְרִיאַת שְׁמַע וְלֹא כִּוֵּן לִבּוֹ בְּפָסוּק רִאשׁוֹן אוֹ בְּ״בָרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד״ — צָרִיךְ לַחֲזֹר וְלִקְרוֹתָן. וַאֲפִלּוּ הָאוֹמְרִים שֶׁמִּצְווֹת אֵין צָרִיךְ כַּוָּנָה מוֹדִים בָּזֶה שֶׁאִם לֹא כִּוֵּן לִבּוֹ — לֹא יָצָא יְדֵי חוֹבָתוֹ מִן הַתּוֹרָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ס׳. וּכְשֶׁקּוֹרֵא פָּסוּק ״שְׁמַע יִשְׂרָאֵל״ פַּעַם שֵׁנִית — יִקְרָא בְּלַחַשׁ אִם הוּא בְּצִבּוּר, שֶׁלֹּא יִהְיֶה נִרְאֶה כִּמְקַבֵּל ב׳ רָשֻׁיּוֹת. וְאִם לֹא נִזְכַּר שֶׁלֹּא כִּוֵּן לִבּוֹ עַד לְאַחַר שֶׁסִּיֵּם כָּל הַפָּרָשָׁה — צָרִיךְ לַחֲזֹר לָרֹאשׁ, שֶׁאִי אֶפְשָׁר לִקְרוֹתוֹ בְּמָקוֹם שֶׁנִּזְכַּר, מִשּׁוּם שֶׁנִּמְצָא קוֹרְאָהּ לְמַפְרֵעַ וְלֹא יָצָא כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ס״ד:

עִיקַּר הַכַּוָּנָה. אַף שֶׁלְּכַתְּחִלָּה צָרִיךְ לְכַוֵּן בְּג׳ פָּרָשִׁיּוֹת — הָעִיקָּר בְּפָסוּק רִאשׁוֹן (קַבָּלַת מַלְכוּת שָׁמַיִם) וְכֵן ״בָּרוּךְ שֵׁם״ ; מִ״וְאָהַבְתָּ״ לְשׁוֹן צַוָּאָה. לֹא כִּוֵּן בְּפָסוּק רִאשׁוֹן אוֹ בְּ״בָרוּךְ שֵׁם״ — חוֹזֵר. וַאֲפִלּוּ הָאוֹמְרִים מִצְווֹת אֵין צְרִיכוֹת כַּוָּנָה מוֹדִים כָּאן (לֹא יָצָא מִן הַתּוֹרָה, עיין סימן 60). ״שְׁמַע״ שֵׁנִית בְּלַחַשׁ בְּצִבּוּר. לֹא נִזְכַּר עַד סוֹף הַפָּרָשָׁה — חוֹזֵר לָרֹאשׁ (עיין סימן 64).

סעיף ו הַיָּשֵׁן — מְצַעֲרִים אוֹתוֹ עַד שֶׁיִּקְרָא פָּסוּק רִאשׁוֹן וְהוּא עֵר מַמָּשׁ.

אִם הָיָה יָשֵׁן — מְצַעֲרִים אוֹתוֹ וּמְעִירִים אוֹתוֹ עַד שֶׁיִּקְרָא פָּסוּק רִאשׁוֹן וְהוּא עֵר מַמָּשׁ שֶׁיּוּכַל לְכַוֵּן. מִכָּאן וְאֵילָךְ — אֵין מְצַעֲרִין אוֹתוֹ כְּדֵי שֶׁיִּקְרָא וְהוּא עֵר מַמָּשׁ, שֶׁאַף עַל פִּי שֶׁהוּא קוֹרֵא מִתְנַמְנֵם — יָצָא. וְאַף עַל פִּי שֶׁלְּכַתְּחִלָּה צָרִיךְ לְכַוֵּן בְּכֻלָּהּ — אֵין מַטְרִיחִין וּמְצַעֲרִים אוֹתוֹ בִּשְׁבִיל כָּךְ, כֵּיוָן שֶׁהוּא אָנוּס בַּשֵּׁנָה. אֲבָל עַל כָּל פָּנִים מְצַעֲרִים אוֹתוֹ עַד שֶׁיִּקְרָא כֻּלּוֹ בְּפִיו. וְלֹא אֹנֶס שֵׁנָה בִּלְבַד, אֶלָּא הוּא הַדִּין אִם אָנוּס אֹנֶס אַחֵר שֶׁלֹּא יוּכַל לְכַוֵּן מִ״וְאָהַבְתָּ״ וְאֵילָךְ — אֵין מַטְרִיחִין עָלָיו לְכַוֵּן, כֵּיוָן שֶׁהוּא אָנוּס:

הַיָּשֵׁן. מְצַעֲרִים וּמְעִירִים אֶת הַיָּשֵׁן עַד שֶׁיִּקְרָא פָּסוּק רִאשׁוֹן וְהוּא עֵר מַמָּשׁ (לְכַוֵּן). מִכָּאן וְאֵילָךְ — אֵין מְצַעֲרִין אוֹתוֹ לִהְיוֹת עֵר מַמָּשׁ, שֶׁאַף מִתְנַמְנֵם יָצָא ; אָנוּס בַּשֵּׁנָה. אֲבָל מְצַעֲרִים עַד שֶׁיִּקְרָא כֻּלּוֹ בְּפִיו. וְהוּא הַדִּין אֹנֶס אַחֵר שֶׁלֹּא יוּכַל לְכַוֵּן מִ״וְאָהַבְתָּ״ וְאֵילָךְ — אֵין מַטְרִיחִין עָלָיו.

סעיף ז לֹא יִרְמֹז בְּעֵינָיו בַּפָּרָשָׁה רִאשׁוֹנָה — ״עֲשֵׂה אוֹתָם קֶבַע״.

הַקּוֹרֵא קְרִיאַת שְׁמַע — לֹא יִרְמֹז לְחֲבֵרוֹ בְּעֵינָיו, וְלֹא יִקְרֹץ בִּשְׂפָתָיו, וְלֹא יַרְאֶה בְּאֶצְבְּעוֹתָיו, וַאֲפִלּוּ לְצֹרֶךְ מִצְוָה. וְאִם עָשָׂה כֵּן — עָלָיו הַכָּתוּב נֶאֱמַר: ״וְלֹא אֹתִי קָרָאתָ יַעֲקֹב״ כו׳, מִפְּנֵי שֶׁנִּרְאֶה כְּקוֹרֵא דֶּרֶךְ עֲרַאי, וְהַתּוֹרָה אָמְרָהּ: ״וְדִבַּרְתָּ בָּם״, וְדָרְשׁוּ חֲכָמִים: עֲשֵׂה אוֹתָם קֶבַע. בַּמֶּה דְּבָרִים אֲמוּרִים? בְּפָרָשָׁה רִאשׁוֹנָה, מִפְּנֵי שֶׁיֵּשׁ בִּתְחִלָּתָהּ קַבָּלַת מַלְכוּת שָׁמַיִם — הִצְרִיכוּ לַעֲשׂוֹתָהּ קֶבַע. אֲבָל מִכָּאן וְאֵילָךְ — מֻתָּר לְצֹרֶךְ מִצְוָה קְצָת, כְּעֵין מַעֲשֶׂה שֶׁהָיָה שֶׁשָּׁנָה תַּלְמִיד בִּפְנֵי רַבּוֹ בַּמִּשְׁנָה ״זְכַרְיָה בֶּן קַפּוֹטָל״, וְרַבּוֹ קָרָא פָּרָשָׁה שְׁנִיָּה שֶׁל קְרִיאַת שְׁמַע וְהוֹרָה לוֹ בְּיָדוֹ לִקְרוֹת ״קַבּוֹטָל״:

לֹא יִרְמֹז. הַקּוֹרֵא לֹא יִרְמֹז בְּעֵינָיו, לֹא יִקְרֹץ בִּשְׂפָתָיו, וְלֹא יַרְאֶה בְּאֶצְבְּעוֹתָיו — וַאֲפִלּוּ לְצֹרֶךְ מִצְוָה. וְאִם עָשָׂה — ״וְלֹא אֹתִי קָרָאתָ יַעֲקֹב״ (כְּקוֹרֵא עֲרַאי) ; ״וְדִבַּרְתָּ בָּם״ — עֲשֵׂה אוֹתָם קֶבַע. בְּמֶה דְּבָרִים אֲמוּרִים בְּפָרָשָׁה רִאשׁוֹנָה ; מִכָּאן וְאֵילָךְ מֻתָּר לְצֹרֶךְ מִצְוָה קְצָת (מַעֲשֵׂה דִזְכַרְיָה בֶּן קַפּוֹטָל).

סעיף ח הָעוֹסֵק בִּמְלָאכָה — יִתְבַּטֵּל עַד שֶׁיִּקְרָא פָּרָשָׁה רִאשׁוֹנָה.

הָיָה עוֹסֵק בִּמְלָאכָה וְרָצָה לִקְרוֹת קְרִיאַת שְׁמַע — יִתְבַּטֵּל מִמְּלַאכְתוֹ עַד שֶׁיִּקְרָא קְרִיאַת שְׁמַע פָּרָשָׁה רִאשׁוֹנָה, כְּדֵי שֶׁלֹּא יִהְיֶה כְּקוֹרֵא דֶּרֶךְ עֲרַאי, וַאֲפִלּוּ יָכוֹל לְכַוֵּן:

בִּיטּוּל מְלָאכָה. הָיָה עוֹסֵק בִּמְלָאכָה וְרָצָה לִקְרוֹת — יִתְבַּטֵּל מִמְּלַאכְתּוֹ עַד שֶׁיִּקְרָא פָּרָשָׁה רִאשׁוֹנָה, שֶׁלֹּא יִהְיֶה כְּקוֹרֵא עֲרַאי, וַאֲפִלּוּ יָכוֹל לְכַוֵּן.

סעיף ט הָאֻמָּנִין בְּרֹאשׁ הָאִילָן — קוֹרִין בִּמְקוֹמָן וְאֵין צְרִיכִים לֵירֵד.

הָאֻמָּנִין וְכֵן בַּעַל הַבַּיִת שֶׁהָיוּ עוֹשִׂין מְלָאכָה בְּרֹאשׁ הָאִילָן אוֹ בְּרֹאשׁ שׁוּרוֹת הַבִּנְיָן — קוֹרִין קְרִיאַת שְׁמַע בִּמְקוֹמָן, וְאֵין צְרִיכִים לֵירֵד. אֲבָל מִכָּל מָקוֹם צְרִיכִים לְהִתְבַּטֵּל מִמְּלַאכְתָם בְּפָרָשָׁה רִאשׁוֹנָה, כְּמוֹ שֶׁנִּתְבָּאֵר:

הָאֻמָּנִין. אֻמָּנִין וּבַעַל הַבַּיִת הָעוֹשִׂים מְלָאכָה בְּרֹאשׁ הָאִילָן אוֹ בְּרֹאשׁ שׁוּרוֹת הַבִּנְיָן — קוֹרִין בִּמְקוֹמָן וְאֵין צְרִיכִים לֵירֵד ; אֲבָל צְרִיכִים לְהִתְבַּטֵּל בְּפָרָשָׁה רִאשׁוֹנָה, כְּמוֹ שֶׁנִּתְבָּאֵר (סעיף ח).

סעיף י הַכַּתָּף קוֹרֵא עִם מַשָּׂאוֹ ; לֹא בִּשְׁעַת טְעִינָה וּפְרִיקָה.

הַכַּתָּף אַף עַל פִּי שֶׁמַּשָּׂאוֹ עַל כְּתֵפוֹ — קוֹרֵא קְרִיאַת שְׁמַע, מִפְּנֵי שֶׁיּוּכַל לְכַוֵּן. אֲבָל לֹא יַתְחִיל בְּשָׁעָה שֶׁטּוֹעֵן וְלֹא בְּשָׁעָה שֶׁפּוֹרֵק, מִפְּנֵי שֶׁאֵין לִבּוֹ מְיֻשָּׁב:

הַכַּתָּף. הַכַּתָּף אַף שֶׁמַּשָּׂאוֹ עַל כְּתֵפוֹ — קוֹרֵא ק״ש, שֶׁיּוּכַל לְכַוֵּן. אֲבָל לֹא יַתְחִיל בְּשָׁעָה שֶׁטּוֹעֵן וְלֹא בְּשָׁעָה שֶׁפּוֹרֵק, שֶׁאֵין לִבּוֹ מְיֻשָּׁב.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ס״ג (מהדורת קה״ת, י׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — לֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע וְשֶׁלֹּא יִישַׁן

היסוד : הַתְּנוּחָה, הַכַּוָּנָה וְהַקֶּבַע שֶׁל קְרִיאַת שְׁמַע

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה תְּנוּחוֹת הַקְּרִיאָה — כְּדַרְכּוֹ אֲבָל לֹא פְּרַקְדָּן, וְהַמַּחְמִיר לַעֲמֹד הַנִּקְרָא עֲבַרְיָן. אחר כך עִיקַּר הַכַּוָּנָה בְּפָסוּק רִאשׁוֹן. אחר כך הַיָּשֵׁן שֶׁמְּעִירִים עַד פָּסוּק רִאשׁוֹן. ולבסוף הַקֶּבַע וְלֹא הָעֲרַאי — לֹא יִרְמֹז, בִּיטּוּל מְלָאכָה, הָאֻמָּנִין וְהַכַּתָּף.

תְּנוּחוֹת הַקְּרִיאָה וְ״עֲבַרְיָן״

תְּנוּחוֹת הַקְּרִיאָה וְ״עֲבַרְיָן״

היסוד. לִקְרוֹת ק״ש כְּדַרְכּוֹ — עוֹמֵד, יוֹשֵׁב, שׁוֹכֵב, מְהַלֵּךְ, רוֹכֵב, אֲבָל לֹא פְּרַקְדָּן ; הַמַּחְמִיר לַעֲמֹד — עֲבַרְיָן (סעיפים א-ד).

המסקנה. אֵין עוֹמְדִים בְּכַוָּנָה תְּחִלָּה : ״וּבְקוּמֶךָ״ זְמַן וְלֹא תְּנוּחָה. עיין סימן 64 (סדר קריאת שמע).

עִיקַּר הַכַּוָּנָה בְּפָסוּק רִאשׁוֹן

עִיקַּר הַכַּוָּנָה בְּפָסוּק רִאשׁוֹן

היסוד. עִיקַּר הַכַּוָּנָה בְּפָסוּק רִאשׁוֹן (קַבָּלַת מַלְכוּת שָׁמַיִם) ; לֹא כִּוֵּן — חוֹזֵר (סעיף ה).

המסקנה. אַף מ״ד מִצְווֹת אֵין צְרִיכוֹת כַּוָּנָה מוֹדֶה כָּאן (לֹא יָצָא מִן הַתּוֹרָה). עיין סימן 60 (כוונת פסוק ראשון).

הַיָּשֵׁן וְהַמִּתְנַמְנֵם

הַיָּשֵׁן וְהַמִּתְנַמְנֵם

היסוד. מְצַעֲרִים אֶת הַיָּשֵׁן עַד פָּסוּק רִאשׁוֹן וְהוּא עֵר מַמָּשׁ ; מִכָּאן וְאֵילָךְ מִתְנַמְנֵם — יָצָא (סעיף ו).

המסקנה. אָנוּס בַּשֵּׁנָה ; שָׁתוּי וְשִׁיכּוֹר — עיין סימן 99 (שתוי לתפילה).

קֶבַע וְלֹא עֲרַאי

קֶבַע וְלֹא עֲרַאי

היסוד. לֹא יִרְמֹז בַּפָּרָשָׁה רִאשׁוֹנָה, יְבַטֵּל מְלָאכָה ; הָאֻמָּנִין וְהַכַּתָּף (סעיפים ז-י).

המסקנה. ״עֲשֵׂה אוֹתָם קֶבַע״ — לֹא כְּקוֹרֵא עֲרַאי. עיין סימן 62 (מדקדק באותיותיה).

הפניה ללימוד. היקף הסימן — הַתְּנוּחוֹת, הַכַּוָּנָה וְדִין לֹא יִרְמֹז — נוגע להנהגת ק״ש היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

לֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ס״ג. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מי׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
תְּנוּחוֹת הַקְּרִיאָה וּפְרַקְדָּן או״ח ס״ג — קוֹרֵא כְּדַרְכּוֹ עוֹמֵד יוֹשֵׁב שׁוֹכֵב וְרוֹכֵב, וְלֹא פְּרַקְדָּן. מ״ב ס״ג — פְּרָטֵי הַשְּׁכִיבָה עַל צִדּוֹ וְהַחוֹלֶה. ערוה״ש ס״ג — עִנְיַן הַתְּנוּחוֹת וְהַהֶכְשֵׁר. דרך אדמו״ר הזקן — כָּל אָדָם כְּדַרְכּוֹ ; פְּרַקְדָּן אָסוּר מִשּׁוּם שְׂרָרָה וְגַאֲוָה, עַד שֶׁיִּתְהַפֵּךְ עַל צִדּוֹ ; וְהַמַּחְמִיר לַעֲמֹד עֲבַרְיָן (סעיפים א-ד).
עִיקַּר הַכַּוָּנָה בְּפָסוּק רִאשׁוֹן או״ח ס״ג — לֹא כִּוֵּן בְּפָסוּק רִאשׁוֹן — לֹא יָצָא וְחוֹזֵר. מ״ב ס״ג — מוֹדֶה מ״ד אֵין צְרִיכוֹת כַּוָּנָה, וּפְרָטֵי הַחֲזָרָה. ערוה״ש ס״ג — עִיקַּר קַבָּלַת מַלְכוּת שָׁמַיִם. דרך אדמו״ר הזקן — עִיקַּר הַכַּוָּנָה בְּפָסוּק רִאשׁוֹן וּבְ״בָרוּךְ שֵׁם״ ; לֹא נִזְכַּר עַד סוֹף הַפָּרָשָׁה — חוֹזֵר לָרֹאשׁ (סעיף ה).
הַיָּשֵׁן וְהַמִּתְנַמְנֵם או״ח ס״ג — מְצַעֲרִין אוֹתוֹ עַד שֶׁיִּקְרָא פָּסוּק רִאשׁוֹן. מ״ב ס״ג — עַד הֵיכָן מְצַעֲרִין, וְהַמִּתְנַמְנֵם. ערוה״ש ס״ג — דִּין הַיָּשֵׁן וְהָאָנוּס. דרך אדמו״ר הזקן — מְצַעֲרִים עַד פָּסוּק רִאשׁוֹן וְהוּא עֵר מַמָּשׁ ; מִכָּאן וְאֵילָךְ מִתְנַמְנֵם יָצָא, כֵּיוָן שֶׁאָנוּס בַּשֵּׁנָה (סעיף ו).
קֶבַע וְלֹא עֲרַאי — רְמִיזָה וּבִיטּוּל מְלָאכָה או״ח ס״ג — לֹא יִרְמֹז בְּעֵינָיו, וְיִתְבַּטֵּל מִמְּלַאכְתּוֹ. מ״ב ס״ג — פְּרָטֵי הָרְמִיזָה וְהָאֻמָּנִין. ערוה״ש ס״ג — עֲשֵׂה אוֹתָם קֶבַע. דרך אדמו״ר הזקן — לֹא יִרְמֹז בַּפָּרָשָׁה רִאשׁוֹנָה (עֲשֵׂה אוֹתָם קֶבַע) ; הָאֻמָּנִין קוֹרִין בִּמְקוֹמָן, וְהַכַּתָּף לֹא בִּטְעִינָה וּפְרִיקָה (סעיפים ז-י).

הטבלה נערכה מטקסט המחבר (אורח חיים ס״ג, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ב, רא״ש, הר״י, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות י׳-ט״ז). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (י׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — תְּנוּחוֹת הַקְּרִיאָה וְ״עֲבַרְיָן״

קו א — תְּנוּחוֹת הַקְּרִיאָה וְ״עֲבַרְיָן״

יסוד הסימן : לִקְרוֹת ק״ש כְּדַרְכּוֹ, אֲבָל לֹא פְּרַקְדָּן (סעיפים א-ד).

קריאת הרב : הַמַּחְמִיר לַעֲמֹד — עֲבַרְיָן ; אֲבָל הַמְהַלֵּךְ שֶׁעוֹמֵד לְכַוֵּן — רַשַּׁאי. עיין סימן 64.

מסקנה למעשה : לֹא לְהַחְמִיר לַעֲמֹד ; לִקְרוֹת בְּכָל תְּנוּחָה מְהֻגֶּנֶת.

קו ב — כַּוָּנַת פָּסוּק רִאשׁוֹן וְהַיָּשֵׁן

קו ב — כַּוָּנַת פָּסוּק רִאשׁוֹן וְהַיָּשֵׁן

יסוד הסימן : עִיקַּר הַכַּוָּנָה בְּפָסוּק רִאשׁוֹן ; הַיָּשֵׁן — עַד שֶׁיִּקְרָאֶנּוּ עֵר מַמָּשׁ (סעיפים ה-ו).

קריאת הרב : לֹא כִּוֵּן בְּפָסוּק רִאשׁוֹן — חוֹזֵר. עיין סימן 60, סימן 99.

מסקנה למעשה : לְהַקְפִּיד עַל כַּוָּנַת הַפָּסוּק הָרִאשׁוֹן, וְלַעֲמֹד בּוֹ בַּדֶּרֶךְ.

קו ג — ״קֶבַע״ וְלֹא ״עֲרַאי״

קו ג — ״קֶבַע״ וְלֹא ״עֲרַאי״

יסוד הסימן : לֹא יִרְמֹז בַּפָּרָשָׁה רִאשׁוֹנָה, יְבַטֵּל מְלָאכָה (סעיפים ז-י).

קריאת הרב : ״עֲשֵׂה אוֹתָם קֶבַע״ ; הָאֻמָּנִין קוֹרִין בִּמְקוֹמָן. עיין סימן 62.

מסקנה למעשה : בַּפָּרָשָׁה רִאשׁוֹנָה — לֹא סִימָנִים וְלֹא מְלָאכָה ; לִקְרוֹת בְּקֶבַע.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ס״ג ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ס״ג (לֵישֵׁב בִּשְׁעַת קְרִיאַת שְׁמַע וְשֶׁלֹּא יִישַׁן)

  • ① תנוחות הקריאה. לִקְרוֹת ק״ש כְּדַרְכּוֹ — יוֹשֵׁב, עוֹמֵד, שׁוֹכֵב עַל צִדּוֹ — אֲבָל לְעוֹלָם לֹא פְּרַקְדָּן. יסוד : שוע״ר ס״ג סעיף א.
  • ② לא להחמיר לעמוד. לֹא לַעֲמֹד בְּכַוָּנָה לִקְרוֹת מְעֻמָּד : הַמַּחְמִיר לַעֲמֹד נִקְרָא עֲבַרְיָן (אֲבָל מְהַלֵּךְ שֶׁעוֹמֵד לְכַוֵּן — רַשַּׁאי). יסוד : שוע״ר ס״ג סעיפים ב-ד.
  • ③ פסוק ראשון. לַעֲמֹד בְּפָסוּק רִאשׁוֹן כְּשֶׁמְּהַלֵּךְ, וּבְעִיקָּר לְכַוֵּן : עִיקַּר הַכַּוָּנָה בּוֹ ; לֹא כִּוֵּן — חוֹזֵר. יסוד : שוע״ר ס״ג סעיפים ג-ה.
  • ④ הישן. לְעוֹרֵר אֶת הַיָּשֵׁן עַד שֶׁיִּקְרָא פָּסוּק רִאשׁוֹן עֵר מַמָּשׁ ; מִכָּאן וְאֵילָךְ מִתְנַמְנֵם — יָצָא. יסוד : שוע״ר ס״ג סעיף ו.
  • ⑤ לא ירמוז / ביטול מלאכה. בַּפָּרָשָׁה רִאשׁוֹנָה — לֹא לִרְמֹז (עֵינַיִם, שְׂפָתַיִם, אֶצְבָּעוֹת) וְלִבְטֵל מְלָאכָה (עֲשֵׂה אוֹתָם קֶבַע) ; הַכַּתָּף לֹא יַתְחִיל בִּטְעִינָה. יסוד : שוע״ר ס״ג סעיפים ז-י.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ס״ג (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַתְּנוּחוֹת, הַכַּוָּנָה, הַיָּשֵׁן — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
