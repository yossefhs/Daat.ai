דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ע״ז · דִּין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם

דרכו של אדמו״ר הזקן בדִּין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם : כֵּיוָן שֶׁאִסּוּרָן אֵינוֹ אֶלָּא מִדִּבְרֵי סוֹפְרִים — מְבַטְּלָן בִּשְׁפִיכַת רְבִיעִית מַיִם, בֵּין בַּקַּרְקַע בֵּין בִּכְלִי ; חוּץ מִכְּלִי הַמְיֻחָד (עָבִיט) ; וְהַשִּׁעוּר לְפִי מִסְפַּר הַפְּעָמִים, וְהַנִּבְלַע בְּבֶגֶד — טוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ע״ז — דִּין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם. אצל הרב הסימן נפרש לב׳ סְעִיפִים : הַבִּטּוּל בִּרְבִיעִית מַיִם (בִּטּוּל אִסּוּר דְּרַבָּנָן, עִם דִּין הַכְּלִי הַמְיֻחָד — עָבִיט), וְשִׁעוּר הָרְבִיעִית לְפִי מִסְפַּר הַפְּעָמִים עִם הַנִּבְלַע בְּבֶגֶד (טוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ).

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ע״ז — ב׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ע״ז). אדמו״ר הזקן דן בו בְּדִין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ע״ז — דִּין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ע״ז — דִּין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם — וּבוֹ ב׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.77), ואחריו תרגומנו. הסימן כולל ב׳ סעיפים (סעיף k → ‎.77.k).

סעיף א מֵי רַגְלַיִם — אִסּוּר דְּרַבָּנָן ; מְבַטְּלָן בִּרְבִיעִית מַיִם, חוּץ מִכְּלִי הַמְיֻחָד לָהֶם.

שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם וּבוֹ ב׳ סְעִיפִים:מֵי רַגְלַיִם לְאַחַר שֶׁנָּפְלוּ לָאָרֶץ, כֵּיוָן שֶׁאֵין אִסּוּרָן אֶלָּא מִדִּבְרֵי סוֹפְרִים — יָכוֹל לְבַטְּלָם עַל יְדֵי שֶׁיִּשְׁפֹּךְ עֲלֵיהֶם רְבִיעִית מַיִם. וַאֲפִלּוּ הֵם בִּכְלִי — מֵטִיל עֲלֵיהֶם רְבִיעִית מַיִם, בֵּין שֶׁהָיוּ הֵם תְּחִלָּה בַּכְּלִי וְנוֹתֵן עֲלֵיהֶם רְבִיעִית מַיִם, בֵּין שֶׁנּוֹתֵן תְּחִלָּה הַמַּיִם בַּכְּלִי. וּבִלְבַד שֶׁלֹּא יִהְיֶה כְּלִי הַמְיֻחָד לְמֵי רַגְלַיִם, שֶׁאָז אֲפִלּוּ אֵין בּוֹ מֵי רַגְלַיִם כְּלָל — אָסוּר לִקְרוֹת כְּנֶגְדָּן, וְאֵין מוֹעִיל בִּנְתִינַת הַמַּיִם לְהַכְשִׁיר הַכְּלִי עַצְמוֹ כְּמוֹ שֶׁיִּתְבָּאֵר סִימָן פ״ז:

בִּטּוּל בִּרְבִיעִית מַיִם. כֵּיוָן שֶׁאִסּוּר מֵי רַגְלַיִם (לְאַחַר שֶׁנָּפְלוּ לָאָרֶץ) אֵינוֹ אֶלָּא מִדִּבְרֵי סוֹפְרִים — מְבַטְּלָם עַל יְדֵי שְׁפִיכַת רְבִיעִית מַיִם. וַאֲפִלּוּ בִּכְלִי — מֵטִיל עֲלֵיהֶם רְבִיעִית, בֵּין שֶׁהַמֵּי רַגְלַיִם תְּחִלָּה וְנוֹתֵן עֲלֵיהֶם מַיִם, בֵּין שֶׁהַמַּיִם תְּחִלָּה. וּבִלְבַד שֶׁלֹּא יְהֵא כְּלִי הַמְיֻחָד לְמֵי רַגְלַיִם — שֶׁאָז אֲפִלּוּ רֵיק אָסוּר לִקְרוֹת כְּנֶגְדּוֹ, וְאֵין הַמַּיִם מַכְשִׁירִים אֶת הַכְּלִי עַצְמוֹ (סימן פ״ז).

סעיף ב שִׁעוּר הָרְבִיעִית לְפִי מִסְפַּר הַפְּעָמִים ; נִבְלַע בְּבֶגֶד — מַטְפִּיחוֹ בַּמַּיִם.

רְבִיעִית שֶׁאָמְרוּ — לְמֵי רַגְלַיִם שֶׁל פַּעַם אֶחָד בֵּין שֶׁהִשְׁתִּין הַרְבֵּה אוֹ מְעַט, וּלְשֶׁל ב׳ פְּעָמִים — ב׳ רְבִיעִית, וּלְשֶׁל ג׳ פְּעָמִים — ג׳, וְכֵן לְעוֹלָם. וְכָל זֶה כְּשֶׁהֵם בְּעַיִן, אֲבָל אִם נִבְלְעוּ בְּבֶגֶד אַף עַל פִּי שֶׁהוּא טוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ, אִם חוֹזֵר וּמַטְפִּיחוֹ בַּמַּיִם — דַּיּוֹ, וּבְשַׁבָּת מְנַגֵּב בּוֹ יָדָיו כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ש״ב. וְכֵן בָּאָרֶץ:

שִׁעוּר הָרְבִיעִית וְנִבְלַע בְּבֶגֶד. הָרְבִיעִית שֶׁאָמְרוּ — לְפַעַם אֶחָד, בֵּין הִשְׁתִּין הַרְבֵּה בֵּין מְעַט ; לְב׳ פְּעָמִים — ב׳ רְבִיעִית, לְג׳ — ג׳, וְכֵן לְעוֹלָם. וְכָל זֶה כְּשֶׁהֵם בְּעַיִן ; אֲבָל אִם נִבְלְעוּ בְּבֶגֶד — אַף שֶׁטּוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ — דַּי בְּכָךְ שֶׁחוֹזֵר וּמַטְפִּיחוֹ בַּמַּיִם. וּבְשַׁבָּת מְנַגֵּב בּוֹ יָדָיו (סימן ש״ב), וְכֵן בָּאָרֶץ.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ע״ז (מהדורת קה״ת, ב׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם

היסוד : דִּין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים. תחילה הַבִּטּוּל בִּרְבִיעִית מַיִם — אִסּוּר דְּרַבָּנָן שֶׁמְּבַטְּלִים בִּשְׁפִיכַת מַיִם. אחר כך דִּין הַכְּלִי הַמְיֻחָד (עָבִיט), שֶׁבּוֹ אֵין הַמַּיִם מוֹעִילִים. ולבסוף שִׁעוּר הָרְבִיעִית לְפִי הַפְּעָמִים וְהַנִּבְלַע בְּבֶגֶד.

בִּטּוּל בִּרְבִיעִית מַיִם — אִסּוּר דְּרַבָּנָן

בִּטּוּל בִּרְבִיעִית מַיִם — אִסּוּר דְּרַבָּנָן

היסוד. אִסּוּר לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם אֵינוֹ אֶלָּא מִדִּבְרֵי סוֹפְרִים ; לְפִיכָךְ מְבַטְּלִים אוֹתוֹ בִּשְׁפִיכַת רְבִיעִית מַיִם — בֵּין עַל גַּבֵּי קַרְקַע בֵּין בִּכְלִי, וְאֵין נַפְקָא מִינָה מַה קָּדַם — הַמַּיִם אוֹ מֵי הָרַגְלַיִם (סעיף א).

המסקנה. מִשּׁוּם שֶׁשֹּׁרֶשׁ הָאִסּוּר דְּרַבָּנָן — רְבִיעִית פְּשׁוּטָה מַסְפֶּקֶת לְבַטְּלוֹ. עיין סימן ע״ו · 76 (צוֹאָה וְנִקָּיוֹן הַמָּקוֹם).

כְּלִי הַמְיֻחָד (עָבִיט) — אֵין הַמַּיִם מוֹעִילִים

כְּלִי הַמְיֻחָד (עָבִיט) — אֵין הַמַּיִם מוֹעִילִים

היסוד. כְּלִי הַמְיֻחָד לְמֵי רַגְלַיִם — אֲפִלּוּ רֵיק מִכָּל מֵי רַגְלַיִם — אָסוּר לִקְרוֹת כְּנֶגְדּוֹ, וְאֵין נְתִינַת הַמַּיִם מַכְשִׁירָה אֶת הַכְּלִי עַצְמוֹ (סעיף א ; ועיין סימן פ״ז).

המסקנה. הַבִּטּוּל פּוֹעֵל עַל מֵי הָרַגְלַיִם שֶׁבְּעַיִן, לֹא עַל מַעֲמַד הַכְּלִי הַמְיֻחָד : הַכְּלִי נִשְׁאָר בְּאִסּוּרוֹ. עיין סימן ע״ח · 78.

שִׁעוּר הָרְבִיעִית וְנִבְלַע בְּבֶגֶד

שִׁעוּר הָרְבִיעִית וְנִבְלַע בְּבֶגֶד

היסוד. רְבִיעִית לְפַעַם אֶחָד (הַרְבֵּה אוֹ מְעַט) ; ב׳ פְּעָמִים — ב׳ רְבִיעִית ; ג׳ — ג׳, וְכֵן לְעוֹלָם. וְכָל זֶה כְּשֶׁהֵם בְּעַיִן (סעיף ב).

המסקנה. נִבְלַע בְּבֶגֶד — אַף שֶׁטּוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ — דַּי בְּמַטְפִּיחוֹ בַּמַּיִם ; וּבְשַׁבָּת מְנַגֵּב בּוֹ יָדָיו (סימן ש״ב) ; וְכֵן בָּאָרֶץ.

הפניה ללימוד. היקף הסימן — הַבִּטּוּל בִּרְבִיעִית, דִּין הָעָבִיט וְהַנִּבְלַע בְּבֶגֶד — נוגע להנהגת ק״ש ותפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ע״ז. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מב׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
בִּטּוּל מֵי רַגְלַיִם בִּרְבִיעִית או״ח ע״ז — אָסוּר לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם עַד שֶׁיִּטִּיל לְתוֹכָן רְבִיעִית מַיִם ; בֵּין עַל גַּבֵּי קַרְקַע בֵּין בִּכְלִי. מ״ב ע״ז — פְּרָטֵי הַבִּטּוּל וְאֹפֶן נְתִינַת הַמַּיִם. ערוה״ש ע״ז — גֶּדֶר הַבִּטּוּל בְּאִסּוּר דְּרַבָּנָן. דרך אדמו״ר הזקן — כֵּיוָן שֶׁאִסּוּרָן מִדִּבְרֵי סוֹפְרִים — מְבַטְּלָן בִּרְבִיעִית מַיִם, בֵּין בַּקַּרְקַע בֵּין בִּכְלִי, בֵּין הַמַּיִם תְּחִלָּה בֵּין מֵי הָרַגְלַיִם (סעיף א).
שִׁעוּר הָרְבִיעִית לְפִי הַפְּעָמִים או״ח ע״ז — רְבִיעִית לְפַעַם אֶחָד, ב׳ רְבִיעִיּוֹת לִשְׁתַּיִם, ג׳ לְשָׁלֹשׁ, וְכֵן לְעוֹלָם. מ״ב ע״ז — הַשִּׁעוּר אֵינוֹ תָּלוּי בַּכַּמּוּת אֶלָּא בְּמִסְפַּר הַפְּעָמִים. ערוה״ש ע״ז — חֶשְׁבּוֹן הָרְבִיעִיּוֹת בְּרִבּוּי פְּעָמִים. דרך אדמו״ר הזקן — פַּעַם אֶחָד בֵּין הַרְבֵּה בֵּין מְעַט — רְבִיעִית ; ב׳ פְּעָמִים — ב׳ רְבִיעִית, וְכֵן לְעוֹלָם (סעיף ב).
עָבִיט וְנִבְלַע בְּבֶגֶד או״ח ע״ז — וּבִלְבַד שֶׁלֹּא יְהֵא עָבִיט הַמְיֻחָד לָהֶם [וְעַיֵּן סִימָן פ״ז]. מ״ב ע״ז — דִּין הַכְּלִי הַמְיֻחָד וְשֶׁאֵין הַמַּיִם מוֹעִילִים לוֹ. ערוה״ש ע״ז — גֶּדֶר הָעָבִיט וְהַנִּבְלַע. דרך אדמו״ר הזקן — כְּלִי הַמְיֻחָד — אֲפִלּוּ רֵיק אָסוּר, וְאֵין הַמַּיִם מַכְשִׁירִים אוֹתוֹ (סימן פ״ז) ; נִבְלַע בְּבֶגֶד — מַטְפִּיחוֹ בַּמַּיִם דַּיּוֹ, וּבְשַׁבָּת מְנַגֵּב בּוֹ יָדָיו (סימן ש״ב) (סעיף ב).

הטבלה נערכה מטקסט המחבר (אורח חיים ע״ז, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ב׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַבִּטּוּל בִּרְבִיעִית

קו א — הַבִּטּוּל בִּרְבִיעִית

יסוד הסימן : אִסּוּר מֵי רַגְלַיִם מִדִּבְרֵי סוֹפְרִים ; מְבַטְּלָן בִּרְבִיעִית מַיִם (סעיף א).

קריאת הרב : אֵין נַפְקָא מִינָה בַּסֵּדֶר — הַמַּיִם קֹדֶם אוֹ מֵי הָרַגְלַיִם, עַל הַקַּרְקַע אוֹ בִּכְלִי. עיין סימן ע״ו · 76.

מסקנה למעשה : שְׁפִיכַת רְבִיעִית מַיִם מְסִירָה אֶת הַמְּנִיעָה וּמַתֶּרֶת לִקְרוֹת אוֹ לְהִתְפַּלֵּל כְּנֶגֶד הַמָּקוֹם.

קו ב — הַכְּלִי הַמְיֻחָד

קו ב — הַכְּלִי הַמְיֻחָד

יסוד הסימן : כְּלִי הַמְיֻחָד לְמֵי רַגְלַיִם — אֲפִלּוּ רֵיק אָסוּר, וְאֵין הַמַּיִם מַכְשִׁירִים אוֹתוֹ (סעיף א ; סימן פ״ז).

קריאת הרב : הַבִּטּוּל פּוֹעֵל עַל הַחֹמֶר שֶׁבְּעַיִן, לֹא עַל מַעֲמַד הַכְּלִי הַמְיֻחָד. עיין סימן ע״ח · 78.

מסקנה למעשה : עָבִיט הַמְיֻחָד נִשְׁאָר בְּאִסּוּרוֹ — נְתִינַת מַיִם בְּתוֹכוֹ אֵינָהּ מוֹעִילָה.

קו ג — שִׁעוּר וְנִבְלַע

קו ג — שִׁעוּר וְנִבְלַע

יסוד הסימן : רְבִיעִית לְכָל פַּעַם (הַרְבֵּה אוֹ מְעַט) ; ב׳ פְּעָמִים — ב׳ רְבִיעִית ; וְכֵן לְעוֹלָם (סעיף ב).

קריאת הרב : כָּל זֶה כְּשֶׁהֵם בְּעַיִן ; נִבְלַע בְּבֶגֶד — מַטְפִּיחוֹ בַּמַּיִם דַּיּוֹ (סימן ש״ב, נִגּוּב יָדַיִם בְּשַׁבָּת). עיין סימן ע״ח · 78.

מסקנה למעשה : לְמְנוֹת רְבִיעִית לְכָל הַשְׁתָּנָה ; וּבַנִּבְלַע — דַּי בְּהַטְפָּחָה בְּמַיִם.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ע״ז ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ע״ז (דִּין שֶׁלֹּא לִקְרוֹת כְּנֶגֶד מֵי רַגְלַיִם)

  • ① בִּטּוּל בִּרְבִיעִית. כְּנֶגֶד מֵי רַגְלַיִם שֶׁנָּפְלוּ לָאָרֶץ, שְׁפִיכַת רְבִיעִית מַיִם מְבַטֶּלֶת אֶת הָאִסּוּר (דְּרַבָּנָן) וּמַתֶּרֶת לִקְרוֹת. יסוד : שוע״ר ע״ז סעיף א.
  • ② סֵדֶר הַנְּתִינָה. אַף בִּכְלִי — רְבִיעִית מַסְפֶּקֶת, וְאֵין נַפְקָא מִינָה אִם הַמַּיִם אוֹ מֵי הָרַגְלַיִם קָדְמוּ. יסוד : שוע״ר ע״ז סעיף א.
  • ③ כְּלִי הַמְיֻחָד (עָבִיט). כְּלִי הַמְיֻחָד לְמֵי רַגְלַיִם — אֲפִלּוּ רֵיק — אָסוּר, וְאֵין הַמַּיִם מַכְשִׁירִים אוֹתוֹ (עיין סימן פ״ז). יסוד : שוע״ר ע״ז סעיף א.
  • ④ שִׁעוּר לְפִי הַפְּעָמִים. רְבִיעִית לְכָל הַשְׁתָּנָה (הַרְבֵּה אוֹ מְעַט) ; ב׳ פְּעָמִים — ב׳ רְבִיעִית ; ג׳ — ג׳, וְכֵן לְעוֹלָם. יסוד : שוע״ר ע״ז סעיף ב.
  • ⑤ נִבְלַע בְּבֶגֶד / בָּאָרֶץ. נִבְלַע (אַף טוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ) — דַּי בְּמַטְפִּיחוֹ בַּמַּיִם ; וּבְשַׁבָּת מְנַגֵּב בּוֹ יָדָיו (סימן ש״ב). יסוד : שוע״ר ע״ז סעיף ב.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ע״ז (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַבִּטּוּל בִּרְבִיעִית, הָעָבִיט, הַנִּבְלַע בְּבֶגֶד — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
