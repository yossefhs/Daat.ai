דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ע״ד · שֶׁלֹּא לְבָרֵךְ כְּשֶׁאֲבָרָיו רוֹאִים אֶת הָעֶרְוָה

דרכו של אדמו״ר הזקן בדִּין הַכִּסּוּי לִקְרִיאַת שְׁמַע : וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר — כִּסּוּי הָעֶרְוָה מן התורה, וְלֵב רוֹאֶה אֶת הָעֶרְוָה — הֶפְסֵק בֵּין הַלֵּב לָעֶרְוָה ; הַמַּיִם כְּכִסּוּי וְהַמְּחַבֵּק בִּזְרוֹעוֹתָיו ; אֲבָרָיו נוֹגְעִים וְחֶגֶר הַמָּתְנַיִם — ק״ש מוּל תְּפִלָּה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ע״ד — שֶׁלֹּא לְבָרֵךְ כְּשֶׁאֲבָרָיו רוֹאִים אֶת הָעֶרְוָה. אצל הרב הסימן נפרש לז׳ סְעִיפִים : יְסוֹד הַכִּסּוּי וְהַלֵּב רוֹאֶה אֶת הָעֶרְוָה, הָעֶרְוָה מְגֻלָּה וְהַכִּלָּה, הַמַּיִם כְּכִסּוּי, הַמְּחַבֵּק בִּזְרוֹעוֹתָיו, הָאִשָּׁה וְהֶחָלוּק, אֲבָרָיו נוֹגְעִים וְהַכִּיס, וְלַסּוֹף חֲגוּרַת מָתְנַיִם עִם הַחִלּוּק שֶׁבֵּין ק״ש לִתְפִלָּה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ע״ד — ז׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ע״ד). אדמו״ר הזקן דן בו בְּדִין שֶׁלֹּא לְבָרֵךְ כְּשֶׁאֲבָרָיו רוֹאִים אֶת הָעֶרְוָה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ע״ד — שֶׁלֹּא לְבָרֵךְ כְּשֶׁאֲבָרָיו רוֹאִים אֶת הָעֶרְוָה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ע״ד — שֶׁלֹּא לְבָרֵךְ כְּשֶׁאֲבָרָיו רוֹאִים אֶת הָעֶרְוָה — וּבוֹ ז׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.74), ואחריו תרגומנו. הסימן כולל ז׳ סעיפים (סעיף k → ‎.74.k).

סעיף א וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר ; שֶׁלֹּא יִהְיֶה הַלֵּב עִם הָעֶרְוָה תַּחַת מִכְסֶה אֶחָד בְּלִי הֶפְסֵק.

שֶׁלֹּא לְבָרֵךְ כְּשֶׁאֲבָרָיו רוֹאִים אֶת הָעֶרְוָה וּבוֹ ז׳ סְעִיפִים:כָּתוּב בַּתּוֹרָה: ״וְשַׁבְתָּ וְכִסִּיתָ וְכו׳, כִּי ה׳ אֱלֹהֶיךָ מִתְהַלֵּךְ בְּקֶרֶב מַחֲנֶךָ וְגו׳, וְהָיָה מַחֲנֶיךָ קָדוֹשׁ, וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר וְגו׳״. מִכָּאן לָמְדוּ חֲכָמִים שֶׁבְּכָל מְקוֹם שֶׁה׳ אֱלֹהֵינוּ מִתְהַלֵּךְ עִמָּנוּ, דְּהַיְנוּ כְּשֶׁאָנוּ עוֹסְקִים בִּקְרִיאַת שְׁמַע וּבִתְפִלָּה אוֹ בְּדִבְרֵי תּוֹרָה — צָרִיךְ לִהְיוֹת הַמַּחֲנֶה קָדוֹשׁ, שֶׁלֹּא תְּהֵא צוֹאָה מְגֻלָּה שָׁם, וְשֶׁלֹּא יִרְאֶה ה׳ בָּנוּ עֶרְוַת דָּבָר, דְּהַיְנוּ שֶׁלֹּא יִהְיֶה דְּבַר עֶרְוָה כְּנֶגֶד פָּנָיו שֶׁל אָדָם הַקּוֹרֵא אוֹ מִתְפַּלֵּל כִּמְלֹא עֵינָיו, מִפְּנֵי שֶׁעִקָּר מַחֲנֵהוּ שֶׁל אָדָם הוּא כְּנֶגֶד פָּנָיו, וְצָרִיךְ שֶׁיִּהְיֶה מַחֲנֶה זֶה קָדוֹשׁ, שֶׁלֹּא יִרְאֶה בּוֹ ה׳ עֶרְוַת דָּבָר. וְאִסּוּר זֶה הוּא מִן הַתּוֹרָה. וְהַחֲכָמִים הוֹסִיפוּ שֶׁהַמְּקוֹמוֹת אֲשֶׁר ה׳ שׁוֹכֵן בָּהֶם בְּקִרְבֵּנוּ, שֶׁהֵם רֹאשׁוֹ וְלִבּוֹ שֶׁל אָדָם הַקּוֹרֵא אוֹ מִתְפַּלֵּל — צָרִיךְ גַּם כֵּן שֶׁלֹּא יִהְיוּ רוֹאִים אֶת הָעֶרְוָה, דְּהַיְנוּ שֶׁלֹּא יִהְיֶה עִם הָעֶרְוָה תַּחַת מִכְסֶה אֶחָד בְּלִי הֶפְסֵק וַחֲצִיצָה בֵּינֵיהֶם (אַף עַל פִּי שֶׁמִּן הַתּוֹרָה אֵין בָּזֶה מִשּׁוּם ״וְלֹא יִרְאֶה בְךָ״ כו׳, כֵּיוָן שֶׁעֶרְוָתוֹ מְכֻסָּה נֶגֶד בְּנֵי אָדָם). לְפִיכָךְ אִם הָיָה יָשֵׁן עָרֹם וּמִתְכַּסֶּה בְּטַלִּיתוֹ, אַף עַל פִּי שֶׁהוֹצִיא רֹאשׁוֹ חוּץ לְטַלִּיתוֹ שֶׁלֹּא יִהְיֶה עִם עֶרְוָתוֹ תַּחַת מִכְסֶה אֶחָד — הֲרֵי זֶה לֹא יִקְרָא עַד שֶּׁיָּחֹץ בְּטַלִּיתוֹ עַל לִבּוֹ לְהַפְסִיק בֵּין לִבּוֹ לָעֶרְוָה, שֶׁלִּבּוֹ רוֹאֶה אֶת הָעֶרְוָה אָסוּר. וְהוּא הַדִּין אִם לִבּוֹ רוֹאֶה אֶת עֶרְוַת חֲבֵרוֹ, דְּהַיְנוּ שֶׁאֵין הֶפְסֵק בֵּין לִבּוֹ לְעֶרְוַת חֲבֵרוֹ, כְּגוֹן שֶׁשְּׁנֵיהֶם שׁוֹכְבִים עֲרֻמִּים וּמִתְכַּסִּים בְּטַלִּית אֶחָד וְהֵם רְחוֹקִים זֶה מִזֶּה שֶׁאֵין בְּשָׂרָם נוֹגֵעַ אֶלָּא שֶׁפְּנֵיהֶם זֶה כְּנֶגֶד זֶה:

וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר. כָּתוּב בַּתּוֹרָה ״וְהָיָה מַחֲנֶיךָ קָדוֹשׁ, וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר״ : בְּעֵת ק״ש, תְּפִלָּה וְדִבְרֵי תּוֹרָה — הַמַּחֲנֶה כְּנֶגֶד פָּנָיו (כִּמְלֹא עֵינָיו) צָרִיךְ לִהְיוֹת קָדוֹשׁ, וְאִסּוּר עֶרְוָה מְגֻלָּה הוּא מִן הַתּוֹרָה. וְהַחֲכָמִים הוֹסִיפוּ שֶׁהָרֹאשׁ וְהַלֵּב לֹא יִהְיוּ עִם הָעֶרְוָה תַּחַת מִכְסֶה אֶחָד בְּלִי הֶפְסֵק. לְפִיכָךְ הַיָּשֵׁן עָרֹם בְּטַלִּיתוֹ, אַף שֶׁהוֹצִיא רֹאשׁוֹ — לֹא יִקְרָא עַד שֶׁיָּחֹץ בַּטַּלִּית עַל לִבּוֹ ; לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה — אָסוּר. וְהוּא הַדִּין בְּעֶרְוַת חֲבֵרוֹ (שְׁנַיִם עֲרֻמִּים תַּחַת טַלִּית אֶחָד).

סעיף ב הָעֶרְוָה מְגֻלָּה — אָסוּר מן התורה ; אֲבָל כִּלָּה שֶׁאֵינָהּ גָּבוֹהַּ י׳ טְפָחִים — כִּמְכֻסֶּה.

אֲפִלּוּ אִם לִבּוֹ מְכֻסֶּה בְּבֶגֶד לְהַפְסִיק בֵּינוֹ לָעֶרְוָה — צָרִיךְ שֶׁתְּהֵא עֶרְוָתוֹ גַּם כֵּן מְכֻסָּה. וְאִם הוּא מְגֻלֶּה — אָסוּר לִקְרוֹת מִן הַתּוֹרָה, אֲפִלּוּ אִם הֶחֱזִיר פָּנָיו מִמֶּנָּה שֶׁלֹּא יִרְאֶנָּה אוֹ שֶׁעָצַם עֵינָיו, שֶׁהֲרֵי לֹא נֶאֱמַר ״לֹא תִּרְאֶה עֶרְוַת דָּבָר״, אֶלָּא ״לֹא יִרְאֶה בְךָ״, כְּלוֹמַר שֶׁתְּהֵא מְכֻסֶּה בְּעִנְיָן שֶׁלֹּא תְּהֵא נִרְאֶה לְאַחֵר הָעוֹמֵד אֶצְלוֹ. לְפִיכָךְ הָעוֹמֵד בַּבַּיִת עָרֹם וְהוֹצִיא רֹאשׁוֹ חוּץ לַחַלּוֹן שֶׁלֹּא יִרְאֶה עֶרְוָתוֹ, אַף עַל פִּי שֶׁכִּסָּה לִבּוֹ בְּבֶגֶד — הֲרֵי זֶה לֹא יִקְרָא, כֵּיוָן שֶׁעֶרְוָתוֹ מִגֻלָּה וְנִרְאֵית בַּבַּיִת. אֲבָל הַיָּשֵׁן עָרֹם בְּכִלָּה שֶׁאֵינוֹ גָּבוֹהַּ י׳ טְפָחִים וְהוֹצִיא רֹאשׁוֹ חוּצָה לָהּ וְגַם כִּסָּה בְּבֶגֶד עַל לִבּוֹ, אַף עַל פִּי שֶׁעֶרְוָתוֹ מִגֻלָּה בַּכִּלָּה — הֲרֵי זֶה כִּמְכֻסֶּה בְּטַלִּית, שֶׁכֵּיוָן שֶׁאֵין הַכִּלָּה גָּבוֹהַּ י׳ טְפָחִים שֵׁם כִּסּוּי עָלֶיהָ, וְאֵינָהּ דּוֹמָה לְבַיִת שֶׁהוּא חָלוּק רְשׁוּת לְעַצְמוֹ, וְכֵיוָן שֶׁהָעֶרְוָה נִרְאֵית בְּתוֹכוֹ — אָסוּר:

הָעֶרְוָה מְגֻלָּה. אַף שֶׁהַלֵּב מְכֻסֶּה — צָרִיךְ שֶׁהָעֶרְוָה עַצְמָהּ תְּהֵא מְכֻסָּה ; מְגֻלָּה — אָסוּר מִן הַתּוֹרָה, אֲפִלּוּ הֶחֱזִיר פָּנָיו אוֹ עָצַם עֵינָיו, שֶׁנֶּאֱמַר ״לֹא יִרְאֶה בְךָ״ (וְלֹא ״לֹא תִּרְאֶה״) — שֶׁתְּהֵא מְכֻסָּה כְּלַפֵּי אֲחֵרִים. לְפִיכָךְ עוֹמֵד עָרֹם בַּבַּיִת וּמוֹצִיא רֹאשׁוֹ לַחַלּוֹן — לֹא יִקְרָא, שֶׁעֶרְוָתוֹ נִרְאֵית בַּבַּיִת. אֲבָל הַיָּשֵׁן בְּכִלָּה שֶׁאֵינָהּ גָּבוֹהַּ י׳ טְפָחִים, רֹאשׁוֹ חוּץ וְלֵב מְכֻסֶּה — כִּמְכֻסֶּה בְּטַלִּית, שֶׁאֵין הַכִּלָּה רְשׁוּת לְעַצְמָהּ.

סעיף ג מַיִם אֲפִלּוּ צְלוּלִין חֲשׁוּבִים כִּסּוּי ; וּבִלְבַד שֶׁיּוֹצִיא רֹאשׁוֹ וְלִבּוֹ חוּץ לַמַּיִם.

מַיִם אֲפִלּוּ הֵן צְלוּלִין חֲשׁוּבִים כִּסּוּי לָעֶרְוָה, וּבִלְבַד שֶׁיּוֹצִיא רֹאשׁוֹ וְלִבּוֹ חוּץ לַמַּיִם, שֶׁלֹּא יִהְיֶה עִם עֶרְוָתוֹ תַּחַת מִכְסֶה אֶחָד כְּמוֹ שֶׁנִּתְבָּאֵר סָעִיף א׳. וְאִם לִבּוֹ תּוֹךְ הַמַּיִם, אִם הֵם צְלוּלִים — אָסוּר מִשּׁוּם לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה. וְאִם הֵן עֲכוּרִים שֶׁאֵין אֲבָרָיו נִרְאִים בָּהֶם — מֻתָּר לִקְרוֹת בְּתוֹכָם, וְהוּא מִפְּנֵי שֶׁהֵם חֲשׁוּבִים הֶפְסֵק בֵּין לִבּוֹ לָעֶרְוָה. וְהוּא שֶׁלֹּא יִהְיֶה רֵיחָם רַע, שֶׁאָז צָרִיךְ לְהַרְחִיק מֵהֶם ד׳ אַמּוֹת כְּמוֹ שֶׁיִּתְבָּאֵר לְקַמָּן. וּמַיִם צְלוּלִים שֶׁעַל גַּבֵּי קַרְקַע עוֹכְרָן בְּרַגְלָיו וּמֻתָּר. וְהָרוֹחֵץ בְּמַיִם צְלוּלִים שֶׁבַּכְּלִי שֶׁאִי אֶפְשָׁר לְעָכְרָן וְרוֹצֶה לִשְׁתּוֹת, אִם לִבּוֹ תּוֹךְ הַמַּיִם — יְכַסֶּה בַּבֶּגֶד מִמַּטָּה לְלִבּוֹ וִיבָרֵךְ. אֲבָל אִם כִּסָּהוּ בְּיָדוֹ — אֵינוֹ חָשׁוּב כִּסּוּי, שֶׁהַלֵּב וְהַיָּד גּוּף אֶחָד הֵן וְאֵין הַגּוּף יָכוֹל לְכַסּוֹת אֶת עַצְמוֹ. וְהוּא הַדִּין אִם מְכַסֶּה אֶת רֹאשׁוֹ בְּיָדָיו — אֵינוֹ חָשׁוּב כִּסּוּי שֶׁיּוּכַל לְבָרֵךְ כָּךְ בְּגִלּוּי הָרֹאשׁ, כְּמוֹ שֶׁיִּתְבָּאֵר לְקַמָּן בְּסִימָן צ״א:

מַיִם כְּכִסּוּי. מַיִם אֲפִלּוּ צְלוּלִים חֲשׁוּבִים כִּסּוּי לָעֶרְוָה, וּבִלְבַד שֶׁרֹאשׁוֹ וְלִבּוֹ חוּץ לַמַּיִם. לִבּוֹ בְּמַיִם צְלוּלִים — אָסוּר (לִבּוֹ רוֹאֶה) ; מַיִם עֲכוּרִים שֶׁאֵין אֲבָרָיו נִרְאִים — מֻתָּר (הֶפְסֵק), וּבִלְבַד שֶׁאֵין רֵיחָם רַע (וְאִם לָאו — ד׳ אַמּוֹת). מַיִם צְלוּלִים שֶׁעַל הַקַּרְקַע — עוֹכְרָן בְּרַגְלָיו. בַּכְּלִי שֶׁאִי אֶפְשָׁר לְעָכְרוֹ וְרוֹצֶה לִשְׁתּוֹת — יְכַסֶּה בְּבֶגֶד מִמַּטָּה לְלִבּוֹ. יָד אֵינָהּ כִּסּוּי (הַלֵּב וְהַיָּד גּוּף אֶחָד), וְהוּא הַדִּין הַמְכַסֶּה רֹאשׁוֹ בְּיָדָיו (סימן צ״א).

סעיף ד מְחַבֵּק גּוּפוֹ בִּזְרוֹעוֹתָיו — חֲשׁוּבָה הַפְסָקָה בֵּין לִבּוֹ לָעֶרְוָה.

אִם הָאָדָם מְחַבֵּק גּוּפוֹ בִּזְרוֹעוֹתָיו — הֲרֵי זֶה חֲשׁוּבָה הַפְסָקָה בֵּין לִבּוֹ לָעֶרְוָה, שֶׁאַף שֶׁהַיָּד אֵינָהּ חֲשׁוּבָה כִּסּוּי — חֲשׁוּבָה הַפְסָקָה, כֵּיוָן שֶׁמִּכָּל מָקוֹם אֵין לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה:

מְחַבֵּק בִּזְרוֹעוֹתָיו. הַמְחַבֵּק גּוּפוֹ בִּזְרוֹעוֹתָיו — חֲשׁוּבָה הַפְסָקָה בֵּין לִבּוֹ לָעֶרְוָה : אַף שֶׁהַיָּד אֵינָהּ כִּסּוּי — חֲשׁוּבָה הַפְסָקָה, שֶׁמִּכָּל מָקוֹם אֵין לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה.

סעיף ה הַנָּשִׁים לְבוּשׁוֹת הֶחָלוּק — יְכוֹלוֹת לְבָרֵךְ ; עֲרֻמּוֹת וְיוֹשְׁבוֹת — מֻתֶּרֶת.

יֵשׁ אוֹמְרִים שֶׁהַנָּשִׁים יְכוֹלוֹת לְבָרֵךְ וּלְהִתְפַּלֵּל כְּשֶׁהֵן לְבוּשׁוֹת הֶחָלוּק אַף עַל פִּי שֶׁאֵינָן מַפְסִיקוֹת לְמַטָּה מִלִּבָּן, לְפִי שֶׁעֶרְוָתָן לְמַטָּה מְאֹד וְאֵין לִבָּם רוֹאֶה אֶת הָעֶרְוָה. וּמִכָּל מָקוֹם אִם הֵן עֲרֻמּוֹת, כֵּיוָן שֶׁעֶרְוָתָן מִגֻלָּה וְנִרְאֵית לַאֲחֵרִים — אָסוּר מִשּׁוּם ״וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר וְגו׳״. וְאִם הִיא יוֹשֶׁבֶת שֶׁאָז אֵין פָּנֶיהָ שֶׁל מַטָּה נִרְאֵית — מֻתֶּרֶת לְבָרֵךְ אַף עַל פִּי שֶׁהִיא עֲרֻמָּה, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ר״ו:

הָאִשָּׁה וְהֶחָלוּק. יש אומרים : הַנָּשִׁים יְכוֹלוֹת לְבָרֵךְ וּלְהִתְפַּלֵּל לְבוּשׁוֹת הֶחָלוּק אַף שֶׁאֵינָן מַפְסִיקוֹת לְמַטָּה מִלִּבָּן, שֶׁעֶרְוָתָן לְמַטָּה מְאֹד וְאֵין לִבָּן רוֹאֶה. עֲרֻמּוֹת — אָסוּר, שֶׁעֶרְוָתָן נִרְאֵית לַאֲחֵרִים (״וְלֹא יִרְאֶה בְךָ״). יוֹשֶׁבֶת שֶׁאֵין פָּנֶיהָ שֶׁל מַטָּה נִרְאֵית — מֻתֶּרֶת אַף עֲרֻמָּה (סימן ר״ו).

סעיף ו שְׁאָר אֲבָרָיו רוֹאִים — מֻתָּר ; נוֹגֵעַ בְּעֶרְוָתוֹ — אָסוּר ; אֲבָל בַּכִּיס אֵין לָחוּשׁ.

שְׁאָר אֲבָרָיו רוֹאִים אֶת הָעֶרְוָה — מֻתָּר. אֲבָל אִם אֵיזֶה מֵאֲבָרָיו נוֹגֵעַ בְּעֶרְוָתוֹ אוֹ בְּעֶרְוַת חֲבֵרוֹ — אָסוּר לִקְרוֹת אוֹ לְהִתְפַּלֵּל. וְיַרְיכוֹתָיו שֶׁהָעֶרְוָה שׁוֹכֶבֶת עֲלֵיהֶן — צָרִיךְ לְהַפְסִיקָן בְּבֶגֶד, אוֹ לְהַרְחִיקָן בְּעִנְיָן שֶׁלֹּא יִגַּע הַגִּיד בָּהֶן. אֲבָל בַּכִּיס אֵין לָחוּשׁ אִם נוֹגֵעַ בָּהֶן, לְפִי שֶׁלֹּא אָסְרוּ אֲבָרָיו נוֹגְעִים בְּעֶרְוָתוֹ אֶלָּא מִשּׁוּם גְּזֵירָה אַטּוּ יָדוֹ נוֹגַעַת בְּעֶרְוָתוֹ שֶׁאָז יָבוֹא לִידֵי הִרְהוּר, לְפִיכָךְ בַּכִּיס שֶׁאֵין לָחוּשׁ בּוֹ מִשּׁוּם הִרְהוּר אַף אִם הָיְתָה נוֹגַעַת בּוֹ — אֵין לִגְזֹר בּוֹ מִשּׁוּם אֲבָרָיו רוֹאִים אֶת הָעֶרְוָה, אַף עַל פִּי שֶׁהוּא עֶרְוָה גְּמוּרָה לְעִנְיָן שֶׁאִם הוּא מְגֻלֶּה אָסוּר לִקְרוֹת כְּנֶגְדּוֹ מִשּׁוּם ״וְלֹא יִרְאֶה בְךָ וְגו׳״:

אֲבָרָיו נוֹגְעִים. שְׁאָר אֲבָרָיו רוֹאִים אֶת הָעֶרְוָה — מֻתָּר. אֲבָל אֵבֶר הַנּוֹגֵעַ בְּעֶרְוָתוֹ אוֹ בְּעֶרְוַת חֲבֵרוֹ — אָסוּר לִקְרוֹת אוֹ לְהִתְפַּלֵּל. יְרֵכַיִם שֶׁהָעֶרְוָה שׁוֹכֶבֶת עֲלֵיהֶן — לְהַפְסִיקָן בְּבֶגֶד אוֹ לְהַרְחִיקָן. אֲבָל בַּכִּיס אֵין לָחוּשׁ, שֶׁאִסּוּר הַנְּגִיעָה אֵינוֹ אֶלָּא גְּזֵירָה אַטּוּ יָד (שֶׁמָּא יָבוֹא לִידֵי הִרְהוּר), וְאֵינָהּ בַּכִּיס — אַף שֶׁהוּא עֶרְוָה גְּמוּרָה לְ״לֹא יִרְאֶה בְךָ״.

סעיף ז טַלִּית חֲגוּרָה עַל מָתְנָיו — מֻתָּר לִקְרוֹת ק״ש ; אֲבָל לְהִתְפַּלֵּל אָסוּר עַד שֶׁיְּכַסֶּה לִבּוֹ.

הָיְתָה טַלִּיתוֹ חֲגוּרָה לוֹ עַל מָתְנָיו לְכַסּוֹתוֹ מִמָּתְנָיו וּלְמַטָּה, אַף עַל פִּי שֶׁמִּמָּתְנָיו וּלְמַעְלָה הוּא עָרֹם — מֻתָּר לִקְרוֹת קְרִיאַת שְׁמַע, כֵּיוָן שֶׁעֶרְוָתוֹ מְכֻסָּה, וְגַם הַטַּלִּית מַפְסֶקֶת בֵּין לִבּוֹ לָעֶרְוָה. אֲבָל לְהִתְפַּלֵּל אָסוּר עַד שֶׁיְּכַסֶּה לִבּוֹ, לְפִי שֶׁבַּתְּפִלָּה צָרִיךְ לִרְאוֹת אֶת עַצְמוֹ כְּעוֹמֵד לִפְנֵי הַמֶּלֶךְ וּמְדַבֵּר עִמּוֹ שֶׁצָּרִיךְ לַעֲמֹד בְּאֵימָה, אֲבָל בִּקְרִיאַת שְׁמַע אֵינוֹ מְדַבֵּר עִם הַמֶּלֶךְ:

חֲגוּרַת מָתְנַיִם. טַלִּיתוֹ חֲגוּרָה עַל מָתְנָיו לְכַסּוֹתוֹ מִמָּתְנָיו וּלְמַטָּה, אַף שֶׁלְּמַעְלָה הוּא עָרֹם — מֻתָּר לִקְרוֹת ק״ש, שֶׁעֶרְוָתוֹ מְכֻסָּה וְהַטַּלִּית מַפְסֶקֶת בֵּין לִבּוֹ לָעֶרְוָה. אֲבָל לְהִתְפַּלֵּל אָסוּר עַד שֶׁיְּכַסֶּה לִבּוֹ, שֶׁבַּתְּפִלָּה צָרִיךְ לִרְאוֹת עַצְמוֹ כְּעוֹמֵד לִפְנֵי הַמֶּלֶךְ בְּאֵימָה — מַה שֶּׁאֵין כֵּן בְּק״ש שֶׁאֵינוֹ מְדַבֵּר עִם הַמֶּלֶךְ.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ע״ד (מהדורת קה״ת, ז׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — שֶׁלֹּא לְבָרֵךְ כְּשֶׁאֲבָרָיו רוֹאִים אֶת הָעֶרְוָה

היסוד : הַכִּסּוּי וְהַלֵּב רוֹאֶה אֶת הָעֶרְוָה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה ה״לֹא יִרְאֶה בְךָ״ — יְסוֹד הַכִּסּוּי מִן הַתּוֹרָה. אחר כך הַלֵּב רוֹאֶה אֶת הָעֶרְוָה — תּוֹסֶפֶת חֲכָמִים. אחר כך הַכִּסּוּי וְהַהֶפְסֵק — מַיִם, זְרוֹעוֹת וְיָד. ולבסוף אֲבָרָיו נוֹגְעִים וְחֲגוּרַת מָתְנַיִם — ק״ש מוּל תְּפִלָּה.

וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר

וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר — יְסוֹד הַתּוֹרָה

היסוד. בְּעֵת ק״ש, תְּפִלָּה וְדִבְרֵי תּוֹרָה הַמַּחֲנֶה קָדוֹשׁ ; עֶרְוָה מְגֻלָּה כְּנֶגְדּוֹ — אָסוּר מִן הַתּוֹרָה, אַף הֶחֱזִיר פָּנָיו אוֹ עָצַם עֵינָיו — ״לֹא יִרְאֶה בְךָ״ וְלֹא ״לֹא תִּרְאֶה״ (סעיפים א-ב).

המסקנה. הָעֶרְוָה צְרִיכָה כִּסּוּי כְּלַפֵּי אֲחֵרִים ; כִּלָּה נְמוּכָה (פָּחוֹת מִי׳ טְפָחִים) חֲשׁוּבָה כִּסּוּי. עיין סימן 73 (כיסוי בשכיבה).

לֵב רוֹאֶה אֶת הָעֶרְוָה

לֵב רוֹאֶה אֶת הָעֶרְוָה — תּוֹסֶפֶת חֲכָמִים

היסוד. הַחֲכָמִים הוֹסִיפוּ שֶׁלֹּא יִהְיוּ הָרֹאשׁ וְהַלֵּב עִם הָעֶרְוָה תַּחַת מִכְסֶה אֶחָד בְּלִי הֶפְסֵק ; לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה — אָסוּר (סעיף א).

המסקנה. הַיָּשֵׁן עָרֹם בְּטַלִּיתוֹ חוֹצֵץ בַּטַּלִּית עַל לִבּוֹ לְהַפְסִיק. עיין סימן 75 (ערוה כנגד הקורא).

כִּסּוּי וְהֶפְסֵק — מַיִם, זְרוֹעוֹת וְיָד

כִּסּוּי וְהֶפְסֵק — מַיִם, זְרוֹעוֹת וְיָד

היסוד. מַיִם אֲפִלּוּ צְלוּלִים כִּסּוּי אִם רֹאשׁ וְלֵב חוּץ ; עֲכוּרִים — הֶפְסֵק ; זְרוֹעוֹת הַמְחַבְּקוֹת — הַפְסָקָה, אֲבָל יָד אֵינָהּ כִּסּוּי (גּוּף אֶחָד) (סעיפים ג-ד).

המסקנה. הַרַב מַבְחִין הֵיטֵב בֵּין כִּסּוּי (בֶּגֶד) לְהֶפְסֵק (בֵּין הַלֵּב לָעֶרְוָה). עיין סימן 76 (מקום נקי).

אֲבָרָיו נוֹגְעִים וְחֶגֶר הַמָּתְנַיִם

אֲבָרָיו נוֹגְעִים וְחֶגֶר הַמָּתְנַיִם

היסוד. נוֹגֵעַ בָּעֶרְוָה — אָסוּר (גְּזֵירָה) ; כִּיס פָּטוּר. הָאִשָּׁה בַּחָלוּק — מֻתֶּרֶת (סעיפים ה-ו).

המסקנה. טַלִּית חֲגוּרָה עַל מָתְנָיו דַּי לְק״ש אֲבָל לֹא לִתְפִלָּה, שֶׁצָּרִיךְ לְכַסּוֹת לִבּוֹ (סעיף ז).

הפניה ללימוד. היקף הסימן — כִּסּוּי הָעֶרְוָה, הַהֶפְסֵק וְהַחִלּוּק שֶׁבֵּין ק״ש לִתְפִלָּה — נוגע להנהגת ק״ש ותפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלשון צנועה ובלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

שֶׁלֹּא לְבָרֵךְ כְּשֶׁאֲבָרָיו רוֹאִים אֶת הָעֶרְוָה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ע״ד. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מז׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
לֹא יִרְאֶה בְךָ וְכִסּוּי הָעֶרְוָה או״ח ע״ד — הָעֶרְוָה מְכֻסָּה כְּנֶגֶד אֲחֵרִים ; לֵב רוֹאֶה אֶת הָעֶרְוָה — אָסוּר. מ״ב ע״ד — גֶּדֶר ״לֹא יִרְאֶה בְךָ״ וַאֲפִלּוּ הֶחֱזִיר פָּנָיו. ערוה״ש ע״ד — הַחִלּוּק שֶׁבֵּין דְּאוֹרַיְתָא לְדִּין הַלֵּב. דרך אדמו״ר הזקן — אִסּוּר הָעֶרְוָה מְגֻלָּה מן התורה (״לֹא יִרְאֶה בְךָ״ וְלֹא ״לֹא תִּרְאֶה״) ; הַלֵּב עִם הָעֶרְוָה תַּחַת מִכְסֶה אֶחָד בְּלִי הֶפְסֵק — מִדִּבְרֵי חֲכָמִים (סעיפים א-ב).
מַיִם וְהֶפְסֵק בֵּין לֵב לָעֶרְוָה או״ח ע״ד — הָרוֹחֵץ בְּמַיִם צְלוּלִים יְכַסֶּה בְּבֶגֶד ; בַּיָּדַיִם לֹא הֲוֵי כִּסּוּי. מ״ב ע״ד — מַיִם עֲכוּרִים וְגֶדֶר הַהֶפְסֵק. ערוה״ש ע״ד — דִּין הַמַּיִם הַצְּלוּלִים וְהָעֲכוּרִים. דרך אדמו״ר הזקן — מַיִם צְלוּלִים כִּסּוּי (רֹאשׁ וְלֵב חוּץ) ; עֲכוּרִים הֶפְסֵק ; מְחַבֵּק בִּזְרוֹעוֹתָיו הַפְסָקָה ; יָד אֵינָהּ כִּסּוּי (סעיפים ג-ד).
הָאִשָּׁה וְהֶחָלוּק או״ח ע״ד — הַנָּשִׁים מְבָרְכוֹת בַּחָלוּק אַף שֶׁאֵינָן מַפְסִיקוֹת מִלְּמַטָּה מִלֵּב. מ״ב ע״ד — טַעַם שֶׁעֶרְוָתָן לְמַטָּה מְאֹד. ערוה״ש ע״ד — פְּרָטֵי הָאִשָּׁה הָעֲרֻמָּה וְהַיּוֹשֶׁבֶת. דרך אדמו״ר הזקן — בַּחָלוּק מֻתֶּרֶת (עֶרְוָתָן לְמַטָּה) ; עֲרֻמָּה — אָסוּר ; יוֹשֶׁבֶת — מֻתֶּרֶת (סימן ר״ו) (סעיף ה).
אֲבָרָיו נוֹגְעִים וְק״ש מוּל תְּפִלָּה או״ח ע״ד — נוֹגֵעַ בְּעֶרְוָתוֹ אָסוּר ; טַלִּית עַל מָתְנָיו — ק״ש מֻתָּר, תְּפִלָּה עַד שֶׁיְּכַסֶּה לֵב. מ״ב ע״ד — דִּין הַכִּיס וְהַיְרֵכַיִם. ערוה״ש ע״ד — טַעַם הַחִלּוּק שֶׁבֵּין ק״ש לִתְפִלָּה. דרך אדמו״ר הזקן — נוֹגֵעַ אָסוּר (גְּזֵירָה), כִּיס פָּטוּר ; חֲגוּרַת מָתְנַיִם — ק״ש מֻתָּר, תְּפִלָּה עַד שֶׁיְּכַסֶּה לִבּוֹ (כְּעוֹמֵד לִפְנֵי הַמֶּלֶךְ) (סעיפים ו-ז).

הטבלה נערכה מטקסט המחבר (אורח חיים ע״ד, מקור ספריא Shulchan_Arukh,_Orach_Chayim.74) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ד-כ״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ז׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — כִּסּוּי הָעֶרְוָה

קו א — כִּסּוּי הָעֶרְוָה

יסוד הסימן : הָעֶרְוָה מְגֻלָּה — אָסוּר מן התורה, אֲפִלּוּ הֶחֱזִיר פָּנָיו (סעיפים א-ב).

קריאת הרב : הָאִסּוּר שֶׁתְּהֵא נִרְאֵית לַאֲחֵרִים (״לֹא יִרְאֶה בְךָ״) ; כִּלָּה נְמוּכָה חֲשׁוּבָה כִּסּוּי. עיין סימן 73.

מסקנה למעשה : לֹא לִקְרוֹת וְלֹא לְהִתְפַּלֵּל אֶלָּא כְּשֶׁהָעֶרְוָה מְכֻסָּה כְּלַפֵּי אֲחֵרִים.

קו ב — הֶפְסֵק בֵּין הַלֵּב לָעֶרְוָה

קו ב — הֶפְסֵק בֵּין הַלֵּב לָעֶרְוָה

יסוד הסימן : לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה — אָסוּר ; מַיִם עֲכוּרִים וּזְרוֹעוֹת — הַפְסָקָה (סעיפים א, ג-ד).

קריאת הרב : הַרַב מַבְחִין הֵיטֵב בֵּין כִּסּוּי לְהֶפְסֵק — הַיָּד אֵינָהּ לֹא זֶה וְלֹא זֶה (גּוּף אֶחָד). עיין סימן 75.

מסקנה למעשה : לַחֲצֹץ בַּטַּלִּית עַל הַלֵּב, אוֹ לְחַבֵּק בִּזְרוֹעוֹתָיו, לְהַפְסִיק.

קו ג — ק״ש מוּל תְּפִלָּה

קו ג — ק״ש מוּל תְּפִלָּה

יסוד הסימן : חֲגוּרַת מָתְנַיִם — ק״ש מֻתָּר ; תְּפִלָּה — עַד שֶׁיְּכַסֶּה לִבּוֹ (סעיף ז).

קריאת הרב : הַתְּפִלָּה חֲמוּרָה יוֹתֵר — כְּעוֹמֵד לִפְנֵי הַמֶּלֶךְ בְּאֵימָה ; ק״ש אֵינוֹ ״מְדַבֵּר עִם הַמֶּלֶךְ״. עיין סימן 76.

מסקנה למעשה : חֲגוּרַת מָתְנַיִם דַּי לְק״ש ; לְכַסּוֹת הַלֵּב קֹדֶם הָעֲמִידָה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ע״ד ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד וּבְלָשׁוֹן צְנוּעָה, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ע״ד (שֶׁלֹּא לְבָרֵךְ כְּשֶׁאֲבָרָיו רוֹאִים אֶת הָעֶרְוָה)

  • ① כִּסּוּי הָעֶרְוָה. אֵין לִקְרוֹת אוֹ לְהִתְפַּלֵּל אֶלָּא כְּשֶׁהָעֶרְוָה מְכֻסָּה כְּלַפֵּי אֲחֵרִים ; הַחֲזָרַת פָּנִים אוֹ עֲצִימַת עֵינַיִם אֵינָהּ מוֹעֶלֶת (״לֹא יִרְאֶה בְךָ״). יסוד : שוע״ר ע״ד סעיפים א-ב.
  • ② לֵב רוֹאֶה. הַיָּשֵׁן עָרֹם בְּטַלִּיתוֹ — חוֹצֵץ בַּטַּלִּית עַל לִבּוֹ לְהַפְסִיק בֵּין הַלֵּב לָעֶרְוָה. יסוד : שוע״ר ע״ד סעיף א.
  • ③ מַיִם. מַיִם צְלוּלִים כִּסּוּי אִם רֹאשׁ וְלֵב חוּץ ; עֲכוּרִים — הֶפְסֵק ; הַיָּד אֵינָהּ כִּסּוּי. יסוד : שוע״ר ע״ד סעיף ג.
  • ④ מְחַבֵּק בִּזְרוֹעוֹתָיו. חִבּוּק הַגּוּף בִּזְרוֹעוֹתָיו חֲשׁוּבָה הַפְסָקָה בֵּין הַלֵּב לָעֶרְוָה. יסוד : שוע״ר ע״ד סעיף ד.
  • ⑤ נָשִׁים. לְבוּשׁוֹת הֶחָלוּק — מֻתָּרוֹת אַף בְּלִי הֶפְסֵק מִתַּחַת לַלֵּב ; עֲרֻמּוֹת וְיוֹשְׁבוֹת — מֻתָּרוֹת. יסוד : שוע״ר ע״ד סעיף ה.
  • ⑥ ק״ש מוּל תְּפִלָּה. חֲגוּרַת מָתְנַיִם (עָרֹם לְמַעְלָה) — ק״ש מֻתָּר ; לָעֲמִידָה — רַק אַחַר כִּסּוּי הַלֵּב. יסוד : שוע״ר ע״ד סעיפים ו-ז.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ע״ד (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַכִּסּוּי, הַהֶפְסֵק, הַחִלּוּק שֶׁבֵּין ק״ש לִתְפִלָּה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
