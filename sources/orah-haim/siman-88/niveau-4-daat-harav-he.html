דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן פ״ח · דִּין בַּעַל קֶרִי בִּקְרִיאַת שְׁמַע

דרכו של אדמו״ר הזקן בדִּין בַּעַל קֶרִי בִּקְרִיאַת שְׁמַע : כָּל הַטְּמֵאִים — קוֹרִין וּמִתְפַּלְלִין ; הַבַּעַל קֶרִי שֶׁהוֹצִיאוֹ עֶזְרָא — תַּקָּנָה שֶׁבָּטְלָה לְגַמְרֵי ; וְמִנְהַג טָהֳרָה שֶׁל נָשִׁים בִּשְׁעַת נִדּוּתָן — בֵּית הַכְּנֶסֶת וְחִיּוּב הַבְּרָכוֹת — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן פ״ח — דִּין בַּעַל קֶרִי בִּקְרִיאַת שְׁמַע. אצל הרב הסימן נפרש לב׳ סְעִיפִים : הַבַּעַל קֶרִי וְתַקָּנַת עֶזְרָא שֶׁבָּטְלָה (עִם פְּרָטֵי ט׳ קַבִּין לְמִי שֶׁנָּהַג בָּהּ), וְאַחַר כָּךְ מִנְהַג הַטָּהֳרָה שֶׁל הַנָּשִׁים בִּשְׁעַת נִדּוּתָן וְחִיּוּבָן בַּבְּרָכוֹת.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן פ״ח — ב׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim פ״ח). אדמו״ר הזקן דן בו בְּדִין בַּעַל קֶרִי בִּקְרִיאַת שְׁמַע. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן פ״ח — דִּין בַּעַל קֶרִי בִּקְרִיאַת שְׁמַע

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן פ״ח — בַּעַל קֶרִי מֻתָּר בִּקְרִיאַת שְׁמַע — וּבוֹ ב׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.88), ואחריו תרגומנו. הסימן כולל ב׳ סעיפים (סעיף k → ‎.88.k).

סעיף א כָּל הַטְּמֵאִים קוֹרִין ק״ש וּמִתְפַּלְלִין ; חוּץ מִבַּעַל קֶרִי — תַּקָּנַת עֶזְרָא שֶׁבָּטְלָה.

בַּעַל קֶרִי מֻתָּר בִּקְרִיאַת שְׁמַע וּבוֹ ב׳ סְעִיפִים:כָּל הַטְּמֵאִים, אֲפִלּוּ טֻמְאָה יוֹצֵא עֲלֵיהֶם מִגּוּפָן כְּגוֹן זָב וְזָבָה וְכַיּוֹצֵא בָּהֶן — קוֹרִין קְרִיאַת שְׁמַע וּמִתְפַּלְלִין. חוּץ מִבַּעַל קֶרִי, שֶׁהוֹצִיאוֹ עֶזְרָא מִן כְּלַל הַטְּמֵאִין וַאֲסָרוֹ בֵּין בְּדִבְרֵי תּוֹרָה בֵּין בִּקְרִיאַת שְׁמַע וּתְפִלָּה, כְּדֵי שֶׁלֹּא יִהְיוּ תַּלְמִידֵי חֲכָמִים מְצוּיִין אֵצֶל נְשׁוֹתֵיהֶן כְּתַרְנְגוֹלִים. וְלֹא פָּשְׁטָה תַּקָּנָה זוֹ בְּרֹב יִשְׂרָאֵל, וְאַחַר כָּךְ בָּטְלָה אוֹתָהּ תַּקָּנָה לְגַמְרֵי, מִשּׁוּם בִּטּוּל תּוֹרָה וּמִשּׁוּם בִּטּוּל פִּרְיָה וְרִבְיָה, וְהֶעֱמִידוּ הַדָּבָר עַל הַדִּין שֶׁאַף בַּעַל קֶרִי מֻתָּר בְּדִבְרֵי תּוֹרָה וּבִקְרִיאַת שְׁמַע וּבִּתְפִלָּה בְּלֹא טְבִילָה וּבְלֹא רְחִיצָה בְּט׳ קַבִּין. וְכֵן פָּשַׁט הַמִּנְהָג. וְאַף בִּזְמַן שֶׁנָּהֲגוּ בִּטְבִילָה — לֹא הִצְרִיכוּהוּ אֶלָּא לְבָרִיא שֶׁשִּׁמֵּשׁ מִטָּתוֹ, אֲבָל חֹלֶה שֶׁרָאָה קֶרִי לְאָנְסוֹ — פָּטוּר מִכְּלוּם. (וְחֹלֶה שֶׁשִּׁמֵּשׁ מִטָּתוֹ) וּבָרִיא לְאָנְסוֹ — דַּיּוֹ בְּט׳ קַבִּין שֶׁיִּשְׁפֹּךְ עָלָיו מִכְּלִי אַחַת אוֹ אֲפִלּוּ מִב׳ וְג׳, אֲבָל מִד׳ כֵּלִים אֵין מִצְטָרְפִין. וְהוּא שֶׁיַּתְחִיל הַשֵּׁנִי עַד שֶׁלֹּא פָּסַק הָרִאשׁוֹן, וְכֵן הַשְּׁלִישִׁי עַד שֶׁלֹּא פָּסַק הַשֵּׁנִי. וְתִשְׁעָה קַבִּין אֵלּוּ אֵינָן מְטַהֲרִין אֶלָּא כְּשֶׁשּׁוֹפְכִין עָלָיו אֲבָל לֹא כְּשֶׁטּוֹבֵל בָּהֶם, בֵּין שֶׁהֵם בַּכְּלִי בֵּין בַּקַּרְקַע, עַד שֶׁיִּהְיֶה אַרְבָּעִים סְאָה (בַּקַּרְקַע), וְאָז מוֹעִילִים לִטְבֹּל בָּהֶם אֲפִלּוּ לְבָרִיא שֶׁשִּׁמֵּשׁ מִטָּתוֹ, אֲפִלּוּ הֵם שְׁאוּבִים:

כָּל הַטְּמֵאִים. כָּל הַטְּמֵאִים — אֲפִלּוּ טֻמְאָה הַיּוֹצֵאת מִגּוּפָן (זָב, זָבָה וְכַיּוֹצֵא בָּהֶן) — קוֹרִין ק״ש וּמִתְפַּלְלִין. חוּץ מִבַּעַל קֶרִי שֶׁהוֹצִיאוֹ עֶזְרָא וַאֲסָרוֹ בְּדִבְרֵי תּוֹרָה, ק״ש וּתְפִלָּה — כְּדֵי שֶׁלֹּא יִהְיוּ תַּלְמִידֵי חֲכָמִים מְצוּיִין אֵצֶל נְשׁוֹתֵיהֶן שֶׁלֹּא כְּדֶרֶךְ. וְלֹא פָּשְׁטָה הַתַּקָּנָה בְּרֹב יִשְׂרָאֵל, וְאַחַר כָּךְ בָּטְלָה לְגַמְרֵי (מִשּׁוּם בִּטּוּל תּוֹרָה וּבִטּוּל פִּרְיָה וְרִבְיָה) : אַף בַּעַל קֶרִי מֻתָּר בְּלֹא טְבִילָה וּבְלֹא ט׳ קַבִּין, וְכֵן פָּשַׁט הַמִּנְהָג. וּבִזְמַן שֶׁנָּהֲגוּ טְבִילָה — לֹא הִצְרִיכוּ אֶלָּא בָּרִיא ; חֹלֶה שֶׁרָאָה קֶרִי לְאָנְסוֹ פָּטוּר מִכְּלוּם. הַט׳ קַבִּין אֵינָן מְטַהֲרִין אֶלָּא כְּשֶׁשּׁוֹפְכִין עָלָיו, לֹא בִּטְבִילָה, עַד שֶׁיִּהְיֶה אַרְבָּעִים סְאָה.

סעיף ב מִנְהַג טָהֳרָה שֶׁל נָשִׁים בְּבֵית הַכְּנֶסֶת ; חַיָּבוֹת בַּבְּרָכוֹת אֲפִלּוּ בִּשְׁעַת נִדָּתָן.

נָהֲגוּ הַנָּשִׁים טָהֳרָה וּפְרִישׁוּת בְּעַצְמָן שֶׁלֹּא לִכָּנֵס לְבֵית הַכְּנֶסֶת וְשֶׁלֹּא לִרְאוֹת סֵפֶר תּוֹרָה בִּשְׁעַת נִדּוּתָן, וְגַם כְּשֶׁמִתְפַּלְלִין אֵינָן עוֹמְדוֹת בִּפְנֵי חַבְרוֹתֵיהֶם. וּמִשּׁוּם מִנְהָג וְכָבוֹד עוֹשִׂין כֵּן וְלֹא מִשּׁוּם אִסּוּר, וּבִימֵי לִבּוּנָן לֹא נָהֲגוּ לְהַחֲמִיר. וּבְיָמִים נוֹרָאִים מִיּוֹם אֶחָד שֶׁל סְלִיחוֹת וְאֵילָךְ שֶׁרַבִּים מִתְאַסְּפִים לֵילֵךְ לְבֵית הַכְּנֶסֶת — מֻתָּרוֹת לֵילֵךְ לְבֵית הַכְּנֶסֶת אַף בִּימֵי נִדּוּתָן כִּשְׁאָר נָשִׁים, כִּי יִהְיֶה לָהֶן לְעִצָּבוֹן גָּדוֹל שֶׁהַכֹּל מִתְאַסְּפִין וְהֵן יַעַמְדוּ חוּץ. וְכֵן כָּל כַּיּוֹצֵא בָּזֶה, כְּגוֹן שֶׁמַּשִּׂיאָה אֶת בְּנָהּ אוֹ בִּתָּהּ, אוֹ שֶׁהִיא בְּעַצְמָהּ יוֹלֶדֶת שֶׁהוֹלְכִין לְבֵית הַכְּנֶסֶת אַחַר ד׳ שָׁבוּעוֹת — מֻתֶּרֶת אֲפִלּוּ הִיא נִדָּה. וְכָל זֶה לֵילֵךְ לְבֵית הַכְּנֶסֶת, אֲבָל לְהִתְפַּלֵּל בְּבֵיתָהּ וּלְבָרֵךְ כָּל הַבְּרָכוֹת וּבִפְרָט בִּרְכַּת הַמָּזוֹן וְקִדּוּשׁ שֶׁהֵן מִן הַתּוֹרָה — חַיָּבוֹת לְכֻלֵּי עָלְמָא אֲפִלּוּ בִּשְׁעַת נִדָּתָן. וּמַה שֶּׁנּוֹהֲגִין שֶׁלֹּא לְהַזְכִּיר אֶת הַשֵּׁם בִּשְׁעַת נִדָּתָן אוֹ שֶׁלֹּא לִגַּע בַּסֵּפֶר — אֵין מִנְהָג זֶה עִקָּר, וְעַל כָּל פָּנִים תִּשְׁמַע בִּרְכַּת הַמָּזוֹן וְקִדּוּשׁ מֵאֲחֵרִים, וְאִם אֵין אַחֵר תְּקַדֵּשׁ וּתְבָרֵךְ בְּעַצְמָהּ בְּלַחַשׁ כָּל הַבְּרָכוֹת שֶׁחַיֶּבֶת בָּהֶן:(סְלִיק הִלְכוֹת קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ בְּעֶזְרַת הַשֵּׁם יִתְבָּרֵךְ):

נָהֲגוּ הַנָּשִׁים. נָהֲגוּ הַנָּשִׁים בְּעַצְמָן טָהֳרָה וּפְרִישׁוּת : שֶׁלֹּא לִכָּנֵס לְבֵית הַכְּנֶסֶת וְשֶׁלֹּא לִרְאוֹת סֵפֶר תּוֹרָה בִּשְׁעַת נִדּוּתָן. וְעוֹשִׂין כֵּן מִשּׁוּם מִנְהָג וְכָבוֹד וְלֹא מִשּׁוּם אִסּוּר, וּבִימֵי לִבּוּנָן לֹא הֶחֱמִירוּ. וּבַיָּמִים נוֹרָאִים, מִיּוֹם רִאשׁוֹן שֶׁל סְלִיחוֹת וְאֵילָךְ, שֶׁרַבִּים מִתְאַסְּפִים — מֻתָּרוֹת לֵילֵךְ לְבֵית הַכְּנֶסֶת כִּשְׁאָר נָשִׁים, שֶׁלֹּא יִהְיֶה לָהֶן עִצָּבוֹן גָּדוֹל ; וְכֵן בְּמַשִּׂיאָה בְּנָהּ אוֹ בִּתָּהּ, אוֹ בְּיוֹלֶדֶת. וְכָל זֶה רַק לְבֵית הַכְּנֶסֶת ; אֲבָל לְהִתְפַּלֵּל בְּבֵיתָהּ וּלְבָרֵךְ כָּל הַבְּרָכוֹת — וּבִפְרָט בִּרְכַּת הַמָּזוֹן וְקִדּוּשׁ שֶׁהֵן מִן הַתּוֹרָה — חַיָּבוֹת לְכֻלֵּי עָלְמָא אֲפִלּוּ בִּשְׁעַת נִדָּתָן. וּמַה שֶּׁנּוֹהֲגִין שֶׁלֹּא לְהַזְכִּיר הַשֵּׁם אוֹ שֶׁלֹּא לִגַּע בַּסֵּפֶר — אֵין מִנְהָג זֶה עִקָּר ; עַל כָּל פָּנִים תִּשְׁמַע בִּרְכַּת הַמָּזוֹן וְקִדּוּשׁ מֵאֲחֵרִים, וְאִם אֵין אַחֵר — תְּבָרֵךְ בְּעַצְמָהּ בְּלַחַשׁ. (סְלִיק הִלְכוֹת קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ.)

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim פ״ח (מהדורת קה״ת, ב׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין בַּעַל קֶרִי בִּקְרִיאַת שְׁמַע

היסוד : הַבַּעַל קֶרִי וְתַקָּנַת עֶזְרָא

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שני צירים. תחילה הַבַּעַל קֶרִי וְתַקָּנַת עֶזְרָא שֶׁבָּטְלָה — עִם פְּרָטֵי ט׳ קַבִּין בַּמָּקוֹם שֶׁנָּהֲגוּ בָּהּ. אחר כך מִנְהַג הַטָּהֳרָה שֶׁל הַנָּשִׁים בִּשְׁעַת נִדּוּתָן — הַהַבְחָנָה בֵּין בֵּית הַכְּנֶסֶת לְחִיּוּב הַבְּרָכוֹת.

בַּעַל קֶרִי — תַּקָּנַת עֶזְרָא שֶׁבָּטְלָה

בַּעַל קֶרִי — תַּקָּנַת עֶזְרָא שֶׁבָּטְלָה

היסוד. כָּל הַטְּמֵאִים קוֹרִין ק״ש וּמִתְפַּלְלִין ; רַק אֶת הַבַּעַל קֶרִי הוֹצִיא עֶזְרָא — וְאַחַר כָּךְ בָּטְלָה הַתַּקָּנָה לְגַמְרֵי (מִשּׁוּם בִּטּוּל תּוֹרָה וּפִרְיָה וְרִבְיָה), וְהֶעֱמִידוּ עַל הַדִּין (סעיף א).

המסקנה. הַיּוֹם הַבַּעַל קֶרִי לוֹמֵד, קוֹרֵא וּמִתְפַּלֵּל בְּלֹא טְבִילָה ; הַט׳ קַבִּין שֶׁשּׁוֹפְכִין עָלָיו נִשְׁאֲרוּ כְּרֶשֶׁם מִנְהָג קַדְמוֹן. עיין סימן 87.

מִנְהַג טָהֳרָה שֶׁל נָשִׁים

מִנְהַג טָהֳרָה שֶׁל נָשִׁים

היסוד. נָהֲגוּ הַנָּשִׁים טָהֳרָה וּפְרִישׁוּת בְּעַצְמָן — מִשּׁוּם מִנְהָג וְכָבוֹד וְלֹא מִשּׁוּם אִסּוּר ; וּבַיָּמִים נוֹרָאִים וְכַיּוֹצֵא בָּהֶם מֻתָּרוֹת לֵילֵךְ לְבֵית הַכְּנֶסֶת (סעיף ב).

המסקנה. הַמִּנְהָג נוֹגֵעַ רַק לְבֵית הַכְּנֶסֶת ; לְהִתְפַּלֵּל בְּבֵיתָהּ וְהַבְּרָכוֹת — וּבִפְרָט בִּרְכַּת הַמָּזוֹן וְקִדּוּשׁ — חַיָּבוֹת לְכֻלֵּי עָלְמָא. עיין סימן 89.

הפניה ללימוד. היקף הסימן — תַּקָּנַת עֶזְרָא שֶׁבָּטְלָה וּמִנְהַג הַטָּהֳרָה שֶׁל הַנָּשִׁים — נוגע לעניינים עדינים של הנהגה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בצניעות ובלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין בַּעַל קֶרִי — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן פ״ח. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מב׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
בַּעַל קֶרִי וְתַקָּנַת עֶזְרָא או״ח פ״ח — כָּל הַטְּמֵאִים קוֹרִין ; חוּץ מִבַּעַל קֶרִי שֶׁהוֹצִיאוֹ עֶזְרָא, וְאַחַר כָּךְ בָּטְלָה הַתַּקָּנָה. מ״ב פ״ח — הַיּוֹם מֻתָּר בְּלֹא טְבִילָה ; דִּין ט׳ קַבִּין וְעִנְיַן טְבִילַת עֶזְרָא. ערוה״ש פ״ח — גֶּדֶר בִּטּוּל הַתַּקָּנָה וְשֹׁרֶשׁ הַנְהָגַת הַנְּקִיּוּת. דרך אדמו״ר הזקן — בָּטְלָה לְגַמְרֵי (בִּטּוּל תּוֹרָה וּפִרְיָה וְרִבְיָה) ; בְּלֹא טְבִילָה ; ט׳ קַבִּין שֶׁשּׁוֹפְכִין עָלָיו, לֹא בִּטְבִילָה, עַד אַרְבָּעִים סְאָה (סעיף א).
מִנְהַג נָשִׁים בִּימֵי נִדּוּתָן או״ח פ״ח (רמ״א) — יֵשׁ שֶׁכָּתְבוּ שֶׁאֵין לְאִשָּׁה נִדָּה לִכָּנֵס לְבֵית הַכְּנֶסֶת ; וּבְיָמִים נוֹרָאִים מֻתָּרוֹת. מ״ב פ״ח — פְּרָטֵי הַמִּנְהָג וְהַהֶתֵּר בַּיָּמִים נוֹרָאִים. ערוה״ש פ״ח — עִקַּר הַדִּין שֶׁהַכֹּל מֻתָּר, וְהַמִּנְהָג לְחֻמְרָא. דרך אדמו״ר הזקן — מִנְהָג וְכָבוֹד וְלֹא אִסּוּר ; מֻתָּרוֹת לְבֵית הַכְּנֶסֶת בַּיָּמִים נוֹרָאִים ; חַיָּבוֹת בַּבְּרָכוֹת (בִּרְכַּת הַמָּזוֹן וְקִדּוּשׁ) אֲפִלּוּ בִּשְׁעַת נִדָּתָן (סעיף ב).

הטבלה נערכה מטקסט המחבר (אורח חיים פ״ח, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ד, רמ״א, תרומת הדשן, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ב). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ב׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שתי מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַתַּקָּנָה שֶׁבָּטְלָה

קו א — הַתַּקָּנָה שֶׁבָּטְלָה

יסוד הסימן : תַּקָּנַת עֶזְרָא עַל הַבַּעַל קֶרִי — בָּטְלָה לְגַמְרֵי, וְהֶעֱמִידוּ הַדָּבָר עַל הַדִּין (סעיף א).

קריאת הרב : הָרַב מַדְגִּישׁ אֶת שְׁנֵי טַעֲמֵי הַבִּטּוּל — בִּטּוּל תּוֹרָה וּבִטּוּל פִּרְיָה וְרִבְיָה — וְקוֹבֵעַ אֶת הַדִּין : בְּלֹא טְבִילָה. עיין סימן 87.

מסקנה למעשה : אֵין דּוֹחִין לִמּוּד אוֹ תְּפִלָּה מֵחֲמַת קֶרִי ; הַט׳ קַבִּין אֵינָן אֶלָּא מִנְהָג רְשׁוּת, שֶׁשּׁוֹפְכִין עָלָיו.

קו ב — מִנְהָג לְעֻמַּת חִיּוּב

קו ב — מִנְהָג לְעֻמַּת חִיּוּב

יסוד הסימן : מִנְהַג הַטָּהֳרָה שֶׁל הַנָּשִׁים הוּא מִשּׁוּם מִנְהָג וְכָבוֹד וְלֹא מִשּׁוּם אִסּוּר ; הַבְּרָכוֹת נִשְׁאָרוֹת חִיּוּב (סעיף ב).

קריאת הרב : הָרַב מַבְחִין בְּבֵרוּר בֵּין הַפְּרִישׁוּת הָרְשׁוּת (בֵּית הַכְּנֶסֶת) לְבֵין הַחִיּוּב (הַבְּרָכוֹת מִן הַתּוֹרָה). עיין סימן 89.

מסקנה למעשה : בִּרְכַּת הַמָּזוֹן וְקִדּוּשׁ אֵינָן נִדְחוֹת לְעוֹלָם ; וְאִם אֵין לָהּ מִי לְהַשְׁמִיעַ — מְבָרֶכֶת בְּלַחַשׁ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן פ״ח ברגישות אדמו״ר הזקן וחב״ד — מוצגות בצניעות, ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן פ״ח (דִּין בַּעַל קֶרִי בִּקְרִיאַת שְׁמַע)

  • ① כָּל הַטְּמֵאִים. הַכֹּל קוֹרִין ק״ש וּמִתְפַּלְלִין ; אֵין דּוֹחִין לֹא לִמּוּד וְלֹא תְּפִלָּה. יסוד : שוע״ר פ״ח סעיף א.
  • ② תַּקָּנַת עֶזְרָא. הַתַּקָּנָה עַל הַבַּעַל קֶרִי בָּטְלָה לְגַמְרֵי — מֻתָּר בְּלֹא טְבִילָה ; וְכֵן פָּשַׁט הַמִּנְהָג. יסוד : שוע״ר פ״ח סעיף א.
  • ③ ט׳ קַבִּין. בַּמָּקוֹם שֶׁנָּהֲגוּ בָּהּ — הַטָּהֳרָה שֶׁשּׁוֹפְכִין עָלָיו, לֹא בִּטְבִילָה, עַד אַרְבָּעִים סְאָה. יסוד : שוע״ר פ״ח סעיף א.
  • ④ מִנְהַג נָשִׁים. הַפְּרִישׁוּת מִבֵּית הַכְּנֶסֶת בִּשְׁעַת נִדּוּתָן — מִנְהָג וְכָבוֹד וְלֹא אִסּוּר ; בַּיָּמִים נוֹרָאִים וּשְׂמָחוֹת — מֻתָּרוֹת. יסוד : שוע״ר פ״ח סעיף ב.
  • ⑤ חִיּוּב הַבְּרָכוֹת. לְהִתְפַּלֵּל בְּבֵיתָהּ וְלוֹמַר הַבְּרָכוֹת — וּבִפְרָט בִּרְכַּת הַמָּזוֹן וְקִדּוּשׁ — חַיָּבוֹת לְכֻלֵּי עָלְמָא ; וְאִם אֵין מִי לְהַשְׁמִיעַ, בְּלַחַשׁ. יסוד : שוע״ר פ״ח סעיף ב.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן פ״ח (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — עִנְיְנֵי הַטָּהֳרָה, הַמִּנְהָג וְחִיּוּב הַבְּרָכוֹת — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
