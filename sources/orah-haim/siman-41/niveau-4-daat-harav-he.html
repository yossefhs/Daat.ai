דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן מ״א · דִּין הַנּוֹשֵׂא מַשּׂאוֹי אֵיךְ יִנְהֹג בַּתְּפִלִּין

דרכו של אדמו״ר הזקן בדין הנושא משאוי וההנהגה בתפילין: חֲלִיצַת הַשֶּׁל רֹאשׁ כָּל זְמַן שֶׁנּוֹשֵׂא מַשּׂאוֹי (גְּנַאי, דֶּרֶךְ בִּזָּיוֹן) ; הַמִּטְפַּחַת אֲסוּרָה אֲבָל כּוֹבַע וּמִצְנֶפֶת מֻתָּרִים (דֶּרֶךְ לְבִישָׁה) ; וּמַשּׂאוֹי כָּבֵד ד׳ קַבִּין שֶׁהַתְּפִלִּין נִדְחָקוֹת — ברגישות חב״ד.

למה רמה 4 מיוחדת לאדמו״ר הזקן? השולחן ערוך של אדמו״ר הזקן אינו פירוש על המחבר — הוא שולחן ערוך עצמאי ושלם, שכתבו אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו: הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. דף זה מאסף את שיטת הרב על סימן מ״א — דִּין הַנּוֹשֵׂא מַשּׂאוֹי אֵיךְ יִנְהֹג בַּתְּפִלִּין. הסימן כולל אצל הרב סָעִיף אֶחָד בלבד, שבו אדמו״ר הזקן פורש: חִיּוּב חֲלִיצַת הַשֶּׁל רֹאשׁ כָּל זְמַן שֶׁנּוֹשֵׂא מַשּׂאוֹי עַל רֹאשׁוֹ ; שֶׁאֲפִלּוּ דָּבָר קַל כְּמִטְפַּחַת אָסוּר עַל הָרֹאשׁ שֶׁיֵּשׁ בּוֹ תְּפִלִּין (מִשּׁוּם גְּנַאי), וְאִלּוּ כּוֹבַע וּמִצְנֶפֶת — שֶׁדַּרְכָּן לִהְיוֹת עַל הָרֹאשׁ (דֶּרֶךְ לְבִישָׁה) — מֻתָּרִים ; וְמַשּׂאוֹי כָּבֵד ד׳ קַבִּין שֶׁהַתְּפִלִּין נִדְחָקוֹת וְצָרִיךְ לַהֲסִירָן.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בדף זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן מ״א — סָעִיף אֶחָד מידי הרב — לפי מהדורת קה״ת כפי שנסרקה בספריא (Shulchan Arukh HaRav, Orach Chayim מ״א). בסעיף יחיד זה, אדמו״ר הזקן דן בנושא משאוי על ראשו (הַנּוֹשֵׂא מַשּׂאוֹי) שצריך לחלוץ את השֶּׁל רֹאשׁ עד שיסיר את המשאוי ; באיסור מִטְפַּחַת על הראש (מִשּׁוּם גְּנַאי, דבר שאינו מלבוש ואין דרכו להיות על הראש) ; במה שמותר — כּוֹבַע וּמִצְנֶפֶת, דֶּרֶךְ לְבִישָׁה, אף על התפילין ; וּבמַשּׂאוֹי כָּבֵד ד׳ קַבִּין שהתפילין נדחקות. הביאורים הם שלנו. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן מ״א — הַנּוֹשֵׂא מַשּׂאוֹי אֵיךְ יִנְהֹג בַּתְּפִלִּין

הטקסט המלא של שולחן ערוך אדמו״ר הזקן

שולחן ערוך הרב, סימן מ״א — הַנּוֹשֵׂא מַשּׂאוֹי אֵיךְ יִנְהֹג בַּתְּפִלִּין — וּבוֹ סָעִיף אֶחָד
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.41), ולאחריו ביאור קצר. הסימן כולל סָעִיף אֶחָד בלבד.

סעיף א הַנּוֹשֵׂא מַשּׂאוֹי עַל רֹאשׁוֹ חוֹלֵץ שֶׁל רֹאשׁ עַד שֶׁיָּסִיר ; מִטְפַּחַת אֲסוּרָה, כּוֹבַע וּמִצְנֶפֶת מֻתָּרִים ; מַשּׂאוֹי כָּבֵד ד׳ קַבִּין וְנִדְחָקוֹת — לַהֲסִירָן.

דִּין הַנּוֹשֵׂא מַשְּׂאוֹי אֵיךְ יִנְהֹג בַּתְּפִלִּין וּבוֹ סָעִיף אֶחָד:הַנּוֹשֵׂא מַשּׂאוֹי עַל רֹאשׁוֹ — צָרִיךְ לַחֲלֹץ תְּפִלִּין שֶׁל רֹאשׁ עַד שֶׁיָּסִיר הַמַּשׂאוֹי. וַאֲפִלּוּ דָּבָר קַל, כְּגוֹן מִטְפַּחַת — אָסוּר לְהַנִּיחַ עַל הָרֹאשׁ שֶׁיֵּשׁ בּוֹ תְּפִלִּין, שֶׁגְּנַאי הוּא כְּשֶׁיֵּשׁ עַל הָרֹאשׁ דָּבָר אַחֵר שֶׁאֵינָהּ מַלְבּוּשׁ וְאֵין דַּרְכָּהּ לִהְיוֹת עַל הָרֹאשׁ. אֲבָל דָּבָר שֶׁדַּרְכָּהּ לִהְיוֹת עַל הָרֹאשׁ כְּגוֹן כּוֹבַע וּמִצְנֶפֶת — מֻתָּר, אֲפִלּוּ אִם מֻנָּח עַל הַתְּפִלִּין. וּמִכָּל מָקוֹם, אִם הוּא מַשּׂאוֹי כָּבֵד אַרְבָּעָה קַבִּין וְהַתְּפִלִּין נִדְחָקוֹת — צָרִיךְ לַהֲסִירָן:

אִיךְ יִנְהֹג הַנּוֹשֵׂא מַשּׂאוֹי. הַנּוֹשֵׂא מַשּׂאוֹי עַל רֹאשׁוֹ צָרִיךְ לַחֲלֹץ אֶת הַתְּפִלִּין שֶׁל רֹאשׁ עַד שֶׁיָּסִיר אֶת הַמַּשּׂאוֹי. וַאֲפִלּוּ דָּבָר קַל כְּמִטְפַּחַת אָסוּר לְהַנִּיחַ עַל הָרֹאשׁ שֶׁיֵּשׁ בּוֹ תְּפִלִּין, שֶׁגְּנַאי הוּא כְּשֶׁיֵּשׁ עַל הָרֹאשׁ דָּבָר אַחֵר שֶׁאֵינוֹ מַלְבּוּשׁ וְאֵין דַּרְכּוֹ לִהְיוֹת עַל הָרֹאשׁ. אֲבָל דָּבָר שֶׁדַּרְכּוֹ לִהְיוֹת עַל הָרֹאשׁ כְּגוֹן כּוֹבַע וּמִצְנֶפֶת — מֻתָּר, אֲפִלּוּ מֻנָּח עַל הַתְּפִלִּין. וּמִכָּל מָקוֹם, אִם הוּא מַשּׂאוֹי כָּבֵד ד׳ קַבִּין וְהַתְּפִלִּין נִדְחָקוֹת — צָרִיךְ לַהֲסִירָן.

מקור: ספריא — Shulchan Arukh HaRav, Orach Chayim מ״א (מהדורת קה״ת, סָעִיף אֶחָד). ביאור: DAAT. אין מובא כאן שום קטע מחוץ לטקסט הניתן לאימות.

2 · היסוד

היסוד — כְּבוֹד הַתְּפִלִּין בִּנְשִׂיאַת מַשּׂאוֹי

היסוד: הנושא משאוי וכבוד התפילין

קריאת אדמו״ר הזקן על סימן זה בשלושה מהלכים. תחילה קְדֻשַּׁת הַתְּפִלִּין וּכְבוֹדָן: נְשִׂיאַת מַשּׂאוֹי עַל הָרֹאשׁ הִיא גְּנַאי / דֶּרֶךְ בִּזָּיוֹן וְהֶסֵּחַ הַדַּעַת, וּמִכָּאן חִיּוּב חֲלִיצַת הַשֶּׁל רֹאשׁ עַד שֶׁיָּסִיר. אַחַר כָּךְ הַקְּרִיטֶרְיוֹן שֶׁל דֶּרֶךְ לְבִישָׁה: מַלְבּוּשׁ שֶׁדַּרְכּוֹ עַל הָרֹאשׁ (כּוֹבַע, מִצְנֶפֶת) מֻתָּר אַף עַל הַתְּפִלִּין, אֲבָל מִטְפַּחַת — שֶׁאֵינָהּ מַלְבּוּשׁ — אֲסוּרָה. וּלְבַסּוֹף שִׁעוּר ד׳ קַבִּין: מַשּׂאוֹי כָּבֵד שֶׁהַתְּפִלִּין נִדְחָקוֹת מְחַיֵּב לַהֲסִירָן, שֶׁמָּא יָזוּזוּ מִמְּקוֹמָן.

כְּבוֹד הַתְּפִלִּין וְאִסּוּר הַגְּנַאי

כבוד התפילין ואיסור הגְּנַאי

היסוד. הַתְּפִלִּין תַּשְׁמִישֵׁי קְדֻשָּׁה ; נְשִׂיאַת מַשּׂאוֹי עַל הָרֹאשׁ שֶׁיֵּשׁ בּוֹ תְּפִלִּין הִיא גְּנַאי, וְיֵשׁ בָּהּ אַף הֶסֵּחַ הַדַּעַת — וּמִכָּאן חִיּוּב חֲלִיצַת הַשֶּׁל רֹאשׁ עַד שֶׁיָּסִיר אֶת הַמַּשּׂאוֹי.

התוצאה. דָּבָר שֶׁאֵינוֹ מַלְבּוּשׁ וְאֵין דַּרְכּוֹ עַל הָרֹאשׁ — אֲפִלּוּ קַל כְּמִטְפַּחַת — אָסוּר. וּלְעִנְיַן הַגּוּף נָקִי וְהֶסֵּחַ הַדַּעַת בַּתְּפִלִּין עַיֵּן סימן 28 וְסימן 44.

דֶּרֶךְ לְבִישָׁה — כּוֹבַע וּמִצְנֶפֶת

דרך לבישה — כובע ומצנפת

העיקרון. הַקְּרִיטֶרְיוֹן הַמַּכְרִיעַ הוּא הַדֶּרֶךְ לְבִישָׁה: דָּבָר שֶׁדַּרְכּוֹ לִהְיוֹת עַל הָרֹאשׁ — כּוֹבַע אוֹ מִצְנֶפֶת — מֻתָּר אֲפִלּוּ מֻנָּח עַל הַתְּפִלִּין, שֶׁאֵין בָּזֶה לֹא גְּנַאי וְלֹא חִסָּרוֹן כָּבוֹד.

ההרחבה. לְעֻמַּת זֹאת, מַה שֶּׁאֵינוֹ מַלְבּוּשׁ (מִטְפַּחַת, מַשּׂאוֹי) מַנִּיחַ עַל הָרֹאשׁ דָּבָר אַחֵר זָר לַקְּדֻשָּׁה — וּמִכָּאן הָאִסּוּר. קַו הַגְּבוּל אֵינוֹ הַכֹּבֶד לְבַדּוֹ, אֶלָּא טִיב הַדָּבָר וְיַחְסוֹ אֶל הָרֹאשׁ.

מַשּׂאוֹי כָּבֵד ד׳ קַבִּין וְנִדְחָקוֹת

משאוי כבד ד׳ קבין והתפילין נדחקות

העיקרון. אַף כּוֹבַע וּמִצְנֶפֶת הַמֻּתָּרִים עֲלוּלִים לְהִתְקַשּׁוֹת: אִם הוּא מַשּׂאוֹי כָּבֵד ד׳ קַבִּין וְהַתְּפִלִּין נִדְחָקוֹת — צָרִיךְ לַהֲסִירָן.

ההנהגה. הַחֲשָׁשׁ כָּפוּל: הַכֹּבֶד הַמֵּזִיז אֶת הַתְּפִלִּין מִמְּקוֹמָן, וּפְגִיעָה בַּכָּבוֹד. שִׁעוּר ד׳ קַבִּין קוֹבֵעַ אֶת הַסַּף שֶׁמֵּעֵבֶר לוֹ הֶתֵּר הַדֶּרֶךְ לְבִישָׁה אֵינוֹ עוֹמֵד.

הפניה ללימוד. משקל היסוד שהונח כאן — הכָּבוֹד לתַשְׁמִישֵׁי קְדֻשָּׁה, הדֶּרֶךְ לְבִישָׁה ושִׁעוּר ד׳ קַבִּין — נוגע גם בהנהגה (כיצד נושאים, מה מניחים על הראש, מתי חולצים) וגם ברגישות חב״ד על הקְדֻשָּׁה שסביב המצווה. דף זה מציג זאת ברמת העיקרון המתועד בטקסט של הרב, בלי לצטט פרק או נוסח מדויק, ובלי להוסיף דבר מומצא במשמעות הרוחנית. ללימוד מעמיק של ממד זה — ולהנהגה אישית במשאוי, במטפחת ובשיעור — יש ללמוד עם רב של חב״ד ובתוך המקורות עצמם.

הַנּוֹשֵׂא מַשּׂאוֹי אֵיךְ יִנְהֹג בַּתְּפִלִּין — בעיקרון · ללימוד, פנה לרב של חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרכו של אדמו״ר הזקן עם המחבר, הרמ״א (רמ״א) והמשנה ברורה (משנה ברורה) בנקודות המפתח של סימן מ״א. שורת שולחן ערוך הרב (שולחן ערוך הרב) מובלטת — היא חושפת את רגישות הרב כפי שהיא עולה מטקסטו (§1, סָעִיף אֶחָד). הרב מפרש את הטעם ואת הפרט במקום שהמחבר מסתפק באמירה.

נושא מחבר רמ״א משנה ברורה דרך אדמו״ר הזקן (שולחן ערוך הרב)
נשיאת משאוי וחליצת של ראש או״ח מ״א — הַנּוֹשֵׂא מַשּׂאוֹי עַל רֹאשׁוֹ חוֹלֵץ שֶׁל רֹאשׁ עַד שֶׁיָּסִיר הַמַּשּׂאוֹי. — מ״ב מ״א / ברכות כ״ג ע״ב — טַעַם הַגְּנַאי וְהֶסֵּחַ הַדַּעַת. דרך אדמו״ר הזקן — הַנּוֹשֵׂא מַשּׂאוֹי עַל רֹאשׁוֹ צָרִיךְ לַחֲלֹץ שֶׁל רֹאשׁ עַד שֶׁיָּסִיר הַמַּשּׂאוֹי (סעיף א).
מטפחת מול כובע ומצנפת או״ח מ״א — מִטְפַּחַת אֲסוּרָה עַל הָרֹאשׁ ; כּוֹבַע וּמִצְנֶפֶת מֻתָּרִים. — מ״ב מ״א — גֶּדֶר דֶּרֶךְ לְבִישָׁה ; דָּבָר שֶׁאֵינוֹ מַלְבּוּשׁ. דרך אדמו״ר הזקן — דָּבָר שֶׁאֵינוֹ מַלְבּוּשׁ וְאֵין דַּרְכּוֹ עַל הָרֹאשׁ — גְּנַאי ; אֲבָל כּוֹבַע וּמִצְנֶפֶת מֻתָּרִים אַף עַל הַתְּפִלִּין (סעיף א).
משאוי כבד ד׳ קבין ונדחקות או״ח מ״א — עִקַּר הַנּוֹשֵׂא מַשּׂאוֹי ; שִׁעוּר הַכֹּבֶד מִדִּבְרֵי הָרְמ״א. רמ״א מ״א — מַשּׂאוֹי כָּבֵד ד׳ קַבִּין וְהַתְּפִלִּין נִדְחָקוֹת — יְסִירֵן. מ״ב מ״א — שִׁעוּר ד׳ קַבִּין וַחֲשַׁשׁ שֶׁיָּזוּזוּ מִמְּקוֹמָן. דרך אדמו״ר הזקן — אִם הוּא מַשּׂאוֹי כָּבֵד ד׳ קַבִּין וְהַתְּפִלִּין נִדְחָקוֹת — צָרִיךְ לַהֲסִירָן (סעיף א).

הטבלה נערכה מטקסט המחבר (אורח חיים מ״א, סָעִיף אֶחָד, מקור ספריא) ומנושאי הכלים הניתנים לאימות (בית יוסף, טור, רמב״ם הלכות תפילין ד, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ג ע״ב). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (סָעִיף אֶחָד) המובא ב§1. ייחוסי סעיף קטן שאינם ודאיים נותרו « על דרך הסברה ».

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה — מעבר ליישום הפשוט של הכלל. כל אחת בנויה: עיקרון הסימן → קריאת הרב → תוצאה מעשית. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — כְּבוֹד הַתְּפִלִּין וְאִסּוּר גְּנַאי

כבוד התפילין ואיסור הגְּנַאי

עיקרון הסימן: חֲלִיצַת הַשֶּׁל רֹאשׁ כָּל זְמַן שֶׁנּוֹשֵׂא מַשּׂאוֹי (סעיף יחיד).

קריאת הרב: הַמַּשּׂאוֹי עַל הָרֹאשׁ הוּא גְּנַאי וְהֶסֵּחַ הַדַּעַת ; חוֹלֵץ אֶת הַשֶּׁל רֹאשׁ עַד שֶׁיָּסִיר אֶת הַמַּשּׂאוֹי.

תוצאה מעשית: לֹא לָשֵׂאת מַשּׂאוֹי עַל הָרֹאשׁ תּוֹךְ כְּדֵי הֲנָחַת הַשֶּׁל רֹאשׁ ; לַחֲלֹץ תְּחִלָּה.

קו ב — דֶּרֶךְ לְבִישָׁה, כּוֹבַע וּמִצְנֶפֶת

דרך לבישה — כובע ומצנפת

עיקרון הסימן: מַה מֻּתָּר וּמַה אָסוּר עַל הָרֹאשׁ (סעיף יחיד).

קריאת הרב: הַקְּרִיטֶרְיוֹן הוּא הַדֶּרֶךְ לְבִישָׁה — כּוֹבַע אוֹ מִצְנֶפֶת (מַלְבּוּשׁ) מֻתָּר אַף עַל הַתְּפִלִּין ; מִטְפַּחַת, שֶׁאֵינָהּ מַלְבּוּשׁ, אֲסוּרָה.

תוצאה מעשית: כִּסּוּי רֹאשׁ רָגִיל מֻנָּח בְּלֹא חֲשָׁשׁ ; מִטְפַּחַת אוֹ כָּל דָּבָר זָר עַל הָרֹאשׁ שֶׁיֵּשׁ בּוֹ תְּפִלִּין — לֹא.

קו ג — שִׁעוּר ד׳ קַבִּין וְהַתְּפִלִּין נִדְחָקוֹת

שיעור ד׳ קבין והתפילין נדחקות

עיקרון הסימן: מַשּׂאוֹי כָּבֵד שֶׁהַתְּפִלִּין נִדְחָקוֹת (סעיף יחיד).

קריאת הרב: מֵעֵבֶר לְשִׁעוּר ד׳ קַבִּין, כְּשֶׁהַתְּפִלִּין נִדְחָקוֹת, הֶתֵּר הַכִּסּוּי אֵינוֹ עוֹמֵד — שֶׁמָּא יָזוּזוּ מִמְּקוֹמָן.

תוצאה מעשית: לְהָסִיר תְּפִלִּין תַּחַת מַשּׂאוֹי כָּבֵד ; וְלִסְמֹךְ עַל הָרַב בַּהֲעָרָכַת הַשִּׁעוּר.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן מ״א ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, בהפניה לרב לפרטי המקרים (המשאוי, כיסוי הראש, השיעור).

לחסיד חב״ד — סימן מ״א (הַנּוֹשֵׂא מַשּׂאוֹי אֵיךְ יִנְהֹג בַּתְּפִלִּין)

  • ① נשיאת משאוי. הַנּוֹשֵׂא מַשּׂאוֹי עַל רֹאשׁוֹ חוֹלֵץ אֶת הַשֶּׁל רֹאשׁ עַד שֶׁיָּסִיר — שֶׁהוּא גְּנַאי וְהֶסֵּחַ הַדַּעַת. יסוד: שוע״ר מ״א סעיף א.
  • ② המטפחת. אֲפִלּוּ דָּבָר קַל כְּמִטְפַּחַת אָסוּר עַל הָרֹאשׁ שֶׁיֵּשׁ בּוֹ תְּפִלִּין, שֶׁאֵינָהּ מַלְבּוּשׁ וְאֵין דַּרְכָּהּ לִהְיוֹת עַל הָרֹאשׁ. יסוד: שוע״ר מ״א סעיף א.
  • ③ כובע ומצנפת. דָּבָר שֶׁדַּרְכּוֹ עַל הָרֹאשׁ — כּוֹבַע, מִצְנֶפֶת (דֶּרֶךְ לְבִישָׁה) — מֻתָּר, אֲפִלּוּ מֻנָּח עַל הַתְּפִלִּין. יסוד: שוע״ר מ״א סעיף א.
  • ④ משאוי כבד. אִם הוּא מַשּׂאוֹי כָּבֵד ד׳ קַבִּין וְהַתְּפִלִּין נִדְחָקוֹת — צָרִיךְ לַהֲסִירָן. יסוד: שוע״ר מ״א סעיף א.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן מ״א (הטקסט האמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — המשאוי, כיסוי הראש, שיעור ד׳ קבין — יש להתייעץ עם רב, ולחב״ד עם רב של חב״ד.
