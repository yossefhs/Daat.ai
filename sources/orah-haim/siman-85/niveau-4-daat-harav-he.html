דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן פ״ה · בְּאֵיזֶה מְקוֹמוֹת אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע

דרכו של אדמו״ר הזקן במְבוֹאוֹת הַמְטֻנָּפוֹת : לֹא יֵלֵךְ בִּמְבוֹאוֹת הַמְטֻנָּפוֹת וְיִקְרָא — הָיָה קוֹרֵא וּבָא, מַפְסִיק ; אִסּוּר דִּבּוּר וְהִרְהוּר בִּמְקוֹם הַטִּנֹּפֶת ; דְּבָרִים שֶׁל חֹל וְכִנּוּיִים — מֻתָּר מִן הַדִּין ; הַשֵּׁמוֹת שֶׁאֵינָן נִמְחָקִין וְהַהַפְרָשָׁה מֵאִסּוּר — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן פ״ה — בְּאֵיזֶה מְקוֹמוֹת אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע. אצל הרב הסימן נפרש לד׳ סְעִיפִים : הַמְּבוֹאוֹת הַמְטֻנָּפוֹת וְהַהַפְסָקָה, אִסּוּר הַדִּבּוּר וְהַהִרְהוּר בִּמְקוֹם הַטִּנֹּפֶת, דְּבָרִים שֶׁל חֹל וְכִנּוּיִים, וְהַשֵּׁמוֹת שֶׁאֵינָן נִמְחָקִין עִם הַהַפְרָשָׁה מֵאִסּוּר.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן פ״ה — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim פ״ה). אדמו״ר הזקן דן בו בְּאֵיזֶה מְקוֹמוֹת אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן פ״ה — בְּאֵיזֶה מְקוֹמוֹת אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן פ״ה — בְּאֵיזֶה מְקוֹמוֹת אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.85), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.85.k).

סעיף א לֹא יֵלֵךְ בִּמְבוֹאוֹת הַמְטֻנָּפוֹת וְיִקְרָא ; הָיָה קוֹרֵא וּבָא — מַפְסִיק ; אִסּוּר דִּבּוּר וְהִרְהוּר.

בְּאֵיזֶה מְקוֹמוֹת אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע וּבוֹ ד׳ סְעִיפִים:לֹא יֵלֵךְ אָדָם בִּמְבוֹאוֹת הַמְטֻנָּפוֹת וְיַנִּיחַ יָדָיו עַל פִּיו וְיִקְרָא קְרִיאַת שְׁמַע. וַאֲפִלּוּ אִם הָיָה קוֹרֵא וּבָא — צָרִיךְ לְהַפְסִיק כְּשֶׁיַּגִּיעַ לִמְבוֹאוֹת הַמְטֻנָּפוֹת. וְאִם לֹא פָּסַק — עָלָיו הַכָּתוּב אוֹמֵר: ״כִּי דְבַר ה׳ בָּזָה וְגו׳ הִכָּרֵת תִּכָּרֵת וְגו׳״. וּכְשֶׁיּוֹצֵא מִשָּׁם — חוֹזֵר לַמָּקוֹם שֶׁפָּסַק, אִם לֹא שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ, כְּמוֹ שנתבאר בְּסִימָן ס״ה. וְלֹא קְרִיאַת שְׁמַע בִּלְבַד, אֶלָּא כָּל עִנְיָן שֶׁהוּא מִדִּבְרֵי הַקֹּדֶשׁ — אָסוּר לְאָמְרָהּ בִּמְקוֹם הַטִּנֹּפֶת, דְּהַיְנוּ בְּמָקוֹם שֶׁיֵּשׁ בּוֹ צוֹאָה אוֹ מֵי רַגְלַיִם אוֹ בְּבֵית הַכִּסֵּא אוֹ בְּבֵית הַמֶּרְחָץ, וַאֲפִלּוּ לְאָמְרוֹ בִּלְשׁוֹן חֹל. וְלֹא לְאָמְרוֹ בִּלְבַד, אֶלָּא אֲפִלּוּ לְהַרְהֵר בְּלִבּוֹ — אָסוּר, אֲפִלּוּ בְּדִבְרֵי תּוֹרָה שֶׁחַיָּב לַהֲגוֹת בָּהּ יוֹמָם וָלַיְלָה. וְלָכֵן כְּשֶׁנִּכְנָס לְבֵית הַכִּסֵּא — טוֹב לַחְשֹׁב חֶשְׁבּוֹנוֹתָיו, שֶׁלֹּא יָבוֹא לִידֵי הִרְהוּר תּוֹרָה. וּבְשַׁבָּת שֶׁאָסוּר לְחָשְׁבָם — יַפְנֶה לִבּוֹ לְהַרְהֵר בְּצִיּוּרִים וּבִנְיָנִים נָאִים וְכַיוֹצֵא בְּדִמְיוֹנִים אֵלּוּ הַמּוֹשְׁכִים וּמְפַנִּים לִבּוֹ מִלָּבוֹא לִידֵי הִרְהוּר דָּבָר שֶׁבְּקְּדֻשָּׁה. וַאֲפִלּוּ הִלְכוֹת בֵּית הַכִּסֵּא אָסוּר לִלְמֹד בְּבֵית הַכִּסֵּא, וְכֵן הִלְכוֹת בֵּית הַמֶּרְחָץ בְּבֵית הַמֶּרְחָץ:

מְבוֹאוֹת הַמְטֻנָּפוֹת. לֹא יֵלֵךְ בִּמְבוֹאוֹת הַמְטֻנָּפוֹת וְיָדָיו עַל פִּיו וְיִקְרָא ק״ש. הָיָה קוֹרֵא וּבָא — מַפְסִיק כְּשֶׁיַּגִּיעַ ; לֹא פָּסַק — עָלָיו הַכָּתוּב אוֹמֵר ״כִּי דְבַר ה׳ בָּזָה… הִכָּרֵת תִּכָּרֵת״. כְּשֶׁיּוֹצֵא — חוֹזֵר לַמָּקוֹם שֶׁפָּסַק אִם לֹא שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ (עיין סימן ס״ה). וְלֹא ק״ש בִּלְבַד : תמצית : כָּל דְּבַר קֹדֶשׁ אָסוּר בִּמְקוֹם הַטִּנֹּפֶת (צוֹאָה, מֵי רַגְלַיִם, בֵּית הַכִּסֵּא, בֵּית הַמֶּרְחָץ), אֲפִלּוּ בִּלְשׁוֹן חֹל וַאֲפִלּוּ לְהַרְהֵר בְּלִבּוֹ — אֲפִלּוּ דִּבְרֵי תּוֹרָה. לָכֵן בְּבֵית הַכִּסֵּא טוֹב לַחְשֹׁב חֶשְׁבּוֹנוֹתָיו, וּבְשַׁבָּת לְהַרְהֵר בְּצִיּוּרִים וּבִנְיָנִים נָאִים. וַאֲפִלּוּ הִלְכוֹת בֵּית הַכִּסֵּא אָסוּר לִלְמֹד שָׁם.

סעיף ב דְּבָרִים שֶׁל חֹל וְכִנּוּיִים — מֻתָּר מִן הַדִּין ; חֲסִידוּת לְהַחֲמִיר.

אֲבָל דְּבָרִים שֶׁל חֹל מֻתָּר לְאָמְרָם שָׁם בִּלְשׁוֹן הַקֹּדֶשׁ, אֶלָּא שֶׁמִּדַּת חֲסִידוּת הוּא לְהַחֲמִיר. וְכֵן הַכִּנּוּיִים כְּגוֹן ״רַחוּם״ וְ״חַנּוּן״ וְ״נֶאֱמָן״ וְכַיוֹצֵא בָּהֶם — מֻתָּר לְאָמְרָם שָׁם מִן הַדִּין, לְפִי שֶׁגַּם בְּנֵי אָדָם מְכֻנִּים בָּהֶם לִפְעָמִים, כְּמוֹ שֶׁנֶּאֱמַר: ״חַנּוּן וְרַחוּם וְצַדִּיק״:

דְּבָרִים שֶׁל חֹל. דְּבָרִים שֶׁל חֹל מֻתָּר לְאָמְרָם שָׁם אֲפִלּוּ בִּלְשׁוֹן הַקֹּדֶשׁ ; מִדַּת חֲסִידוּת לְהַחֲמִיר. וְכֵן הַכִּנּוּיִים — ״רַחוּם״, ״חַנּוּן״, ״נֶאֱמָן״ וְכַיּוֹצֵא — מֻתָּר מִן הַדִּין, שֶׁגַּם בְּנֵי אָדָם מְכֻנִּים בָּהֶם, כְּמוֹ שֶׁנֶּאֱמַר ״חַנּוּן וְרַחוּם וְצַדִּיק״.

סעיף ג שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין — אָסוּר לְהַזְכִּירָן, אֲפִלּוּ בִּלְשׁוֹן חֹל.

שֵׁמוֹת הַמְיֻחָדִין לְהַקָּדוֹשׁ בָּרוּךְ הוּא וְהֵן ז׳ שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין — אָסוּר לְהַזְכִּירָן שָׁם אֲפִלּוּ בִּלְשׁוֹן חֹל, דְּהַיְנוּ שֵׁם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נִקְרָא בְּפִי כָּל עַם וְלָשׁוֹן שֶׁיִּהְיֶה (כְּגוֹן גאָ״ט בִּלְשׁוֹן אַשְׁכְּנַז אוֹ בוג״א בִּלְשׁוֹן פּוֹלִין וְרוּסְיָא), שֶׁאַף שֶׁשֵּׁם זֶה אֵין בּוֹ קְדֻשָּׁה בְּאוֹתִיּוֹת כְּתִיבָתוֹ וּמֻתָּר לְמָחְקָן, מִכָּל מָקוֹם יֵשׁ בּוֹ מִשּׁוּם בִּזָּיוֹן בְּהַזְכָּרָתוֹ בִּמְקוֹם טִנֹּפֶת, כְּמוֹ בְּהַזְכָּרַת שָׁלוֹם שֶׁמֻּתָּר גַּם כֵּן לְמָחְקוֹ וְאַף עַל פִּי כֵן כֵּיוָן שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נִקְרָא בּוֹ אַף עַל פִּי שֶׁאֵינוֹ מְיֻחָד לוֹ — אָסוּר לְהַזְכִּירוֹ כְּשֶׁמִּתְכַוֵּן עַל עִנְיַן הַשָּׁלוֹם, כְּמוֹ שֶׁנִּתְבָּאֵר סִימָן פ״ד. (ב) כָּל שֶׁכֵּן שֵׁמוֹת הַמְיֻחָדִים לוֹ בִּלְשׁוֹנוֹת הַנָּכְרִים, שֶׁלְּכַמָּה דְּבָרִים דִּינָם כְּשֵׁמוֹת שֶׁבִּלְשׁוֹן הַקֹּדֶשׁ, (ג) כְּגוֹן לְעִנְיַן שְׁבוּעָה, וּלְעִנְיַן הַזְכָּרַת שֵׁם שָׁמַיִם לְבַטָּלָה, וּלְעִנְיָן קִלְלַת חֲבֵרוֹ בְּשֵׁם (עַיֵּן חֹשֶׁן מִשְׁפָּט סִימָן כ״ז) (וּלְעִנְיַן בִּרְכַּת הַשֵּׁם — עַיֵּן בְּיוֹרֵה דֵּעָה סִימָן ש״מ (סל״ו) [סל״ז]):

שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין. הַשֵּׁמוֹת הַמְיֻחָדִין (ז׳ שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין) — אָסוּר לְהַזְכִּירָן שָׁם אֲפִלּוּ בִּלְשׁוֹן חֹל. תמצית : אֲפִלּוּ שֵׁם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נִקְרָא בְּפִי כָּל עַם וְלָשׁוֹן (גאָ״ט, בוג״א) — אַף שֶׁמֻּתָּר לְמָחְקוֹ, יֵשׁ בּוֹ מִשּׁוּם בִּזָּיוֹן בִּמְקוֹם טִנֹּפֶת, כְּמוֹ בְּהַזְכָּרַת שָׁלוֹם (עיין סימן פ״ד). כָּל שֶׁכֵּן הַשֵּׁמוֹת הַמְיֻחָדִים לוֹ בִּלְשׁוֹנוֹת הַנָּכְרִים, שֶׁדִּינָם כְּשֵׁמוֹת שֶׁבִּלְשׁוֹן הַקֹּדֶשׁ (שְׁבוּעָה, שֵׁם שָׁמַיִם לְבַטָּלָה, קִלְלַת חֲבֵרוֹ בְּשֵׁם — חֹשֶׁן מִשְׁפָּט כ״ז ; בִּרְכַּת הַשֵּׁם — יוֹרֵה דֵּעָה ש״מ).

סעיף ד לְהַפְרִישׁ מֵאִסּוּר וְנִכְנְסָה אִשָּׁה בְּלִבּוֹ — מֻתָּר.

אִם נִזְדַּמֵּן לוֹ לְהַפְרִישׁ מִדְּבַר אִסּוּר בַּמֶּרְחָץ אוֹ בְּבֵית הַכִּסֵּא — מַפְרִישׁ אֲפִלּוּ בִּלְשׁוֹן הַקֹּדֶשׁ וּבְעִנְיַן קֹדֶשׁ, דְּהַיְנוּ לוֹמַר לוֹ אֲפִלּוּ בִּלְשׁוֹן הוֹרָאָה שֶׁאָסוּר לַעֲשׂוֹת כָּךְ. וְאֵין צָרִיךְ לְדַקְדֵּק וְלוֹמַר אַל תַּעֲשֶׂה כָּךְ בְּלִי לְשׁוֹן אִסּוּר. וְהוּא הַדִּין אִם נִכְנְסָה אִשָּׁה בְּלִבּוֹ — מֻתָּר לְהַרְהֵר בְּדִבְרֵי תּוֹרָה, מִפְּנֵי שֶׁהוּא כְּמוֹ לְהַפְרִישׁ מִדְּבַר אִסּוּר, שֶׁהַתּוֹרָה מַצֶּלֶת מֵהִרְהוּרִים רָעִים. וַאֲפִלּוּ שֶׁלֹּא לְהַפְרִישׁ מֵאִסּוּר מֻתָּר לוֹמַר לַחֲבֵרוֹ: עֲשֵׂה לִי כָּךְ וְכָךְ, אֲפִלּוּ אִם הוּא בְּעִנְיָן שֶׁמִּמֵּילָא הוּא הוֹרָאָה שֶׁבָּזֶה מוֹרֶה לוֹ שֶׁהוּא מֻתָּר לוֹ לַעֲשׂוֹת כָּךְ, כֵּיוָן שֶׁאֵינוֹ אוֹמְרוֹ בִּלְשׁוֹן הוֹרָאָה:

לְהַפְרִישׁ מֵאִסּוּר. נִזְדַּמֵּן לְהַפְרִישׁ מִדְּבַר אִסּוּר בַּמֶּרְחָץ אוֹ בְּבֵית הַכִּסֵּא — מַפְרִישׁ אֲפִלּוּ בִּלְשׁוֹן הַקֹּדֶשׁ וּבְעִנְיַן קֹדֶשׁ, אֲפִלּוּ בִּלְשׁוֹן הוֹרָאָה, וְאֵין צָרִיךְ לְדַקְדֵּק בְּלִי לְשׁוֹן אִסּוּר. וְהוּא הַדִּין נִכְנְסָה אִשָּׁה בְּלִבּוֹ — מֻתָּר לְהַרְהֵר בְּדִבְרֵי תּוֹרָה, שֶׁהַתּוֹרָה מַצֶּלֶת מֵהִרְהוּרִים רָעִים. וַאֲפִלּוּ שֶׁלֹּא לְהַפְרִישׁ — מֻתָּר לוֹמַר לַחֲבֵרוֹ עֲשֵׂה לִי כָּךְ וְכָךְ, כֵּיוָן שֶׁאֵינוֹ אוֹמְרוֹ בִּלְשׁוֹן הוֹרָאָה.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim פ״ה (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — בְּאֵיזֶה מְקוֹמוֹת אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע

היסוד : מְבוֹאוֹת הַמְטֻנָּפוֹת

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַמְּבוֹאוֹת הַמְטֻנָּפוֹת וְהַהַפְסָקָה. אחר כך אִסּוּר הַדִּבּוּר וְהַהִרְהוּר בִּמְקוֹם הַטִּנֹּפֶת. אחר כך דְּבָרִים שֶׁל חֹל וְכִנּוּיִים. ולבסוף הַשֵּׁמוֹת שֶׁאֵינָן נִמְחָקִין וְהַהַפְרָשָׁה מֵאִסּוּר.

מְבוֹאוֹת הַמְטֻנָּפוֹת — הַפְסָקָה וַחֲזָרָה

מְבוֹאוֹת הַמְטֻנָּפוֹת — הַפְסָקָה וַחֲזָרָה

היסוד. לֹא יֵלֵךְ בִּמְבוֹאוֹת הַמְטֻנָּפוֹת וְיִקְרָא ; הָיָה קוֹרֵא וּבָא מַפְסִיק כְּשֶׁיַּגִּיעַ, וּכְשֶׁיּוֹצֵא חוֹזֵר לַמָּקוֹם שֶׁפָּסַק אִם לֹא שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ (סעיף א).

המסקנה. הַחֲזָרָה תְּלוּיָה בְּדִין שָׁהָה כְּדֵי לִגְמֹר. עיין סימן 65 (הַפְסָקוֹת בַּקְּרִיאָה).

אִסּוּר דִּבּוּר וְהִרְהוּר

אִסּוּר דִּבּוּר וְהִרְהוּר

היסוד. כָּל דְּבַר קֹדֶשׁ אָסוּר בִּמְקוֹם הַטִּנֹּפֶת (צוֹאָה, מֵי רַגְלַיִם, בֵּית הַכִּסֵּא, בֵּית הַמֶּרְחָץ), אֲפִלּוּ בִּלְשׁוֹן חֹל וַאֲפִלּוּ לְהַרְהֵר בְּלִבּוֹ (סעיף א).

המסקנה. לָכֵן טוֹב לַחְשֹׁב חֶשְׁבּוֹנוֹתָיו, וּבְשַׁבָּת לְהַרְהֵר בְּצִיּוּרִים וּבִנְיָנִים נָאִים. עיין סימן 84 (הִרְהוּר בַּמֶּרְחָץ).

דְּבָרִים שֶׁל חֹל וְכִנּוּיִים

דְּבָרִים שֶׁל חֹל וְכִנּוּיִים

היסוד. דְּבָרִים שֶׁל חֹל מֻתָּר לְאָמְרָם שָׁם אֲפִלּוּ בִּלְשׁוֹן הַקֹּדֶשׁ (מִדַּת חֲסִידוּת לְהַחֲמִיר) ; וְהַכִּנּוּיִים (רַחוּם, חַנּוּן, נֶאֱמָן) — מֻתָּר מִן הַדִּין (סעיף ב).

המסקנה. הַחֲלֻקָּה בֵּין שֵׁם לְכִנּוּי, לֹא בֵּין הַלְּשׁוֹנוֹת. עיין סימן 84 (כִּנּוּיֵי הַשֵּׁם).

שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין וְהַפְרָשָׁה

שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין וְהַפְרָשָׁה

היסוד. ז׳ שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין — אָסוּר לְהַזְכִּירָן שָׁם אֲפִלּוּ בִּלְשׁוֹן חֹל ; וְאִם נִזְדַּמֵּן לְהַפְרִישׁ מֵאִסּוּר — מַפְרִישׁ אֲפִלּוּ בִּלְשׁוֹן הַקֹּדֶשׁ (סעיפים ג-ד).

המסקנה. שֶׁהַתּוֹרָה מַצֶּלֶת מֵהִרְהוּרִים רָעִים — לָכֵן הֶתֵּר הַהִרְהוּר תּוֹרָה כְּשֶׁנִּכְנְסָה אִשָּׁה בְּלִבּוֹ. עיין סימן 84 (בִּרְכַּת הַשֵּׁם).

הפניה ללימוד. היקף הסימן — הַהַפְסָקָה, אִסּוּר הַהִרְהוּר וְהַהַפְרָשָׁה מֵאִסּוּר — נוגע להנהגת ק״ש ומקומות הטינופת. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

בְּאֵיזֶה מְקוֹמוֹת אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן פ״ה. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
מְבוֹאוֹת הַמְטֻנָּפוֹת וְהַהַפְסָקָה או״ח פ״ה — לֹא יֵלֵךְ בִּמְבוֹאוֹת הַמְטֻנָּפוֹת וְיִקְרָא ; הָיָה קוֹרֵא וּבָא מַפְסִיק, וּכְשֶׁיּוֹצֵא חוֹזֵר לַמָּקוֹם שֶׁפָּסַק. מ״ב פ״ה — פְּרָטֵי הַהַפְסָקָה וְשִׁעוּר שָׁהָה כְּדֵי לִגְמֹר. ערוה״ש פ״ה — גֶּדֶר הַמָּבוֹי הַמְטֻנָּף וְהַהֶמְשֵׁךְ בַּקְּרִיאָה. דרך אדמו״ר הזקן — מַפְסִיק כְּשֶׁיַּגִּיעַ ; לֹא פָּסַק — עָלָיו הַכָּתוּב אוֹמֵר ; חוֹזֵר לַמָּקוֹם שֶׁפָּסַק אִם לֹא שָׁהָה (סעיף א).
אִסּוּר דִּבּוּר וְהִרְהוּר או״ח פ״ה — אֲפִלּוּ לְהַרְהֵר בְּדִבְרֵי תּוֹרָה אָסוּר בְּבֵית הַכִּסֵּא וּבַמֶּרְחָץ וּבִמְקוֹם הַטִּנֹּפֶת. מ״ב פ״ה — גֶּדֶר מְקוֹם הַטִּנֹּפֶת וְעֵצוֹת שֶׁלֹּא לָבוֹא לִידֵי הִרְהוּר. ערוה״ש פ״ה — דִּין הַהִרְהוּר וְלִמּוּד הִלְכוֹת הַמֶּרְחָץ בַּמֶּרְחָץ. דרך אדמו״ר הזקן — כָּל דְּבַר קֹדֶשׁ אָסוּר אֲפִלּוּ בִּלְשׁוֹן חֹל וַאֲפִלּוּ בְּהִרְהוּר ; טוֹב לַחְשֹׁב חֶשְׁבּוֹנוֹתָיו, וּבְשַׁבָּת צִיּוּרִים וּבִנְיָנִים נָאִים (סעיף א).
דְּבָרִים שֶׁל חֹל וְכִנּוּיִים או״ח פ״ה — דְּבָרִים שֶׁל חֹל מֻתָּר בִּלְשׁוֹן הַקֹּדֶשׁ ; הַכִּנּוּיִים כְּגוֹן רַחוּם נֶאֱמָן מֻתָּר. מ״ב פ״ה — הַהֶבְדֵּל בֵּין שֵׁמוֹת לְכִנּוּיִים וְעִנְיַן מִדַּת חֲסִידוּת. ערוה״ש פ״ה — גֶּדֶר הַכִּנּוּיִים הַנֶּאֱמָרִים גַּם בִּבְנֵי אָדָם. דרך אדמו״ר הזקן — מֻתָּר מִן הַדִּין, אֶלָּא שֶׁמִּדַּת חֲסִידוּת לְהַחֲמִיר ; הַכִּנּוּיִים לְפִי שֶׁגַּם בְּנֵי אָדָם מְכֻנִּים בָּהֶם (סעיף ב).
שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין וְהַפְרָשָׁה או״ח פ״ה — הַשֵּׁמוֹת שֶׁאֵינָן נִמְחָקִין אָסוּר לְהַזְכִּיר ; נִזְדַּמֵּן לְהַפְרִישׁ מַפְרִישׁ אֲפִלּוּ בִּלְשׁוֹן הַקֹּדֶשׁ. מ״ב פ״ה — דִּין הַשֵּׁמוֹת בִּלְשׁוֹנוֹת הַנָּכְרִים וְהֶתֵּר הַהַפְרָשָׁה. ערוה״ש פ״ה — הַזְכָּרַת הַשֵּׁם בְּלַעַז וְעִנְיַן בִּזָּיוֹן. דרך אדמו״ר הזקן — ז׳ שֵׁמוֹת אֲפִלּוּ בִּלְשׁוֹן חֹל ; מַפְרִישׁ אֲפִלּוּ בִּלְשׁוֹן הוֹרָאָה ; נִכְנְסָה אִשָּׁה — מֻתָּר לְהַרְהֵר, שֶׁהַתּוֹרָה מַצֶּלֶת (סעיפים ג-ד).

הטבלה נערכה מטקסט המחבר (אורח חיים פ״ה, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ד-כ״ה, שבת מ׳). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — מְקוֹם הַטִּנֹּפֶת

קו א — מְקוֹם הַטִּנֹּפֶת

יסוד הסימן : אָסוּר לִקְרוֹת אוֹ לְהַרְהֵר דְּבַר קֹדֶשׁ בִּמְקוֹם הַטִּנֹּפֶת (סעיף א).

קריאת הרב : הַגֶּדֶר נִקְבָּע בַּצּוֹאָה, בַּמֵּי רַגְלַיִם, בְּבֵית הַכִּסֵּא וּבְבֵית הַמֶּרְחָץ. עיין סימן 84.

מסקנה למעשה : לְהַפְסִיק כְּשֶׁיַּגִּיעַ, וְלַחֲזֹר כְּשֶׁיּוֹצֵא לְפִי דִּין שָׁהָה.

קו ב — הַהִרְהוּר וְהֶסַּח הַדַּעַת

קו ב — הַהִרְהוּר וְהֶסַּח הַדַּעַת

יסוד הסימן : אֲפִלּוּ לְהַרְהֵר בְּדִבְרֵי תּוֹרָה אָסוּר ; טוֹב לַחְשֹׁב חֶשְׁבּוֹנוֹתָיו (סעיף א).

קריאת הרב : בְּשַׁבָּת שֶׁאָסוּר לְחָשְׁבָם, יַפְנֶה לִבּוֹ לְהַרְהֵר בְּצִיּוּרִים וּבִנְיָנִים נָאִים. עיין סימן 83.

מסקנה למעשה : לְהַסִּיחַ אֶת הַדַּעַת בְּפֹעַל, שֶׁלֹּא יָבוֹא לִידֵי הִרְהוּר תּוֹרָה.

קו ג — כְּבוֹד הַשֵּׁם וְהַהַפְרָשָׁה

קו ג — כְּבוֹד הַשֵּׁם וְהַהַפְרָשָׁה

יסוד הסימן : שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין אָסוּר לְהַזְכִּיר ; לְהַפְרִישׁ מֵאִסּוּר מֻתָּר אֲפִלּוּ בִּלְשׁוֹן הַקֹּדֶשׁ (סעיפים ג-ד).

קריאת הרב : אֲפִלּוּ הַשֵּׁמוֹת בִּלְשׁוֹנוֹת הַנָּכְרִים דִּינָם קָרוֹב ; שֶׁהַתּוֹרָה מַצֶּלֶת מֵהִרְהוּרִים רָעִים. עיין סימן 86.

מסקנה למעשה : לְהַרְחִיק הִרְהוּר רַע עַל יְדֵי תּוֹרָה, וּלְהַפְרִישׁ מֵאִסּוּר אֲפִלּוּ בְּמָקוֹם זֶה.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן פ״ה ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן פ״ה (בְּאֵיזֶה מְקוֹמוֹת אָסוּר לִקְרוֹת קְרִיאַת שְׁמַע)

  • ① מְבוֹאוֹת הַמְטֻנָּפוֹת. אֵין לִקְרוֹת ק״ש בְּמָבוֹי מְטֻנָּף. הָיָה קוֹרֵא וּבָא — מַפְסִיק כְּשֶׁיַּגִּיעַ ; כְּשֶׁיּוֹצֵא חוֹזֵר לַמָּקוֹם שֶׁפָּסַק אִם לֹא שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ. יסוד : שוע״ר פ״ה סעיף א.
  • ② דִּבּוּר וְהִרְהוּר. בִּמְקוֹם הַטִּנֹּפֶת (בֵּית הַכִּסֵּא, בֵּית הַמֶּרְחָץ) — אֵין דְּבַר קֹדֶשׁ, אֲפִלּוּ בִּלְשׁוֹן חֹל, וַאֲפִלּוּ בְּהִרְהוּר. יסוד : שוע״ר פ״ה סעיף א.
  • ③ עֵצָה. בִּכְנִיסָה — טוֹב לַחְשֹׁב חֶשְׁבּוֹנוֹתָיו ; בְּשַׁבָּת — לְהַרְהֵר בְּצִיּוּרִים וּבִנְיָנִים נָאִים שֶׁלֹּא יָבוֹא לִידֵי הִרְהוּר תּוֹרָה. יסוד : שוע״ר פ״ה סעיף א.
  • ④ דְּבָרִים שֶׁל חֹל וְכִנּוּיִים. דְּבָרִים שֶׁל חֹל — מֻתָּר (מִדַּת חֲסִידוּת לְהַחֲמִיר) ; כִּנּוּיִים (רַחוּם, חַנּוּן, נֶאֱמָן) — מֻתָּר מִן הַדִּין ; אֲבָל ז׳ שֵׁמוֹת שֶׁאֵינָן נִמְחָקִין — אָסוּר. יסוד : שוע״ר פ״ה סעיפים ב-ג.
  • ⑤ הַפְרָשָׁה. לְהַפְרִישׁ מֵאִסּוּר — מַפְרִישׁ אֲפִלּוּ בִּלְשׁוֹן הַקֹּדֶשׁ ; נִכְנְסָה אִשָּׁה בְּלִבּוֹ — מֻתָּר לְהַרְהֵר בְּדִבְרֵי תּוֹרָה, שֶׁהַתּוֹרָה מַצֶּלֶת מֵהִרְהוּרִים רָעִים. יסוד : שוע״ר פ״ה סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן פ״ה (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַהַפְסָקָה, אִסּוּר הַהִרְהוּר, הַהַפְרָשָׁה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
