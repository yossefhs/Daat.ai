דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״ד · שֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה

דרכו של אדמו״ר הזקן בשֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה : מֶלֶךְ יִשְׂרָאֵל לֹא יְשִׁיבֶנּוּ, אֲבָל מֶלֶךְ אֻמּוֹת שֶׁמָּא יַהַרְגֶנּוּ — יַפְסִיק ; נָחָשׁ, עַקְרָב, שׁוֹר וּפִקּוּחַ נֶפֶשׁ ; הַחֲזָרָה לְאַחַר הֶפְסֵק ; קַדִּישׁ וּקְדֻשָּׁה — שׁוֹמֵעַ כְּעוֹנֶה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״ד — שֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה. אצל הרב הסימן נפרש לז׳ סְעִיפִים : לֹא לְהַפְסִיק בְּדִבּוּר וְעִנְיַן הַמֶּלֶךְ, פִּקּוּחַ נֶפֶשׁ (נָחָשׁ, עַקְרָב, שׁוֹר), הַחֲזָרָה לְאַחַר הֶפְסֵק (שָׁהָה, מֵזִיד וְשׁוֹגֵג), וְקַדִּישׁ וּקְדֻשָּׁה בְּ״שׁוֹמֵעַ כְּעוֹנֶה״ עִם הַקְּרִיאָה לַתּוֹרָה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״ד — ז׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״ד). אדמו״ר הזקן דן בו בְּשֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״ד — שֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״ד — שֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה — וּבוֹ ז׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.104), ואחריו תרגומנו. הסימן כולל ז׳ סעיפים (סעיף k → ‎.104.k).

סעיף א לֹא יַפְסִיק — מֶלֶךְ יִשְׂרָאֵל לֹא יְשִׁיבֶנּוּ ; מֶלֶךְ אֻמּוֹת שֶׁמָּא יַהַרְגֶנּוּ — יַפְסִיק.

שֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה וּבוֹ ז׳ סְעִיפִים:לֹא יַפְסִיק בִּתְפִלָּתוֹ. וַאֲפִלּוּ מֶלֶךְ יִשְׂרָאֵל שׁוֹאֵל בִּשְׁלוֹמוֹ — לֹא יְשִׁיבֶנּוּ. אֲבָל מֶלֶךְ אֻמּוֹת הָעוֹלָם שֶׁיֵּשׁ לָחֹש שֶׁמָּא יַהַרְגֶנּוּ — יַפְסִיק. וּמִכָּל מָקוֹם, אִם אֶפְשָׁר לוֹ לְקַצֵּר, דְּהַיְנוּ שֶׁיֹּאמַר תְּחִלַּת כָּל בְּרָכָה וְסוֹפָהּ קֹדֶם שֶׁיַּגִּיעַ אֵלָיו — יְקַצֵּר, שֶׁלֹּא יִצְטָרֵךְ לְהַפְסִיק. אוֹ אִם אֶפְשָׁר לוֹ לְהַטּוֹת מִן הַדֶּרֶךְ — יִטֶּה, וְלֹא יַפְסִיק בְּדִבּוּר. וְכֵן אִם הָיָה מִתְפַּלֵּל בַּדֶּרֶךְ וּבָאָה בְּהֵמָה אוֹ קָרוֹן כְּנֶגְדּוֹ — יִטֶּה מִן הַדֶּרֶךְ וְלֹא יַפְסִיק בְּדִבּוּר. אֲבָל בְּעִנְיָן אַחֵר — אֵין לוֹ לָצֵאת מִמְּקוֹמוֹ עַד שֶׁיִּגְמֹר תְּפִלָּתוֹ, אֶלָּא אִם כֵּן הוּא בְּתַחֲנוּנִים שֶׁאַחַר הַתְּפִלָּה, שֶׁאָז מֻתָּר לוֹ לַעֲקֹר מִמְּקוֹמוֹ עַל יְדֵי כְּרִיעוֹת וְהִשְׁתַּחֲוָיוֹת (אוֹ לְצֹרֶךְ אַחֵר). אֲבָל שֶׁלֹּא לְצֹרֶךְ — לֹא יָזוּז מִמְּקוֹמוֹ עַד שֶׁיִּפְסַע ג׳ פְּסִיעוֹת לַאֲחו‍ֹרָיו כְּדֶרֶךְ שֶׁנִּפְטָרִים מִן הַמֶּלֶךְ:

שֶׁלֹּא לְהַפְסִיק. אֲפִלּוּ מֶלֶךְ יִשְׂרָאֵל שׁוֹאֵל בִּשְׁלוֹמוֹ — לֹא יְשִׁיבֶנּוּ ; מֶלֶךְ אֻמּוֹת שֶׁיֵּשׁ לָחֹשׁ שֶׁמָּא יַהַרְגֶנּוּ — יַפְסִיק. אִם אֶפְשָׁר — יְקַצֵּר אוֹ יִטֶּה מִן הַדֶּרֶךְ וְלֹא יַפְסִיק בְּדִבּוּר, וְכֵן מִבְּהֵמָה אוֹ קָרוֹן בַּדֶּרֶךְ. שֶׁלֹּא לְצֹרֶךְ — לֹא יָזוּז עַד שֶׁיִּפְסַע ג׳ פְּסִיעוֹת לַאֲחוֹרָיו.

סעיף ב נָחָשׁ כָּרוּךְ — לֹא יַפְסִיק בְּדִבּוּר ; עַקְרָב וּפִקּוּחַ נֶפֶשׁ — פּוֹסֵק.

וַאֲפִלּוּ נָחָשׁ כָּרוּךְ עַל עֲקֵבוֹ — לֹא יַפְסִיק בְּדִבּוּר לוֹמַר לְאַחֵר לְהָסִיר מֵעָלָיו הַנָּחָשׁ. אֲבָל יָכוֹל לֵילֵךְ לְמָקוֹם אַחֵר כְּדֵי שֶׁיִּפֹּל הַנָּחָשׁ מֵרַגְלוֹ, שֶׁלֹּא מָצִינוּ הֲלִיכָה שֶׁנִּקְרֵאת הַפְסָקָה בְּשׁוּם מָקוֹם בְּאֶמְצַע עֲשִׂיַּת הַמִּצְוָה, אֶלָּא שֶׁבַּתְּפִלָּה הוֹאִיל וְעוֹמֵד לִפְנֵי הַמֶּלֶךְ — אֵין לָזוּז מִמְּקוֹמוֹ שֶׁלֹּא לְצֹרֶךְ עַד לְאַחַר הַתְּפִלָּה שֶׁיִּפָּטֵר מִלִּפְנֵי הַמֶּלֶךְ בְּג׳ פְּסִיעוֹת לַאֲחוֹרָיו כְּמוֹ שֶׁנִּתְבָּאֵר, אֲבָל לְצֹרֶךְ כָּזֶה מֻתָּר אֲפִלּוּ בְּאֶמְצַע בִּרְכוֹת י״ח. בַּמֶּה דְּבָרִים אֲמוּרִים? בְּנָחָשׁ שֶׁרֹב הַפַּעַם אֵינוֹ נוֹשֵׁךְ, אֲבָל עַקְרָב שֶׁהוּא מוּעָד לַעֲקֹץ וּלְהָמִית — פּוֹסֵק אֲפִלּוּ בְּדִבּוּר, שֶׁאֵין לְךָ דָּבָר שֶׁעוֹמֵד בִּפְנֵי פִּקּוּחַ נֶפֶשׁ. וַאֲפִלּוּ נָחָשׁ אִם רוֹאֶה שֶׁהוּא כָּעוּס כְּנֶגְדּוֹ וּבָא לְהַזִּיקוֹ — פּוֹסֵק אֲפִלּוּ בְּדִבּוּר:

נָחָשׁ וְעַקְרָב. נָחָשׁ כָּרוּךְ עַל עֲקֵבוֹ — לֹא יַפְסִיק בְּדִבּוּר, אֲבָל יָכוֹל לֵילֵךְ לְמָקוֹם אַחֵר. עַקְרָב הַמּוּעָד לַעֲקֹץ וּלְהָמִית — פּוֹסֵק אֲפִלּוּ בְּדִבּוּר, שֶׁאֵין דָּבָר עוֹמֵד בִּפְנֵי פִּקּוּחַ נֶפֶשׁ ; וְכֵן נָחָשׁ כָּעוּס הַבָּא לְהַזִּיקוֹ.

סעיף ג רָאָה שׁוֹר בָּא כְּנֶגְדּוֹ — פּוֹסֵק ; מֻחְזָקִים שֶׁאֵינָם מַזִּיקִין — אֵינוֹ פּוֹסֵק.

אִם רָאָה שׁוֹר בָּא כְּנֶגְדּוֹ — פּוֹסֵק, שֶׁמַּרְחִיקִין מִשּׁוֹר תָּם חֲמִשִּׁים אַמָּה וּמִמּוּעָד כִּמְלֹא עֵינָיו. וְאִם שְׁוָרִים שֶׁבַּמָּקוֹם הַהוּא מֻחְזָקִים שֶׁאֵינָם מַזִּיקִין — אֵינוֹ פּוֹסֵק, אֶלָּא אִם כֵּן רָאָה שׁוֹר שֶׁהִזִּיק כְּבָר פַּעַם אַחַת:

שׁוֹר בָּא כְּנֶגְדּוֹ. פּוֹסֵק, שֶׁמַּרְחִיקִין מִשּׁוֹר תָּם חֲמִשִּׁים אַמָּה וּמִמּוּעָד כִּמְלֹא עֵינָיו. וְאִם מֻחְזָקִים שֶׁאֵינָם מַזִּיקִין — אֵינוֹ פּוֹסֵק, אֶלָּא אִם כֵּן רָאָה שׁוֹר שֶׁהִזִּיק כְּבָר.

סעיף ד שָׁהָה כְּדֵי לִגְמֹר — חוֹזֵר לָרֹאשׁ ; מֵזִיד וְשׁוֹגֵג וּסְדֵר הַחֲזָרָה.

בְּכָל מָקוֹם שֶׁפּוֹסֵק, אִם שָׁהָה בְּהַפְסָקָתוֹ מֵחֲמַת אָנְסוֹ כְּדֵי לִגְמֹר אֶת כָּל הַתְּפִלָּה מֵרֹאשָׁהּ לְסוֹפָהּ — חוֹזֵר לָרֹאשׁ כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ס״ה, וּמְשַׁעֲרִין לְפִי הַמִּתְפַּלֵּל כְּמוֹ שֶׁנִּתְבָּאֵר שָׁם. וְאִם לֹא שָׁהָה כָּל כָּךְ — חוֹזֵר לִתְחִלַּת הַבְּרָכָה שֶׁפָּסַק בָּהּ, אִם פָּסַק בְּדִבּוּר בְּאֶמְצַע אֵיזוֹ בְּרָכָה מִי״ג בְּרָכוֹת אֶמְצָעִיּוֹת. אֲבָל אִם פָּסַק בְּג׳ רִאשׁוֹנוֹת — חוֹזֵר לָרֹאשׁ, מִפְּנֵי שֶׁהֵן חֲשׁוּבוֹת כִּבְרָכָה אַחַת. וְאִם בְּג׳ אַחֲרוֹנוֹת — חוֹזֵר לִ״רְצֵה״, שֶׁהֵן חֲשׁוּבוֹת גַּם כֵּן כְּאַחַת, שֶׁג׳ רִאשׁוֹנוֹת הֵן כְּעֶבֶד שֶׁמְּסַדֵּר שֶׁבַח לִפְנֵי רַבּוֹ וְג׳ אַחֲרוֹנוֹת הֵן כְּעֶבֶד שֶׁקִּבֵּל פְּרָס מֵרַבּוֹ וְנוֹטֵל רְשׁוּת וְהוֹלֵךְ לוֹ. בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁפָּסַק בְּדִבּוּר מֵחֲמַת אֹנֶס אוֹ בְּשׁוֹגֵג, אֲבָל אִם שָׂח בְּמֵזִיד — חוֹזֵר לָרֹאשׁ בְּכָל עִנְיָן (וְעַכְשָׁו שֶׁרֹב הָעוֹלָם אֵין נִזְהָרִין לְהָשִׂיחַ בְּבִרְכוֹת קְרִיאַת שְׁמַע — חֲשׁוּבִים שׁוֹגְגִים, שֶׁאוֹמֵר מֻתָּר שׁוֹגֵג הוּא). וַאֲפִלּוּ אִם הִפְסִיק בְּדִבּוּר מֵחֲמַת אֹנֶס אוֹ בְּשׁוֹגֵג בְּאֶמְצַע אֵיזוֹ בְּרָכָה (מֵהָאֶמְצָעִיּוֹת) וְלֹא חָזַר לִתְחִלַּת הַבְּרָכָה אֶלָּא לַמָּקוֹם שֶׁפָּסַק וְגָמַר תְּפִלָּתוֹ וְעָקַר רַגְלָיו — צָרִיךְ לַחֲזֹר לְרֹאשׁ הַתְּפִלָּה, לְפִי שֶׁאוֹתָהּ בְּרָכָה שֶׁהִפְסִיק בְּדִבּוּר בְּאֶמְצָעָהּ אֵינָהּ עוֹלָה לוֹ וּכְאִלּוּ לֹא אֲמָרָהּ כְּלָל, וְי״ח בְּרָכוֹת מְעַכְּבוֹת זוֹ אֶת זוֹ, לְעִנְיָן שֶׁאִם דִּלֵּג אַחַת מֵהֶן וְלֹא נִזְכַּר עַד שֶׁעָקַר רַגְלָיו — צָרִיךְ לַחֲזֹר לָרֹאשׁ כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קי״ט (אֲבָל מִי שֶׁאֵינוֹ יוֹדֵעַ לְהִתְפַּלֵּל כֻּלָּן אֶלָּא מִקְצָתָן — יֹאמַר אוֹתָן שֶׁיּוֹדֵעַ, וְאֵינָן מְעַכְּבוֹת זוֹ אֶת זוֹ כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תקצ״ג[)]. אֲבָל אִם הִפְסִיק בֵּין בְּרָכָה לִבְרָכָה — גּוֹמֵר תְּפִלָּתוֹ עַל הַסֵּדֶר אֲפִלּוּ לְכַתְּחִלָּה, וַאֲפִלּוּ שָׂח בְּמֵזִיד (חוּץ מִג׳ רִאשׁוֹנוֹת וְג׳ אַחֲרוֹנוֹת, שֶׁהֵן חֲשׁוּבוֹת כְּאַחַת לְכָל דָּבָר). (וְיֵשׁ אוֹמְרִים שֶׁאֲפִלּוּ שָׂח בְּמֵזִיד בְּאֶמְצַע בְּרָכָה יֵשׁ לוֹ תַּקָּנָה לַחֲזֹר לִתְחִלַּת אוֹתָהּ בְּרָכָה מִיָּד אַחַר הַשִּׂיחָה, אֶלָּא שֶׁאִם אַחַר הַשִּׂיחָה חָזַר לַמָּקוֹם שֶׁפָּסַק וְגָמַר תְּפִלָּתוֹ), (אוֹ אֲפִלּוּ אוֹתָהּ בְּרָכָה שֶׁשָּׂח בָּהּ לְבַדָּהּ), אַף עַל פִּי שֶׁעֲדַיִן לֹא עָקַר רַגְלָיו, שֶׁאִם הָיָה שׁוֹגֵג הָיָה חוֹזֵר לִתְחִלַּת הַבְּרָכָה שֶׁשָּׂח בָּהּ, עַכְשָׁו שֶׁהוּא מֵזִיד — צָרִיךְ לַחֲזֹר לָרֹאשׁ. וְכֵן הַדִּין אִם הֵזִיד וְלֹא הִזְכִּיר גְּבוּרוֹת גְּשָׁמִים בִּתְחִיַּת הַמֵּתִים אוֹ שְׁאֵלָה בְּ״בִרְכַּת הַשָּׁנִים״ אוֹ שֶׁל רֹאשׁ חֹדֶשׁ וְחֻלּוֹ שֶׁל מוֹעֵד בָּעֲבוֹדָה:

הַחֲזָרָה לְאַחַר הֶפְסֵק. שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ — חוֹזֵר לָרֹאשׁ (סימן ס״ה) ; וְאִם לֹא — לִתְחִלַּת הַבְּרָכָה שֶׁפָּסַק בָּהּ מִן הָאֶמְצָעִיּוֹת. ג׳ רִאשׁוֹנוֹת — לָרֹאשׁ ; ג׳ אַחֲרוֹנוֹת — לִרְצֵה. כָּל זֶה בְּאֹנֶס אוֹ בְּשׁוֹגֵג ; שָׂח בְּמֵזִיד — חוֹזֵר לָרֹאשׁ. בֵּין בְּרָכָה לִבְרָכָה — גּוֹמֵר עַל הַסֵּדֶר אֲפִלּוּ לְכַתְּחִלָּה.

סעיף ה קַדִּישׁ וּקְדֻשָּׁה — יִשְׁתֹּק וִיכַוֵּן, שׁוֹמֵעַ כְּעוֹנֶה.

אֲפִלּוּ לַעֲנוֹת קַדִּישׁ וּבָרְכוּ וּקְדֻשָּׁה — לֹא יַפְסִיק בִּתְפִלַּת י״ח, אֶלָּא יִשְׁתֹּק וִיכַוֵּן לְמַה שֶּׁאוֹמֵר הַשְּׁלִיחַ צִבּוּר וְיִהְיֶה כְּעוֹנֶה בְּפִיו, שֶׁשּׁוֹמֵעַ כְּעוֹנֶה בְּכָל מָקוֹם, וְאַף עַל פִּי כֵן אֵין בָּזֶה מִשּׁוּם הֶפְסֵק תְּפִלָּה, כֵּיוָן שֶׁאֵינוֹ מוֹצִיא בְּפִיו. וּכְשֶׁמַּגִּיעַ הַשְּׁלִיחַ צִבּוּר בַּקַּדִּישׁ לְ״יִתְבָּרֵךְ וְיִשְׁתַּבַּח״ — חוֹזֵר הוּא לִתְפִלָּתוֹ:

קַדִּישׁ וּקְדֻשָּׁה. אֲפִלּוּ לַעֲנוֹת קַדִּישׁ בָּרְכוּ וּקְדֻשָּׁה — לֹא יַפְסִיק בַּי״ח, אֶלָּא יִשְׁתֹּק וִיכַוֵּן לַשְּׁלִיחַ צִבּוּר וְיִהְיֶה כְּעוֹנֶה (שׁוֹמֵעַ כְּעוֹנֶה) ; וּכְשֶׁמַּגִּיעַ לְיִתְבָּרֵךְ וְיִשְׁתַּבַּח — חוֹזֵר לִתְפִלָּתוֹ.

סעיף ו קְרָאוּהוּ לְסֵפֶר תּוֹרָה — אֵינוֹ פּוֹסֵק.

הָיָה עוֹמֵד בִּתְפִלָּה וּקְרָאוּהוּ לְסֵפֶר תּוֹרָה — אֵינוֹ פּוֹסֵק, אֲפִלּוּ לְהָאוֹמְרִים שֶׁבִּקְרִיאַת שְׁמַע פּוֹסֵק וְעוֹלֶה לְסֵפֶר תּוֹרָה כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ס״ו:

קְרָאוּהוּ לְסֵפֶר תּוֹרָה. עוֹמֵד בִּתְפִלָּה וּקְרָאוּהוּ לַתּוֹרָה — אֵינוֹ פּוֹסֵק, אֲפִלּוּ לְהָאוֹמְרִים שֶׁבִּקְרִיאַת שְׁמַע פּוֹסֵק וְעוֹלֶה (סימן ס״ו).

סעיף ז אַחַר שְׁמוֹנֶה עֶשְׂרֵה קֹדֶם ״אֱלֹהַי נְצֹר״ — יָכוֹל לַעֲנוֹת.

אַחַר שֶׁסִּיֵּם שְׁמוֹנֶה עֶשְׂרֵה בְּרָכוֹת קֹדֶם ״אֱלֹהַי נְצֹר״ — יָכוֹל לַעֲנוֹת קְדֻשָּׁה וְקַדִּישׁ וּבָרְכוּ כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ב:

אַחַר שְׁמוֹנֶה עֶשְׂרֵה. אַחַר שֶׁסִּיֵּם שְׁמוֹנֶה עֶשְׂרֵה בְּרָכוֹת קֹדֶם אֱלֹהַי נְצֹר — יָכוֹל לַעֲנוֹת קְדֻשָּׁה וְקַדִּישׁ וּבָרְכוּ (סימן קכ״ב).

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק״ד (מהדורת קה״ת, ז׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — שֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה

היסוד : שֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה לֹא לְהַפְסִיק בְּדִבּוּר וְעִנְיַן הַמֶּלֶךְ. אחר כך פִּקּוּחַ נֶפֶשׁ — נָחָשׁ, עַקְרָב, שׁוֹר. אחר כך הַחֲזָרָה לְאַחַר הֶפְסֵק. ולבסוף קַדִּישׁ וּקְדֻשָּׁה בְּשׁוֹמֵעַ כְּעוֹנֶה וְהַקְּרִיאָה לַתּוֹרָה.

לֹא לְהַפְסִיק בְּדִבּוּר — וְעִנְיַן הַמֶּלֶךְ

לֹא לְהַפְסִיק בְּדִבּוּר — וְעִנְיַן הַמֶּלֶךְ

היסוד. מֶלֶךְ יִשְׂרָאֵל שׁוֹאֵל בִּשְׁלוֹמוֹ — לֹא יְשִׁיבֶנּוּ ; מֶלֶךְ אֻמּוֹת שֶׁמָּא יַהַרְגֶנּוּ — יַפְסִיק. אִם אֶפְשָׁר — יְקַצֵּר אוֹ יִטֶּה, וְלֹא יַפְסִיק בְּדִבּוּר (סעיף א).

המסקנה. שֶׁלֹּא לְצֹרֶךְ — לֹא יָזוּז עַד ג׳ פְּסִיעוֹת לַאֲחוֹרָיו. עיין סימן 103.

פִּקּוּחַ נֶפֶשׁ — נָחָשׁ, עַקְרָב

פִּקּוּחַ נֶפֶשׁ — נָחָשׁ, עַקְרָב

היסוד. נָחָשׁ כָּרוּךְ עַל עֲקֵבוֹ — לֹא יַפְסִיק בְּדִבּוּר, אֲבָל יָכוֹל לֵילֵךְ ; עַקְרָב מוּעָד לְהָמִית — פּוֹסֵק אֲפִלּוּ בְּדִבּוּר (סעיף ב).

המסקנה. אֵין לְךָ דָּבָר שֶׁעוֹמֵד בִּפְנֵי פִּקּוּחַ נֶפֶשׁ. עיין רמת הבסיס.

רָאָה שׁוֹר בָּא כְּנֶגְדּוֹ

רָאָה שׁוֹר בָּא כְּנֶגְדּוֹ

היסוד. רָאָה שׁוֹר בָּא כְּנֶגְדּוֹ — פּוֹסֵק (שׁוֹר תָּם חֲמִשִּׁים אַמָּה, מוּעָד כִּמְלֹא עֵינָיו) ; מֻחְזָקִים שֶׁאֵינָם מַזִּיקִין — אֵינוֹ פּוֹסֵק (סעיף ג).

המסקנה. מְשַׁעֲרִים אֶת הַסַּכָּנָה הָאֲמִתִּית שֶׁל הַמָּקוֹם וְהַבְּהֵמָה. עיין רמת הלמדן.

הַחֲזָרָה וְקַדִּישׁ וּקְדֻשָּׁה

הַחֲזָרָה וְקַדִּישׁ וּקְדֻשָּׁה

היסוד. שָׁהָה כְּדֵי לִגְמֹר — חוֹזֵר לָרֹאשׁ (סעיף ד) ; קַדִּישׁ וּקְדֻשָּׁה — יִשְׁתֹּק וִיכַוֵּן, שׁוֹמֵעַ כְּעוֹנֶה (סעיף ה).

המסקנה. קְרָאוּהוּ לַתּוֹרָה — אֵינוֹ פּוֹסֵק (סעיף ו) ; אַחַר הַי״ח — יָכוֹל לַעֲנוֹת (סעיף ז). עיין סימן 66.

הפניה ללימוד. היקף הסימן — לֹא לְהַפְסִיק, פִּקּוּחַ נֶפֶשׁ, הַחֲזָרָה וְשׁוֹמֵעַ כְּעוֹנֶה — נוגע להנהגת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

שֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק״ד. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מז׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הַהֶפְסֵק בְּדִבּוּר וְהַמֶּלֶךְ או״ח ק״ד — מֶלֶךְ יִשְׂרָאֵל לֹא יְשִׁיבֶנּוּ ; מֶלֶךְ אֻמּוֹת — יְקַצֵּר אוֹ יִטֶּה, וְאִם אִי אֶפְשָׁר — יַפְסִיק. מ״ב ק״ד — פְּרָטֵי הַהֲשָׁבָה וְהַהַטָּיָה וְהַקִּצּוּר. ערוה״ש ק״ד — גֶּדֶר אֵימַת הַמֶּלֶךְ בַּתְּפִלָּה. דרך אדמו״ר הזקן — מֶלֶךְ אֻמּוֹת שֶׁמָּא יַהַרְגֶנּוּ — יַפְסִיק ; שֶׁלֹּא לְצֹרֶךְ — לֹא יָזוּז עַד ג׳ פְּסִיעוֹת (סעיף א).
פִּקּוּחַ נֶפֶשׁ — נָחָשׁ, עַקְרָב, שׁוֹר או״ח ק״ד — נָחָשׁ לֹא יַפְסִיק ; עַקְרָב פּוֹסֵק ; שׁוֹר בָּא כְּנֶגְדּוֹ פּוֹסֵק. מ״ב ק״ד — שִׁעוּרֵי הַרְחָקָה מִשּׁוֹר תָּם וּמוּעָד. ערוה״ש ק״ד — סַכָּנָה הַדּוֹחָה אֶת הַתְּפִלָּה. דרך אדמו״ר הזקן — פּוֹסֵק אֲפִלּוּ בְּדִבּוּר, שֶׁאֵין דָּבָר עוֹמֵד בִּפְנֵי פִּקּוּחַ נֶפֶשׁ (סעיפים ב-ג).
הַחֲזָרָה לְאַחַר הֶפְסֵק או״ח ק״ד — שָׁהָה כְּדֵי לִגְמֹר כֻּלָּהּ חוֹזֵר לָרֹאשׁ ; ג׳ רִאשׁוֹנוֹת לָרֹאשׁ, ג׳ אַחֲרוֹנוֹת לִרְצֵה. מ״ב ק״ד — דִּקְדּוּקֵי מֵזִיד וְשׁוֹגֵג בַּחֲזָרָה. ערוה״ש ק״ד — סְדֵר הַחֲזָרָה בַּבְּרָכוֹת הָאֶמְצָעִיּוֹת. דרך אדמו״ר הזקן — שָׂח בְּמֵזִיד — חוֹזֵר לָרֹאשׁ ; בְּרָכָה שֶׁהִפְסִיק בְּאֶמְצָעָהּ אֵינָהּ עוֹלָה (סעיף ד).
קַדִּישׁ, קְדֻשָּׁה וּקְרִיאַת הַתּוֹרָה או״ח ק״ד — אֵינוֹ פּוֹסֵק לְקַדִּישׁ וּקְדֻשָּׁה, יִשְׁתֹּק וִיכַוֵּן ; קְרָאוּהוּ לַתּוֹרָה אֵינוֹ פּוֹסֵק. מ״ב ק״ד — דִּין שׁוֹמֵעַ כְּעוֹנֶה בִּתְפִלָּה. ערוה״ש ק״ד — גֶּדֶר הַכַּוָּנָה בְּלֹא הוֹצָאָה בַּפֶּה. דרך אדמו״ר הזקן — יִשְׁתֹּק וִיכַוֵּן, שׁוֹמֵעַ כְּעוֹנֶה, וְאֵין בָּזֶה הֶפְסֵק ; קְרָאוּהוּ לַתּוֹרָה — אֵינוֹ פּוֹסֵק (סעיפים ה-ז).

הטבלה נערכה מטקסט המחבר (אורח חיים ק״ד, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ל״ב-ל״ג). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ז׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — לֹא לְהַפְסִיק בְּדִבּוּר

קו א — לֹא לְהַפְסִיק בְּדִבּוּר

יסוד הסימן : מֶלֶךְ יִשְׂרָאֵל — לֹא יְשִׁיבֶנּוּ ; אִם אֶפְשָׁר — יְקַצֵּר אוֹ יִטֶּה (סעיף א).

קריאת הרב : הָאָדָם עוֹמֵד לִפְנֵי הַמֶּלֶךְ ; שֶׁלֹּא לְצֹרֶךְ אֵינוֹ זָז אֶלָּא אַחַר ג׳ פְּסִיעוֹת. עיין סימן 103.

מסקנה למעשה : לְהַעֲדִיף מַעֲשֶׂה (לְהַטּוֹת, לְקַצֵּר) עַל דִּבּוּר ; לֹא לְהַפְסִיק אֶלָּא מִפְּנֵי סַכָּנָה אֲמִתִּית.

קו ב — פִּקּוּחַ נֶפֶשׁ דּוֹחֶה

קו ב — פִּקּוּחַ נֶפֶשׁ דּוֹחֶה

יסוד הסימן : נָחָשׁ — לֹא יַפְסִיק בְּדִבּוּר (אֲבָל יָכוֹל לֵילֵךְ) ; עַקְרָב אוֹ נָחָשׁ כָּעוּס — פּוֹסֵק אֲפִלּוּ בְּדִבּוּר (סעיפים ב-ג).

קריאת הרב : אֵין לְךָ דָּבָר שֶׁעוֹמֵד בִּפְנֵי פִּקּוּחַ נֶפֶשׁ ; הַשּׁוֹר נִדּוֹן לְפִי תָּם / מוּעָד. עיין רמת הלמדן.

מסקנה למעשה : מוּל סַכָּנָה בְּרוּרָה — פּוֹסֵק אֲפִלּוּ בְּדִבּוּר ; וְאִם לָאו — מַטֶּה בְּלִי לְדַבֵּר.

קו ג — שׁוֹמֵעַ כְּעוֹנֶה וְהַחֲזָרָה

קו ג — שׁוֹמֵעַ כְּעוֹנֶה וְהַחֲזָרָה

יסוד הסימן : קַדִּישׁ וּקְדֻשָּׁה — יִשְׁתֹּק וִיכַוֵּן (שׁוֹמֵעַ כְּעוֹנֶה) ; שָׁהָה כְּדֵי לִגְמֹר — חוֹזֵר לָרֹאשׁ (סעיפים ד-ה).

קריאת הרב : שְׁמִיעָה בְּלֹא הוֹצָאָה בַּפֶּה אֵינָהּ הֶפְסֵק ; הַמֵּזִיד חָמוּר מִן הַשּׁוֹגֵג. עיין סימן 66.

מסקנה למעשה : בַּתְּפִלָּה — לִשְׁתֹּק וְלִשְׁמֹעַ אֶת הַשְּׁלִיחַ צִבּוּר ; לַעֲנוֹת רַק אַחַר אֱלֹהַי נְצֹר (סעיף ז).

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״ד ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״ד (שֶׁלֹּא לְהַפְסִיק בַּתְּפִלָּה)

  • ① הַמֶּלֶךְ. מֶלֶךְ יִשְׂרָאֵל שׁוֹאֵל בִּשְׁלוֹמוֹ — לֹא יְשִׁיבֶנּוּ ; מֶלֶךְ אֻמּוֹת שֶׁמָּא יַהַרְגֶנּוּ — יַפְסִיק ; אִם אֶפְשָׁר — יְקַצֵּר אוֹ יִטֶּה. יסוד : שוע״ר ק״ד סעיף א.
  • ② נָחָשׁ. נָחָשׁ כָּרוּךְ — לֹא יַפְסִיק בְּדִבּוּר, אֲבָל יָכוֹל לֵילֵךְ לְמָקוֹם אַחֵר. יסוד : שוע״ר ק״ד סעיף ב.
  • ③ עַקְרָב / פִּקּוּחַ נֶפֶשׁ. עַקְרָב, אוֹ נָחָשׁ כָּעוּס הַבָּא לְהַזִּיק — פּוֹסֵק אֲפִלּוּ בְּדִבּוּר. יסוד : שוע״ר ק״ד סעיף ב.
  • ④ שׁוֹר. רָאָה שׁוֹר בָּא כְּנֶגְדּוֹ — פּוֹסֵק ; מֻחְזָקִים שֶׁאֵינָם מַזִּיקִין — אֵינוֹ פּוֹסֵק. יסוד : שוע״ר ק״ד סעיף ג.
  • ⑤ הַחֲזָרָה. שָׁהָה כְּדֵי לִגְמֹר כֻּלָּהּ — לָרֹאשׁ ; שָׂח בְּמֵזִיד — לָרֹאשׁ ; בֵּין בְּרָכָה לִבְרָכָה — עַל הַסֵּדֶר. יסוד : שוע״ר ק״ד סעיף ד.
  • ⑥ קַדִּישׁ וּקְדֻשָּׁה. לֹא יַפְסִיק ; יִשְׁתֹּק וִיכַוֵּן (שׁוֹמֵעַ כְּעוֹנֶה) ; קְרָאוּהוּ לַתּוֹרָה — אֵינוֹ פּוֹסֵק. יסוד : שוע״ר ק״ד סעיפים ה-ו.
  • ⑦ אַחַר שְׁמוֹנֶה עֶשְׂרֵה. קֹדֶם אֱלֹהַי נְצֹר — יָכוֹל לַעֲנוֹת קְדֻשָּׁה וְקַדִּישׁ וּבָרְכוּ. יסוד : שוע״ר ק״ד סעיף ז.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק״ד (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — לֹא לְהַפְסִיק, פִּקּוּחַ נֶפֶשׁ, שׁוֹמֵעַ כְּעוֹנֶה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
