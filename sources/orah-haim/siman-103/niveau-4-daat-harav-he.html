דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״ג · מִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ אוֹ עִטּוּשׁ

דרכו של אדמו״ר הזקן במִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ אוֹ עִטּוּשׁ : רוּחַ מִלְּמַטָּה בַּתְּפִלָּה — מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ ; נִצְטַעֵר וְאֵינוֹ יָכוֹל — מִנְהַג עַכְשָׁו וּכְבוֹד הַבְּרִיּוֹת ; הַמִּתְעַטֵּשׁ — מִלְּמַטָּה סִימָן רַע וּמִלְמַעְלָה סִימָן יָפֶה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״ג — מִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ אוֹ עִטּוּשׁ, עניין הנידון בצניעות. אצל הרב הסימן נפרש לג׳ סְעִיפִים : הָרוּחַ מִלְּמַטָּה בַּתְּפִלָּה וְהַמַּמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ, הַנִּצְטַעֵר וְאֵינוֹ יָכוֹל עִם הַחִלּוּק בֵּין בֵּיתוֹ לְצִבּוּר וְהַכְּבוֹד הַבְּרִיּוֹת, וְהַמִּתְעַטֵּשׁ — סִימָן רַע וְסִימָן יָפֶה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״ג — ג׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״ג). אדמו״ר הזקן דן בו, בצניעות, בְּמִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ אוֹ עִטּוּשׁ. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״ג — מִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ אוֹ עִטּוּשׁ

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״ג — מִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ אוֹ עִטּוּשׁ — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.103), ואחריו תרגומנו. הסימן כולל ג׳ סעיפים (סעיף k → ‎.103.k).

סעיף א רוּחַ מִלְּמַטָּה בַּתְּפִלָּה — מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ וְחוֹזֵר לַמָּקוֹם שֶׁפָּסַק.

מִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ אוֹ עִטּוּשׁ וּבוֹ ג׳ סְעִיפִים:הָיָה עוֹמֵד בִּתְפִלָּה וְיָצָא מִמֶּנּוּ רוּחַ מִלְּמַטָּה — מַמְתִּין עַד שֶׁיִּכְלֶה (הָרוּחַ) [הָרֵיחַ] וְחוֹזֵר לַמָּקוֹם שֶׁפָּסַק:

רוּחַ מִלְּמַטָּה. מִי שֶׁהָיָה עוֹמֵד בִּתְפִלָּה וְיָצָא מִמֶּנּוּ רוּחַ מִלְּמַטָּה — מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ, וְאַחַר כָּךְ חוֹזֵר לַמָּקוֹם שֶׁפָּסַק. אֵין הַתְּפִלָּה נִפְסֶלֶת, אֶלָּא מַמְתִּין בְּצִנְעָה וּמַשְׁלִים.

סעיף ב נִצְטַעֵר וְאֵינוֹ יָכוֹל — מִנְהַג עַכְשָׁו, כְּבוֹד הַבְּרִיּוֹת ; וְאִם יָכוֹל — חַיָּב לְהַעֲמִיד עַצְמוֹ.

בִּקֵּשׁ לָצֵאת מִמֶּנּוּ רוּחַ מִלְּמַטָּה וְנִצְטַעֵר הַרְבֵּה וְאֵינוֹ יָכוֹל לְהַעֲמִיד עַצְמוֹ — הוֹלֵךְ לַאֲחוֹרָיו ד׳ אַמּוֹת וּמוֹצִיא הָרוּחַ. וְלֹא יֵלֵךְ לִצְדָדִים, מִפְּנֵי שֶׁצָּרִיךְ שֶׁיִּהְיוּ פָּנָיו מְכֻוָּנִים כְּנֶגֶד הַמָּקוֹם שֶׁהִתְחִיל לְהִתְפַּלֵּל, שֶׁלֹּא יְהֵא נִרְאֶה כְּפוֹסֵק לְגַמְרֵי וְאֵין כַּוָּנָתוֹ לַחֲזֹר לִתְפִלָּתוֹ. וּמַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ וְאוֹמֵר: ״רִבּוֹן הָעוֹלָמִים, יְצַרְתָּנוּ נְקָבִים נְקָבִים, חֲלוּלִים חֲלוּלִים, גָּלוּי וְיָדוּעַ לְפָנֶיךָ חֶרְפָּתֵנוּ וּכְלִמָּתֵנוּ, חֶרְפָּה וּכְלִמָּה בְּחַיֵּינוּ, רִמָּה וְתוֹלֵעָה בְּמוֹתֵנוּ״, וְאֵין אֲמִירָה זוֹ חָשׁוּב הֶפְסֵק בַּתְּפִלָּה, הוֹאִיל וּכְבָר מֻפְסֶקֶת תְּפִלָּתוֹ עַל יְדֵי הָרוּחַ, וְחוֹזֵר לִמְקוֹמוֹ, וְחוֹזֵר לַמָּקוֹם שֶׁפָּסַק. וְלֹא יִגְמֹר תְּפִלָּתוֹ בַּמָּקוֹם שֶׁנִּתְרַחֵק שָׁם, הוֹאִיל וּכְבָר הִתְחִיל לְהִתְפַּלֵּל כָּאן וְקָבְעָה הַשְּׁכִינָה מָקוֹם כָּאן, וּכְשֶׁבִּקֵּשׁ לְהִתְעַטֵּשׁ נִתְרַחֵק מִמְּקוֹם הַשְּׁכִינָה, לְפִיכָךְ צָרִיךְ הוּא לַחֲזֹר לַמָּקוֹם שֶׁקָּבַע לַשְּׁכִינָה. וְעַכְשָׁו לֹא נָהֲגוּ אֲפִלּוּ הַמְדַקְדְּקִים לְהַרְחִיק ד׳ אַמּוֹת וְלוֹמַר ״רִבּוֹן וְכו׳״. וְיֵשׁ שֶׁלִּמְּדוּ זְכוּת לוֹמַר שֶׁלֹּא הִצְרִיכוּ לְהַרְחִיק אֶלָּא כְּשֶׁמִּתְפַּלֵּל בְּבֵיתוֹ, אֲבָל כְּשֶׁמִּתְפַּלֵּל בְּצִבּוּר שֶׁזֶּה לוֹ בִּזָּיוֹן גָּדוֹל שֶׁנִּרְאֶה לַכֹּל בָּשְׁתּוֹ וּכְלִמָּתוֹ — אֵינוֹ צָרִיךְ לְהַרְחִיק כְּלָל, שֶׁגָּדוֹל כְּבוֹד הַבְּרִיּוֹת. וְכֵיוָן שֶׁאֵין צָרִיךְ לְהַרְחִיק, אֵין לוֹמַר גַּם כֵּן ״רִבּוֹן כו׳״, כֵּיוָן שֶׁאֵין חֶרְפָּתוֹ וּכְלִמָּתוֹ נִכֶּרֶת, רַק מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ. וְכָל זֶה כְּשֶׁאֵינוֹ יָכוֹל לְהַעֲמִיד עַצְמוֹ, אֲבָל אִם יָכוֹל — חַיָּב לְהַעֲמִיד עַצְמוֹ בְּכָל מַה דְּאֶפְשָׁר שֶׁלֹּא לְהָפִיחַ בַּתְּפִלָּה, וְאֵין בָּזֶה מִשּׁוּם ״בַּל תְּשַׁקְּצוּ״. וַאֲפִלּוּ אִם נִצְרָךְ לִנְקָבָיו — יַעֲמִיד עַצְמוֹ כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן צ״ב:

נִצְטַעֵר וְאֵינוֹ יָכוֹל. תמצית : מִי שֶׁנִּצְטַעֵר הַרְבֵּה וְאֵינוֹ יָכוֹל לְהַעֲמִיד עַצְמוֹ — הָיָה הוֹלֵךְ לַאֲחוֹרָיו ד׳ אַמּוֹת ; וְלֹא לִצְדָדִים, שֶׁפָּנָיו צְרִיכִים לִהְיוֹת כְּנֶגֶד מְקוֹם הַתְחָלָתוֹ, שֶׁלֹּא יֵרָאֶה כְּפוֹסֵק לְגַמְרֵי. מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ וְאוֹמֵר ״רִבּוֹן הָעוֹלָמִים״ — וְאֵין זוֹ הֶפְסֵק, שֶׁכְּבָר מֻפְסֶקֶת תְּפִלָּתוֹ — וְחוֹזֵר לִמְקוֹמוֹ וְלַמָּקוֹם שֶׁפָּסַק. וְעַכְשָׁו אַף הַמְדַקְדְּקִים אֵינָם נוֹהֲגִים לְהַרְחִיק וְלוֹמַר ״רִבּוֹן״ ; יֵשׁ שֶׁלִּמְּדוּ זְכוּת : לֹא הִצְרִיכוּ אֶלָּא בְּבֵיתוֹ, אֲבָל בְּצִבּוּר שֶׁהוּא בִּזָּיוֹן גָּדוֹל — אֵינוֹ מַרְחִיק כְּלָל, שֶׁגָּדוֹל כְּבוֹד הַבְּרִיּוֹת, וְאַף אֵינוֹ אוֹמֵר ״רִבּוֹן״, רַק מַמְתִּין. וְכָל זֶה כְּשֶׁאֵינוֹ יָכוֹל ; אֲבָל אִם יָכוֹל — חַיָּב לְהַעֲמִיד עַצְמוֹ בְּכָל מַה דְּאֶפְשָׁר, וְאֵין בָּזֶה מִשּׁוּם ״בַּל תְּשַׁקְּצוּ״ ; וְכֵן אִם נִצְרָךְ לִנְקָבָיו (עיין סִימָן צ״ב).

סעיף ג הַמִּתְעַטֵּשׁ — מִלְּמַטָּה סִימָן רַע לוֹ, מִלְמַעְלָה סִימָן יָפֶה לוֹ.

הַמִּתְעַטֵּשׁ בִּתְפִלָּתוֹ מִלְּמַטָּה, אֲפִלּוּ הֲפָחָה מוּעֶטֶת שֶׁאֵינָהּ מַשְׁמַעַת קוֹל — סִימָן רַע לוֹ, מִלְמַעְלָה — סִימָן יָפֶה לוֹ, שֶׁכְּשֵׁם שֶׁעוֹשִׂין לוֹ נַחַת רוּחַ מִלְּמַטָּה שֶׁהָעִטּוּשׁ הוּא נוֹחַ לָאָדָם — כָּךְ עוֹשִׂין לוֹ נַחַת רוּחַ מִלְמַעְלָה לְמַלֹּאות שְׁאֵלָתוֹ וּבַקָּשָׁתוֹ:

הַמִּתְעַטֵּשׁ. בַּתְּפִלָּה : מִלְּמַטָּה, אֲפִלּוּ הֲפָחָה מוּעֶטֶת שֶׁאֵינָהּ מַשְׁמַעַת קוֹל — סִימָן רַע לוֹ ; מִלְמַעְלָה, עִטּוּשׁ — סִימָן יָפֶה לוֹ : כְּשֵׁם שֶׁעוֹשִׂין לוֹ נַחַת רוּחַ מִלְּמַטָּה, שֶׁהָעִטּוּשׁ נוֹחַ לָאָדָם, כָּךְ עוֹשִׂין לוֹ נַחַת רוּחַ מִלְמַעְלָה לְמַלֹּאות שְׁאֵלָתוֹ וּבַקָּשָׁתוֹ.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק״ג (מהדורת קה״ת, ג׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — מִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ אוֹ עִטּוּשׁ

היסוד : רוּחַ אוֹ עִטּוּשׁ בַּתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים, כולם בצניעות. תחילה הָרוּחַ מִלְּמַטָּה בַּתְּפִלָּה וְהַמַּמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ. אחר כך הַנִּצְטַעֵר וְאֵינוֹ יָכוֹל עִם הַחִלּוּק בֵּין בֵּיתוֹ לְצִבּוּר וְהַכְּבוֹד הַבְּרִיּוֹת. ולבסוף הַמִּתְעַטֵּשׁ — סִימָן רַע וְסִימָן יָפֶה.

רוּחַ בַּתְּפִלָּה — מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ

רוּחַ בַּתְּפִלָּה — מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ

היסוד. עוֹמֵד בִּתְפִלָּה וְיָצָא מִמֶּנּוּ רוּחַ מִלְּמַטָּה — אֵינוֹ מַרְחִיק וְאֵינוֹ מְדַבֵּר : מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ, וְחוֹזֵר לַמָּקוֹם שֶׁפָּסַק (סעיף א).

המסקנה. אֵין הַתְּפִלָּה נִפְסֶלֶת ; הַהַמְתָּנָה שׁוֹמֶרֶת עַל כְּבוֹד מְקוֹם הַתְּפִלָּה. עיין סימן 97 (הפסק ונקיון בתפילה).

נִצְטַעֵר וְאֵינוֹ יָכוֹל — כְּבוֹד הַבְּרִיּוֹת

נִצְטַעֵר וְאֵינוֹ יָכוֹל — כְּבוֹד הַבְּרִיּוֹת

היסוד. מִי שֶׁאֵינוֹ יָכוֹל לְהַעֲמִיד עַצְמוֹ הָיָה מַרְחִיק ד׳ אַמּוֹת וְאוֹמֵר ״רִבּוֹן״ ; אֲבָל עַכְשָׁו אֵין נוֹהֲגִין כֵּן, וּבְיִחוּד בְּצִבּוּר — שֶׁגָּדוֹל כְּבוֹד הַבְּרִיּוֹת — רַק מַמְתִּין (סעיף ב).

המסקנה. אִם יָכוֹל לְהַעֲמִיד עַצְמוֹ — חַיָּב, וְאֵין בָּזֶה ״בַּל תְּשַׁקְּצוּ״. עיין סימן 92 (העמדת עצמו בתפילה).

הַמִּתְעַטֵּשׁ — סִימָן רַע וְסִימָן יָפֶה

הַמִּתְעַטֵּשׁ — סִימָן רַע וְסִימָן יָפֶה

היסוד. מִלְּמַטָּה, אֲפִלּוּ הֲפָחָה מוּעֶטֶת — סִימָן רַע לוֹ ; מִלְמַעְלָה, עִטּוּשׁ — סִימָן יָפֶה לוֹ, נַחַת רוּחַ לְמַלֹּאות שְׁאֵלָתוֹ (סעיף ג).

המסקנה. אֵינוֹ נִבְהָל, אֶלָּא מַמְשִׁיךְ תְּפִלָּתוֹ בְּיִשּׁוּב הַדַּעַת. עיין סימן 104 (הפסק וסדר בתפילה).

הפניה ללימוד. היקף הסימן — הַמַּמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ, הַכְּבוֹד הַבְּרִיּוֹת וְהַסִּימָן שֶׁל הַמִּתְעַטֵּשׁ — נוגע להנהגה הצנועה של המתפלל. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בצניעות ובלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

מִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק״ג. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מג׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
רוּחַ מִלְּמַטָּה בַּתְּפִלָּה או״ח ק״ג — יָצָא מִמֶּנּוּ רוּחַ מִלְּמַטָּה, מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ וְחוֹזֵר וּמִתְפַּלֵּל. מ״ב ק״ג — גֶּדֶר הַהַמְתָּנָה וְשֶׁאֵין בָּזֶה הֶפְסֵק. ערוה״ש ק״ג — דִּין הָרוּחַ שֶׁנִּזְדַּמְּנָה בַּתְּפִלָּה. דרך אדמו״ר הזקן — מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ וְחוֹזֵר לַמָּקוֹם שֶׁפָּסַק (סעיף א).
נִצְטַעֵר וְאֵינוֹ יָכוֹל וּכְבוֹד הַבְּרִיּוֹת או״ח ק״ג — הוֹלֵךְ לַאֲחוֹרָיו ד׳ אַמּוֹת וְאוֹמֵר ״רִבּוֹן הָעוֹלָמִים״ ; וּבַהַגָּה : בְּצִבּוּר אֵין צָרִיךְ לְהַרְחִיק. מ״ב ק״ג — מִנְהַג עַכְשָׁו וּתְנַאי הַהַרְחָקָה. ערוה״ש ק״ג — גֶּדֶר הָאִי אֶפְשָׁר לְהַעֲמִיד עַצְמוֹ. דרך אדמו״ר הזקן — עַכְשָׁו אַף הַמְדַקְדְּקִים אֵין מַרְחִיקִין וְאֵין אוֹמְרִים ״רִבּוֹן״ ; בְּצִבּוּר — שֶׁגָּדוֹל כְּבוֹד הַבְּרִיּוֹת ; וְאִם יָכוֹל — חַיָּב לְהַעֲמִיד עַצְמוֹ (סעיף ב).
הַמִּתְעַטֵּשׁ — סִימָן או״ח ק״ג — מִלְּמַטָּה סִימָן רַע לוֹ, מִלְמַעְלָה סִימָן יָפֶה לוֹ. מ״ב ק״ג — טַעַם הַסִּימָן וְהַנְהָגָה בּוֹ. ערוה״ש ק״ג — עִנְיַן הָעִטּוּשׁ בַּתְּפִלָּה. דרך אדמו״ר הזקן — אֲפִלּוּ הֲפָחָה מוּעֶטֶת מִלְּמַטָּה סִימָן רַע ; מִלְמַעְלָה נַחַת רוּחַ לְמַלֹּאות שְׁאֵלָתוֹ (סעיף ג).

הטבלה נערכה מטקסט המחבר (אורח חיים ק״ג, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם, תרומת הדשן סי׳ ט״ז, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ד, כ״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ

קו א — מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ

יסוד הסימן : רוּחַ מִלְּמַטָּה בַּתְּפִלָּה — מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ וְחוֹזֵר לַמָּקוֹם שֶׁפָּסַק (סעיף א).

קריאת הרב : הַתְּפִלָּה נִמְשֶׁכֶת וְאֵינָהּ נִפְסֶלֶת ; הַהַמְתָּנָה מְכַבֶּדֶת אֶת מְקוֹם הַתְּפִלָּה. עיין סימן 97.

מסקנה למעשה : לֹא לְדַבֵּר וְלֹא לָנוּעַ — לְהַמְתִּין בְּשֶׁקֶט, וְאַחַר כָּךְ לַחֲזֹר לַמָּקוֹם שֶׁפָּסַק.

קו ב — כְּבוֹד הַבְּרִיּוֹת וּמִנְהַג עַכְשָׁו

קו ב — כְּבוֹד הַבְּרִיּוֹת וּמִנְהַג עַכְשָׁו

יסוד הסימן : נִצְטַעֵר וְאֵינוֹ יָכוֹל — עַכְשָׁו אֵין מַרְחִיקִין וְאֵין אוֹמְרִים ״רִבּוֹן״, וּבְצִבּוּר שֶׁגָּדוֹל כְּבוֹד הַבְּרִיּוֹת (סעיף ב).

קריאת הרב : הַהֲלָכָה חוֹשֶׁשֶׁת לַחֲסֹךְ אֶת בָּשְׁתּוֹ הַנִּרְאֵית שֶׁל הַמִּתְפַּלֵּל. עיין סימן 92.

מסקנה למעשה : בַּצִּבּוּר אֵין מוֹשְׁכִים תְּשׂוּמֶת לֵב : מַמְתִּינִים בְּצִנְעָה עַד שֶׁיִּכְלֶה הָרֵיחַ.

קו ג — לְהַעֲמִיד עַצְמוֹ וְהַסִּימָן

קו ג — לְהַעֲמִיד עַצְמוֹ וְהַסִּימָן

יסוד הסימן : אִם יָכוֹל — חַיָּב לְהַעֲמִיד עַצְמוֹ, וְאֵין בָּזֶה ״בַּל תְּשַׁקְּצוּ״ ; וְהַמִּתְעַטֵּשׁ מִלְּמַטָּה סִימָן רַע וּמִלְמַעְלָה סִימָן יָפֶה (סעיפים ב-ג).

קריאת הרב : מַאֲמַץ הַהַעֲמָדָה הוּא מִכְּבוֹד הַתְּפִלָּה, בְּלִי חֻמְרָה יְתֵרָה. עיין סימן 104.

מסקנה למעשה : לְהַעֲמִיד עַצְמוֹ בְּכָל מַה דְּאֶפְשָׁר ; וְאֵין לְהִבָּהֵל מִן הַסִּימָן — לְהַמְשִׁיךְ בְּיִשּׁוּב הַדַּעַת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״ג ברגישות אדמו״ר הזקן וחב״ד — מוצגות בצניעות, ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״ג (מִי שֶׁנִּזְדַּמֵּן לוֹ רוּחַ בִּתְפִלָּתוֹ אוֹ עִטּוּשׁ)

  • ① רוּחַ בַּתְּפִלָּה. יָצָא מִמֶּנּוּ רוּחַ מִלְּמַטָּה בַּתְּפִלָּה — אֵינוֹ מַרְחִיק וְאֵינוֹ מְדַבֵּר : מַמְתִּין עַד שֶׁיִּכְלֶה הָרֵיחַ, וְחוֹזֵר לַמָּקוֹם שֶׁפָּסַק. יסוד : שוע״ר ק״ג סעיף א.
  • ② נִצְטַעֵר וְאֵינוֹ יָכוֹל. עַכְשָׁו אֵין מַרְחִיקִין ד׳ אַמּוֹת וְאֵין אוֹמְרִים ״רִבּוֹן״ ; וּבִפְרָט בְּצִבּוּר — שֶׁגָּדוֹל כְּבוֹד הַבְּרִיּוֹת — רַק מַמְתִּין. יסוד : שוע״ר ק״ג סעיף ב.
  • ③ אִם יָכוֹל. אִם יָכוֹל לְהַעֲמִיד עַצְמוֹ — חַיָּב בְּכָל מַה דְּאֶפְשָׁר, וְאֵין בָּזֶה ״בַּל תְּשַׁקְּצוּ״ ; וְכֵן אִם נִצְרָךְ לִנְקָבָיו (עיין סִימָן צ״ב). יסוד : שוע״ר ק״ג סעיף ב.
  • ④ הַמִּתְעַטֵּשׁ. מִלְּמַטָּה, אֲפִלּוּ מוּעֶטֶת — סִימָן רַע ; מִלְמַעְלָה, עִטּוּשׁ — סִימָן יָפֶה. אֵינוֹ נִבְהָל, אֶלָּא מַמְשִׁיךְ תְּפִלָּתוֹ. יסוד : שוע״ר ק״ג סעיף ג.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק״ג (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַמַּמְתִּין, הַכְּבוֹד הַבְּרִיּוֹת, הַהַעֲמָדָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
