דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ס״ה · דִּין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע

דרכו של אדמו״ר הזקן בדִּין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע : קְרָאָהּ סֵרוּגִין — יָצָא, וְהַהֶפְסֵק מֵחֲמַת אֹנֶס — פְּלוּגְתָּא ; הַנִּכְנָס וּמָצָא צִבּוּר קוֹרְאִין — קוֹרֵא פָּסוּק רִאשׁוֹן עִם הַצִּבּוּר ; וְטוֹב לִקְרֹא עִמָּהֶם כָּל ק״ש — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ס״ה — דִּין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע. אצל הרב הסימן נפרש לג׳ סְעִיפִים : הַקְּרָאָהּ סֵרוּגִין עִם הַהֶפְסֵק מֵחֲמַת אֹנֶס, הַנִּכְנָס וּמָצָא צִבּוּר קוֹרְאִין וְהַפָּסוּק רִאשׁוֹן עִם הַצִּבּוּר, וְלִבְסוֹף הַטּוֹב לִקְרֹא עִמָּהֶם כָּל קְרִיאַת שְׁמַע.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ס״ה — ג׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ס״ה). אדמו״ר הזקן דן בו בְּדִין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ס״ה — דִּין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ס״ה — דִּין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.65), ואחריו תרגומנו. הסימן כולל ג׳ סעיפים (סעיף k → ‎.65.k).

סעיף א קְרָאָהּ סֵרוּגִין — יָצָא ; הֶפְסֵק בִּרְצוֹנוֹ — לַמָּקוֹם שֶׁפָּסַק ; מֵחֲמַת אֹנֶס — פְּלוּגְתָּא.

הַנִּכְנָס לְבֵית הַכְּנֶסֶת וּמָצָא צִבּוּר קוֹרְאִין קְרִיאַת שְׁמַע אוֹ שֶׁהִפְסִיק בִּשְׁעַת הַקְּרִיאָה וּבוֹ ג׳ סְעִיפִים:קְרָאָהּ סֵרוּגִין, דְּהַיְנוּ שֶׁהִתְחִיל לִקְרוֹת וְהִפְסִיק בֵּין בִּשְׁתִיקָה בֵּין בְּדִבּוּר וְחָזַר וּגְמָרָהּ, אֲפִלּוּ שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ — יָצָא, אֲפִלּוּ אִם הָיָה הַהֶפְסֵק מֵחֲמַת אֹנֶס. וְכֵן בִּמְגִלָּה וּבִתְקִיעַת שׁוֹפָר. אֲבָל בִּתְפִלָּה אִם שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ מֵרֹאשׁ עַד סוֹף — צָרִיךְ לַחֲזֹר לָרֹאשׁ. וְיֵשׁ אוֹמְרִים שֶׁאֵין חִלּוּק בֵּין קְרִיאַת שְׁמַע לִתְפִלָּה, וּבְכָל מָקוֹם אִם שָׁהָה מֵחֲמַת אֹנֶס, שֶׁאַף אִם הָיָה רוֹצֶה שֶׁלֹּא לִשְׁהוֹת לֹא הָיָה יָכוֹל — צָרִיךְ לַחֲזֹר לָרֹאשׁ, אִם שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ מֵרֹאשׁ עַד סוֹף וְלֹא מִמָּקוֹם שֶׁשּׁוֹהֶה וּמַפְסִיק בּוֹ עַד סוֹף, וּמְשַׁעֲרִין לְפִי הַקּוֹרֵא וְלֹא לְפִי רֹב בְּנֵי אָדָם. וַאֲפִלּוּ בְּבִרְכַּת קְרִיאַת שְׁמַע אִם שָׁהָה בְּאֶמְצַע הַבְּרָכָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ מֵרֹאשָׁהּ לְסוֹפָהּ — צָרִיךְ לַחֲזֹר לָרֹאשׁ. אֲבָל אִם הִפְסִיק שֶׁלֹּא מֵחֲמַת אֹנֶס, שֶׁאִם הָיָה רוֹצֶה הָיָה קוֹרֵא וְשָׁהָה בִּרְצוֹנוֹ וְהִפְסִיק בֵּין בְּדִבּוּר בֵּין בִּשְׁתִיקָה אֲפִלּוּ זְמַן רַב — אֵין צָרִיךְ לַחֲזֹר אֶלָּא לַמָּקוֹם שֶׁפָּסַק. וְכֵן יֵשׁ לִנְהֹג הֲלָכָה לְמַעֲשֶׂה, כִּי כֵּן עִקָּר. וְאֹנֶס זֶה, יֵשׁ אוֹמְרִים אֲפִלּוּ אֹנֶס לִסְטִים אוֹ אֹנֶס אַחֵר, כָּל שֶׁלֹּא הָיָה יָכוֹל לִקְרוֹת אֲפִלּוּ אִם רָצָה. וְיֵשׁ אוֹמְרִים שֶׁאֵינוֹ חָשׁוּב הֶפְסֵק מֵחֲמַת אֹנֶס לַחֲזֹר לָרֹאשׁ אֶלָּא אִם כֵּן לֹא הָיָה יָכוֹל לִקְרוֹת מֵחֲמַת שֶׁהָאִישׁ אֵינוֹ רָאוּי, כְּגוֹן שֶׁהִתְחִיל לִקְרוֹת קְרִיאַת שְׁמַע וְהִתְחִילוּ מֵי רַגְלָיו שׁוֹתְתִין עַל בִּרְכָּיו וְהֻצְרַךְ לְהַפְסִיק עַד שֶׁיִּכְלוּ, אוֹ מֵחֲמַת שֶׁהַמָּקוֹם אֵינוֹ רָאוּי לִקְרִיאָה, כְּגוֹן שֶׁקּוֹרְאָהּ מְהַלֵּךְ וְהִגִּיעַ לִמְבוֹאוֹת הַמְטֻנָּפוֹת וְהֻצְרַךְ לְהַפְסִיק. אֲבָל אִם הָאִישׁ אוֹ הַמָּקוֹם רְאוּיִין אֶלָּא שֶׁאֹנֶס אַחֵר מְעַכְּבוֹ וּמַצְרִיכוֹ לְהַפְסִיק, אֲפִלּוּ שָׁהָה זְמַן רַב — חוֹזֵר לַמָּקוֹם שֶׁפָּסַק. וּלְעִנְיַן הֲלָכָה — יֵשׁ מִי שֶׁאוֹמֵר שֶׁבַּתְּפִלָּה אֲפִלּוּ בְּאֹנֶס אַחֵר חוֹזֵר לָרֹאשׁ, מֵאַחַר שֶׁיֵּשׁ אוֹמְרִים שֶׁאֲפִלּוּ בְּלִי אֹנֶס כְּלָל חוֹזֵר לָרֹאשׁ, אֲבָל בִּקְרִיאַת שְׁמַע אֵין צָרִיךְ לַחֲזֹר לָרֹאשׁ אֶלָּא אִם כֵּן הָאִישׁ אוֹ הַמָּקוֹם אֵינוֹ רָאוּי:

קְרָאָהּ סֵרוּגִין. הִתְחִיל לִקְרוֹת וְהִפְסִיק (בִּשְׁתִיקָה אוֹ בְּדִבּוּר) וְחָזַר וּגְמָרָהּ — אֲפִלּוּ שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ — יָצָא, אֲפִלּוּ מֵחֲמַת אֹנֶס (וְכֵן בִּמְגִלָּה וּבְשׁוֹפָר). אֲבָל בִּתְפִלָּה — שָׁהָה מֵרֹאשׁ עַד סוֹף — חוֹזֵר לָרֹאשׁ. לְיֵשׁ אוֹמְרִים אֵין חִלּוּק בֵּין ק״ש לִתְפִלָּה : בְּאֹנֶס חוֹזֵר לָרֹאשׁ (וּמְשַׁעֲרִין לְפִי הַקּוֹרֵא), אֲפִלּוּ בְּבִרְכוֹת ק״ש. אֲבָל הֶפְסֵק בִּרְצוֹנוֹ (שֶׁלֹּא מֵחֲמַת אֹנֶס) — אֲפִלּוּ זְמַן רַב — חוֹזֵר לַמָּקוֹם שֶׁפָּסַק (וְכֵן עִקָּר). וְעַל הָאֹנֶס : לְאֶחָד אֲפִלּוּ אֹנֶס לִסְטִים ; לְאַחֵר רַק כְּשֶׁהָאִישׁ אוֹ הַמָּקוֹם אֵינוֹ רָאוּי. וְלַהֲלָכָה : בִּתְפִלָּה אֲפִלּוּ אֹנֶס אַחֵר חוֹזֵר לָרֹאשׁ, אֲבָל בק״ש אֵין חוֹזֵר לָרֹאשׁ אֶלָּא אִם כֵּן הָאִישׁ אוֹ הַמָּקוֹם אֵינוֹ רָאוּי.

סעיף ב הַנִּכְנָס וּמָצָא צִבּוּר קוֹרְאִין — פָּסוּק רִאשׁוֹן עִמָּהֶם ; בְּמָקוֹם שֶׁאָסוּר לְהַפְסִיק — בְּנִגּוּן הַצִּבּוּר.

קָרָא קְרִיאַת שְׁמַע וְנִכְנַס לְבֵית הַכְּנֶסֶת וּמָצָא צִבּוּר שֶׁקּוֹרִין קְרִיאַת שְׁמַע — צָרִיךְ לִקְרוֹת עִמָּהֶם פָּסוּק רִאשׁוֹן, שֶׁלֹּא יֵרָאֶה כְּאִלּוּ אֵינוֹ רוֹצֶה לְקַבֵּל עֹל מַלְכוּת שָׁמַיִם עִם חֲבֵרָיו. וְהוּא הַדִּין שְׁאָר דָּבָר שֶׁהַצִּבּוּר אוֹמְרִים, כְּגוֹן ״תְּהִלָּה לְדָוִד״ — צָרִיךְ לִקְרֹא עִמָּהֶם, שֶׁכֵּן דֶּרֶךְ אֶרֶץ. וְהוּא הַדִּין אִם הוּא בְּבֵית הַכְּנֶסֶת וְאוֹמֵר דִּבְרֵי תַּחֲנוּנִים אוֹ פְּסוּקִים בְּמָקוֹם שֶׁרַשַּׁאי לִפְסֹק. אֲבָל אִם הוּא עָסוּק בְּמָקוֹם שֶׁאֵינוֹ רַשַּׁאי לִפְסֹק, כְּגוֹן מִ״בָּרוּךְ שֶׁאָמַר״ וְאֵילָךְ — לֹא יַפְסִיק, אֶלָּא יֹאמַר הַתֵּבוֹת שֶׁהוּא אוֹמֵר בְּשָׁעָה שֶׁהַצִּבּוּר אוֹמְרִים פָּסוּק רִאשׁוֹן בְּנִגּוּן הַצִּבּוּר, שֶׁיְּהֵא נִרְאֶה כְּאִלּוּ קוֹרֵא עִמָּהֶם. וְיֵשׁ אוֹמְרִים דְּמַפְסִיק אֲפִלּוּ בְּאֶמְצַע בִּרְכוֹת קְרִיאַת שְׁמַע, לְקַבֵּל עֹל מַלְכוּת שָׁמַיִם עִם הַצִּבּוּר. וְיֵשׁ אוֹמְרִים דְּבִפְסוּקֵי דְּזִמְרָה יֵשׁ לְהַפְסִיק, אֲבָל בַּבְּרָכוֹת שֶׁלִּפְנֵי קְרִיאַת שְׁמַע אֵין לְהַפְסִיק כְּלָל בִּשְׁבִיל פָּסוּק רִאשׁוֹן שֶׁל קְרִיאַת שְׁמַע, אֶלָּא יֹאמַר אוֹתָן הַתֵּבוֹת בְּנִגּוּן כְּמוֹ שֶׁהַצִּבּוּר אוֹמְרִים קְרִיאַת שְׁמַע. וּלְעִנְיַן הֲלָכָה לְמַעֲשֶׂה — כִּסְבָרָא הַג׳, דְּבִפְסוּקֵי דְּזִמְרָה — יֵשׁ לְהַפְסִיק, וּבְאֶמְצַע הַבְּרָכוֹת שֶׁל קְרִיאַת שְׁמַע — אֵין לְהַפְסִיק, אֶלָּא יֹאמַר הַתֵּבוֹת בְּנִגּוּן הַצִּבּוּר. וּמָאן דְּמַפְסִיק בִּבְרָכוֹת דִּקְרִיאַת שְׁמַע לִקְרֹא פָּסוּק רִאשׁוֹן עִם הַצִּבּוּר — רָאוּי לִגְעֹר בּוֹ:

פָּסוּק רִאשׁוֹן עִם הַצִּבּוּר. קָרָא ק״ש וְנִכְנַס וּמָצָא צִבּוּר קוֹרְאִין — צָרִיךְ לִקְרוֹת עִמָּהֶם פָּסוּק רִאשׁוֹן, שֶׁלֹּא יֵרָאֶה כְּאִלּוּ אֵינוֹ רוֹצֶה לְקַבֵּל עֹל מַלְכוּת שָׁמַיִם עִם חֲבֵרָיו. וְהוּא הַדִּין ״תְּהִלָּה לְדָוִד״ (דֶּרֶךְ אֶרֶץ) וְתַחֲנוּנִים בְּמָקוֹם שֶׁרַשַּׁאי לִפְסֹק. אֲבָל בְּמָקוֹם שֶׁאֵינוֹ רַשַּׁאי — מִ״בָּרוּךְ שֶׁאָמַר״ וְאֵילָךְ — לֹא יַפְסִיק, אֶלָּא יֹאמַר הַתֵּבוֹת בְּנִגּוּן הַצִּבּוּר. הֲלָכָה לְמַעֲשֶׂה (כִּסְבָרָא הַג׳) : בְּפְסוּקֵי דְּזִמְרָה — יֵשׁ לְהַפְסִיק ; בְּבִרְכוֹת ק״ש — אֵין לְהַפְסִיק, אֶלָּא בְּנִגּוּן הַצִּבּוּר. וּמָאן דְּמַפְסִיק בַּבְּרָכוֹת — רָאוּי לִגְעֹר בּוֹ.

סעיף ג טוֹב לִקְרֹא עִמָּהֶם כָּל ק״ש — שָׂכָר כְּקוֹרֵא בַּתּוֹרָה ; אֵינוֹ חַיָּב אֶלָּא בְּפָסוּק רִאשׁוֹן.

קָרָא קְרִיאַת שְׁמַע וְנִכְנַס לְבֵית הַכְּנֶסֶת וּמָצָא צִבּוּר שֶׁקּוֹרִין קְרִיאַת שְׁמַע — טוֹב שֶׁיִּקְרָא עִמָּהֶם כָּל קְרִיאַת שְׁמַע וִיקַבֵּל שָׂכָר כְּקוֹרֵא בַּתּוֹרָה. אֲבָל אֵינוֹ חַיָּב רַק בְּפָסוּק רִאשׁוֹן, כְּמוֹ שֶׁנִּתְבָּאֵר:

לִקְרֹא עִמָּהֶם כָּל ק״ש. קָרָא ק״ש וְנִכְנַס וּמָצָא צִבּוּר קוֹרְאִין — טוֹב שֶׁיִּקְרָא עִמָּהֶם כָּל קְרִיאַת שְׁמַע וִיקַבֵּל שָׂכָר כְּקוֹרֵא בַּתּוֹרָה. אֲבָל אֵינוֹ חַיָּב אֶלָּא בְּפָסוּק רִאשׁוֹן, כְּמוֹ שֶׁנִּתְבָּאֵר (סעיף ב).

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ס״ה (מהדורת קה״ת, ג׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע

היסוד : דִּין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים. תחילה הַקְּרָאָהּ סֵרוּגִין וְהַהֶפְסֵק מֵחֲמַת אֹנֶס. אחר כך הַנִּכְנָס וּמָצָא צִבּוּר קוֹרְאִין וְהַפָּסוּק רִאשׁוֹן עִם הַצִּבּוּר. ולבסוף הַטּוֹב לִקְרֹא עִמָּהֶם כָּל ק״ש וְגֶדֶר הַחִיּוּב.

קְרָאָהּ סֵרוּגִין — יָצָא

קְרָאָהּ סֵרוּגִין — יָצָא

היסוד. קְרָאָהּ סֵרוּגִין — הִפְסִיק (בִּשְׁתִיקָה אוֹ בְּדִבּוּר) וְחָזַר וּגְמָרָהּ — יָצָא, אֲפִלּוּ שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ, וַאֲפִלּוּ מֵחֲמַת אֹנֶס (וְכֵן בִּמְגִלָּה וּבְשׁוֹפָר) (סעיף א).

המסקנה. הַסֵּרוּגִין אֵינוֹ שׁוֹבֵר אֶת הַקְּרִיאָה — הַקְּרִיאָה שְׁרִירָה. עיין סימן ס״ו · 66 (הפסק בק״ש ובברכותיה).

הֶפְסֵק מֵחֲמַת אֹנֶס — פְּלוּגְתָּא

הֶפְסֵק מֵחֲמַת אֹנֶס — פְּלוּגְתָּא

היסוד. תְּפִלָּה : שָׁהָה מֵרֹאשׁ עַד סוֹף — חוֹזֵר לָרֹאשׁ ; ק״ש : הֶפְסֵק בִּרְצוֹנוֹ אֲפִלּוּ זְמַן רַב — לַמָּקוֹם שֶׁפָּסַק ; מֵחֲמַת אֹנֶס — לָרֹאשׁ רַק אִם הָאִישׁ אוֹ הַמָּקוֹם אֵינוֹ רָאוּי (סעיף א).

המסקנה. מְשַׁעֲרִין לְפִי הַקּוֹרֵא (לֹא לְפִי רֹב בְּנֵי אָדָם). עיין סימן ס״ד · 64 (טעה בק״ש).

פָּסוּק רִאשׁוֹן עִם הַצִּבּוּר

פָּסוּק רִאשׁוֹן עִם הַצִּבּוּר

היסוד. הַנִּכְנָס שֶׁכְּבָר קָרָא וּמָצָא צִבּוּר קוֹרְאִין — צָרִיךְ לִקְרוֹת עִמָּהֶם פָּסוּק רִאשׁוֹן (עֹל מַלְכוּת שָׁמַיִם עִם חֲבֵרָיו) ; בְּפְסוּקֵי דְּזִמְרָה מַפְסִיק, בְּבִרְכוֹת ק״ש אֵין מַפְסִיק (סעיף ב).

המסקנה. בַּבְּרָכוֹת אוֹמֵר הַתֵּבוֹת בְּנִגּוּן הַצִּבּוּר ; הַמַּפְסִיק — רָאוּי לִגְעֹר בּוֹ. עיין סימן נ״א · 51 (ברוך שאמר ופסוקי דזמרה).

הפניה ללימוד. היקף הסימן — הַסֵּרוּגִין, הַהֶפְסֵק מֵחֲמַת אֹנֶס וְהַפָּסוּק רִאשׁוֹן עִם הַצִּבּוּר — נוגע להנהגת ק״ש היומיומית בבית הכנסת. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ס״ה. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מג׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
קְרָאָהּ סֵרוּגִין או״ח ס״ה — קְרָאָהּ סֵרוּגִין אֲפִלּוּ שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ — יָצָא, אֲפִלּוּ מֵחֲמַת אֹנֶס. מ״ב ס״ה — גֶּדֶר הַסֵּרוּגִין וְשִׁעוּר כְּדֵי לִגְמֹר. ערוה״ש ס״ה — דִּין הַשְּׁהִיָּה בַּקְּרִיאָה. דרך אדמו״ר הזקן — סֵרוּגִין יָצָא, אֲפִלּוּ מֵחֲמַת אֹנֶס ; וְכֵן בִּמְגִלָּה וּבְשׁוֹפָר (סעיף א).
הֶפְסֵק מֵחֲמַת אֹנֶס וַהֲלָכָה או״ח ס״ה — רָמָ״א : אָנוּס וְהִפְסִיק כְּדֵי לִגְמֹר אֶת כֻּלָּהּ חוֹזֵר לָרֹאשׁ ; מְשַׁעֲרִין לְפִי הַקּוֹרֵא. מ״ב ס״ה — פְּרָטֵי הָאֹנֶס וּפְלוּגְתָּא בֵּין ק״ש לִתְפִלָּה. ערוה״ש ס״ה — הַהֶכְרֵעַ בְּאֹנֶס אַחֵר. דרך אדמו״ר הזקן — בִּרְצוֹנוֹ חוֹזֵר לַמָּקוֹם שֶׁפָּסַק (וְכֵן עִקָּר) ; בק״ש חוֹזֵר לָרֹאשׁ רַק אִם הָאִישׁ אוֹ הַמָּקוֹם אֵינוֹ רָאוּי (סעיף א).
פָּסוּק רִאשׁוֹן וְלִקְרֹא עִם הַצִּבּוּר או״ח ס״ה — צָרִיךְ לִקְרוֹת עִמָּהֶם פָּסוּק רִאשׁוֹן ; טוֹב לִקְרֹא כָּל ק״ש ; אֵינוֹ חַיָּב אֶלָּא בְּפָסוּק רִאשׁוֹן. מ״ב ס״ה — גֶּדֶר הַהֶפְסֵק בַּבְּרָכוֹת וּבִפְסוּקֵי דְּזִמְרָה. ערוה״ש ס״ה — דִּין הָאוֹמֵר עִם הַצִּבּוּר. דרך אדמו״ר הזקן — בְּפְסוּקֵי דְּזִמְרָה מַפְסִיק, בְּבִרְכוֹת ק״ש אוֹמֵר בְּנִגּוּן הַצִּבּוּר ; הַמַּפְסִיק בַּבְּרָכוֹת רָאוּי לִגְעֹר בּוֹ (סעיפים ב-ג).

הטבלה נערכה מטקסט המחבר (אורח חיים ס״ה, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ב, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות י״ג, מגילה י״ח). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — קְרָאָהּ סֵרוּגִין

קו א — קְרָאָהּ סֵרוּגִין

יסוד הסימן : קְרָאָהּ סֵרוּגִין — יָצָא, אֲפִלּוּ שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ (סעיף א).

קריאת הרב : הַסֵּרוּגִין אֵינוֹ שׁוֹבֵר אֶת הַקְּרִיאָה ; וְכֵן בִּמְגִלָּה וּבְשׁוֹפָר. עיין סימן ס״ו · 66.

מסקנה למעשה : קְרִיאָה שֶׁהֻפְסְקָה וְחָזְרָה — שְׁרִירָה, בְּדִיעֲבַד אֵין צֹרֶךְ לַחֲזֹר.

קו ב — הֶפְסֵק מֵחֲמַת אֹנֶס

קו ב — הֶפְסֵק מֵחֲמַת אֹנֶס

יסוד הסימן : בִּרְצוֹנוֹ — לַמָּקוֹם שֶׁפָּסַק ; בק״ש מֵחֲמַת אֹנֶס — לָרֹאשׁ רַק אִם הָאִישׁ אוֹ הַמָּקוֹם אֵינוֹ רָאוּי (סעיף א).

קריאת הרב : וְכֵן עִקָּר ; מְשַׁעֲרִין לְפִי הַקּוֹרֵא. עיין סימן ס״ד · 64.

מסקנה למעשה : לְהַבְדִּיל בֵּין הֶפְסֵק בִּרְצוֹנוֹ (לַמָּקוֹם שֶׁפָּסַק) לְאֹנֶס שֶׁמְּבַטֵּל אֶת כְּשֵׁרוּת הָאִישׁ אוֹ הַמָּקוֹם.

קו ג — פָּסוּק רִאשׁוֹן עִם הַצִּבּוּר

קו ג — פָּסוּק רִאשׁוֹן עִם הַצִּבּוּר

יסוד הסימן : צָרִיךְ לִקְרוֹת עִמָּהֶם פָּסוּק רִאשׁוֹן ; טוֹב לִקְרֹא כָּל ק״ש (שָׂכָר כְּקוֹרֵא בַּתּוֹרָה) ; אֵינוֹ חַיָּב אֶלָּא בְּפָסוּק רִאשׁוֹן (סעיפים ב-ג).

קריאת הרב : בְּפְסוּקֵי דְּזִמְרָה מַפְסִיק ; בְּבִרְכוֹת ק״ש אוֹמֵר בְּנִגּוּן הַצִּבּוּר. עיין סימן נ״א · 51 (ברוך שאמר).

מסקנה למעשה : לְהִצְטָרֵף לַצִּבּוּר לְפָסוּק רִאשׁוֹן ; לֹא לְהַפְסִיק בַּבְּרָכוֹת — הַמַּפְסִיק רָאוּי לִגְעֹר בּוֹ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ס״ה ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ס״ה (דִּין הַמַּפְסִיק בִּקְרִיאַת שְׁמַע)

  • ① סֵרוּגִין. הִפְסִיק (בִּשְׁתִיקָה אוֹ בְּדִבּוּר) וְחָזַר — יָצָא, אֲפִלּוּ שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ (וְכֵן בִּמְגִלָּה וּבְשׁוֹפָר). יסוד : שוע״ר ס״ה סעיף א.
  • ② הֶפְסֵק בִּרְצוֹנוֹ. הֶפְסֵק בִּרְצוֹנוֹ, אֲפִלּוּ זְמַן רַב — אֵין צָרִיךְ לַחֲזֹר אֶלָּא לַמָּקוֹם שֶׁפָּסַק (וְכֵן עִקָּר). יסוד : שוע״ר ס״ה סעיף א.
  • ③ הֶפְסֵק מֵחֲמַת אֹנֶס. בק״ש חוֹזֵר לָרֹאשׁ רַק אִם הָאִישׁ אוֹ הַמָּקוֹם אֵינוֹ רָאוּי ; מְשַׁעֲרִין לְפִי הַקּוֹרֵא. יסוד : שוע״ר ס״ה סעיף א.
  • ④ פָּסוּק רִאשׁוֹן עִם הַצִּבּוּר. כְּבָר קָרָא וְנִכְנַס : לִקְרוֹת עִמָּהֶם פָּסוּק רִאשׁוֹן ; בְּפְסוּקֵי דְּזִמְרָה מַפְסִיק, בְּבִרְכוֹת ק״ש אוֹמֵר בְּנִגּוּן הַצִּבּוּר. יסוד : שוע״ר ס״ה סעיף ב.
  • ⑤ לִקְרֹא כָּל ק״ש. טוֹב שֶׁיִּקְרָא עִמָּהֶם כָּל ק״ש (שָׂכָר כְּקוֹרֵא בַּתּוֹרָה) ; אֵינוֹ חַיָּב אֶלָּא בְּפָסוּק רִאשׁוֹן. יסוד : שוע״ר ס״ה סעיף ג.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ס״ה (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַסֵּרוּגִין, הַהֶפְסֵק מֵחֲמַת אֹנֶס, הַפָּסוּק רִאשׁוֹן עִם הַצִּבּוּר — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
