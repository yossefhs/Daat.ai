דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן צ״א · שֶׁיֶּאֱזֹר מָתְנָיו וִיכַסֶּה רֹאשׁוֹ בִּשְׁעַת תְּפִלָּה

דרכו של אדמו״ר הזקן בשֶׁיֶּאֱזֹר מָתְנָיו וִיכַסֶּה רֹאשׁוֹ בִּשְׁעַת תְּפִלָּה : אֵזוֹר וְכִסּוּי הַלֵּב — חֲצִיצָה בֵּין הַלֵּב לָעֶרְוָה, מִשּׁוּם ״הִכּוֹן״ ; כִּסּוּי הָרֹאשׁ — אַזְכָּרָה וּמִדַּת חֲסִידוּת ; כְּבוֹד הָרַגְלַיִם וְהַלְּבוּשׁ ; וַעֲטִיפָה וְהִתְקַשְּׁטוּת — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן צ״א — שֶׁיֶּאֱזֹר מָתְנָיו וִיכַסֶּה רֹאשׁוֹ בִּשְׁעַת תְּפִלָּה. אצל הרב הסימן נפרש לו׳ סְעִיפִים : הַחֲגִירָה וְהַחֲצִיצָה בֵּין הַלֵּב לָעֶרְוָה, הָאֵזוֹר מִשּׁוּם ״הִכּוֹן״, כִּסּוּי הָרֹאשׁ וְהָאַזְכָּרָה, כּוֹבַע קַשׁ וְהַנָּחַת יָד, כְּבוֹד הָרַגְלַיִם וְהָאֲפוּנְדָּא, וּלְבַסּוֹף הָעֲטִיפָה עִם הַהִתְקַשְּׁטוּת וַחֲבִיקַת הַיָּדַיִם בְּעֵת הַזַּעַם.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן צ״א — ו׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim צ״א). אדמו״ר הזקן דן בו בְּשֶׁיֶּאֱזֹר מָתְנָיו וִיכַסֶּה רֹאשׁוֹ בִּשְׁעַת תְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן צ״א — שֶׁיֶּאֱזֹר מָתְנָיו וִיכַסֶּה רֹאשׁוֹ בִּשְׁעַת תְּפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן צ״א — שֶׁיֶּאֱזֹר מָתְנָיו וִיכַסֶּה רֹאשׁוֹ בִּשְׁעַת תְּפִלָּה — וּבוֹ ו׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.91), ואחריו תרגומנו. הסימן כולל ו׳ סעיפים (סעיף k → ‎.91.k).

סעיף א טַלִּית חֲגוּרָה עַל מָתְנָיו — אָסוּר עַד שֶׁיְּכַסֶּה לִבּוֹ ; נֶאֱנַס וְהִתְפַּלֵּל — יָצָא.

שֶׁיֶּאֱזֹר מָתְנָיו וִיכַסֶּה רֹאשׁוֹ בִּשְׁעַת תְּפִלָּה וּבוֹ ו׳ סְעִיפִים:בְּגָדָיו כֵּיצַד? אֵין צָרִיךְ לוֹמַר שֶׁתְּהֵא עֶרְוָתוֹ מְכֻסָּה וְלֹא יְהֵא לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה, אֶלָּא אֲפִלּוּ אִם הָיְתָה טַלִּית חֲגוּרָה לוֹ עַל מָתְנָיו לְכַסּוֹתוֹ מִמָּתְנָיו וּלְמַטָּה, וְחוֹצֵץ בְּטַלִּיתוֹ עַל מָתְנָיו שֶׁלֹּא יְהֵא לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה — אָסוּר לְהִתְפַּלֵּל עַד שֶׁיְּכַסֶּה לִבּוֹ, מִטַּעַם שֶׁנִּתְבָּאֵר סִימָן ע״ד. וְאִם לֹא כִּסָּה אֶת לִבּוֹ, אוֹ שֶׁנֶּאֱנַס וְאֵין לוֹ בְּמַה שֶּׁיִּתְכַּסֶּה, הוֹאִיל וְכִסָּה עֶרְוָתוֹ וְהִתְפַּלֵּל — יָצָא:

כִּסּוּי הַלֵּב. אֵין צָרִיךְ לוֹמַר שֶׁתְּהֵא עֶרְוָתוֹ מְכֻסָּה וְלֹא יְהֵא לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה, אֶלָּא אֲפִלּוּ טַלִּית חֲגוּרָה עַל מָתְנָיו וְחוֹצֶצֶת — אָסוּר לְהִתְפַּלֵּל עַד שֶׁיְּכַסֶּה לִבּוֹ, מִטַּעַם סימן ע״ד. וְאִם לֹא כִּסָּה, אוֹ נֶאֱנַס וְאֵין לוֹ בַּמֶּה — הוֹאִיל וְכִסָּה עֶרְוָתוֹ וְהִתְפַּלֵּל — יָצָא.

סעיף ב צָרִיךְ לֶאֱזֹר אֵזוֹר בַּתְּפִלָּה מִשּׁוּם ״הִכּוֹן״ ; בִּשְׁאָר בְּרָכוֹת אֵין צָרִיךְ.

צָרִיךְ לֶאֱזֹר אֵזוֹר בִּשְׁעַת הַתְּפִלָּה אֲפִלּוּ אִם יֵשׁ לוֹ אַבְנֵט שֶׁאֵין לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה, מִשּׁוּם שֶׁנֶּאֱמַר: ״הִכּוֹן לִקְרַאת אֱלֹהֶיךָ יִשְׂרָאֵל״. אֲבָל בִּשְׁאָר בְּרָכוֹת מֻתָּר לְבָרֵךְ בְּלֹא חֲגוֹרָה, מֵאַחַר שֶׁיֵּשׁ לוֹ אַבְנֵט בַּמִּכְנָסַיִם וְאֵין לִבּוֹ רוֹאֶה אֶת הָעֶרְוָה. וְהוּא הַדִּין אֲפִלּוּ בְּלֹא מִכְנָסַיִם אִם בְּגָדָיו מֻנָּחִים מַמָּשׁ עַל בִּטְנוֹ וְחוֹצְצִים בֵּין לִבּוֹ לָעֶרְוָה. וּמִי שֶׁהוֹלֵךְ כָּל הַיּוֹם בְּלֹא חֲגוֹרָה — אֵין צָרִיךְ לֶאֱזֹר בִּשְׁעַת תְּפִלָּה:

אֵזוֹר מִשּׁוּם ״הִכּוֹן״. צָרִיךְ לֶאֱזֹר אֵזוֹר בַּתְּפִלָּה אֲפִלּוּ יֵשׁ לוֹ אַבְנֵט, מִשּׁוּם ״הִכּוֹן לִקְרַאת אֱלֹהֶיךָ יִשְׂרָאֵל״. אֲבָל בִּשְׁאָר בְּרָכוֹת מֻתָּר בְּלֹא חֲגוֹרָה, כְּשֶׁיֵּשׁ אַבְנֵט בַּמִּכְנָסַיִם ; וְהוּא הַדִּין אִם בְּגָדָיו עַל בִּטְנוֹ וְחוֹצְצִים. וּמִי שֶׁהוֹלֵךְ כָּל הַיּוֹם בְּלֹא חֲגוֹרָה — אֵין צָרִיךְ לֶאֱזֹר בַּתְּפִלָּה.

סעיף ג אָסוּר לְהוֹצִיא אַזְכָּרָה בְּרֹאשׁ מְגֻלֶּה ; מִדַּת חֲסִידוּת — ד׳ אַמּוֹת.

אָסוּר לְהוֹצִיא אַזְכָּרָה מִפִּיו בְּרֹאשׁ מְגֻלֶּה. וְיֵשׁ אוֹמְרִים שֶׁיֵּשׁ לִמְחוֹת שֶׁלֹּא לִכָּנֵס לְבֵית הַכְּנֶסֶת בְּגִלּוּי הָרֹאשׁ. וּמִדַּת חֲסִידוּת הוּא שֶׁלֹּא לֵילֵךְ ד׳ אַמּוֹת בְּגִלּוּי הָרֹאשׁ אֲפִלּוּ שֶׁלֹּא בְּבֵית הַכְּנֶסֶת, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ב׳:

כִּסּוּי הָרֹאשׁ. אָסוּר לְהוֹצִיא אַזְכָּרָה מִפִּיו בְּרֹאשׁ מְגֻלֶּה. וְיֵשׁ אוֹמְרִים שֶׁיֵּשׁ לִמְחוֹת שֶׁלֹּא לִכָּנֵס לְבֵית הַכְּנֶסֶת בְּגִלּוּי הָרֹאשׁ. וּמִדַּת חֲסִידוּת שֶׁלֹּא לֵילֵךְ ד׳ אַמּוֹת בְּגִלּוּי הָרֹאשׁ אֲפִלּוּ שֶׁלֹּא בְּבֵית הַכְּנֶסֶת, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ב׳.

סעיף ד כּוֹבַע קַשׁ — כִּסּוּי ; יָדוֹ עַל רֹאשׁוֹ — אֵינָהּ כִּסּוּי ; יַד אַחֵר — יֵשׁ לְהָקֵל.

כּוֹבָעִים הַקְּלוּעִים מִקַּשׁ חֲשׁוּבִים כִּסּוּי, אֲבָל הַנָּחַת יָדוֹ עַל רֹאשׁוֹ אֵינָהּ חֲשׁוּבָה כִּסּוּי, לְפִי שֶׁהַיָּד וְהָרֹאשׁ גּוּף אֶחָד הֵן וְאֵין הַגּוּף יָכוֹל לְכַסּוֹת אֶת עַצְמוֹ. אֲבָל אִם אַחֵר מַנִּיחַ יָדוֹ עַל רֹאשׁוֹ — יֵשׁ לְהָקֵל (כֵּיוָן שֶׁרש״ל מֵקֵל לְגַמְרֵי):

כּוֹבַע קַשׁ וְהַנָּחַת יָד. כּוֹבָעִים הַקְּלוּעִים מִקַּשׁ חֲשׁוּבִים כִּסּוּי, אֲבָל הַנָּחַת יָדוֹ עַל רֹאשׁוֹ אֵינָהּ כִּסּוּי, לְפִי שֶׁהַיָּד וְהָרֹאשׁ גּוּף אֶחָד הֵן. אֲבָל אִם אַחֵר מַנִּיחַ יָדוֹ עַל רֹאשׁוֹ — יֵשׁ לְהָקֵל (שֶׁרש״ל מֵקֵל לְגַמְרֵי).

סעיף ה לֹא בַּאֲפוּנְדָּא, לֹא בְּרַגְלַיִם מְגֻלִּים כְּמִנְהַג הַמָּקוֹם ; וְלֹא בָּתֵּי יָדַיִם.

לֹא יַעֲמֹד אָדָם בִּתְפִלָּה בַּאֲפוּנְדָּתוֹ, וְלֹא בְּרַגְלַיִם מְגֻלִּים אִם דֶּרֶךְ אַנְשֵׁי הַמָּקוֹם שֶׁלֹּא יַעַמְדוּ לִפְנֵי הַגְּדוֹלִים אֶלָּא בְּבָתֵּי רַגְלַיִם. אֲבָל אִם דַּרְכָּם לַעֲמֹד יְחֵפִים בִּפְנֵי הַגְּדוֹלִים כְּמוֹ בְּאֶרֶץ הָעֲרָב — מֻתָּר לְהִתְפַּלֵּל יָחֵף. וּבִמְדִינוֹת אֵלּוּ אֵין לְהִתְפַּלֵּל בְּבָתֵּי שׁוֹקַיִם שֶׁל פִּשְׁתָּן אִם לֹא יִלְבַּשׁ עֲלֵיהֶם אַנְפִּילָאוֹת (שֶׁקּוֹרִין זאקי״ן), שֶׁגְּנַאי הוּא לַעֲמֹד בִּפְנֵי הַגְּדוֹלִים בְּבָתֵּי שׁוֹקַיִם שֶׁל פִּשְׁתָּן בִּלְבַד, וּמִכָּל שֶׁכֵּן כְּשֶׁנּוֹעֲלִין סַנְדָּל (שֶׁקּוֹרִין פאנטאפי״ל) שֶׁהֶעָקֵב מְגֻלֶּה (אֲבָל בְּאַנְפִּילָאוֹת מֻתָּר לְהִתְפַּלֵּל אַף בְּלֹא בָּתֵּי שׁוֹקַיִם, מֵאַחַר שֶׁיֵּשׁ לוֹ אֵזוֹר לְהַפְסִיק בֵּין לִבּוֹ לָעֶרְוָה, אוֹ שֶׁבְּגָדָיו מֻנָּחִים מַמָּשׁ עַל בִּטְנוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסָעִיף ב׳). וְכֵן אֵין לִלְבֹּשׁ בִּשְׁעַת הַתְּפִלָּה בָּתֵּי יָדַיִם כְּדֶרֶךְ עוֹבְרֵי דְּרָכִים:

כְּבוֹד הָרַגְלַיִם וְהַלְּבוּשׁ. לֹא יַעֲמֹד בִּתְפִלָּה בַּאֲפוּנְדָּתוֹ, וְלֹא בְּרַגְלַיִם מְגֻלִּים אִם דֶּרֶךְ אַנְשֵׁי הַמָּקוֹם שֶׁלֹּא לַעֲמֹד לִפְנֵי הַגְּדוֹלִים אֶלָּא בְּבָתֵּי רַגְלַיִם. אֲבָל אִם דַּרְכָּם יְחֵפִים כְּמוֹ בְּאֶרֶץ הָעֲרָב — מֻתָּר יָחֵף. וּבִמְדִינוֹת אֵלּוּ אֵין לְהִתְפַּלֵּל בְּבָתֵּי שׁוֹקַיִם שֶׁל פִּשְׁתָּן בְּלֹא אַנְפִּילָאוֹת, וְכָל שֶׁכֵּן בְּסַנְדָּל שֶׁהֶעָקֵב מְגֻלֶּה. וְכֵן אֵין לִלְבֹּשׁ בָּתֵּי יָדַיִם כְּדֶרֶךְ עוֹבְרֵי דְּרָכִים.

סעיף ו דֶּרֶךְ חֲכָמִים — עֲטוּפִים ; בְּעֵת שָׁלוֹם מִתְקַשְּׁטִים, בְּעֵת זַעַם כְּעֶבֶד.

דֶּרֶךְ הַחֲכָמִים וְתַלְמִידֵיהֶם שֶׁלֹּא יִתְפַּלְּלוּ אֶלָּא כְּשֶׁהֵן עֲטוּפִים. וּמִנְהַג הַחֲכָמִים הָרִאשׁוֹנִים כְּשֶׁהָיָה שָׁלוֹם בָּעוֹלָם הָיוּ מִתְקַשְּׁטִים בִּבְגָדִים נָאִים וּמִתְפַּלְלִים, מִשּׁוּם ״הִכּוֹן כו׳״, כְּלוֹמַר הִתְנָאֶה לְפָנָיו. וּבְעֵת הַזַּעַם הָיוּ עוֹמְדִים כְּעֶבֶד לִפְנֵי רַבּוֹ שֶׁמַּשְׁלִיךְ אַדַּרְתּוֹ מֵעָלָיו שֶׁלֹּא יֵרָאֶה כְּחָשׁוּב, וְחוֹבֵק אֶת יָדָיו בְּאֶצְבְּעוֹתָיו מֵאֵימַת רַבּוֹ וְיִרְאָתוֹ, דְּהַיְנוּ שֶׁחוֹבֵק אֶצְבְּעוֹתָיו שֶׁל יָדָיו זֶה בְּשֶׁל זֶה כְּאָדָם שֶׁשּׁוֹבֵר אֶצְבְּעוֹתָיו כְּשֶׁמִּצְטַעֵר. וְיֵשׁ מֵהַחֲכָמִים שֶׁהָיוּ עוֹשִׂים כֵּן אַף בְּעֵת שָׁלוֹם. וּמִכָּל מָקוֹם, טוֹב לִזָּהֵר שֶׁלֹּא לְחַבֵּק אֶצְבְּעוֹתָיו בְּעֵת שָׁלוֹם, כִּי בָּזֶה מוֹרִיד דִּין עַל עַצְמוֹ, אֶלָּא יַנִּיחַ יָדָיו זֶה עַל זֶה כְּפוּתִין, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן צ״ה:

עֲטִיפָה וְהִתְקַשְּׁטוּת. דֶּרֶךְ הַחֲכָמִים וְתַלְמִידֵיהֶם שֶׁלֹּא יִתְפַּלְּלוּ אֶלָּא עֲטוּפִים. בְּעֵת שָׁלוֹם הָיוּ מִתְקַשְּׁטִים בִּבְגָדִים נָאִים, מִשּׁוּם ״הִכּוֹן כו׳״ — הִתְנָאֶה לְפָנָיו. וּבְעֵת הַזַּעַם — כְּעֶבֶד לִפְנֵי רַבּוֹ, וְחוֹבֵק אֶצְבְּעוֹתָיו מֵאֵימָתוֹ. וּמִכָּל מָקוֹם טוֹב שֶׁלֹּא לְחַבֵּק אֶצְבְּעוֹתָיו בְּעֵת שָׁלוֹם, כִּי מוֹרִיד דִּין עַל עַצְמוֹ, אֶלָּא יַנִּיחַ יָדָיו כְּפוּתִין, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן צ״ה.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim צ״א (מהדורת קה״ת, ו׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — חֲגִירָה, כִּסּוּי הָרֹאשׁ וַעֲטִיפָה

היסוד : הַתְּלַבּוּשׁ בִּשְׁעַת הַתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הָאֵזוֹר וְכִסּוּי הַלֵּב — הַחֲצִיצָה בֵּין הַלֵּב לָעֶרְוָה, מִשּׁוּם ״הִכּוֹן״. אחר כך כִּסּוּי הָרֹאשׁ — הָאַזְכָּרָה וּמִדַּת חֲסִידוּת. אחר כך כְּבוֹד הָרַגְלַיִם וְהַלְּבוּשׁ — הָאֲפוּנְדָּא וּמִנְהַג הַמָּקוֹם. ולבסוף הָעֲטִיפָה וְהַהִתְקַשְּׁטוּת — עִם חֲבִיקַת הַיָּדַיִם בְּעֵת הַזַּעַם.

אֵזוֹר וְכִסּוּי הַלֵּב — מִשּׁוּם ״הִכּוֹן״

אֵזוֹר וְכִסּוּי הַלֵּב — מִשּׁוּם ״הִכּוֹן״

היסוד. אֵין לְהִתְפַּלֵּל אֶלָּא בְּכִסּוּי הַלֵּב וַחֲצִיצָה בֵּין הַלֵּב לָעֶרְוָה ; וְאוֹזֵר אֵזוֹר מִשּׁוּם ״הִכּוֹן לִקְרַאת אֱלֹהֶיךָ יִשְׂרָאֵל״ (סעיפים א-ב).

המסקנה. הַחֲגִירָה בַּתְּפִלָּה יְתֵרָה עַל צְנִיעוּת בְּעָלְמָא — חוֹבַת הַכָּנָה לִפְנֵי הַמֶּלֶךְ. עיין סימן 74 (חֲצִיצַת הַלֵּב).

כִּסּוּי הָרֹאשׁ — אַזְכָּרָה וּמִדַּת חֲסִידוּת

כִּסּוּי הָרֹאשׁ — אַזְכָּרָה וּמִדַּת חֲסִידוּת

היסוד. אָסוּר לְהוֹצִיא אַזְכָּרָה בְּרֹאשׁ מְגֻלֶּה ; מִדַּת חֲסִידוּת שֶׁלֹּא לֵילֵךְ ד׳ אַמּוֹת בְּגִלּוּי הָרֹאשׁ (סעיף ג).

המסקנה. כּוֹבַע קַשׁ חֲשׁוּב כִּסּוּי ; יָדוֹ אֵינָהּ כִּסּוּי, אֲבָל יַד אַחֵר — יֵשׁ לְהָקֵל. עיין סימן 2 (גִּלּוּי הָרֹאשׁ).

כְּבוֹד הָרַגְלַיִם וְהַלְּבוּשׁ

כְּבוֹד הָרַגְלַיִם וְהַלְּבוּשׁ

היסוד. לֹא בַּאֲפוּנְדָּא, וְלֹא בְּרַגְלַיִם מְגֻלִּים כְּמִנְהַג הַמָּקוֹם לִפְנֵי הַגְּדוֹלִים ; וְלֹא בָּתֵּי יָדַיִם (סעיף ה).

המסקנה. הַקְּנֶה־מִדָּה הוּא הַכָּבוֹד הַיַּחֲסִי — מַה שֶּׁגְּנַאי לִפְנֵי אָדָם חָשׁוּב גְּנַאי לִפְנֵי הַמֶּלֶךְ. עיין סימן 90 (מְקוֹם הַתְּפִלָּה).

עֲטִיפָה וְהִתְקַשְּׁטוּת

עֲטִיפָה וְהִתְקַשְּׁטוּת

היסוד. דֶּרֶךְ חֲכָמִים לְהִתְפַּלֵּל עֲטוּפִים ; בְּעֵת שָׁלוֹם מִתְקַשְּׁטִים בִּבְגָדִים נָאִים, בְּעֵת זַעַם כְּעֶבֶד לִפְנֵי רַבּוֹ (סעיף ו).

המסקנה. הַרַב מַזְהִיר : שֶׁלֹּא לַחֲבֹּק אֶצְבְּעוֹתָיו בְּעֵת שָׁלוֹם, כִּי מוֹרִיד דִּין עַל עַצְמוֹ. עיין סימן 95 (הַנָּחַת הַיָּדַיִם).

הפניה ללימוד. היקף הסימן — הַחֲגִירָה, כִּסּוּי הָרֹאשׁ, הַכָּבוֹד וְהָעֲטִיפָה — נוגע להכנה היומיומית לתפלה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

שֶׁיֶּאֱזֹר מָתְנָיו וִיכַסֶּה רֹאשׁוֹ — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן צ״א. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מו׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
אֵזוֹר וְכִסּוּי הַלֵּב או״ח צ״א — טַלִּית חֲגוּרָה עַל מָתְנָיו אָסוּר עַד שֶׁיְּכַסֶּה לִבּוֹ ; צָרִיךְ אֵזוֹר מִשּׁוּם ״הִכּוֹן״. מ״ב צ״א — גֶּדֶר חֲצִיצַת הַלֵּב וְדִין הָאֵזוֹר בַּתְּפִלָּה. ערוה״ש צ״א — חִיּוּב הַחֲגוֹרָה וּטְעַם הַהַכָּנָה. דרך אדמו״ר הזקן — נֶאֱנַס וְהִתְפַּלֵּל יָצָא ; אֵזוֹר מִשּׁוּם ״הִכּוֹן״, וּמִי שֶׁהוֹלֵךְ בְּלֹא חֲגוֹרָה — אֵין צָרִיךְ (סעיפים א-ב).
כִּסּוּי הָרֹאשׁ וְאַזְכָּרָה או״ח צ״א — אָסוּר אַזְכָּרָה בְּרֹאשׁ מְגֻלֶּה ; כּוֹבַע קַשׁ כִּסּוּי, יָדוֹ לֹא. מ״ב צ״א — פְּרָטֵי הַכִּסּוּי וּמַחֲלֹקֶת הַהֲנָחַת יָד. ערוה״ש צ״א — גֶּדֶר גִּלּוּי הָרֹאשׁ וּמִדַּת חֲסִידוּת. דרך אדמו״ר הזקן — מִדַּת חֲסִידוּת ד׳ אַמּוֹת ; יָדוֹ עַל רֹאשׁוֹ אֵינָהּ כִּסּוּי, יַד אַחֵר — יֵשׁ לְהָקֵל (סעיפים ג-ד).
רַגְלַיִם וַאֲפוּנְדָּא או״ח צ״א — לֹא בַּאֲפוּנְדָּא וְלֹא בְּרַגְלַיִם מְגֻלִּים כְּמִנְהַג הַמָּקוֹם. מ״ב צ״א — דִּין בָּתֵּי שׁוֹקַיִם וּבָתֵּי יָדַיִם בַּתְּפִלָּה. ערוה״ש צ״א — הַשְׁוָאַת כְּבוֹד הַתְּפִלָּה לִכְבוֹד הַגְּדוֹלִים. דרך אדמו״ר הזקן — בְּאֶרֶץ הָעֲרָב יָחֵף מֻתָּר ; בִּמְדִינוֹת אֵלּוּ אַנְפִּילָאוֹת, וְלֹא בָּתֵּי יָדַיִם (סעיף ה).
עֲטִיפָה וַחֲבִיקַת הַיָּדַיִם או״ח צ״א — דֶּרֶךְ חֲכָמִים עֲטוּפִים ; בְּעֵת זַעַם חוֹבֵק יָדָיו, בְּעֵת שָׁלוֹם מִתְקַשֵּׁט. מ״ב צ״א — הַנְהָגוֹת הָעֲטִיפָה וְהַהִתְקַשְּׁטוּת. ערוה״ש צ״א — גֶּדֶר הָעֲטִיפָה וּבִגְדֵי הַכָּבוֹד. דרך אדמו״ר הזקן — טוֹב שֶׁלֹּא לַחֲבֹּק אֶצְבְּעוֹתָיו בְּעֵת שָׁלוֹם, כִּי מוֹרִיד דִּין עַל עַצְמוֹ ; יַנִּיחַ יָדָיו כְּפוּתִין (סעיף ו).

הטבלה נערכה מטקסט המחבר (אורח חיים צ״א, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפלה ה, רא״ש, ר״ן, הגהות מיימוני, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד שבת ט׳, י׳, עמוס ד׳ י״ב). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ו׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַחֲגִירָה מִשּׁוּם ״הִכּוֹן״

קו א — הַחֲגִירָה מִשּׁוּם ״הִכּוֹן״

יסוד הסימן : צָרִיךְ אֵזוֹר בַּתְּפִלָּה אֲפִלּוּ יֵשׁ לוֹ אַבְנֵט, מִשּׁוּם ״הִכּוֹן לִקְרַאת אֱלֹהֶיךָ יִשְׂרָאֵל״ (סעיף ב).

קריאת הרב : הַחֲגִירָה עִנְיַן הַכָּנָה לִפְנֵי הַמֶּלֶךְ, לֹא צְנִיעוּת בִּלְבַד ; מִי שֶׁהוֹלֵךְ בְּלֹא חֲגוֹרָה — אֵין צָרִיךְ. עיין סימן 74.

מסקנה למעשה : לֶאֱזֹר אֵזוֹר לָעֲמִידָה, אַף כְּשֶׁכְּבָר לוֹבֵשׁ בֶּגֶד הַחוֹצֵץ.

קו ב — כִּסּוּי הָרֹאשׁ וְהָאַזְכָּרָה

קו ב — כִּסּוּי הָרֹאשׁ וְהָאַזְכָּרָה

יסוד הסימן : אָסוּר לְהוֹצִיא אַזְכָּרָה בְּרֹאשׁ מְגֻלֶּה ; מִדַּת חֲסִידוּת ד׳ אַמּוֹת (סעיפים ג-ד).

קריאת הרב : כּוֹבַע קַשׁ חֲשׁוּב כִּסּוּי ; יָדוֹ לֹא (גּוּף אֶחָד), אֲבָל יַד אַחֵר — יֵשׁ לְהָקֵל. עיין סימן 2.

מסקנה למעשה : לְעוֹלָם לֹא לְהוֹצִיא הַשֵּׁם בְּרֹאשׁ מְגֻלֶּה ; לְכַסּוֹת אַף מִחוּץ לְבֵית הַכְּנֶסֶת (מִדַּת חֲסִידוּת).

קו ג — כָּבוֹד, עֲטִיפָה וְהַיָּדַיִם

קו ג — כָּבוֹד, עֲטִיפָה וְהַיָּדַיִם

יסוד הסימן : לֹא בַּאֲפוּנְדָּא ; דֶּרֶךְ חֲכָמִים עֲטוּפִים ; בְּעֵת שָׁלוֹם מִתְקַשְּׁטִים (סעיפים ה-ו).

קריאת הרב : טוֹב שֶׁלֹּא לַחֲבֹּק אֶצְבְּעוֹתָיו בְּעֵת שָׁלוֹם, כִּי מוֹרִיד דִּין עַל עַצְמוֹ. עיין סימן 95 (הַנָּחַת הַיָּדַיִם).

מסקנה למעשה : לְהִתְפַּלֵּל עָטוּף וּמְלֻבָּשׁ בְּכָבוֹד ; לְהַנִּיחַ יָדָיו כְּפוּתִין, בְּלִי לַחֲבֹּק אֶצְבָּעוֹת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן צ״א ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן צ״א (שֶׁיֶּאֱזֹר מָתְנָיו וִיכַסֶּה רֹאשׁוֹ)

  • ① כִּסּוּי הַלֵּב. אֵין לְהִתְפַּלֵּל עַד שֶׁיְּכַסֶּה לִבּוֹ וְיַחֲצֹץ בֵּין הַלֵּב לָעֶרְוָה. נֶאֱנַס וְהִתְפַּלֵּל — יָצָא. יסוד : שוע״ר צ״א סעיף א.
  • ② אֵזוֹר. לֶאֱזֹר אֵזוֹר לָעֲמִידָה מִשּׁוּם ״הִכּוֹן״, אַף עִם אַבְנֵט ; וּבִשְׁאָר בְּרָכוֹת — אֵין צָרִיךְ. יסוד : שוע״ר צ״א סעיף ב.
  • ③ כִּסּוּי הָרֹאשׁ. לְעוֹלָם לֹא אַזְכָּרָה בְּרֹאשׁ מְגֻלֶּה ; מִדַּת חֲסִידוּת שֶׁלֹּא לֵילֵךְ ד׳ אַמּוֹת מְגֻלֶּה. כּוֹבַע קַשׁ — כִּסּוּי. יסוד : שוע״ר צ״א סעיפים ג-ד.
  • ④ כְּבוֹד הָרַגְלַיִם. לֹא בַּאֲפוּנְדָּא וְלֹא בְּרַגְלַיִם מְגֻלִּים כְּמִנְהַג הַמָּקוֹם ; לֹא בָּתֵּי יָדַיִם בַּתְּפִלָּה. יסוד : שוע״ר צ״א סעיף ה.
  • ⑤ עֲטִיפָה. לְהִתְפַּלֵּל עָטוּף וּמְלֻבָּשׁ בְּכָבוֹד ; לְהַנִּיחַ יָדָיו זֶה עַל זֶה כְּפוּתִין, בְּלִי לַחֲבֹּק אֶצְבָּעוֹת בְּעֵת שָׁלוֹם (מוֹרִיד דִּין). יסוד : שוע״ר צ״א סעיף ו.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן צ״א (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַחֲגִירָה, כִּסּוּי הָרֹאשׁ, הָעֲטִיפָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
