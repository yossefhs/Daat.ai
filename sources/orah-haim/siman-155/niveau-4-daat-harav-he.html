דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קנ״ה · לֵילֵךְ מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ

דרכו של אדמו״ר הזקן בלֵילֵךְ מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ : חִיּוּב תַּלְמוּד תּוֹרָה לְכָל אִישׁ יִשְׂרָאֵל, וּקְבִיעוּת עֵת מִיָּד אַחַר הַתְּפִלָּה — קֹדֶם שֶׁיֵּלֵךְ לַעֲסָקָיו ; קַבָּלַת פְּנֵי שְׁכִינָה — ״יֵלְכוּ מֵחַיִל אֶל חָיִל״ ; פַּת שַׁחֲרִית וּשְׁמִירַת הַבְּרִיאוּת וְהַמִּדּוֹת ; וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קנ״ה — לֵילֵךְ מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ. אצל הרב, סימן זה — שֶׁכּוֹתַרְתּוֹ שֶׁכָּל אֶחָד חַיָּב בְּתַלְמוּד תּוֹרָה — נפרש לב׳ סְעִיפִים : חִיּוּב תַּלְמוּד תּוֹרָה לְכָל אִישׁ וּקְבִיעוּת הָעֵת מִיָּד אַחַר הַתְּפִלָּה, קַבָּלַת פְּנֵי שְׁכִינָה בַּהֲלִיכָה לְבֵית הַמִּדְרָשׁ, פַּת שַׁחֲרִית וּשְׁמִירַת הַבְּרִיאוּת וְהַמִּדּוֹת, וְלִלְווֹת אֶת הָעֵת הַקָּבוּעַ — וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. סימן קנ״ה הוא חידוש כתיבת שולחן ערוך הרב לאחר הסימנים שהאדמו״ר הזקן לא כתב. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קנ״ה — ב׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קנ״ה). אדמו״ר הזקן דן בו בְּלֵילֵךְ מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קנ״ה — שֶׁכָּל אֶחָד חַיָּב בְּתַלְמוּד תּוֹרָה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קנ״ה — שֶׁכָּל אֶחָד חַיָּב בְּתַלְמוּד תּוֹרָה — וּבוֹ ב׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.155), ואחריו תרגומנו. הסימן כולל ב׳ סעיפים (סעיף k → ‎.155.k).

סעיף א חִיּוּב תַּלְמוּד תּוֹרָה לְכָל אִישׁ ; קְבִיעוּת עֵת מִיָּד אַחַר הַתְּפִלָּה ; פַּת שַׁחֲרִית וּמִדּוֹת.

שֶׁכָּל אֶחָד חַיָּב בְּתַלְמוּד תּוֹרָה וּבוֹ ב׳ סְעִיפִים:כָּל אִישׁ יִשְׂרָאֵל חַיָּב בְּתַלְמוּד תּוֹרָה, בֵּין עָנִי בֵּין עָשִׁיר, בֵּין שָׁלֵם בְּגוּפוֹ בֵּין בַּעַל יִסּוּרִין, וַאֲפִלּוּ עָנִי הַמְחַזֵּר עַל הַפְּתָחִים, וַאֲפִלּוּ בַּעַל אִשָּׁה וּבָנִים — חַיָּב לִקְבֹּעַ לוֹ עֵת לְתַלְמוּד תּוֹרָה בַּיּוֹם וּבַלַּיְלָה, שֶׁנֶּאֱמַר: ״וְהָגִיתָ בּוֹ יוֹמָם וָלַיְלָה״. וּבְעֵת זוֹ צָרִיךְ שֶׁיִּלְמֹד תּוֹרָה שֶׁבִּכְתָב וְתוֹרָה שֶׁבְּעַל פֶּה הֲלָכוֹת פְּסוּקוֹת וְתַלְמוּד. וְלֹא תַּלְמוּד בִּלְבַד אִם הוּא תְּחִלַּת לִמּוּדוֹ, לְפִי שֶׁבַּתְּחִלָּה צָרִיךְ לֵידַע הַרְבֵּה מִן הָאִסּוּר וְהַמֻּתָּר בְּלֹא טְעָמִים וּרְאָיוֹת, וְאֵיךְ לַעֲשׂוֹת הַמִּצְווֹת הַנְּהוּגוֹת כְּדֵי שֶׁיֵּדַע לִשְׁמֹר וְלַעֲשׂוֹת וְלֹא יֶחֱטָא, קֹדֶם שֶׁיֵּדַע הַכֹּל עַל נָכוֹן מֵעֹמֶק לִמּוּד הַתַּלְמוּד. וְעֵת זוֹ שֶׁבַּיּוֹם מִצְוָה מִן הַמֻּבְחָר לִקְבֹּעַ אוֹתָהּ מִיָּד אַחַר הַתְּפִלָּה, קֹדֶם שֶׁיֵּלֵךְ לַעֲסָקָיו. וְכָל הַיּוֹצֵא מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ וְהוֹלֵךְ לִלְמֹד — זוֹכֶה וּמְקַבֵּל פְּנֵי הַשְּׁכִינָה, שֶׁנֶּאֱמַר: ״יֵלְכוּ מֵחַיִל אֶל חָיִל יֵרָאֶה אֶל אֱלֹהִים בְּצִיּוֹן״ (וְעַיֵּן שָׁם בְּתִלִּים בְּתַרְגּוּם יוֹנָתָן וּבַגְּמָרָא סוֹף מַסֶּכֶת מוֹעֵד קָטָן דַּף כ״ט ע״א וּבְמִדְרָשׁ רַבָּה פָּרָשַׁת שְׁמוֹת סוֹף פֶּרֶק ב׳ וּבִירוּשַׁלְמִי סוֹף פֶּרֶק ד׳ דִּשְׁבִיעִית). וּקְבִיעוּת הָעֵת צָרִיךְ שֶׁיִּהְיֶה קְבוּעָה וּתְקוּעָה בְּעִנְיָן שֶׁלֹּא יַעֲבִירֶנּוּ אַף אִם הוּא סָבוּר לְהַרְוִיחַ הַרְבֵּה. וּמִכָּל מָקוֹם קֹדֶם שֶׁיֵּלֵךְ לְבֵית הַמִּדְרָשׁ — יָכוֹל לֶאֱכֹל פַּת שַׁחֲרִית אִם הוּא רָגִיל. וְטוֹב שֶׁיַּרְגִּיל בּוֹ, כִּי פ״ג מִינֵי חֳלָאִים כְּמִנְיָן ״מַחֲלָה״ תְּלוּיִים בַּמָּרָה, וְכֻלָּם פַּת בַּמֶּלַח וְקִיתוֹן שֶׁל מַיִם שַׁחֲרִית מְבַטַּלְתָּם. וּמִצְוָה לְהַנְהִיג עַצְמוֹ בְּהַנְהָגָה טוֹבָה לִשְׁמֹר בְּרִיאוּתוֹ, כְּדֵי שֶׁיִּהְיֶה בָּרִיא וְחָזָק לַעֲבוֹדַת בּוֹרְאוֹ. וְעַל כֵּן אֵינוֹ רַשַּׁאי לְסַגֵּף עַצְמוֹ אֶלָּא אִם כֵּן יֵשׁ בּוֹ צֹרֶךְ לָעֲבוֹדָה, כְּגוֹן שֶׁטִּבְעוֹ נוֹטֶה לִהְיוֹת שָׁטוּף בְּתַאֲוַת אֲכִילָה וּשְׁתִיָּה אוֹ שְׁאָר תַּאֲווֹת, וְאִי אֶפְשָׁר לוֹ לִכְפּוֹת טִבְעוֹ אֶלָּא אִם כֵּן יַטֶּה עַצְמוֹ לְקָצֶה אַחֲרוֹן וְלִנְהֹג בְּעַצְמוֹ בִּפְרִישׁוּת לְגַמְרֵי מֵאוֹתוֹ דָּבָר שֶׁטִּבְעוֹ כּוֹסֵף, וּמַנְהִיג בָּזֶה זְמַן רַב עַד שֶׁיֵּדַע בְּנַפְשׁוֹ שֶׁאַף אִם יִנְהַג עַצְמוֹ בְּזֶה הַדֶּרֶךְ הַמְמֻצָּע לֹא יַטֶּנּוּ טִבְעוֹ מִדֶּרֶךְ הַיְשָׁרָה. וְזֶהוּ כְּלָל לְכָל הַמִּדּוֹת שֶׁבָּאָדָם, חוּץ מִן הַגַּאֲוָה וְהַכַּעַס שֶׁבָּהֶם אֵין רָאוּי לֵילֵךְ בְּמִדָּה בֵּינוֹנִית, אֶלָּא לְהִתְרַחֵק מִגֹּבַהּ הַלֵּב וְהַכַּעַס עַד קָצֶה הָאַחֲרוֹן. וּכְמוֹ שֶׁאָמְרוּ חֲכָמִים: ״מְאֹד מְאֹד הֱוֵי שְׁפַל רוּחַ״ בְּפֶרֶק ד׳ דְּאָבוֹת מִשְׁנָה ד׳, יְעֻיַּן שָׁם בְּפֵרוּשׁ הָרע״ב וּבְפֵרוּשׁ הַמִּשְׁנָיוֹת לְהָרַמְבַּ״ם שֶׁהֶאֱרִיךְ בָּזֶה וְכֵן בְּעִנְיַן הַכַּעַס, וּכְמוֹ שֶׁיִּתְבָּאֵר לְקַמָּן סִימָן קנ״ו ס״ג:

חִיּוּב תַּלְמוּד תּוֹרָה וּקְבִיעוּת הָעֵת. כָּל אִישׁ יִשְׂרָאֵל חַיָּב בְּתַלְמוּד תּוֹרָה — עָנִי וְעָשִׁיר, שָׁלֵם וּבַעַל יִסּוּרִין, אֲפִלּוּ עָנִי הַמְחַזֵּר עַל הַפְּתָחִים וּבַעַל אִשָּׁה וּבָנִים — וְחַיָּב לִקְבֹּעַ עֵת בַּיּוֹם וּבַלַּיְלָה (״וְהָגִיתָ בּוֹ יוֹמָם וָלַיְלָה״). בָּעֵת הַזֹּאת לוֹמֵד תּוֹרָה שֶׁבִּכְתָב וְשֶׁבְּעַל פֶּה — הֲלָכוֹת פְּסוּקוֹת וְתַלְמוּד ; בַּתְּחִלָּה לֵידַע הָאִסּוּר וְהַמֻּתָּר לְמַעֲשֶׂה, קֹדֶם עֹמֶק הַתַּלְמוּד. מִצְוָה מִן הַמֻּבְחָר לִקְבֹּעַ הָעֵת מִיָּד אַחַר הַתְּפִלָּה, קֹדֶם שֶׁיֵּלֵךְ לַעֲסָקָיו. וְכָל הַיּוֹצֵא מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ — מְקַבֵּל פְּנֵי הַשְּׁכִינָה (״יֵלְכוּ מֵחַיִל אֶל חָיִל״). הַקְּבִיעוּת קְבוּעָה וּתְקוּעָה, שֶׁלֹּא יַעֲבִירֶנּוּ אַף אִם סָבוּר לְהַרְוִיחַ הַרְבֵּה. קֹדֶם הַהֲלִיכָה — יָכוֹל לֶאֱכֹל פַּת שַׁחֲרִית אִם רָגִיל, וְטוֹב שֶׁיַּרְגִּיל (פ״ג מִינֵי חֳלָאִים כְּמִנְיָן ״מַחֲלָה״ שֶׁפַּת בַּמֶּלַח וְקִיתוֹן מַיִם מְבַטְּלִים). מִצְוָה לִשְׁמֹר הַבְּרִיאוּת לַעֲבוֹדַת בּוֹרְאוֹ, וְלֹא לְסַגֵּף אֶלָּא לְצֹרֶךְ — מִדָּה בֵּינוֹנִית בְּכָל הַמִּדּוֹת, חוּץ מִגַּאֲוָה וְכַעַס שֶׁמִּתְרַחֵק עַד קָצֶה הָאַחֲרוֹן (״מְאֹד מְאֹד הֱוֵי שְׁפַל רוּחַ״).

סעיף ב לִלְווֹת אֶת הָעֵת הַקָּבוּעַ ; וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד — יֵלֵךְ לְבֵית הַמִּדְרָשׁ.

אִם צָרִיךְ לַעֲשׂוֹת דָּבָר נָחוּץ מִיָּד אַחַר הַתְּפִלָּה וְרוֹצֶה לִלְווֹת עֵת הַקָּבוּעַ לוֹ וּלְפָרְעָהּ אַחַר כָּךְ בַּיּוֹם אוֹ בַּלַּיְלָה (כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן רל״ח סָעִיף ב׳ מֵהַשַּׁ״ס דְּעֵרוּבִין פֶּרֶק הַדָּר דַּף ס״ה ע״א) — אַף עַל פִּי כֵן טוֹב שֶׁיִּלְמֹד אֲפִלּוּ הֲלָכָה אַחַת אוֹ פָּסוּק אֶחָד מִיָּד אַחַר הַתְּפִלָּה. וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד — יֵלֵךְ לְבֵית הַמִּדְרָשׁ וּשְׂכַר הֲלִיכָה בְּיָדוֹ, אוֹ יִקְבַּע לוֹ מָקוֹם וְיִלְמֹד מְעַט בְּזֶה שֶׁיּוֹדֵעַ, וְיַכְנִיס יִרְאַת שָׁמַיִם בְּלִבּוֹ, וִיפַשְׁפֵּשׁ בְּמַעֲשָׂיו, אוּלַי יִמְצָא בָּהֶם אֵיזֶה צַד עֲבֵרָה וְיִפְרֹשׁ מִמֶּנָּה:

לִלְווֹת אֶת הָעֵת — וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד. אִם צָרִיךְ דָּבָר נָחוּץ מִיָּד אַחַר הַתְּפִלָּה וְרוֹצֶה לִלְווֹת אֶת הָעֵת הַקָּבוּעַ לוֹ וּלְפָרְעָהּ אַחַר כָּךְ בַּיּוֹם אוֹ בַּלַּיְלָה (כִּדְאִיתָא בְּסִימָן רל״ח סָעִיף ב׳, מֵהַשַּׁ״ס דְּעֵרוּבִין פֶּרֶק הַדָּר דַּף ס״ה ע״א) — אַף עַל פִּי כֵן טוֹב שֶׁיִּלְמֹד אֲפִלּוּ הֲלָכָה אַחַת אוֹ פָּסוּק אֶחָד מִיָּד אַחַר הַתְּפִלָּה. וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד — יֵלֵךְ לְבֵית הַמִּדְרָשׁ וּשְׂכַר הֲלִיכָה בְּיָדוֹ, אוֹ יִקְבַּע לוֹ מָקוֹם וְיִלְמֹד מְעַט בְּמַה שֶּׁיּוֹדֵעַ, וְיַכְנִיס יִרְאַת שָׁמַיִם בְּלִבּוֹ, וִיפַשְׁפֵּשׁ בְּמַעֲשָׂיו — אוּלַי יִמְצָא צַד עֲבֵרָה וְיִפְרֹשׁ מִמֶּנָּה.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קנ״ה (מהדורת קה״ת, ב׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — לֵילֵךְ מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ

היסוד : לֵילֵךְ מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה חִיּוּב תַּלְמוּד תּוֹרָה וּקְבִיעוּת עִתִּים לְכָל אִישׁ. אחר כך הַהֲלִיכָה מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ מִיָּד אַחַר הַתְּפִלָּה עִם קַבָּלַת פְּנֵי שְׁכִינָה. אחר כך פַּת שַׁחֲרִית וּשְׁמִירַת הַבְּרִיאוּת וְהַמִּדּוֹת. ולבסוף לִלְווֹת אֶת הָעֵת וּמְקוֹם מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד.

חִיּוּב תַּלְמוּד תּוֹרָה וּקְבִיעוּת עִתִּים

חִיּוּב תַּלְמוּד תּוֹרָה וּקְבִיעוּת עִתִּים

היסוד. כָּל אִישׁ יִשְׂרָאֵל חַיָּב בְּתַלְמוּד תּוֹרָה — עָנִי אוֹ עָשִׁיר — וְחַיָּב לִקְבֹּעַ עֵת בַּיּוֹם וּבַלַּיְלָה (״וְהָגִיתָ בּוֹ יוֹמָם וָלַיְלָה״) (סעיף א).

המסקנה. הַקְּבִיעוּת קְבוּעָה וּתְקוּעָה, שֶׁלֹּא יַעֲבִירֶנּוּ אַף אִם סָבוּר לְהַרְוִיחַ הַרְבֵּה. עיין סימן קנ״ו · 156.

מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ — קֹדֶם הָעֲסָקִים

מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ — קֹדֶם הָעֲסָקִים

היסוד. מִצְוָה מִן הַמֻּבְחָר לִקְבֹּעַ הָעֵת מִיָּד אַחַר הַתְּפִלָּה, קֹדֶם שֶׁיֵּלֵךְ לַעֲסָקָיו ; וְכָל הַיּוֹצֵא מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ — מְקַבֵּל פְּנֵי הַשְּׁכִינָה (סעיף א).

המסקנה. מְחַבֵּר אֶת הַלִּמּוּד לַתְּפִלָּה ; ״יֵלְכוּ מֵחַיִל אֶל חָיִל״ — מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ. עיין סימן קנ״ו · 156.

פַּת שַׁחֲרִית וּשְׁמִירַת הַבְּרִיאוּת

פַּת שַׁחֲרִית וּשְׁמִירַת הַבְּרִיאוּת

היסוד. קֹדֶם שֶׁיֵּלֵךְ לְבֵית הַמִּדְרָשׁ — יָכוֹל לֶאֱכֹל פַּת שַׁחֲרִית אִם רָגִיל ; טוֹב שֶׁיַּרְגִּיל בּוֹ (״מַחֲלָה״ תְּלוּיִים בַּמָּרָה) (סעיף א).

המסקנה. מִצְוָה לִשְׁמֹר הַבְּרִיאוּת לַעֲבוֹדַת בּוֹרְאוֹ ; מִדָּה בֵּינוֹנִית בְּכָל הַמִּדּוֹת, חוּץ מִגַּאֲוָה וְכַעַס. עיין סימן קנ״ו · 156.

לִלְווֹת אֶת הָעֵת · וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ

לִלְווֹת אֶת הָעֵת · וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ

היסוד. דָּבָר נָחוּץ — יָכוֹל לִלְווֹת אֶת הָעֵת וּלְפָרְעָהּ אַחַר כָּךְ, וְאַף עַל פִּי כֵן טוֹב שֶׁיִּלְמֹד אֲפִלּוּ הֲלָכָה אַחַת אוֹ פָּסוּק אֶחָד (סעיף ב).

המסקנה. וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד — יֵלֵךְ לְבֵית הַמִּדְרָשׁ וּשְׂכַר הֲלִיכָה בְּיָדוֹ, וְיַכְנִיס יִרְאַת שָׁמַיִם בְּלִבּוֹ וִיפַשְׁפֵּשׁ בְּמַעֲשָׂיו. עיין סימן קנ״ו · 156.

הפניה ללימוד. היקף הסימן — חִיּוּב תַּלְמוּד תּוֹרָה, קְבִיעוּת עִתִּים, הַהֲלִיכָה לְבֵית הַמִּדְרָשׁ וּמְקוֹם מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד — נוגע להנהגה היומיומית ביציאה מן התפילה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קנ״ה. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מב׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
חִיּוּב תַּלְמוּד תּוֹרָה וּקְבִיעוּת עִתִּים או״ח קנ״ה — קוֹבֵעַ עֵת לִלְמֹד, וְהָעֵת קְבוּעָה שֶׁלֹּא יַעֲבִירֶנּוּ. מ״ב קנ״ה — חִיּוּב קְבִיעוּת עִתִּים לַתּוֹרָה בַּיּוֹם וּבַלַּיְלָה. ערוה״ש קנ״ה — גֶּדֶר קְבִיעוּת הָעֵת וַחֲשִׁיבוּתָהּ. דרך אדמו״ר הזקן — כָּל אִישׁ חַיָּב, עָנִי וְעָשִׁיר, לִקְבֹּעַ עֵת בַּיּוֹם וּבַלַּיְלָה, קְבוּעָה וּתְקוּעָה (סעיף א).
מַה שֶּׁלּוֹמֵד בָּעֵת הַקְּבוּעָה או״ח קנ״ה — סְתַם קְבִיעוּת עֵת לְתַלְמוּד תּוֹרָה. מ״ב קנ״ה — מִקְרָא מִשְׁנָה וְתַלְמוּד לְפִי כֹּחוֹ. ערוה״ש קנ״ה — סֵדֶר הַלִּמּוּד לְמַתְחִיל וּלְמֻפְלָג. דרך אדמו״ר הזקן — תּוֹרָה שֶׁבִּכְתָב וְשֶׁבְּעַל פֶּה, הֲלָכוֹת פְּסוּקוֹת וְתַלְמוּד ; בַּתְּחִלָּה לֵידַע הָאִסּוּר וְהַמֻּתָּר (סעיף א).
מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ וּפַת שַׁחֲרִית או״ח קנ״ה — אַחַר בֵּית הַכְּנֶסֶת יֵלֵךְ לְבֵית הַמִּדְרָשׁ ; פַּת שַׁחֲרִית אִם רָגִיל. מ״ב קנ״ה — קֹדֶם הָעֲסָקִים, וְשֶׁכַר הַהֲלִיכָה. ערוה״ש קנ״ה — טַעַם פַּת שַׁחֲרִית וּשְׁמִירַת הַגּוּף. דרך אדמו״ר הזקן — מִיָּד אַחַר הַתְּפִלָּה, קֹדֶם עֲסָקָיו, וּמְקַבֵּל פְּנֵי שְׁכִינָה ; פַּת שַׁחֲרִית וּשְׁמִירַת הַבְּרִיאוּת וְהַמִּדּוֹת (סעיף א).
לִלְווֹת הָעֵת וּמִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד או״ח קנ״ה — אַף מִי שֶׁאֵינוֹ יוֹדֵעַ יֵלֵךְ ; שְׂכַר הֲלִיכָה בְּיָדוֹ ; יִקְבַּע מָקוֹם. מ״ב קנ״ה — יִרְאַת שָׁמַיִם וּפִשְׁפּוּשׁ בַּמַּעֲשִׂים. ערוה״ש קנ״ה — חוֹבַת כָּל אָדָם אַף בְּמַה שֶּׁיּוֹדֵעַ. דרך אדמו״ר הזקן — דָּבָר נָחוּץ — לִלְווֹת הָעֵת וּלְפָרְעָהּ ; אֲפִלּוּ הֲלָכָה אַחַת ; מִי שֶׁאֵינוֹ יוֹדֵעַ — יֵלֵךְ וִיפַשְׁפֵּשׁ בְּמַעֲשָׂיו (סעיף ב).

הטבלה נערכה מטקסט המחבר (אורח חיים קנ״ה, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תלמוד תורה, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ח׳, מועד קטן כ״ט, בבא קמא פ״ב). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ב׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — חִיּוּב וּקְבִיעוּת

קו א — חִיּוּב וּקְבִיעוּת

יסוד הסימן : כָּל אִישׁ יִשְׂרָאֵל חַיָּב בְּתַלְמוּד תּוֹרָה וְלִקְבֹּעַ עֵת בַּיּוֹם וּבַלַּיְלָה (סעיף א).

קריאת הרב : הַקְּבִיעוּת קְבוּעָה וּתְקוּעָה, שֶׁלֹּא יַעֲבִירֶנּוּ אַף לְרֶוַח גָּדוֹל. עיין סימן קנ״ו · 156.

מסקנה למעשה : לִקְבֹּעַ עֵת לִמּוּד יוֹמִית קְבוּעָה, בְּלִי לְהַעֲבִירָהּ מִפְּנֵי הָעֲסָקִים.

קו ב — הַהֲלִיכָה וּפַת שַׁחֲרִית

קו ב — הַהֲלִיכָה וּפַת שַׁחֲרִית

יסוד הסימן : מִיָּד אַחַר הַתְּפִלָּה, קֹדֶם עֲסָקָיו — וְהַיּוֹצֵא לְבֵית הַמִּדְרָשׁ מְקַבֵּל פְּנֵי שְׁכִינָה (סעיף א).

קריאת הרב : קֹדֶם הַהֲלִיכָה, פַּת שַׁחֲרִית אִם רָגִיל, וּשְׁמִירַת הַבְּרִיאוּת לַעֲבוֹדַת בּוֹרְאוֹ. עיין סימן קנ״ו · 156.

מסקנה למעשה : לְחַבֵּר הַלִּמּוּד לַתְּפִלָּה, וְלִשְׁמֹר עַל הַגּוּף בְּלִי לְסַגֵּף.

קו ג — לִלְווֹת וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ

קו ג — לִלְווֹת וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ

יסוד הסימן : דָּבָר נָחוּץ — לִלְווֹת אֶת הָעֵת וּלְפָרְעָהּ, אֲבָל טוֹב לִלְמֹד אֲפִלּוּ הֲלָכָה אַחַת מִיָּד (סעיף ב).

קריאת הרב : וְאַף מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד — יֵלֵךְ לְבֵית הַמִּדְרָשׁ וּשְׂכַר הֲלִיכָה בְּיָדוֹ. עיין סימן קנ״ו · 156.

מסקנה למעשה : לֹא לְוַתֵּר לְגַמְרֵי — לְפָחוֹת פָּסוּק אוֹ הֲלָכָה, וְיִרְאַת שָׁמַיִם עִם פִּשְׁפּוּשׁ בַּמַּעֲשִׂים.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קנ״ה ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קנ״ה (לֵילֵךְ מִבֵּית הַכְּנֶסֶת לְבֵית הַמִּדְרָשׁ)

  • ① חִיּוּב. כָּל אִישׁ יִשְׂרָאֵל חַיָּב בְּתַלְמוּד תּוֹרָה — עָנִי וְעָשִׁיר — וְחַיָּב לִקְבֹּעַ עֵת בַּיּוֹם וּבַלַּיְלָה. יסוד : שוע״ר קנ״ה סעיף א.
  • ② קְבִיעוּת. הָעֵת קְבוּעָה וּתְקוּעָה ; שֶׁלֹּא יַעֲבִירֶנּוּ אַף אִם סָבוּר לְהַרְוִיחַ הַרְבֵּה. יסוד : שוע״ר קנ״ה סעיף א.
  • ③ מִיָּד אַחַר הַתְּפִלָּה. לִקְבֹּעַ הָעֵת מִיָּד אַחַר הַתְּפִלָּה, קֹדֶם עֲסָקָיו ; הַיּוֹצֵא לְבֵית הַמִּדְרָשׁ — מְקַבֵּל פְּנֵי שְׁכִינָה. יסוד : שוע״ר קנ״ה סעיף א.
  • ④ פַּת שַׁחֲרִית וּמִדּוֹת. קֹדֶם הַהֲלִיכָה — פַּת שַׁחֲרִית אִם רָגִיל ; לִשְׁמֹר הַבְּרִיאוּת וְלֵילֵךְ בְּמִדָּה בֵּינוֹנִית, חוּץ מִגַּאֲוָה וְכַעַס. יסוד : שוע״ר קנ״ה סעיף א.
  • ⑤ לִלְווֹת · מִי שֶׁאֵינוֹ יוֹדֵעַ. דָּבָר נָחוּץ — לִלְווֹת הָעֵת, אֲבָל תָּמִיד אֲפִלּוּ הֲלָכָה אַחַת מִיָּד ; וּמִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד — יֵלֵךְ לְבֵית הַמִּדְרָשׁ וִיפַשְׁפֵּשׁ בְּמַעֲשָׂיו. יסוד : שוע״ר קנ״ה סעיף ב.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קנ״ה (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַחִיּוּב, הַקְּבִיעוּת, פַּת שַׁחֲרִית, מְקוֹם מִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
