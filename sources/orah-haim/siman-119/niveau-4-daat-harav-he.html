דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן קי״ט · דִּין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת

דרכו של אדמו״ר הזקן בדִּין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת : הוֹסָפָה מֵעֵין הַבְּרָכָה — יָחִיד וְצִבּוּר, בְּסוֹף אוֹ בְּאֶמְצַע ; הוִדּוּי בְּ״שׁוֹמֵעַ תְּפִלָּה״ ; הַזְכָּרַת שֵׁם הַחוֹלֶה ; וְסֵדֶר הַבְּרָכוֹת עִם דִּין הַ״עֲנֵנוּ״ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן קי״ט — דִּין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת. אצל הרב הסימן נפרש לו׳ סְעִיפִים : הַהוֹסָפָה מֵעֵין הַבְּרָכָה (יָחִיד וְצִבּוּר), הַוִּדּוּי בְּ״שׁוֹמֵעַ תְּפִלָּה״, הַזְכָּרַת שֵׁם הַחוֹלֶה, סֵדֶר הַבְּרָכוֹת וְהַדִּלֵּג בְּרָכָה, דִּין הַ״עֲנֵנוּ״ לִשְׁלִיחַ צִבּוּר, וְהַטָּעוּת בְּ״עֲנֵנוּ״ קֹדֶם ״רְאֵה נָא״.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן קי״ט — ו׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim קי״ט). אדמו״ר הזקן דן בו בְּדִין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן קי״ט — דִּין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן קי״ט — דִּין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת — וּבוֹ ו׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.119), ואחריו תרגומנו. הסימן כולל ו׳ סעיפים (סעיף k → ‎.119.k).

סעיף א הוֹסָפָה מֵעֵין הַבְּרָכָה ; יָחִיד וְצִבּוּר ; בְּסוֹף אוֹ בְּאֶמְצַע.

דִּין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת וּבוֹ ו׳ סְעִיפִים:אִם רָצָה לְהוֹסִיף בְּכָל בְּרָכָה מֵהָאֶמְצָעִיּוֹת מֵעֵין הַבְּרָכָה — מוֹסִיף. כֵּיצַד? הָיָה לוֹ חוֹלֶה — מְבַקֵּשׁ עָלָיו רַחֲמִים בְּ״בִרְכַּת רְפָאֵנוּ״. הָיָה צָרִיךְ לְפַרְנָסָה — מְבַקֵּשׁ בְּ״בִרְכַּת הַשָּׁנִים״. אֲבָל בְּ״שׁוֹמֵעַ תְּפִלָּה״ יָכוֹל לִשְׁאֹל כָּל צְרָכָיו, מִפְּנֵי שֶׁהִיא כּוֹלֶלֶת כָּל הַבַּקָּשׁוֹת. וּכְשֶׁהוּא מוֹסִיף — מַתְחִיל בַּבְּרָכָה וְאַחַר כָּךְ מוֹסִיף, אֲבָל לֹא יוֹסִיף וְאַחַר כָּךְ יַתְחִיל הַבְּרָכָה, מִפְּנֵי שֶׁצָּרִיךְ לַעֲשׂוֹת עִקָּר מִמַּטְבֵּעַ שֶׁתִּקְּנוּ חֲכָמִים וּבַקָּשָׁתוֹ תִּהְיֶה טְפֵלָה. וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁכְּשֶׁמּוֹסִיף בַּבְּרָכָה מֵעֵין אוֹתָהּ בְּרָכָה, אִם מוֹסִיף וּמְבַקֵּשׁ רַחֲמִים בִּשְׁבִיל כָּל יִשְׂרָאֵל — צָרִיךְ לוֹמַר בִּלְשׁוֹן רַבִּים וְלֹא בִּלְשׁוֹן יָחִיד, שֶׁלֹּא יִהְיֶה כְּהֶפְסֵק בַּתְּפִלָּה. וְלֹא יוֹסִיף אֶלָּא בְּסוֹף הַבְּרָכָה וְלֹא בְּאֶמְצָעָהּ, שֶׁכְּשֶׁשּׁוֹאֵל צָרְכֵי רַבִּים בְּאֶמְצַע הַבְּרָכָה — נִרְאֶה כְּמוֹסִיף עַל מַטְבֵּעַ שֶׁתִּקְּנוּ חֲכָמִים. בַּמֶּה דְּבָרִים אֲמוּרִים? בְּיָחִיד, אֲבָל בְּצִבּוּר — מֻתָּרִים. וְלָכֵן אוֹמְרִים סְלִיחוֹת בְּאֶמְצַע בִּרְכַּת ״סְלַח לָנוּ״, וְכֵן בְּאֶמְצַע בְּרָכָה אֶמְצָעִית שֶׁל יוֹם הַכִּפּוּרִים. וְאִם שׁוֹאֵל צְרָכָיו מַמָּשׁ, כְּגוֹן שֶׁיֵּשׁ לוֹ חֹלִי בְּתוֹךְ בֵּיתוֹ, אוֹ שֶׁהוּא צָרִיךְ לְפַרְנָסָה — יָכוֹל לִשְׁאֹל אֲפִלּוּ בְּאֶמְצַע הַבְּרָכָה, וְהוּא שֶׁיֹּאמַר בִּלְשׁוֹן יָחִיד וְלֹא בִּלְשׁוֹן רַבִּים, שֶׁלֹּא יִדְמֶה כְּמוֹסִיף עַל מַטְבֵּעַ שֶׁתִּקְּנוּ חֲכָמִים. וּבְ״שׁוֹמֵעַ תְּפִלָּה״ יָכוֹל לִשְׁאֹל בֵּין בִּלְשׁוֹן יָחִיד בֵּין בִּלְשׁוֹן רַבִּים, בֵּין צְרָכָיו מַמָּשׁ בֵּין צָרְכֵי רַבִּים, שֶׁבִּרְכַּת ״שׁוֹמֵעַ תְּפִלָּה״ כּוֹלֶלֶת כָּל הַצְּרָכִים וּלְכָךְ נִתְקְנָה. וּמִכָּל מָקוֹם כְּשֶׁמּוֹסִיף לְצֹרֶךְ יָחִיד — לֹא יַאֲרִיךְ אֲפִלּוּ הוּא מֵעֵין הַבְּרָכָה אוֹ בְּ״שׁוֹמֵעַ תְּפִלָּה״, אֲבָל לְאַחַר הַתְּפִלָּה קֹדֶם שֶׁעָקַר רַגְלָיו — רַשַּׁאי לְהַאֲרִיךְ אֲפִלּוּ כְּסֵדֶר יוֹם הַכִּפּוּרִים. וּמַה שֶּׁנּוֹהֲגִין לוֹמַר סְלִיחוֹת בִּ״סְלַח לָנוּ״ וּמַרְבִּים בְּתַחֲנוּנִים בְּאֶמְצַע בְּרָכָה, זֶהוּ מִפְּנֵי שֶׁהֵם צָרְכֵי רַבִּים. וְכֵן אֲפִלּוּ לְצֹרֶךְ יָחִיד מֻתָּר לוֹמַר סְלִיחוֹת וְתַחֲנוּנִים הַרְבֵּה בְּאֶמְצַע בְּרָכָה אִם רַבִּים צְרִיכִים לְתוֹרָתוֹ, שֶׁזֶּהוּ כְּצֹרֶךְ רַבִּים. וְטוֹב לִזָּהֵר בְּכָל זֶה, אַף עַל פִּי שֶׁיֵּשׁ חוֹלְקִים עַל דְּבָרָיו וּמַתִּירִים כָּל זֶה (וְדִבְרֵיהֶם נִרְאִים עִקָּר):

מֵעֵין הַבְּרָכָה. מוֹסִיף בָּאֶמְצָעִיּוֹת מֵעֵין הַבְּרָכָה : חוֹלֶה בְּ״רְפָאֵנוּ״, פַּרְנָסָה בְּ״בִרְכַּת הַשָּׁנִים״, וְכָל צְרָכָיו בְּ״שׁוֹמֵעַ תְּפִלָּה״. מַתְחִיל בַּבְּרָכָה וְאַחַר כָּךְ מוֹסִיף, שֶׁהָעִקָּר מִמַּטְבֵּעַ חֲכָמִים. לְכָל יִשְׂרָאֵל — בִּלְשׁוֹן רַבִּים וּבְסוֹף הַבְּרָכָה ; בְּצִבּוּר — מֻתָּרִים אֲפִלּוּ בְּאֶמְצַע (סְלִיחוֹת). לְצֹרֶךְ יָחִיד מַמָּשׁ — בִּלְשׁוֹן יָחִיד, וְלֹא יַאֲרִיךְ, מִלְּבַד לְאַחַר הַתְּפִלָּה קֹדֶם שֶׁעָקַר רַגְלָיו.

סעיף ב וִדּוּי בְּ״שׁוֹמֵעַ תְּפִלָּה״ בִּלְשׁוֹן יָחִיד ; מְזוֹנוֹתָיו אֲפִלּוּ עָשִׁיר.

טוֹב לְהִתְוַדּוֹת בְּ״שׁוֹמֵעַ תְּפִלָּה״ בִּלְשׁוֹן יָחִיד, מִלְּבַד הַוִּדּוּי שֶׁבִּלְשׁוֹן רַבִּים שֶׁאַחַר הַתְּפִלָּה, וְלִשְׁאֹל אַחַר כָּךְ מְזוֹנוֹתָיו אֲפִלּוּ הוּא עָשִׁיר:

וִדּוּי בְּ״שׁוֹמֵעַ תְּפִלָּה״. טוֹב לְהִתְוַדּוֹת בְּ״שׁוֹמֵעַ תְּפִלָּה״ בִּלְשׁוֹן יָחִיד, מִלְּבַד הַוִּדּוּי בִּלְשׁוֹן רַבִּים שֶׁאַחַר הַתְּפִלָּה ; וְלִשְׁאֹל מְזוֹנוֹתָיו אֲפִלּוּ הוּא עָשִׁיר.

סעיף ג מְבַקֵּשׁ עַל חֲבֵרוֹ בְּפָנָיו — אֵין צָרִיךְ לְהַזְכִּיר שְׁמוֹ.

הַמְבַקֵּשׁ רַחֲמִים עַל חֲבֵרוֹ בְּפָנָיו — אֵין צָרִיךְ לְהַזְכִּיר שְׁמוֹ, שֶׁנֶּאֱמַר: ״אֵל נָא רְפָא נָא לָהּ״, וְלֹא הִזְכִּיר שֵׁם מִרְיָם. אֲבָל שֶׁלֹּא בְּפָנָיו — צָרִיךְ לְהַזְכִּיר שְׁמוֹ:

הַזְכָּרַת שֵׁם. הַמְבַקֵּשׁ עַל חֲבֵרוֹ בְּפָנָיו — אֵין צָרִיךְ לְהַזְכִּיר שְׁמוֹ, שֶׁנֶּאֱמַר ״אֵל נָא רְפָא נָא לָהּ״ (וְלֹא הִזְכִּיר שֵׁם מִרְיָם) ; אֲבָל שֶׁלֹּא בְּפָנָיו — צָרִיךְ לְהַזְכִּיר שְׁמוֹ.

סעיף ד דִּלֵּג בְּרָכָה — חוֹזֵר לְרֹאשׁ הַבְּרָכָה, וּמִשָּׁם עַל הַסֵּדֶר.

אַף עַל פִּי שֶׁג׳ רִאשׁוֹנוֹת חֲשׁוּבוֹת כְּאַחַת וְכֵן ג׳ אַחֲרוֹנוֹת, מִכָּל מָקוֹם י״ג אֶמְצָעִיּוֹת אֵינָן חֲשׁוּבִים כְּאַחַת. לְפִיכָךְ, אִם דִּלֵּג בְּרָכָה אַחַת, אוֹ שֶׁטָּעָה בָּהּ טָעוּת שֶׁמַּחֲזִירִין אוֹתוֹ — אֵין צָרִיךְ לַחֲזֹר לְבִרְכַּת ״אַתָּה חוֹנֵן״ אֶלָּא לְרֹאשׁ הַבְּרָכָה שֶׁטָּעָה בָּהּ אוֹ שֶׁדִּלֵּג. וּמִכָּל מָקוֹם מִשָּׁם וְאֵילָךְ צָרִיךְ לוֹמַר עַל הַסֵּדֶר, שֶׁאִם לֹא נִזְכַּר עַד לְאַחַר כַּמָּה בְּרָכוֹת — צָרִיךְ לַחֲזֹר וְלוֹמַר כֻּלָּן אַחֲרֶיהָ, שֶׁאִם יֹאמַר אוֹתָהּ בְּרָכָה לְבַדָּהּ בַּמָּקוֹם שֶׁנִּזְכַּר — נִמְצָא שִׁנָּה סִדְּרָן שֶׁל בְּרָכוֹת וְלֹא יָצָא יְדֵי חוֹבָתוֹ, שֶׁסֵּדֶר בְּרָכוֹת הוּא מֵאַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה וּמֵהֶם כַּמָּה נְבִיאִים, וְסָמְכוּ סִדְרָן עַל הַמִּקְרָאוֹת:

סֵדֶר הַבְּרָכוֹת. ג׳ רִאשׁוֹנוֹת (וְג׳ אַחֲרוֹנוֹת) חֲשׁוּבוֹת כְּאַחַת, אֲבָל י״ג אֶמְצָעִיּוֹת לֹא. דִּלֵּג בְּרָכָה אוֹ טָעוּת שֶׁמַּחֲזִירִין — אֵין חוֹזֵר לְ״אַתָּה חוֹנֵן״ אֶלָּא לְרֹאשׁ הַבְּרָכָה שֶׁטָּעָה, וּמִשָּׁם עַל הַסֵּדֶר (וְאוֹמֵר כֻּלָּן אַחֲרֶיהָ). סֵדֶר בְּרָכוֹת מֵאַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה, סְמוּכִים עַל הַמִּקְרָאוֹת.

סעיף ה שְׁלִיחַ צִבּוּר שֶׁשָּׁכַח ״עֲנֵנוּ״ אַחַר ״רְפָאֵנוּ״ — לֹא יַחֲזֹר.

לְפִיכָךְ, שְׁלִיחַ צִבּוּר שֶׁשָּׁכַח לוֹמַר ״עֲנֵנוּ״ בְּתַעֲנִית צִבּוּר בֵּין ״גּוֹאֵל״ לְ״רוֹפֵא״ וְלֹא נִזְכַּר עַד לְאַחַר שֶׁחָתַם בִּרְכַּת ״רְפָאֵנוּ״ — לֹא יַחֲזֹר, שֶׁאִם הָיָה חוֹזֵר הָיָה צָרִיךְ לוֹמַר ״עֲנֵנוּ״ וְאַחַר כָּךְ ״רְפָאֵנוּ״ שֶׁלֹּא לְשַׁנּוֹת סִדְרָן שֶׁל בְּרָכוֹת, וְזֶה אִי אֶפְשָׁר, שֶׁאִם כֵּן בִּרְכַּת ״רְפָאֵנוּ״ הִיא בְּרָכָה לְבַטָּלָה, כֵּיוָן שֶׁאֵין מַחֲזִירִין אוֹתוֹ בִּשְׁבִיל ״עֲנֵנוּ״ כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תקס״ה. אֲבָל אִם לֹא חָתַם ״רְפָאֵנוּ״ — יֹאמַר ״עֲנֵנוּ״ וְאַחַר כָּךְ ״רְפָאֵנוּ״. וְלֹא אָמְרוּ שֶׁאֵין מַחֲזִירִין אוֹתוֹ אֶלָּא כְּשֶׁאִי אֶפְשָׁר לוֹ לְתַקֵּן טָעוּתוֹ אֶלָּא אִם כֵּן יַחֲזֹר וִיבָרֵךְ בִּרְכוֹת הַתְּפִלָּה שֵׁנִית, אֲבָל לוֹמַר בִּרְכַּת ״עֲנֵנוּ״ לְבַדָּהּ — מַחֲזִירִין אוֹתוֹ, שֶׁזֶּה אֵינָהּ נִקְרָא חֲזָרָה כֵּיוָן שֶׁאוֹמְרָהּ בִּמְקוֹמָהּ בֵּין ״גּוֹאֵל״ לְ״רוֹפֵא״. וְאִם נִזְכַּר אַחַר חֲתִימַת ״רְפָאֵנוּ״ קֹדֶם חֲתִימַת ״שׁוֹמֵעַ תְּפִלָּה״ — יֹאמַר ״עֲנֵנוּ״ בְּ״שׁוֹמֵעַ תְּפִלָּה״ כְּיָחִיד. וְהוּא הַדִּין אִם לֹא הָיוּ עֲשָׂרָה הַמִּתְעַנִּים בְּבֵית הַכְּנֶסֶת וּבָאוּ אַחַר ״רְפָאֵנוּ״ — אוֹמֵר בְּ״שׁוֹמֵעַ תְּפִלָּה״, וְיַחְתֹּם ״בָּרוּךְ אַתָּה ה׳ הָעוֹנֶה בְּעֵת צָרָה וְשׁוֹמֵעַ תְּפִלָּה״. וְאֵין זוֹ חֲתִימָה בִּשְׁתַּיִם, מִפְּנֵי שֶׁבִּרְכַּת ״שׁוֹמֵעַ תְּפִלָּה״ וַחֲתִימָתָהּ מַשְׁמָעָהּ בֵּין עַל תְּפִלַּת תַּעֲנִית בֵּין עַל שְׁאָר הַתְּפִלָּה. לְפִיכָךְ, אִם חָתַם בְּ״שׁוֹמֵעַ תְּפִלָּה״ לְבַד — יָצָא. וְאִם שָׁכַח גַּם בְּ״שׁוֹמֵעַ תְּפִלָּה״ — אוֹמְרָהּ בְּרָכָה בִּפְנֵי עַצְמָהּ אַחַר ״בַּשָּׁלוֹם״. וְאֵין זֶה שִׁנּוּי מִסֵּדֶר הַבְּרָכוֹת, כֵּיוָן שֶׁכְּבָר נִסְתַּיְּמוּ בִּרְכוֹת י״ח:

שְׁלִיחַ צִבּוּר שֶׁשָּׁכַח ״עֲנֵנוּ״. ש״ץ שֶׁשָּׁכַח ״עֲנֵנוּ״ בֵּין ״גּוֹאֵל״ לְ״רוֹפֵא״ וְלֹא נִזְכַּר עַד שֶׁחָתַם ״רְפָאֵנוּ״ — לֹא יַחֲזֹר (שֶׁ״רְפָאֵנוּ״ תִּהְיֶה בְּרָכָה לְבַטָּלָה) ; אוֹמֵר ״עֲנֵנוּ״ בְּ״שׁוֹמֵעַ תְּפִלָּה״ כְּיָחִיד. לֹא חָתַם ״רְפָאֵנוּ״ — ״עֲנֵנוּ״ וְאַחַר כָּךְ ״רְפָאֵנוּ״. בְּתַעֲנִית חוֹתֵם ״הָעוֹנֶה בְּעֵת צָרָה וְשׁוֹמֵעַ תְּפִלָּה״ ; שָׁכַח גַּם בְּ״שׁוֹמֵעַ תְּפִלָּה״ — בְּרָכָה בִּפְנֵי עַצְמָהּ אַחַר ״בַּשָּׁלוֹם״, שֶׁכְּבָר נִסְתַּיְּמוּ בִּרְכוֹת י״ח.

סעיף ו אָמַר ״עֲנֵנוּ״ קֹדֶם ״רְאֵה נָא״ — חוֹזֵר, וְסֵדֶר הַבְּרָכוֹת מְעַכֵּב.

אִם טָעָה וְאָמַר ״עֲנֵנוּ״ קֹדֶם ״רְאֵה נָא בְעָנְיֵנוּ״ — צָרִיךְ לַחֲזֹר וְלוֹמַר ״עֲנֵנוּ״ אַחַר ״רְאֵה נָא״, שֶׁאִם לֹא יַחֲזֹר — נִמְצָא שֶׁאָמַר בִּרְכַּת ״רְאֵה נָא״ שֶׁלֹּא בִּמְקוֹמָהּ, שֶׁמְּקוֹמָהּ קֹדֶם ״עֲנֵנוּ״, וְסֵדֶר הַבְּרָכוֹת מְעַכֵּב:

״עֲנֵנוּ״ קֹדֶם ״רְאֵה נָא״. טָעָה וְאָמַר ״עֲנֵנוּ״ קֹדֶם ״רְאֵה נָא בְעָנְיֵנוּ״ — צָרִיךְ לַחֲזֹר וְלוֹמַר ״עֲנֵנוּ״ אַחַר ״רְאֵה נָא״, שֶׁאַחֶרֶת ״רְאֵה נָא״ נֶאֶמְרָה שֶׁלֹּא בִּמְקוֹמָהּ (מְקוֹמָהּ קֹדֶם ״עֲנֵנוּ״), וְסֵדֶר הַבְּרָכוֹת מְעַכֵּב.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim קי״ט (מהדורת קה״ת, ו׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת

היסוד : דִּין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַהוֹסָפָה מֵעֵין הַבְּרָכָה וְהַהַבְחָנָה יָחִיד / צִבּוּר. אחר כך הַוִּדּוּי בְּ״שׁוֹמֵעַ תְּפִלָּה״. אחר כך הַזְכָּרַת שֵׁם הַחוֹלֶה. ולבסוף סֵדֶר הַבְּרָכוֹת וְדִין הַ״עֲנֵנוּ״ (שֶׁסֵּדֶר בְּרָכוֹת מְעַכֵּב).

הוֹסָפָה מֵעֵין הַבְּרָכָה — יָחִיד וְצִבּוּר

הוֹסָפָה מֵעֵין הַבְּרָכָה — יָחִיד וְצִבּוּר

היסוד. מוֹסִיף מֵעֵין הַבְּרָכָה : חוֹלֶה בְּ״רְפָאֵנוּ״, פַּרְנָסָה בְּ״בִרְכַּת הַשָּׁנִים״, וְכָל צְרָכָיו בְּ״שׁוֹמֵעַ תְּפִלָּה״. מַתְחִיל בַּבְּרָכָה וְאַחַר כָּךְ מוֹסִיף (סעיף א).

המסקנה. בְּצִבּוּר — מֻתָּר אֲפִלּוּ בְּאֶמְצַע ; בְּיָחִיד — בְּסוֹף וּבִלְשׁוֹן יָחִיד. עיין סימן 120.

וִדּוּי בְּ״שׁוֹמֵעַ תְּפִלָּה״

וִדּוּי בְּ״שׁוֹמֵעַ תְּפִלָּה״

היסוד. טוֹב לְהִתְוַדּוֹת בְּ״שׁוֹמֵעַ תְּפִלָּה״ בִּלְשׁוֹן יָחִיד, מִלְּבַד הַוִּדּוּי שֶׁאַחַר הַתְּפִלָּה (סעיף ב).

המסקנה. לִשְׁאֹל מְזוֹנוֹתָיו אֲפִלּוּ הוּא עָשִׁיר — הַבְּרָכָה כּוֹלֶלֶת כָּל הַצְּרָכִים. עיין סימן 118.

הַזְכָּרַת שֵׁם הַחוֹלֶה

הַזְכָּרַת שֵׁם הַחוֹלֶה

היסוד. בְּפָנָיו — אֵין צָרִיךְ לְהַזְכִּיר שְׁמוֹ (״אֵל נָא רְפָא נָא לָהּ״) ; שֶׁלֹּא בְּפָנָיו — צָרִיךְ לְהַזְכִּיר (סעיף ג).

המסקנה. בִּפְנֵי הַחוֹלֶה הַתְּפִלָּה עוֹלָה בְּלֹא שֵׁם ; מֵרָחוֹק — מַזְכִּיר שְׁמוֹ. עיין סימן 122.

סֵדֶר הַבְּרָכוֹת וַ״עֲנֵנוּ״

סֵדֶר הַבְּרָכוֹת וַ״עֲנֵנוּ״

היסוד. דִּלֵּג בְּרָכָה — חוֹזֵר לְרֹאשׁ הַבְּרָכָה וּמִשָּׁם עַל הַסֵּדֶר ; ש״ץ שֶׁשָּׁכַח ״עֲנֵנוּ״ אַחַר ״רְפָאֵנוּ״ — לֹא יַחֲזֹר, אוֹמְרוֹ בְּ״שׁוֹמֵעַ תְּפִלָּה״ (סעיפים ד-ה).

המסקנה. סֵדֶר בְּרָכוֹת מְעַכֵּב ; ״עֲנֵנוּ״ קֹדֶם ״רְאֵה נָא״ — חוֹזֵר (סעיף ו). עיין סימן 124.

הפניה ללימוד. היקף הסימן — הַהוֹסָפָה בַּבְּרָכוֹת, הַסֵּדֶר וְדִין הַ״עֲנֵנוּ״ — נוגע להנהגת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן קי״ט. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מו׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הוֹסָפָה מֵעֵין הַבְּרָכָה או״ח קי״ט — מוֹסִיף מֵעֵין הַבְּרָכָה ; יָחִיד בְּסוֹף, וּבְ״שׁוֹמֵעַ תְּפִלָּה״ כָּל צְרָכָיו. מ״ב קי״ט — פְּרָטֵי הַהוֹסָפָה, מַתְחִיל בַּבְּרָכָה וְאַחַר כָּךְ מוֹסִיף. ערוה״ש קי״ט — גֶּדֶר צָרְכֵי יָחִיד וְצָרְכֵי רַבִּים. דרך אדמו״ר הזקן — יָחִיד בְּסוֹף וּבִלְשׁוֹן יָחִיד ; צִבּוּר מֻתָּרִים אֲפִלּוּ בְּאֶמְצַע (סְלִיחוֹת) ; לֹא יַאֲרִיךְ לְצֹרֶךְ יָחִיד (סעיף א).
וִדּוּי וּבַקָּשַׁת מְזוֹנוֹת או״ח קי״ט — כְּלַל דִּין הַהוֹסָפָה בַּבְּרָכוֹת הָאֶמְצָעִיּוֹת. מ״ב קי״ט — עִנְיַן הַוִּדּוּי וְהַתַּחֲנוּנִים בַּתְּפִלָּה. ערוה״ש קי״ט — הַמִּתְוַדֶּה וְשׁוֹאֵל צְרָכָיו. דרך אדמו״ר הזקן — טוֹב לְהִתְוַדּוֹת בְּ״שׁוֹמֵעַ תְּפִלָּה״ בִּלְשׁוֹן יָחִיד, וְלִשְׁאֹל מְזוֹנוֹתָיו אֲפִלּוּ עָשִׁיר (סעיף ב).
הַזְכָּרַת שֵׁם הַחוֹלֶה או״ח קי״ט — בָּא בִּכְלַל בַּקָּשַׁת רַחֲמִים עַל הַחוֹלֶה. מ״ב קי״ט — בְּפָנָיו וְשֶׁלֹּא בְּפָנָיו. ערוה״ש קי״ט — טַעַם הַהַזְכָּרָה שֶׁלֹּא בְּפָנָיו. דרך אדמו״ר הזקן — בְּפָנָיו אֵין צָרִיךְ לְהַזְכִּיר שְׁמוֹ (״אֵל נָא רְפָא נָא לָהּ״) ; שֶׁלֹּא בְּפָנָיו — צָרִיךְ (סעיף ג).
סֵדֶר הַבְּרָכוֹת וַ״עֲנֵנוּ״ או״ח קי״ט — דִּלֵּג בְּרָכָה חוֹזֵר לְרֹאשׁ הַבְּרָכָה ; ש״ץ שֶׁגָּמַר גּוֹאֵל וְשָׁכַח עֲנֵנוּ לֹא יַחֲזִיר. מ״ב קי״ט — פְּרָטֵי הַ״עֲנֵנוּ״ בְּתַעֲנִית וּסְדַר הַחֲזָרָה. ערוה״ש קי״ט — גֶּדֶר עִכּוּב הַסֵּדֶר בָּאֶמְצָעִיּוֹת. דרך אדמו״ר הזקן — לְרֹאשׁ הַבְּרָכָה וּמִשָּׁם עַל הַסֵּדֶר ; שָׁכַח ״עֲנֵנוּ״ אַחַר ״רְפָאֵנוּ״ — בְּ״שׁוֹמֵעַ תְּפִלָּה״ ; קֹדֶם ״רְאֵה נָא״ — חוֹזֵר (סעיפים ד-ו).

הטבלה נערכה מטקסט המחבר (אורח חיים קי״ט, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות ל״ד, עבודה זרה ח׳). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ו׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — עִקָּר הַמַּטְבֵּעַ וְהַתּוֹסֶפֶת טְפֵלָה

קו א — עִקָּר הַמַּטְבֵּעַ וְהַתּוֹסֶפֶת טְפֵלָה

יסוד הסימן : מַתְחִיל בַּבְּרָכָה וְאַחַר כָּךְ מוֹסִיף, שֶׁהָעִקָּר מִמַּטְבֵּעַ חֲכָמִים וּבַקָּשָׁתוֹ טְפֵלָה (סעיף א).

קריאת הרב : הַתּוֹסֶפֶת הָאִישִׁית לֹא תַּעֲקֹר אֶת הַמַּטְבֵּעַ שֶׁתִּקְּנוּ חֲכָמִים. עיין סימן 120.

מסקנה למעשה : לְהַתְחִיל אֶת הַבְּרָכָה, וְרַק אַחַר כָּךְ לְהוֹסִיף — לְעוֹלָם לֹא לְהֵפֶךְ.

קו ב — יָחִיד וְצִבּוּר

קו ב — יָחִיד וְצִבּוּר

יסוד הסימן : יָחִיד — בְּסוֹף וּבִלְשׁוֹן יָחִיד ; צִבּוּר — מֻתָּרִים אֲפִלּוּ בְּאֶמְצַע (סְלִיחוֹת בְּ״סְלַח לָנוּ״) (סעיף א).

קריאת הרב : צֹרֶךְ רַבִּים מַרְחִיב אֶת מַה שֶּׁצֹּרֶךְ יָחִיד מְצַמְצֵם. עיין סימן 118.

מסקנה למעשה : צְרָכָיו — בְּקִצּוּר וּבִלְשׁוֹן יָחִיד ; לֹא לְהַאֲרִיךְ לְצֹרֶךְ יָחִיד.

קו ג — סֵדֶר בְּרָכוֹת מְעַכֵּב

קו ג — סֵדֶר בְּרָכוֹת מְעַכֵּב

יסוד הסימן : דִּלֵּג בְּרָכָה — לְרֹאשׁ הַבְּרָכָה וְעַל הַסֵּדֶר ; ״עֲנֵנוּ״ קֹדֶם ״רְאֵה נָא״ — חוֹזֵר (סעיפים ד-ו).

קריאת הרב : הַסֵּדֶר מֵאַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה וּמְעַכֵּב. עיין סימן 124.

מסקנה למעשה : לְעוֹלָם לֹא לוֹמַר בְּרָכָה בּוֹדֶדֶת שֶׁלֹּא בִּמְקוֹמָהּ — לַחֲזֹר לְרֹאשָׁהּ וְלֶאֱמֹר עַל הַסֵּדֶר.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן קי״ט ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן קי״ט (דִּין הָרוֹצֶה לְהוֹסִיף בַּבְּרָכוֹת)

  • ① הוֹסָפָה. לְהוֹסִיף מֵעֵין הַבְּרָכָה : חוֹלֶה בְּ״רְפָאֵנוּ״, פַּרְנָסָה בְּ״בִרְכַּת הַשָּׁנִים״, וְהַכֹּל בְּ״שׁוֹמֵעַ תְּפִלָּה״. יסוד : שוע״ר קי״ט סעיף א.
  • ② סֵדֶר הַתּוֹסֶפֶת. מַתְחִיל בַּבְּרָכָה וְאַחַר כָּךְ מוֹסִיף ; הָעִקָּר מִמַּטְבֵּעַ חֲכָמִים. יסוד : שוע״ר קי״ט סעיף א.
  • ③ יָחִיד וְצִבּוּר. יָחִיד — בְּסוֹף וּבִלְשׁוֹן יָחִיד, לֹא יַאֲרִיךְ ; צִבּוּר — מֻתָּרִים אֲפִלּוּ בְּאֶמְצַע. יסוד : שוע״ר קי״ט סעיף א.
  • ④ וִדּוּי. טוֹב לְהִתְוַדּוֹת בְּ״שׁוֹמֵעַ תְּפִלָּה״ בִּלְשׁוֹן יָחִיד, וְלִשְׁאֹל מְזוֹנוֹתָיו אֲפִלּוּ עָשִׁיר. יסוד : שוע״ר קי״ט סעיף ב.
  • ⑤ שֵׁם הַחוֹלֶה. בְּפָנָיו — אֵין צָרִיךְ לְהַזְכִּיר שְׁמוֹ ; שֶׁלֹּא בְּפָנָיו — צָרִיךְ. יסוד : שוע״ר קי״ט סעיף ג.
  • ⑥ סֵדֶר וַ״עֲנֵנוּ״. דִּלֵּג בְּרָכָה — לְרֹאשׁ הַבְּרָכָה וְעַל הַסֵּדֶר ; ש״ץ שֶׁשָּׁכַח ״עֲנֵנוּ״ אַחַר ״רְפָאֵנוּ״ — בְּ״שׁוֹמֵעַ תְּפִלָּה״ ; קֹדֶם ״רְאֵה נָא״ — חוֹזֵר. יסוד : שוע״ר קי״ט סעיפים ד-ו.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן קי״ט (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַהוֹסָפָה, הַסֵּדֶר, דִּין הַ״עֲנֵנוּ״ — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
