דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן פ״ז · כַּמָּה צָרִיךְ לְהַרְחִיק מִגְּרָף שֶׁל רְעִי

דרכו של אדמו״ר הזקן בכַּמָּה צָרִיךְ לְהַרְחִיק בִּשְׁעַת קְרִיאַת שְׁמַע מִגְּרָף שֶׁל רְעִי : גְּרָף וְעָבִיט — הַרְחָקַת ד׳ אַמּוֹת כְּמִצּוֹאָה ; חֹמֶר הַכְּלִי — חֶרֶס וְעֵץ מוּל מַתֶּכֶת וּזְכוּכִית ; כְּפִיָּה וְכִסּוּי — בָּרוֹ כְּתוֹכוֹ ; וְהַבַּיִת וְהַמְּחִצָּה — מִטָּה, לָבוּד וּכְפִיַּת כְּלִי — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן פ״ז — כַּמָּה צָרִיךְ לְהַרְחִיק מִגְּרָף שֶׁל רְעִי. אצל הרב הסימן נפרש לד׳ סְעִיפִים : הַגְּרָף וְעָבִיט וְהַרְחָקַת ד׳ אַמּוֹת, חֹמֶר הַכְּלִי (חֶרֶס וְעֵץ מוּל מַתֶּכֶת וּזְכוּכִית), הַכְּפִיָּה וְהַכִּסּוּי, וְדִין הַבַּיִת עִם הַמְּחִצָּה, הַלָּבוּד וּכְפִיַּת הַכְּלִי.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן פ״ז — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim פ״ז). אדמו״ר הזקן דן בו בְּכַמָּה צָרִיךְ לְהַרְחִיק מִגְּרָף שֶׁל רְעִי. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן פ״ז — כַּמָּה צָרִיךְ לְהַרְחִיק מִגְּרָף שֶׁל רְעִי

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן פ״ז — כַּמָּה צָרִיךְ לְהַרְחִיק מִגְּרָף שֶׁל רְעִי — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.87), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.87.k).

סעיף א גְּרָף וְעָבִיט — מַרְחִיק ד׳ אַמּוֹת כְּמִצּוֹאָה ; אֵין הֲטָלַת מַיִם מוֹעֶלֶת.

כַּמָּה צָרִיךְ לְהַרְחִיק בִּשְׁעַת קְרִיאַת שְׁמַע מִן גְּרָף שֶׁל רְעִי וּבוֹ ד׳ סְעִיפִים:גְּרָף שֶׁל רְעִי וְעָבִיט שֶׁל מֵי רַגְלַיִם — צָרִיךְ לְהַרְחִיק מֵהֶם ד׳ אַמּוֹת וּלְפָנָיו כִּמְלֹא עֵינָיו כְּמוֹ מִן הַצּוֹאָה וּמֵי רַגְלַיִם עַצְמָן, אַף עַל פִּי שֶׁאֵין בָּהֶן כְּלוּם, וַאֲפִלּוּ אֵין בָּהֶם רֵיחַ רַע, שֶׁכֵּיוָן שֶׁהֵם מְיֻחָדִים לְכָךְ — הֲרֵי הֵם כִּמְחִצּוֹת שֶׁל בֵּית הַכִּסֵּא, שֶׁאָסוּר מִן הַתּוֹרָה מִשּׁוּם ״וְהָיָה מַחֲנֶיךָ קָדוֹשׁ״. וְלָכֵן אֵין הֲטָלַת רְבִיעִית מַיִם אוֹ יוֹתֵר מוֹעֶלֶת כְּלוּם בִּגְרָף שֶׁל רְעִי וְעָבִיט כְּמוֹ שֶׁמּוֹעֶלֶת בְּמֵי רַגְלַיִם עַצְמָן, לְפִי שֶׁמֵּי רַגְלַיִם לְאַחַר שֶׁנָּפְלוּ לָאָרֶץ — אֵין אִסּוּרָן אֶלָּא מִדִּבְרֵי סוֹפְרִים, לְפִיכָךְ הֵקֵלּוּ בָּהֶן לְבַטְּלָם בַּמַּיִם, מַה שֶּׁאֵין כֵּן בִּגְרָף וְעָבִיט שֶׁאִסּוּרָן מִן הַתּוֹרָה. וְעוֹד שֶׁמֵּי רַגְלַיִם הֵם צְלוּלִין וּמִתְעָרְבִין עִם רְבִיעִית מַיִם שֶׁנּוֹתֵן בָּהֶם וּמִתְבַּטְּלִים בָּהֶם, אֲבָל גְּרָף וְעָבִיט אֵין מַה שֶּׁבָּלוּעַ בָּהֶם מִתְעָרֵב בַּמַּיִם שֶׁיִּתְבַּטֵּל בָּהֶן:

גְּרָף וְעָבִיט. גְּרָף שֶׁל רְעִי וְעָבִיט שֶׁל מֵי רַגְלַיִם — מַרְחִיק מֵהֶם ד׳ אַמּוֹת וּלְפָנָיו כִּמְלֹא עֵינָיו, כְּמוֹ מִן הַצּוֹאָה וּמֵי רַגְלַיִם עַצְמָן, אַף שֶׁהֵם רֵיקִים וַאֲפִלּוּ בְּאֵין רֵיחַ רַע ; שֶׁכֵּיוָן שֶׁהֵם מְיֻחָדִים לְכָךְ — הֲרֵי הֵם כִּמְחִצּוֹת בֵּית הַכִּסֵּא, אִסּוּר מִן הַתּוֹרָה (״וְהָיָה מַחֲנֶיךָ קָדוֹשׁ״). לְפִיכָךְ אֵין הֲטָלַת רְבִיעִית מַיִם מוֹעֶלֶת בָּהֶם — בְּנִגּוּד לְמֵי רַגְלַיִם עַצְמָן, שֶׁאִסּוּרָן מִדִּבְרֵי סוֹפְרִים ; וְעוֹד, שֶׁמֵּי רַגְלַיִם צְלוּלִין וּמִתְעָרְבִין וּמִתְבַּטְּלִים בַּמַּיִם, אֲבָל הַבָּלוּעַ בַּגְּרָף וּבֶעָבִיט אֵינוֹ מִתְעָרֵב בַּמַּיִם.

סעיף ב חֶרֶס וְעֵץ — מֻבְלָע וּמָאוּס ; מַתֶּכֶת וּזְכוּכִית רְחוּצִים — מֻתָּר.

גְּרָף וְעָבִיט שְׁנֵיהֶם שֶׁל חֶרֶס הֵם, אֶלָּא שֶׁהַמְיֻחָד לִרְעִי — נִקְרָא גְּרָף, וְהַמְיֻחָד לְמֵי רַגְלַיִם — נִקְרָא עָבִיט. וְהוּא הַדִּין אִם הֵם שֶׁל עֵץ, מִפְּנֵי שֶׁהַחֶרֶס אוֹ עֵץ שֶׁל גְּרָף וְעָבִיט הוּא מֻבְלָע בְּתוֹכוֹ הַרְבֵּה וּמָלֵא זֻהֲמָא וּמָאוּס כְּבֵית הַכִּסֵּא. אֲבָל אִם הֵם שֶׁל מַתֶּכֶת אוֹ שֶׁל זְכוּכִית אוֹ שֶׁל חֶרֶס מְצֻפֶּה בִּזְכוּכִית (אוֹ בִּבְדִיל וְעוֹפֶרֶת), כֵּיוָן שֶׁאֵינָן מֻבְלָעִים וּמְלֵאִים זֻהֲמָא כָּל כָּךְ — מֻתָּר לִקְרוֹת כְּנֶגְדָּן אִם הֵם רְחוּצִים יָפֶה, וּבִלְבַד שֶׁלֹּא יִהְיֶה בָּהֶן רֵיחַ. וַאֲפִלּוּ שֶׁל חֶרֶס אוֹ שֶׁל עֵץ, אִם הֵם מְיֻחָדִים גַּם כֵּן לְתַשְׁמִישׁ אַחֵר, כְּגוֹן לִשְׁפֹּךְ לְתוֹכָן שׁוֹפְכִין — אֵינָן דּוֹמִין לְבֵית הַכִּסֵּא, וְאִם אֵין בָּהֶם צוֹאָה וְלֹא רֵיחַ — מֻתָּר. וְהוּא הַדִּין אֲפִלּוּ כְּלִי הַמְיֻחָד לְהַשְׁתִּין בִּלְבַד אֶלָּא שֶׁבְּכָל פַּעַם שֶׁמַּשְׁתִּין נוֹתֵן בְּתוֹכוֹ רְבִיעִית מַיִם לְבַטְּלָם:

חֹמֶר הַכְּלִי. גְּרָף וְעָבִיט שְׁנֵיהֶם שֶׁל חֶרֶס — הַמְיֻחָד לִרְעִי נִקְרָא גְּרָף, וְהַמְיֻחָד לְמֵי רַגְלַיִם נִקְרָא עָבִיט ; וְהוּא הַדִּין שֶׁל עֵץ, שֶׁהַחֶרֶס אוֹ הָעֵץ מֻבְלָע וּמָלֵא זֻהֲמָא וּמָאוּס כְּבֵית הַכִּסֵּא. אֲבָל שֶׁל מַתֶּכֶת, זְכוּכִית אוֹ חֶרֶס מְצֻפֶּה — שֶׁאֵינָן מֻבְלָעִים — מֻתָּר לִקְרוֹת כְּנֶגְדָּן אִם רְחוּצִים יָפֶה וּבְלֹא רֵיחַ. וַאֲפִלּוּ חֶרֶס אוֹ עֵץ הַמְיֻחָד גַּם לְתַשְׁמִישׁ אַחֵר (שׁוֹפְכִין) — אֵינוֹ כְּבֵית הַכִּסֵּא, וּבְאֵין צוֹאָה וְרֵיחַ — מֻתָּר ; וְהוּא הַדִּין כְּלִי הַמְיֻחָד לְהַשְׁתִּין שֶׁנּוֹתֵן בּוֹ בְּכָל פַּעַם רְבִיעִית מַיִם.

סעיף ג כָּפָה עֲלֵיהֶם כְּלִי — מֻתָּר ; כָּפָה הַגְּרָף עַצְמוֹ — לֹא הוֹעִיל.

גְּרָף וְעָבִיט שֶׁכָּפָה עֲלֵיהֶם כְּלִי אַחֵר — מֻתָּר, שֶׁכִּסּוּי מוֹעִיל לָהֶם כְּמוֹ לְצוֹאָה וּמֵי רַגְלַיִם. אֲבָל אִם כָּפָה הַגְּרָף אוֹ הֶעָבִיט עַצְמוֹ — לֹא הוֹעִיל כְּלוּם, שֶׁכֵּיוָן שֶׁהוּא בָּלוּעַ בְּתוֹכוֹ — הֲרֵי בָּרוֹ כְּתוֹכוֹ. אֲבָל שֶׁל מַתֶּכֶת וּזְכוּכִית שֶׁאֵינוֹ רָחוּץ מִתּוֹכוֹ — מוֹעֶלֶת לוֹ כְּפִיָּה, כֵּיוָן שֶׁאֵין אִסּוּרוֹ אֶלָּא מִתּוֹכוֹ:

כְּפִיָּה וְכִסּוּי. גְּרָף וְעָבִיט שֶׁכָּפָה עֲלֵיהֶם כְּלִי אַחֵר — מֻתָּר, שֶׁהַכִּסּוּי מוֹעִיל לָהֶם כְּמוֹ לְצוֹאָה וּמֵי רַגְלַיִם. אֲבָל כָּפָה הַגְּרָף אוֹ הֶעָבִיט עַצְמוֹ — לֹא הוֹעִיל, שֶׁכֵּיוָן שֶׁהוּא בָּלוּעַ בְּתוֹכוֹ — בָּרוֹ כְּתוֹכוֹ. וְשֶׁל מַתֶּכֶת אוֹ זְכוּכִית שֶׁאֵינוֹ רָחוּץ מִתּוֹכוֹ — מוֹעֶלֶת לוֹ כְּפִיָּה, שֶׁאֵין אִסּוּרוֹ אֶלָּא מִתּוֹכוֹ.

סעיף ד בַּיִת — מַרְחִיק ד׳ אַמּוֹת ; מִטָּה גְּבוֹהָה י׳ טְפָחִים כִּמְחִצָּה ; פָּחוֹת מִג׳ טְפָחִים כְּלָבוּד.

בַּיִת שֶׁיֵּשׁ בּוֹ צוֹאָה אוֹ מֵי רַגְלַיִם אוֹ גְּרָף אוֹ עָבִיט — מֻתָּר לִקְרוֹת בּוֹ כְּשֶׁמַּרְחִיק מֵהֶם ד׳ אַמּוֹת מִלְּאַחֲרָיו, וְאֵין אוֹמְרִים שֶׁכָּל הַבַּיִת הוּא כְּדִי[ן] ד׳ אַמּוֹת. וְאִם יֵשׁ בַּבַּיִת מִטָּה גָּבוֹהַּ י׳ טְפָחִים וּמְחִצּוֹתֶיהָ מַגִּיעוֹת (לְפָחוֹת מִג׳ טְפָחִים סָמוּךְ) לָאָרֶץ וְהַגְּרָף וְהַצּוֹאָה הֵם אֲחוֹרֵי הַמִּטָּה — הֲרֵי הַמִּטָּה חֲשׁוּבָה כִּמְחִצָּה לְהַפְסִיק בֵּינוֹ לְבֵינֵיהֶם, וְאֵין צָרִיךְ לְהַרְחִיק מֵהֶם ד׳ אַמּוֹת. וְהוּא שֶׁתִּהְיֶה הַמִּטָּה חוֹצֶצֶת מִכֹּתֶל לְכֹתֶל, וְאִם לָאו — אֵינָהּ חֲשׁוּבָה כִּמְחִצָּה, שֶׁהִיא בְּטֵלָה לְגַבֵּי הַבַּיִת (אֲבָל אִם עָשָׂה מְחִצָּה לְשֵׁם כָּךְ — אֵין צָרִיךְ לַחְצֹץ מִכֹּתֶל לְכֹתֶל, אֲפִלּוּ הִיא מְחִצַּת עֲרַאי כְּגוֹן מַחְצֶלֶת אוֹ סָדִין) (עַיֵּן סִימָן שט״ו). וְאִם הִנִּיחָם תַּחַת הַמִּטָּה אַף עַל פִּי שֶׁאֵינוֹ גָּבוֹהַּ י׳ טְפָחִים, אִם מְחִצּוֹתֶיהָ מַגִּיעוֹת עַד פָּחוֹת מִג׳ טְפָחִים סָמוּךְ לָאָרֶץ, שֶׁכִּסְתוּמִים וּמְכֻסִּים הֵם בִּמְחִצּוֹת הַמִּטָּה, שֶׁכָּל פָּחוֹת מִג׳ טְפָחִים כְּלָבוּד הוּא. וְהוּא הַדִּין אִם נָפַל עֲלֵיהֶם כְּלִי וְיֵשׁ אֲוִיר קְצָת בֵּין הַכְּלִי לַקַּרְקַע, כְּגוֹן שֶׁהַכְּלִי בּוֹלֵט קְצָת מִצַּד אֶחָד, אִם הוּא אֲוִיר פָּחוֹת מִג׳ טְפָחִים בַּגֹּבַהּ — כְּלָבוּד הוּא. אֲבָל אִם הַכְּלִי פָּתוּחַ לְמַעְלָה אֲפִלּוּ פָּחוֹת מִג׳ טְפָחִים — אָסוּר, מִפְּנֵי שֶׁצָּרִיךְ שֶׁיִּהְיֶה מְכֻסִּים וְלֹא מְגֻלִּים:

בַּיִת וּמְחִצָּה. בַּיִת שֶׁיֵּשׁ בּוֹ צוֹאָה, מֵי רַגְלַיִם, גְּרָף אוֹ עָבִיט — מֻתָּר לִקְרוֹת בּוֹ כְּשֶׁמַּרְחִיק ד׳ אַמּוֹת מִלְּאַחֲרָיו, וְאֵין כָּל הַבַּיִת כְּדִין ד׳ אַמּוֹת. מִטָּה גָּבוֹהַּ י׳ טְפָחִים שֶׁמְּחִצּוֹתֶיהָ מַגִּיעוֹת עַד פָּחוֹת מִג׳ טְפָחִים סָמוּךְ לָאָרֶץ וְהַגְּרָף אֲחוֹרֵי הַמִּטָּה — חֲשׁוּבָה כִּמְחִצָּה לְהַפְסִיק, וְאֵין צָרִיךְ לְהַרְחִיק ; וּבִלְבַד שֶׁחוֹצֶצֶת מִכֹּתֶל לְכֹתֶל (וְאִם עָשָׂה מְחִצָּה לְשֵׁם כָּךְ — אַף מְחִצַּת עֲרַאי כְּמַחְצֶלֶת אוֹ סָדִין, עַיֵּן סִימָן שט״ו). הִנִּיחָם תַּחַת הַמִּטָּה, אַף שֶׁאֵינָהּ גְּבוֹהָה י׳ טְפָחִים — כָּל פָּחוֹת מִג׳ טְפָחִים כְּלָבוּד. וְהוּא הַדִּין כְּלִי שֶׁנָּפַל עֲלֵיהֶם וְיֵשׁ אֲוִיר פָּחוֹת מִג׳ טְפָחִים — כְּלָבוּד ; אֲבָל אִם הַכְּלִי פָּתוּחַ לְמַעְלָה, אֲפִלּוּ פָּחוֹת מִג׳ טְפָחִים — אָסוּר, שֶׁצָּרִיךְ שֶׁיִּהְיוּ מְכֻסִּים וְלֹא מְגֻלִּים.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim פ״ז (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — כַּמָּה צָרִיךְ לְהַרְחִיק מִגְּרָף שֶׁל רְעִי

היסוד : הַרְחָקָה מִגְּרָף שֶׁל רְעִי

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַגְּרָף וְעָבִיט וְהַרְחָקַת ד׳ אַמּוֹת. אחר כך חֹמֶר הַכְּלִי — חֶרֶס וְעֵץ מוּל מַתֶּכֶת וּזְכוּכִית. אחר כך הַכְּפִיָּה וְהַכִּסּוּי. ולבסוף דִּין הַבַּיִת עִם הַמְּחִצָּה, הַלָּבוּד וּכְפִיַּת הַכְּלִי.

גְּרָף וְעָבִיט — הַרְחָקַת ד׳ אַמּוֹת

גְּרָף וְעָבִיט — הַרְחָקַת ד׳ אַמּוֹת

היסוד. גְּרָף וְעָבִיט — מַרְחִיק ד׳ אַמּוֹת וּלְפָנָיו כִּמְלֹא עֵינָיו כְּמוֹ מִן הַצּוֹאָה, אַף בְּאֵין רֵיחַ, שֶׁהֵם כִּמְחִצּוֹת בֵּית הַכִּסֵּא — אִסּוּר מִן הַתּוֹרָה (סעיף א).

המסקנה. הֲטָלַת רְבִיעִית מַיִם אֵינָהּ מוֹעֶלֶת בָּהֶם (בְּנִגּוּד לְמֵי רַגְלַיִם עַצְמָן, שֶׁאִסּוּרָן מִדִּבְרֵי סוֹפְרִים). עיין סימן 79 (הַרְחָקָה מִצּוֹאָה).

חֹמֶר הַכְּלִי — חֶרֶס וְעֵץ מוּל מַתֶּכֶת

חֹמֶר הַכְּלִי — חֶרֶס וְעֵץ מוּל מַתֶּכֶת

היסוד. חֶרֶס אוֹ עֵץ — מֻבְלָע וּמָלֵא זֻהֲמָא, מָאוּס כְּבֵית הַכִּסֵּא ; מַתֶּכֶת, זְכוּכִית אוֹ חֶרֶס מְצֻפֶּה — רְחוּצִים יָפֶה וּבְלֹא רֵיחַ — מֻתָּר (סעיף ב).

המסקנה. כְּלִי הַמְיֻחָד גַּם לְתַשְׁמִישׁ אַחֵר (שׁוֹפְכִין) — אֵינוֹ כְּבֵית הַכִּסֵּא. עיין סימן 76 (הֲכָנַת הַמָּקוֹם וְהַנְּקִיּוּת).

כְּפִיָּה וְכִסּוּי — בָּרוֹ כְּתוֹכוֹ

כְּפִיָּה וְכִסּוּי — בָּרוֹ כְּתוֹכוֹ

היסוד. כָּפָה עֲלֵיהֶם כְּלִי אַחֵר — מֻתָּר ; כָּפָה הַגְּרָף עַצְמוֹ (חֶרֶס/עֵץ) — לֹא הוֹעִיל, שֶׁבָּרוֹ כְּתוֹכוֹ (סעיף ג).

המסקנה. שֶׁל מַתֶּכֶת וּזְכוּכִית שֶׁאֵין אִסּוּרוֹ אֶלָּא מִתּוֹכוֹ — מוֹעֶלֶת לוֹ כְּפִיָּה. עיין סימן 83 (כִּסּוּי וְהַרְחָקָה).

בַּיִת וּמְחִצָּה — מִטָּה, לָבוּד וּכְפִיַּת כְּלִי

בַּיִת וּמְחִצָּה — מִטָּה, לָבוּד וּכְפִיַּת כְּלִי

היסוד. בַּיִת — מַרְחִיק ד׳ אַמּוֹת מִלְּאַחֲרָיו ; מִטָּה גְּבוֹהָה י׳ טְפָחִים חוֹצֶצֶת מִכֹּתֶל לְכֹתֶל — כִּמְחִצָּה (סעיף ד).

המסקנה. תַּחַת הַמִּטָּה אוֹ כְּלִי שֶׁנָּפַל — פָּחוֹת מִג׳ טְפָחִים כְּלָבוּד ; פָּתוּחַ לְמַעְלָה — אָסוּר. עיין סימן 315 (מְחִצָּה בְּשַׁבָּת).

הפניה ללימוד. היקף הסימן — הַרְחָקַת הַגְּרָף, חֹמֶר הַכְּלִי, הַכְּפִיָּה וְהַמְּחִצָּה — נוגע להנהגת ק״ש היומיומית בְּמָקוֹם שֶׁיֵּשׁ בּוֹ גְּרָף. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

כַּמָּה צָרִיךְ לְהַרְחִיק מִגְּרָף שֶׁל רְעִי — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן פ״ז. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הַרְחָקַת גְּרָף וְעָבִיט או״ח פ״ז — גְּרָף וְעָבִיט שֶׁל חֶרֶס אוֹ עֵץ צָרִיךְ לְהַרְחִיק כְּמִצּוֹאָה, אֲפִלּוּ הֵטִיל בָּהֶם מַיִם. מ״ב פ״ז — טַעַם הַהַרְחָקָה וְשִׁעוּר ד׳ אַמּוֹת. ערוה״ש פ״ז — גֶּדֶר הַכְּלִי הַמְיֻחָד לִרְעִי. דרך אדמו״ר הזקן — מַרְחִיק ד׳ אַמּוֹת כְּמִן הַצּוֹאָה, שֶׁהֵם כִּמְחִצּוֹת בֵּית הַכִּסֵּא — אִסּוּר מִן הַתּוֹרָה ; אֵין הֲטָלַת מַיִם מוֹעֶלֶת (סעיף א).
חֹמֶר הַכְּלִי או״ח פ״ז — שֶׁל מַתֶּכֶת אוֹ זְכוּכִית אוֹ חֶרֶס מְצֻפֶּה — מֻתָּר אִם רְחוּצִים יָפֶה. מ״ב פ״ז — תְּנַאי הָרְחִיצָה וְהֶעְדֵּר הָרֵיחַ. ערוה״ש פ״ז — כְּלִי הַמְיֻחָד גַּם לְתַשְׁמִישׁ אַחֵר. דרך אדמו״ר הזקן — חֶרֶס וְעֵץ מֻבְלָעִים וּמָאוּס ; מַתֶּכֶת וּזְכוּכִית רְחוּצִים בְּלֹא רֵיחַ — מֻתָּר ; מְיֻחָד גַּם לְשׁוֹפְכִין — אֵינוֹ כְּבֵית הַכִּסֵּא (סעיף ב).
כְּפִיָּה עַל פִּי הַכְּלִי או״ח פ״ז — גְּרָף וְעָבִיט שֶׁכְּפָאוֹ עַל פִּיו — יֵשׁ מַתִּירִים וְיֵשׁ אוֹסְרִים, וַהֲלָכָה כָּאוֹסְרִים. מ״ב פ״ז — טַעַם הָאוֹסְרִים בִּכְפִיַּת הַכְּלִי עַצְמוֹ. ערוה״ש פ״ז — חִלּוּק בֵּין כִּסּוּי בִּכְלִי אַחֵר לִכְפִיַּת עַצְמוֹ. דרך אדמו״ר הזקן — כָּפָה כְּלִי אַחֵר — מֻתָּר ; כָּפָה הַגְּרָף עַצְמוֹ — לֹא הוֹעִיל (בָּרוֹ כְּתוֹכוֹ) ; מַתֶּכֶת שֶׁאֵין אִסּוּרוֹ אֶלָּא מִתּוֹכוֹ — מוֹעֶלֶת כְּפִיָּה (סעיף ג).
בַּיִת וּמְחִצָּה או״ח פ״ז — מֻתָּר לִקְרוֹת בְּבַיִת שֶׁהִרְחִיק כַּשִּׁעוּר, אוֹ כָּפָה עֲלֵיהֶם כְּלִי — כִּקְבוּרִים. מ״ב פ״ז — שִׁעוּר הַהַרְחָקָה וְדִין הַמִּטָּה כִּמְחִצָּה. ערוה״ש פ״ז — גֶּדֶר לָבוּד וּמְחִצָּה בַּבַּיִת. דרך אדמו״ר הזקן — מַרְחִיק ד׳ אַמּוֹת מִלְּאַחֲרָיו ; מִטָּה גְּבוֹהָה י׳ טְפָחִים חוֹצֶצֶת מִכֹּתֶל לְכֹתֶל — כִּמְחִצָּה ; פָּחוֹת מִג׳ טְפָחִים כְּלָבוּד ; פָּתוּחַ לְמַעְלָה — אָסוּר (סעיף ד).

הטבלה נערכה מטקסט המחבר (אורח חיים פ״ז, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — כִּמְחִצּוֹת בֵּית הַכִּסֵּא

קו א — כִּמְחִצּוֹת בֵּית הַכִּסֵּא

יסוד הסימן : גְּרָף וְעָבִיט מְיֻחָדִים — כִּמְחִצּוֹת בֵּית הַכִּסֵּא, אִסּוּר מִן הַתּוֹרָה (סעיף א).

קריאת הרב : הַחִלּוּק בֵּין תּוֹרָה לִדְרַבָּנָן מְבָאֵר לָמָּה הֲטָלַת מַיִם אֵינָהּ מוֹעֶלֶת בַּגְּרָף כְּמוֹ בְּמֵי רַגְלַיִם. עיין סימן 79.

מסקנה למעשה : לְהַרְחִיק ד׳ אַמּוֹת אַף שֶׁהַכְּלִי רֵיק וּבְלֹא רֵיחַ.

קו ב — חֹמֶר וְזֻהֲמָא

קו ב — חֹמֶר וְזֻהֲמָא

יסוד הסימן : חֶרֶס וְעֵץ מֻבְלָעִים וּמָאוּס ; מַתֶּכֶת וּזְכוּכִית — לֹא (סעיף ב).

קריאת הרב : הַמַּדָּד הוּא הַבְּלִיעָה וְהַזֻּהֲמָא, לֹא צוּרַת הַכְּלִי ; כְּלִי הַמְיֻחָד גַּם לְאַחֵר יוֹצֵא מִגֶּדֶר בֵּית הַכִּסֵּא. עיין סימן 76.

מסקנה למעשה : כְּלִי מְצֻפֶּה אוֹ שֶׁל מַתֶּכֶת רָחוּץ יָפֶה וּבְלֹא רֵיחַ — מֻתָּר לִקְרוֹת כְּנֶגְדּוֹ.

קו ג — כִּסּוּי, לָבוּד וּמְחִצָּה

קו ג — כִּסּוּי, לָבוּד וּמְחִצָּה

יסוד הסימן : כָּפָה כְּלִי אַחֵר — מֻתָּר ; מִטָּה כִּמְחִצָּה ; פָּחוֹת מִג׳ טְפָחִים כְּלָבוּד (סעיפים ג-ד).

קריאת הרב : הַיְסוֹד הוּא לְהַפְסִיק אוֹ לְכַסּוֹת אֶת הַחֵפֶץ, לֹא רַק לְהַרְחִיקוֹ ; מִכָּאן כֹּחַ הַכִּסּוּי וְהַלָּבוּד. עיין סימן 315.

מסקנה למעשה : כִּסּוּי בִּכְלִי אַחֵר אוֹ הַפְסָקַת מְחִצָּה — פּוֹטֶרֶת מֵהַרְחָקַת ד׳ אַמּוֹת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן פ״ז ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן פ״ז (כַּמָּה צָרִיךְ לְהַרְחִיק מִגְּרָף שֶׁל רְעִי)

  • ① הַרְחָקָה. גְּרָף וְעָבִיט — לְהַרְחִיק ד׳ אַמּוֹת וּלְפָנָיו כִּמְלֹא עֵינָיו, אַף רֵיקִים וּבְלֹא רֵיחַ (כְּמִן הַצּוֹאָה). יסוד : שוע״ר פ״ז סעיף א.
  • ② הֲטָלַת מַיִם. הֲטָלַת מַיִם אֵינָהּ מוֹעֶלֶת בַּגְּרָף וּבֶעָבִיט (אִסּוּר מִן הַתּוֹרָה), בְּנִגּוּד לְמֵי רַגְלַיִם עַצְמָן. יסוד : שוע״ר פ״ז סעיף א.
  • ③ חֹמֶר הַכְּלִי. מַתֶּכֶת, זְכוּכִית אוֹ חֶרֶס מְצֻפֶּה רְחוּצִים יָפֶה וּבְלֹא רֵיחַ — מֻתָּר ; חֶרֶס וְעֵץ מֻבְלָעִים — אָסוּר. יסוד : שוע״ר פ״ז סעיף ב.
  • ④ כְּפִיָּה. כָּפָה כְּלִי אַחֵר עֲלֵיהֶם — מֻתָּר ; כָּפָה הַגְּרָף עַצְמוֹ (חֶרֶס/עֵץ) — לֹא הוֹעִיל (בָּרוֹ כְּתוֹכוֹ). יסוד : שוע״ר פ״ז סעיף ג.
  • ⑤ בַּיִת וּמְחִצָּה. בַּבַּיִת — מַרְחִיק ד׳ אַמּוֹת מִלְּאַחֲרָיו ; מִטָּה גְּבוֹהָה י׳ טְפָחִים חוֹצֶצֶת מִכֹּתֶל לְכֹתֶל, אוֹ כִּסּוּי פָּחוֹת מִג׳ טְפָחִים (לָבוּד) — פּוֹטֶרֶת. יסוד : שוע״ר פ״ז סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן פ״ז (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַהַרְחָקָה, חֹמֶר הַכְּלִי, הַכְּפִיָּה וְהַמְּחִצָּה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
