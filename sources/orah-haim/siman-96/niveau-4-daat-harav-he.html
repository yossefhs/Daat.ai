דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן צ״ו · הֲסָרַת הַטְּרָדוֹת כְּדֵי שֶׁיְּכַוֵּן

דרכו של אדמו״ר הזקן בְּדִּין הֲסָרַת הַטְּרָדוֹת בַּתְּפִלָּה : לֹא יֶאֱחֹז מַה שֶּׁלִּבּוֹ עָלָיו — פֶּן יִפֹּל וְתִתְבַּטֵּל הַכַּוָּנָה ; הַסִּדּוּר מֻתָּר לְצֹרֶךְ הַתְּפִלָּה, אֲבָל אֵין חוֹזְרִים אַחֲרָיו (הֶפְסֵק) ; סֵפֶר שֶׁנָּפַל — מַגְבִּיהוֹ בְּסוֹף הַבְּרָכָה — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן צ״ו — הֲסָרַת הַטְּרָדוֹת כְּדֵי שֶׁיְּכַוֵּן. אצל הרב הסימן נפרש לג׳ סְעִיפִים : הַמַּה שֶּׁלִּבּוֹ עָלָיו וְהַלּוּלָב, הַסִּדּוּר וְהַהֶפְסֵק, וְהַסֵּפֶר שֶׁנָּפַל הַמּוֹנֵעַ הַכַּוָּנָה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן צ״ו — ג׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim צ״ו). אדמו״ר הזקן דן בו בַּהֲסָרַת הַטְּרָדוֹת כְּדֵי שֶׁיְּכַוֵּן. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן צ״ו — הֲסָרַת הַטְּרָדוֹת כְּדֵי שֶׁיְּכַוֵּן

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן צ״ו — הֲסָרַת הַטְּרָדוֹת כְּדֵי שֶׁיְּכַוֵּן — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.96), ואחריו תרגומנו. הסימן כולל ג׳ סעיפים (סעיף k → ‎.96.k).

סעיף א לֹא יֶאֱחֹז מַה שֶּׁלִּבּוֹ עָלָיו ; לוּלָב מֻתָּר שֶׁאֲחִיזָתוֹ מִצְוָה.

שֶׁיִּמְנַע כָּל הַטְּרָדוֹת כְּדֵי שֶׁיְּכַוֵּן וּבוֹ ג׳ סְעִיפִים:כְּשֶׁהוּא מִתְפַּלֵּל לֹא יֶאֱחֹז בְּיָדוֹ תְּפִלִּין, וְלֹא סֵפֶר מִכִּתְבֵי הַקֹּדֶשׁ, וְלֹא קְעָרָה מְלֵאָה, וְלֹא סַכִּין, וְלֹא מָעוֹת, וְלֹא כִּכָּר, מִפְּנֵי שֶׁלִּבּוֹ עֲלֵיהֶם שֶׁלֹּא יִפֹּל וְיִטָּרֵד וְתִתְבַּטֵּל כַּוָּנָתוֹ. אֲבָל שְׁאָר דְּבָרִים שֶׁאִם יִפְּלוּ לֹא יִפָּסְדוּ וְלֹא יַזִּיקוּ לוֹ — מֻתָּר לֶאֱחֹז מִן הַדִּין. אֶלָּא שֶׁמִּצְוָה מִן הַמֻּבְחָר שֶׁלֹּא לֶאֱחֹז כְּלוּם, אֶלָּא יִהְיוּ יָדָיו כְּפוּתִין כְּעוֹמֵד לִפְנֵי הַמֶּלֶךְ, כְּמוֹ שֶׁנִּתְבָּאֵר סִימָן צ״ה סָעִיף ד׳. וְלוּלָב בִּזְמַנּוֹ מֻתָּר לֶאֱחֹז, אַף עַל פִּי שֶׁאִם יִפֹּל אֶפְשָׁר שֶׁיִּפָּסֵל, מִכָּל מָקוֹם כֵּיוָן שֶׁהָאֲחִיזָה הִיא מִצְוָה — אֵינוֹ נִטְרָד בִּשְׁבִילוֹ:

מַה שֶּׁלִּבּוֹ עָלָיו. בְּשָׁעָה שֶׁמִּתְפַּלֵּל לֹא יֶאֱחֹז תְּפִלִּין, סֵפֶר מִכִּתְבֵי הַקֹּדֶשׁ, קְעָרָה מְלֵאָה, סַכִּין, מָעוֹת אוֹ כִּכָּר, מִפְּנֵי שֶׁלִּבּוֹ עֲלֵיהֶם (שֶׁמָּא יִפְּלוּ) — יִטָּרֵד וְתִתְבַּטֵּל כַּוָּנָתוֹ. אֲבָל שְׁאָר דְּבָרִים שֶׁאֵין נִפְסָדִים וְאֵין מַזִּיקִים — מֻתָּר לֶאֱחֹז מִן הַדִּין ; אֶלָּא שֶׁמִּצְוָה מִן הַמֻּבְחָר שֶׁלֹּא לֶאֱחֹז כְּלוּם, יָדָיו כְּפוּתִין כְּעוֹמֵד לִפְנֵי הַמֶּלֶךְ (סימן צ״ה סעיף ד׳). וְלוּלָב בִּזְמַנּוֹ — מֻתָּר, שֶׁאַף שֶׁאֶפְשָׁר יִפָּסֵל בִּנְפִילָה, כֵּיוָן שֶׁהָאֲחִיזָה הִיא מִצְוָה — אֵינוֹ נִטְרָד.

סעיף ב מֻתָּר לֶאֱחֹז סִדּוּר לְצֹרֶךְ הַתְּפִלָּה ; לֹא יַחֲזֹר אַחֲרָיו מִשּׁוּם הֶפְסֵק.

מֻתָּר לֶאֱחֹז סִדּוּר בְּיָדוֹ בְּשָׁעָה שֶׁמִּתְפַּלֵּל, שֶׁכֵּיוָן שֶׁאוֹחֲזוֹ לְצֹרֶךְ הַתְּפִלָּה — אֵינוֹ נִטְרָד בִּשְׁבִילוֹ. וּמִכָּל מָקוֹם, אִם לֹא הָיָה בְּיָדוֹ קֹדֶם שֶׁהִתְחִיל — לֹא יַחֲזֹר אַחֲרָיו בַּתְּפִלָּה לִטְּלוֹ, מִשּׁוּם הֶפְסֵק. אֶלָּא אִם כֵּן הָיָה בְּמָקוֹם מְיֻחָד שֶׁהוּא מוּכָן אָז מֻתָּר לִטְּלוֹ אֲפִלּוּ בְּתוֹךְ הַתְּפִלָּה, וְהוּא שֶׁנּוֹטְלוֹ כְּדֵי לְהִתְפַּלֵּל מִתּוֹכוֹ. וְכָל הַמִּתְפַּלֵּל מִתּוֹךְ הַסִּדּוּר — יִזָּהֵר שֶׁיִּרְשֹׁם תְּחִלָּה כָּל הַמְּקוֹמוֹת שֶׁצָּרִיךְ לְהִתְפַּלֵּל בְּאוֹתָהּ תְּפִלָּה, וְלֹא יְחַפֵּשׂ אַחֲרֵיהֶם בְּתוֹךְ הַתְּפִלָּה, כִּי יַפְסִיק עַל יְדֵי זֶה וְלֹא יוּכַל לְכַוֵּן יָפֶה:

הַסִּדּוּר וְהַהֶפְסֵק. מֻתָּר לֶאֱחֹז סִדּוּר בַּתְּפִלָּה, שֶׁכֵּיוָן שֶׁאוֹחֲזוֹ לְצֹרֶךְ הַתְּפִלָּה — אֵינוֹ נִטְרָד. וּמִכָּל מָקוֹם, אִם לֹא הָיָה בְּיָדוֹ קֹדֶם — לֹא יַחֲזֹר אַחֲרָיו בַּתְּפִלָּה מִשּׁוּם הֶפְסֵק ; אֶלָּא אִם כֵּן הָיָה בְּמָקוֹם מְיֻחָד מוּכָן, אָז מֻתָּר לִטְּלוֹ אֲפִלּוּ בְּתוֹךְ הַתְּפִלָּה כְּדֵי לְהִתְפַּלֵּל מִתּוֹכוֹ. וְכָל הַמִּתְפַּלֵּל מִתּוֹךְ הַסִּדּוּר — יִזָּהֵר לִרְשֹׁם תְּחִלָּה כָּל הַמְּקוֹמוֹת, וְלֹא יְחַפֵּשׂ בְּתוֹךְ הַתְּפִלָּה, כִּי יַפְסִיק וְלֹא יוּכַל לְכַוֵּן יָפֶה.

סעיף ג נָפַל סֵפֶר לְפָנָיו — מֻתָּר לְהַגְבִּיהוֹ כְּשֶׁיְּסַיֵּם הַבְּרָכָה.

נָפַל סֵפֶר לְפָנָיו עַל הָאָרֶץ וְאֵינוֹ יָכוֹל לְכַוֵּן יָפֶה מִפְּנֵי זֶה — מֻתָּר לְהַגְבִּיהוֹ כְּשֶׁיְּסַיֵּם אוֹתָהּ בְּרָכָה שֶׁהוּא עוֹמֵד בָּהּ:

סֵפֶר שֶׁנָּפַל. נָפַל סֵפֶר לְפָנָיו עַל הָאָרֶץ וְאֵינוֹ יָכוֹל לְכַוֵּן יָפֶה מִפְּנֵי זֶה — מֻתָּר לְהַגְבִּיהוֹ כְּשֶׁיְּסַיֵּם אוֹתָהּ בְּרָכָה שֶׁהוּא עוֹמֵד בָּהּ.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim צ״ו (מהדורת קה״ת, ג׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — הֲסָרַת הַטְּרָדוֹת כְּדֵי שֶׁיְּכַוֵּן

היסוד : הֲסָרַת הַטְּרָדוֹת לְמַעַן הַכַּוָּנָה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים. תחילה הַמַּה שֶּׁלִּבּוֹ עָלָיו — מַה שֶּׁאֵין אוֹחֲזִים פֶּן יִפֹּל — וְהַלּוּלָב. אחר כך הַסִּדּוּר הַמֻּתָּר לְצֹרֶךְ הַתְּפִלָּה וְהַהֶפְסֵק. ולבסוף הַסֵּפֶר שֶׁנָּפַל הַמּוֹנֵעַ הַכַּוָּנָה.

מַה שֶּׁלִּבּוֹ עָלָיו — הַחֲשָׁשׁ שֶׁמָּא יִפֹּל

מַה שֶּׁלִּבּוֹ עָלָיו — הַחֲשָׁשׁ שֶׁמָּא יִפֹּל

היסוד. אֵין אוֹחֲזִין מַה שֶּׁלִּבּוֹ עָלָיו (תְּפִלִּין, סֵפֶר, קְעָרָה מְלֵאָה, סַכִּין, מָעוֹת, כִּכָּר), פֶּן יִפֹּל וְתִתְבַּטֵּל הַכַּוָּנָה ; מִצְוָה מִן הַמֻּבְחָר שֶׁלֹּא לֶאֱחֹז כְּלוּם, יָדָיו כְּפוּתִין כְּעוֹמֵד לִפְנֵי הַמֶּלֶךְ (סעיף א).

המסקנה. הַגֶּדֶר אֵינוֹ טִיב הַחֵפֶץ אֶלָּא הַטִּרְדָּא : מַה שֶּׁאֵינוֹ נִפְסָד וְאֵינוֹ מַזִּיק — מֻתָּר מִן הַדִּין. עיין סימן צ״ה · 95 (עמידה לפני המלך).

הַסִּדּוּר וְהַהֶפְסֵק

הַסִּדּוּר וְהַהֶפְסֵק

היסוד. הַסִּדּוּר — לְצֹרֶךְ הַתְּפִלָּה — אֵינוֹ נִטְרָד ; אֲבָל אֵין חוֹזְרִים אַחֲרָיו בְּאֶמְצַע הַתְּפִלָּה (הֶפְסֵק), אֶלָּא בְּמָקוֹם מְיֻחָד מוּכָן (סעיף ב).

המסקנה. רוֹשְׁמִים מֵרֹאשׁ אֶת כָּל מְקוֹמוֹת הַתְּפִלָּה שֶׁלֹּא לְהַפְסִיק. עיין סימן צ״ח · 98 (כוונת התפילה).

סֵפֶר שֶׁנָּפַל — הַגְבָּהָה בְּסוֹף הַבְּרָכָה

סֵפֶר שֶׁנָּפַל — הַגְבָּהָה בְּסוֹף הַבְּרָכָה

היסוד. סֵפֶר שֶׁנָּפַל לָאָרֶץ וּמוֹנֵעַ הַכַּוָּנָה — מֻתָּר לְהַגְבִּיהוֹ כְּשֶׁיְּסַיֵּם אוֹתָהּ בְּרָכָה (סעיף ג).

המסקנה. אֵין מִתְכּוֹפְפִים בְּאֶמְצַע הַבְּרָכָה : מַמְתִּינִים לְסִיּוּמָהּ, וְאָז מַגְבִּיהִים אֶת שֶׁמּוֹנֵעַ הַכַּוָּנָה. עיין סימן צ״ז · 97 (הנהגות בתפילה).

הפניה ללימוד. היקף הסימן — הַמַּה שֶּׁלִּבּוֹ עָלָיו, הַסִּדּוּר וְהַהֶפְסֵק וְהַסֵּפֶר שֶׁנָּפַל — נוגע להנהגת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

הֲסָרַת הַטְּרָדוֹת כְּדֵי שֶׁיְּכַוֵּן — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן צ״ו. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מג׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
מַה שֶּׁלִּבּוֹ עָלָיו וְלוּלָב או״ח צ״ו — לֹא יֶאֱחֹז תְּפִלִּין, סֵפֶר, קְעָרָה מְלֵאָה, סַכִּין, מָעוֹת, כִּכָּר ; לוּלָב מֻתָּר. מ״ב צ״ו — דְּבָרִים שֶׁלִּבּוֹ עֲלֵיהֶם, וְטַעַם הַחֲשָׁשׁ שֶׁמָּא יִפְּלוּ. ערוה״ש צ״ו — גֶּדֶר מַה שֶּׁמַּטְרִיד וּמַה שֶּׁאֵינוֹ מַטְרִיד. דרך אדמו״ר הזקן — לֹא יֶאֱחֹז מַה שֶּׁלִּבּוֹ עֲלֵיהֶם ; שְׁאָר דְּבָרִים מֻתָּר מִן הַדִּין, אֲבָל מִצְוָה מִן הַמֻּבְחָר יָדָיו כְּפוּתִין (צ״ה ד׳) ; לוּלָב — אֲחִיזָתוֹ מִצְוָה (סעיף א).
הַסִּדּוּר וְהַהֶפְסֵק או״ח צ״ו — מֻתָּר לֶאֱחֹז מַחֲזוֹר תְּפִלּוֹת ; הגה : לֹא הָיָה בְּיָדוֹ לֹא יַחֲזֹר אֶלָּא בְּמָקוֹם מוּכָן. מ״ב צ״ו — פְּרָטֵי הַהֶפְסֵק וְהַהֲכָנָה לִפְנֵי הַתְּפִלָּה. ערוה״ש צ״ו — דִּין הַנּוֹטֵל סִדּוּר בְּאֶמְצַע הַתְּפִלָּה. דרך אדמו״ר הזקן — מֻתָּר לֶאֱחֹז סִדּוּר לְצֹרֶךְ הַתְּפִלָּה ; לֹא יַחֲזֹר אַחֲרָיו מִשּׁוּם הֶפְסֵק אֶלָּא מִמָּקוֹם מוּכָן ; יִרְשֹׁם תְּחִלָּה כָּל הַמְּקוֹמוֹת (סעיף ב).
סֵפֶר שֶׁנָּפַל בָּאָרֶץ או״ח צ״ו — לֹא נִתְפָּרֵשׁ בַּשֻּׁלְחָן עָרוּךְ (ב׳ סְעִיפִים) ; מְקוֹרוֹ בָּאַחֲרוֹנִים. מ״ב צ״ו — דִּין סֵפֶר שֶׁנָּפַל וְהַגְבָּהָתוֹ בַּתְּפִלָּה. ערוה״ש צ״ו — מָתַי מֻתָּר לְהַגְבִּיהַּ בְּאֶמְצַע הַתְּפִלָּה. דרך אדמו״ר הזקן — נָפַל סֵפֶר לְפָנָיו וְאֵינוֹ יָכוֹל לְכַוֵּן — מֻתָּר לְהַגְבִּיהוֹ כְּשֶׁיְּסַיֵּם אוֹתָהּ בְּרָכָה (סעיף ג).

הטבלה נערכה מטקסט המחבר (אורח חיים צ״ו, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ג-כ״ד). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — מַה שֶּׁלִּבּוֹ עָלָיו

קו א — מַה שֶּׁלִּבּוֹ עָלָיו

יסוד הסימן : לֹא יֶאֱחֹז מַה שֶּׁלִּבּוֹ עֲלֵיהֶם ; לוּלָב מֻתָּר שֶׁאֲחִיזָתוֹ מִצְוָה (סעיף א).

קריאת הרב : הַמַּדָּד הוּא הַטִּרְדָּא, לֹא הַחֵפֶץ ; שְׁאָר דְּבָרִים מֻתָּר מִן הַדִּין, אֲבָל מִצְוָה מִן הַמֻּבְחָר שֶׁלֹּא לֶאֱחֹז כְּלוּם. עיין סימן צ״ה · 95.

מסקנה למעשה : לַעֲמֹד יָדָיו כְּפוּתִין לִפְנֵי הַמֶּלֶךְ, בְּלִי לֶאֱחֹז דָּבָר שֶׁלִּבּוֹ עָלָיו.

קו ב — הַסִּדּוּר וְהַהֶפְסֵק

קו ב — הַסִּדּוּר וְהַהֶפְסֵק

יסוד הסימן : מֻתָּר לֶאֱחֹז סִדּוּר לְצֹרֶךְ הַתְּפִלָּה ; לֹא יַחֲזֹר אַחֲרָיו מִשּׁוּם הֶפְסֵק (סעיף ב).

קריאת הרב : מַה שֶּׁמְּשַׁמֵּשׁ אֶת הַתְּפִלָּה אֵינוֹ מַטְרִיד ; מַה שֶּׁמַּפְסִיק (חִפּוּשׂ, הִלּוּךְ) נִדְחֶה. עיין סימן צ״ח · 98.

מסקנה למעשה : לְהָכִין וְלִרְשֹׁם מֵרֹאשׁ אֶת כָּל מְקוֹמוֹת הַסִּדּוּר לְאוֹתָהּ תְּפִלָּה.

קו ג — סֵפֶר שֶׁנָּפַל

קו ג — סֵפֶר שֶׁנָּפַל

יסוד הסימן : נָפַל סֵפֶר וְאֵינוֹ יָכוֹל לְכַוֵּן — מֻתָּר לְהַגְבִּיהוֹ כְּשֶׁיְּסַיֵּם הַבְּרָכָה (סעיף ג).

קריאת הרב : אֵין קוֹטְעִים בְּרָכָה שֶׁבְּאֶמְצַע ; מַמְתִּינִים לְסִיּוּמָהּ לְהָסִיר אֶת מוֹנֵעַ הַכַּוָּנָה. עיין סימן צ״ז · 97.

מסקנה למעשה : לְהַנִּיחַ אֶת הַסֵּפֶר בָּאָרֶץ עַד סוֹף הַבְּרָכָה, וְאָז לְהַגְבִּיהוֹ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן צ״ו ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן צ״ו (הֲסָרַת הַטְּרָדוֹת כְּדֵי שֶׁיְּכַוֵּן)

  • ① מַה שֶּׁלִּבּוֹ עָלָיו. בַּתְּפִלָּה, לֹא לֶאֱחֹז מַה שֶּׁלִּבּוֹ עָלָיו — תְּפִלִּין, סֵפֶר, קְעָרָה מְלֵאָה, סַכִּין, מָעוֹת, כִּכָּר — פֶּן יִפֹּל. יסוד : שוע״ר צ״ו סעיף א.
  • ② מִצְוָה מִן הַמֻּבְחָר. שְׁאָר דְּבָרִים מֻתָּרִים מִן הַדִּין, אֲבָל הַמֻּבְחָר שֶׁלֹּא לֶאֱחֹז כְּלוּם : יָדָיו כְּפוּתִין כְּעוֹמֵד לִפְנֵי הַמֶּלֶךְ. יסוד : שוע״ר צ״ו סעיף א.
  • ③ לוּלָב. לוּלָב בִּזְמַנּוֹ — מֻתָּר לֶאֱחֹז, כֵּיוָן שֶׁהָאֲחִיזָה הִיא מִצְוָה — אֵינוֹ נִטְרָד. יסוד : שוע״ר צ״ו סעיף א.
  • ④ סִדּוּר. סִדּוּר לְצֹרֶךְ הַתְּפִלָּה — מֻתָּר ; לֹא הָיָה בְּיָדוֹ, אֵין חוֹזְרִים אַחֲרָיו (הֶפְסֵק) אֶלָּא בְּמָקוֹם מוּכָן ; לִרְשֹׁם מֵרֹאשׁ אֶת הַמְּקוֹמוֹת. יסוד : שוע״ר צ״ו סעיף ב.
  • ⑤ סֵפֶר שֶׁנָּפַל. סֵפֶר שֶׁנָּפַל וּמוֹנֵעַ הַכַּוָּנָה — לְהַגְבִּיהוֹ כְּשֶׁיְּסַיֵּם אוֹתָהּ בְּרָכָה, לֹא בָּאֶמְצַע. יסוד : שוע״ר צ״ו סעיף ג.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן צ״ו (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַמַּה שֶּׁלִּבּוֹ עָלָיו, הַסִּדּוּר וְהַהֶפְסֵק, הַסֵּפֶר שֶׁנָּפַל — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
