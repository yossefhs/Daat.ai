דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ע״ה · דִּין גִּלּוּי שֵׂעָר וְקוֹל אִשָּׁה בִּשְׁעַת קְרִיאַת שְׁמַע

דרכו של אדמו״ר הזקן בדִּין גִּלּוּי שֵׂעָר וְקוֹל אִשָּׁה בִּשְׁעַת קְרִיאַת שְׁמַע : טֶפַח מְגֻלֶּה בָּאִשָּׁה — עֶרְוָה, מִפְּנֵי הַהִרְהוּר ; שֵׂעָר וְקוֹל אִשָּׁה — עֶרְוָה ; הַעֶרְוָה בַּעֲשָׁשִׁית וְהַהֶחֱזִיר פָּנָיו ; וְהַרְהוּר בְּדִבְרֵי תּוֹרָה — דִּבּוּר אָסוּר, הִרְהוּר מֻתָּר — ברגישות חב״ד, בִּצְנִיעוּת.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ע״ה — דִּין גִּלּוּי שֵׂעָר וְקוֹל אִשָּׁה בִּשְׁעַת קְרִיאַת שְׁמַע, נושא הנדון כאן בִּצְנִיעוּת וּבְרֹגַע. אצל הרב הסימן נפרש לי׳ סְעִיפִים : הַטֶּפַח הַמְגֻלֶּה בָּאִשָּׁה, קְרִיאַת הָאִשָּׁה עַצְמָהּ, הַלְּבוּשׁ הַדַּק, הַשֵּׂעָר בָּאִשָּׁה וּבָאִישׁ, קוֹל אִשָּׁה, עֶרְוַת בְּהֵמָה גּוֹי וְקָטָן, הָעֶרְוָה בַּעֲשָׁשִׁית, הַהֶחֱזִיר פָּנָיו, וְהַהַרְהוּר בְּדִבְרֵי תּוֹרָה.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ע״ה — י׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ע״ה). אדמו״ר הזקן דן בו בְּדִין גִּלּוּי שֵׂעָר וְקוֹל אִשָּׁה בִּשְׁעַת קְרִיאַת שְׁמַע, בִּצְנִיעוּת הָרְאוּיָה לַנּוֹשֵׂא. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ע״ה — דִּין גִּלּוּי שֵׂעָר וְקוֹל אִשָּׁה בִּשְׁעַת קְרִיאַת שְׁמַע

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ע״ה — דִּין גִּלּוּי שֵׂעָר וְקוֹל אִשָּׁה בִּשְׁעַת קְרִיאַת שְׁמַע — וּבוֹ י׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.75), ואחריו תרגומנו. הסימן כולל י׳ סעיפים (סעיף k → ‎.75.k).

סעיף א טֶפַח מְגֻלֶּה בָּאִשָּׁה — עֶרְוָה, אֲפִלּוּ אִשְׁתּוֹ ; פָּנֶיהָ וְיָדֶיהָ מֻתָּר, שׁוֹקָהּ אָסוּר.

לְהִזָּהֵר מִגִּלּוּי וְשֵׂעָר [וְקוֹל אִשָּׁה] בִּשְׁעַת קְרִיאַת שְׁמַע וְכֵן שֶׁלֹּא לִקְרוֹתָהּ כְּנֶגֶד הָעֶרְוָה וּבוֹ י׳ סְעִיפִים:טֶפַח מְגֻלֶּה בָּאִשָּׁה בְּמָקוֹם שֶׁדַּרְכָּהּ לְכַסּוֹת — עֶרְוָה הוּא אֲפִלּוּ הִיא אִשְׁתּוֹ, מִפְּנֵי שֶׁמֵּבִיא לִידֵי הִרְהוּר כְּשֶׁמִּסְתַּכֵּל בּוֹ, וְלָכֵן אָסוּר לִקְרוֹת וּלְהִתְפַּלֵּל כְּנֶגְדּוֹ (כְּמוֹ כְּנֶגֶד עֶרְוָה מַמָּשׁ, עַד שֶׁיַּחֲזֹר פָּנָיו לְצַד אַחֵר). וְיֵשׁ מַתִּירִין (א) אִם אֵינוֹ מִסְתַּכֵּל בּוֹ אֲפִלּוּ רוֹאֶה אוֹתוֹ בְּהַעֲבָרָה בְּעָלְמָא, כֵּיוָן שֶׁאֵינוֹ עֶרְוָה אֶלָּא מִשּׁוּם שֶׁמֵּבִיא לִידֵי הִרְהוּר, וְאֵין לָחוּשׁ לְהִרְהוּר אֶלָּא כְּשֶׁמִּסְתַּכֵּל בּוֹ וְלֹא בִּרְאִיָּה בְּעָלְמָא. וְכָל שֶׁכֵּן אִם מַעֲצִים עֵינָיו וְאֵינוֹ רוֹאֵהוּ כְּלָל שֶׁמֻּתָּר, וְאֵין צָרִיךְ לְהַחֲזִיר פָּנָיו (וְיֵשׁ לִסְמֹךְ עַל דִּבְרֵיהֶם כְּשֶׁאִי אֶפְשָׁר בְּעִנְיָן אַחֵר). (וְיֵשׁ אוֹמְרִים (ב) דִּלְפִי סְבָרָא הָרִאשׁוֹנָה לֹא אָמְרוּ טֶפַח אֶלָּא בְּאִשְׁתּוֹ, אֲבָל בְּאִשָּׁה אַחֶרֶת אֲפִלּוּ פָּחוֹת מִטֶּפַח אָסוּר. וְאֵין דִּבְרֵיהֶם עִקָּר).אֲבָל פָּנֶיהָ וְיָדֶיהָ שֶׁדַּרְכָּן לִהְיוֹת מְגֻלִּים — מֻתָּר לִקְרוֹת כְּנֶגְדָּן, שֶׁכֵּיוָן שֶׁרָגִיל בָּהֶן אֵינוֹ בָּא לִידֵי הִרְהוּר. אֲבָל כְּנֶגֶד שׁוֹקָהּ אָסוּר, אַף עַל פִּי שֶׁהַשּׁוֹק אֵינוֹ מָקוֹם צָנוּעַ בָּאִישׁ שֶׁדַּרְכּוֹ לִהְיוֹת מְגֻלֶּה (בִּמְקוֹמוֹת שֶׁהוֹלְכִין בְּלֹא אַנְפִּילָאוֹת), כֵּיוָן שֶׁהוּא מָקוֹם צָנוּעַ בָּאִשָּׁה — עֶרְוָה הִיא. וְיֵשׁ אוֹמְרִים שֶׁאֲפִלּוּ פָּחוֹת מִטֶּפַח שֶׁמְּגֻלֶּה בְּשׁוֹקָהּ עֶרְוָה הִיא אֲפִלּוּ בְּאִשְׁתּוֹ. וְנָכוֹן לָחוּשׁ לְדִבְרֵיהֶם:

טֶפַח בָּאִשָּׁה. טֶפַח מְגֻלֶּה בְּמָקוֹם שֶׁדַּרְכָּהּ לְכַסּוֹת — עֶרְוָה, אֲפִלּוּ אִשְׁתּוֹ, שֶׁמֵּבִיא לִידֵי הִרְהוּר בְּמִסְתַּכֵּל ; אָסוּר לִקְרוֹת וּלְהִתְפַּלֵּל כְּנֶגְדּוֹ. וְיֵשׁ מַתִּירִין בְּאֵין מִסְתַּכֵּל (רְאִיָּה בְּעָלְמָא), וְכָל שֶׁכֵּן בְּעוֹצֵם עֵינָיו. פָּנֶיהָ וְיָדֶיהָ שֶׁדַּרְכָּן מְגֻלִּים — מֻתָּר, שֶׁרָגִיל בָּהֶן ; אֲבָל שׁוֹקָהּ — אָסוּר, וְיֵשׁ אוֹמְרִים אֲפִלּוּ פָּחוֹת מִטֶּפַח, וְנָכוֹן לָחוּשׁ.

סעיף ב הָאִשָּׁה עַצְמָהּ — מֻתֶּרֶת לִקְרוֹת אַף עֲרֻמָּה אִם אֵין פָּנֶיהָ שֶׁלְּמַטָּה נִרְאִין.

וְכָל זֶה לָאֲנָשִׁים, אֲבָל הָאִשָּׁה עַצְמָהּ אֲפִלּוּ הִיא עֲרֻמָּה — מֻתֶּרֶת לִקְרוֹת וּלְהִתְפַּלֵּל אִם הִיא יוֹשֶׁבֶת שֶׁאֵין פָּנֶיהָ שֶׁלְּמַטָּה נִרְאִין, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ע״ד. וְאֵין בְּגִלּוּי כָּל גּוּפָה מִשּׁוּם ״וְלֹא יִרְאֶה בְךָ עֶרְוַת דָּבָר״, לְפִי שֶׁאֵינָהּ עֶרְוָה אֶלָּא לָאֲנָשִׁים וּמִשּׁוּם הִרְהוּר. וּלְפִיכָךְ אַף אִשָּׁה אַחֶרֶת מֻתֶּרֶת לִקְרוֹת וּלְהִתְפַּלֵּל כְּנֶגֶד אִשָּׁה עֲרֻמָּה, אִם אֵין פָּנֶיהָ שֶׁלְּמַטָּה נִרְאִין. וְיֵשׁ מִי שֶׁאוֹסֵר לְאִשָּׁה אַחֶרֶת (וְהָעִקָּר כַּסְּבָרָא הָרִאשׁוֹנָה):

הָאִשָּׁה עַצְמָהּ. כָּל זֶה לָאֲנָשִׁים ; אֲבָל הָאִשָּׁה עַצְמָהּ, אַף עֲרֻמָּה — מֻתֶּרֶת לִקְרוֹת וּלְהִתְפַּלֵּל בְּיוֹשֶׁבֶת שֶׁאֵין פָּנֶיהָ שֶׁלְּמַטָּה נִרְאִין (כְּסִימָן ע״ד), שֶׁאֵין גּוּפָהּ עֶרְוָה אֶלָּא לָאֲנָשִׁים וּמִשּׁוּם הִרְהוּר. וְלָכֵן אַף אִשָּׁה אַחֶרֶת מֻתֶּרֶת כְּנֶגְדָּהּ. וְיֵשׁ מִי שֶׁאוֹסֵר לְאִשָּׁה אַחֶרֶת, וְהָעִקָּר כַּסְּבָרָא הָרִאשׁוֹנָה.

סעיף ג לְבוּשָׁהּ דַּק וְנִרְאֶה בָּשָׂר מִתּוֹכוֹ — אָסוּר, כְּעֶרְוָה בַּעֲשָׁשִׁית.

אִם לְבוּשָׁהּ דַּק וְנִרְאֶה בָּשָׂר מִתּוֹכָהּ — אָסוּר, כְּמוֹ בָּעֶרְוָה בַּעֲשָׁשִׁית כְּמוֹ (שֶׁנִּתְבָּאֵר) [שֶׁיִּתְבָּאֵר]:

לְבוּשׁ דַּק. לְבוּשָׁהּ דַּק וְהַבָּשָׂר נִרְאֶה מִתּוֹכוֹ — אָסוּר, כְּעֶרְוָה בַּעֲשָׁשִׁית, כְּמוֹ שֶׁיִּתְבָּאֵר לְהַלָּן.

סעיף ד שֵׂעָר שֶׁדַּרְכָּהּ לְכַסּוֹת — עֶרְוָה ; בְּתוּלוֹת וּפֵאָה נָכְרִית — מֻתָּר.

שֵׂעָר שֶׁל אִשָּׁה שֶׁדַּרְכָּהּ לְכַסּוֹתָהּ — עֶרְוָה הִיא, מִפְּנֵי שֶׁמֵּבִיא לִידֵי הִרְהוּר, וְאָסוּר לִקְרוֹת אוֹ לְהִתְפַּלֵּל כְּנֶגְדָּהּ עַל דֶּרֶךְ שֶׁנִּתְבָּאֵר, אֲפִלּוּ הִיא אִשְׁתּוֹ. אֲבָל בְּתוּלוֹת שֶׁדַּרְכָּן לֵילֵךְ פְּרוּעוֹת רֹאשׁ, וְכֵן שְׂעָרוֹת שֶׁל נָשִׁים שֶׁרְגִילִין לָצֵאת מִחוּץ לְצַמָּתָן בִּקְצָת אֲרָצוֹת — מֻתָּר לִקְרוֹת כְּנֶגְדָּן, שֶׁכֵּיוָן שֶׁרְגִילִין בָּהֶן אֵינָן מְבִיאִין לִידֵי הִרְהוּר. וּפֵאָה נָכְרִית אֲפִלּוּ דַּרְכָּהּ לְכַסּוֹתָהּ — מֻתָּר לִקְרוֹת כְּנֶגְדָּהּ, וְגַם מֻתָּר לְגַלּוֹתָהּ, וְאֵין בָּהּ מִשּׁוּם ״יָצְאָה וְרֹאשָׁהּ פְּרוּעָה״ שֶׁהוּא אָסוּר מִן הַתּוֹרָה בְּאֵשֶׁת אִישׁ:

שֵׂעָר בָּאִשָּׁה. שֵׂעָר שֶׁדַּרְכָּהּ לְכַסּוֹת — עֶרְוָה, שֶׁמֵּבִיא לִידֵי הִרְהוּר, וְאָסוּר כְּנֶגְדּוֹ, אֲפִלּוּ אִשְׁתּוֹ. אֲבָל בְּתוּלוֹת שֶׁדַּרְכָּן פְּרוּעוֹת רֹאשׁ, וּשְׂעָרוֹת הָרְגִילוֹת לָצֵאת בִּקְצָת אֲרָצוֹת — מֻתָּר, שֶׁרְגִילִין בָּהֶן. וּפֵאָה נָכְרִית אֲפִלּוּ דַּרְכָּהּ לְכַסּוֹת — מֻתָּר לִקְרוֹת כְּנֶגְדָּהּ וְאַף לְגַלּוֹתָהּ, וְאֵין בָּהּ מִשּׁוּם ״יָצְאָה וְרֹאשָׁהּ פְּרוּעָה״.

סעיף ה שֵׂעָר שֶׁל עֶרְוַת אִישׁ הַיּוֹצֵא דֶּרֶךְ נֶקֶב שֶׁבְּבִגְדּוֹ — מֻתָּר.

שֵׂעָר שֶׁל עֶרְוַת אִישׁ יוֹצֵא דֶּרֶךְ נֶקֶב שֶׁבְּבִגְדוֹ — מֻתָּר לִקְרוֹת כְּנֶגְדּוֹ:

שֵׂעָר בָּאִישׁ. שֵׂעָר שֶׁל עֶרְוַת אִישׁ הַיּוֹצֵא דֶּרֶךְ נֶקֶב שֶׁבְּבִגְדּוֹ — מֻתָּר לִקְרוֹת כְּנֶגְדּוֹ.

סעיף ו קוֹל זֶמֶר שֶׁל אִשָּׁה — עֶרְוָה ; קוֹל דִּבּוּרָהּ שֶׁרָגִיל בּוֹ — אֵינוֹ עֶרְוָה.

קוֹל זֶמֶר שֶׁל אִשָּׁה בֵּין בִּפְנוּיָה בֵּין בְּאֵשֶׁת אִישׁ אֲפִלּוּ שֶׁל אִשְׁתּוֹ — הוּא עֶרְוָה, וְאָסוּר לִקְרוֹת קְרִיאַת שְׁמַע כְּשֶׁשּׁוֹמְעוֹ. וְהוּא הַדִּין לְהִתְפַּלֵּל אוֹ לְדַבֵּר בְּדִבְרֵי תּוֹרָה. אֲבָל קוֹל דִּבּוּרָהּ כֵּיוָן שֶׁרָגִיל בּוֹ — אֵינוֹ עֶרְוָה אֲפִלּוּ שֶׁל אֵשֶׁת אִישׁ. וַאֲפִלּוּ קוֹל זֶמֶר, אִם יָכוֹל לְכַוֵּן לִבּוֹ לִתְפִלָּתוֹ בְּעִנְיָן שֶׁאֵינוֹ שׁוֹמְעוֹ וְאֵינוֹ מֵשִׂים לֵב אֵלָיו — מֻתָּר:

קוֹל אִשָּׁה. קוֹל זֶמֶר שֶׁל אִשָּׁה, בֵּין פְּנוּיָה בֵּין אֵשֶׁת אִישׁ אֲפִלּוּ אִשְׁתּוֹ — עֶרְוָה, וְאָסוּר לִקְרוֹת ק״ש כְּשֶׁשּׁוֹמְעוֹ, וְהוּא הַדִּין לְהִתְפַּלֵּל אוֹ לְדַבֵּר בַּתּוֹרָה. אֲבָל קוֹל דִּבּוּרָהּ שֶׁרָגִיל בּוֹ — אֵינוֹ עֶרְוָה. וְאַף קוֹל זֶמֶר, אִם יָכוֹל לְכַוֵּן לִבּוֹ בְּלִי לְהַשְׁגִּיחַ בּוֹ — מֻתָּר.

סעיף ז עֶרְוַת בְּהֵמָה — מֻתָּר ; עֶרְוַת גּוֹי וְקָטָן — אָסוּר (וְיֵשׁ מַתִּירִין בְּקָטָן).

עֶרְוַת בְּהֵמָה — אֵינוֹ נִקְרָא עֶרְוָה, וּמֻתָּר לִקְרוֹת כְּנֶגְדָּהּ. אֲבָל הַגּוֹיִם אַף עַל פִּי שֶׁנֶּאֱמַר בָּהֶם ״בְּשַׂר חֲמוֹרִים בְּשָׂרָם״ — עֶרְוָתָם נִקְרָא עֶרְוָה, שֶׁנֶּאֱמַר: ״וְעֶרְוַת אֲבִיהֶם לֹא רָאוּ וְגו׳״, וְאָסוּר לִקְרוֹת כְּנֶגְדָּם. וְכֵן כְּנֶגֶד עֶרְוַת קָטָן אֲפִלּוּ בֶּן יוֹמוֹ. וְיֵשׁ מַתִּירִין כְּנֶגֶד עֶרְוַת קָטָן כָּל זְמַן שֶׁאֵינוֹ רָאוּי לְבִיאָה, דְּהַיְנוּ עַד שֶׁיִּהְיֶה בֶּן ט׳ שָׁנִים וְיוֹם אֶחָד. וְכֵן עִקָּר. וְלָכֵן אֵין צָרִיךְ לְכַסּוֹת עֶרְוַת הַקָּטָן בִּשְׁעַת בִּרְכַּת הַמִּילָה. וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁמִּכָּל מָקוֹם לֹא יֶאֱחֹז הַמּוֹהֵל הָעֶרְוָה בְּיָדוֹ בִּשְׁעַת בְּרָכָה:

עֶרְוַת בְּהֵמָה, גּוֹי וְקָטָן. עֶרְוַת בְּהֵמָה — אֵינָהּ עֶרְוָה, וּמֻתָּר. אֲבָל הַגּוֹיִם — עֶרְוָתָם עֶרְוָה (״וְעֶרְוַת אֲבִיהֶם לֹא רָאוּ״), וְאָסוּר, וְכֵן עֶרְוַת קָטָן אֲפִלּוּ בֶּן יוֹמוֹ. וְיֵשׁ מַתִּירִין בְּקָטָן שֶׁאֵינוֹ רָאוּי לְבִיאָה — עַד בֶּן ט׳ שָׁנִים וְיוֹם אֶחָד, וְכֵן עִקָּר ; וְלָכֵן אֵין צָרִיךְ לְכַסּוֹתוֹ בְּבִרְכַּת הַמִּילָה, וְיֵשׁ מִי שֶׁאוֹמֵר שֶׁלֹּא יֶאֱחֹז הַמּוֹהֵל בְּיָדוֹ בִּשְׁעַת בְּרָכָה.

סעיף ח עֶרְוָה בַּעֲשָׁשִׁית וְנִרְאֵית דֶּרֶךְ דָּפְנוֹתֶיהָ — אָסוּר, שֶׁבִּרְאִיָּה תָּלָה הַכָּתוּב.

עֶרְוָה בַּעֲשָׁשִׁית וְנִרְאֵית דֶּרֶךְ דָּפְנוֹתֶיהָ — אָסוּר לִקְרוֹת כְּנֶגְדָּהּ. וַאֲפִלּוּ הִיא בְּבַיִת אַחֵר וְנִרְאֵית דֶּרֶךְ עֲשָׁשִׁית שֶׁבַּחַלּוֹן, לְפִי שֶׁבִּרְאִיָּה תָּלָה הַכָּתוּב, שֶׁנֶּאֱמַר: ״וְלֹא יִרְאֶה בְךָ וְגו׳״, וַהֲרֵי הִיא נִרְאֵית:

עֶרְוָה בַּעֲשָׁשִׁית. עֶרְוָה בַּעֲשָׁשִׁית הַנִּרְאֵית דֶּרֶךְ דָּפְנוֹתֶיהָ — אָסוּר, וַאֲפִלּוּ בְּבַיִת אַחֵר וְנִרְאֵית דֶּרֶךְ עֲשָׁשִׁית שֶׁבַּחַלּוֹן, שֶׁבִּרְאִיָּה תָּלָה הַכָּתוּב (״וְלֹא יִרְאֶה בְךָ״) וַהֲרֵי נִרְאֵית.

סעיף ט הֶחֱזִיר פָּנָיו מִן הָעֶרְוָה — מֻתָּר ; עָצַם עֵינָיו אוֹ סוּמָא אוֹ בַּלַּיְלָה — אֵין מוֹעִיל.

הָיְתָה עֶרְוָה כְּנֶגְדּוֹ וְהֶחֱזִיר פָּנָיו מִמֶּנָּה עִם כָּל גּוּפוֹ עַד שֶׁנַּעֲשֵׂית לוֹ מִן הַצַּד, אַף עַל פִּי שֶׁעוֹמֵד בְּסָמוּךְ לָהּ מִן הַצַּד תּוֹךְ ד׳ אַמּוֹת — מֻתָּר לִקְרוֹת אֶצְלָהּ. וְאַף עַל פִּי שֶׁנֶּאֱמַר ״וְהָיָה מַחֲנֶיךָ קָדוֹשׁ וְלֹא יִרְאֶה בְךָ וְגו׳״, וּמַחֲנֵהוּ שֶׁל אָדָם הוּא ד׳ אַמּוֹת מִכָּל צַד כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ע״ט, מִכָּל מָקוֹם כֵּיוָן שֶׁאָמְרָה תּוֹרָה ״וְלֹא יִרְאֶה בְךָ״ — תָּלָה הַכָּתוּב בָּרְאִיָּה, וְלֹא אָסַר אֶלָּא עֶרְוָה הַנִּרְאִית. וּמִכָּל מָקוֹם אִם יְכוֹלָה לִהְיוֹת נִרְאֵית אֶלָּא שֶׁהוּא עוֹצֶם עֵינָיו מֵרְאוֹת בָּרַע אוֹ שֶׁהוּא סוּמָא אוֹ בַּלַּיְלָה — אֵין זֶה מוֹעִיל כְּלוּם, שֶׁהֲרֵי לֹא נֶאֱמַר ״לֹא תִּרְאֶה״ אֶלָּא ״לֹא יִרְאֶה״ כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ע״ד. וְיֵשׁ מַתִּירִין בָּזֶה, לְפִי שֶּׁמִּ״לֹא יִרְאֶה בְךָ וְגו׳״ אֵין לִלְמֹד אֶלָּא לְעֶרְוַת עַצְמוֹ, שֶׁכְּשֶׁעֶרְוָתוֹ מְגֻלָּה — אָסוּר לִקְרוֹת אֲפִלּוּ עָצַם עֵינָיו אוֹ סוּמָא אוֹ בַּלַּיְלָה, שֶׁנֶּאֱמַר ״וְלֹא יִרְאֶה בְךָ וְגו׳״, אֲבָל לֹא לְעֶרְוַת חֲבֵרוֹ. וְהָעִקָּר כַּסְּבָרָא הָרִאשׁוֹנָה:

הֶחֱזִיר פָּנָיו. הָיְתָה עֶרְוָה כְּנֶגְדּוֹ וְהֶחֱזִיר פָּנָיו עִם כָּל גּוּפוֹ עַד שֶׁנַּעֲשֵׂית מִן הַצַּד, אַף בְּתוֹךְ ד׳ אַמּוֹת — מֻתָּר, שֶׁתָּלָה הַכָּתוּב בָּרְאִיָּה (״וְלֹא יִרְאֶה בְךָ״) וְלֹא אָסַר אֶלָּא הַנִּרְאִית. אֲבָל אִם יְכוֹלָה לְהֵרָאוֹת וְהוּא עוֹצֵם עֵינָיו אוֹ סוּמָא אוֹ בַּלַּיְלָה — אֵין מוֹעִיל, שֶׁלֹּא נֶאֱמַר ״לֹא תִּרְאֶה״ אֶלָּא ״לֹא יִרְאֶה״. וְיֵשׁ מַתִּירִין, וְהָעִקָּר כַּסְּבָרָא הָרִאשׁוֹנָה.

סעיף י מֻתָּר לְהַרְהֵר בְּדִבְרֵי תּוֹרָה כְּשֶׁהוּא עָרֹם — דִּבּוּר אָסוּר, הִרְהוּר מֻתָּר.

מֻתָּר לְהַרְהֵר בְּדִבְרֵי תּוֹרָה כְּשֶׁהוּא עָרֹם, וְאֵין צָרִיךְ לוֹמַר כְּנֶגֶד עֶרְוָה אַחֶרֶת, שֶׁנֶּאֱמַר: ״עֶרְוַת דָּבָר״, דִּבּוּר — אָסוּר, הִרְהוּר — מֻתָּר. וּמִכָּל מָקוֹם אֵין לוֹ לִשְׁמֹעַ בְּרָכָה מֵחֲבֵרוֹ (וְלָצֵאת יְדֵי חוֹבָתוֹ) (כִּי אִי אֶפְשָׁר לוֹמַר) שׁוֹמֵעַ כְּעוֹנֶה (כֵּיוָן שֶׁאִי אֶפְשָׁר לוֹ לַעֲנוֹת):

הַרְהוּר בְּדִבְרֵי תּוֹרָה. מֻתָּר לְהַרְהֵר בְּדִבְרֵי תּוֹרָה כְּשֶׁהוּא עָרֹם, וְכָל שֶׁכֵּן כְּנֶגֶד עֶרְוָה אַחֶרֶת, שֶׁנֶּאֱמַר ״עֶרְוַת דָּבָר״ — דִּבּוּר אָסוּר, הִרְהוּר מֻתָּר. וּמִכָּל מָקוֹם אֵין לוֹ לִשְׁמֹעַ בְּרָכָה מֵחֲבֵרוֹ לָצֵאת בָּהּ בְּשׁוֹמֵעַ כְּעוֹנֶה, כֵּיוָן שֶׁאִי אֶפְשָׁר לוֹ לַעֲנוֹת.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ע״ה (מהדורת קה״ת, י׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — גִּלּוּי, שֵׂעָר וְקוֹל אִשָּׁה כְּעֶרְוָה

היסוד : הַצְּנִיעוּת כְּנֶגֶד הָעֶרְוָה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים, הנדונים בִּצְנִיעוּת. תחילה הַטֶּפַח וְהַגִּלּוּי בָּאִשָּׁה (הַשִּׁעוּר וְטַעַם הַהִרְהוּר). אחר כך הַשֵּׂעָר בָּאִשָּׁה וּבָאִישׁ. אחר כך קוֹל אִשָּׁה. ולבסוף הָעֶרְוָה, הָרְאִיָּה וְהַהַרְהוּר.

טֶפַח וְגִלּוּי בָּאִשָּׁה — מִשּׁוּם הִרְהוּר

טֶפַח וְגִלּוּי בָּאִשָּׁה — מִשּׁוּם הִרְהוּר

היסוד. טֶפַח מְגֻלֶּה בְּמָקוֹם שֶׁדַּרְכָּהּ לְכַסּוֹת — עֶרְוָה, אֲפִלּוּ אִשְׁתּוֹ, שֶׁמֵּבִיא לִידֵי הִרְהוּר ; פָּנֶיהָ וְיָדֶיהָ — מֻתָּר ; שׁוֹקָהּ — אָסוּר (סעיפים א-ג).

המסקנה. הַטַּעַם הוּא הַהִרְהוּר : בִּמְקוֹם שֶׁאֵין הִסְתַּכְּלוּת (רְאִיָּה בְּעָלְמָא, עוֹצֵם עֵינָיו) — יֵשׁ מַתִּירִין. עיין סימן 74 (הכנת הגוף לק״ש).

שֵׂעָר בָּאִשָּׁה וּבָאִישׁ

שֵׂעָר בָּאִשָּׁה וּבָאִישׁ

היסוד. שֵׂעָר שֶׁדַּרְכָּהּ לְכַסּוֹת — עֶרְוָה, אֲפִלּוּ אִשְׁתּוֹ ; בְּתוּלוֹת, שֵׂעָר הָרָגִיל לָצֵאת וּפֵאָה נָכְרִית — מֻתָּר ; שֵׂעָר הַיּוֹצֵא מִבֶּגֶד הָאִישׁ — מֻתָּר (סעיפים ד-ה).

המסקנה. הַמַּדָּד הוּא הָרְגִילוּת : מַה שֶּׁדַּרְכּוֹ מְגֻלֶּה אֵינוֹ מֵבִיא לִידֵי הִרְהוּר. עיין סימן 76 (מקום הראוי לק״ש).

קוֹל אִשָּׁה

קוֹל אִשָּׁה

היסוד. קוֹל זֶמֶר שֶׁל אִשָּׁה — עֶרְוָה, וְאָסוּר לִקְרוֹת ק״ש כְּשֶׁשּׁוֹמְעוֹ ; קוֹל דִּבּוּרָהּ הָרָגִיל — אֵינוֹ עֶרְוָה (סעיף ו).

המסקנה. אִם יָכוֹל לְכַוֵּן לִבּוֹ בְּלִי לְהַשְׁגִּיחַ בּוֹ — מֻתָּר. עיין סימן 79 (ד׳ אמות בק״ש).

עֶרְוָה, רְאִיָּה וְהַרְהוּר

עֶרְוָה, רְאִיָּה וְהַרְהוּר

היסוד. עֶרְוַת גּוֹי וְקָטָן — אָסוּר ; בַּעֲשָׁשִׁית הַנִּרְאֵית — אָסוּר ; הֶחֱזִיר פָּנָיו — מֻתָּר ; הַרְהוּר בְּדִבְרֵי תּוֹרָה עָרֹם — מֻתָּר (סעיפים ז-י).

המסקנה. הַכֹּל תָּלוּי בָּרְאִיָּה (״לֹא יִרְאֶה״) : דִּבּוּר אָסוּר, הִרְהוּר מֻתָּר. עיין סימן 73 (הנהגת הגוף בק״ש).

הפניה ללימוד. היקף הסימן — הַטֶּפַח, הַשֵּׂעָר, קוֹל אִשָּׁה וְהַהַרְהוּר — נוגע לַצְּנִיעוּת בְּהַנְהָגַת ק״ש וּתְפִלָּה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין גִּלּוּי שֵׂעָר וְקוֹל אִשָּׁה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ע״ה. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מי׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
טֶפַח וְגִלּוּי בָּאִשָּׁה או״ח ע״ה — טֶפַח מְגֻלֶּה בָּאִשָּׁה עֶרְוָה, אֲפִלּוּ אִשְׁתּוֹ ; הרמ״א — בְּאִשָּׁה אַחֶרֶת אֲפִלּוּ פָּחוֹת מִטֶּפַח. מ״ב ע״ה — גֶּדֶר הַטֶּפַח, מָקוֹם שֶׁדַּרְכּוֹ לְכַסּוֹת וּפְרָטֵי הַשּׁוֹק. ערוה״ש ע״ה — הַהֶכְרֵעַ בִּמְקוֹמוֹת שֶׁנִּשְׁתַּנּוּ הַמִּנְהָגִים. דרך אדמו״ר הזקן — טֶפַח עֶרְוָה מִשּׁוּם הִרְהוּר ; יֵשׁ מַתִּירִין בְּאֵין מִסְתַּכֵּל ; פָּנֶיהָ וְיָדֶיהָ מֻתָּר, שׁוֹקָהּ אָסוּר (סעיפים א-ג).
שֵׂעָר בָּאִשָּׁה וּבָאִישׁ או״ח ע״ה — שֵׂעָר שֶׁדַּרְכָּהּ לְכַסּוֹת אָסוּר ; בְּתוּלוֹת וּשְׂעָר נָכְרִית — מֻתָּר. מ״ב ע״ה — גֶּדֶר הַשְּׂעָרוֹת הַיּוֹצְאוֹת וּפֵאָה נָכְרִית. ערוה״ש ע״ה — דִּין הַשֵּׂעָר לְפִי הַמָּקוֹם וְהַזְּמַן. דרך אדמו״ר הזקן — שֵׂעָר שֶׁדַּרְכָּהּ לְכַסּוֹת עֶרְוָה ; בְּתוּלוֹת, שֵׂעָר הָרָגִיל וּפֵאָה נָכְרִית — מֻתָּר ; שֵׂעָר הָאִישׁ הַיּוֹצֵא מִבֶּגֶד — מֻתָּר (סעיפים ד-ה).
קוֹל אִשָּׁה או״ח ע״ה — יֵשׁ לִזָּהֵר מִקּוֹל זֶמֶר אִשָּׁה ; קוֹל הָרָגִיל אֵינוֹ עֶרְוָה. מ״ב ע״ה — גֶּדֶר קוֹל זֶמֶר וְקוֹל דִּבּוּר בִּשְׁעַת הַקְּרִיאָה. ערוה״ש ע״ה — פְּרָטֵי קוֹל אִשָּׁה בַּתְּפִלָּה וּבַתּוֹרָה. דרך אדמו״ר הזקן — קוֹל זֶמֶר עֶרְוָה, אָסוּר לִקְרוֹת ; קוֹל דִּבּוּר הָרָגִיל אֵינוֹ עֶרְוָה ; יָכוֹל לְכַוֵּן בְּלִי לְהַשְׁגִּיחַ — מֻתָּר (סעיף ו).
עֶרְוָה, רְאִיָּה וְהַרְהוּר או״ח ע״ה — עֶרְוָה בַּעֲשָׁשִׁית אָסוּר ; הֶחֱזִיר פָּנָיו אוֹ עָצַם עֵינָיו אוֹ סוּמָא — מֻתָּר. מ״ב ע״ה — גֶּדֶר הָרְאִיָּה וְהַהַרְחָקָה וְעֶרְוַת גּוֹי וְקָטָן. ערוה״ש ע״ה — הַהַכְרָעָה בָּרְאִיָּה וּבַהַרְהוּר. דרך אדמו״ר הזקן — גּוֹי וְקָטָן וַעֲשָׁשִׁית — אָסוּר ; הֶחֱזִיר פָּנָיו — מֻתָּר, וְעָצַם עֵינָיו לֹא מוֹעִיל ; הַרְהוּר בַּתּוֹרָה עָרֹם — מֻתָּר, דִּבּוּר אָסוּר (סעיפים ז-י).

הטבלה נערכה מטקסט המחבר (אורח חיים ע״ה, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ד). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (י׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1) וְנֶאֱמָר בִּצְנִיעוּת.

קו א — הַהִרְהוּר הוּא הַטַּעַם

קו א — הַהִרְהוּר הוּא הַטַּעַם

יסוד הסימן : טֶפַח, שֵׂעָר וְקוֹל — עֶרְוָה מִפְּנֵי שֶׁמֵּבִיא לִידֵי הִרְהוּר (סעיפים א, ד, ו).

קריאת הרב : כֵּיוָן שֶׁהַטַּעַם הוּא הַהִרְהוּר, מַה שֶּׁרָגִיל (פָּנֶיהָ, יָדֶיהָ, קוֹל דִּבּוּר) אֵינוֹ עֶרְוָה. עיין סימן 74.

מסקנה למעשה : לְהַקְפִּיד עַל צְנִיעוּת הַמָּקוֹם וְהָרְאִיָּה, בְּלִי הַחְמָרָה יְתֵרָה בִּמְקוֹם שֶׁאֵין הִסְתַּכְּלוּת.

קו ב — הַכֹּל תָּלוּי בָּרְאִיָּה

קו ב — הַכֹּל תָּלוּי בָּרְאִיָּה

יסוד הסימן : ״וְלֹא יִרְאֶה בְךָ״ — עֶרְוָה בַּעֲשָׁשִׁית אָסוּר, הֶחֱזִיר פָּנָיו מֻתָּר (סעיפים ח-ט).

קריאת הרב : הַכָּתוּב תָּלָה בָּרְאִיָּה ; הַחְזָרַת הַגּוּף מוֹעִילָה, אֲבָל עֲצִימַת עֵינַיִם אֵינָהּ מוֹעִילָה (״לֹא יִרְאֶה״ וְלֹא ״לֹא תִּרְאֶה״). עיין סימן 79.

מסקנה למעשה : כְּנֶגֶד עֶרְוָה — לְהַחֲזִיר כָּל הַגּוּף ; לַיְלָה אוֹ עֲצִימַת עֵינַיִם אֵינָם פּוֹטְרִים בְּנִרְאֵית.

קו ג — דִּבּוּר אָסוּר, הִרְהוּר מֻתָּר

קו ג — דִּבּוּר אָסוּר, הִרְהוּר מֻתָּר

יסוד הסימן : ״עֶרְוַת דָּבָר״ — דִּבּוּר אָסוּר, הִרְהוּר מֻתָּר (סעיף י).

קריאת הרב : הַהַרְהוּר בַּתּוֹרָה מֻתָּר אַף עָרֹם, אֲבָל אֵינוֹ יוֹצֵא בְּשׁוֹמֵעַ כְּעוֹנֶה, שֶׁאֵינוֹ יָכוֹל לַעֲנוֹת. עיין סימן 73.

מסקנה למעשה : הַרְהוּר הַתּוֹרָה מְלַוֶּה תָּמִיד ; הַדִּבּוּר וְהַבְּרָכָה מַמְתִּינִים לְמָקוֹם וּלְמַצָּב הָרָאוּי.

5 · למעשה

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ע״ה ברגישות אדמו״ר הזקן וחב״ד — מוצגות בִּצְנִיעוּת, ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ע״ה (דִּין גִּלּוּי שֵׂעָר וְקוֹל אִשָּׁה)

  • ① טֶפַח. לֹא לִקְרוֹת ק״ש וְלֹא לְהִתְפַּלֵּל כְּנֶגֶד טֶפַח מְגֻלֶּה בְּמָקוֹם שֶׁדַּרְכּוֹ לְכַסּוֹת (אֲפִלּוּ אִשְׁתּוֹ) ; פָּנֶיהָ וְיָדֶיהָ — מֻתָּר, אֲבָל שׁוֹקָהּ — אָסוּר. יסוד : שוע״ר ע״ה סעיפים א-ג.
  • ② שֵׂעָר. שֵׂעָר שֶׁדַּרְכּוֹ לְכַסּוֹת — עֶרְוָה ; אֲבָל בְּתוּלוֹת, שֵׂעָר הָרָגִיל לָצֵאת וּפֵאָה נָכְרִית — מֻתָּר. יסוד : שוע״ר ע״ה סעיפים ד-ה.
  • ③ קוֹל אִשָּׁה. לֹא לִקְרוֹת ק״ש כְּשֶׁשּׁוֹמֵעַ קוֹל זֶמֶר ; קוֹל דִּבּוּר הָרָגִיל אֵינוֹ עֶרְוָה ; יָכוֹל לְכַוֵּן בְּלִי לְהַשְׁגִּיחַ — מֻתָּר. יסוד : שוע״ר ע״ה סעיף ו.
  • ④ רְאִיָּה. עֶרְוָה בַּעֲשָׁשִׁית הַנִּרְאֵית — אָסוּר ; הַחְזָרַת כָּל הַגּוּף — מֻתָּר, אֲבָל עֲצִימַת עֵינַיִם, סוּמָא וְלַיְלָה — אֵין מוֹעִיל. יסוד : שוע״ר ע״ה סעיפים ח-ט.
  • ⑤ הַרְהוּר. מֻתָּר לְהַרְהֵר בְּדִבְרֵי תּוֹרָה אַף עָרֹם (דִּבּוּר אָסוּר, הִרְהוּר מֻתָּר) ; אֵין יוֹצְאִים בִּבְרָכָה בְּשׁוֹמֵעַ כְּעוֹנֶה. יסוד : שוע״ר ע״ה סעיף י.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ע״ה (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַטֶּפַח, הַשֵּׂעָר, קוֹל אִשָּׁה, הָרְאִיָּה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
