דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן פ׳ · דִּין מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ

דרכו של אדמו״ר הזקן בדִּין מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ : תְּפִלַּת י״ח דּוֹרֶשֶׁת גּוּף נָקִי — מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה, וְתַשְׁלוּמִין בְּמִנְחָה ; פְּסוּקֵי דְּזִמְרָה — בְּלֹא בְּרָכוֹת כְּקוֹרֵא בַּתּוֹרָה ; אֲבָל קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ — בִּזְמַנָּן אֲפִלּוּ בְּלֹא גּוּף נָקִי, עִם תְּפִלִּין בֵּין ״אַהֲבָה״ לִק״ש — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן פ׳ — דִּין מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ. אצל הרב הסימן נפרש לג׳ סְעִיפִים : תְּפִלַּת י״ח הַדּוֹרֶשֶׁת גּוּף נָקִי (מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה, תַּשְׁלוּמִין בְּמִנְחָה), פְּסוּקֵי דְּזִמְרָה בְּלֹא בְּרָכוֹת, וּקְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ הַנֶּאֱמָרוֹת בִּזְמַנָּן אֲפִלּוּ בְּלֹא גּוּף נָקִי, עִם תְּפִלִּין בֵּין ״אַהֲבָה״ לִק״ש.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן פ׳ — ג׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim פ׳). אדמו״ר הזקן דן בו בְּדִין מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן פ׳ — דִּין מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן פ׳ — דִּין מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ — וּבוֹ ג׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.80), ואחריו תרגומנו. הסימן כולל ג׳ סעיפים (סעיף k → ‎.80.k).

סעיף א הַתְּפִלָּה דּוֹרֶשֶׁת גּוּף נָקִי — מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה, וְתַשְׁלוּמִין בְּמִנְחָה.

מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ וּבוֹ ג׳ סְעִיפִים:מִי שֶׁמִּתְיַלֵּד לוֹ רוּחַ בְּבִטְנוֹ וּמוֹצִיא רוּחוֹת בַּבֹּקֶר וּבָרִי לוֹ שֶׁאֵינוֹ יָכוֹל לַעֲמֹד עַל עַצְמוֹ מִלְּהָפִיחַ עַד שֶׁיִּגְמֹר כָּל תְּפִלַּת י״ח — מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה מִמַּה שֶּׁיִּתְפַּלֵּל בְּלֹא גּוּף נָקִי, שֶׁהַתְּפִלָּה יֵשׁ לָהּ תַּשְׁלוּמִין, שֶׁאִם נֶאֱנַס וְלֹא הִתְפַּלֵּל שַׁחֲרִית — מִתְפַּלֵּל מִנְחָה שְׁתַּיִם, וְזֶה גַּם כֵּן אָנוּס הוּא שֶׁאִי אֶפְשָׁר לוֹ לְהִתְפַּלֵּל י״ח בְּלֹא גּוּף נָקִי, שֶׁגְּנַאי גָּדוֹל הוּא לְהָפִיחַ כְּשֶׁעוֹמֵד לִפְנֵי הַמֶּלֶךְ וּמְדַבֵּר עִמּוֹ וּכְדַי בִּזָּיוֹן וָקָצֶף, שֶׁהֲרֵי הִצְרִיכוּ לְהַרְחִיק ד׳ אַמּוֹת כְּשֶׁמְּבַקֵּשׁ לְהָפִיחַ בַּתְּפִלָּה וְאִי אֶפְשָׁר לוֹ לַעֲמֹד עַל עַצְמוֹ כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן ק״ג:

גּוּף נָקִי בַּתְּפִלָּה. מִי שֶׁמִּתְיַלֵּד לוֹ רוּחַ בְּבִטְנוֹ וּבָרִי לוֹ שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ עַד גְּמַר כָּל תְּפִלַּת י״ח — מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה מִלְּהִתְפַּלֵּל בְּלֹא גּוּף נָקִי, שֶׁהַתְּפִלָּה יֵשׁ לָהּ תַּשְׁלוּמִין : נֶאֱנַס וְלֹא הִתְפַּלֵּל שַׁחֲרִית — מִתְפַּלֵּל מִנְחָה שְׁתַּיִם ; וְגַם זֶה אָנוּס, שֶׁגְּנַאי גָּדוֹל לְהָפִיחַ לִפְנֵי הַמֶּלֶךְ (בִּזָּיוֹן וָקָצֶף), וּכְבָר הִצְרִיכוּ לְהַרְחִיק ד׳ אַמּוֹת (סימן ק״ג).

סעיף ב פְּסוּקֵי דְּזִמְרָה — לֹא יֹאמַר, אֶלָּא בְּלֹא בִּרְכוֹתֶיהָ כְּקוֹרֵא בַּתּוֹרָה.

וְכֵיוָן (א) שֶׁאֵינוֹ מִתְפַּלֵּל תְּפִלַּת י״ח שֶׁל שַׁחֲרִית — לֹא יֹאמַר גַּם כֵּן פְּסוּקֵי דְּזִמְרָה, אֶלָּא אִם כֵּן יֹאמַר בְּלֹא בִּרְכוֹתֶיהָ כְּקוֹרֵא בַּתּוֹרָה, שֶׁהֵם נִתְקְנוּ בִּבְרָכָה כְּדֵי לְסַדֵּר שְׁבָחָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא קֹדֶם הַתְּפִלָּה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן נ״א נ״ב:

פְּסוּקֵי דְּזִמְרָה. וְכֵיוָן שֶׁאֵינוֹ מִתְפַּלֵּל תְּפִלַּת י״ח שֶׁל שַׁחֲרִית — לֹא יֹאמַר גַּם כֵּן פְּסוּקֵי דְּזִמְרָה, אֶלָּא אִם יֹאמְרֵם בְּלֹא בִּרְכוֹתֶיהָ כְּקוֹרֵא בַּתּוֹרָה, שֶׁנִּתְקְנוּ בִּבְרָכָה לְסַדֵּר שְׁבָחָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא קֹדֶם הַתְּפִלָּה (סימן נ״א נ״ב).

סעיף ג קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ בִּזְמַנָּן ; תְּפִלִּין בֵּין ״אַהֲבָה״ לִק״ש ; טַלִּית מֻתָּר.

אֲבָל (ב) קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ צָרִיךְ לוֹמַר בִּזְמַנָּן אֲפִלּוּ בְּלֹא גּוּף נָקִי, שֶׁאֵינוֹ נִקְרָא מְדַבֵּר עִם הַמֶּלֶךְ אֶלָּא כְּשֶׁעוֹמֵד לְפָנָיו בִּתְפִלַּת י״ח כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ע״ד. וְאַף שֶׁיִּקְרָא קְרִיאַת שְׁמַע בְּלֹא תְּפִלִּין שֶׁהַתְּפִלִּין צְרִיכִין גּוּף נָקִי — לֹא יַמְתִּין וְיַעֲבֹר זְמַן קְרִיאַת שְׁמַע כְּדֵי לִקְרוֹתָהּ בִּתְפִלִּין, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ס״ו. וְאִם יֵרָאֶה לוֹ שֶׁיּוּכַל לַעֲמֹד עַל עַצְמוֹ בִּשְׁעַת קְרִיאַת שְׁמַע — יָנִיחַ תְּפִלִּין בֵּין ״אַהֲבָה״ לִקְרִיאַת שְׁמַע וִיבָרֵךְ עֲלֵיהֶן וְאַחַר כָּךְ יַחְלְצֵם מִיָּד. אֲבָל הַטַּלִּית יָכוֹל לִלְבֹּשׁ וּלְבָרֵךְ קֹדֶם שֶׁיַּתְחִיל הַבְּרָכוֹת, שֶׁאֵין צָרִיךְ גּוּף נָקִי לַצִּיצִית:

קְרִיאַת שְׁמַע וְהַתְּפִלִּין. אֲבָל קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ צָרִיךְ לוֹמַר בִּזְמַנָּן אֲפִלּוּ בְּלֹא גּוּף נָקִי, שֶׁאֵינוֹ נִקְרָא מְדַבֵּר עִם הַמֶּלֶךְ אֶלָּא בִּתְפִלַּת י״ח (סימן ע״ד). וְאַף שֶׁיִּקְרָא בְּלֹא תְּפִלִּין — לֹא יַמְתִּין וְיַעֲבֹר זְמַן ק״ש כְּדֵי לִקְרוֹתָהּ בִּתְפִלִּין (סימן ס״ו) ; וְאִם יֵרָאֶה לוֹ שֶׁיּוּכַל לַעֲמֹד עַל עַצְמוֹ — יָנִיחַ תְּפִלִּין בֵּין ״אַהֲבָה״ לִק״ש וִיבָרֵךְ וְאַחַר כָּךְ יַחְלְצֵם מִיָּד. אֲבָל הַטַּלִּית יָכוֹל לִלְבֹּשׁ וּלְבָרֵךְ, שֶׁאֵין צָרִיךְ גּוּף נָקִי לַצִּיצִית.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim פ׳ (מהדורת קה״ת, ג׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ

היסוד : מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב שלושה צירים. תחילה תְּפִלַּת י״ח הַדּוֹרֶשֶׁת גּוּף נָקִי — מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה. אחר כך דִּין פְּסוּקֵי דְּזִמְרָה לְמִי שֶׁאֵינוֹ מִתְפַּלֵּל. ולבסוף קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ הַנֶּאֱמָרוֹת בִּזְמַנָּן, וְדִין הַתְּפִלִּין וְהַטַּלִּית.

גּוּף נָקִי בִּתְפִלַּת י״ח

גּוּף נָקִי בִּתְפִלַּת י״ח

היסוד. בָּרִי לוֹ שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר עַד גְּמַר י״ח — מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה מִלְּהִתְפַּלֵּל בְּלֹא גּוּף נָקִי, שֶׁיֵּשׁ לָהּ תַּשְׁלוּמִין (סעיף א).

המסקנה. הָעֲמִידָה לִפְנֵי הַמֶּלֶךְ מְחַיֶּבֶת נְקִיּוּת ; הָאָנוּס מִתְפַּלֵּל מִנְחָה שְׁתַּיִם. עיין סימן ע״ד · 74 (גוף נקי ומחיצות).

פְּסוּקֵי דְּזִמְרָה בְּלֹא בְּרָכוֹת

פְּסוּקֵי דְּזִמְרָה בְּלֹא בְּרָכוֹת

היסוד. כֵּיוָן שֶׁאֵינוֹ מִתְפַּלֵּל י״ח — לֹא יֹאמַר פְּסוּקֵי דְּזִמְרָה, אֶלָּא בְּלֹא בִּרְכוֹתֶיהָ כְּקוֹרֵא בַּתּוֹרָה (סעיף ב).

המסקנה. הַשֶּׁבַח עִם בִּרְכָתוֹ נִתְקַן לְסַדֵּר שְׁבָחָיו קֹדֶם הַתְּפִלָּה ; בְּלֹא תְּפִלָּה — קוֹרֵא כִּקְרִיאָה בְּעָלְמָא. עיין סימן נ״א · 51 (פסוקי דזמרה).

קְרִיאַת שְׁמַע וְהַתְּפִלִּין

קְרִיאַת שְׁמַע וְהַתְּפִלִּין

היסוד. קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ בִּזְמַנָּן אֲפִלּוּ בְּלֹא גּוּף נָקִי — שֶׁאֵינוֹ מְדַבֵּר עִם הַמֶּלֶךְ (סימן ע״ד) ; תְּפִלִּין בֵּין ״אַהֲבָה״ לִק״ש אִם יָכוֹל לַעֲמֹד עַל עַצְמוֹ, וְאַחַר כָּךְ חוֹלֵץ (סעיף ג).

המסקנה. אֵין מַעֲבִירִין זְמַן ק״ש בִּשְׁבִיל תְּפִלִּין ; הַטַּלִּית מֻתֶּרֶת (אֵין צָרִיךְ גּוּף נָקִי לַצִּיצִית). עיין סימן ס״ו · 66 (הפסק בקריאת שמע).

הפניה ללימוד. היקף הסימן — הַגּוּף נָקִי בַּתְּפִלָּה, הַפְּסוּקֵי דְּזִמְרָה וְהַקְּרִיאַת שְׁמַע בִּזְמַנָּהּ — נוגע להנהגת התְּפִלָּה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן פ׳. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מג׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
גּוּף נָקִי בִּתְפִלַּת י״ח או״ח פ׳ — מוּטָב שֶׁיַּעֲבֹר זְמַן קְרִיאַת שְׁמַע וּתְפִלָּה מִלְּהִתְפַּלֵּל בְּלֹא גּוּף נָקִי ; אָנוּס — מִנְחָה שְׁתַּיִם. מ״ב פ׳ — פְּרָטֵי הָאָנוּס וְהַתַּשְׁלוּמִין וּמִי שֶׁרָגִיל בְּכָךְ. ערוה״ש פ׳ — גֶּדֶר גּוּף נָקִי בַּעֲמִידָה לִפְנֵי הַמֶּלֶךְ. דרך אדמו״ר הזקן — בָּרִי לוֹ שֶׁאֵינוֹ יָכוֹל לַעֲמֹד עַד גְּמַר י״ח — מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה ; תַּשְׁלוּמִין בְּמִנְחָה שְׁתַּיִם ; גְּנַאי גָּדוֹל, ד׳ אַמּוֹת (סימן ק״ג) (סעיף א).
פְּסוּקֵי דְּזִמְרָה או״ח פ׳ — לֹא נִתְפָּרֵשׁ בַּמְּחַבֵּר ; עִקָּרוֹ בְּזְמַן קְרִיאַת שְׁמַע וּתְפִלָּה. מ״ב פ׳ — דִּין אֲמִירַת פְּסוּקֵי דְּזִמְרָה לְמִי שֶׁאֵינוֹ מִתְפַּלֵּל. ערוה״ש פ׳ — אֲמִירָתָן בְּלֹא בְּרָכוֹת כְּקוֹרֵא בַּתּוֹרָה. דרך אדמו״ר הזקן — כֵּיוָן שֶׁאֵינוֹ מִתְפַּלֵּל י״ח — לֹא יֹאמַר פְּסוּקֵי דְּזִמְרָה, אֶלָּא בְּלֹא בִּרְכוֹתֶיהָ כְּקוֹרֵא בַּתּוֹרָה (סימן נ״א נ״ב) (סעיף ב).
קְרִיאַת שְׁמַע וְהַתְּפִלִּין או״ח פ׳ — יָנִיחַ תְּפִלִּין בֵּין אַהֲבָה לִק״ש ; הַגָּה (רמ״א) — וּמְבָרֵךְ עֲלֵיהֶם. מ״ב פ׳ — קְרִיאַת שְׁמַע בִּזְמַנָּהּ וּתְפִלִּין בֵּין אַהֲבָה לִק״ש. ערוה״ש פ׳ — דִּין הַטַּלִּית וְהַתְּפִלִּין בְּלֹא גּוּף נָקִי. דרך אדמו״ר הזקן — ק״ש וּבִרְכוֹתֶיהָ בִּזְמַנָּן אֲפִלּוּ בְּלֹא גּוּף נָקִי (אֵינוֹ מְדַבֵּר עִם הַמֶּלֶךְ, סימן ע״ד) ; תְּפִלִּין בֵּין ״אַהֲבָה״ לִק״ש וּמְבָרֵךְ וְחוֹלֵץ מִיָּד ; טַלִּית מֻתָּר (סעיף ג).

הטבלה נערכה מטקסט המחבר (אורח חיים פ׳, מקור ספריא), מהגהת הרמ״א, ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפלה ד, מגן אברהם, ט״ז, אליה רבה, פרי מגדים, משנה ברורה, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ב–כ״ג). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ג׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — גּוּף נָקִי בַּתְּפִלָּה

קו א — גּוּף נָקִי בַּתְּפִלָּה

יסוד הסימן : בָּרִי לוֹ שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר עַד גְּמַר י״ח — מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה (סעיף א).

קריאת הרב : הָעֲמִידָה בִּתְפִלַּת י״ח הִיא מְדַבֵּר עִם הַמֶּלֶךְ — וְהַפִיחַ בָּהּ בִּזָּיוֹן וָקָצֶף. עיין סימן ע״ד · 74.

מסקנה למעשה : הָאָנוּס מִתְפַּלֵּל מִנְחָה שְׁתַּיִם ; אֵין מִתְפַּלְּלִין י״ח בְּלֹא גּוּף נָקִי.

קו ב — פְּסוּקֵי דְּזִמְרָה

קו ב — פְּסוּקֵי דְּזִמְרָה

יסוד הסימן : כֵּיוָן שֶׁאֵינוֹ מִתְפַּלֵּל — לֹא יֹאמַר פְּסוּקֵי דְּזִמְרָה, אֶלָּא בְּלֹא בִּרְכוֹתֶיהָ (סעיף ב).

קריאת הרב : בִּרְכַּת פְּסוּקֵי דְּזִמְרָה נִתְקְנָה לְסַדֵּר שְׁבָחָיו קֹדֶם הַתְּפִלָּה. עיין סימן נ״א · 51.

מסקנה למעשה : בְּלֹא תְּפִלָּה — מַשְׁמִיט הַבְּרָכוֹת אוֹ קוֹרֵא כְּקוֹרֵא בַּתּוֹרָה.

קו ג — קְרִיאַת שְׁמַע וְהַתְּפִלִּין

קו ג — קְרִיאַת שְׁמַע וְהַתְּפִלִּין

יסוד הסימן : ק״ש וּבִרְכוֹתֶיהָ בִּזְמַנָּן אֲפִלּוּ בְּלֹא גּוּף נָקִי ; תְּפִלִּין בֵּין ״אַהֲבָה״ לִק״ש (סעיף ג).

קריאת הרב : ק״ש אֵינָהּ מְדַבֵּר עִם הַמֶּלֶךְ (סימן ע״ד) ; אֵין מַעֲבִירִין זְמַנָּהּ בִּשְׁבִיל תְּפִלִּין (סימן ס״ו). עיין סימן ס״ו · 66.

מסקנה למעשה : תְּפִלִּין בֵּין ״אַהֲבָה״ לִק״ש וְאַחַר כָּךְ חוֹלֵץ ; הַטַּלִּית מֻתֶּרֶת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן פ׳ ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן פ׳ (דִּין מִי שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר מֵהָפִיחַ)

  • ① גּוּף נָקִי בָּעֲמִידָה. בָּרִי לוֹ שֶׁאֵינוֹ יָכוֹל לְהִשָּׁמֵר עַד גְּמַר י״ח — מוּטָב שֶׁיַּעֲבֹר זְמַן הַתְּפִלָּה מִלְּהִתְפַּלֵּל בְּלֹא גּוּף נָקִי. יסוד : שוע״ר פ׳ סעיף א.
  • ② תַּשְׁלוּמִין וְאָנוּס. אָנוּס הוּא ; חָסֵר שַׁחֲרִית — מִתְפַּלֵּל מִנְחָה שְׁתַּיִם (גְּנַאי גָּדוֹל לִפְנֵי הַמֶּלֶךְ, ד׳ אַמּוֹת, סימן ק״ג). יסוד : שוע״ר פ׳ סעיף א.
  • ③ פְּסוּקֵי דְּזִמְרָה. כֵּיוָן שֶׁאֵינוֹ מִתְפַּלֵּל י״ח — לֹא יֹאמַר, אוֹ בְּלֹא בִּרְכוֹתֶיהָ כְּקוֹרֵא בַּתּוֹרָה (סימן נ״א נ״ב). יסוד : שוע״ר פ׳ סעיף ב.
  • ④ קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ. בִּזְמַנָּן אֲפִלּוּ בְּלֹא גּוּף נָקִי (אֵינוֹ מְדַבֵּר עִם הַמֶּלֶךְ, סימן ע״ד) ; אֵין מַעֲבִירִין זְמַן ק״ש בִּשְׁבִיל תְּפִלִּין (סימן ס״ו). יסוד : שוע״ר פ׳ סעיף ג.
  • ⑤ תְּפִלִּין וְטַלִּית. יָכוֹל לַעֲמֹד עַל עַצְמוֹ בִּשְׁעַת ק״ש — תְּפִלִּין בֵּין ״אַהֲבָה״ לִק״ש, בְּרָכָה, וְאַחַר כָּךְ חוֹלֵץ מִיָּד ; הַטַּלִּית נִלְבֶּשֶׁת וּמְבֹרֶכֶת (אֵין צָרִיךְ גּוּף נָקִי לַצִּיצִית). יסוד : שוע״ר פ׳ סעיף ג.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן פ׳ (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַגּוּף נָקִי בַּתְּפִלָּה, הַפְּסוּקֵי דְּזִמְרָה, הַקְּרִיאַת שְׁמַע — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
