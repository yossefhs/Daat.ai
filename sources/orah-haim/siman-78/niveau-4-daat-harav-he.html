דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ע״ח · דִּין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה

דרכו של אדמו״ר הזקן בדִּין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה : מֵי רַגְלָיו שׁוֹתְתִין — פּוֹסֵק עַד שֶׁיִּכְלוּ וְחוֹזֵר ; נָפְלוּ בָּאָרֶץ — מַרְחִיק ד׳ אַמּוֹת אוֹ מַמְתִּין עַד שֶׁיִּבָּלְעוּ ; בֶּגֶד עֶלְיוֹן נָקִי אוֹ מְגֻלֶּה ; הַחוֹלֶה הַמְטַפְטֵף לְאָנְסוֹ — בֶּגֶד סְבִיב אֵיבָרוֹ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ע״ח — דִּין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה. אצל הרב הסימן נפרש לד׳ סְעִיפִים : הַפּוֹסֵק בִּשְׁעַת שְׁתִיתָה וְהַשָּׁהָה כְּדֵי לִגְמֹר, הַמַּיִם שֶׁנָּפְלוּ בָּאָרֶץ (הַרְחָקָה אוֹ הַמְתָּנָה) וְהַחִלּוּק בֵּין ק״ש לִתְפִלָּה, הַבֶּגֶד הָעֶלְיוֹן נָקִי מוּל מְגֻלֶּה, וְהַחוֹלֶה הַמְטַפְטֵף מֵי רַגְלַיִם לְאָנְסוֹ.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ע״ח — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ע״ח). אדמו״ר הזקן דן בו בְּדִין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ע״ח — דִּין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ע״ח — דִּין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.78), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.78.k).

סעיף א מֵי רַגְלָיו שׁוֹתְתִין — פּוֹסֵק עַד שֶׁיִּכְלוּ וְחוֹזֵר ; שָׁהָה כְּדֵי לִגְמֹר — לָרֹאשׁ.

מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה וּבוֹ ד׳ סְעִיפִים:הָיָה קוֹרֵא קְרִיאַת שְׁמַע וְהִתְחִילוּ מֵי רַגְלָיו שׁוֹתְתִין עַל בִּרְכָּיו — פּוֹסֵק עַד שֶׁיִּכְלוּ לִשְׁתֹת, וְחוֹזֵר לִקְרוֹת, אֲפִלּוּ נָפְלוּ עַל בְּגָדָיו וְיֵשׁ בָּהֶם טוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ, כֵּיוָן שֶׁהֵם מְכֻסִים בַּבֶּגֶד הָעֶלְיוֹן. וְאִם שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ — חוֹזֵר לָרֹאשׁ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ס״ה:

פּוֹסֵק עַד שֶׁיִּכְלוּ. הָיָה קוֹרֵא ק״ש וְהִתְחִילוּ מֵי רַגְלָיו שׁוֹתְתִין עַל בִּרְכָּיו — פּוֹסֵק עַד שֶׁיִּכְלוּ לִשְׁתֹת וְחוֹזֵר לִקְרוֹת ; וַאֲפִלּוּ יֵשׁ טוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ, כֵּיוָן שֶׁהוּא מְכֻסֶּה בַּבֶּגֶד הָעֶלְיוֹן. וְאִם שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ — חוֹזֵר לָרֹאשׁ (כְּסִימָן ס״ה).

סעיף ב נָפְלוּ בָּאָרֶץ — מַרְחִיק ד׳ אַמּוֹת אוֹ מַמְתִּין עַד שֶׁיִּבָּלְעוּ ; חִלּוּק בֵּין ק״ש לִתְפִלָּה.

וְאִם נָפְלוּ מֵי רַגְלַיִם בָּאָרֶץ — מַרְחִיק מֵהֶם ד׳ אַמּוֹת. וְאִם הוּא מָקוֹם שֶׁיְּכוֹלִין לִבָּלַע בַּקַּרְקַע בִּמְהֵרָה — יָכוֹל לְהַמְתִּין עַד שֶׁיִּבָּלְעוּ, וְאֵין חוֹשְׁשִׁין שֶׁמָּא יִשְׁהֶה בְּהַמְתָּנָה זוֹ כְּדֵי לִגְמֹר אֶת כֻּלָּהּ וְיִצְטָרֵךְ לַחֲזֹר לָרֹאשׁ, לְפִי שֶׁאַף אִם יִהְיֶה כֵּן — אֵין כָּאן אִסּוּר בַּמֶּה שֶׁקּוֹרֵא קְרִיאַת שְׁמַע ב׳ פְּעָמִים, מִפְּנֵי שֶׁהוּא כְּקוֹרֵא בַּתּוֹרָה (וּמִכָּל מָקוֹם לְכַתְּחִלָּה אֵין לִשְׁהוֹת כָּל כָּךְ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן פ״ה). אֲבָל אִם אֵרַע כֵּן כְּשֶׁעוֹמֵד בַּתְּפִלָּה — לֹא יַמְתִּין עַד שֶׁיִּבָּלְעוּ, כִּי שֶׁמָּא יִשְׁהֶה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ וְיִצְטָרֵךְ לַחֲזֹר לָרֹאשׁ וְיִגְרֹם בְּרָכָה שֶׁאֵינָהּ צְרִיכָה, אֶלָּא יַרְחִיק ד׳ אַמּוֹת אוֹ יֵצֵא לְחֶדֶר אַחֵר וְיִגְמֹר שָׁם תְּפִלָּתוֹ:

נָפְלוּ בָּאָרֶץ. נָפְלוּ מֵי רַגְלַיִם בָּאָרֶץ — מַרְחִיק ד׳ אַמּוֹת ; מָקוֹם שֶׁנִּבְלָע בִּמְהֵרָה — יָכוֹל לְהַמְתִּין עַד שֶׁיִּבָּלְעוּ. וּבַ־ק״ש אֵין חוֹשְׁשִׁין לַשְּׁהִיָּה, שֶׁהוּא כְּקוֹרֵא בַּתּוֹרָה (וּמִכָּל מָקוֹם לְכַתְּחִלָּה אֵין לִשְׁהוֹת, סִימָן פ״ה). אֲבָל בַּתְּפִלָּה — לֹא יַמְתִּין, מִשּׁוּם בְּרָכָה שֶׁאֵינָהּ צְרִיכָה, אֶלָּא יַרְחִיק ד׳ אַמּוֹת אוֹ יֵצֵא לְחֶדֶר אַחֵר.

סעיף ג בֶּגֶד עֶלְיוֹן נָקִי — מַרְחִיק ; מֵי רַגְלַיִם מְגֻלִּין — גּוֹמֵר בִּמְקוֹמוֹ.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁבִּגְדוֹ הָעֶלְיוֹן נָקִי וּמְכַסֶּה עַל מֵי רַגְלַיִם שֶׁנָּפְלוּ עַל בְּגָדִים הַתַּחְתּוֹנִים, אֲבָל אִם לִכְלוּךְ מֵי רַגְלַיִם שֶׁבִּבְגָדָיו הוּא מְגֻלֶּה, שֶׁאַף אִם יִרְחַק ד׳ אַמּוֹת לֹא יִתְפַּלֵּל בִּנְקִיּוּת — לֹא יַפְסִיק תְּפִלָּתוֹ כְּלָל, אֶלָּא גּוֹמְרָהּ בְּמָקוֹם שֶׁעוֹמֵד שָׁם אַף עַל פִּי שֶׁמֵּי רַגְלָיו מְגֻלִּין. וְאֵין צָרִיךְ לְהַחֲלִיף בְּגָדָיו אֲפִלּוּ יֵשׁ לוֹ, שֶׁכֵּיוָן שֶׁלְּאַחַר שֶׁנָּפְלוּ מִמֶּנּוּ מֵי רַגְלָיו אֵין אִסּוּרָן אֶלָּא מִדִּבְרֵי סוֹפְרִים — לֹא הִטְרִיחוּהוּ לְהַפְסִיק תְּפִלָּתוֹ וְלִלְבֹּשׁ בְּגָדִים אֲחֵרִים, כֵּיוָן שֶׁהוּא עוֹמֵד כְּבָר בִּתְפִלָּתוֹ שֶׁאָסוּר לְהַפְסִיק בָּהּ בִּלְבִישַׁת הַבֶּגֶד. וְגַם אִם בָּא לְהַחֲמִיר עַל עַצְמוֹ — אֵינוֹ רַשַּׁאי. מַה שֶּׁאֵין כֵּן בְּהַרְחָקַת ד׳ אַמּוֹת כְּשֶׁבִּגְדוֹ הָעֶלְיוֹן נָקִי — אֵין בָּהּ טֹרַח כְּלָל (וְגַם אֵינָהּ חֲשׁוּבָה הֶפְסֵק כָּל כָּךְ כְּמוֹ לְבִישַׁת בֶּגֶד):

בֶּגֶד עֶלְיוֹן נָקִי מוּל מְגֻלֶּה. כָּל זֶה כְּשֶׁהַבֶּגֶד הָעֶלְיוֹן נָקִי וּמְכַסֶּה. אֲבָל אִם הַלִּכְלוּךְ מְגֻלֶּה, שֶׁאַף בְּהַרְחָקַת ד׳ אַמּוֹת לֹא יִתְפַּלֵּל בִּנְקִיּוּת — לֹא יַפְסִיק תְּפִלָּתוֹ כְּלָל אֶלָּא גּוֹמְרָהּ בִּמְקוֹמוֹ. וְאֵין צָרִיךְ לְהַחֲלִיף בְּגָדָיו אֲפִלּוּ יֵשׁ לוֹ, וְאַף לְהַחֲמִיר אֵינוֹ רַשַּׁאי, שֶׁאֵין אִסּוּרָן אֶלָּא מִדִּבְרֵי סוֹפְרִים.

סעיף ד חוֹלֶה הַמְטַפְטֵף לְאָנְסוֹ — בֶּגֶד סְבִיב אֵיבָרוֹ, פּוֹסֵק בִּשְׁעַת הַטָּפָה וְחוֹזֵר.

מִי שֶׁיֵּשׁ לוֹ חֳלִי שֶׁמְּטַפְטֵף תָּמִיד מֵי רַגְלַיִם לְאָנְסוֹ, וְאִם כֵּן לֹא יוּכַל לְהִתְפַּלֵּל לְעוֹלָם — יַעֲשֶׂה לוֹ בֶּגֶד סְבִיב אֵיבָרוֹ, כְּדֵי שֶׁבְּגָדִים הָעֶלְיוֹנִים יִהְיוּ נְקִיִּים וְיִכָּנֵס לְבֵית הַכְּנֶסֶת וְיִתְפַּלֵּל. וּבִשְׁעַת הַטָּפַת מֵי רַגְלַיִם — יַפְסִיק וְאַחַר כָּךְ יַחֲזֹר לְמָקוֹם שֶׁפָּסַק, רַק יִזָּהֵר שֶׁלֹּא יַתְחִיל לְהִתְפַּלֵּל כְּשֶׁהֻצְרַךְ כְּבָר לְטַפְטֵף. וְאַף עַל פִּי שֶׁיּוֹדֵעַ שֶׁבְּתוֹךְ תְּפִלָּתוֹ יִצְטָרֵךְ לְטַפְטֵף — אֵין בְּכָךְ כְּלוּם, כֵּיוָן שֶׁאִי אֶפְשָׁר לוֹ לְהַמְתִּין עַד שֶׁיִּטַּהֵר (עַיֵּן סִימָן פ׳):

הַחוֹלֶה הַמְטַפְטֵף. חֳלִי שֶׁמְּטַפְטֵף תָּמִיד לְאָנְסוֹ, שֶׁאִם לֹא כֵן לֹא יִתְפַּלֵּל לְעוֹלָם — יַעֲשֶׂה בֶּגֶד סְבִיב אֵיבָרוֹ שֶׁהַבְּגָדִים הָעֶלְיוֹנִים נְקִיִּים, וְיִכָּנֵס וְיִתְפַּלֵּל. וּבִשְׁעַת הַטָּפָה — יַפְסִיק וְיַחֲזֹר לְמָקוֹם שֶׁפָּסַק, וְרַק יִזָּהֵר שֶׁלֹּא יַתְחִיל כְּשֶׁכְּבָר הֻצְרַךְ לְטַפְטֵף ; שֶׁאִי אֶפְשָׁר לוֹ לְהַמְתִּין עַד שֶׁיִּטַּהֵר (עַיֵּן סִימָן פ׳).

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ע״ח (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה

היסוד : דִּין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַפּוֹסֵק בִּשְׁעַת שְׁתִיתָה וְהַשָּׁהָה כְּדֵי לִגְמֹר. אחר כך הַמַּיִם שֶׁנָּפְלוּ בָּאָרֶץ (הַרְחָקָה אוֹ הַמְתָּנָה) וְהַחִלּוּק בֵּין ק״ש לִתְפִלָּה. אחר כך הַבֶּגֶד הָעֶלְיוֹן נָקִי מוּל מְגֻלֶּה. ולבסוף הַחוֹלֶה הַמְטַפְטֵף מֵי רַגְלַיִם לְאָנְסוֹ.

פּוֹסֵק בִּשְׁעַת שְׁתִיתָה — וְחוֹזֵר

פּוֹסֵק בִּשְׁעַת שְׁתִיתָה — וְחוֹזֵר

היסוד. מֵי רַגְלָיו שׁוֹתְתִין עַל בִּרְכָּיו — פּוֹסֵק עַד שֶׁיִּכְלוּ לִשְׁתֹת וְחוֹזֵר לִקְרוֹת ; אֲפִלּוּ טוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ, כְּשֶׁמְּכֻסֶּה בַּבֶּגֶד הָעֶלְיוֹן (סעיף א).

המסקנה. וְאִם שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ — חוֹזֵר לָרֹאשׁ. עיין סימן ס״ה · 65 (שהה כדי לגמור).

נָפְלוּ בָּאָרֶץ — הַרְחָקָה אוֹ הַמְתָּנָה

נָפְלוּ בָּאָרֶץ — הַרְחָקָה אוֹ הַמְתָּנָה

היסוד. מֵי רַגְלַיִם בָּאָרֶץ — מַרְחִיק ד׳ אַמּוֹת ; מָקוֹם שֶׁנִּבְלָע בִּמְהֵרָה — יָכוֹל לְהַמְתִּין עַד שֶׁיִּבָּלְעוּ (סעיף ב).

המסקנה. חִלּוּק : בַּ־ק״ש אֵין חוֹשְׁשִׁין (כְּקוֹרֵא בַּתּוֹרָה) ; בַּתְּפִלָּה לֹא יַמְתִּין, מִשּׁוּם בְּרָכָה שֶׁאֵינָהּ צְרִיכָה. עיין סימן ע״ט · 79 (מרחק מן הצואה).

בֶּגֶד עֶלְיוֹן — נָקִי מוּל מְגֻלֶּה

בֶּגֶד עֶלְיוֹן — נָקִי מוּל מְגֻלֶּה

היסוד. בֶּגֶד עֶלְיוֹן נָקִי וּמְכַסֶּה — מַרְחִיק ; אֲבָל מְגֻלֶּה — לֹא יַפְסִיק תְּפִלָּתוֹ כְּלָל, גּוֹמְרָהּ בִּמְקוֹמוֹ (סעיף ג).

המסקנה. אֵין צָרִיךְ לְהַחֲלִיף בְּגָדָיו, וְאַף לְהַחֲמִיר אֵינוֹ רַשַּׁאי (אִסּוּר מִדִּבְרֵי סוֹפְרִים). עיין סימן ע״ו · 76 (נקיות הגוף בתפילה).

חוֹלֶה הַמְטַפְטֵף — בֶּגֶד סְבִיב אֵיבָרוֹ

חוֹלֶה הַמְטַפְטֵף — בֶּגֶד סְבִיב אֵיבָרוֹ

היסוד. חֳלִי שֶׁמְּטַפְטֵף תָּמִיד לְאָנְסוֹ — יַעֲשֶׂה בֶּגֶד סְבִיב אֵיבָרוֹ, שֶׁהַבְּגָדִים הָעֶלְיוֹנִים נְקִיִּים, וְיִתְפַּלֵּל (סעיף ד).

המסקנה. בִּשְׁעַת הַטָּפָה — יַפְסִיק וְיַחֲזֹר לְמָקוֹם שֶׁפָּסַק ; אִי אֶפְשָׁר לוֹ לְהַמְתִּין עַד שֶׁיִּטַּהֵר. עיין סימן ע״ז · 77 (מי רגלים שותתין).

הפניה ללימוד. היקף הסימן — הַפּוֹסֵק, הַהַרְחָקָה וְדִין הַחוֹלֶה — נוגע להנהגת ק״ש והתפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ע״ח. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
מֵי רַגְלַיִם שׁוֹתְתִין בִּשְׁעַת ק״ש או״ח ע״ח — פּוֹסֵק עַד שֶׁיִּכְלוּ הַמַּיִם וְחוֹזֵר לִקְרוֹת, אַף שֶׁיֵּשׁ טוֹפֵחַ הַמְכֻסֶּה בַּבֶּגֶד. מ״ב ע״ח — גֶּדֶר הַהֶפְסֵק וְהַחֲזָרָה, וּשְׁהִיָּה כְּדֵי לִגְמֹר. ערוה״ש ע״ח — דִּין הַשּׁוֹתֵת וְהַמְכֻסֶּה בַּבֶּגֶד הָעֶלְיוֹן. דרך אדמו״ר הזקן — פּוֹסֵק עַד שֶׁיִּכְלוּ לִשְׁתֹת וְחוֹזֵר ; שָׁהָה כְּדֵי לִגְמֹר אֶת כֻּלָּהּ — חוֹזֵר לָרֹאשׁ (סעיף א).
נָפְלוּ בָּאָרֶץ — הַרְחָקָה אוֹ הַמְתָּנָה או״ח ע״ח — מַרְחִיק ד׳ אַמּוֹת ; אוֹ מַמְתִּין עַד שֶׁיִּבָּלְעוּ בַּקַּרְקַע. מ״ב ע״ח — שִׁעוּר הַהַרְחָקָה וּפְרָטֵי הַהַמְתָּנָה. ערוה״ש ע״ח — גֶּדֶר בְּלִיעַת הַמַּיִם בַּקַּרְקַע. דרך אדמו״ר הזקן — בַּ־ק״ש אֵין חוֹשְׁשִׁין (כְּקוֹרֵא בַּתּוֹרָה) ; בַּתְּפִלָּה לֹא יַמְתִּין, מִשּׁוּם בְּרָכָה שֶׁאֵינָהּ צְרִיכָה (סעיף ב).
בֶּגֶד עֶלְיוֹן נָקִי מוּל מְגֻלֶּה או״ח ע״ח — כְּשֶׁמְּכֻסֶּה מַרְחִיק ד׳ אַמּוֹת ; לִכְלוּךְ מְגֻלֶּה בַּתְּפִלָּה. מ״ב ע״ח — כְּשֶׁמֵּי רַגְלָיו מְגֻלִּין גּוֹמֵר בִּמְקוֹמוֹ. ערוה״ש ע״ח — הֶפְסֵק הַתְּפִלָּה בְּלִכְלוּךְ מְגֻלֶּה. דרך אדמו״ר הזקן — מְגֻלֶּה — לֹא יַפְסִיק כְּלָל אֶלָּא גּוֹמְרָהּ בִּמְקוֹמוֹ ; אֵין מַחֲלִיף בְּגָדָיו וְאַף לְהַחֲמִיר אֵינוֹ רַשַּׁאי (סעיף ג).
חוֹלֶה הַמְטַפְטֵף לְאָנְסוֹ או״ח ע״ח — עוֹשֶׂה בֶּגֶד סְבִיב אֵיבָרוֹ וּמִתְפַּלֵּל ; פּוֹסֵק בִּשְׁעַת הַטָּפָה. מ״ב ע״ח — דִּין הַחוֹלֶה שֶׁאִי אֶפְשָׁר לוֹ לְהִזָּהֵר. ערוה״ש ע״ח — הַנְהָגַת הַמְטַפְטֵף בְּאֹנֶס. דרך אדמו״ר הזקן — בֶּגֶד סְבִיב אֵיבָרוֹ ; יַפְסִיק בִּשְׁעַת הַטָּפָה וְיַחֲזֹר לְמָקוֹם שֶׁפָּסַק ; אִי אֶפְשָׁר לְהַמְתִּין עַד שֶׁיִּטַּהֵר (סעיף ד).

הטבלה נערכה מטקסט המחבר (אורח חיים ע״ח, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג והלכות תפילה ד, רא״ש, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ה). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — פּוֹסֵק וְחוֹזֵר

קו א — פּוֹסֵק וְחוֹזֵר

יסוד הסימן : מֵי רַגְלָיו שׁוֹתְתִין — פּוֹסֵק עַד שֶׁיִּכְלוּ לִשְׁתֹת וְחוֹזֵר לִקְרוֹת (סעיף א).

קריאת הרב : הַטּוֹפֵחַ הַמְכֻסֶּה בַּבֶּגֶד הָעֶלְיוֹן אֵינוֹ מְעַכֵּב ; אֲבָל שָׁהָה כְּדֵי לִגְמֹר — חוֹזֵר לָרֹאשׁ. עיין סימן ס״ה · 65.

מסקנה למעשה : לְהַפְסִיק עַד שֶׁתִּכְלֶה הַשְּׁתִיתָה, וְאַחַר כָּךְ לַחֲזֹר לְמָקוֹם שֶׁפָּסַק.

קו ב — הַרְחָקָה מוּל הַמְתָּנָה

קו ב — הַרְחָקָה מוּל הַמְתָּנָה

יסוד הסימן : נָפְלוּ בָּאָרֶץ — מַרְחִיק ד׳ אַמּוֹת אוֹ מַמְתִּין עַד שֶׁיִּבָּלְעוּ (סעיף ב).

קריאת הרב : חִלּוּק בֵּין ק״ש (כְּקוֹרֵא בַּתּוֹרָה, אֵין חוֹשְׁשִׁין) לִתְפִלָּה (לֹא יַמְתִּין, בְּרָכָה שֶׁאֵינָהּ צְרִיכָה). עיין סימן ע״ט · 79.

מסקנה למעשה : בַּתְּפִלָּה לֹא יַמְתִּין לַבְּלִיעָה — יַרְחִיק ד׳ אַמּוֹת אוֹ יֵצֵא לְחֶדֶר אַחֵר.

קו ג — נְקִיּוּת וְאֹנֶס

יסוד הסימן : מְגֻלֶּה — לֹא יַפְסִיק אֶלָּא גּוֹמְרָהּ בִּמְקוֹמוֹ ; חוֹלֶה הַמְטַפְטֵף לְאָנְסוֹ — בֶּגֶד סְבִיב אֵיבָרוֹ (סעיפים ג-ד).

קריאת הרב : אִסּוּר הַלִּכְלוּךְ שֶׁלְּאַחַר מַעֲשֶׂה אֵינוֹ אֶלָּא מִדִּבְרֵי סוֹפְרִים — וְלָכֵן אֵין טֹרַח וְאָסוּר לְהַחֲמִיר. עיין סימן ע״ז · 77.

מסקנה למעשה : אֵין מַפְסִיקִין עַל לִכְלוּךְ מְגֻלֶּה וְלֹא לְהַחֲלִיף בְּגָדִים ; הַחוֹלֶה מִתְפַּלֵּל עִם בֶּגֶד הַמָּגֵן.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ע״ח ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ע״ח (דִּין מִי שֶׁנִּצְטָרֵךְ לְהָטִיל מַיִם בִּשְׁעַת קְרִיאָה)

  • ① מֵי רַגְלָיו שׁוֹתְתִין. נִשְׁפְּכוּ מֵי רַגְלָיו בִּשְׁעַת ק״ש — פּוֹסֵק עַד שֶׁיִּכְלוּ וְחוֹזֵר לִקְרוֹת ; הַטּוֹפֵחַ הַמְכֻסֶּה בַּבֶּגֶד הָעֶלְיוֹן אֵינוֹ מְעַכֵּב. יסוד : שוע״ר ע״ח סעיף א.
  • ② שָׁהָה כְּדֵי לִגְמֹר. שָׁהָה כְּדֵי לִגְמֹר אֶת כָּל הַק״ש — חוֹזֵר לָרֹאשׁ (כְּסִימָן ס״ה). יסוד : שוע״ר ע״ח סעיף א.
  • ③ נָפְלוּ בָּאָרֶץ. נָפְלוּ בָּאָרֶץ — מַרְחִיק ד׳ אַמּוֹת אוֹ מַמְתִּין לַבְּלִיעָה ; אֲבָל בַּתְּפִלָּה לֹא יַמְתִּין (בְּרָכָה שֶׁאֵינָהּ צְרִיכָה). יסוד : שוע״ר ע״ח סעיף ב.
  • ④ בֶּגֶד מְגֻלֶּה. לִכְלוּךְ מְגֻלֶּה בַּתְּפִלָּה — לֹא יַפְסִיק כְּלָל אֶלָּא גּוֹמְרָהּ בִּמְקוֹמוֹ ; אֵין מַחֲלִיף בְּגָדָיו וְאָסוּר לְהַחֲמִיר. יסוד : שוע״ר ע״ח סעיף ג.
  • ⑤ חוֹלֶה הַמְטַפְטֵף. חוֹלֶה הַמְטַפְטֵף לְאָנְסוֹ — בֶּגֶד סְבִיב אֵיבָרוֹ, מִתְפַּלֵּל, יַפְסִיק בִּשְׁעַת הַטָּפָה וְיַחֲזֹר לְמָקוֹם שֶׁפָּסַק. יסוד : שוע״ר ע״ח סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ע״ח (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַפּוֹסֵק, הַהַרְחָקָה, דִּין הַחוֹלֶה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
