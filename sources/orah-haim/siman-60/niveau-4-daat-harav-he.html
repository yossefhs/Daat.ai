דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ס׳ · דִּין בְּרָכוֹת קְרִיאַת שְׁמַע וְאִם מִצְווֹת צְרִיכוֹת כַּוָּנָה

דרכו של אדמו״ר הזקן בבְּרָכוֹת קְרִיאַת שְׁמַע וּבַכְּלָל מִצְווֹת צְרִיכוֹת כַּוָּנָה : הַבְּרָכָה הַשְּׁנִיָּה (אַהֲבָה רַבָּה / אַהֲבַת עוֹלָם), הַבְּרָכוֹת שֶׁאֵינָן מְעַכְּבוֹת, הַזְּכִירוֹת, וְכַוָּנַת הַפָּסוּק הָרִאשׁוֹן — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. הרבי הורה במפורש ללמוד ברצינות את שולחן ערוך הרב. עמוד זה מאסף את דרך הרב בסימן ס׳ — דִּין בְּרָכוֹת קְרִיאַת שְׁמַע וְאִם מִצְווֹת צְרִיכוֹת כַּוָּנָה. אצל הרב הסימן נפרש לה׳ סְעִיפִים : הַבְּרָכָה הַשְּׁנִיָּה (אַהֲבָה רַבָּה / אַהֲבַת עוֹלָם), הַבְּרָכוֹת וְסֵדֶר הַבְּרָכוֹת שֶׁאֵינָן מְעַכְּבוֹת, הַזְּכִירוֹת אֵצֶל קְרִיאַת שְׁמַע, וְהַכְּלָל הַגָּדוֹל מִצְווֹת צְרִיכוֹת כַּוָּנָה עִם כַּוָּנַת הַפָּסוּק הָרִאשׁוֹן.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ס׳ — ה׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ס׳). אדמו״ר הזקן דן בו בדִּין בְּרָכוֹת קְרִיאַת שְׁמַע וְאִם מִצְווֹת צְרִיכוֹת כַּוָּנָה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ס׳ — דִּין בְּרָכוֹת קְרִיאַת שְׁמַע וְאִם מִצְווֹת צְרִיכוֹת כַּוָּנָה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ס׳ — דִּין בְּרָכוֹת קְרִיאַת שְׁמַע וְאִם מִצְווֹת צְרִיכוֹת כַּוָּנָה — וּבוֹ ה׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.60), ואחריו תרגומנו. הסימן כולל חמישה סעיפים (ה׳ סְעִיפִים ; סעיף k → ‎.60.k).

סעיף א בְּרָכָה שְׁנִיָּה נֶחְלְקוּ חֲכָמִים בִּפְתִיחָתָהּ — אַהֲבָה רַבָּה / אַהֲבַת עוֹלָם.

דִּין בְּרָכוֹת לִקְרִיאַת שְׁמַע וְאִם צְרִיכִים כַּוָּנָה וּבוֹ ה׳ סְעִיפִים:בְּרָכָה שְׁנִיָּה נֶחְלְקוּ חֲכָמִים בִּפְתִיחָתָהּ, יֵשׁ אוֹמְרִים ״אַהֲבָה רַבָּה״, וְיֵשׁ אוֹמְרִים ״אַהֲבַת עוֹלָם אֲהַבְתָּנוּ כו׳״. וְהַגְּאוֹנִים הִכְרִיעוּ לוֹמַר בְּשַׁחֲרִית ״אַהֲבָה רַבָּה״ וּבְעַרְבִית ״אַהֲבַת עוֹלָם״. וְכֵן נוֹהֲגִין בִּמְדִינוֹת אֵלּוּ. וְלָמָּה אֵין בְּרָכָה זוֹ פּוֹתַחַת בְּ״בָרוּךְ״? מִפְּנֵי שֶׁהִיא סְמוּכָה לְ״יוֹצֵר אוֹר״ וְדַי לָהּ בְּ״בָרוּךְ״ שֶׁבְּ״יוֹצֵר אוֹר״:

הַבְּרָכָה הַשְּׁנִיָּה. נֶחְלְקוּ חֲכָמִים בִּפְתִיחָתָהּ — יֵשׁ אוֹמְרִים ״אַהֲבָה רַבָּה״, יֵשׁ אוֹמְרִים ״אַהֲבַת עוֹלָם״. הַגְּאוֹנִים הִכְרִיעוּ : שַׁחֲרִית ״אַהֲבָה רַבָּה״, עַרְבִית ״אַהֲבַת עוֹלָם״ — וְכֵן נוֹהֲגִין. וְלָמָּה אֵינָהּ פּוֹתַחַת בְּ״בָרוּךְ״ ? מִפְּנֵי שֶׁהִיא סְמוּכָה לְ״יוֹצֵר אוֹר״, וְדַי לָהּ בְּ״בָרוּךְ״ שֶׁבְּ״יוֹצֵר אוֹר״.

סעיף ב הַבְּרָכוֹת אֵינָן מְעַכְּבוֹת אֶת ק״ש — קְרָאָהּ בְּלֹא בִּרְכוֹתֶיהָ יָצָא.

הַבְּרָכוֹת אֵינָן מְעַכְּבוֹת אֶת קְרִיאַת שְׁמַע, שֶׁאִם קְרָאָהּ בְּלֹא בִּרְכוֹתֶיהָ — יָצָא יְדֵי חוֹבַת קְרִיאַת שְׁמַע, וְחוֹזֵר וְקוֹרֵא הַבְּרָכוֹת בְּלֹא קְרִיאַת שְׁמַע, שֶׁבִּרְכוֹת קְרִיאַת שְׁמַע אֵינָן כְּבִרְכוֹת שְׁאָר מִּצְווֹת שֶׁאֵין מְבָרְכִין אוֹתָן אֶלָּא עוֹבֵר לַעֲשִׂיָּתָן, שֶׁהֲרֵי אֵינוֹ מְבָרֵךְ ״אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ לִקְרוֹת אֶת שְׁמַע״, וּכְמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן נ״ט. וְאַף עַל פִּי כֵן טוֹב לַחֲזֹר וְלִקְרוֹת קְרִיאַת שְׁמַע עִם הַבְּרָכוֹת. וְאִם לֹא חָזַר וְקָרָא לֹא הַקְרִיאַת שְׁמַע וְלֹא הַבְּרָכוֹת — יָצָא יְדֵי חוֹבַת קְרִיאַת שְׁמַע, בֵּין יָחִיד שֶׁעָשָׂה כֵּן בֵּין צִבּוּר:

הַבְּרָכוֹת אֵינָן מְעַכְּבוֹת. קְרָאָהּ בְּלֹא בִּרְכוֹתֶיהָ — יָצָא יְדֵי חוֹבַת ק״ש, וְחוֹזֵר וְקוֹרֵא הַבְּרָכוֹת בְּלֹא ק״ש, שֶׁאֵינָן כְּבִרְכוֹת שְׁאָר מִצְווֹת (עוֹבֵר לַעֲשִׂיָּתָן) — עַיֵּן סִימָן נ״ט. וְטוֹב לַחֲזֹר ; וְאַף אִם לֹא חָזַר — יָצָא, יָחִיד וְצִבּוּר.

סעיף ג סֵדֶר הַבְּרָכוֹת אֵינוֹ מְעַכֵּב — הִקְדִּים שְׁנִיָּה לָרִאשׁוֹנָה יָצָא.

סֵדֶר הַבְּרָכוֹת אֵינָם מְעַכְּבוֹת, שֶׁאִם הִקְדִּים שְׁנִיָּה לָרִאשׁוֹנָה — יָצָא יְדֵי חוֹבַת בְּרָכוֹת. וְאַף עַל פִּי שֶׁהַשְּׁנִיָּה אֵינָהּ פּוֹתַחַת בְּ״בָרוּךְ״ וְכָל בְּרָכָה שֶׁאֵין בָּהּ מַלְכוּת — אֵינָהּ בְּרָכָה, מִכָּל מָקוֹם כֵּיוָן שֶׁנִתַּקְנָה סְמוּכָה לַחֲבֶרְתָּהּ, אַף כְּשֶׁקּוֹרֵא אוֹתָהּ בִּפְנֵי עַצְמָהּ — אֵין בָּהּ שֵׁם וּמַלְכוּת בִּפְתִיחָתָהּ:

סֵדֶר הַבְּרָכוֹת. אֵינוֹ מְעַכֵּב : הִקְדִּים שְׁנִיָּה לָרִאשׁוֹנָה — יָצָא. וְאַף שֶׁהַשְּׁנִיָּה אֵינָהּ פּוֹתַחַת בְּ״בָרוּךְ״ וְכָל בְּרָכָה שֶׁאֵין בָּהּ מַלְכוּת אֵינָהּ בְּרָכָה — כֵּיוָן שֶׁנִּתַּקְנָה סְמוּכָה, אַף בִּפְנֵי עַצְמָהּ אֵין בָּהּ שֵׁם וּמַלְכוּת בִּפְתִיחָתָהּ.

סעיף ד זְכִירַת הַר סִינַי, עֲמָלֵק, מִרְיָם וְהָעֵגֶל — וְטוֹב לְזָכְרָן אֵצֶל ק״ש.

זְכִירַת מַעֲמַד הַר סִינַי וּמַעֲשֵׂה עֲמָלֵק וּמַעֲשֵּׂה מִרְיָם וּמַעֲשֵׂה הָעֵגֶל הֵן מִּצְווֹת עֲשֵׂה שֶׁל תּוֹרָה. וְגַם זְכִירַת שַׁבָּת יֵשׁ אוֹמְרִים שֶׁמִּצְוָתָהּ בְּכָל יוֹם. וְטוֹב לְזָכְרָן אֵצֶל קְרִיאַת שְׁמַע, שֶׁכְּשֶׁיֹּאמַר ״וּבָנוּ בָחַרְתָּ״ — יִזְכֹּר מַתַּן תּוֹרָה, ״וְקֵרַבְתָּנוּ״ — לְהַר סִינַי, ״לְשִׁמְךָ הַגָּדוֹל״ — מַעֲשֵׂה עֲמָלֵק, שֶׁאֵין הַשֵּׁם שָׁלֵם עַד שֶׁיִּמָּחֶה זַרְעוֹ, ״לְהוֹדוֹת לְךָ״ — הַפֶּה לֹא נִבְרָא אֶלָּא לְהוֹדוֹת וְלֹא לְדַבֵּר לְשׁוֹן הָרָע, וְזֶהוּ זְכִירַת מַעֲשֵׂה מִרְיָם, ״וּזְכַרְתֶּם אֶת כָּל מִּצְווֹת ה׳״ — זוֹ הִיא שַׁבָּת שֶׁשְּׁקוּלָה כְּנֶגֶד כָּל הַמִּצְווֹת, וּמַעֲשֵׂה הָעֵגֶל — יֵשׁ מִי שֶׁאוֹמֵר שֶׁיִּזְכֹּר כְּשֶׁאוֹמֵר ״(לְיַחֶדְךָ) בְּאַהֲבָה״, וְלֹא כְּאוֹתָהּ שָׁעָה שֶׁעָשׂוּ אֶת הָעֵגֶל שֶׁלֹּא הָיָה בְּאַהֲבָה עִם הַקָּדוֹשׁ בָּרוּךְ הוּא:

הַזְּכִירוֹת. זְכִירַת מַעֲמַד הַר סִינַי, עֲמָלֵק, מִרְיָם וְהָעֵגֶל — מִצְווֹת עֲשֵׂה. וְגַם זְכִירַת שַׁבָּת בְּכָל יוֹם. וְטוֹב לְזָכְרָן אֵצֶל ק״ש : ״וּבָנוּ בָחַרְתָּ״ — מַתַּן תּוֹרָה ; ״וְקֵרַבְתָּנוּ״ — הַר סִינַי ; ״לְשִׁמְךָ הַגָּדוֹל״ — עֲמָלֵק ; ״לְהוֹדוֹת לְךָ״ — מִרְיָם ; ״וּזְכַרְתֶּם״ — שַׁבָּת ; וְהָעֵגֶל בְּ״בְּאַהֲבָה״.

סעיף ה כָּל הַמִּצְווֹת צְרִיכוֹת כַּוָּנָה — וַהֲלָכָה כַּסְּבָרָא הָרִאשׁוֹנָה ; פָּסוּק רִאשׁוֹן.

כָּל הַמִּצְווֹת צְרִיכוֹת כַּוָּנָה לָצֵאת יְדֵי חוֹבָה בַּעֲשִׂיַּת אוֹתָהּ מִצְוָה. וְאִם עֲשָׂאָהּ בְּלֹא כַּוָּנָה לָצֵאת יְדֵי חוֹבָתוֹ אֶלָּא כְּמִתְעַסֵּק בְּעָלְמָא אוֹ לְכַוָּנָה אַחֶרֶת וְלֹא לְשֵׁם אוֹתָהּ מִצְוָה — לֹא יָצָא יְדֵי חוֹבָתוֹ מִן הַתּוֹרָה. וְיֵשׁ אוֹמְרִים שֶׁמִּצְווֹת אֵין צְרִיכוֹת כַּוָּנָה, וְאַף הַמִּתְעַסֵּק יָצָא בְּדִיעֲבַד. וַהֲלָכָה כַּסְּבָרָא הָרִאשׁוֹנָה. בַּמֶּה דְּבָרִים אֲמוּרִים? בְּמִּצְווֹת שֶׁל תּוֹרָה, אֲבָל בְּשֶׁל דִּבְרֵי סוֹפְרִים — יֵשׁ אוֹמְרִים שֶׁהֲלָכָה שֶׁאֵין צָרִיךְ כַּוָּנָה. וְיֵשׁ חוֹלְקִין, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן תע״ה. וּקְרִיאַת שְׁמַע הִיא מִן הַתּוֹרָה, וּצְרִיכָה כַּוָּנָה לְדִבְרֵי הַכֹּל. וְאַף לְהָאוֹמְרִים שֶׁאֲפִלּוּ מִּצְווֹת שֶׁל תּוֹרָה אֵין צָרִיךְ כַּוָּנָה, מִכָּל מָקוֹם בִּקְרִיאַת שְׁמַע צָרִיךְ כַּוָּנַת הַלֵּב שֶׁיָּבִין מַה שֶּׁהוּא אוֹמֵר בְּפִיו, וְלֹא אָמְרוּ שֶׁאֵינוֹ צָרִיךְ כַּוָּנָה אֶלָּא בְּכַוָּנָה לָצֵאת יְדֵי חוֹבָה, אֲבָל צָרִיךְ כַּוָּנַת הָעִנְיָן, כְּלוֹמַר שֶׁלֹּא יְהַרְהֵר בִּדְבָרִים אֲחֵרִים, כְּדֵי שֶׁיְּקַבֵּל מַלְכוּת שָׁמַיִם בְּהַסְכָּמַת הַלֵּב. וְאֵינוֹ דּוֹמֶה לִשְׁאָר מִּצְווֹת שֶׁהֵן מִּצְווֹת עֲשִׂיָּה, וְכָל שֶׁעָשָׂה מִצְוָתָהּ אַף עַל פִּי שֶׁלֹּא כִּוֵּן לָהּ — הֲרֵי קִיֵּם מִצְוַת עֲשִׂיָּתָהּ, אֶלָּא שֶׁאֵין זֶה מִן הַמֻּבְחָר, וְכָל שֶׁכֵּן אִם כִּוֵּן לָצֵאת אַף עַל פִּי שֶׁהִרְהֵר בָּהּ בָּאֶמְצַע בִּדְבָרִים אֲחֵרִים — יָצָא לְדִבְרֵי הַכֹּל. אֲבָל קְרִיאַת שְׁמַע וּתְפִלָּה שֶׁהֵן קַבָּלַת עֹל מַלְכוּת שָׁמַיִם אוֹ סִדּוּר שְׁבָחִים — אֵינוֹ בְּדִין שֶׁיִּהְיֶה לְבָבוֹ פּוֹנֶה לִדְבָרִים אֲחֵרִים. וְכָל זֶה בְּפָסוּק רִאשׁוֹן שֶׁל קְרִיאַת שְׁמַע שֶׁהוּא ״שְׁמַע יִשְׂרָאֵל״, שֶׁהוּא עִקָּר קַבָּלַת מַלְכוּת שָׁמַיִם. וְכֵן בִּבְרָכָה רִאשׁוֹנָה שֶׁל תְּפִלָּה. אֲבָל מִ״וְאָהַבְתָּ״ וְאֵילָךְ — אֵין הַכַּוָּנָה מְעַכֶּבֶת בְּדִיעֲבַד. וַאֲפִלּוּ הָיָה קוֹרֵא בַּתּוֹרָה אוֹ מַגִּיהַּ הַפָּרָשִׁיּוֹת אֵלּוּ בְּעוֹנַת קְרִיאַת שְׁמַע — יָצָא יְדֵי חוֹבָתוֹ, אַף עַל פִּי שֶׁלֹּא נִתְכַוֵּן לָצֵאת יְדֵי חוֹבָתוֹ. וְהוּא שֶׁנִתְכַוֵּן בְּפָסוּק רִאשׁוֹן (מִפְּנֵי שֶׁעִקָּר קְרִיאַת שְׁמַע אֵינָהּ אֶלָּא פָּסוּק רִאשׁוֹן בִּלְבַד שֶׁהוּא מִן הַתּוֹרָה, וְהַשְּׁאָר שֶׁתִקְּנוּ חֲכָמִים לֹא הִצְרִיכוּ כַּוָּנָה אֶלָּא לְכַתְּחִלָּה, אֲבָל אִם קָרָא הַשְּׁאָר בְּלֹא כַּוָּנָה — אֵין צָרִיךְ לַחֲזֹר וְלִקְרוֹת, בֵּין שֶׁקָּרָא בְּלֹא כַּוָּנָה לָצֵאת בֵּין שֶׁלֹּא בְּכַוָּנַת הָעִנְיָן. וְאַף לְהָאוֹמְרִים שֶׁאַף מִּצְווֹת שֶׁל דִּבְרֵי סוֹפְרִים צְרִיכִים כַּוָּנָה לָצֵאת, מִכָּל מָקוֹם בִּקְרִיאַת שְׁמַע כֵּיוָן שֶׁאֵינָהּ מִצְוַת עֲשִׂיָּה אֶלָּא קַבָּלַת מַלְכוּת שָׁמַיִם וְעֹל מִּצְווֹת שֶׁהוּא דָּבָר הַתָּלוּי בְּכַוָּנַת הָעִנְיָן, לְפִיכָךְ כֵּיוָן שֶׁלֹּא הִצְרִיכוּ לַחֲזֹר וְלִקְרוֹת מִפְּנֵי חֶסְרוֹן כַּוָּנַת הָעִנְיָן — לֹא הִצְרִיכוּ גַּם כֵּן לַחֲזֹר מִפְּנֵי חֶסְרוֹן כַּוָּנָה לָצֵאת, כִּי מַה בֶּצַע שֶׁיַּחֲזֹר לְדַבֵּר בְּלִי לֵב, וְדַי בְּמַה שֶּׁהִצְרִיכוּהוּ לִקְרוֹת בְּכָל יוֹם, שֶׁעַל יְדֵי כֵּן יָבוֹא גַּם כֵּן לְקַבָּלָה בַּלֵּב בְּרֹב הַפְּעָמִים, שֶׁהֲרֵי לְכַתְּחִלָּה הִצְרִיכוּ כַּוָּנָה, אֶלָּא שֶׁאִם אֵרַע מִקְרֶה שֶׁלֹּא נִתְכַוֵּן — לֹא הִטְרִיחוּ לַחֲזֹר. אֲבָל לְהָאוֹמְרִים שֶׁמִּ״וְאָהַבְתָּ״ וְאֵילָךְ הוּא גַּם כֵּן מִן הַתּוֹרָה — צָרִיךְ כַּוָּנָה לָצֵאת בְּכָל מַה שֶּׁהוּא מִן הַתּוֹרָה. אֲבָל כַּוָּנַת הָעִנְיָן דַּי בְּפָסוּק רִאשׁוֹן לְדִבְרֵי הַכֹּל בְּדִיעֲבַד):

מִצְווֹת צְרִיכוֹת כַּוָּנָה. כָּל הַמִּצְווֹת צְרִיכוֹת כַּוָּנָה לָצֵאת ; מִתְעַסֵּק אוֹ לְכַוָּנָה אַחֶרֶת — לֹא יָצָא מִן הַתּוֹרָה. וְיֵשׁ אוֹמְרִים אֵין צְרִיכוֹת כַּוָּנָה, וַהֲלָכָה כַּסְּבָרָא הָרִאשׁוֹנָה. בְּמִצְווֹת דִּבְרֵי סוֹפְרִים — מַחֲלֹקֶת (עַיֵּן סִימָן תע״ה). וְק״ש מִן הַתּוֹרָה וּצְרִיכָה כַּוָּנָה לְדִבְרֵי הַכֹּל. וְכָל זֶה בְּפָסוּק רִאשׁוֹן ״שְׁמַע יִשְׂרָאֵל״ — עִיקַּר קַבָּלַת מַלְכוּת שָׁמַיִם ; מִ״וְאָהַבְתָּ״ וְאֵילָךְ אֵין הַכַּוָּנָה מְעַכֶּבֶת בְּדִיעֲבַד.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ס׳ (מהדורת קה״ת, ה׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — בְּרָכוֹת קְרִיאַת שְׁמַע וְהַכַּוָּנָה

היסוד : בְּרָכוֹת קְרִיאַת שְׁמַע וְהַכַּוָּנָה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה הַבְּרָכָה הַשְּׁנִיָּה — אַהֲבָה רַבָּה / אַהֲבַת עוֹלָם וְהַפְּתִיחָה בְּבָרוּךְ. אחר כך הַבְּרָכוֹת וְסֵדֶר הַבְּרָכוֹת שֶׁאֵינָן מְעַכְּבוֹת. אחר כך הַזְּכִירוֹת אֵצֶל קְרִיאַת שְׁמַע. ולבסוף הַכְּלָל מִצְווֹת צְרִיכוֹת כַּוָּנָה וְכַוָּנַת הַפָּסוּק הָרִאשׁוֹן.

הַבְּרָכָה הַשְּׁנִיָּה — אַהֲבָה רַבָּה / אַהֲבַת עוֹלָם

הברכה השנייה — אהבה רבה / אהבת עולם

היסוד. נֶחְלְקוּ חֲכָמִים בִּפְתִיחָתָהּ ; הַגְּאוֹנִים : שַׁחֲרִית אַהֲבָה רַבָּה, עַרְבִית אַהֲבַת עוֹלָם (סעיף א).

הַמַּסְקָנָה. אֵינָהּ פּוֹתַחַת בְּ״בָרוּךְ״ (סְמוּכָה לְ״יוֹצֵר אוֹר״). עַיֵּן סימן 47 (ברכת התורה).

הַבְּרָכוֹת וְסֵדֶר הַבְּרָכוֹת אֵינָן מְעַכְּבוֹת

הברכות וסדר הברכות

היסוד. קְרָאָהּ בְּלֹא בְּרָכוֹת יָצָא ; הִקְדִּים שְׁנִיָּה לָרִאשׁוֹנָה יָצָא (סעיפים ב-ג).

הַמַּסְקָנָה. בִּרְכוֹת ק״ש אֵינָן כְּבִרְכוֹת שְׁאָר מִצְווֹת. עַיֵּן סימן 59 (ברכות ק״ש).

הַזְּכִירוֹת אֵצֶל קְרִיאַת שְׁמַע

הזכירות אצל ק״ש

היסוד. זְכִירַת הַר סִינַי, עֲמָלֵק, מִרְיָם, הָעֵגֶל וְשַׁבָּת — מִצְווֹת עֲשֵׂה (סעיף ד).

הַמַּסְקָנָה. וְטוֹב לְזָכְרָן אֵצֶל קְרִיאַת שְׁמַע. עַיֵּן סימן 46.

מִצְווֹת צְרִיכוֹת כַּוָּנָה וּפָסוּק רִאשׁוֹן

מצוות צריכות כוונה והפסוק הראשון

היסוד. כָּל הַמִּצְווֹת צְרִיכוֹת כַּוָּנָה, וַהֲלָכָה כַּסְּבָרָא הָרִאשׁוֹנָה ; פָּסוּק רִאשׁוֹן מְעַכֵּב (סעיף ה).

הַמַּסְקָנָה. מִ״וְאָהַבְתָּ״ וְאֵילָךְ אֵין הַכַּוָּנָה מְעַכֶּבֶת בְּדִיעֲבַד. עַיֵּן סימן 61.

הפניה ללימוד. היקף הסימן — בְּרָכוֹת קְרִיאַת שְׁמַע, הַזְּכִירוֹת וּמִצְווֹת צְרִיכוֹת כַּוָּנָה — נוגע להנהגת התפילה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּין בְּרָכוֹת קְרִיאַת שְׁמַע וְאִם מִצְווֹת צְרִיכוֹת כַּוָּנָה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ס׳. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מחמשת סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
הַבְּרָכָה הַשְּׁנִיָּה או״ח ס׳ — אַהֲבַת עוֹלָם ; וְהָרְמָ״א : יֵשׁ אוֹמְרִים אַהֲבָה רַבָּה. מ״ב ס׳ — נֻסַּח הַבְּרָכָה וּמִנְהַג אַשְׁכְּנַז וּסְפָרַד. ערוה״ש ס׳ — טַעַם ״אַהֲבָה רַבָּה״ וְ״אַהֲבַת עוֹלָם״. דרך אדמו״ר הזקן — מַחֲלֹקֶת בִּפְתִיחָתָהּ ; הַגְּאוֹנִים : שַׁחֲרִית אַהֲבָה רַבָּה, עַרְבִית אַהֲבַת עוֹלָם ; אֵינָהּ פּוֹתַחַת בְּ״בָרוּךְ״ (סְמוּכָה) (סעיף א).
הַבְּרָכוֹת וְסֵדֶר הַבְּרָכוֹת או״ח ס׳ — קְרָאָהּ בְּלֹא בְּרָכוֹת יָצָא ; סֵדֶר הַבְּרָכוֹת אֵינוֹ מְעַכֵּב. מ״ב ס׳ — חוֹזֵר וְאוֹמֵר הַבְּרָכוֹת בְּלֹא ק״ש. ערוה״ש ס׳ — הַבְּרָכוֹת אֵינָן מְעַכְּבוֹת אֶת ק״ש. דרך אדמו״ר הזקן — קְרָאָהּ בְּלֹא בִּרְכוֹתֶיהָ יָצָא, חוֹזֵר לַבְּרָכוֹת ; הִקְדִּים שְׁנִיָּה לָרִאשׁוֹנָה יָצָא (סעיפים ב-ג).
הַזְּכִירוֹת אֵצֶל ק״ש או״ח ס׳ — שֵׁשׁ זְכִירוֹת סְמוּכוֹת לִקְרִיאַת שְׁמַע. מ״ב ס׳ — טַעַם הַזְּכִירוֹת אֵצֶל ק״ש. ערוה״ש ס׳ — עִנְיַן הַזְּכִירוֹת. דרך אדמו״ר הזקן — הַר סִינַי, עֲמָלֵק, מִרְיָם, הָעֵגֶל וְשַׁבָּת — מִצְווֹת עֲשֵׂה, וְטוֹב לְזָכְרָן אֵצֶל ק״ש (סעיף ד).
מִצְווֹת צְרִיכוֹת כַּוָּנָה וּפָסוּק רִאשׁוֹן או״ח ס׳ — מִצְווֹת צְרִיכוֹת כַּוָּנָה, וְכֵן הֲלָכָה ; לֹא כִּוֵּן בְּפָסוּק רִאשׁוֹן לֹא יָצָא. מ״ב ס׳ — כַּוָּנַת הַלֵּב בְּפָסוּק רִאשׁוֹן מְעַכֶּבֶת. ערוה״ש ס׳ — דִּין מִצְווֹת צְרִיכוֹת כַּוָּנָה. דרך אדמו״ר הזקן — צְרִיכוֹת כַּוָּנָה, וַהֲלָכָה כַּסְּבָרָא הָרִאשׁוֹנָה ; ק״ש מִן הַתּוֹרָה, פָּסוּק רִאשׁוֹן מְעַכֵּב, מִ״וְאָהַבְתָּ״ אֵין מְעַכֵּב בְּדִיעֲבַד (סעיף ה).

הטבלה נערכה מטקסט המחבר (אורח חיים ס׳, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע א-ב, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות י״ג, ראש השנה כ״ח-כ״ט). טור אדמו״ר הזקן נשען על הטקסט האמיתי של שולחן ערוך הרב (ה׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — הַבְּרָכָה הַשְּׁנִיָּה וְהַבְּרָכוֹת

הברכה השנייה והברכות

יסוד הסימן : בְּרָכָה שְׁנִיָּה אַהֲבָה רַבָּה / אַהֲבַת עוֹלָם ; הַבְּרָכוֹת וְסֵדֶר הַבְּרָכוֹת אֵינָן מְעַכְּבוֹת (סעיפים א-ג).

קריאת הרב : אֵינָהּ פּוֹתַחַת בְּבָרוּךְ (סְמוּכָה) ; קְרָאָהּ בְּלֹא בְּרָכוֹת יָצָא. עַיֵּן סימן 59.

מסקנה למעשה : לוֹמַר הַבְּרָכוֹת בְּסִדְרָן, וְלָדַעַת שֶׁאֵינָן מְעַכְּבוֹת אֶת הַק״ש.

קו ב — הַזְּכִירוֹת אֵצֶל קְרִיאַת שְׁמַע

הזכירות אצל ק״ש

יסוד הסימן : זְכִירַת הַר סִינַי, עֲמָלֵק, מִרְיָם, הָעֵגֶל וְשַׁבָּת — מִצְווֹת עֲשֵׂה (סעיף ד).

קריאת הרב : וְטוֹב לְזָכְרָן אֵצֶל בִּרְכוֹת ק״ש בְּכָל מַטְבֵּעַ וּמַטְבֵּעַ. עַיֵּן סימן 46.

מסקנה למעשה : לְכַוֵּן אֶת הַזְּכִירוֹת בַּתֵּבוֹת הַמַּתְאִימוֹת שֶׁבַּבְּרָכוֹת.

קו ג — מִצְווֹת צְרִיכוֹת כַּוָּנָה וּפָסוּק רִאשׁוֹן

מצוות צריכות כוונה והפסוק הראשון

יסוד הסימן : כָּל הַמִּצְווֹת צְרִיכוֹת כַּוָּנָה, וַהֲלָכָה כַּסְּבָרָא הָרִאשׁוֹנָה ; פָּסוּק רִאשׁוֹן מְעַכֵּב (סעיף ה).

קריאת הרב : ק״ש מִן הַתּוֹרָה, כַּוָּנַת הַלֵּב ; מִ״וְאָהַבְתָּ״ וְאֵילָךְ אֵין הַכַּוָּנָה מְעַכֶּבֶת בְּדִיעֲבַד. עַיֵּן סימן 61.

מסקנה למעשה : לְהִתְרַכֵּז בְּ״שְׁמַע יִשְׂרָאֵל״ — עִיקַּר קַבָּלַת עֹל מַלְכוּת שָׁמַיִם.

5 · למעשה

הלכה למעשה — מנהג חב״ד

ההנהגה החב״דית למעשה

נקודות הנהגה הנובעות ישירות מסימן ס׳ ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ס׳ (דִּין בְּרָכוֹת קְרִיאַת שְׁמַע וְאִם מִצְווֹת צְרִיכוֹת כַּוָּנָה)

  • ① הברכה השנייה. הַבְּרָכָה הַשְּׁנִיָּה : לְפִי הַגְּאוֹנִים שַׁחֲרִית ״אַהֲבָה רַבָּה״, עַרְבִית ״אַהֲבַת עוֹלָם״ ; אֵינָהּ פּוֹתַחַת בְּבָרוּךְ (סְמוּכָה לְ״יוֹצֵר אוֹר״). מקור : שוע״ר ס׳ סעיף א.
  • ② הברכות אינן מעכבות. קְרָאָהּ בְּלֹא בְּרָכוֹת — יָצָא ; חוֹזֵר וְאוֹמֵר הַבְּרָכוֹת בְּלֹא ק״ש (וְטוֹב לַחֲזֹר וְלִקְרוֹת עִם הַבְּרָכוֹת). מקור : שוע״ר ס׳ סעיף ב.
  • ③ סדר הברכות. סֵדֶר הַבְּרָכוֹת אֵינוֹ מְעַכֵּב : הִקְדִּים שְׁנִיָּה לָרִאשׁוֹנָה — יָצָא יְדֵי חוֹבַת בְּרָכוֹת. מקור : שוע״ר ס׳ סעיף ג.
  • ④ הזכירות. לִזְכֹּר אֶת הַזְּכִירוֹת (הַר סִינַי, עֲמָלֵק, מִרְיָם, הָעֵגֶל, שַׁבָּת) בַּתֵּבוֹת הַמַּתְאִימוֹת שֶׁבַּבְּרָכוֹת. מקור : שוע״ר ס׳ סעיף ד.
  • ⑤ מצוות צריכות כוונה. צְרִיכוֹת כַּוָּנָה (וְכֵן הֲלָכָה) ; ק״ש צְרִיכָה כַּוָּנַת הַלֵּב בְּפָסוּק רִאשׁוֹן ״שְׁמַע יִשְׂרָאֵל״ — וּבִלְעָדֶיהָ לֹא יָצָא. מקור : שוע״ר ס׳ סעיף ה.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ס׳ (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַבְּרָכָה הַשְּׁנִיָּה, הַבְּרָכוֹת, הַזְּכִירוֹת, הַכַּוָּנָה — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
