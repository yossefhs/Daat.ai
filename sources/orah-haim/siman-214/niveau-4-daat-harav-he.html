דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן רי״ד · בְּכָל בְּרָכָה צָרִיךְ שֶׁיִּהְיֶה שֵׁם וּמַלְכוּת

דרכו של אדמו״ר הזקן בְּעִנְיַן בְּכָל בְּרָכָה צָרִיךְ שֶׁיִּהְיֶה שֵׁם וּמַלְכוּת : הַזְכָּרַת הַשֵּׁם ; מַלְכוּת בִּפְתִיחָתָן ; בְּרָכָה רִאשׁוֹנָה שֶׁל תְּפִלַּת י״ח ; מֵעֵין שֶׁבַע ; תֵּבַת הָעוֹלָם ; מֶלֶךְ לְבַדּוֹ ; מוּסָב עַל בָּרוּךְ ; הַמֶּלֶךְ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ומכריע בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן רי״ד — בְּכָל בְּרָכָה צָרִיךְ שֶׁיִּהְיֶה שֵׁם וּמַלְכוּת, הַמַּחֲזִיק אֵצֶל הָרַב ב׳ סְעִיפִים, וְאִלּוּ אֵצֶל הַמְחַבֵּר אֵין בּוֹ אֶלָּא סָעִיף אֶחָד. וְהָרַב מַעֲמִיד בּוֹ אֶת שְׁלַד הַבְּרָכָה כֻּלָּהּ. תְּחִלָּה הַסַּף : מַטְבֵּעַ שֶׁאֵין בּוֹ הַזְכָּרַת הַשֵּׁם אֵינוֹ בְּרָכָה פְּגוּמָה אֶלָּא אֵינוֹ בְּרָכָה כְּלָל. וְאַחַר כָּךְ הַמַּלְכוּת, הַנִּדְרֶשֶׁת מִכָּל הַבְּרָכוֹת הַפּוֹתְחוֹת בְּבָרוּךְ, וְנִדְרֶשֶׁת בִּפְתִיחָתָן דַּוְקָא : וְאִם לָאו — לֹא יָצָא. וּשְׁנֵי מַטְבְּעוֹת יוֹצְאִים מִן הַכְּלָל הַשֵּׁנִי, וּמִטַּעַם אֶחָד : בְּרָכָה רִאשׁוֹנָה שֶׁל תְּפִלַּת י״ח אֵינָהּ הוֹדָאָה עַל הֲנָאָה וְלֹא בִּרְכַּת מִצְוָה, אֶלָּא פְּתִיחָה שֶׁל בַּקָּשָׁה — מְסַדֵּר שְׁבָחָיו שֶׁל מָקוֹם תְּחִלָּה וְאַחַר כָּךְ מִתְפַּלֵּל ; לְפִיכָךְ לֹא תִּקְּנוּ בָּהּ מַלְכוּת. וּבִרְכַּת מֵעֵין שֶׁבַע שֶׁבְּלֵיל שַׁבָּת הוֹלֶכֶת אַחֲרֶיהָ, שֶׁפְּתִיחָתָהּ מֵעֵין פְּתִיחָתָהּ. וְהַסָּעִיף הַשֵּׁנִי יוֹרֵד עַד כְּדֵי תֵּבָה אַחַת : הַמְדַלֵּג אֶת תֵּבַת הָעוֹלָם לְבַדָּהּ — חוֹזֵר וּמְבָרֵךְ, שֶׁאֵין מֶלֶךְ לְבַדּוֹ אוֹמֵר כְּלוּם ; וְשָׁם פּוֹרֵשׂ הָרַב בְּמֻסְגָּר אָרֹךְ אֶת דִּקְדּוּק הַמַּטְבֵּעַ : עַל מָה מוּסָב מֶלֶךְ, עַל מָה מוּסָב שְׁאָר הַבְּרָכָה, וְלָמָּה אֵין הַכַּוָּנָה מִתְקַיֶּמֶת אֶלָּא בְּבָרוּךְ מֶלֶךְ הָעוֹלָם.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן רי״ד — ב׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim רי״ד). אדמו״ר הזקן דן בו בְּעִנְיַן בְּכָל בְּרָכָה צָרִיךְ שֶׁיִּהְיֶה שֵׁם וּמַלְכוּת. הטקסט מובא כלשונו, ובכלל זה פרטי הדפוס שבו (סוגריים, מרכאות, רווחים). אין מובא בעמוד זה שום קטע מחוץ לטקסט בר-בדיקה זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן רי״ד — בְּכָל בְּרָכָה צָרִיךְ שֶׁיִּהְיֶה שֵׁם וּמַלְכוּת

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן רי״ד — בְּכָל בְּרָכָה צָרִיךְ שֶׁיִּהְיֶה שֵׁם וּמַלְכוּת — וּבוֹ ב׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.214). הסימן כולל ב׳ סעיפים (סעיף k → ‎.214.k).

סעיף א שֵׁם וּמַלְכוּת בְּכָל בְּרָכָה — חוּץ מִבְּרָכָה רִאשׁוֹנָה שֶׁל תְּפִלַּת י״ח וּמֵעֵין שֶׁבַע.

בְּכָל בְּרָכָה צָרִיךְ שֶׁיִּהְיֶה שֵׁם וּמַלְכוּת, וּבוֹ ב׳ סְעִיפִים:
כָּל בְּרָכָה שֶׁאֵין בָּהּ הַזְכָּרַת הַשֵּׁם — אֵינָהּ בְּרָכָה. וְכָל בְּרָכוֹת הַפּוֹתְחוֹת בְּ"בָרוּךְ" — צָרִיךְ לוֹמַר גַּם מַלְכוּת בִּפְתִיחָתָן, וְאִם לָאו — לֹא יָצָא. חוּץ מִבְּרָכָה רִאשׁוֹנָה שֶׁל תְּפִלַּת י״ח, שֶׁאֵינָהּ הוֹדָאָה עַל הֲנָאָה, אוֹ מִצְוָה, אֶלָּא עַל תְּבִיעַת צְרָכָיו שֶׁל אָדָם, כְּדֵי שֶׁיְּסַדֵּר שְׁבָחָיו שֶׁל מָקוֹם וְאַחַר כָּךְ יִתְפַּלֵּל, לְפִיכָךְ לֹא תִּקְּנוּ בָּהּ מַלְכוּת. וְכֵן בְּבִרְכוֹת מֵעֵין שֶׁבַע שֶׁבְּלֵיל שַׁבָּת, הוֹאִיל וּפְתִיחָתָהּ הִיא מֵעֵין בְּרָכָה רִאשׁוֹנָה שֶׁל תְּפִלַּת י״ח:

סעיף ב דִּלֵּג תֵּבַת הָעוֹלָם — חוֹזֵר וּמְבָרֵךְ, שֶׁמֶּלֶךְ לְבַדּוֹ אֵינוֹ מַלְכוּת.

אֲפִלּוּ אִם לֹא דִּלֵּג אֶלָּא תֵּבַת "הָעוֹלָם" לְבַד — צָרִיךְ לַחֲזֹר וּלְבָרֵךְ, שֶׁ"מֶּלֶךְ" לְבַדּוֹ אֵינוֹ חָשׁוּב מַלְכוּת (כִּי אֵין לוֹ בֵּאוּר כְּלָל בְּלֹא תֵּבַת "הָעוֹלָם" שֶׁהוּא סָמוּךְ אֲלֵיהֶם, שֶׁאִם תִּמְצֵי לוֹמַר שֶׁהוּא מוּסָב עַל תֵּבַת "אֱלֹהֵינוּ", כְּלוֹמַר שֶׁ"אֱלֹהֵינוּ" הוּא "מֶלֶךְ" — אֵין זוֹ כַּוָּנַת חֲכָמִים שֶׁתִּקְּנוּ מַלְכוּת בִּבְרָכוֹת, אֶלָּא שֶׁיִּהְיֶה מוּסָב עַל "בָּרוּךְ" שֶׁבִּתְחִלַּת הַבְּרָכָה, כְּלוֹמַר "בָּרוּךְ מֶלֶךְ הָעוֹלָם", שֶׁהֲרֵי מְסַיֵּם "שֶׁהַכֹּל כוּ׳", אוֹ "אֲשֶׁר כוּ׳", אוֹ "שֶׁעָשָׂה כוּ׳", שֶׁכָּל זֶה מוּסָב עַל "בָּרוּךְ", וּכְשֶׁאֵינוֹ אוֹמֵר "הָעוֹלָם" — אֵינוֹ מוּסָב עַל "בָּרוּךְ" אֶלָּא תֵּבַת "ה׳ אֱלֹהֵינוּ", אֲבָל "מֶלֶךְ" נִשְׁאַר לְבַדּוֹ, אֶלָּא אִם כֵּן הָיָה אוֹמֵר "הַמֶּלֶךְ". וְאִם תִּמְצֵי לוֹמַר שֶׁ"מֶּלֶךְ" מוּסָב לְמַטָּה, "מֶלֶךְ שֶׁעָשָׂה כוּ׳", אוֹ "מֶלֶךְ שֶׁהַכֹּל כוּ׳" — גַּם זוֹ אֵינָהּ כַּוָּנַת חֲכָמִים בְּמַטְבֵּעַ הַבְּרָכוֹת, אֶלָּא שֶׁשְּׁאָר הַבְּרָכָה יְהֵא מוּסָב עַל "הַשֵּׁם" שֶׁהוּא עִקַּר הַבְּרָכָה, וְלֹא עַל "מֶלֶךְ" לְבַד):

2 · למעשה

הלכה למעשה — מנהג חב״ד

הנהגה מעשית לחב״ד

נקודות הנהגה הנובעות ישירות מסימן רי״ד ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן רי״ד (בְּכָל בְּרָכָה צָרִיךְ שֶׁיִּהְיֶה שֵׁם וּמַלְכוּת)

  • ① סעיף א. שְׁנֵי תְּנָאִים הֵם, וְלֹא אֶחָד. תְּחִלָּה הַזְכָּרַת הַשֵּׁם : מַטְבֵּעַ שֶׁלֹּא הֻזְכַּר בּוֹ הַשֵּׁם אֵינוֹ בְּרָכָה חֲסֵרָה אֶלָּא אֵינָהּ בְּרָכָה כְּלָל. וְאַחֲרָיו הַמַּלְכוּת, בְּכָל הַבְּרָכוֹת הַפּוֹתְחוֹת בְּבָרוּךְ : צָרִיךְ לְאָמְרָהּ בִּפְתִיחָתָן מַמָּשׁ, וְאִם לֹא אֲמָרָהּ — לֹא יָצָא. וְאַחַר כָּךְ מַעֲמִיד הָרַב שְׁתֵּי יְצִיאוֹת מִן הַכְּלָל, וּשְׁתֵּיהֶן מִטַּעַם אֶחָד. בְּרָכָה רִאשׁוֹנָה שֶׁל תְּפִלַּת י״ח אֵינָהּ הוֹדָאָה עַל הֲנָאָה וְאֵינָהּ בִּרְכַּת מִצְוָה, אֶלָּא בָּאָה עַל תְּבִיעַת צְרָכָיו שֶׁל אָדָם, כַּסֵּדֶר הָרָאוּי — מְסַדֵּר שְׁבָחָיו שֶׁל מָקוֹם תְּחִלָּה וְאַחַר כָּךְ מִתְפַּלֵּל ; לְפִיכָךְ לֹא תִּקְּנוּ בָּהּ מַלְכוּת. וּבִרְכַּת מֵעֵין שֶׁבַע שֶׁבְּלֵיל שַׁבָּת הוֹלֶכֶת אַחֲרֶיהָ, שֶׁפְּתִיחָתָהּ מֵעֵין אוֹתָהּ בְּרָכָה רִאשׁוֹנָה. וְהַנּוֹגֵעַ לְמַעֲשֶׂה : אֵין אֵלּוּ מַטְבְּעוֹת חֲסֵרִים שֶׁיֵּשׁ לְהַשְׁלִימָם — כָּךְ תַּקָּנָתָם מֵעִקָּרָא, וְאֵין חוֹזְרִים עֲלֵיהֶם מֵחֲמַת חֶסְרוֹן מַלְכוּת.
  • ② סעיף ב. כָּאן יוֹרֵד הָרַב עַד כְּדֵי תֵּבָה אַחַת : הַמְדַלֵּג אֶת תֵּבַת הָעוֹלָם לְבַדָּהּ, וְאָמַר אֶת כָּל הַשְּׁאָר, חוֹזֵר וּמְבָרֵךְ, שֶׁאֵין מֶלֶךְ לְבַדּוֹ חָשׁוּב מַלְכוּת. וּבְמֻסְגָּר אֶחָד אָרֹךְ נִתָּן כָּאן אֶחָד מִדִּקְדּוּקֵי הַלָּשׁוֹן הַיָּפִים שֶׁבְּשֻׁלְחַן עָרוּךְ הָרַב : שָׁלֹשׁ דְּרָכִים לְהָסֵב אֶת תֵּבַת מֶלֶךְ, וּשְׁתַּיִם מֵהֶן נִדְחוֹת. (א) לַהֲסִבָּהּ לְמַעְלָה עַל אֱלֹהֵינוּ — כְּלוֹמַר שֶׁאֱלֹהֵינוּ הוּא מֶלֶךְ — אֵין זוֹ כַּוָּנַת חֲכָמִים שֶׁתִּקְּנוּ מַלְכוּת בִּבְרָכוֹת. (ב) לַהֲסִבָּהּ לְמַטָּה עַל הֶמְשֵׁךְ הַבְּרָכָה — מֶלֶךְ שֶׁעָשָׂה, מֶלֶךְ שֶׁהַכֹּל — אַף הִיא אֵינָהּ מַטְבֵּעַ הַבְּרָכוֹת, שֶׁשְּׁאָר הַבְּרָכָה צָרִיךְ לִהְיוֹת מוּסָב עַל הַשֵּׁם שֶׁהוּא עִקַּר הַבְּרָכָה, וְלֹא עַל מֶלֶךְ לְבַד. (ג) וְנִשְׁאֶרֶת הַדֶּרֶךְ הָאַחַת : מֶלֶךְ מוּסָב עַל בָּרוּךְ שֶׁבִּתְחִלַּת הַבְּרָכָה — בָּרוּךְ מֶלֶךְ הָעוֹלָם — כְּדֶרֶךְ שֶׁמּוּסָבִים עָלָיו הַסִּיּוּמִים שֶׁהַכֹּל, אֲשֶׁר, שֶׁעָשָׂה. וְאֵין הֲסִבָּה זוֹ מִתְקַיֶּמֶת אֶלָּא בְּתֵבַת הָעוֹלָם הַסְּמוּכָה לָהּ ; וּבְלֹא הָעוֹלָם מִתְיַשֶּׁבֶת הַלָּשׁוֹן עַל תֵּבוֹת ה׳ אֱלֹהֵינוּ, וּמֶלֶךְ נִשְׁאָר לְבַדּוֹ. וּמְסַיֵּם הָרַב בְּשִׁנּוּי אֶחָד שֶׁהָיָה מַצִּיל אֶת הַמַּטְבֵּעַ : אִלּוּ אָמַר הַמֶּלֶךְ, בְּה״א הַיְדִיעָה — שֶׁאָז עוֹמֶדֶת הַתֵּבָה בִּפְנֵי עַצְמָהּ.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן רי״ד (הטקסט האמיתי מובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — בְּרָכָה שֶׁנֶּאֶמְרָה בְּלֹא שֵׁם אוֹ בְּלֹא מַלְכוּת, תֵּבָה שֶׁנִּשְׁמְטָה בְּאֶמְצַע הַפְּתִיחָה, וּסְפֵק אִם חוֹזֵר — יש לשאול את רבך, ולחב״ד רב חב״די.
