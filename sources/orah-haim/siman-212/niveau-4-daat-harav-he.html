דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן רי״ב · שֶׁהָעִקָּר פּוֹטֵר אֶת הַטָּפֵל

דרכו של אדמו״ר הזקן בְּעִנְיַן שֶׁהָעִקָּר פּוֹטֵר אֶת הַטָּפֵל : עִקָּר וְטָפֵל ; תַּעֲרֹבֶת דָּגָן ; דָּג מָלוּחַ וּפַת ; יֵין שָׂרָף ; נְטִילַת יָדַיִם ; לְמַתֵּק הַשְּׁתִיָּה ; חָבִיב ; דַּעְתּוֹ עָלָיו ; שִׁנּוּי מָקוֹם — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ומכריע בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן רי״ב — שֶׁהָעִקָּר פּוֹטֵר אֶת הַטָּפֵל. הַמְּחַבֵּר לֹא הִקְצָה לְסִימָן זֶה אֶלָּא ב׳ סְעִיפִים, וְהָרַב כּוֹתֵב בּוֹ י״א סְעִיפִים — וְהַהֶפְרֵשׁ הוּא כָּל הָעִנְיָן : מַה שֶּׁהַשֻּׁלְחָן עָרוּךְ אוֹמֵר כִּכְלָל, אַדְמוּ״ר הַזָּקֵן מְפָרְקוֹ לִתְנָאָיו. תְּחִלָּה הוּא מְסַדֵּר אֶת מַדְרֵגוֹת הַטְּפֵלוּת — מִתַּעֲרֹבֶת שֶׁהַשֵּׁנִי בָּא לְתַקֵּן אֶת הָרִאשׁוֹן וְעַד שְׁנֵי מַאֲכָלִים שֶׁאֵינָם מְעֹרָבִים כְּלָל, שֶׁשָּׁם אַף הַפַּת נַעֲשֵׂית טְפֵלָה לְדָג מָלוּחַ — וְאַחַר כָּךְ קוֹבֵעַ אֶת הַכְּלָל הַמַּנְהִיג אֶת הַסִּימָן : כָּל תַּעֲרֹבֶת שֶׁיֵּשׁ בָּהּ מִין דָּגָן, הַדָּגָן עִקָּר לְעוֹלָם, כָּל שֶׁמִּתְכַּוֵּן גַּם לַאֲכִילָתוֹ. וּמִשָּׁם לְמִקְרֵי הַגְּבוּל : פַּת שְׁרוּיָה בְּיֵין שָׂרָף לִפְנֵי הַסְּעוּדָה וּלְאַחֲרֶיהָ, פַּת הַבָּאָה בְּכִיסָנִין הַשְּׁרוּיָה בְּמַשְׁקֶה אוֹ מְטֻבֶּלֶת בִּדְבַשׁ, וּפֵרוּרֵי לֶחֶם בְּתוֹךְ שֵׁכָר חַם שֶׁאֵין בָּהֶם מַמָּשׁוּת. וְאַחַר כָּךְ הַתְּנָאִים : אֵין הַטְּפֵלוּת אֶלָּא בִּשְׁעַת אֲכִילָה יַחַד ; וְהִיא פּוֹטֶרֶת אַף מִנְּטִילַת יָדַיִם ; וְאֵינָהּ יְכוֹלָה לִפְטֹר אֶת הַנֶּאֱכָל קֹדֶם הַשְּׁתִיָּה, שֶׁאָסוּר לֵהָנוֹת בְּלֹא בְּרָכָה תְּחִלָּה ; וְהִיא נִדְחֵית מִפְּנֵי הֶחָבִיב לְעוֹלָם ; וּצְרִיכָה שֶׁיִּהְיֶה הַטָּפֵל לְפָנָיו אוֹ שֶׁתִּהְיֶה דַּעְתּוֹ עָלָיו ; וּבְטֵלָה בְּשִׁנּוּי מָקוֹם מֵחֶדֶר לְחֶדֶר.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן רי״ב — י״א סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim רי״ב). אדמו״ר הזקן דן בו בְּעִנְיַן שֶׁהָעִקָּר פּוֹטֵר אֶת הַטָּפֵל. הטקסט מובא כלשונו, ובכלל זה סוגריים מרובעים של המהדורה (כגון גֻּדְגְּ[דָ]נִיּוֹת, [בָּ]זֶה, [הָאַחֲרוֹנוֹת]). אין מובא בעמוד זה שום קטע מחוץ לטקסט בר-בדיקה זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן רי״ב — שֶׁהָעִקָּר פּוֹטֵר אֶת הַטָּפֵל

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן רי״ב — שֶׁהָעִקָּר פּוֹטֵר אֶת הַטָּפֵל — וּבוֹ י״א סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.212). הסימן כולל י״א סעיפים (סעיף k → ‎.212.k).

סעיף א כָּל שֶׁהוּא עִקָּר וְעִמּוֹ טְפֵלָה — מְבָרֵךְ עַל הָעִקָּר וּפוֹטֵר אֶת הַטְּפֵלָה.

שֶׁהָעִקָּר פּוֹטֵר אֶת הַטָּפֵל, וּבוֹ י״א סְעִיפִים:
כָּל שֶׁהוּא עִקָּר וְעִמּוֹ טְפֵלָה — מְבָרֵךְ עַל הָעִקָּר וּפוֹטֵר אֶת הַטְּפֵלָה, בֵּין מִבְּרָכָה רִאשׁוֹנָה בֵּין מִבְּרָכָה אַחֲרוֹנָה.
אֵין צָרִיךְ לוֹמַר אִם הַטְּפֵלָה מְעֹרֶבֶת עִם הָעִקָּר, כְּגוֹן כָּל תַּעֲרֹבֶת שְׁנֵי מִינִים שֶׁהָאֶחָד הוּא עִקָּר וְהַשֵּׁנִי אֵינוֹ אֶלָּא לְתַקְּנוֹ וּלְהַכְשִׁירוֹ, אוֹ אֲפִלּוּ שְׁנֵיהֶם עִקָּר אֶלָּא שֶׁהָאֶחָד מְרֻבֶּה וְהָאֶחָד מוּעָט, שֶׁהָרֹב נֶחְשָׁב עִקָּר וְהַמּוּעָט טָפֵל, אוֹ בְּתַעֲרֹבֶת מַמָּשׁוּת דָּגָן, שֶׁאֲפִלּוּ הוּא מוּעָט נֶחְשָׁב עִקָּר.
וַאֲפִלּוּ אֵינוֹ מְעֹרָב מַמָּשׁ עִם הַטָּפֵל, אֶלָּא שֶׁנַּעֲשָׂה עַל דַּעַת לֶאֱכֹל שְׁנֵיהֶם יַחַד, כְּגוֹן עִסָּה מְמֻלֵּאת בְּפֵרוֹת, וְאָז אֲפִלּוּ אוֹכֵל הַטָּפֵל לְבַדּוֹ אַחַר שֶׁבֵּרַךְ עַל הָעִקָּר — אֵין צָרִיךְ לַחֲזֹר וּלְבָרֵךְ.
אֶלָּא אֲפִלּוּ אֵינָן מְעֹרָבִין כְּלָל זֶה עִם זֶה, אֶלָּא שֶׁעִקַּר הָאֲכִילָה הוּא מִמִּין אֶחָד וְהַשֵּׁנִי אֵינוֹ בָּא אֶלָּא מֵחֲמַת הָרִאשׁוֹן, שֶׁאִם לֹא הָיָה אוֹכֵל הָרִאשׁוֹן לֹא הָיָה אוֹכְלוֹ כְּלָל עַתָּה, הֲרֵי הוּא טָפֵל אֵלָיו וְנִפְטָר בְּבִרְכָתוֹ, אֲפִלּוּ הוּא לֶחֶם, שֶׁהוּא חָשׁוּב מִכָּל הַמִּינִים, כְּגוֹן מִי שֶׁחָפֵץ לֶאֱכֹל דָּג מָלוּחַ וּכְדֵי שֶׁלֹּא יַזִּיקֶנּוּ בִּגְרוֹנוֹ אוֹכֵל עִמּוֹ פַּת — הֲרֵי הוּא טְפֵלָה לַדָּג וְנִפְטֶרֶת בְּבִרְכָתוֹ.
וְהוּא שֶׁאֵינוֹ רָעֵב וְאֵינוֹ תָּאֵב כְּלָל לַאֲכִילַת הַפַּת, וְלֹא הָיָה אוֹכְלוֹ כְּלָל עַכְשָׁו אִם לֹא הַמָּלִיחַ, אֲבָל אִם הוּא תָּאֵב גַּם לַאֲכִילַת פַּת, אַף עַל פִּי שֶׁאוֹכְלוֹ עִם הַמָּלִיחַ — אֵינָהּ טְפֵלָה אֵלָיו, אֲפִלּוּ אִם תָּאֵב לוֹ יוֹתֵר, וְצָרִיךְ לְבָרֵךְ עָלֶיהָ "הַמּוֹצִיא" וּפוֹטֵר אֶת הַמָּלִיחַ, הוֹאִיל וְהוּא מִדְּבָרִים שֶׁדַּרְכָּן לָבוֹא תּוֹךְ הַסְּעוּדָה, שֶׁהֵם נִפְטָרִים לְעוֹלָם בְּבִרְכַּת הַפַּת, אַף שֶׁאֵינָם בָּאִים מֵחֲמָתָהּ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן קע״ז:

סעיף ב פַּת שְׁרוּיָה בְּיֵין שָׂרָף לִפְנֵי הַסְּעוּדָה — הַפַּת עִקָּר וְהַיַּיִן טָפֵל.

וְכֵן מִי שֶׁאוֹכֵל לִפְנֵי סְעוּדָה פַּת שְׁרוּיָה בְּיֵין שָׂרָף, אַף עַל פִּי שֶׁעִקַּר כַּוָּנָתוֹ בָּזֶה עַל הַיֵּין שָׂרָף, אֶלָּא מִפְּנֵי שֶׁהוּא חָזָק וְקָשֶׁה לוֹ לִשְׁתּוֹתוֹ לְבַדּוֹ שׁוֹרֶה בּוֹ פַּת וְאוֹכְלוֹ, מִכָּל מָקוֹם, כֵּיוָן שֶׁהוּא לִפְנֵי הַסְּעוּדָה וַדַּאי כַּוָּנָתוֹ גַּם כֵּן לִסְעֹד הַלֵּב, וּלְפִיכָךְ אֵין הַפַּת טְפֵלָה לְהַיֵּין שָׂרָף, וְצָרִיךְ לְבָרֵךְ עָלֶיהָ "הַמּוֹצִיא", וּפוֹטֵר אֶת הַיֵּין שָׂרָף אֲפִלּוּ אֵין כַּוָּנָתוֹ בּוֹ כְּדֵי לְעוֹרֵר תַּאֲוַת הַמַּאֲכָל, שֶׁלֹּא הָיָה נִפְטַר בְּבִרְכַּת הַפַּת אִם הָיָה שׁוֹתֵהוּ בִּפְנֵי עַצְמוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן קע״ד.
וְכֵן אֲפִלּוּ אִם שׁוֹרֶה פַּת בְּיַיִן שֶׁצָּרִיךְ לְבָרֵךְ עָלָיו "בּוֹרֵא פְּרִי הַגָּפֶן" אַף בְּתוֹךְ הַסְּעוּדָה — אֵין צָרִיךְ לְבָרֵךְ עַל הַבָּלוּעַ בַּפַּת הַשְּׁרוּיָה בּוֹ, שֶׁכֵּיוָן שֶׁכַּוָּנָתוֹ גַּם כֵּן לִסְעֹד הַלֵּב — הַפַּת עִקָּר וְהַיַּיִן טָפֵל, שֶׁכָּל תַּעֲרֹבֶת מִין דָּגָן עִם מִין אַחֵר — הַדָּגָן עִקָּר לְעוֹלָם, כָּל שֶׁמִּתְכַּוְּנִין גַּם כֵּן לַאֲכִילָתוֹ, אַף עַל פִּי שֶׁעִקַּר הַכַּוָּנָה הוּא בִּשְׁבִיל מִין הָאַחֵר, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ר״ח:

סעיף ג פַּת הַבָּאָה בְּכִיסָנִין הַשְּׁרוּיָה בְּמַשְׁקֶה — וּמָתַי נַעֲשֵׂית טְפֵלָה לוֹ.

וּלְפִיכָךְ, אֲפִלּוּ פַּת הַבָּאָה בְּכִיסָנִין שֶׁמְּבָרֵךְ עָלֶיהָ "בּוֹרֵא מִינֵי מְזוֹנוֹת", אִם שׁוֹרֶה אוֹתָהּ בְּיַיִן אוֹ יֵין שָׂרָף וּשְׁאָר מַשְׁקִין — אֵין צָרִיךְ לְבָרֵךְ עַל מַשְׁקֶה הַבָּלוּעַ בָּהּ, וְלֹא עַל מַשְׁקֶה שֶׁעַל גַּבָּהּ, כְּגוֹן שֶׁמְּטַבֵּל בִּדְבַשׁ וְאוֹכֵל, וַאֲפִלּוּ אִם עִקַּר כַּוָּנָתוֹ עַל הַמַּשְׁקֶה (שֶׁכֵּיוָן שֶׁהִיא מֵה׳ מִינִים הִיא עִקָּר).
אֶלָּא אִם כֵּן אֵינוֹ מִתְכַּוֵּן לַאֲכִילָתָם כְּלָל, אֶלָּא כְּדֵי שֶׁלֹּא יְטַנְּפוּ אֶת הַיָּדַיִם מֵהַדְּבַשׁ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן קס״ח, אוֹ שׁוֹרֶה בְּיֵין שָׂרָף וְכַוָּנָתוֹ עַל הַיֵּין שָׂרָף בִּלְבַד, אֶלָּא שֶׁמִּפְּנֵי שֶׁהוּא חָזָק וְקָשֶׁה לוֹ לִשְׁתּוֹתוֹ לְבַדּוֹ הוּא שׁוֹרֶה בּוֹ פַּת וְאוֹכְלָהּ, אַף עַל פִּי שֶׁאֵינוֹ מִתְאַוֶּה לָהּ כְּלָל, כְּגוֹן מִי שֶׁשּׁוֹרֶה אַחַר הַסְּעוּדָה מְעַט פַּת בְּיֵין שָׂרָף וְאוֹכְלָהּ לְעַכֵּל הַמַּאֲכָל, שֶׁאָז נַעֲשֵׂית הַפַּת טְפֵלָה לַיֵּין שָׂרָף, וְצָרִיךְ לְבָרֵךְ עָלֶיהָ "שֶׁהַכֹּל", כְּדִין יֵין שָׂרָף הַבָּא לְעַכֵּל הַמַּאֲכָל, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן קע״ד.
(וְאִם שׁוֹרֶה בּוֹ מַאֲכָל שֶׁמִּתְאַוֶּה לוֹ וְאֵינוֹ מֵחֲמֵשֶׁת הַמִּינִים — בָּאנוּ לַמַּחֲלֹקֶת שֶׁנִּתְבָּאֵר בְּסִימָן ר״ב, שֶׁיֵּשׁ אוֹמְרִים שֶׁכָּל תַּעֲרֹבֶת ב׳ מִינִים שֶׁאֶחָד מֵהֶם הָיָה אוֹכְלוֹ גַּם אִלּוּ הָיָה לְבַדּוֹ וְהַשֵּׁנִי לֹא הָיָה אוֹכְלוֹ לְבַדּוֹ — נַעֲשָׂה זֶה טָפֵל לָזֶה הוֹאִיל וְאֵינוֹ אוֹכְלוֹ אֶלָּא עַל יְדֵי שֶׁמְּעֹרָב [בָּ]זֶה, וְאַף עַל פִּי שֶׁכַּוָּנָתוֹ עַל שְׁנֵיהֶם בְּשָׁוֶה. וְיֵשׁ חוֹלְקִים בָּזֶה. וּסְפֵק בְּרָכוֹת לְהָקֵל, כְּמוֹ שֶׁנִּתְבָּאֵר שָׁם):

סעיף ד מְפָרֵר לֶחֶם דַּק דַּק לְתוֹךְ שֵׁכָר — הַשֵּׁכָר עִקָּר.

וַאֲפִלּוּ אִם מִתְכַּוֵּן וּמִתְאַוֶּה לִטְעֹם הַפַּת הַשְּׁרוּיָה בַּמַּשְׁקֶה, אֶלָּא שֶׁאֵינוֹ עוֹשֶׂה לַאֲכִילָה אֶלָּא לִשְׁתִיָּה, כְּגוֹן שֶׁמְּפָרֵר לֶחֶם דַּק דַּק וְנוֹתֵן לְתוֹךְ שֵׁכָר חַם כְּדֵי שֶׁיִּתֵּן בּוֹ טַעַם, כֵּיוָן שֶׁאֵין בַּלֶּחֶם מַמָּשׁוּת וְדָבָר חָשׁוּב — הַשֵּׁכָר עִקָּר (שֶׁלֹּא אָמְרוּ שֶׁכָּל תַּעֲרֹבֶת שֶׁיֵּשׁ בּוֹ דָּגָן הַדָּגָן עִקָּר, אֶלָּא כְּשֶׁיֵּשׁ בּוֹ מַמָּשׁוּת הַדָּגָן, וְלֹא רִבָּה בּוֹ מַשְׁקֶה הַרְבֵּה עַד שֶׁנַּעֲשָׂה רַךְ וְרָאוּי לִשְׁתִיָּה אֶלָּא הוּא עָב וְרָאוּי לַאֲכִילָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ר״ח):

סעיף ה מַה שֶּׁאוֹכֵל אַחַר כָּךְ לְבַדּוֹ — צָרִיךְ לְבָרֵךְ עָלָיו תְּחִלָּה וָסוֹף.

וְכָל זֶה בְּפַת אוֹ מַאֲכָל אַחֵר שֶׁאוֹכֵל עִם הַמָּלִיחַ אוֹ עִם הַיֵּין שָׂרָף בְּיַחַד, אֲבָל מַה שֶּׁאוֹכֵל אַחַר כָּךְ לְבַדּוֹ — צָרִיךְ לְבָרֵךְ עָלָיו תְּחִלָּה וָסוֹף, כֵּיוָן שֶׁבַּאֲכִילָה זוֹ אֵינֶנּוּ טָפֵל כְּלָל, וְגַם מִתְּחִלָּה לֹא נַעֲשָׂה טָפֵל אֶלָּא בִּשְׁעַת אֲכִילָתוֹ אוֹתָם יַחַד, וְלֹא מִתְּחִלַּת עֲשִׂיָּתָם יַחַד בַּאֲפִיָּה אוֹ בִּשּׁוּל, שֶׁעַל יְדֵי זֶה נַעֲשׂוּ כְּדָבָר אֶחָד מַמָּשׁ לִבְרָכָה, לְעִנְיָן שֶׁהָעִקָּר פּוֹטֵר אֶת הַטָּפֵל אַף אִם אוֹכְלוֹ אַחַר כָּךְ לְבַד (אֲבָל לֹא לְעִנְיָן שֶׁיְּבָרֵךְ עַל הַטָּפֵל בִּרְכַּת הָעִקָּר אִם בָּא לְאָכְלוֹ לְבַדּוֹ מִתְּחִלָּה, מִטַּעַם שֶׁיִּתְבָּאֵר).
וְאִם גַּם בַּאֲכִילָה זוֹ הוּא טָפֵל — אֵין צָרִיךְ לְבָרֵךְ עָלָיו אַף עַל פִּי שֶׁאוֹכְלוֹ לְבַדּוֹ, כְּגוֹן מִי שֶׁאוֹכֵל פַּת לְמַתֵּק הַשְּׁתִיָּה שֶׁשָּׁתָה כְּבָר — אֵין צָרִיךְ לְבָרֵךְ עָלֶיהָ, שֶׁהִיא טְפֵלָה לַמַּשְׁקֶה וְנִפְטֶרֶת בְּבִרְכָתוֹ:

סעיף ו כָּל פַּת שֶׁהִיא טְפֵלָה לְעִנְיַן בְּרָכָה — טְפֵלָה גַּם לִנְטִילַת יָדַיִם.

וְכָל פַּת שֶׁהִיא טְפֵלָה לְעִנְיַן בְּרָכָה — הִיא טְפֵלָה גַּם כֵּן לְעִנְיַן נְטִילַת יָדַיִם, וְאֵין צָרִיךְ לִטּוֹל יָדָיו, אֲפִלּוּ אוֹכֵל כְּבֵיצָה.
וְטוֹב לִמְנֹעַ מִלֶּאֱכֹל פַּת לְמַתֵּק הַשְּׁתִיָּה, כִּי מִי יוּכַל לְהַבְחִין אִם אוֹכְלָהּ לְמַתֵּק הַשְּׁתִיָּה אוֹ לְמִלּוּי כֶּרֶס, שֶׁאָז צָרִיךְ לְבָרֵךְ עָלֶיהָ "הַמּוֹצִיא":

סעיף ז הָאוֹכֵל קֹדֶם הַשְּׁתִיָּה — צָרִיךְ לְבָרֵךְ בְּרָכָה הַמְיֻחֶדֶת לָהּ.

וְכָל זֶה בְּמִתּוּק הַשְּׁתִיָּה שֶׁשָּׁתָה כְּבָר, אֲבָל הָאוֹכֵל גַּרְעִינֵי גֻּדְגְּ[דָ]נִיּוֹת וְכַיּוֹצֵא בָּהֶם לְמַתֵּק הַשְּׁתִיָּה שֶׁיִּשְׁתֶּה, שֶׁאֲכִילָתָם גּוֹרֶרֶת לוֹ תַּאֲוָה לִשְׁתִיָּה, אַף עַל פִּי שֶׁאֲכִילָתָם טְפֵלָה לְהַשְּׁתִיָּה וְהָיְתָה נִפְטֶרֶת בְּבִרְכָתָהּ אִם הָיָה אוֹכֵל אַחַר שֶׁבֵּרַךְ עַל הַשְּׁתִיָּה, מִכָּל מָקוֹם, כְּשֶׁאוֹכֵל קֹדֶם הַשְּׁתִיָּה אִי אֶפְשָׁר שֶׁיִּפָּטֵר בְּבִרְכַּת הַשְּׁתִיָּה שֶׁיְּבָרֵךְ אַחַר אֲכִילָתוֹ, כִּי אָסוּר לֵהָנוֹת מֵהָעוֹלָם הַזֶּה בְּלֹא בְּרָכָה בַּתְּחִלָּה, לָכֵן צָרִיךְ לְבָרֵךְ עֲלֵיהֶם בְּרָכָה הַמְיֻחֶדֶת לָהֶם.
וְכֵן הָרוֹצֶה לִשְׁתּוֹת בַּבֹּקֶר אוֹ אַחַר שֵׁנַת צָהֳרַיִם, וְרוֹצֶה לֶאֱכֹל מְעַט תְּחִלָּה כְּדֵי שֶׁלֹּא יִשְׁתֶּה עַל לֵב רֵיקָן, אַף עַל פִּי שֶׁאֲכִילָה זוֹ הִיא צֹרֶךְ הַשְּׁתִיָּה וּטְפֵלָה אֵלֶיהָ — צָרִיךְ לְבָרֵךְ עָלֶיהָ בְּרָכָה הַמְיֻחֶדֶת לָהּ, כְּדֵי שֶׁלֹּא יֵהָנֶה מֵהָעוֹלָם הַזֶּה בְּלֹא בְּרָכָה:

סעיף ח בַּמֶּה דְּבָרִים אֲמוּרִים — וּלְעוֹלָם יַרְבֶּה הָאָדָם בַּבְּרָכוֹת הַצְּרִיכוֹת.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁשּׁוֹתֶה יַיִן שֶׁבִּרְכָתוֹ "בּוֹרֵא פְּרִי הַגָּפֶן", שֶׁאִי אֶפְשָׁר לְבָרֵךְ בְּרָכָה זוֹ עַל הַמַּאֲכָל, אֲבָל הַשּׁוֹתֶה מַשְׁקֶה שֶׁבִּרְכָתוֹ "שֶׁהַכֹּל" — יֵשׁ אוֹמְרִים שֶׁיְּבָרֵךְ עַל הַמַּאֲכָל שֶׁאוֹכֵל תְּחִלָּה "שֶׁהַכֹּל" וְלֹא בִּרְכָתוֹ הַמְיֻחֶדֶת לוֹ, שֶׁכֵּיוָן שֶׁאֲכִילָתוֹ אֵינָהּ אֶלָּא לְצֹרֶךְ הַשְּׁתִיָּה וּטְפֵלָה אֵלֶיהָ — מְבָרֵךְ עָלֶיהָ בִּרְכַּת הַשְּׁתִיָּה שֶׁהִיא עִקָּר.
אֲבָל כָּל טָפֵל שֶׁאֵין אֲכִילָתוֹ טְפֵלָה אֶל הָעִקָּר אֶלָּא כְּשֶׁאוֹכֵל שְׁנֵיהֶם יַחַד אוֹ הַטָּפֵל אַחַר הָעִקָּר, וּכְשֶׁאוֹכְלוֹ קֹדֶם הָעִקָּר אֵין אֲכִילָתוֹ טְפֵלָה אֵלָיו כְּלָל, כְּגוֹן שֶׁהֵבִיאוּ אֵלָיו פַּת אוֹ מַאֲכָל אַחֵר לֶאֱכֹל עִם הַמָּלִיחַ שֶׁלֹּא יַזִּיקֶנּוּ בִּגְרוֹנוֹ אוֹ לְמַתֵּק הַשְּׁתִיָּה אַחַר שֶׁיִּשְׁתֶּה, וּכְשֶׁבָּא לֶאֱכֹל מִמֶּנּוּ לִפְנֵי הַמָּלִיחַ אוֹ לִפְנֵי הַשְּׁתִיָּה אֵין אֲכִילָה זוֹ טְפֵלָה אֵלֶיהָ כְּלָל — צָרִיךְ לְבָרֵךְ עָלֶיהָ בְּרָכָה הַמְיֻחֶדֶת לָהּ וְלֹא בִּרְכַּת הַמָּלִיחַ אוֹ הַמַּשְׁקֶה, אַף עַל פִּי שֶׁהִיא "שֶׁהַכֹּל".
לְפִיכָךְ, טוֹב שֶׁיֹּאכַל מִמֶּנּוּ מְעַט קֹדֶם בִּרְכַּת הַמָּלִיחַ אוֹ הַמַּשְׁקֶה, כְּדֵי שֶׁיְּבָרֵךְ עָלָיו בְּרָכָה הַמְיֻחֶדֶת לוֹ, וְלֹא יִפְטְרֶנּוּ בִּבְרָכָה אַחֶרֶת בְּתוֹרַת טְפֵלָה.
וְאַף עַל פִּי שֶׁמִּן הַדִּין אֵין לָחֹשׁ לְכָךְ, שֶׁאַף שֶׁבִּרְכַּת הָעִקָּר הִיא "שֶׁהַכֹּל" שֶׁהִיא בְּרָכָה כּוֹלֶלֶת וּבִרְכַּת הַטְּפֵלָה הִיא בְּרָכָה מְבֹרֶרֶת וּפוֹרֶטֶת — אֵין לָהּ דִּין קְדִימָה עַל הָעִקָּר, כֵּיוָן שֶׁאֵינוֹ אוֹכֵל עַתָּה הַטָּפֵל אֶלָּא בִּשְׁבִיל אֲכִילַת הָעִקָּר, אִם כֵּן קֹדֶם שֶׁהִתְחִיל בַּאֲכִילַת הָעִקָּר שֶׁעֲדַיִן אֵינוֹ חָפֵץ לֶאֱכֹל הַטָּפֵל — אֵינוֹ מְחֻיָּב לְאָכְלוֹ בִּשְׁבִיל קְדִימַת בִּרְכָתוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן רי״א, מִכָּל מָקוֹם כֵּיוָן שֶׁהָעִקָּר וְהַטָּפֵל אֵין בִּרְכוֹתֵיהֶם שָׁווֹת, וְאִם יְבָרֵךְ עַל הָעִקָּר תְּחִלָּה לֹא יְבָרֵךְ כְּלָל עַל הַטָּפֵל, וְאִם יְבָרֵךְ עַל הַטָּפֵל תְּחִלָּה יַחֲזֹר וִיבָרֵךְ עַל הָעִקָּר — טוֹב יוֹתֵר לַעֲשׂוֹת כֵּן, לְהַרְבּוֹת בַּבְּרָכוֹת, כִּי לְעוֹלָם יַרְבֶּה הָאָדָם בַּבְּרָכוֹת הַצְּרִיכוֹת וִימַעֵט בְּשֶׁאֵינָן צְרִיכוֹת, וְזוֹ הִיא בְּרָכָה הַצְּרִיכָה, שֶׁהֲרֵי הִיא מְיֻחֶדֶת לְהַטָּפֵל וְלֹא לְהָעִקָּר, וְגַם (בְּ)בִרְכַּת הָעִקָּר אֵינָהּ מְיֻחֶדֶת אֶל הַטָּפֵל אֶלָּא כּוֹלֶלֶת גַּם אֶת הַטָּפֵל, וְלֹא צִוּוּ לְמַעֵט אֶלָּא כְּשֶׁיָּכוֹל לִפְטֹר שְׁנֵי דְּבָרִים בִּבְרָכָה אַחַת הַמְיֻחֶדֶת לִשְׁנֵיהֶם:

סעיף ט עִקָּר הֶחָבִיב לְעוֹלָם יוֹתֵר מִן הַטָּפֵל — מְבָרֵךְ עָלָיו תְּחִלָּה.

בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁאֵין הָעִקָּר חָבִיב לוֹ יוֹתֵר מֵהַטָּפֵל, אֶלָּא שֶׁהַשָּׁעָה צְרִיכָה לְכָךְ לַעֲשׂוֹת מִמֶּנּוּ עִקָּר, כְּגוֹן שֶׁאָכַל כְּבָר פֵּרוֹת מְתוּקִין בְּיוֹתֵר וְהֻצְרַךְ לֶאֱכֹל אַחֲרֵיהֶם מָלִיחַ לְחַתֵּךְ הַלֵּחוֹת הַנִּקְבָּצוֹת בָּאִצְטוֹמְכָא, וְאוֹכֵל עִמּוֹ אֵיזֶה מַאֲכָל שֶׁלֹּא יַזִּיקֶנּוּ בִּגְרוֹנוֹ, אֲבָל אֵין הַמָּלִיחַ חָבִיב עָלָיו וְאֵינוֹ מִתְאַוֶּה לוֹ כְּלָל, כְּמוֹ שֶׁאֵינוֹ מִתְאַוֶּה לְהַמַּאֲכָל עַכְשָׁו אִם לֹא הַמָּלִיחַ.
אוֹ אֲפִלּוּ אִם הָעִקָּר חָבִיב עָלָיו עַכְשָׁו, אֶלָּא שֶׁבְּרֹב הַפְּעָמִים הַטָּפֵל חָבִיב עָלָיו יוֹתֵר, וְאִם כֵּן אֵין קְדִימָה לְהָעִקָּר, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן רי״א.
וְאֵין צָרִיךְ לוֹמַר אִם גַּם עַכְשָׁו הַטָּפֵל חָבִיב עָלָיו יוֹתֵר, אֶלָּא שֶׁאַף עַל פִּי כֵן לֹא הָיָה אוֹכְלוֹ אִם לֹא שֶׁהֻצְרַךְ לֶאֱכֹל הַמָּלִיחַ, וּמִפְּנֵי זֶה הוּא טָפֵל אֵלָיו. וְאַף שֶׁאַף עַל פִּי כֵן אֵין צָרִיךְ לְהַקְדִּים הַטָּפֵל מִן הַדִּין אַף עַל פִּי שֶׁהוּא חָבִיב, כֵּיוָן שֶׁלֹּא הָיָה אוֹכְלוֹ כְּלָל אִם לֹא הָעִקָּר, מִכָּל מָקוֹם, כְּדֵי לְהַרְבּוֹת בַּבְּרָכוֹת — טוֹב לְהַקְדִּימוֹ, כֵּיוָן שֶׁאֵין לְהָעִקָּר דִּין קְדִימָה מִצַּד עַצְמוֹ.
וְאַף אִם מִצַּד בִּרְכָתוֹ יֵשׁ לוֹ דִּין קְדִימָה, כְּגוֹן שֶׁבִּרְכָתוֹ "בּוֹרֵא פְּרִי הָאֲדָמָה" אוֹ "בּוֹרֵא פְּרִי הָעֵץ" וּבִרְכַּת הַטָּפֵל הִיא "שֶׁהַכֹּל", מִכָּל מָקוֹם, כֵּיוָן שֶׁאִם יְבָרֵךְ עַל הָעִקָּר תְּחִלָּה "בּוֹרֵא פְּרִי הָעֵץ" אוֹ "בּוֹרֵא פְּרִי הָאֲדָמָה" לֹא יְבָרֵךְ כְּלָל עַל הַטָּפֵל שֶׁבִּרְכָתוֹ "שֶׁהַכֹּל נִהְיָה בִּדְבָרוֹ" — לֹא שַׁיָּךְ כְּלָל בָּזֶה קְדִימָה, וְיוֹתֵר טוֹב לְהַרְבּוֹת בַּבְּרָכוֹת, כְּמוֹ שֶׁנִּתְבָּאֵר.
אֲבָל אִם הָעִקָּר חָבִיב עָלָיו לְעוֹלָם יוֹתֵר מִן הַטָּפֵל — צָרִיךְ לְבָרֵךְ עָלָיו תְּחִלָּה, שֶׁכֵּיוָן שֶׁהוּא חָשׁוּב מִמֶּנּוּ וּמֻקְדָּם לִבְרָכָה מֵחֲמַת חֲבִיבוּת עַצְמוֹ וְלֹא מֵחֲמַת בִּרְכָתוֹ — רָאוּי הוּא לְבָרֵךְ עָלָיו וְלִפְטֹר הַטָּפֵל בְּבִרְכָתוֹ מֵחֲמַת טְפֵלָתוֹ, שֶׁזֶּהוּ עִקַּר מַעֲלַת חֲשִׁיבוּתוֹ כְּשֶׁפּוֹטֵר דָּבָר אַחֵר בְּבִרְכָתוֹ, וְלֹא קְדִימָה לְבַדָּהּ, שֶׁאֵין בָּהּ מַעֲלָה כָּל כָּךְ לַדְּבָרִים הַחֲשׁוּבִים מִצַּד עַצְמָן, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן רי״א.
וְאַף אִם בִּרְכַּת הָעִקָּר הִיא "שֶׁהַכֹּל" וּבִרְכַּת הַטָּפֵל הִיא מְבֹרֶרֶת וּפְרָטִית — אֵין זֶה מוֹעִיל לְהַקְדִּימוֹ, כֵּיוָן שֶׁלֹּא הָיָה אוֹכְלוֹ כְּלָל אִם לֹא הָעִקָּר.
וַאֲפִלּוּ אִם עַכְשָׁו כְּשֶׁאוֹכֵל הַטָּפֵל עִם הָעִקָּר הַטָּפֵל חָבִיב עָלָיו יוֹתֵר אֶלָּא שֶׁכְּשֶׁאוֹכְלוֹ לְבַדּוֹ אֵינוֹ חָבִיב לוֹ כְּמוֹ הָעִקָּר — אֵין לוֹ לְבָרֵךְ עָלָיו כְּלָל, כֵּיוָן שֶׁכְּשֶׁבָּא לְאָכְלוֹ לְבַדּוֹ הֲרֵי אֵינוֹ חָבִיב עָלָיו כְּמוֹ הָעִקָּר, וּכְשֶׁבָּא לְאָכְלוֹ עִם הָעִקָּר הֲרֵי הוּא טָפֵל אֵלָיו, וְכָל עִקָּר וְטָפֵל אַף עַל פִּי שֶׁהַטָּפֵל חָבִיב — מְבָרֵךְ עַל הָעִקָּר וּפוֹטֵר אֶת הַטָּפֵל הַנֶּאֱכָל עִמּוֹ אוֹ אַחֲרָיו:

סעיף י אֵין הַטָּפֵל נִפְטָר אֶלָּא כְּשֶׁהָיָה לְפָנָיו אוֹ שֶׁהָיְתָה דַּעְתּוֹ עָלָיו.

לְעוֹלָם אֵין הַטָּפֵל נִפְטָר בְּבִרְכַּת הָעִקָּר, וַאֲפִלּוּ אִם הֵם שְׁנֵי מִינִים שֶׁבִּרְכוֹתֵיהֶם שָׁווֹת, אֶלָּא כְּשֶׁהָיָה הַטָּפֵל לְפָנָיו, אוֹ שֶׁהָיְתָה דַּעְתּוֹ עָלָיו בְּבֵרוּר בְּשָׁעָה שֶׁבֵּרַךְ עַל הָעִקָּר, אֲבָל אִם אַחַר הַבְּרָכָה הֱבִיאוּהוּ לְפָנָיו וְלֹא הָיְתָה דַּעְתּוֹ עָלָיו בְּבֵרוּר, בְּעִנְיָן שֶׁאִם לֹא הָיָה טָפֵל הָיָה צָרִיךְ לַחֲזֹר וּלְבָרֵךְ — אֵין טְפֵלָתוֹ מוֹעֶלֶת כְּלוּם לְפָטְרוֹ בְּבִרְכַּת הָעִקָּר, לֹא מִבְּרָכָה רִאשׁוֹנָה אַף עַל פִּי שֶׁבִּרְכוֹתֵיהֶם שָׁווֹת, וְעַל דֶּרֶךְ מַה שֶּׁנִּתְבָּאֵר בְּסִימָן ר״ו, וְלֹא מִבְּרָכָה אַחֲרוֹנָה כְּשֶׁאֵין בִּרְכוֹתֵיהֶם [הָאַחֲרוֹנוֹת] שָׁווֹת, אֶלָּא אִם כֵּן קָבַע עַצְמוֹ עַל הָעִקָּר, וְהוּא דָּבָר שֶׁקְּבִיעוּת מוֹעֶלֶת בּוֹ מִפְּנֵי שֶׁדַּרְכּוֹ בְּכָךְ, כְּגוֹן פַּת, וְכֵן יַיִן, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ר״ו, וְכֵן שֵׁכָר וּדְבַשׁ בִּמְדִינוֹת אֵלּוּ, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן רי״ג.
וּמִכָּל מָקוֹם, מִי שֶׁאוֹכֵל פַּת אוֹ מַאֲכָל אַחֵר לְמַתֵּק הַשְּׁתִיָּה וְדַרְכּוֹ בְּכָךְ בְּרֹב הַפְּעָמִים — הֲרֵי זֶה כְּאִלּוּ הָיְתָה דַּעְתּוֹ עָלָיו.
וְאִם אֵין דַּרְכּוֹ בְּכָךְ — מְבָרֵךְ עַל הַפַּת "שֶׁהַכֹּל" כְּבִרְכַּת הַמַּשְׁקֶה שֶׁהִיא טְפֵלָה אֵלָיו.
(אֲבָל אֵין צָרִיךְ לִטּוֹל יָדָיו תְּחִלָּה, אַף אִם הוּא בְּעִנְיָן שֶׁצָּרִיךְ לְבָרֵךְ עָלֶיהָ "הַמּוֹצִיא", כְּגוֹן שֶׁהִיא טְפֵלָה לְדָבָר שֶׁאֵין בִּרְכָתוֹ "שֶׁהַכֹּל", וּלְעִנְיַן נְטִילַת יָדַיִם לֹא שַׁיָּךְ לְחַלֵּק בֵּין אִם הָיְתָה דַּעְתּוֹ עַל הַטָּפֵל בִּשְׁעַת בִּרְכַּת הָעִקָּר אוֹ שֶׁנִּמְלַךְ עָלָיו אַחַר הַבְּרָכָה, אֶלָּא כֵּיוָן שֶׁאוֹכְלָהּ בְּתוֹרַת טְפֵלָה לְדָבָר אַחֵר — פָּטוּר מִנְּטִילַת יָדַיִם, שֶׁלֹּא תִּקְּנוּ לַפַּת אֶלָּא כְּשֶׁאוֹכְלָהּ לִסְעֹד הַלֵּב, שֶׁסְּעוּדָתָהּ רְגִילָה וּמְצוּיָה תָּמִיד, וְיֵשׁ לָחֹשׁ לְסֶרֶךְ תְּרוּמָה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן קנ״ח):

סעיף י״א טְפֵלָה הַנֶּאֱכֶלֶת אַחַר הָעִקָּר — בַּמָּקוֹם שֶׁאָכַל הָעִקָּר וְלֹא שִׁנָּה מְקוֹמוֹ.

וְכָל טְפֵלָה הַנֶּאֱכֶלֶת אַחַר אֲכִילַת הָעִקָּר אֵינָהּ נִפְטֶרֶת בְּבִרְכָתוֹ אֶלָּא אִם כֵּן אוֹכְלָהּ בַּמָּקוֹם שֶׁאָכַל הָעִקָּר וְלֹא שִׁנָּה מְקוֹמוֹ בֵּינְתַיִם מֵחֶדֶר לְחֶדֶר, כְּמוֹ שֶׁלֹּא הָיְתָה נִפְטֶרֶת אִם לֹא הָיְתָה טְפֵלָה, מִפְּנֵי שֶׁעֲמִידָתוֹ מִמְּקוֹמוֹ הוּא גְּמַר סְעוּדָתוֹ, וּמַה שֶּׁאוֹכֵל אַחַר כָּךְ — סְעוּדָה אַחֶרֶת הִיא, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן קע״ח.
אֶלָּא אִם כֵּן הִיא טְפֵלָה לְדָבָר הַטָּעוּן בְּרָכָה לְאַחֲרָיו בִּמְקוֹמוֹ, שֶׁאָז אֵין עֲמִידָתוֹ גְּמַר סְעוּדָתוֹ כָּל שֶׁלֹּא בֵּרַךְ בְּרָכָה הָאַחֲרוֹנָה, כְּמוֹ שֶׁנִּתְבָּאֵר:

2 · למעשה

הלכה למעשה — מנהג חב״ד

הנהגה מעשית לחב״ד

נקודות הנהגה הנובעות ישירות מסימן רי״ב ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן רי״ב (שֶׁהָעִקָּר פּוֹטֵר אֶת הַטָּפֵל)

  • ① סעיף א. כְּלָלוֹ שֶׁל סִימָן : עִקָּר וְטָפֵל. כָּל שֶׁהָאֶחָד עִקָּר וְהַשֵּׁנִי אֵינוֹ בָּא אֶלָּא מֵחֲמָתוֹ — בְּרָכָה אַחַת נֶאֱמֶרֶת, בִּרְכַּת הָעִקָּר, וְהִיא פּוֹטֶרֶת אֶת הַטָּפֵל בֵּין מִבְּרָכָה רִאשׁוֹנָה בֵּין מִבְּרָכָה אַחֲרוֹנָה. וְהָרַב מְסַדֵּר אַרְבַּע מַדְרֵגוֹת מִן הַקַּל אֶל הֶחָמוּר : תַּעֲרֹבֶת שֶׁהַשֵּׁנִי בָּא לְתַקֵּן אֶת הָרִאשׁוֹן ; תַּעֲרֹבֶת שֶׁשְּׁנֵיהֶם עִקָּר וְהָרֹב מַכְרִיעַ ; תַּעֲרֹבֶת שֶׁיֵּשׁ בָּהּ מַמָּשׁוּת דָּגָן, שֶׁאַף בְּמִעוּט הוּא עִקָּר ; וְלִבְסוֹף אַף שֶׁאֵינָם מְעֹרָבִין כְּלָל. וְשָׁם אַף הַפַּת, הַחֲשׁוּבָה מִכָּל הַמִּינִים, נַעֲשֵׂית טְפֵלָה — בְּדָג מָלוּחַ שֶׁאוֹכֵל עִמּוֹ פַּת שֶׁלֹּא יַזִּיקֶנּוּ בִּגְרוֹנוֹ. וּמִיָּד מַעֲמִיד הָרַב אֶת הַתְּנַאי הַמַּכְרִיעַ לְמַעֲשֶׂה : דַּוְקָא כְּשֶׁאֵינוֹ תָּאֵב כְּלָל לַפַּת ; אֲבָל תָּאֵב גַּם לָהּ — שׁוּב אֵינָהּ טְפֵלָה, וּמְבָרֵךְ "הַמּוֹצִיא" וְהִיא פּוֹטֶרֶת אֶת הַמָּלִיחַ.
  • ② סעיף ב. פַּת שְׁרוּיָה בְּיֵין שָׂרָף לִפְנֵי הַסְּעוּדָה : אַף שֶׁכַּוָּנָתוֹ עַל הַמַּשְׁקֶה, כֵּיוָן שֶׁהוּא לִפְנֵי הַסְּעוּדָה וַדַּאי כַּוָּנָתוֹ גַּם כֵּן לִסְעֹד הַלֵּב — וְאֵין הַפַּת טְפֵלָה, אֶלָּא מְבָרֵךְ "הַמּוֹצִיא" וּפוֹטֵר אֶת הַיֵּין שָׂרָף. וְכָאן נִקְבָּע הַכְּלָל הַגָּדוֹל שֶׁל הַסִּימָן : כָּל תַּעֲרֹבֶת מִין דָּגָן עִם מִין אַחֵר — הַדָּגָן עִקָּר לְעוֹלָם, כָּל שֶׁמִּתְכַּוֵּן גַּם לַאֲכִילָתוֹ, וְאַף שֶׁעִקַּר הַכַּוָּנָה לַמִּין הָאַחֵר ; וּלְפִיכָךְ אֵין מְבָרְכִין עַל הַיַּיִן הַבָּלוּעַ בַּפַּת הַשְּׁרוּיָה.
  • ③ סעיף ג. וּמִכֹּחַ אוֹתוֹ כְּלָל אַף פַּת הַבָּאָה בְּכִיסָנִין : שְׁרוּיָה בְּיַיִן אוֹ בְּיֵין שָׂרָף אוֹ מְטֻבֶּלֶת בִּדְבַשׁ — הִיא הָעִקָּר, מִפְּנֵי שֶׁהִיא מֵחֲמֵשֶׁת מִינֵי דָּגָן, וְאֵין מְבָרְכִין עַל הַמַּשְׁקֶה. וּשְׁתֵּי יוֹצְאוֹת מִן הַכְּלָל : כְּשֶׁאֵינוֹ מִתְכַּוֵּן לַאֲכִילָתָהּ כְּלָל אֶלָּא שֶׁלֹּא יְטַנְּפוּ יָדָיו, וּכְשֶׁכַּוָּנָתוֹ עַל הַיֵּין שָׂרָף בִּלְבַד וְהַפַּת אֵינָהּ אֶלָּא לְהַכְשִׁירוֹ לִשְׁתִיָּה — כְּגוֹן הַשּׁוֹרֶה אַחַר הַסְּעוּדָה לְעַכֵּל הַמַּאֲכָל, שֶׁאָז הַפַּת טְפֵלָה וּמְבָרֵךְ עָלֶיהָ "שֶׁהַכֹּל". וּלְבַסּוֹף מֵבִיא הָרַב אֶת מַחֲלֹקֶת סִימָן ר״ב · 202 וּמַכְרִיעַ סְפֵק בְּרָכוֹת לְהָקֵל.
  • ④ סעיף ד. גְּבוּל דַּק וּמַעֲשִׂי : פֵּרוּרֵי לֶחֶם בְּתוֹךְ מַשְׁקֶה חַם לִתֵּן בּוֹ טַעַם. כָּאן יֵשׁ תַּאֲוָה לִטְעֹם, אֶלָּא שֶׁהַמַּעֲשֶׂה לִשְׁתִיָּה וְלֹא לַאֲכִילָה, וְאֵין בַּלֶּחֶם מַמָּשׁוּת — וּלְפִיכָךְ אֵין כְּלַל "הַדָּגָן עִקָּר" נוֹהֵג, וְהַשֵּׁכָר עִקָּר. וְזֶהוּ הַמִּבְחָן שֶׁמְּגַלֶּה הָרַב : אֵין הַדָּגָן מַכְרִיעַ אֶלָּא כְּשֶׁיֵּשׁ בּוֹ מַמָּשׁוּת, עָב וְרָאוּי לַאֲכִילָה, וְלֹא רַךְ וְרָאוּי לִשְׁתִיָּה.
  • ⑤ סעיף ה. לַטְּפֵלָה יֵשׁ זְמַן. אֵינָהּ אֶלָּא בְּמַה שֶּׁאוֹכֵל יַחַד עִם הָעִקָּר ; וּמַה שֶּׁאוֹכֵל אַחַר כָּךְ לְבַדּוֹ — צָרִיךְ בְּרָכָה תְּחִלָּה וָסוֹף. וְיוֹצֵא מִן הַכְּלָל : כְּשֶׁנַּעֲשׂוּ יַחַד בַּאֲפִיָּה אוֹ בְּבִשּׁוּל, שֶׁנַּעֲשׂוּ כְּדָבָר אֶחָד מַמָּשׁ לִבְרָכָה וְהָעִקָּר פּוֹטֵר אַף בַּאֲכִילָה נִפְרֶדֶת. וְאִם הַטְּפֵלוּת נִמְשֶׁכֶת בְּעֶצֶם הָאֲכִילָה — כְּגוֹן פַּת לְמַתֵּק הַשְּׁתִיָּה שֶׁשָּׁתָה כְּבָר — אֵין צָרִיךְ בְּרָכָה אַף שֶׁאוֹכְלָהּ לְבַדָּהּ.
  • ⑥ סעיף ו. נָפְקָא מִנָּה מַעֲשִׂית שֶׁרְגִילִים לִשְׁכֹּחַ : פַּת הַטְּפֵלָה לְעִנְיַן בְּרָכָה טְפֵלָה גַּם לִנְטִילַת יָדַיִם, וְאַף בִּכְבֵיצָה אֵין צָרִיךְ לִטֹּל. וּמוֹסִיף הָרַב עֵצָה שֶׁל זְהִירוּת, כְּדַרְכּוֹ : טוֹב לִמְנֹעַ מִלֶּאֱכֹל פַּת לְמַתֵּק הַשְּׁתִיָּה, כִּי מִי יוּכַל לְהַבְחִין אִם לְמִתּוּק הוּא אוֹכֵל אוֹ לְמִלּוּי כֶּרֶס — וְאָז חַיָּב "הַמּוֹצִיא".
  • ⑦ סעיף ז. הִפּוּךְ הַסֵּדֶר : מִתּוּק שְׁתִיָּה שֶׁכְּבָר שָׁתָה אֵינוֹ כַּאֲכִילָה שֶׁלִּפְנֵי הַשְּׁתִיָּה. הַנֶּאֱכָל תְּחִלָּה אִי אֶפְשָׁר שֶׁיִּפָּטֵר בִּבְרָכָה שֶׁתָּבוֹא אַחֲרָיו, שֶׁאָסוּר לֵהָנוֹת מֵהָעוֹלָם הַזֶּה בְּלֹא בְּרָכָה בַּתְּחִלָּה ; וּלְפִיכָךְ, אַף שֶׁהוּא טָפֵל, מְבָרֵךְ עָלָיו בְּרָכָה הַמְיֻחֶדֶת לוֹ. וּשְׁנֵי מַעֲשִׂים בְּכָל יוֹם : גַּרְעִינִים הַגּוֹרְרִים תַּאֲוַת שְׁתִיָּה, וּמְעַט אֲכִילָה בַּבֹּקֶר אוֹ אַחַר שֵׁנַת צָהֳרַיִם שֶׁלֹּא יִשְׁתֶּה עַל לֵב רֵיקָן.
  • ⑧ סעיף ח. וּמְחַלֵּק הָרַב : בְּיַיִן — אִי אֶפְשָׁר לוֹמַר "הַגָּפֶן" עַל הַמַּאֲכָל ; וּבְמַשְׁקֶה שֶׁבִּרְכָתוֹ "שֶׁהַכֹּל" — יֵשׁ אוֹמְרִים שֶׁיְּבָרֵךְ "שֶׁהַכֹּל" עַל הַמַּאֲכָל שֶׁאוֹכֵל תְּחִלָּה. אֲבָל כָּל שֶׁבַּאֲכִילָה שֶׁלִּפְנֵי הָעִקָּר אֵינוֹ טָפֵל כְּלָל — חוֹזֶרֶת לוֹ בִּרְכָתוֹ הַמְיֻחֶדֶת. וּמִכָּאן עֵצָתוֹ הַמַּעֲשִׂית : יֹאכַל מִמֶּנּוּ מְעַט קֹדֶם, כְּדֵי לְבָרֵךְ עָלָיו בְּרָכָה הַמְיֻחֶדֶת לוֹ. וְטַעַם הַדָּבָר רָאוּי לְעִיּוּן : אֵין זוֹ שְׁאֵלַת קְדִימָה — שֶׁאֵין לְהָעִקָּר קְדִימָה כָּאן — אֶלָּא שֶׁלֹּא תֹּאבַד בְּרָכָה צְרִיכָה — שֶׁלְּעוֹלָם יַרְבֶּה הָאָדָם בַּבְּרָכוֹת הַצְּרִיכוֹת וִימַעֵט בְּשֶׁאֵינָן צְרִיכוֹת.
  • ⑨ סעיף ט. וְכָל עֵצָה זוֹ דַּוְקָא כְּשֶׁאֵין הָעִקָּר חָבִיב. הָרַב מְצַיֵּר אֶת הַמִּקְרִים שֶׁבָּהֶם שֵׁם "עִקָּר" בָּא מִן הַשָּׁעָה וְלֹא מִן הַחֲבִיבוּת — כְּגוֹן מָלִיחַ אַחַר פֵּרוֹת מְתוּקִין בְּיוֹתֵר לְחַתֵּךְ הַלֵּחוֹת שֶׁבָּאִצְטוֹמְכָא — וְשָׁם טוֹב לְהַקְדִּים אֶת הַטָּפֵל לְהַרְבּוֹת בַּבְּרָכוֹת. אֲבָל עִקָּר הֶחָבִיב לְעוֹלָם — מְבָרֵךְ עָלָיו תְּחִלָּה וּפוֹטֵר, שֶׁעִקַּר מַעֲלַת הַחֲשִׁיבוּת אֵינָהּ בַּקְּדִימָה אֶלָּא בִּפְטִירַת דָּבָר אַחֵר בְּבִרְכָתוֹ. וְלֹא בִּרְכָה מְבֹרֶרֶת שֶׁל הַטָּפֵל וְלֹא חֲבִיבוּת שֶׁל שָׁעָה מְשַׁנּוֹת אֶת הַדִּין.
  • ⑩ סעיף י. תְּנַאי יְסוֹדִי, וְהוּא הַשָּׁכִיחַ בְּיוֹתֵר לְמַעֲשֶׂה : אֵין הַטְּפֵלוּת מוֹעֶלֶת אֶלָּא כְּשֶׁהַטָּפֵל הָיָה לְפָנָיו אוֹ שֶׁהָיְתָה דַּעְתּוֹ עָלָיו בְּבֵרוּר בִּשְׁעַת בִּרְכַּת הָעִקָּר. הוּבָא לְאַחַר הַבְּרָכָה וְלֹא הָיְתָה דַּעְתּוֹ עָלָיו — אֵינוֹ נִפְטָר לֹא בָּרִאשׁוֹנָה וְלֹא בָּאַחֲרוֹנָה, אֶלָּא אִם כֵּן קָבַע עַצְמוֹ עַל דָּבָר שֶׁקְּבִיעוּת מוֹעֶלֶת בּוֹ (פַּת, יַיִן ; וְשֵׁכָר וּדְבַשׁ בִּמְדִינוֹת אֵלּוּ). וּשְׁנֵי פְּרָטִים : הָרָגִיל בְּכָךְ בְּרֹב הַפְּעָמִים — כְּאִלּוּ הָיְתָה דַּעְתּוֹ עָלָיו ; וְשֶׁאֵין דַּרְכּוֹ בְּכָךְ — מְבָרֵךְ עַל הַפַּת "שֶׁהַכֹּל" כְּבִרְכַּת הַמַּשְׁקֶה. וּלְעִנְיַן יָדַיִם — פַּת הַנֶּאֱכֶלֶת בְּתוֹרַת טְפֵלָה פְּטוּרָה לְעוֹלָם מִנְּטִילָה.
  • ⑪ סעיף י״א. וְתְנַאי אַחֲרוֹן : הַמָּקוֹם. טְפֵלָה הַנֶּאֱכֶלֶת אַחַר הָעִקָּר אֵינָהּ נִפְטֶרֶת אֶלָּא בַּמָּקוֹם שֶׁאָכַל הָעִקָּר וְלֹא שִׁנָּה מְקוֹמוֹ מֵחֶדֶר לְחֶדֶר, שֶׁעֲמִידָתוֹ מִמְּקוֹמוֹ הִיא גְּמַר סְעוּדָתוֹ וּמַה שֶּׁאַחֲרֶיהָ סְעוּדָה אַחֶרֶת. וְיוֹצֵא מִן הַכְּלָל : כְּשֶׁהִיא טְפֵלָה לְדָבָר הַטָּעוּן בְּרָכָה לְאַחֲרָיו בִּמְקוֹמוֹ, שֶׁכָּל שֶׁלֹּא בֵּרַךְ בְּרָכָה אַחֲרוֹנָה אֵין עֲמִידָתוֹ גְּמַר סְעוּדָה.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן רי״ב (הטקסט האמיתי מובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — אֵיזֶה מַאֲכָל עִקָּר וְאֵיזֶה טָפֵל, אִם צָרִיךְ לְבָרֵךְ בִּפְנֵי עַצְמוֹ, וְאִם צָרִיךְ נְטִילַת יָדַיִם — יש לשאול את רבך, ולחב״ד רב חב״די.
