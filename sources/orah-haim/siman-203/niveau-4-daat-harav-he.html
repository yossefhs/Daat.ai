דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ר״ג · דִּין בִּרְכַּת פֵּרוֹת הָאָרֶץ

דרכו של אדמו״ר הזקן בְּעִנְיַן דִּין בִּרְכַּת פֵּרוֹת הָאָרֶץ : סִימָן לֵידַע פְּרִי הָעֵץ מִפְּרִי הָאֲדָמָה ; תּוּתִים הַגְּדֵלִים בִּסְנֶה ; פֵּרוֹת אִילָנֵי סְרָק וְעִנְּבֵי הַהֲדַס ; אֱגוֹזִים קְטַנִּים שֶׁבַּיַּעַר ; תּוּתִים שֶׁאֵין בָּהֶם אֶלָּא שָׂרָף ; זַנְגְּבִיל רָטֹב וְיָבֵשׁ ; עִקָּר וְטָפֵל בִּדְבַשׁ וּבְצוּקֶ״ר ; תַּבְלִין שֶׁאֵין דַּרְכָּן לְהֵאָכֵל ; צְנוֹן שֶׁסּוֹפוֹ לְהַקְשׁוֹת — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ומכריע בחריפות תלמודית מיוחדת.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ר״ג — דִּין בִּרְכַּת פֵּרוֹת הָאָרֶץ, הנפרש אצל הרב לְז׳ סְעִיפִים : מִן הַסִּימָן הַמַּבְחִין בֵּין פְּרִי הָעֵץ לִפְרִי הָאֲדָמָה — הֶעָנָף הַקַּיָּם לְשָׁנָה הַבָּאָה מוּל עֵצִים הַכָּלִים בַּחֹרֶף וְחוֹזְרִים וְצוֹמְחִים מִשָּׁרָשָׁיו — לְפֵרוֹת שֶׁאֵין שֵׁם פְּרִי עֲלֵיהֶם כְּלָל, פֵּרוֹת אִילָנֵי סְרָק וְעִנְּבֵי הַהֲדַס, לְעֻמַּת אֱגוֹזִים קְטַנִּים שֶׁבַּיַּעַר שֶׁפֵּרוֹת חֲשׁוּבִים הֵם ; מִתּוּתִים שֶׁאֵין בָּהֶם אֶלָּא שָׂרָף כָּנוּס בַּחַרְצַנִּים, שֶׁמּוֹצְצִים וְזוֹרְקִים, לְזַנְגְּבִיל שֶׁהוּא שֹׁרֶשׁ תַּחַת הַקַּרְקַע וְאֵין בִּרְכָתוֹ בִּרְכַּת הָעֵץ ; מֵעִקָּר וְטָפֵל — הַדְּבַשׁ וְהַצּוּקֶ״ר הַמְתַקְּנִים אֶת הַתַּבְלִין וְאֵינָם נוֹטְלִים אֶת בִּרְכָתוֹ — לִגְבוּל הַדָּבָר בְּתַבְלִין שֶׁאֵין דַּרְכָּן אֶלָּא לְתַבֵּל בָּהֶם אֶת הַתַּבְשִׁיל ; וְעַד הַצְּנוֹן, שֶׁאֵין סוֹפוֹ לְהַקְשׁוֹת מוֹצִיאוֹ מִפֵּרוֹת הָאָרֶץ.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ר״ג — ז׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ר״ג). אדמו״ר הזקן דן בו בְּדִין בִּרְכַּת פֵּרוֹת הָאָרֶץ. אין מובא בעמוד זה שום קטע מחוץ לטקסט בר-בדיקה זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ר״ג — דִּין בִּרְכַּת פֵּרוֹת הָאָרֶץ

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ר״ג — דִּין בִּרְכַּת פֵּרוֹת הָאָרֶץ — וּבוֹ ז׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.203). הסימן כולל ז׳ סעיפים (סעיף k → ‎.203.k).

סעיף א עַל פֵּרוֹת הָאָרֶץ "בּוֹרֵא פְּרִי הָאֲדָמָה" — וְסִימָן לֵידַע פְּרִי הָעֵץ.

דִּין בִּרְכַּת פֵּרוֹת הָאָרֶץ, וּבוֹ ז׳ סְעִיפִים:
עַל פֵּרוֹת הָאָרֶץ מְבָרְכִין "בּוֹרֵא פְּרִי הָאֲדָמָה". וְסִימָן לֵידַע אֵיזֶה פְּרִי הָעֵץ וְאֵיזֶה פְּרִי הָאֲדָמָה: כָּל שֶׁאִלּוּ יִנָּטֵל הַפְּרִי וְנִשְׁאַר הֶעָנָף קַיָּם לְשָׁנָה הַבָּאָה וּמוֹצִיא הַפְּרִי — הֲרֵי זֶה פְּרִי הָעֵץ. אֲבָל כָּל שֶׁכָּלוּ עֵצָיו לְגַמְרֵי בַּחֹרֶף וְחוֹזְרִים וְצוֹמְחִים מִשָּׁרָשָׁיו — אֵין פֵּרוֹתָיו חֲשׁוּבִים פְּרִי הָעֵץ, וּמְבָרְכִין עֲלֵיהֶם "בּוֹרֵא פְּרִי הָאֲדָמָה". לְפִיכָךְ, מְבָרְכִין עַל תּוּתִים הַגְּדֵלִים בִּסְנֶה "בּוֹרֵא פְּרִי הָאֲדָמָה":

סעיף ב פֵּרוֹת אִילָנֵי סְרָק "שֶׁהַכֹּל" — וֶאֱגוֹזֵי הַיַּעַר "בּוֹרֵא פְּרִי הָעֵץ".

בַּמֶּה דְּבָרִים אֲמוּרִים? בְּתוּתִים, שֶׁהֵם טוֹבִים לַאֲכִילָה וּפְרִי גָּמוּר הֵן, אֶלָּא שֶׁאֵין הֶעָנָף מִתְקַיֵּם מִשָּׁנָה לַחֲבֶרְתָּהּ, אֲבָל מִינֵי פֵּרוֹת גְּרוּעִים הַגְּדֵלִים בְּאִילָנֵי סְרָק, כְּגוֹן תַּפּוּחִים קְטַנִּים וְאַגָּסִים קְטַנִּים הַגְּדֵלִים בַּיַּעַר, וְכֵן עִנְּבֵי הַהֲדַס, אַף עַל פִּי שֶׁהֵם רְאוּיִים לַאֲכִילָה — אֵינָם חֲשׁוּבִים פְּרִי, וְאֵין מְבָרְכִין עֲלֵיהֶם אֶלָּא "שֶׁהַכֹּל נִהְיָה בִּדְבָרוֹ", בֵּין חַיִּים בֵּין מְבֻשָּׁלִים.
אֲבָל אֱגוֹזִים קְטַנִּים הַגְּדֵלִים בַּיַּעַר — פֵּרוֹת חֲשׁוּבִים הֵם, וּמְבָרְכִין עֲלֵיהֶם "בּוֹרֵא פְּרִי הָעֵץ". וְכֵן פְּרִי שֶׁקּוֹרִין ברומבערי״ן, וּפְרִי אָדֹם שֶׁקּוֹרִין ערפור״ט. וְכֵן אוֹתָם שֶׁקּוֹרִין יאגידע״ס הַשְּׁחוֹרִים. אֲבָל הָאֲדֻמִּים גְּדֵלִים בָּעֵשֶׂב, וּבִרְכָתָם "בּוֹרֵא פְּרִי הָאֲדָמָה":

סעיף ג תּוּתִים שֶׁאֵין בָּהֶם אֶלָּא שָׂרָף — "שֶׁהַכֹּל" כְּעַל מֵי פֵּרוֹת.

אוֹתָן תּוּתִים אֲדֻמִּים הַנִּמְצָאִים בְּתַמּוּז, (וְגַם) יֵשׁ (מִין) הַנִּמְצָאִים בַּחֹרֶף, וְאֵין בָּהֶם אֶלָּא שָׂרָף לְבַד כָּנוּס בְּתוֹךְ הַחַרְצַנִּים, וּמוֹצְצִים אוֹתָן וְזוֹרְקִין אוֹתָן — מְבָרְכִין עֲלֵיהֶם "שֶׁהַכֹּל" כְּעַל שְׁאָר מֵי פֵּרוֹת. וְאַף אִם בּוֹלְעָן עִם הַקְּלִפָּה וְהַגַּרְעִין, מִכָּל מָקוֹם, עֵינֵינוּ רוֹאוֹת שֶׁהַקְּלִפָּה וְהַגַּרְעִין אֵין רְאוּיִים לַאֲכִילָה, אִם כֵּן אֵין לְבָרֵךְ אֶלָּא עַל הַמַּשְׁקֶה שֶׁבְּתוֹכָהּ:

סעיף ד זַנְגְּבִיל רָטֹב — "בּוֹרֵא פְּרִי הָאֲדָמָה", שֶׁהוּא שֹׁרֶשׁ תַּחַת הַקַּרְקַע.

זַנְגְּבִיל בְּעוֹדוֹ רָטֹב שֶׁהוּא רָאוּי לַאֲכִילָה — מְבָרְכִים עָלָיו "בּוֹרֵא פְּרִי הָאֲדָמָה" וְלֹא "בּוֹרֵא פְּרִי הָעֵץ", לְפִי שֶׁאֵינוֹ עִקַּר הַפְּרִי, כִּי הוּא שֹׁרֶשׁ תַּחַת הַקַּרְקַע. וְכֵן עַל שֹׁרֶשׁ שֶׁקּוֹרִין לאקרי״ץ אִם בּוֹלְעוֹ:

סעיף ה זַנְגְּבִיל יָבֵשׁ בְּמִרְקַחַת — הַדְּבַשׁ וְהַצּוּקֶ״ר טָפֵל וְלֹא עִקָּר.

וְכֵן זַנְגְּבִיל יָבֵשׁ שֶׁעָשָׂה מִמֶּנּוּ מִרְקַחַת — מְבָרְכִים עָלָיו "בּוֹרֵא פְּרִי הָאֲדָמָה". וְכֵן פִּלְפְּלִים יְבֵשִׁים, שֶׁאַף שֶׁנַּעֲשׂוּ רְאוּיִים לַאֲכִילָה עַל יְדֵי הַדְּבַשׁ — אֵין הַדְּבַשׁ חָשׁוּב עִקָּר אֶלָּא טָפֵל, שֶׁהֲרֵי הוּא בָּא לְהַכְשִׁירָם וּלְתַקְּנָם. וְכֵן בְּשָׂמִים שְׁחוּקִים וּמְעֹרָבִים עִם צוּקֶ״ר, כְּמוֹ שֶׁרְגִילִים לִכְתֹּשׁ זַנְגְּבִיל יָבֵשׁ וּלְעָרְבוֹ עִם צוּקֶ״ר (שֶׁקּוֹרִין מאגי״ן פולווע״ר), אַף שֶׁאֵינָן רְאוּיִם לַאֲכִילָה כְּלָל בִּפְנֵי עַצְמָן — אֵין הַצּוּקֶ״ר חָשׁוּב עִקָּר אֶלָּא טָפֵל, שֶׁהוּא בָּא לְהַכְשִׁירָם וּלְתַקְּנָם:

סעיף ו תַּבְלִין שֶׁאֵין דַּרְכָּן לְהֵאָכֵל — "שֶׁהַכֹּל", וְזַנְגְּבִיל צָלוּי — "הָאֲדָמָה".

בַּמֶּה דְּבָרִים אֲמוּרִים? בְּמִינֵי בְּשָׂמִים וְתַבְלִין, שֶׁדַּרְכָּן לְהֵאָכֵל עַל יְדֵי טִגּוּן בִּדְבַשׁ, אוֹ עַל יְדֵי תַּעֲרֹבֶת צוּקֶ״ר אוֹ דָּבָר אַחֵר, בְּעִנְיָן שֶׁהֵם חֲשׁוּבִים עִקָּר, כְּגוֹן זַנְגְּבִיל וּפִלְפְּלִין וְצִפֹּרֶן וְכַיּוֹצֵא בָּהֶם. אֲבָל מִינֵי בְּשָׂמִים וְתַבְלִין שֶׁאֵין דַּרְכָּן לְהֵאָכֵל אֶלָּא כְּשֶׁמְּתַבְּלִין בָּהֶם הַתַּבְשִׁיל, וְאָז הַתַּבְשִׁיל הוּא עִקָּר, וְאֵין הָאֲכִילָה נִקְרֵאת עַל שֵׁם הַתַּבְלִין כְּלָל, וּכְאִלּוּ אֵינָן עוֹמְדִין לַאֲכִילָה כְּלָל, כְּגוֹן ציטווא״ר וְכַיּוֹצֵא בּוֹ — אֵין שֵׁם פְּרִי עָלָיו כְּלָל. וְאַף אִם תִּקְּנוֹ וְהִכְשִׁירוֹ לַאֲכִילָה כְּמוֹת שֶׁהוּא, כְּמוֹ שֶׁרְגִילִין שֶׁצּוֹלִין ציטווא״ר וְאוֹכְלִין לִרְפוּאָה — אֵין מְבָרְכִין עָלָיו אֶלָּא "שֶׁהַכֹּל", כֵּיוָן שֶׁאֵין נוֹטְעִין אוֹתוֹ עַל דַּעַת לְאָכְלוֹ כְּלָל, אֶלָּא עַל דַּעַת לִתֵּן בַּתַּבְשִׁיל. וְאֵין צָרִיךְ לוֹמַר אִם טִגְּנוֹ בִּדְבַשׁ אוֹ שֶׁעֵרְבוֹ עִם צוּקֶ״ר. אֲבָל כְּשֶׁצּוֹלִין זַנְגְּבִי״ל לִרְפוּאָה — מְבָרְכִין עָלָיו "בּוֹרֵא פְּרִי הָאֲדָמָה", כֵּיוָן שֶׁנַּעֲשָׂה רָאוּי לַאֲכִילָה עַל יְדֵי זֶה:

סעיף ז צְנוֹן — "בּוֹרֵא פְּרִי הָאֲדָמָה", אַף שֶׁסּוֹפוֹ לְהַקְשׁוֹת.

צְנוֹן מְבָרְכִין עָלָיו "בּוֹרֵא פְּרִי הָאֲדָמָה", אַף עַל פִּי שֶׁהָיָה סוֹפוֹ לְהַקְשׁוֹת:

2 · למעשה

הלכה למעשה — מנהג חב״ד

הנהגה מעשית לחב״ד

נקודות הנהגה הנובעות ישירות מסימן ר״ג ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת העיקרון, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ר״ג (דִּין בִּרְכַּת פֵּרוֹת הָאָרֶץ)

  • ① סעיף א. הַסִּימָן אֵינוֹ בְּגֹדֶל הַצֶּמַח אֶלָּא בְּקִיּוּם הָעֵץ : נִנְטַל הַפְּרִי וְנִשְׁאַר הֶעָנָף קַיָּם לְשָׁנָה הַבָּאָה וּמוֹצִיא פְּרִי — פְּרִי הָעֵץ ; כָּלוּ עֵצָיו לְגַמְרֵי בַּחֹרֶף וְחוֹזְרִים וְצוֹמְחִים מִשָּׁרָשָׁיו — פֵּרוֹתָיו פְּרִי הָאֲדָמָה. וּמִכָּאן תּוּתִים הַגְּדֵלִים בִּסְנֶה : "בּוֹרֵא פְּרִי הָאֲדָמָה".
  • ② סעיף ב. שְׁתֵּי שְׁאֵלוֹת נִפְגָּשׁוֹת כָּאן : אִם פְּרִי עֵץ הוּא, וְאִם שֵׁם פְּרִי עָלָיו כְּלָל. פֵּרוֹת גְּרוּעִים שֶׁל אִילָנֵי סְרָק — תַּפּוּחִים וְאַגָּסִים קְטַנִּים שֶׁבַּיַּעַר, וְעִנְּבֵי הַהֲדַס — אֵין שֵׁם פְּרִי עֲלֵיהֶם, וּבִרְכָתָם "שֶׁהַכֹּל" בֵּין חַיִּים בֵּין מְבֻשָּׁלִים. וְאִלּוּ אֱגוֹזִים קְטַנִּים שֶׁבַּיַּעַר פֵּרוֹת חֲשׁוּבִים הֵם — "בּוֹרֵא פְּרִי הָעֵץ". וּבַתּוּתִים חִלֵּק הָרַב אַף בַּמַּרְאֶה : הַשְּׁחוֹרִים פְּרִי הָעֵץ, וְהָאֲדֻמִּים הַגְּדֵלִים בָּעֵשֶׂב פְּרִי הָאֲדָמָה.
  • ③ סעיף ג. מַה שֶּׁמּוֹצְצִים וְזוֹרְקִים אֵינוֹ אָכוּל : תּוּת שֶׁאֵין בּוֹ אֶלָּא שָׂרָף כָּנוּס בְּתוֹךְ הַחַרְצַנִּים — בִּרְכָתוֹ כְּמֵי פֵּרוֹת, "שֶׁהַכֹּל". וְאַף הַבּוֹלְעָן עִם הַקְּלִפָּה וְהַגַּרְעִין אֵינוֹ מְשַׁנֶּה, שֶׁעֵינֵינוּ רוֹאוֹת שֶׁאֵינָם רְאוּיִים לַאֲכִילָה, וְאֵין הַבְּרָכָה אֶלָּא עַל הַמַּשְׁקֶה שֶׁבְּתוֹכָהּ.
  • ④ סעיף ד. אֵין הַשֹּׁרֶשׁ פְּרִי הָעֵץ : זַנְגְּבִיל רָטֹב הָרָאוּי לַאֲכִילָה — "בּוֹרֵא פְּרִי הָאֲדָמָה" וְלֹא "הָעֵץ", שֶׁהוּא שֹׁרֶשׁ תַּחַת הַקַּרְקַע. וְכֵן שֹׁרֶשׁ לאקרי״ץ — וְהוּא שֶׁבּוֹלְעוֹ.
  • ⑤ סעיף ה. עִקָּר וְטָפֵל : הַדְּבַשׁ וְהַצּוּקֶ״ר הַמַּכְשִׁירִים אֶת הַתַּבְלִין אֵינָם נוֹטְלִים אֶת הַבְּרָכָה לְעַצְמָם, שֶׁאֵינָם בָּאִים אֶלָּא לְתַקְּנוֹ. לְפִיכָךְ זַנְגְּבִיל יָבֵשׁ שֶׁעָשָׂה מִמֶּנּוּ מִרְקַחַת, פִּלְפְּלִים יְבֵשִׁים, וּבְשָׂמִים שְׁחוּקִים וּמְעֹרָבִים בְּצוּקֶ״ר — בִּרְכָתָם בִּרְכַּת הַתַּבְלִין עַצְמוֹ, "בּוֹרֵא פְּרִי הָאֲדָמָה", אַף שֶׁאֵינָם רְאוּיִים כְּלָל בִּפְנֵי עַצְמָן.
  • ⑥ סעיף ו. וְיֵשׁ גְּבוּל לְדָבָר : לֹא נֶאֶמְרוּ דְּבָרִים אֵלּוּ אֶלָּא בְּתַבְלִין שֶׁדַּרְכָּן לְהֵאָכֵל כָּךְ — זַנְגְּבִיל, פִּלְפְּלִין וְצִפֹּרֶן. אֲבָל תַּבְלִין שֶׁאֵין דַּרְכָּן אֶלָּא לְתַבֵּל בָּהֶם אֶת הַתַּבְשִׁיל — אֵין שֵׁם פְּרִי עָלָיו כְּלָל, שֶׁאֵין נוֹטְעִין אוֹתוֹ עַל דַּעַת לְאָכְלוֹ ; וַאֲפִלּוּ צָלוּי וְנֶאֱכָל לִרְפוּאָה — "שֶׁהַכֹּל". וְזַנְגְּבִיל צָלוּי לִרְפוּאָה — חוֹזֵר לְבִרְכַּת "בּוֹרֵא פְּרִי הָאֲדָמָה", שֶׁנַּעֲשָׂה רָאוּי לַאֲכִילָה עַל יְדֵי זֶה.
  • ⑦ סעיף ז. צְנוֹן בִּרְכָתוֹ "בּוֹרֵא פְּרִי הָאֲדָמָה" אַף עַל פִּי שֶׁסּוֹפוֹ לְהַקְשׁוֹת : אֵין הֱיוֹתוֹ עָתִיד לְהִתְקַשּׁוֹת מוֹצִיאוֹ מִבִּרְכַּת פֵּרוֹת הָאָרֶץ.

⚠ חלק זה מציג את הנהגת חב״ד המבוססת על שולחן ערוך הרב, סימן ר״ג (הטקסט האמיתי מובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — אֵיזוֹ בְּרָכָה עַל תּוּת זֶה, עַל זַנְגְּבִיל בְּמִרְקַחַת, וְעַל תַּבְלִין מְעֹרָב בְּצוּקֶ״ר — יש לשאול את רבך, ולחב״ד רב חב״די.
