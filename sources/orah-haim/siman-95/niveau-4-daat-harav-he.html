דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן צ״ה · כִּוּוּן הָאֵיבָרִים בִּשְׁעַת הַתְּפִלָּה

דרכו של אדמו״ר הזקן בכִּוּוּן הָאֵיבָרִים בִּשְׁעַת הַתְּפִלָּה : רַגְלָיו זוֹ אֵצֶל זוֹ — כְּמַלְאָכִים ; ג׳ פְּסִיעוֹת — דֶּרֶךְ קֵרוּב וְהַגָּשָׁה ; עֵינָיו לְמַטָּה לָאָרֶץ וְלִבּוֹ לְמַעְלָה לַשָּׁמַיִם ; וְכְּעֶבֶד לִפְנֵי רַבּוֹ — יָדָיו עַל לִבּוֹ — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן צ״ה — כִּוּוּן הָאֵיבָרִים בִּשְׁעַת הַתְּפִלָּה. אצל הרב הסימן נפרש לד׳ סְעִיפִים : כִּוּוּן הָרַגְלַיִם וְהַהִדָּמּוּת לַמַּלְאָכִים, הַג׳ פְּסִיעוֹת לִפְנֵי הַתְּפִלָּה, הָעֵינַיִם לְמַטָּה וְהַלֵּב לְמַעְלָה, וְהָעֲמִידָה כְּעֶבֶד לִפְנֵי רַבּוֹ עִם כְּפִיפַת הַיָּדַיִם.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן צ״ה — ד׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim צ״ה). אדמו״ר הזקן דן בו בְּכִוּוּן הָאֵיבָרִים בִּשְׁעַת הַתְּפִלָּה. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן צ״ה — כִּוּוּן אֵיבָרָיו בִּשְׁעַת הַתְּפִלָּה

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן צ״ה — כִּוּוּן אֵיבָרָיו בִּשְׁעַת הַתְּפִלָּה — וּבוֹ ד׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.95), ואחריו תרגומנו. הסימן כולל ד׳ סעיפים (סעיף k → ‎.95.k).

סעיף א כִּוּוּן הָרַגְלַיִם — זוֹ אֵצֶל זוֹ כְּמַלְאָכִים ; וְגַם בִּקְדֻשָּׁה.

כִּוּוּן אֵיבָרָיו בִּשְׁעַת הַתְּפִלָּה וּבוֹ ד׳ סְעִיפִים:הַמִּתְפַּלֵּל צָרִיךְ שֶׁיְכַוֵּן רַגְלָיו זוֹ אֵצֶל זוֹ בְּכִוּוּן כְּאִלּוּ אֵינָן אֶלָּא רֶגֶל אֶחָד, לְהִדָּמוֹת לַמַּלְאָכִים שֶׁנֶּאֱמַר בָּהֶם ״וְרַגְלֵיהֶם רֶגֶל יְשָׁרָה״, כְּלוֹמַר רַגְלֵיהֶם נִרְאוֹת כְּרֶגֶל אֶחָד. וְטוֹב לְכַוֵּן רַגְלָיו גַּם כְּשֶׁאוֹמֵר קְדֻשָּׁה עִם שְׁלִיחַ צִבּוּר, שֶׁהֲרֵי אוֹמֵר ״נְקַדֵּשׁ … כְּשֵׁם שֶׁמַּקְדִּישִׁים … בִּשְׁמֵי מָרוֹם״, וּבַמָּרוֹם מַקְדִּישִׁים כֵּן בְּרֶגֶל יְשָׁרָה:

כִּוּוּן הָרַגְלַיִם. הַמִּתְפַּלֵּל צָרִיךְ לְכַוֵּן רַגְלָיו זוֹ אֵצֶל זוֹ כְּאִלּוּ אֵינָן אֶלָּא רֶגֶל אֶחָד, לְהִדָּמוֹת לַמַּלְאָכִים שֶׁנֶּאֱמַר בָּהֶם ״וְרַגְלֵיהֶם רֶגֶל יְשָׁרָה״ — רַגְלֵיהֶם נִרְאוֹת כְּרֶגֶל אֶחָד. וְטוֹב לְכַוֵּן רַגְלָיו גַּם בִּקְדֻשָּׁה עִם הַשַּׁ״ץ, שֶׁהֲרֵי אוֹמְרִים ״כְּשֵׁם שֶׁמַּקְדִּישִׁים … בִּשְׁמֵי מָרוֹם״, וּבַמָּרוֹם מַקְדִּישִׁים בְּרֶגֶל יְשָׁרָה.

סעיף ב ג׳ פְּסִיעוֹת לְפָנָיו — דֶּרֶךְ קֵרוּב וְהַגָּשָׁה.

יֵשׁ מִי שֶׁאוֹמֵר שֶׁכְּשֶׁעוֹמֵד לְהִתְפַּלֵּל יֵלֵךְ לְפָנָיו ג׳ פְּסִיעוֹת, דֶּרֶךְ קֵרוּב וְהַגָּשָׁה לְדָבָר שֶׁצָּרִיךְ לַעֲשׂוֹת. וְטוֹב לָחֹש לִדְבָרָיו:

ג׳ פְּסִיעוֹת. יֵשׁ מִי שֶׁאוֹמֵר : כְּשֶׁעוֹמֵד לְהִתְפַּלֵּל יֵלֵךְ לְפָנָיו ג׳ פְּסִיעוֹת, דֶּרֶךְ קֵרוּב וְהַגָּשָׁה לְדָבָר שֶׁצָּרִיךְ לַעֲשׂוֹת. וְטוֹב לָחֹש לִדְבָרָיו.

סעיף ג עֵינָיו לְמַטָּה לָאָרֶץ נֶגֶד הַמִּקְדָּשׁ, וְלִבּוֹ לְמַעְלָה לַשָּׁמַיִם.

כָּתוּב אֶחָד אוֹמֵר: ״וְהָיוּ עֵינַי וְלִבִּי שָׁם כָּל הַיָּמִים״, וְכָתוּב אֶחָד אוֹמֵר: ״נִשָּׂא לְבָבֵנוּ אֶל כַּפָּיִם אֶל אֵל בַּשָּׁמָיִם״, וּכְדֵי שֶׁיִּתְקַיְּמוּ ב׳ מִקְרָאוֹת אֵלּוּ אָמְרוּ שֶׁהַמִּתְפַּלֵּל צָרִיךְ שֶׁיָּכֹף רֹאשׁוֹ מְעַט שֶׁיִּהְיוּ עֵינָיו לְמַטָּה לָאָרֶץ, שֶׁהֲרֵי אָנוּ מִתְפַּלְלִים נֶגֶד בֵּית הַמִּקְדָּשׁ שֶׁנֶּאֱמַר בּוֹ ״וְהָיוּ עֵינַי וְלִבִּי שָׁם כו׳״, לָכֵן יִתֵּן עֵינָיו לְמַטָּה כְּנֶגְדּוֹ וְיַחֲשֹׁב כְּאִלּוּ עוֹמֵד בּוֹ וּמִתְפַּלֵּל, וְעַל יְדֵי זֶה תְּפִלָּתוֹ רְצוּיָה, וּבְלִבּוֹ יְכַוֵּן לְמַעְלָה ״אֶל אֵל שֶׁבַּשָּׁמַיִם״ (וְאוֹתָן הַמַּגְבִּיהִים רָאשֵׁיהֶם וְעֵינֵיהֶם לְמַעְלָה כְּמַבִּיטִים עַל הַגַּג — הַמַּלְאָכִים מַלְעִיגִים עֲלֵיהֶם). וְקֹדֶם הַתְּפִלָּה יִשָּׂא עֵינָיו לַשָּׁמַיִם, לְעוֹרֵר הַכַּוָּנָה. וְכֵן בְּתוֹךְ הַתְּפִלָּה אִם נִתְבַּטְּלָה כַּוָּנָתוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן צ׳:

עֵינַיִם וְלֵב. כָּתוּב אֶחָד אוֹמֵר ״וְהָיוּ עֵינַי וְלִבִּי שָׁם כָּל הַיָּמִים״ וְכָתוּב אֶחָד ״נִשָּׂא לְבָבֵנוּ אֶל כַּפָּיִם אֶל אֵל בַּשָּׁמָיִם״ ; וּכְדֵי שֶׁיִּתְקַיְּמוּ ב׳ הַמִּקְרָאוֹת — יָכֹף רֹאשׁוֹ מְעַט, עֵינָיו לְמַטָּה לָאָרֶץ נֶגֶד הַמִּקְדָּשׁ, וּבְלִבּוֹ יְכַוֵּן לְמַעְלָה אֶל אֵל שֶׁבַּשָּׁמַיִם. וְהַמַּגְבִּיהִים רָאשֵׁיהֶם כְּמַבִּיטִים עַל הַגַּג — הַמַּלְאָכִים מַלְעִיגִים עֲלֵיהֶם. וְקֹדֶם הַתְּפִלָּה יִשָּׂא עֵינָיו לַשָּׁמַיִם לְעוֹרֵר הַכַּוָּנָה (וְכֵן בְּתוֹךְ הַתְּפִלָּה, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן צ׳).

סעיף ד כְּעֶבֶד לִפְנֵי רַבּוֹ ; יָדָיו עַל לִבּוֹ, לֹא עַל חֲלָצָיו.

צָרִיךְ לַעֲמֹד בַּתְּפִלָּה כְּעֶבֶד לִפְנֵי רַבּוֹ בְּאֵימָה בְּיִרְאָה בְּפַחַד. וְאַל יַנִּיחַ יָדָיו עַל חֲלָצָיו, מִפְּנֵי שֶׁהוּא דֶּרֶךְ יוּהֲרָא. וְטוֹב שֶׁיַּנִּיחֵן עַל לִבּוֹ כְּפוּתִין הַיְמָנִית עַל הַשְּׂמָאלִית, בְּמָקוֹם שֶׁנּוֹהֲגִין לַעֲמֹד כֵּן לִפְנֵי הַמֶּלֶךְ כְּשֶׁמְּדַבְּרִים עִמּוֹ וְשׁוֹאֲלִין מֵאִתּוֹ צָרְכֵיהֶם. וְהַכֹּל כְּמִנְהַג הַמָּקוֹם. וְנוֹהֲגִים קְצָת מִטַּעַם הַיָּדוּעַ לָהֶם לָכֹף הָאֲגוּדָל בְּתוֹךְ פִּסַּת הַיָּד. וְיֵשׁ נוֹהֲגִין מִטַּעַם הַיָּדוּעַ לָהֶם לְהַנִּיחַ זְרוֹעוֹתֵיהֶם עַל לִבָּם יְמָנִית עַל שְׂמָאלִית שֶׁיִּהְיֶה פֶּרֶק אֶמְצָעִי מִשְׁתֵּי זְרוֹעוֹת נֶגֶד פְּנֵי הַמִּתְפַּלֵּל:

כְּעֶבֶד לִפְנֵי רַבּוֹ. צָרִיךְ לַעֲמֹד בַּתְּפִלָּה כְּעֶבֶד לִפְנֵי רַבּוֹ בְּאֵימָה בְּיִרְאָה בְּפַחַד. וְאַל יַנִּיחַ יָדָיו עַל חֲלָצָיו — דֶּרֶךְ יוּהֲרָא. וְטוֹב שֶׁיַּנִּיחֵן עַל לִבּוֹ כְּפוּתִין הַיְמָנִית עַל הַשְּׂמָאלִית, כְּדֶרֶךְ שֶׁעוֹמְדִים לִפְנֵי הַמֶּלֶךְ. וְהַכֹּל כְּמִנְהַג הַמָּקוֹם ; יֵשׁ הַכּוֹפְפִים הָאֲגוּדָל בְּתוֹךְ פִּסַּת הַיָּד, וְיֵשׁ הַמַּנִּיחִים זְרוֹעוֹתֵיהֶם יְמָנִית עַל שְׂמָאלִית שֶׁיִּהְיֶה פֶּרֶק אֶמְצָעִי נֶגֶד פְּנֵי הַמִּתְפַּלֵּל.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim צ״ה (מהדורת קה״ת, ד׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — כִּוּוּן הָאֵיבָרִים בִּשְׁעַת הַתְּפִלָּה

היסוד : כִּוּוּן הָאֵיבָרִים בַּתְּפִלָּה

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה כִּוּוּן הָרַגְלַיִם וְהַהִדָּמּוּת לַמַּלְאָכִים. אחר כך הַג׳ פְּסִיעוֹת דֶּרֶךְ קֵרוּב. אחר כך הָעֵינַיִם לְמַטָּה וְהַלֵּב לְמַעְלָה. ולבסוף הָעֲמִידָה כְּעֶבֶד לִפְנֵי רַבּוֹ וּכְפִיפַת הַיָּדַיִם.

כִּוּוּן הָרַגְלַיִם — כְּמַלְאָכִים

כִּוּוּן הָרַגְלַיִם — כְּמַלְאָכִים

היסוד. לְכַוֵּן רַגְלָיו זוֹ אֵצֶל זוֹ כְּאִלּוּ אֵינָן אֶלָּא רֶגֶל אֶחָד, לְהִדָּמוֹת לַמַּלְאָכִים (״וְרַגְלֵיהֶם רֶגֶל יְשָׁרָה״) ; וְטוֹב גַּם בִּקְדֻשָּׁה עִם הַשַּׁ״ץ (סעיף א).

המסקנה. גּוּף הַמִּתְפַּלֵּל נַעֲשֶׂה כְּמַלְאָךְ עוֹמֵד — עֲמִידָה יַצִּיבָה וּמְאֻחֶדֶת. עיין סימן 94 (נֹכַח הַמִּקְדָּשׁ).

ג׳ פְּסִיעוֹת — דֶּרֶךְ קֵרוּב

ג׳ פְּסִיעוֹת — דֶּרֶךְ קֵרוּב

היסוד. יֵשׁ מִי שֶׁאוֹמֵר : קֹדֶם הַתְּפִלָּה יֵלֵךְ לְפָנָיו ג׳ פְּסִיעוֹת דֶּרֶךְ קֵרוּב וְהַגָּשָׁה ; וְטוֹב לָחֹש לִדְבָרָיו (סעיף ב).

המסקנה. נִגָּשִׁים לַתְּפִלָּה כְּקֵרוּב אֶל הַמֶּלֶךְ, לֹא כְּהַתְחָלָה סְתָם. עיין סימן 90 (מְקוֹם הַתְּפִלָּה).

עֵינַיִם לְמַטָּה וְהַלֵּב לְמַעְלָה

עֵינַיִם לְמַטָּה וְהַלֵּב לְמַעְלָה

היסוד. יָכֹף רֹאשׁוֹ מְעַט, עֵינָיו לְמַטָּה לָאָרֶץ נֶגֶד הַמִּקְדָּשׁ, וּבְלִבּוֹ לְמַעְלָה אֶל אֵל שֶׁבַּשָּׁמַיִם ; וְכָךְ יִתְקַיְּמוּ ב׳ הַמִּקְרָאוֹת (סעיף ג).

המסקנה. לֹא רֹאשׁ מֻגְבָּהּ כְּלַפֵּי הַגַּג וְלֹא עֵינַיִם מְשׁוֹטְטוֹת : עַנְוַת הַגּוּף עִם עֲלִיַּת הַלֵּב. עיין סימן 98 (כַּוָּנַת הַתְּפִלָּה).

כְּעֶבֶד לִפְנֵי רַבּוֹ — הַיָּדַיִם

כְּעֶבֶד לִפְנֵי רַבּוֹ — הַיָּדַיִם

היסוד. עֲמִידָה כְּעֶבֶד לִפְנֵי רַבּוֹ בְּאֵימָה ; יָדָיו עַל לִבּוֹ כְּפוּתִין יְמָנִית עַל שְׂמָאלִית, וְלֹא עַל חֲלָצָיו (דֶּרֶךְ יוּהֲרָא) ; וְהַכֹּל כְּמִנְהַג הַמָּקוֹם (סעיף ד).

המסקנה. הָעֲמִידָה כֻּלָּהּ מְבַטֵּאת הַכְנָעָה לִפְנֵי הַמֶּלֶךְ. עיין סימן 97 (הַנְהָגַת הַגּוּף בַּתְּפִלָּה).

הפניה ללימוד. היקף הסימן — כִּוּוּן הָרַגְלַיִם, הָעֵינַיִם וָלֵב וְהָעֲמִידָה כְּעֶבֶד — נוגע להנהגת העמידה היומיומית. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

כִּוּוּן הָאֵיבָרִים בִּשְׁעַת הַתְּפִלָּה — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן צ״ה. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מד׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
כִּוּוּן הָרַגְלַיִם או״ח צ״ה — יְכַוֵּן רַגְלָיו זוֹ אֵצֶל זוֹ כְּמַלְאָכִים ; וְטוֹב גַּם בִּקְדֻשָּׁה (סעיף א, ד). מ״ב צ״ה — גֶּדֶר צֵרוּף הָרַגְלַיִם וְהַהִדָּמּוּת לַמַּלְאָכִים. ערוה״ש צ״ה — עִנְיַן ״רֶגֶל יְשָׁרָה״ וְאֹפֶן הָעֲמִידָה. דרך אדמו״ר הזקן — יְכַוֵּן רַגְלָיו זוֹ אֵצֶל זוֹ כְּאִלּוּ אֵינָן אֶלָּא רֶגֶל אֶחָד, לְהִדָּמוֹת לַמַּלְאָכִים ; וְטוֹב גַּם בִּקְדֻשָּׁה עִם הַשַּׁ״ץ (סעיף א).
ג׳ פְּסִיעוֹת לִפְנֵי הַתְּפִלָּה או״ח צ״ה — יֵשׁ אוֹמְרִים יֵלֵךְ ג׳ פְּסִיעוֹת דֶּרֶךְ קֵרוּב וְהַגָּשָׁה (רוֹקֵחַ). מ״ב צ״ה — פְּרָטֵי הַהַגָּשָׁה וְהַמִּנְהָג בָּהּ. ערוה״ש צ״ה — גֶּדֶר הַקְּרִיבָה לַתְּפִלָּה. דרך אדמו״ר הזקן — יֵשׁ מִי שֶׁאוֹמֵר יֵלֵךְ ג׳ פְּסִיעוֹת דֶּרֶךְ קֵרוּב וְהַגָּשָׁה ; וְטוֹב לָחֹש לִדְבָרָיו (סעיף ב).
עֵינַיִם וָלֵב או״ח צ״ה — יָכֹף רֹאשׁוֹ מְעַט עֵינָיו לָאָרֶץ, וּבְלִבּוֹ לְמַעְלָה לַשָּׁמַיִם (סעיף ב). מ״ב צ״ה — יִשּׁוּב הַב׳ מִקְרָאוֹת וְטַעַם הַכְּפִיפָה. ערוה״ש צ״ה — כִּוּוּן הָעֵינַיִם וְהַלֵּב בַּתְּפִלָּה. דרך אדמו״ר הזקן — יָכֹף רֹאשׁוֹ מְעַט עֵינָיו לְמַטָּה נֶגֶד הַמִּקְדָּשׁ, וּבְלִבּוֹ לְמַעְלָה ; וְקֹדֶם הַתְּפִלָּה יִשָּׂא עֵינָיו לַשָּׁמַיִם לְעוֹרֵר הַכַּוָּנָה (סעיף ג).
עֲמִידָה כְּעֶבֶד וְהַיָּדַיִם או״ח צ״ה — יָדָיו עַל לִבּוֹ כְּפוּתִין יְמָנִית עַל שְׂמָאלִית ; לֹא עַל חֲלָצָיו — יוּהֲרָא (סעיף ג). מ״ב צ״ה — פְּרָטֵי הַנָּחַת הַיָּדַיִם וְהַמִּנְהָגִים. ערוה״ש צ״ה — אֵימָה וְיִרְאָה וְהַנְהָגַת הַיָּדַיִם. דרך אדמו״ר הזקן — כְּעֶבֶד לִפְנֵי רַבּוֹ בְּאֵימָה ; יָדָיו עַל לִבּוֹ כְּפוּתִין יְמָנִית עַל שְׂמָאלִית ; לֹא עַל חֲלָצָיו ; וְהַכֹּל כְּמִנְהַג הַמָּקוֹם (סעיף ד).

הטבלה נערכה מטקסט המחבר (אורח חיים צ״ה, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות תפילה ה, רוקח, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות י׳, י״ג, י״ד). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ד׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — כִּוּוּן הָרַגְלַיִם וְהִדָּמּוּת לַמַּלְאָכִים

קו א — כִּוּוּן הָרַגְלַיִם וְהִדָּמּוּת לַמַּלְאָכִים

יסוד הסימן : רַגְלָיו זוֹ אֵצֶל זוֹ כְּרֶגֶל אֶחָד, לְהִדָּמוֹת לַמַּלְאָכִים ; וְגַם בִּקְדֻשָּׁה (סעיף א).

קריאת הרב : גּוּף הַמִּתְפַּלֵּל נַעֲשֶׂה כְּמַלְאָךְ עוֹמֵד — עֲמִידָה יַצִּיבָה וּמְאֻחֶדֶת. עיין סימן 94.

מסקנה למעשה : לְצָרֵף הָרַגְלַיִם בִּתְחִלַּת הָעֲמִידָה, וְגַם כְּשֶׁעוֹנֶה קְדֻשָּׁה.

קו ב — עֵינַיִם לְמַטָּה וְהַלֵּב לְמַעְלָה

קו ב — עֵינַיִם לְמַטָּה וְהַלֵּב לְמַעְלָה

יסוד הסימן : יָכֹף רֹאשׁוֹ מְעַט, עֵינָיו לָאָרֶץ נֶגֶד הַמִּקְדָּשׁ וּבְלִבּוֹ לְמַעְלָה (סעיף ג).

קריאת הרב : עַנְוַת הַמַּבָּט מִתְאַחֶדֶת עִם עֲלִיַּת הַלֵּב ; וְקֹדֶם הַתְּפִלָּה יִשָּׂא עֵינָיו לַשָּׁמַיִם לְעוֹרֵר הַכַּוָּנָה. עיין סימן 98.

מסקנה למעשה : לֹא לְהַגְבִּיהַּ הָרֹאשׁ כְּלַפֵּי הַגַּג ; לְהַשְׁפִּיל מְעַט הָעֵינַיִם וּלְכַוֵּן הַלֵּב לַשָּׁמַיִם.

קו ג — עֲמִידָה כְּעֶבֶד וּכְפִיפַת הַיָּדַיִם

קו ג — עֲמִידָה כְּעֶבֶד וּכְפִיפַת הַיָּדַיִם

יסוד הסימן : כְּעֶבֶד לִפְנֵי רַבּוֹ בְּאֵימָה ; יָדָיו עַל לִבּוֹ כְּפוּתִין, לֹא עַל חֲלָצָיו (סעיף ד).

קריאת הרב : וְהַכֹּל כְּמִנְהַג הַמָּקוֹם — הָרַב מֵבִיא הַמִּנְהָגִים (אֲגוּדָל, זְרוֹעוֹת) וְקוֹבֵעַ אֶת יְסוֹד הַהַכְנָעָה. עיין סימן 97.

מסקנה למעשה : לְהַנִּיחַ הַיָּדַיִם עַל הַלֵּב יְמָנִית עַל שְׂמָאלִית ; לְעוֹלָם לֹא עַל הַחֲלָצַיִם (יוּהֲרָא).

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן צ״ה ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן צ״ה (כִּוּוּן הָאֵיבָרִים בִּשְׁעַת הַתְּפִלָּה)

  • ① רַגְלַיִם. לְצָרֵף הָרַגְלַיִם זוֹ אֵצֶל זוֹ כְּרֶגֶל אֶחָד, כְּמַלְאָכִים ; וְגַם כְּשֶׁאוֹמֵר קְדֻשָּׁה עִם הַשַּׁ״ץ. יסוד : שוע״ר צ״ה סעיף א.
  • ② ג׳ פְּסִיעוֹת. וְטוֹב לָחֹש לְהַלֵּךְ ג׳ פְּסִיעוֹת קֹדֶם הָעֲמִידָה, דֶּרֶךְ קֵרוּב וְהַגָּשָׁה. יסוד : שוע״ר צ״ה סעיף ב.
  • ③ עֵינַיִם. יָכֹף רֹאשׁוֹ מְעַט, עֵינָיו לָאָרֶץ נֶגֶד הַמִּקְדָּשׁ ; לֹא לְהַגְבִּיהַּ הָרֹאשׁ כְּלַפֵּי הַגַּג. יסוד : שוע״ר צ״ה סעיף ג.
  • ④ לֵב וְכַוָּנָה. לְכַוֵּן הַלֵּב לְמַעְלָה, אֶל אֵל שֶׁבַּשָּׁמַיִם ; וְקֹדֶם הַתְּפִלָּה יִשָּׂא עֵינָיו לַשָּׁמַיִם לְעוֹרֵר הַכַּוָּנָה. יסוד : שוע״ר צ״ה סעיף ג.
  • ⑤ יָדַיִם. לַעֲמֹד כְּעֶבֶד לִפְנֵי רַבּוֹ ; יָדָיו עַל לִבּוֹ כְּפוּתִין, יְמָנִית עַל שְׂמָאלִית ; לְעוֹלָם לֹא עַל הַחֲלָצַיִם (יוּהֲרָא). יסוד : שוע״ר צ״ה סעיף ד.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן צ״ה (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — כִּוּוּן הָרַגְלַיִם, הָעֵינַיִם וָלֵב, הָעֲמִידָה כְּעֶבֶד — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
