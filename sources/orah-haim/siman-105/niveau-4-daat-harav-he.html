דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן ק״ה · דִּין הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת

דרכו של אדמו״ר הזקן בדִּין הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת : הַמִּתְפַּלֵּל ב׳ תְּפִלּוֹת זוֹ אַחַר זוֹ — כְּגוֹן שַׁחֲרִית וּמוּסָף, אוֹ שֶׁשָּׁכַח וְלֹא הִתְפַּלֵּל וְצָרִיךְ לְהַשְׁלִימָהּ (תַּשְׁלוּמִין) — צָרִיךְ לְהַמְתִּין בֵּין זוֹ לָזוֹ, כְּדֵי שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת לַעֲרֹךְ דְּבָרָיו בִּלְשׁוֹן תְּחִנָּה, דְּהַיְנוּ כְּדֵי הִלּוּךְ ד׳ אַמּוֹת ; וְלַחֲזֹר לִמְקוֹמוֹ אֲפִלּוּ בְּלֹא תְּפִלָּה שֵׁנִית — כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ג — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן ק״ה — דִּין הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת. אצל הרב הסימן הוא סְעִיף אֶחָד : הַמִּתְפַּלֵּל ב׳ תְּפִלּוֹת זוֹ אַחַר זוֹ — שַׁחֲרִית וּמוּסָף אוֹ תַּשְׁלוּמִין — צָרִיךְ לְהַמְתִּין בֵּין זוֹ לָזוֹ, כְּדֵי שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת לַעֲרֹךְ דְּבָרָיו בִּלְשׁוֹן תְּחִנָּה, שִׁעוּרוֹ כְּדֵי הִלּוּךְ ד׳ אַמּוֹת ; וְהַחֲזָרָה לִמְקוֹמוֹ — אֲפִלּוּ בְּלֹא תְּפִלָּה שֵׁנִית — מְחַיֶּבֶת אַף הִיא הַמְתָּנָה, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ג.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן ק״ה — סְעִיף אֶחָד מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim ק״ה). אדמו״ר הזקן דן בו בְּדִין הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת זוֹ אַחַר זוֹ. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן ק״ה — דִּין הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן ק״ה — דִּין הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת — וּבוֹ סְעִיף אֶחָד
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.105), ואחריו תרגומנו. הסימן כולל סעיף אחד (סְעִיף אֶחָד → ‎.105.1).

סעיף א הַמִּתְפַּלֵּל ב׳ תְּפִלּוֹת זוֹ אַחַר זוֹ — צָרִיךְ לְהַמְתִּין בֵּינֵיהֶן כְּדֵי הִלּוּךְ ד׳ אַמּוֹת.

דִּין הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת וּבוֹ סְעִיף אֶחָד:הַמִּתְפַּלֵּל ב׳ תְּפִלּוֹת זוֹ אַחַר זוֹ, כְּגוֹן שַׁחֲרִית וּמוּסָף, אוֹ שֶׁשָּׁכַח וְלֹא הִתְפַּלֵּל, וְצָרִיךְ לְהַשְׁלִימָהּ בִּזְמַן תְּפִלָּה שֶׁלְּאַחֲרֶיהָ וּמִתְפַּלֵּל שְׁתַּיִם זוֹ אַחַר זוֹ — צָרִיךְ לְהַמְתִּין בֵּין זוֹ לָזוֹ, כְּדֵי שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת לַעֲרֹךְ דְּבָרָיו בִּלְשׁוֹן תְּחִנָּה, דְּהַיְנוּ כְּדֵי הִלּוּךְ ד׳ אַמּוֹת, אֲפִלּוּ אֵינוֹ רוֹצֶה לַחֲזֹר לִמְקוֹמוֹ אֶלָּא מִתְפַּלֵּל שֵׁנִית בְּמָקוֹם שֶׁעוֹמֵד שָׁם כְּשֶׁפָּסַע אַחַר תְּפִלָּה הָרִאשׁוֹנָה, אֲבָל לַחֲזֹר לִמְקוֹמוֹ אֲפִלּוּ אֵינוֹ רוֹצֶה לְהִתְפַּלֵּל שֵׁנִית — צָרִיךְ לְהַמְתִּין כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ג:

הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת. הַמִּתְפַּלֵּל שְׁתֵּי עֲמִידוֹת זוֹ אַחַר זוֹ — כְּגוֹן שַׁחֲרִית וּמוּסָף, אוֹ שֶׁשָּׁכַח וְלֹא הִתְפַּלֵּל וְצָרִיךְ לְהַשְׁלִימָהּ בִּזְמַן תְּפִלָּה שֶׁלְּאַחֲרֶיהָ, וּמִתְפַּלֵּל שְׁתַּיִם בָּזוֹ אַחַר זוֹ — צָרִיךְ לְהַמְתִּין בֵּין זוֹ לָזוֹ, כְּדֵי שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת לַעֲרֹךְ דְּבָרָיו בִּלְשׁוֹן תְּחִנָּה, דְּהַיְנוּ כְּדֵי הִלּוּךְ ד׳ אַמּוֹת. אֲפִלּוּ אֵינוֹ רוֹצֶה לַחֲזֹר לִמְקוֹמוֹ אֶלָּא מִתְפַּלֵּל שֵׁנִית בַּמָּקוֹם שֶׁעוֹמֵד שָׁם כְּשֶׁפָּסַע אַחַר הַתְּפִלָּה הָרִאשׁוֹנָה. אֲבָל לַחֲזֹר לִמְקוֹמוֹ — אֲפִלּוּ אֵינוֹ רוֹצֶה לְהִתְפַּלֵּל שֵׁנִית — צָרִיךְ לְהַמְתִּין, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ג.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim ק״ה (מהדורת קה״ת, סְעִיף אֶחָד). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּין הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת

היסוד : הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת זוֹ אַחַר זוֹ

סעיפו היחיד של אדמו״ר הזקן נפרש סביב ארבע נקודות. תחילה חוֹבַת הַהַמְתָּנָה — שני המקרים שבהם מתפלל שתי עמידות זו אחר זו. אחר כך הַטַּעַם — שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת לַעֲרֹךְ דְּבָרָיו בִּלְשׁוֹן תְּחִנָּה. אחר כך הַשִּׁעוּר — כְּדֵי הִלּוּךְ ד׳ אַמּוֹת. ולבסוף הַחוֹזֵר לִמְקוֹמוֹ וְהַהַפְנָיָה לְסִימָן קכ״ג.

חוֹבַת הַהַמְתָּנָה — שַׁחֲרִית וּמוּסָף אוֹ תַּשְׁלוּמִין

חוֹבַת הַהַמְתָּנָה — שְׁנֵי הַמִּקְרִים

היסוד. הַמִּתְפַּלֵּל ב׳ תְּפִלּוֹת זוֹ אַחַר זוֹ — כְּגוֹן שַׁחֲרִית וּמוּסָף, אוֹ שֶׁשָּׁכַח וְלֹא הִתְפַּלֵּל וְצָרִיךְ לְהַשְׁלִימָהּ — צָרִיךְ לְהַמְתִּין בֵּין זוֹ לָזוֹ (סעיף א).

המסקנה. הַהֲלָכָה חָלָה בֵּין עַל תְּפִלּוֹת רְצוּפוֹת רְגִילוֹת (מוּסָף) וּבֵין עַל הַתַּשְׁלוּמִין שֶׁל תְּפִלָּה שֶׁשָּׁכַח. עיין רמה 1.

הַטַּעַם — דַּעַת מְיֻשֶּׁבֶת בִּלְשׁוֹן תְּחִנָּה

הַטַּעַם — דַּעַת מְיֻשֶּׁבֶת בִּלְשׁוֹן תְּחִנָּה

היסוד. תַּכְלִית הַהַמְתָּנָה : כְּדֵי שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת לַעֲרֹךְ דְּבָרָיו בִּלְשׁוֹן תְּחִנָּה (סעיף א).

המסקנה. הַתְּפִלָּה הִיא תְּחִנָּה — הִיא תּוֹבַעַת דַּעַת מְיֻשֶּׁבֶת, שֶׁאֵינָהּ כְּבוּלָה בַּתְּפִלָּה שֶׁקָּדְמָה. עיין רמה 3.

הַשִּׁעוּר — כְּדֵי הִלּוּךְ ד׳ אַמּוֹת

הַשִּׁעוּר — כְּדֵי הִלּוּךְ ד׳ אַמּוֹת

היסוד. דְּהַיְנוּ כְּדֵי הִלּוּךְ ד׳ אַמּוֹת — שִׁעוּר הַהַמְתָּנָה כִּזְמַן הִלּוּךְ אַרְבַּע אַמּוֹת (סעיף א).

המסקנה. שִׁעוּר זֶה מַסְפִּיק אֲפִלּוּ אֵינוֹ רוֹצֶה לַחֲזֹר לִמְקוֹמוֹ אֶלָּא מִתְפַּלֵּל בַּמָּקוֹם שֶׁעוֹמֵד שָׁם אַחַר הַפְּסִיעָה. עיין רמה 2.

הַחוֹזֵר לִמְקוֹמוֹ — סִימָן קכ״ג

הַחוֹזֵר לִמְקוֹמוֹ — סִימָן קכ״ג

היסוד. אֲבָל לַחֲזֹר לִמְקוֹמוֹ אֲפִלּוּ אֵינוֹ רוֹצֶה לְהִתְפַּלֵּל שֵׁנִית — צָרִיךְ לְהַמְתִּין כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ג (סעיף א).

המסקנה. הַחֲזָרָה לִמְקוֹמוֹ אַחַר הַפְּסִיעָה מְחַיֶּבֶת הַמְתָּנָה, אֲפִלּוּ בְּלֹא תְּפִלָּה שֵׁנִית — וְהָרַב מַפְנֶה לִפְרָטִים לְסִימָן קכ״ג. עיין רמה 1.

הפניה ללימוד. היקף הסימן — חוֹבַת הַהַמְתָּנָה, שִׁעוּרָהּ ד׳ אַמּוֹת וְהַחֲזָרָה לִמְקוֹמוֹ — נוגע לרצף שתי עמידות זו אחר זו. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן ק״ה. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מסעיפו היחיד.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
חוֹבַת הַהַמְתָּנָה או״ח ק״ה — הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת זוֹ אַחַר זוֹ צָרִיךְ לְהַמְתִּין בֵּין זוֹ לָזוֹ. מ״ב ק״ה — טַעַם הַהַמְתָּנָה, שֶׁלֹּא יְהֵא כְּמָשׂוֹי וּכְתוֹרַח עָלָיו. ערוה״ש ק״ה — גֶּדֶר חוֹבַת הַהַמְתָּנָה בֵּין הַתְּפִלּוֹת. דרך אדמו״ר הזקן — הַמִּתְפַּלֵּל ב׳ תְּפִלּוֹת זוֹ אַחַר זוֹ (שַׁחֲרִית וּמוּסָף אוֹ תַּשְׁלוּמִין) צָרִיךְ לְהַמְתִּין בֵּין זוֹ לָזוֹ (סעיף א).
שִׁעוּר וְטַעַם או״ח ק״ה — כְּדֵי הִלּוּךְ ד׳ אַמּוֹת, כְּדֵי שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת לְהִתְפַּלֵּל בִּלְשׁוֹן תְּחִנָּה. מ״ב ק״ה — שִׁעוּר ד׳ אַמּוֹת וְעִנְיַן יִשּׁוּב הַדַּעַת בַּתְּפִלָּה. ערוה״ש ק״ה — בֵּאוּר הַשִּׁעוּר וְהַכַּוָּנָה שֶׁבַּתְּחִנָּה. דרך אדמו״ר הזקן — כְּדֵי שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת לַעֲרֹךְ דְּבָרָיו בִּלְשׁוֹן תְּחִנָּה, דְּהַיְנוּ כְּדֵי הִלּוּךְ ד׳ אַמּוֹת (סעיף א).
הַחוֹזֵר לִמְקוֹמוֹ או״ח ק״ה — עִקַּר הַדִּין בַּהַמְתָּנָה בֵּין הַתְּפִלּוֹת (בְּלֹא פֵּרוּט הַחֲזָרָה לַמָּקוֹם). מ״ב ק״ה — פְּרָטֵי הַפְּסִיעָה וְהַחֲזָרָה לִמְקוֹמוֹ. ערוה״ש ק״ה — דִּין הַחוֹזֵר לִמְקוֹמוֹ אַחַר הַתְּפִלָּה. דרך אדמו״ר הזקן — אֲפִלּוּ מִתְפַּלֵּל בַּמָּקוֹם שֶׁפָּסַע — לְהַמְתִּין ; וְלַחֲזֹר לִמְקוֹמוֹ אֲפִלּוּ בְּלֹא תְּפִלָּה שֵׁנִית — צָרִיךְ לְהַמְתִּין, כְּסִימָן קכ״ג (סעיף א).

הטבלה נערכה מטקסט המחבר (אורח חיים ק״ה, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, מגן אברהם, ט״ז, משנה ברורה, ערוך השולחן, כף החיים ; תלמוד ברכות ל׳). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (סְעִיף אֶחָד) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — חוֹבַת הַהַמְתָּנָה וּטְעָמָהּ

קו א — חוֹבַת הַהַמְתָּנָה וּטְעָמָהּ

יסוד הסימן : הַמִּתְפַּלֵּל ב׳ תְּפִלּוֹת זוֹ אַחַר זוֹ צָרִיךְ לְהַמְתִּין בֵּין זוֹ לָזוֹ, כְּדֵי שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת (סעיף א).

קריאת הרב : הָרַב נוֹתֵן אֶת הַטַּעַם — יִשּׁוּב הַדַּעַת כְּדֵי לַעֲרֹךְ דְּבָרָיו בִּלְשׁוֹן תְּחִנָּה. עיין רמה 1.

מסקנה למעשה : לַעֲשׂוֹת הַפְסָקָה בֵּין שְׁתֵּי עֲמִידוֹת רְצוּפוֹת — שַׁחֲרִית וּמוּסָף אוֹ תַּשְׁלוּמִין.

קו ב — שִׁעוּר ד׳ אַמּוֹת

קו ב — שִׁעוּר ד׳ אַמּוֹת

יסוד הסימן : דְּהַיְנוּ כְּדֵי הִלּוּךְ ד׳ אַמּוֹת (סעיף א).

קריאת הרב : הָרַב קוֹבֵעַ אֶת הַשִּׁעוּר כִּזְמַן הִלּוּךְ אַרְבַּע אַמּוֹת ; דַּי בּוֹ אֲפִלּוּ אֵינוֹ חוֹזֵר לִמְקוֹמוֹ. עיין רמה 2.

מסקנה למעשה : לְהַמְתִּין לְכָל הַפָּחוֹת שִׁעוּר זֶה, אַף בְּמִתְפַּלֵּל בַּמָּקוֹם שֶׁעוֹמֵד שָׁם אַחַר הַפְּסִיעָה.

קו ג — הַחֲזָרָה לִמְקוֹמוֹ

קו ג — הַחֲזָרָה לִמְקוֹמוֹ

יסוד הסימן : אֲבָל לַחֲזֹר לִמְקוֹמוֹ אֲפִלּוּ אֵינוֹ רוֹצֶה לְהִתְפַּלֵּל שֵׁנִית — צָרִיךְ לְהַמְתִּין, כְּמוֹ שֶׁיִּתְבָּאֵר בְּסִימָן קכ״ג (סעיף א).

קריאת הרב : הָרַב מַבְחִין בֵּין מִתְפַּלֵּל בַּמָּקוֹם (אַחַר הַפְּסִיעָה) לְבֵין הַחוֹזֵר לִמְקוֹמוֹ — שֶׁמְּחַיֵּב הַמְתָּנָה אֲפִלּוּ בְּלֹא תְּפִלָּה שֵׁנִית. עיין רמה 3.

מסקנה למעשה : הַפְנָיָה לְסִימָן קכ״ג לִפְרָטֵי הַחֲזָרָה לִמְקוֹמוֹ.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן ק״ה ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן ק״ה (הַמִּתְפַּלֵּל שְׁתֵּי תְּפִלּוֹת)

  • ① הַהַמְתָּנָה. בֵּין שְׁתֵּי עֲמִידוֹת רְצוּפוֹת — לְהַמְתִּין בֵּין זוֹ לָזוֹ קֹדֶם שֶׁמַּתְחִיל אֶת הַשְּׁנִיָּה. יסוד : שוע״ר ק״ה סעיף א.
  • ② הַמִּקְרִים. שְׁנֵי מִקְרִים : שַׁחֲרִית וּמוּסָף, אוֹ תְּפִלָּה שֶׁשָּׁכַח וּמַשְׁלִימָהּ (שֶׁשָּׁכַח וְלֹא הִתְפַּלֵּל, תַּשְׁלוּמִין). יסוד : שוע״ר ק״ה סעיף א.
  • ③ הַטַּעַם. כְּדֵי שֶׁתְּהֵא דַּעְתּוֹ מְיֻשֶּׁבֶת לַעֲרֹךְ דְּבָרָיו בִּלְשׁוֹן תְּחִנָּה. יסוד : שוע״ר ק״ה סעיף א.
  • ④ הַשִּׁעוּר. כְּדֵי הִלּוּךְ ד׳ אַמּוֹת — אֲפִלּוּ מִתְפַּלֵּל בַּמָּקוֹם שֶׁפָּסַע. יסוד : שוע״ר ק״ה סעיף א.
  • ⑤ הַחֲזָרָה לִמְקוֹמוֹ. לַחֲזֹר לִמְקוֹמוֹ — צָרִיךְ לְהַמְתִּין אֲפִלּוּ בְּלֹא תְּפִלָּה שֵׁנִית, כְּסִימָן קכ״ג. יסוד : שוע״ר ק״ה סעיף א.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן ק״ה (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — הַהַמְתָּנָה, שִׁעוּר ד׳ אַמּוֹת, הַחֲזָרָה לִמְקוֹמוֹ — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
