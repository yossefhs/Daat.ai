דעת הרב

רמה 4 — דעת הרב (אדמו״ר הזקן) — סימן פ״ג · דִּינֵי בֵּית הַכִּסֵּא בִּקְרִיאַת שְׁמַע

דרכו של אדמו״ר הזקן בדִּינֵי בֵּית הַכִּסֵּא : בֵּית הַכִּסֵּא יָשָׁן — אָסוּר אַף שֶׁפִּנּוּ הַצּוֹאָה ; הַהַזְמָנָה אוֹסֶרֶת לִקְרוֹת בְּתוֹכוֹ ; הַחֲפִירָה בְּמִדְרוֹן — כְּסָתוּם דָּמִי ; הַסַּפְסָל נָקוּב — כְּבֵית הַכִּסֵּא — ברגישות חב״ד.

למה רמה 4 המוקדשת לאדמו״ר הזקן ? שולחן ערוך הרב אינו פירוש על המחבר — הוא שולחן ערוך שלם ועצמאי, שכתב אדמו״ר הזקן (רבי שניאור זלמן מליאדי, מייסד חב״ד, תלמיד המגיד ממעזריטש). ייחודו : הוא משלב הלכה + טעמי המצוות + פנימיות בחיבור אחד, ופוסק בחריפות תלמודית ייחודית.

לחב״ד, אדמו״ר הזקן הוא הפוסק האחרון — פסקו הוא ההכרעה. עמוד זה מאסף את דרך הרב בסימן פ״ג — דִּינֵי בֵּית הַכִּסֵּא בִּקְרִיאַת שְׁמַע. אצל הרב הסימן נפרש לה׳ סְעִיפִים : בֵּית הַכִּסֵּא יָשָׁן וּמְחִצּוֹתָיו, הַהַזְמָנָה, שְׁנֵי הַחֲדָרִים, הַחֲפִירָה בְּמִדְרוֹן (כְּסָתוּם), וְהַסַּפְסָל וְכִסְאוֹת נְקוּבִים.

→ לקריאת המבוא הכללי על שיטת אדמו״ר הזקן

הערת מקור. הטקסט העברי המובא בעמוד זה הוא הטקסט המלא והאמיתי של שולחן ערוך הרב, אורח חיים סימן פ״ג — ה׳ סְעִיפִים מידי הרב — לפי מהדורת קה״ת כפי שנדפסה בספריא (Shulchan Arukh HaRav, Orach Chayim פ״ג). אדמו״ר הזקן דן בו בְּדִינֵי בֵּית הַכִּסֵּא בִּקְרִיאַת שְׁמַע. התרגומים הם שלנו. אין מובא כל קטע מחוץ לטקסט מאומת זה.
1 · טקסט אדמו״ר הזקן

שולחן ערוך הרב — סימן פ״ג — דִּינֵי בֵּית הַכִּסֵּא בִּקְרִיאַת שְׁמַע

הטקסט המלא של שולחן ערוך הרב

שולחן ערוך הרב, סימן פ״ג — דִּינֵי בֵּית הַכִּסֵּא בִּקְרִיאַת שְׁמַע — וּבוֹ ה׳ סְעִיפִים
טקסט אדמו״ר הזקן עצמו (מקור ספריא Shulchan_Arukh_HaRav,_Orach_Chayim.83), ואחריו תרגומנו. הסימן כולל ה׳ סעיפים (סעיף k → ‎.83.k).

סעיף א בֵּית הַכִּסֵּא יָשָׁן — אָסוּר אַף שֶׁפִּנּוּ הַצּוֹאָה ; חֶדֶר מְיֻחָד לְעֻמַּת מְחִצּוֹת בַּיִת.

דִּינֵי בֵּית הַכִּסֵּא בִּקְרִיאַת שְׁמַע וּבוֹ ה׳ סְעִיפִים:כְּשֵׁם שֶׁאָסוּר לִקְרוֹת קְרִיאַת שְׁמַע כְּנֶגֶד הַצּוֹאָה מִן הַתּוֹרָה — כָּךְ אָסוּר לִקְרוֹת כְּנֶגֶד בֵּית הַכִּסֵּא יָשָׁן, דְּהַיְנוּ שֶׁנִּשְׁתַּמֵּשׁ בּוֹ כְּבָר פַּעַם אֶחָד בַּעֲשִׂיַּת צְרָכָיו, וַאֲפִלּוּ אִם פִּנָּה מִמֶּנּוּ הַצּוֹאָה אַחַר כָּךְ, שֶׁכֵּיוָן שֶׁחָל עָלָיו שֵׁם בֵּית הַכִּסֵּא פַּעַם אֶחָד — מָאוּס הוּא וְאֵין זֶה מַחֲנֶה קָדוֹשׁ, וְאָסוּר מִן הַתּוֹרָה. וַאֲפִלּוּ הוּא מִמְּחִצּוֹת גְּבוֹהוֹת י׳ טְפָחִים, שֶׁהֵן חֹלקוֹת רְשׁוּת לְעַצְמָן לְכָל מַה שֶּׁבְּתוֹכָן, וְנִמְצָא שֶׁהַצּוֹאָה שֶׁבְּתוֹכָן הִיא בִּרְשׁוּת בִּפְנֵי עַצְמוֹ, מִכָּל מָקוֹם הַמְּחִצּוֹת עַצְמָן כֵּיוָן שֶׁהֵם מְיֻחָדִים לְכָךְ הֵם מְאוּסִין כְּמוֹ גְּרָף שֶׁל רְעִי, וְצָרִיךְ לְהַרְחִיק מֵהֶם ד׳ אַמּוֹת וּלְפָנָיו כִּמְלֹא עֵינָיו אֲפִלּוּ פִּנּוּ מִמֶּנּוּ הַצּוֹאָה. וְלָכֵן צָרִיךְ לִזָּהֵר שֶׁלֹּא לִקְרוֹת אוֹ לְהִתְפַּלֵּל אוֹ לְהַרְהֵר בְּדִבְרֵי תּוֹרָה בְּחָצֵר כְּנֶגֶד בֵּית הַכִּסֵּא כִּמְלֹא עֵינָיו אַף עַל פִּי שֶׁהוּא סָגוּר וְאִי אֶפְשָׁר לִרְאוֹת הַצּוֹאָה שֶׁבּוֹ. בַּמֶּה דְּבָרִים אֲמוּרִים? כְּשֶׁעָשָׂה בַּחֲצֵירוֹ חֶדֶר מְיֻחָד לָזֶה אֲבָל אִם עָשָׂה בֵּית הַכִּסֵּא בִּמְחִצּוֹת שֶׁל בַּיִת אוֹ חָצֵר — אֵין עַל הַמְּחִצּוֹת אֵלּוּ דִּין בֵּית הַכִּסֵּא אַף עַל פִּי שֶׁהֵן גַּם כֵּן מְחִצּוֹת שֶׁל בֵּית הַכִּסֵּא, (ו)כֵּיוָן שֶׁאֵינָן מְיֻחָדוֹת לוֹ בִּלְבַד. [וּ]מֻתָּר לִקְרוֹת אֲפִלּוּ בְּסָמוּךְ לָהֶן מַמָּשׁ (רַק שֶׁלֹּא יִהְיֶה סָמוּךְ לְהֶחָלָל שֶׁבֵּינֵיהֶם שֶׁנִּפְלָה שָׁם וְלֹא כְּנֶגְדּוֹ מְלֹא עֵינָיו):

בֵּית הַכִּסֵּא יָשָׁן. כְּשֵׁם שֶׁאָסוּר מִן הַתּוֹרָה לִקְרוֹת כְּנֶגֶד הַצּוֹאָה — כָּךְ אָסוּר כְּנֶגֶד בֵּית הַכִּסֵּא יָשָׁן שֶׁנִּשְׁתַּמֵּשׁ בּוֹ, וַאֲפִלּוּ פִּנּוּ מִמֶּנּוּ הַצּוֹאָה, שֶׁחָל עָלָיו שֵׁם בֵּית הַכִּסֵּא וּמָאוּס הוּא וְאֵין זֶה מַחֲנֶה קָדוֹשׁ. מַרְחִיק ד׳ אַמּוֹת וְכִמְלֹא עֵינָיו. בַּמֶּה דְּבָרִים אֲמוּרִים? בְּחֶדֶר מְיֻחָד ; אֲבָל מְחִצּוֹת שֶׁל בַּיִת אוֹ חָצֵר — אֵין עֲלֵיהֶן דִּין בֵּית הַכִּסֵּא, וּמֻתָּר לִקְרוֹת אֲפִלּוּ בְּסָמוּךְ.

סעיף ב הִזְמִין וְלֹא נִשְׁתַּמֵּשׁ — אָסוּר בְּתוֹכוֹ ; מֻתָּר לִקְרוֹת כְּנֶגְדּוֹ.

הִזְמִין מָקוֹם אוֹ חֶדֶר לְבֵית הַכִּסֵּא וַעֲדַיִן לֹא נִשְׁתַּמֵּשׁ בּוֹ — אָסוּר לִקְרוֹת בְּתוֹכוֹ מִדִּבְרֵי סוֹפְרִים, הוֹאִיל וְנִקְרָא עָלָיו שֵׁם בֵּית הַכִּסֵּא — יֵשׁ בּוֹ בִּזָּיוֹן לַקְּרִיאָה שֶׁבְּתוֹכוֹ. אֲבָל מֻתָּר לִקְרוֹת כְּנֶגְדּוֹ אֲפִלּוּ בְּסָמוּךְ לוֹ תּוֹךְ ד׳ אַמּוֹת, שֶׁאֵין קְרִיאַת הַשֵּׁם בִּלְבַד כְּדַאי לֶאֱסֹר חוּצָה לוֹ:

הַזְמָנָה. הִזְמִין מָקוֹם לְבֵית הַכִּסֵּא וַעֲדַיִן לֹא נִשְׁתַּמֵּשׁ — אָסוּר לִקְרוֹת בְּתוֹכוֹ מִדִּבְרֵי סוֹפְרִים, הוֹאִיל וְנִקְרָא עָלָיו שֵׁם בֵּית הַכִּסֵּא. אֲבָל כְּנֶגְדּוֹ, אֲפִלּוּ בְּסָמוּךְ תּוֹךְ ד׳ אַמּוֹת — מֻתָּר.

סעיף ג ״וְגַם זֶה״ — שְׁנֵיהֶם אֲסוּרִים ; ״וְחֶדֶר זֶה״ — סָפֵק.

הָיוּ לְפָנָיו ב׳ חֲדָרִים וְאָמַר ״חֶדֶר זֶה יִהְיֶה לְבֵית הַכִּסֵּא וְגַם זֶה״ — הֲרֵי שְׁנֵיהֶם אֲסוּרִים לִקְרוֹת בְּתוֹכָם. אֲבָל אִם אָמַר עַל הַשֵּׁנִי ״וְחֶדֶר זֶה״ וְלֹא אָמַר ״וְגַם זֶה״ — הֲרֵי הַשֵּׁנִי סָפֵק אִם הִזְמִינוֹ לְכָךְ אִם לֹא. לְפִיכָךְ אֵין קוֹרִין לְכַתְּחִלָּה, וְאִם קָרָא — יָצָא. אֲבָל אִם קָרָא בְּמָקוֹם שֶׁאָסוּר לִקְרוֹת בְּוַדַּאי, אֲפִלּוּ אֵין אִסּוּרוֹ אֶלָּא מִדִּבְרֵי סוֹפְרִים — צָרִיךְ לַחֲזֹר וְלִקְרוֹת, כְּמוֹ שֶׁנִּתְבָּאֵר בְּסִימָן ע״א:

שְׁנֵי חֲדָרִים. אָמַר ״חֶדֶר זֶה יִהְיֶה לְבֵית הַכִּסֵּא וְגַם זֶה״ — שְׁנֵיהֶם אֲסוּרִים. אָמַר ״וְחֶדֶר זֶה״ וְלֹא ״וְגַם זֶה״ — הַשֵּׁנִי סָפֵק : אֵין קוֹרִין לְכַתְּחִלָּה, וְאִם קָרָא יָצָא. וְהַקּוֹרֵא בְּמָקוֹם הָאָסוּר בְּוַדַּאי, אֲפִלּוּ מִדִּבְרֵי סוֹפְרִים — צָרִיךְ לַחֲזֹר (סִימָן ע״א).

סעיף ד חֲפִירָה בְּמִדְרוֹן — כְּסָתוּם ; מַשְׁתִּינִים חוּצָה — אָסוּר.

בֵּית הַכִּסֵּא שֶׁהוּא בַּחֲפִירָה וּפִיו בְּרִחוּק ד׳ אַמּוֹת מִן הַגֻּמָּא וְהוּא עָשׂוּי בְּמִדְרוֹן, בְּעִנְיָן שֶׁהַרְעִי מִתְגַּלְגֵּל וְנוֹפֵל מִיָּד לְמֵרָחוֹק, וְכֵן הַמֵּי רַגְלַיִם יוֹרְדִין מִיָּד לַגֻּמָּא — הֲרֵי הוּא כְּסָתוּם וּכְאִלּוּ אֵין שָׁם בֵּית הַכִּסֵּא כְּלָל חוּץ לַגֻּמָּא (בֵּין שֶׁיֵּשׁ לוֹ מְחִצּוֹת בֵּין שֶׁאֵין לוֹ), וּמֻתָּר לִקְרוֹת שָׁם אִם אֵין שָׂם רֵיחַ רַע וְגַם אֵין מַשְׁתִּינִים חוּץ לַגֻּמָּא. אֲבָל אִם מַשְׁתִּינִים שָׁם לִפְעָמִים (וּרְגִילִים בְּכָךְ) — אָסוּר לְהַרְהֵר שָׁם בְּדִבְרֵי תּוֹרָה וְאֵין צָרִיךְ לוֹמַר לִקְרוֹת קְרִיאַת שְׁמַע (וַאֲפִלּוּ בְּשָׁעָה שֶׁאֵין שָׁם מֵי רַגְלַיִם), לְפִי שֶׁיֵּשׁ שֵׁם בֵּית הַכִּסֵּא עַל מָקוֹם זֶה שֶׁחוּץ לַגֻּמָּא שֶׁמַּשְׁתִּינִים שָׁם, וְצָרִיךְ לְהַרְחִיק מִמָּקוֹם זֶה ד׳ אַמּוֹת וּלְפָנָיו כִּמְלֹא עֵינָיו (וּמִכָּל מָקוֹם אִם יֵשׁ לוֹ מְחִצּוֹת — אֵין לָהֶם דִּין מְחִצּוֹת בֵּית הַכִּסֵּא לְהַרְחִיק מֵהֶן, כֵּיוָן שֶׁאֵין הַמְּחִצּוֹת אֶלָּא לְבֵית הַכִּסֵּא שֶׁל הֲטָלַת מֵי רַגְלַיִם, שֶׁאֵין אֲסוּרִין אֶלָּא מִדִּבְרֵי סוֹפְרִים אַחַר שֶׁנָּפְלוּ לָאָרֶץ, שֶׁהַצּוֹאָה שֶׁבַּגֻּמָּה כְּסָתוּם דָּמִי וְאֵין מְחִצּוֹת אֵלּוּ נִקְרָאִים בִּשְׁבִילָהּ מְחִצּוֹת בֵּית הַכִּסֵּא):

חֲפִירָה בְּמִדְרוֹן. בֵּית הַכִּסֵּא שֶׁבַּחֲפִירָה, פִּיו רָחוֹק ד׳ אַמּוֹת מִן הַגֻּמָּא וְעָשׂוּי בְּמִדְרוֹן — הָרְעִי וְהַמֵּי רַגְלַיִם נוֹפְלִים מִיָּד לַגֻּמָּא — כְּסָתוּם דָּמִי, וּמֻתָּר לִקְרוֹת אִם אֵין רֵיחַ רַע וְאֵין מַשְׁתִּינִים חוּץ לַגֻּמָּא. אֲבָל מַשְׁתִּינִים שָׁם לִפְעָמִים — אָסוּר לְהַרְהֵר, וְכָל שֶׁכֵּן ק״ש, וּמַרְחִיק ד׳ אַמּוֹת.

סעיף ה סַפְסָל וְכִסְאוֹת נְקוּבִים — מְכֻסִּים אוֹ מְסֻלָּקִים בִּשְׁעַת ק״ש.

בְּנֵי אָדָם שֶׁיֵּשׁ לָהֶם סַפְסָל נָקוּב שֶׁנִּפְנִים עָלָיו, וְכֵן בְּאוֹתָם בָּתִּים שֶׁיֵּשׁ בְּתוֹכָם כִּסְאוֹת קְטַנִּים נְקוּבִים לְצֹרֶךְ הַיְלָדִים קְטַנִּים שֶׁבַּבַּיִת — צָרִיךְ לִזָּהֵר שֶׁיִּהְיוּ מְכֻסִּים בִּשְׁעַת קְרִיאַת שְׁמַע וּתְפִלָּה וְדִבְרֵי תּוֹרָה, אוֹ יוֹצִיאֵם מֵהַחֶדֶר שֶׁמִּתְפַּלֵל אוֹ לוֹמֵד בּוֹ, אֲפִלּוּ הֵם נְקִיִּים בְּתוֹכָם וְעַל גַּבָּם, שֶׁכֵּיוָן שֶׁהֵם מְיֻחָדִים לִפָּנוֹת עֲלֵיהֶם — הֲרֵי הֵם כְּבֵית הַכִּסֵּא, שֶׁאֲפִלּוּ אֵין בּוֹ צוֹאָה צָרִיךְ לְהַרְחִיק מִמֶּנּוּ ד׳ אַמּוֹת וּלְפָנָיו כִּמְלֹא עֵינָיו מִן הַתּוֹרָה:

סַפְסָל וְכִסְאוֹת נְקוּבִים. סַפְסָל נָקוּב שֶׁנִּפְנִים עָלָיו, וְכֵן כִּסְאוֹת קְטַנִּים נְקוּבִים שֶׁל יְלָדִים — צָרִיךְ שֶׁיִּהְיוּ מְכֻסִּים בִּשְׁעַת ק״ש וּתְפִלָּה וְדִבְרֵי תּוֹרָה אוֹ יוֹצִיאֵם, אֲפִלּוּ נְקִיִּים, שֶׁמִּתּוֹךְ שֶׁהֵם מְיֻחָדִים לִפָּנוֹת — הֲרֵי הֵם כְּבֵית הַכִּסֵּא, וּמַרְחִיק ד׳ אַמּוֹת וּכִמְלֹא עֵינָיו מִן הַתּוֹרָה.

מקור : ספריא — Shulchan Arukh HaRav, Orach Chayim פ״ג (מהדורת קה״ת, ה׳ סְעִיפִים). תרגום : DAAT. אין מובא כל קטע מחוץ לטקסט מאומת זה.

2 · היסוד

היסוד — דִּינֵי בֵּית הַכִּסֵּא בִּקְרִיאַת שְׁמַע

היסוד : דִּינֵי בֵּית הַכִּסֵּא

קריאת אדמו״ר הזקן בסימן זה נפרשת סביב ארבעה צירים. תחילה בֵּית הַכִּסֵּא יָשָׁן וְשֵׁמוֹ. אחר כך הַהַזְמָנָה וּשְׁנֵי הַחֲדָרִים. אחר כך הַחֲפִירָה בְּמִדְרוֹן (כְּסָתוּם). ולבסוף הַסַּפְסָל וְכִסְאוֹת נְקוּבִים.

בֵּית הַכִּסֵּא יָשָׁן — שֵׁם וּמְאִיסוּת

בֵּית הַכִּסֵּא יָשָׁן — שֵׁם וּמְאִיסוּת

הַיְסוֹד. בֵּית הַכִּסֵּא יָשָׁן אָסוּר אַף שֶׁפִּנּוּ מִמֶּנּוּ הַצּוֹאָה : שֵׁם בֵּית הַכִּסֵּא עֲשָׂאוֹ מָאוּס, וְאֵין זֶה מַחֲנֶה קָדוֹשׁ (סעיף א).

הַמַּסְקָנָה. לֹא נוֹכְחוּת הַלִּכְלוּךְ אוֹסֶרֶת אֶלָּא הַשֵּׁם שֶׁחָל. עיין סימן 79 (צוֹאָה כְּנֶגֶד ק״ש).

הַזְמָנָה — קְרִיאָה בְּתוֹכוֹ וּכְנֶגְדּוֹ

הַזְמָנָה — קְרִיאָה בְּתוֹכוֹ וּכְנֶגְדּוֹ

הַיְסוֹד. חֶדֶר שֶׁהִזְמִין לְבֵית הַכִּסֵּא וְלֹא נִשְׁתַּמֵּשׁ — אָסוּר לִקְרוֹת בְּתוֹכוֹ מִדִּבְרֵי סוֹפְרִים ; אֲבָל כְּנֶגְדּוֹ אֲפִלּוּ בְּסָמוּךְ — מֻתָּר (סעיף ב).

הַמַּסְקָנָה. ״וְגַם זֶה״ — שְׁנֵיהֶם אֲסוּרִים ; ״וְחֶדֶר זֶה״ לְבַד — סָפֵק, בְּדִיעֲבַד יָצָא ; וּבִמְקוֹם הָאִסּוּר הַוַּדַּאי — חוֹזֵר. עיין סימן 82.

חֲפִירָה בְּמִדְרוֹן — כְּסָתוּם דָּמִי

חֲפִירָה בְּמִדְרוֹן — כְּסָתוּם דָּמִי

הַיְסוֹד. חֲפִירָה בְּמִדְרוֹן הַמְגַלְגֶּלֶת מִיָּד אֶת הָרְעִי וְהַמֵּי רַגְלַיִם — כְּסָתוּם : מֻתֶּרֶת בְּלֹא רֵיחַ רַע וּבְלֹא הֲטָלַת מֵי רַגְלַיִם חוּצָה (סעיף ד).

הַמַּסְקָנָה. אֲבָל אִם מַשְׁתִּינִים חוּצָה — אָסוּר אֲפִלּוּ לְהַרְהֵר. עיין סימן 81.

סַפְסָל וְכִסְאוֹת נְקוּבִים — כְּבֵית הַכִּסֵּא

סַפְסָל וְכִסְאוֹת נְקוּבִים — כְּבֵית הַכִּסֵּא

הַיְסוֹד. סַפְסָל נָקוּב וְכִסְאוֹת נְקוּבִים שֶׁל יְלָדִים, מְיֻחָדִים לִפָּנוֹת — כְּבֵית הַכִּסֵּא אַף כְּשֶׁנְּקִיִּים (סעיף ה).

הַמַּסְקָנָה. לְכַסּוֹתָם אוֹ לְהוֹצִיאָם, וּלְהַרְחִיק ד׳ אַמּוֹת מִן הַתּוֹרָה. עיין סימן 76.

הפניה ללימוד. היקף הסימן — שֵׁם בֵּית הַכִּסֵּא, הַמְּאִיסוּת וְהַכְּסָתוּם — נוגע לטהרת מקום הַקְּרִיאַת שְׁמַע וְהַתְּפִלָּה. עמוד זה מציג אותו ברמת היסוד המאומת בטקסט הרב, בלי להוסיף דבר. ללימוד מעמיק ולהנהגה אישית, ללמוד עם רב חב״ד ובתוך המקורות עצמם.

דִּינֵי בֵּית הַכִּסֵּא בִּקְרִיאַת שְׁמַע — ביסוד · ללימוד, לפנות לרב חב״ד
3 · כח הפסק

כחו של אדמו״ר הזקן בפסק

כוחו של אדמו״ר הזקן בפסק

השוואת דרך אדמו״ר הזקן עם המחבר, המשנה ברורה (משנה ברורה) וערוך השולחן (ערוך השולחן) בנקודות המרכזיות של סימן פ״ג. שורת שולחן ערוך הרב מודגשת — היא חושפת את רגישות הרב כפי שהיא עולה מה׳ סעיפיו.

נושא מחבר משנה ברורה ערוך השולחן דרך אדמו״ר הזקן (שולחן ערוך הרב)
בֵּית הַכִּסֵּא יָשָׁן וּמְחִצּוֹתָיו או״ח פ״ג — בֵּית הַכִּסֵּא יָשָׁן אָסוּר אַף שֶׁפִּנּוּ הַצּוֹאָה ; יֵשׁ מְחִצּוֹת — קוֹרֵא בְּסָמוּךְ. מ״ב פ״ג — גֶּדֶר ״יָשָׁן״ וְשִׁעוּר הַהַרְחָקָה כִּמְלֹא עֵינָיו. ערוה״ש פ״ג — דִּין מְחִצּוֹת בֵּית הַכִּסֵּא בַּזְּמַן הַזֶּה. דרך אדמו״ר הזקן — יָשָׁן אָסוּר מִן הַתּוֹרָה אַף שֶׁפִּנּוּ ; חֶדֶר מְיֻחָד — מַרְחִיק ד׳ אַמּוֹת וְכִמְלֹא עֵינָיו ; מְחִצּוֹת בַּיִת אוֹ חָצֵר — אֵין עֲלֵיהֶן דִּין בֵּית הַכִּסֵּא (סעיף א).
הַזְמָנָה וּשְׁנֵי חֲדָרִים או״ח פ״ג — הִזְמִינוֹ וְלֹא נִשְׁתַּמֵּשׁ מֻתָּר כְּנֶגְדּוֹ וְלֹא בְּתוֹכוֹ. מ״ב פ״ג — הַזְמָנָה בְּפֶה מְהַנְיָא לֶאֱסֹר הַקְּרִיאָה בְּתוֹכוֹ. ערוה״ש פ״ג — כֹּחַ הַהַזְמָנָה וְגִדְרֵי הַסָּפֵק בָּהּ. דרך אדמו״ר הזקן — אָסוּר בְּתוֹכוֹ מִדִּבְרֵי סוֹפְרִים (בִּזָּיוֹן) ; כְּנֶגְדּוֹ אֲפִלּוּ בְּסָמוּךְ — מֻתָּר (סעיף ב). ״וְגַם זֶה״ — שְׁנֵיהֶם אֲסוּרִים ; ״וְחֶדֶר זֶה״ — סָפֵק, בְּדִיעֲבַד יָצָא (סעיף ג).
חֲפִירָה בְּמִדְרוֹן או״ח פ״ג — חֲפִירָה בְּמִדְרוֹן כְּסָתוּם ; רמ״א — מַשְׁתִּינִים אָסוּר לְהַרְהֵר. מ״ב פ״ג — גֶּדֶר ״כְּסָתוּם״ וְרֵיחַ רַע. ערוה״ש פ״ג — בָּתֵּי כִּסֵּא שֶׁלָּנוּ וְדִין הַמֵּי רַגְלַיִם. דרך אדמו״ר הזקן — כְּסָתוּם דָּמִי, מֻתָּר בְּלֹא רֵיחַ רַע וּבְלֹא הֲטָלַת מֵי רַגְלַיִם חוּצָה ; מַשְׁתִּינִים שָׁם — אָסוּר לְהַרְהֵר וְכָל שֶׁכֵּן ק״ש, מַרְחִיק ד׳ אַמּוֹת (סעיף ד).
סַפְסָל וְכִסְאוֹת נְקוּבִים או״ח פ״ג — סַפְסָל נָקוּב מֻתָּר לִקְרוֹת כְּשֶׁאֵין צוֹאָה וּמְכֻסֶּה. מ״ב פ״ג — פְּרָטֵי הַכִּסּוּי וְהַגְּרָף שֶׁתַּחַת הַנֶּקֶב. ערוה״ש פ״ג — כִּסְאוֹת הַיְלָדִים בַּזְּמַן הַזֶּה. דרך אדמו״ר הזקן — מְיֻחָדִים לִפָּנוֹת — כְּבֵית הַכִּסֵּא אַף כְּשֶׁנְּקִיִּים ; מְכֻסִּים אוֹ מְסֻלָּקִים, מַרְחִיק ד׳ אַמּוֹת מִן הַתּוֹרָה (סעיף ה).

הטבלה נערכה מטקסט המחבר (אורח חיים פ״ג, מקור ספריא) ומנושאי הכלים המאומתים (בית יוסף, טור, רמב״ם הלכות קריאת שמע ג, רא״ש, תלמידי רבנו יונה, מגן אברהם, ט״ז, משנה ברורה, ביאור הלכה, פרי מגדים, ערוך השולחן, כף החיים ; תלמוד ברכות כ״ה-כ״ו). שורת אדמו״ר הזקן נשענת על הטקסט האמיתי של שולחן ערוך הרב (ה׳ סְעִיפִים) המובא ב§1.

4 · קווי היסוד של הרב

קווי היסוד של שיטת הרב

קווי היסוד של שיטת הרב בסימן זה

שלוש מגמות המאפיינות את רגישות אדמו״ר הזקן בסימן זה. כל אחת לפי המבנה : יסוד הסימן → קריאת הרב → מסקנה למעשה. כל קו נשען על הטקסט האמיתי של הרב (§1).

קו א — שֵׁם בֵּית הַכִּסֵּא

קו א — שֵׁם בֵּית הַכִּסֵּא

יסוד הסימן : שֵׁם בֵּית הַכִּסֵּא אוֹסֵר אַף בְּלֹא צוֹאָה — יָשָׁן, חֶדֶר מְיֻחָד, סַפְסָל נָקוּב (סעיפים א, ב, ה).

קריאת הרב : הַמְּאִיסוּת שֶׁבַּשֵּׁם, לֹא הַלִּכְלוּךְ שֶׁכָּעֵת, מְסַלֶּקֶת אֶת מַחֲנֶה קָדוֹשׁ. עיין סימן 79.

מסקנה למעשה : לְהִתְיַחֵס לְמָקוֹם הַמְיֻחָד לְבֵית הַכִּסֵּא כְּאָסוּר אַף שֶׁרֵיק וְסָגוּר.

קו ב — הֶפְסֵק וּמְחִצּוֹת

קו ב — הֶפְסֵק וּמְחִצּוֹת

יסוד הסימן : מְחִצּוֹת מְיֻחָדוֹת נוֹשְׂאוֹת דִּין בֵּית הַכִּסֵּא ; מְחִצּוֹת בַּיִת אוֹ חָצֵר — לֹא (סעיף א).

קריאת הרב : לְהַבְחִין בֵּין מְחִצָּה מְיֻחֶדֶת לְכֹתֶל סְתָם ; מַרְחִיק ד׳ אַמּוֹת וְכִמְלֹא עֵינָיו. עיין סימן 82.

מסקנה למעשה : אֵצֶל כֹּתֶל שֶׁאֵינוֹ מְיֻחָד — מֻתָּר לִקְרוֹת אֲפִלּוּ בְּסָמוּךְ מַמָּשׁ.

קו ג — כְּסָתוּם וְהַהֶרְחֵק לְמַעֲשֶׂה

קו ג — כְּסָתוּם וְהַהֶרְחֵק לְמַעֲשֶׂה

יסוד הסימן : הַחֲפִירָה בְּמִדְרוֹן כְּסָתוּם ; אֲבָל הֲטָלַת מֵי רַגְלַיִם חוּצָה מַחֲזִירָה שֵׁם בֵּית הַכִּסֵּא (סעיף ד).

קריאת הרב : הַהֶתֵּר תָּלוּי בְּהֶעְדֵּר רֵיחַ רַע וּבַשִּׁמּוּשׁ הַמַּמָּשִׁי שֶׁל הַמָּקוֹם. עיין סימן 81.

מסקנה למעשה : לֹא לִקְרוֹת וְלֹא לְהַרְהֵר בְּמָקוֹם שֶׁמַּשְׁתִּינִים בּוֹ, אַף שֶׁנָּקִי כָּעֵת.

5 · למעשה

הלכה למעשה — מנהג חב״ד

הלכה למעשה — מנהג חב״ד

נקודות הנהגה הנובעות ישירות מסימן פ״ג ברגישות אדמו״ר הזקן וחב״ד — מוצגות ברמת היסוד, ומפנות לרב לפרטי המקרים.

לחסיד חב״ד — סימן פ״ג (דִּינֵי בֵּית הַכִּסֵּא בִּקְרִיאַת שְׁמַע)

  • ① בֵּית הַכִּסֵּא יָשָׁן. מָקוֹם שֶׁכְּבָר שִׁמֵּשׁ כְּבֵית הַכִּסֵּא — אָסוּר אַף שֶׁפִּנּוּ הַצּוֹאָה ; מַרְחִיק ד׳ אַמּוֹת וְכִמְלֹא עֵינָיו. יסוד : שוע״ר פ״ג סעיף א.
  • ② הַזְמָנָה. חֶדֶר שֶׁהֻזְמַן וְלֹא נִשְׁתַּמֵּשׁ — אֵין קוֹרִין בְּתוֹכוֹ (מִדִּבְרֵי סוֹפְרִים) ; אֲבָל כְּנֶגְדּוֹ אֲפִלּוּ בְּסָמוּךְ — מֻתָּר. יסוד : שוע״ר פ״ג סעיף ב.
  • ③ שְׁנֵי חֲדָרִים. ״וְגַם זֶה״ — שְׁנֵיהֶם אֲסוּרִים ; ״וְחֶדֶר זֶה״ לְבַד — סָפֵק (לְכַתְּחִלָּה לֹא, בְּדִיעֲבַד יָצָא) ; קָרָא בַּוַּדַּאי הָאָסוּר — חוֹזֵר. יסוד : שוע״ר פ״ג סעיף ג.
  • ④ חֲפִירָה בְּמִדְרוֹן. חֲפִירָה בְּמִדְרוֹן = כְּסָתוּם — מֻתֶּרֶת בְּלֹא רֵיחַ רַע ; אֲבָל אִם מַשְׁתִּינִים חוּצָה — אָסוּר אֲפִלּוּ לְהַרְהֵר. יסוד : שוע״ר פ״ג סעיף ד.
  • ⑤ סַפְסָל וְכִסְאוֹת נְקוּבִים. סַפְסָל נָקוּב וְכִסְאוֹת יְלָדִים נְקוּבִים — לְכַסּוֹת אוֹ לְהוֹצִיא, אַף נְקִיִּים, כְּבֵית הַכִּסֵּא. יסוד : שוע״ר פ״ג סעיף ה.

⚠ חלק זה מציג את הנהגת חב״ד המיוסדת על שולחן ערוך הרב, סימן פ״ג (טקסט אמיתי המובא ב§1). אין הוא מחליף הכרעת רב למקרה פרטי. לכל שאלה מעשית — בֵּית הַכִּסֵּא יָשָׁן, הַזְמָנָה, חֲפִירָה בְּמִדְרוֹן, סַפְסָל נָקוּב — לְהִתְיָעֵץ עִם רַבְּךָ, וּלְחַב״ד — עִם רַב חב״ד.
